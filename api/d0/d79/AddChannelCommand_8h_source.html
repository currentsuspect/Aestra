108 void captureRoutesToChannel() {
109 if (m_routesCaptured || m_createdChannelId == 0) {
110 return;
111 }
112 for (const auto& channel : m_manager.getChannelsSnapshot()) {
113 if (!channel || channel->getChannelId() == m_createdChannelId) {
114 continue;
115 }
116 const auto sends = channel->getSends();
117 const bool routesToChannel = channel->getMainOutputId() == m_createdChannelId ||
118 std::any_of(sends.begin(), sends.end(), [this](const AudioRoute& route) {
119 return route.targetChannelId == m_createdChannelId;
120 });
121 if (routesToChannel) {
122 m_routes.push_back({channel->getChannelId(), channel->getMainOutputId(), sends});
123 }
124 }
125 m_routesCaptured = true;
126 }
127
128 void restoreRoutesToChannel() {
129 for (const auto& routeSnapshot : m_routes) {
130 if (auto* channel = m_manager.getChannelById(routeSnapshot.sourceChannelId)) {
131 if (routeSnapshot.mainOutputId == m_createdChannelId) {
132 channel->setMainOutputId(routeSnapshot.mainOutputId);
133 }
134 if (!routeSnapshot.sends.empty()) {
135 channel->replaceSends(routeSnapshot.sends);
136 }
137 }
138 }
139 m_routes.clear();
140 m_routesCaptured = false;
141 }
