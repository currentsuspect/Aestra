126HeadlessMusicGenerator& HeadlessMusicGenerator::addEuclideanRhythm(uint8_t pulses, uint8_t steps,
127 uint8_t pitch, uint8_t velocity) {
128 if (pulses == 0 || steps == 0 || pulses > steps) {
129 Log::warning("[HeadlessGenerator] Invalid Euclidean rhythm parameters");
130 return *this;
131 }
132
133 auto* pattern = getCurrentPattern();
134 if (!pattern) return *this;
135
136 // Generate Euclidean rhythm using Bresenham's algorithm
137 std::vector<bool> rhythm(steps, false);
138 int error = steps / 2;
139 int step = 0;
140
141 for (int i = 0; i < pulses; ++i) {
142 rhythm[step] = true;
143 step += steps / pulses;
144 error += steps % pulses;
145 if (error >= pulses) {
146 step++;
147 error -= pulses;
148 }
149 }
150
151 // Add notes where rhythm is true
152 for (size_t i = 0; i < rhythm.size(); ++i) {
153 if (rhythm[i]) {
154 addNote(static_cast<uint8_t>(i), pitch, velocity, 0.25);
155 }
156 }
157
158 return *this;
159}

261HeadlessMusicGenerator& HeadlessMusicGenerator::generateFromTemplate(const std::string& templateName,
262 uint32_t durationBars) {
263 if (templateName == "Techno") {
264 setTempo(130);
265
266 // Create patterns
267 loadFourOnTheFloor("Kick");
268 loadHiHatPattern("HiHat");
269 createPattern("Clap", 16)
270 .addNote(4, 39, 100, 0.5)
271 .addNote(12, 39, 100, 0.5);
272 loadBassline("Bass");
273
274 // Assemble playlist
275 double barBeats = static_cast<double>(m_timeSigNum);
276 for (uint32_t bar = 0; bar < durationBars; ++bar) {
277 double startBeat = bar * barBeats;
278 addClipToPlaylist("Kick", startBeat, barBeats, 0);
279 addClipToPlaylist("HiHat", startBeat, barBeats, 1);
280 addClipToPlaylist("Clap", startBeat, barBeats, 2);
281 addClipToPlaylist("Bass", startBeat, barBeats, 3);
282 }
283
284 // Setup mixer
285 addChannel("Kick Channel");
286 addChannel("HiHat Channel");
287 addChannel("Clap Channel");
288 addChannel("Bass Channel");
289
290 } else if (templateName == "House") {
291 setTempo(125);
292
293 loadFourOnTheFloor("Kick");
294 createPattern("OffBeatBass", 16)
295 .addNote(2, 33, 100, 0.5)
296 .addNote(6, 33, 100, 0.5)
297 .addNote(10, 33, 100, 0.5)
298 .addNote(14, 33, 100, 0.5);
299
300 double barBeats = static_cast<double>(m_timeSigNum);
301 for (uint32_t bar = 0; bar < durationBars; ++bar) {
302 double startBeat = bar * barBeats;
303 addClipToPlaylist("Kick", startBeat, barBeats, 0);
304 addClipToPlaylist("OffBeatBass", startBeat, barBeats, 1);
305 }
306
307 addChannel("Kick");
308 addChannel("Bass");
309
310 } else if (templateName == "Minimal") {
311 setTempo(128);
312
313 // Minimal kick - sparse
314 createPattern("MinKick", 16)
315 .addNote(0, 36, 110, 0.5)
316 .addNote(10, 36, 90, 0.5);
317
318 // Click hi-hat
319 createPattern("Click", 16)
320 .addEuclideanRhythm(5, 16, 42, 70);
321
322 double barBeats = static_cast<double>(m_timeSigNum);
323 for (uint32_t bar = 0; bar < durationBars; ++bar) {
324 double startBeat = bar * barBeats;
325 addClipToPlaylist("MinKick", startBeat, barBeats, 0);
326 addClipToPlaylist("Click", startBeat, barBeats, 1);
327 }
328
329 addChannel("Kick");
330 addChannel("Perc");
331 } else {
332 Log::warning("[HeadlessGenerator] Unknown template: " + templateName);
333 }
334
335 return *this;
336}

349bool HeadlessMusicGenerator::exportTo(const std::string& outputPath,
350 std::function<void(float)> progressCallback,
351 uint32_t sampleRate,
352 AudioExporter::BitDepth bitDepth) {
353 // Validate
354 std::string error = validate();
355 if (!error.empty()) {
356 Log::error("[HeadlessGenerator] Validation failed: " + error);
357 return false;
358 }
359
360 // Reconcile the render sample rate before committing so the sampler
361 // instruments are initialized at the same rate the engine renders at (the
362 // exportTo sampleRate argument overrides the configured project rate).
363 const uint32_t effectiveRate = sampleRate > 0 ? sampleRate : m_sampleRate;
364 m_sampleRate = effectiveRate;
365
366 // Commit patterns to track manager
367 commitPatternsToManager();
368
369 // Commit playlist
370 commitPlaylistToModel();
371
372 // Remove the temp tone sample on every exit path, including an exception
373 // out of render() (the sampler decoded it into memory at load time).
374 struct ToneSampleCleanup {
375 const std::string& path;
376 ~ToneSampleCleanup() {
377 if (!path.empty()) {
378 std::error_code ec;
379 std::filesystem::remove(path, ec);
380 }
381 }
382 } toneCleanup{m_toneSamplePath};
383
384 // --- Wire the export engine to render the committed timeline ---------------
385 // AudioExporter::render drives AudioEngine::processBlock — the same path the
386 // live engine uses — so the committed MIDI clips synthesize through their
387 // sampler units (live/export parity, AGENTS.md §20).
388 //
389 // setTrackManager stores a weak_ptr, so `borrowedTrackManager` must stay
390 // alive for the whole render. The guard restores the engine to an unwired
391 // state on every exit path (including failures) — the engine keeps a raw
392 // UnitManager*, an owned slot map and a graph copy, none of which may
393 // outlive this call. The render is fully synchronous: no callback or
394 // retained object can touch this wiring once exportTo returns.
395 // Restores everything this render mutates on every exit path, so a
396 // synchronous export leaves the caller's engine and session unchanged:
397 // - engine wiring (weak TrackManager ref, raw UnitManager*, owned slot map,
398 // graph copy, pattern-engine pointer) is cleared;
399 // - the pattern-playback engine is cleared so the timeline instances this
400 // render scheduled do not leak into the caller's TrackManager.
401 // The transport flags/position are never touched (see
402 // scheduleTimelineForOfflineRender below), so there is nothing else to undo.
403 struct RenderStateGuard {
404 AudioEngine& engine;
405 TrackManager& trackManager;
406 ~RenderStateGuard() {
407 engine.setGraph(AudioGraph{});
408 engine.setPatternPlaybackEngine(nullptr);
409 engine.setUnitManager(nullptr);
410 engine.setChannelSlotMap(nullptr);
411 engine.setTrackManager(nullptr);
412 // Remove, don't rewind: this render's instances must not survive into the
413 // caller's session (rewindScheduledInstances() would leave them scheduled and
414 // merely restarted).
415 trackManager.getPatternPlaybackEngine().clearScheduledInstances();
416 }
417 };
418
419 // Non-owning shared_ptr: setTrackManager currently requires shared ownership,
420 // but the caller owns m_trackManager and guarantees it outlives this render.
421 // The no-op deleter means the borrowed TrackManager is never freed here.
422 std::shared_ptr<TrackManager> borrowedTrackManager(&m_trackManager, [](TrackManager*) {});
423 RenderStateGuard renderGuard{m_engine, m_trackManager}; // destroyed before borrowedTrackManager
424
425 m_engine.setSampleRate(effectiveRate);
426 m_engine.setBufferConfig(512, 2);
427 m_engine.setBPM(static_cast<float>(m_tempo));
428 m_engine.setTrackManager(borrowedTrackManager);
429 m_engine.setUnitManager(&m_trackManager.getUnitManager());
430 m_engine.setPatternPlaybackEngine(&m_trackManager.getPatternPlaybackEngine());
431 m_trackManager.buildAndShareSlotMap();
432 if (auto slotMap = m_trackManager.getChannelSlotMapShared()) {
433 m_engine.setChannelSlotMap(slotMap);
434 }
435 m_engine.setGraph(AudioGraphBuilder::buildFromTrackManager(m_trackManager));
436 m_engine.initialize();
437
438 // MIDI clips reach their units through the pattern-playback engine
439 // (AudioEngine::processBlock pops scheduled notes into unit MIDI routes).
440 // Schedule the committed timeline into it WITHOUT starting live transport —
441 // the exporter drives the engine's own transport, so we must not mutate the
442 // caller's playing flag / position. clearScheduledInstances() first removes any prior
443 // contents; the render guard clears again on exit so these instances don't leak.
444 // rewindScheduledInstances() cannot serve here — it only REWINDS active instances
445 // (re-emit from the top, which a loop restart wants), so anything already scheduled
446 // would have been rendered into the export alongside the timeline we just asked for.
447 m_trackManager.getPatternPlaybackEngine().clearScheduledInstances();
448 m_trackManager.scheduleTimelineForOfflineRender(0.0);
449
450 // Setup exporter
451 AudioExporter exporter(m_engine, m_trackManager);
452
453 if (progressCallback) {
454 exporter.setProgressCallback(progressCallback);
455 }
456
457 AudioExporter::Config config;
458 config.outputPath = outputPath;
459 config.sampleRate = sampleRate > 0 ? sampleRate : m_sampleRate;
460 config.bitDepth = bitDepth;
461 config.scope = AudioExporter::RenderScope::FullSong;
462
463 // Render
464 Log::info("[HeadlessGenerator] Exporting to: " + outputPath);
465 auto result = exporter.render(config); // toneCleanup removes the temp sample on scope exit
466
467 if (result.success) {
468 Log::info("[HeadlessGenerator] Export complete: " +
469 std::to_string(result.durationSeconds) + "s, peak: " +
470 std::to_string(result.peakDb) + " dB");
471 return true;
472 } else {
473 Log::error("[HeadlessGenerator] Export failed: " + result.errorMessage);
474 return false;
475 }
476}

548bool HeadlessMusicGenerator::prepareToneSample() {
549 // A short synthetic sine the built-in sampler loads as its source and
550 // pitch-shifts per MIDI note. Written to a unique temp file because the
551 // sampler loads by path (UnitManager::setUnitAudioClip -> loadSample).
552 namespace fs = std::filesystem;
553 std::error_code ec;
554 const fs::path dir = fs::temp_directory_path(ec);
555 if (ec) {
556 Log::error("[HeadlessGenerator] No temp directory for tone sample: " + ec.message());
557 return false;
558 }
559 const fs::path path =
560 dir / ("aestra_headless_tone_" + std::to_string(reinterpret_cast<uintptr_t>(this)) + ".wav");
561
562 const uint32_t rate = m_sampleRate > 0 ? m_sampleRate : 48000u;
563 const uint32_t frames = rate / 2; // 0.5 s
564 const double freq = 440.0;
565 const double twoPi = 6.283185307179586;
566
567 std::ofstream out(path, std::ios::binary | std::ios::trunc);
568 if (!out) {
569 Log::error("[HeadlessGenerator] Cannot write tone sample: " + path.string());
570 return false;
571 }
572 auto w16 = [&](uint16_t v) { out.put(static_cast<char>(v & 0xff)); out.put(static_cast<char>((v >> 8) & 0xff)); };
573 auto w32 = [&](uint32_t v) {
574 out.put(static_cast<char>(v & 0xff));
575 out.put(static_cast<char>((v >> 8) & 0xff));
576 out.put(static_cast<char>((v >> 16) & 0xff));
577 out.put(static_cast<char>((v >> 24) & 0xff));
578 };
579 const uint32_t dataBytes = frames * 2; // mono, 16-bit
580 out.write("RIFF", 4); w32(36 + dataBytes); out.write("WAVE", 4);
581 out.write("fmt ", 4); w32(16); w16(1); w16(1); w32(rate); w32(rate * 2); w16(2); w16(16);
582 out.write("data", 4); w32(dataBytes);
583 for (uint32_t i = 0; i < frames; ++i) {
584 // Short fades at both ends so the looped/one-shot sample has no click.
585 double env = 1.0;
586 const uint32_t fade = rate / 100; // 10 ms
587 if (i < fade) env = static_cast<double>(i) / fade;
588 else if (i >= frames - fade) env = static_cast<double>(frames - i) / fade;
589 const double s = std::sin(twoPi * freq * static_cast<double>(i) / rate) * 0.8 * env;
590 w16(static_cast<uint16_t>(static_cast<int16_t>(std::clamp(s, -1.0, 1.0) * 32767.0)));
591 }
592 if (!out.good()) {
593 Log::error("[HeadlessGenerator] Failed writing tone sample: " + path.string());
594 return false;
595 }
596 m_toneSamplePath = path.string();
597 return true;
598}

600uint64_t HeadlessMusicGenerator::ensureLaneInstrument(uint32_t laneIndex) {
601 if (auto it = m_laneUnits.find(laneIndex); it != m_laneUnits.end()) {
602 return it->second;
603 }
604
605 auto& um = m_trackManager.getUnitManager();
606 const std::string label = "Lane " + std::to_string(laneIndex + 1);
607
608 // Mixer channel that carries this lane's instrument to the master bus.
609 auto* mixerChannel = m_trackManager.addChannel(label);
610
611 // Playlist lane the clips are placed on.
612 PlaylistLaneID laneId = m_trackManager.getPlaylistModel().createLane(label);
613 m_laneIds[laneIndex] = laneId;
614
615 // Built-in sampler — the only always-available (license-free) instrument.
616 auto& pluginManager = PluginManager::getInstance();
617 pluginManager.initialize();
618 auto sampler = pluginManager.createInstanceById("com.Aestrastudios.sampler");
619 if (!sampler) {
620 Log::error("[HeadlessGenerator] Failed to create built-in sampler instrument");
621 return 0;
622 }
623 sampler->initialize(m_sampleRate > 0 ? m_sampleRate : 48000u, 512);
624 sampler->activate();
625
626 const UnitID unit = um.createUnit("Instrument " + std::to_string(laneIndex + 1), UnitType::Sampler);
627 um.attachPlugin(unit, "com.Aestrastudios.sampler", sampler);
628 um.setUnitEnabled(unit, true);
629 if (!m_toneSamplePath.empty()) {
630 um.setUnitAudioClip(unit, m_toneSamplePath); // loads the tone into the sampler
631 }
632 um.setUnitMixerChannel(unit, mixerChannel ? mixerChannel->getChannelId() : MASTER_MIXER_CHANNEL_ID);
633 um.assignUnitToTimelineLane(unit, static_cast<int>(laneIndex));
634
635 m_laneUnits[laneIndex] = unit;
636 return unit;
637}

639void HeadlessMusicGenerator::commitPatternsToManager() {
640 // Commit once per project: re-committing would append duplicate lanes,
641 // units and clips to the TrackManager on a second exportTo() call. State is
642 // reset by createProject(). The MIDI PatternSources themselves are created
643 // in commitPlaylistToModel(): each note routes to the sampler unit backing
644 // the lane its clip is placed on, so the pattern can only be built
645 // once the placement (and thus the lane instrument) is known. Here we just
646 // prepare the shared sampler source.
647 if (m_committed) {
648 return;
649 }
650 if (!prepareToneSample()) {
651 Log::warning("[HeadlessGenerator] Tone sample unavailable — render will be silent");
652 }
653 Log::info("[HeadlessGenerator] Prepared " + std::to_string(m_patterns.size()) +
654 " patterns for placement");
655}

657void HeadlessMusicGenerator::commitPlaylistToModel() {
658 if (m_committed) {
659 return;
660 }
661 auto& patternManager = m_trackManager.getPatternManager();
662 auto& playlist = m_trackManager.getPlaylistModel();
663
664 size_t committed = 0;
665 for (const auto& clip : m_playlistClips) {
666 auto patternIt = m_patterns.find(clip.patternName);
667 if (patternIt == m_patterns.end()) {
668 continue; // validate() already rejected unknown patterns
669 }
670 const PatternData& data = patternIt->second;
671
672 const uint64_t unitId = ensureLaneInstrument(clip.laneIndex);
673 if (unitId == 0) {
674 continue;
675 }
676
677 // Convert the step-grid notes to beat-domain MidiNotes routed to the
678 // lane's sampler unit.
679 MidiPayload payload;
680 payload.notes.reserve(data.notes.size());
681 for (const auto& n : data.notes) {
682 MidiNote mn;
683 mn.pitch = n.pitch;
684 mn.startBeat = static_cast<double>(n.step) * kStepBeats;
685 mn.durationBeats = n.duration * kStepBeats;
686 mn.velocity = std::clamp(static_cast<float>(n.velocity) / 127.0f, 0.0f, 1.0f);
687 mn.unitId = unitId;
688 payload.notes.push_back(mn);
689 }
690
691 const double lengthBeats = static_cast<double>(data.length) * kStepBeats;
692 const PatternID patternId = patternManager.createMidiPattern(clip.patternName, lengthBeats, payload);
693
694 auto laneIt = m_laneIds.find(clip.laneIndex);
695 if (laneIt == m_laneIds.end()) {
696 continue;
697 }
698 playlist.addClipFromPattern(laneIt->second, patternId, clip.startBeat, clip.durationBeats);
699 ++committed;
700 }
701
702 m_committed = true;
703 Log::info("[HeadlessGenerator] Committed " + std::to_string(committed) + " clips to playlist");
704}
