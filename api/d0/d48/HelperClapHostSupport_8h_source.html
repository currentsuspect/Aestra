4// The host side of a CLAP plugin in the sandbox helper (V82-H1 / H3): the extensions a plugin may ask its host for and
5// the state they share. Included by HelperClapModule.h only.
6//
7// * clap.thread-check which thread is "main" and which is "audio": audio = inside process(), main = anywhere else
8// * clap.params the plugin changed or wants a flush. process() runs continuously, so a flush is never needed
9// on its own: its parameter output arrives with the next block
10// * clap.latency/state noted and ignored: the app re-reads latency every 64 blocks and saves state on request
11// * clap.gui the plugin asks for another size
12// * clap.timer-support, clap.posix-fd-support (POSIX only)
13// what an X11 editor needs to run (and what some plugins use with no editor at all): an
14// EditorLoopPump the helper's command loop services between commands.

33struct ClapHostContext {
34 const ClapPlugin* plugin = nullptr; // set as soon as the plugin exists: it may register timers inside init()
35
36 // The plugin's timers and descriptors live as long as the plugin does, editor open or not (some plugins register a
37 // timer in init() and depend on it for their own bookkeeping). The helper's command loop runs this pump between
38 // commands, so on_timer / on_fd arrive on the thread that loaded the plugin.
39 Aestra::Audio::EditorLoopPump pump;
40
41 std::mutex mutex; // guards the lists below
42 std::function<void(int, int)> requestResize; // set while an editor is open
43 std::vector<std::pair<uint32_t, int>> timers; // CLAP timer id -> pump token
44 std::vector<std::pair<int, int>> fds; // file descriptor -> pump token
45 uint32_t nextTimerId = 1;
46
47 std::atomic<bool> callbackRequested{false}; // clap_host.request_callback: call plugin->on_main_thread()
48
49 static bool& inProcess() {
50 static thread_local bool flag = false;
51 return flag;
52 }
53
55 void clear() {
56 std::lock_guard<std::mutex> lock(mutex);
57 for (const auto& t : timers) pump.removeTimer(t.second);
58 for (const auto& f : fds) pump.removeFd(f.second);
59 timers.clear();
60 fds.clear();
61 requestResize = nullptr;
62 plugin = nullptr;
63 }

66 void limitWait(int& timeoutMs) {
67 const int wait = pump.msUntilNextTimer();
68 if (wait >= 0 && (timeoutMs < 0 || wait < timeoutMs)) timeoutMs = wait;
69 }
70#ifndef _WIN32
72 void pollFds(std::vector<pollfd>& out, int& timeoutMs) {
73 for (const int fd : pump.fds()) out.push_back({fd, POLLIN, 0});
74 limitWait(timeoutMs);
75 }
76#endif
78 void service() {
79#ifndef _WIN32
80 std::vector<pollfd> ready;
81 for (const int fd : pump.fds()) ready.push_back({fd, POLLIN, 0});
82 if (!ready.empty() && poll(ready.data(), static_cast<nfds_t>(ready.size()), 0) > 0)
83 for (const auto& p : ready)
84 if (p.revents & (POLLIN | POLLHUP)) pump.dispatchFd(p.fd);
85#endif
86 pump.fireDueTimers();
87 }

127bool hostRegisterTimer(const ClapHost* host, uint32_t periodMs, uint32_t* timerId) {
128 auto* ctx = contextOf(host);
129 if (!ctx || !timerId) return false;
130 std::lock_guard<std::mutex> lock(ctx->mutex);
131 if (!ctx->plugin) return false;
132 const uint32_t id = ctx->nextTimerId++;
133 const int token = ctx->pump.addTimer(std::chrono::milliseconds(periodMs < 1 ? 1 : periodMs), [ctx, id]() {
134 // The extension is looked up when the timer fires: the plugin may have asked for it before it could answer.
135 const ClapPlugin* plugin = ctx->plugin;
136 const auto* ext = plugin && plugin->getExtension
137 ? static_cast<const ClapPluginTimerSupport*>(plugin->getExtension(plugin, kClapExtTimerSupport))
138 : nullptr;
139 if (ext && ext->onTimer) ext->onTimer(plugin, id);
140 });
141 ctx->timers.emplace_back(id, token);
142 *timerId = id;
143 return true;
144}
145bool hostUnregisterTimer(const ClapHost* host, uint32_t timerId) {
146 auto* ctx = contextOf(host);
147 if (!ctx) return false;
148 std::lock_guard<std::mutex> lock(ctx->mutex);
149 for (auto it = ctx->timers.begin(); it != ctx->timers.end(); ++it) {
150 if (it->first == timerId) {
151 ctx->pump.removeTimer(it->second);
152 ctx->timers.erase(it);
153 return true;
154 }
155 }
156 return false;
157}
158const ClapHostTimerSupport kHostTimerSupport = {hostRegisterTimer, hostUnregisterTimer};

160// Descriptors are watched for readability. A plugin that also asks to be told when one is writable gets read events
161// only; editors that need write readiness are not known (the VST3 run loop has no such thing either).
162bool hostRegisterFd(const ClapHost* host, int fd, uint32_t) {
163 auto* ctx = contextOf(host);
164 if (!ctx) return false;
165 std::lock_guard<std::mutex> lock(ctx->mutex);
166 if (!ctx->plugin) return false;
167 const int token = ctx->pump.addFd(fd, [ctx, fd]() {
168 const ClapPlugin* plugin = ctx->plugin;
169 const auto* ext = plugin && plugin->getExtension
170 ? static_cast<const ClapPluginPosixFdSupport*>(plugin->getExtension(plugin, kClapExtPosixFdSupport))
171 : nullptr;
172 if (ext && ext->onFd) ext->onFd(plugin, fd, kClapPosixFdRead);
173 });
174 ctx->fds.emplace_back(fd, token);
175 return true;
176}
177bool hostModifyFd(const ClapHost*, int, uint32_t) { return true; }
178bool hostUnregisterFd(const ClapHost* host, int fd) {
179 auto* ctx = contextOf(host);
180 if (!ctx) return false;
181 std::lock_guard<std::mutex> lock(ctx->mutex);
182 bool found = false;
183 for (auto it = ctx->fds.begin(); it != ctx->fds.end();) {
184 if (it->first == fd) {
185 ctx->pump.removeFd(it->second);
186 it = ctx->fds.erase(it);
187 found = true;
188 } else {
189 ++it;
190 }
191 }
192 return found;
193}

196const void* hostGetExtension(const ClapHost* host, const char* id) {
197 if (!id) return nullptr;
198 if (!std::strcmp(id, kClapExtThreadCheck)) return &kHostThreadCheck;
199 if (!std::strcmp(id, kClapExtHostParams)) return &kHostParams;
200 if (!std::strcmp(id, kClapExtHostLatency)) return &kHostLatency;
201 if (!std::strcmp(id, kClapExtHostState)) return &kHostState;
202 if (!std::strcmp(id, kClapExtGui)) return &kHostGui;
203 if (!std::strcmp(id, kClapExtTimerSupport)) return &kHostTimerSupport;
204#ifndef _WIN32
205 // File descriptors are a POSIX idea: CLAP offers posix-fd-support only there.
206 if (!std::strcmp(id, kClapExtPosixFdSupport)) return &kHostPosixFdSupport;
207#endif
208 (void)host;
209 return nullptr;
210}
