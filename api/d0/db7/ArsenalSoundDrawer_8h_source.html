49 static std::string lower(std::string s) {
50 for (auto& c : s) c = static_cast<char>(std::tolower(static_cast<unsigned char>(c)));
51 return s;
52 }

55 static std::vector<int> filtered(const std::vector<std::string>& list, const std::string& filter) {
56 std::vector<std::string> words;
57 std::string word;
58 for (char c : lower(filter)) {
59 if (c == ' ') { if (!word.empty()) words.push_back(word), word.clear(); }
60 else word.push_back(c);
61 }
62 if (!word.empty()) words.push_back(word);
63 std::vector<int> out;
64 for (size_t i = 0; i < list.size(); ++i) {
65 const std::string name = lower(nameOf(list[i]));
66 const bool all = std::all_of(words.begin(), words.end(), [&](const std::string& w) { return name.find(w) != std::string::npos; });
67 if (all) out.push_back(static_cast<int>(i));
68 }
69 return out;
70 }

73 static State open(const std::vector<std::string>& list, const std::string& current) {
74 State s;
75 s.original = current;
76 s.selected = -1;
77 const auto idx = filtered(list, "");
78 for (size_t i = 0; i < idx.size(); ++i)
79 if (list[static_cast<size_t>(idx[i])] == current) s.selected = static_cast<int>(i);
80 reveal(s, idx.size());
81 return s;
82 }

85 static void reveal(State& s, size_t count) {
86 if (s.selected < 0) { s.scroll = 0; return; }
87 if (s.selected < s.scroll) s.scroll = s.selected;
88 if (s.selected >= s.scroll + kVisibleRows) s.scroll = s.selected - kVisibleRows + 1;
89 s.scroll = std::clamp(s.scroll, 0, std::max(0, static_cast<int>(count) - kVisibleRows));
90 }

92 static void scrollBy(State& s, size_t count, int rows) {
93 s.scroll = std::clamp(s.scroll + rows, 0, std::max(0, static_cast<int>(count) - kVisibleRows));

98 static Outcome onKey(State& s, const std::vector<std::string>& list, const AestraUI::NUIKeyEvent& e) {
99 using AestraUI::NUIKeyCode;
100 if (!e.pressed) return {};
101 if (e.modifiers & AestraUI::NUIModifiers::Ctrl) return {}; // shortcuts stay shortcuts
102 auto idx = filtered(list, s.filter);
103 const auto pathAt = [&](int i) { return (i >= 0 && i < static_cast<int>(idx.size())) ? list[static_cast<size_t>(idx[static_cast<size_t>(i)])] : std::string(); };
104
105 if (e.keyCode == NUIKeyCode::Escape) return {Result::Revert, s.original};
106 if (e.keyCode == NUIKeyCode::Enter) return {Result::Keep, pathAt(s.selected)};
107 if (e.keyCode == NUIKeyCode::Down || e.keyCode == NUIKeyCode::Up) {
108 if (idx.empty()) return {Result::Consumed, {}};
109 const int step = e.keyCode == NUIKeyCode::Down ? 1 : -1;
110 s.selected = s.selected < 0 ? (step > 0 ? 0 : static_cast<int>(idx.size()) - 1)
111 : std::clamp(s.selected + step, 0, static_cast<int>(idx.size()) - 1);
112 reveal(s, idx.size());
113 return {Result::Swap, pathAt(s.selected)};
114 }
115 if (e.keyCode == NUIKeyCode::Backspace) {
116 if (!s.filter.empty()) s.filter.pop_back();
117 } else if (e.character >= 32 && e.character < 127) {
118 s.filter.push_back(e.character);
119 } else {
120 return {Result::Consumed, {}}; // the drawer owns the keyboard while it is open
121 }
122 idx = filtered(list, s.filter);
123 s.selected = idx.empty() ? -1 : 0;
124 s.scroll = 0;
125 return {Result::Consumed, {}};
126 }

129 static int rowAt(const State& s, size_t count, const AestraUI::NUIRect& r, const AestraUI::NUIPoint& p) {
130 if (!r.contains(p) || p.y < r.y + kSearchH) return -1;
131 const int row = s.scroll + static_cast<int>((p.y - r.y - kSearchH) / kRowH);
132 return (row >= 0 && row < static_cast<int>(count) && row < s.scroll + kVisibleRows) ? row : -1;
133 }

138 static void draw(AestraUI::NUIRenderer& renderer, const AestraUI::NUIRect& r, const State& s,
139 const std::vector<std::string>& list, const std::string& current, const AestraUI::NUIColor& accent,
140 const std::string& setLabel = {}, size_t missing = 0, const AestraUI::NUIRect& outer = {}) {
141 auto& theme = AestraUI::NUIThemeManager::getInstance();
142 const auto& tp = theme.getCurrentTheme();
143 renderer.fillRect(r, theme.getColor("backgroundPrimary"));
144
145 // Search line: what was typed, or what you can type; how many sounds match at the right.
146 const auto idx = filtered(list, s.filter);
147 const AestraUI::NUIRect search{r.x, r.y, r.width, kSearchH};
148 renderer.fillRect({r.x, search.bottom() - 1.0f, r.width, 1.0f}, theme.getColor("borderSubtle"));
149 const std::string shown = s.filter.empty() ? (setLabel.empty() ? "Type to search this folder" : "Type to search this set") : s.filter + "|";
150 renderer.drawText(shown, {r.x + 12.0f, std::round(renderer.calculateTextY(search, tp.fontSizeS))}, tp.fontSizeS,
151 s.filter.empty() ? theme.getColor("textSecondary").withAlpha(0.6f) : theme.getColor("textPrimary"));
152 const auto chip = chipRect(r);
153 renderer.fillRect(chip, theme.getColor("surfaceTertiary"));
154 const std::string label = setLabel.empty() ? "Folder" : setLabel;
155 ArsenalClip::within(renderer, {chip.x, chip.y, chip.width - 16.0f, chip.height}, outer);
156 renderer.drawText(label, {chip.x + 8.0f, std::round(renderer.calculateTextY(chip, tp.fontSizeXS))}, tp.fontSizeXS,
157 setLabel.empty() ? theme.getColor("textSecondary") : theme.getColor("textPrimary"));
158 ArsenalClip::restore(renderer, outer);
159 renderer.drawLine({chip.right() - 13.0f, chip.y + chip.height * 0.5f - 1.5f}, {chip.right() - 10.0f, chip.y + chip.height * 0.5f + 1.5f}, 1.2f, theme.getColor("textSecondary"));
160 renderer.drawLine({chip.right() - 10.0f, chip.y + chip.height * 0.5f + 1.5f}, {chip.right() - 7.0f, chip.y + chip.height * 0.5f - 1.5f}, 1.2f, theme.getColor("textSecondary"));
161 std::string count = std::to_string(idx.size()) + (idx.size() == 1 ? " sound" : " sounds");
162 float right = chip.x - 12.0f;
163 if (missing > 0) { // a member that is gone is said so, in amber
164 const std::string gone = std::to_string(missing) + " missing";
165 const float gw = renderer.measureText(gone, tp.fontSizeXS).width;
166 renderer.drawText(gone, {right - gw, std::round(renderer.calculateTextY(search, tp.fontSizeXS))}, tp.fontSizeXS, theme.getColor("warning"));
167 right -= gw + 12.0f;
168 }
169 const float cw = renderer.measureText(count, tp.fontSizeXS).width;
170 renderer.drawText(count, {right - cw, std::round(renderer.calculateTextY(search, tp.fontSizeXS))}, tp.fontSizeXS,
171 theme.getColor("textSecondary").withAlpha(0.8f));
172
173 if (idx.empty()) {
174 renderer.drawTextCentered("No match", {r.x, search.bottom(), r.width, kRowH * kVisibleRows}, tp.fontSizeS,
175 theme.getColor("textSecondary").withAlpha(0.6f));
176 return;
177 }
178 ArsenalClip::within(renderer, {r.x, search.bottom(), r.width, kRowH * kVisibleRows}, outer);
179 for (int v = 0; v < kVisibleRows; ++v) {
180 const int row = s.scroll + v;
181 if (row >= static_cast<int>(idx.size())) break;
182 const std::string& path = list[static_cast<size_t>(idx[static_cast<size_t>(row)])];
183 const AestraUI::NUIRect rr{r.x, search.bottom() + v * kRowH, r.width - 8.0f, kRowH};
184 if (row == s.selected) renderer.fillRect(rr, theme.getColor("surfaceRaised"));
185 if (path == current) renderer.fillRect({rr.x, rr.y + 4.0f, 3.0f, rr.height - 8.0f}, accent); // the sound it has now
186 renderer.drawText(nameOf(path), {rr.x + 14.0f, std::round(renderer.calculateTextY(rr, tp.fontSizeS))}, tp.fontSizeS,
187 row == s.selected ? theme.getColor("textPrimary") : theme.getColor("textPrimary").withAlpha(0.78f));
188 }
189 ArsenalClip::restore(renderer, outer);
190
191 // A thin scroll thumb when there are more matches than rows.
192 if (static_cast<int>(idx.size()) > kVisibleRows) {
193 const float track = kRowH * kVisibleRows;
194 const float thumbH = std::max(16.0f, track * kVisibleRows / static_cast<float>(idx.size()));
195 const float thumbY = search.bottom() + (track - thumbH) * s.scroll / std::max(1.0f, static_cast<float>(idx.size() - kVisibleRows));
196 renderer.fillRect({r.right() - 5.0f, thumbY, 2.0f, thumbH}, theme.getColor("borderSubtle").withAlpha(0.9f));
197 }
198 }
