3// muse-agent — interchangeable intelligence for the Muse protocol.
4//
5// Connects to a Muse socket (the live app with AESTRA_MUSE_PORT, or
6// `MuseRepl --port N` headless), hands the tool manifest to a model
7// provider, and runs the session loop under hard budgets. Muse owns the
8// musical system; the model only proposes intent.
9//
10// muse-agent --connect 41952 --provider anthropic --brief "make a 140 BPM trap loop"
11// muse-agent --connect 41952 --provider openai --base-url http://127.0.0.1:8080 \
12// --model llama-3.1-8b --brief "add reverb to track 0"
13//
14// API keys come from the environment, never flags: ANTHROPIC_API_KEY or
15// OPENAI_API_KEY / MUSE_AGENT_API_KEY. Local OpenAI-compatible servers
16// (llama.cpp, Ollama, LM Studio) usually need no key.

32void printUsage() {
33 std::cout <<
34 "muse-agent: drive a Muse session with a model.\n"
35 " --connect <port|host:port> Muse socket (default host 127.0.0.1)\n"
36 " --brief \"...\" the production brief (required)\n"
37 " --provider anthropic|openai (default openai — local-first)\n"
38 " --model <id> model id (defaults: claude-opus-4-8 / server default)\n"
39 " --base-url <url> OpenAI-compatible endpoint\n"
40 " (default http://127.0.0.1:8080 — llama.cpp server)\n"
41 " --mode direct|collab|auto interruption policy (default auto)\n"
42 " --max-iterations <n> model-turn budget (default 25)\n"
43 " --max-tool-calls <n> Muse-request budget (default 200)\n"
44 " --max-seconds <n> wall-clock budget (default 900)\n"
45 " --verbose log every verb and response to stderr\n";
46}

48std::string envOr(const char* primary, const char* fallback) {
49 if (const char* value = std::getenv(primary); value && *value) return value;

58int main(int argc, char** argv) {
59 std::string connect = "41952";
60 std::string brief;
61 std::string providerName = "openai";
62 std::string model;
63 std::string baseUrl = "http://127.0.0.1:8080";
64 AgentLoop::Options options;
65
66 for (int i = 1; i < argc; ++i) {
67 const std::string arg = argv[i];
68 const auto next = [&](const char* flag) -> std::string {
69 if (i + 1 >= argc) {
70 std::cerr << "missing value for " << flag << "\n";
71 std::exit(2);
72 }
73 return argv[++i];
74 };
75 if (arg == "--connect") connect = next("--connect");
76 else if (arg == "--brief") brief = next("--brief");
77 else if (arg == "--provider") providerName = next("--provider");
78 else if (arg == "--model") model = next("--model");
79 else if (arg == "--base-url") baseUrl = next("--base-url");
80 else if (arg == "--mode") {
81 const std::string mode = next("--mode");
82 if (mode == "direct") options.mode = AgentLoop::Mode::Direct;
83 else if (mode == "collab") options.mode = AgentLoop::Mode::Collaborative;
84 else if (mode == "auto") options.mode = AgentLoop::Mode::Autonomous;
85 else { std::cerr << "unknown mode: " << mode << "\n"; return 2; }
86 }
87 else if (arg == "--max-iterations") options.maxIterations = std::stoi(next(arg.c_str()));
88 else if (arg == "--max-tool-calls") options.maxToolCalls = std::stoi(next(arg.c_str()));
89 else if (arg == "--max-seconds") options.maxSeconds = std::stoi(next(arg.c_str()));
90 else if (arg == "--verbose") options.verbose = true;
91 else if (arg == "--help" || arg == "-h") { printUsage(); return 0; }
92 else { std::cerr << "unknown argument: " << arg << "\n"; return 2; }
93 }
94
95 if (brief.empty()) {
96 std::cerr << "a --brief is required\n";
97 printUsage();
98 return 2;
99 }
100
101 // Resolve the Muse endpoint.
102 std::string host = "127.0.0.1";
103 std::string portText = connect;
104 if (const auto colon = connect.find(':'); colon != std::string::npos) {
105 host = connect.substr(0, colon);
106 portText = connect.substr(colon + 1);
107 }
108 const int port = std::atoi(portText.c_str());
109 if (port <= 0 || port > 65535) {
110 std::cerr << "invalid port in --connect: " << connect << "\n";
111 return 2;
112 }
113
114 MuseSocketClient client;
115 std::string error;
116 if (!client.connect(host, static_cast<uint16_t>(port), error)) {
117 std::cerr << error << "\n"
118 << "Is Muse listening? Start the app with AESTRA_MUSE_PORT=" << port
119 << " or run: MuseRepl --port " << port << "\n";
120 return 1;
121 }
122
123 // Resolve the brain. Local-first: the OpenAI-compatible adapter covers
124 // llama.cpp server, Ollama, and LM Studio out of the box; hosted keys
125 // plug in the same way (BYOK).
126 std::unique_ptr<ModelProvider> provider;
127 if (providerName == "anthropic") {
128 const std::string apiKey = envOr("ANTHROPIC_API_KEY", "MUSE_AGENT_API_KEY");
129 if (apiKey.empty()) {
130 std::cerr << "ANTHROPIC_API_KEY is not set\n";
131 return 1;
132 }
133 provider = std::make_unique<AnthropicProvider>(
134 apiKey, model.empty() ? "claude-opus-4-8" : model);
135 } else if (providerName == "openai") {
136 const std::string apiKey = envOr("OPENAI_API_KEY", "MUSE_AGENT_API_KEY");
137 provider = std::make_unique<OpenAICompatProvider>(
138 apiKey, model.empty() ? "default" : model, baseUrl);
139 } else {
140 std::cerr << "unknown provider: " << providerName << "\n";
141 return 2;
142 }
143
144 AgentLoop loop(*provider,
145 [&client](const std::string& line) { return client.request(line); },
146 options);
147
148 std::cerr << "[muse-agent] " << provider->name() << " -> " << host << ":" << port
149 << "\n";
150 const AgentLoop::Outcome outcome = loop.run(brief);
151
152 std::cout << "\n=== muse-agent " << outcome.stopReason << " ===\n"
153 << outcome.summary << "\n"
154 << "turns: " << outcome.iterations << ", muse calls: " << outcome.toolCalls
155 << ", tokens in/out: " << outcome.usage.inputTokens << "/"
156 << outcome.usage.outputTokens << "\n";
157 return outcome.finished ? 0 : 1;
158}
