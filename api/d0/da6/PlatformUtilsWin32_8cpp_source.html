40std::string PlatformUtilsWin32::openFileDialog(const std::string& title, const std::string& filter) const {
41 OPENFILENAMEA ofn = {};
42 char filename[MAX_PATH] = "";
43
44 // Windows filter format: "Description\0*.ext\0Description2\0*.ext2\0\0"
45 // If filter is empty or doesn't contain embedded nulls, use a default
46 // The filter string must contain the full null-separated format
47 const char* defaultFilter = "All Files\0*.*\0";
48 const char* filterPtr = filter.empty() ? defaultFilter : filter.data();
49
50 ofn.lStructSize = sizeof(OPENFILENAMEA);
51 ofn.hwndOwner = nullptr;
52 ofn.lpstrFilter = filterPtr;
53 ofn.lpstrFile = filename;
54 ofn.nMaxFile = MAX_PATH;
55 ofn.lpstrTitle = title.c_str();
56 ofn.Flags = OFN_FILEMUSTEXIST | OFN_PATHMUSTEXIST | OFN_NOCHANGEDIR;
57
58 if (GetOpenFileNameA(&ofn)) {
59 return std::string(filename);
60 }
61
62 return "";
63}

65std::string PlatformUtilsWin32::saveFileDialog(const SaveFileDialogOptions& options) const {
66 OPENFILENAMEA ofn = {};
67 char filename[MAX_PATH] = "";
68
69 const char* defaultFilter = "All Files\0*.*\0";
70 std::string normalizedFilter = options.filter;
71 if (!normalizedFilter.empty() && normalizedFilter.back() != '\0') {
72 normalizedFilter.push_back('\0');
73 }
74 const char* filterPtr = normalizedFilter.empty() ? defaultFilter : normalizedFilter.c_str();
75
76 if (!options.defaultPath.empty()) {
77 strncpy_s(filename, options.defaultPath.c_str(), _TRUNCATE);
78 }
79
80 ofn.lStructSize = sizeof(OPENFILENAMEA);
81 ofn.hwndOwner = nullptr;
82 ofn.lpstrFilter = filterPtr;
83 ofn.lpstrFile = filename;
84 ofn.nMaxFile = MAX_PATH;
85 ofn.lpstrTitle = options.title.c_str();
86 ofn.lpstrDefExt = options.defaultExtension.empty() ? nullptr : options.defaultExtension.c_str();
87 ofn.Flags = OFN_OVERWRITEPROMPT | OFN_NOCHANGEDIR;
88
89 if (GetSaveFileNameA(&ofn)) {
90 return std::string(filename);
91 }
92
93 return "";
94}

96std::string PlatformUtilsWin32::selectFolderDialog(const std::string& title) const {
97 BROWSEINFOA bi = {};
98 char path[MAX_PATH];
99
100 bi.lpszTitle = title.c_str();
101 bi.ulFlags = BIF_RETURNONLYFSDIRS | BIF_NEWDIALOGSTYLE;
102
103 LPITEMIDLIST pidl = SHBrowseForFolderA(&bi);
104 if (pidl != nullptr) {
105 if (SHGetPathFromIDListA(pidl, path)) {
106 CoTaskMemFree(pidl);
107 return std::string(path);
108 }
109 CoTaskMemFree(pidl);
110 }
111
112 return "";
113}

119void PlatformUtilsWin32::setClipboardText(const std::string& text) const {
120 if (!OpenClipboard(nullptr)) {
121 return;
122 }
123
124 EmptyClipboard();
125
126 HGLOBAL hMem = GlobalAlloc(GMEM_MOVEABLE, text.size() + 1);
127 if (hMem) {
128 char* pMem = (char*)GlobalLock(hMem);
129 if (pMem) {
130 memcpy(pMem, text.c_str(), text.size() + 1);
131 GlobalUnlock(hMem);
132 SetClipboardData(CF_TEXT, hMem);
133 }
134 }
135
136 CloseClipboard();
137}

139std::string PlatformUtilsWin32::getClipboardText() const {
140 if (!OpenClipboard(nullptr)) {
141 return "";
142 }
143
144 std::string result;
145 HANDLE hData = GetClipboardData(CF_TEXT);
146 if (hData) {
147 char* pData = (char*)GlobalLock(hData);
148 if (pData) {
149 result = std::string(pData);
150 GlobalUnlock(hData);
151 }
152 }
153
154 CloseClipboard();
155 return result;
156}
