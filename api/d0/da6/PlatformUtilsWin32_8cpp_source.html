43std::string PlatformUtilsWin32::openFileDialog(const std::string& title, const std::string& filter) const {
44 OPENFILENAMEA ofn = {};
45 char filename[MAX_PATH] = "";
46
47 // Windows filter format: "Description\0*.ext\0Description2\0*.ext2\0\0"
48 // If filter is empty or doesn't contain embedded nulls, use a default
49 // The filter string must contain the full null-separated format
50 const char* defaultFilter = "All Files\0*.*\0";
51 std::string normalizedFilter = filter;
52 if (!normalizedFilter.empty() && normalizedFilter.back() != '\0') {
53 normalizedFilter.push_back('\0');
54 }
55 const char* filterPtr = normalizedFilter.empty() ? defaultFilter : normalizedFilter.c_str();
56
57 ofn.lStructSize = sizeof(OPENFILENAMEA);
58 ofn.hwndOwner = nullptr;
59 ofn.lpstrFilter = filterPtr;
60 ofn.lpstrFile = filename;
61 ofn.nMaxFile = MAX_PATH;
62 ofn.lpstrTitle = title.c_str();
63 ofn.Flags = OFN_FILEMUSTEXIST | OFN_PATHMUSTEXIST | OFN_NOCHANGEDIR;
64
65 if (GetOpenFileNameA(&ofn)) {
66 return std::string(filename);
67 }
68
69 return "";
70}

73 OPENFILENAMEA ofn = {};
74 char filename[MAX_PATH] = "";

76 const char* defaultFilter = "All Files\0*.*\0";
77 std::string normalizedFilter = options.filter;
78 if (!normalizedFilter.empty() && normalizedFilter.back() != '\0') {

88 ofn.hwndOwner = nullptr;
89 ofn.lpstrFilter = filterPtr;
90 ofn.lpstrFile = filename;

105std::wstring widenUtf8(const std::string& s) {
106 if (s.empty()) return {};
107 const int n = MultiByteToWideChar(CP_UTF8, 0, s.data(), static_cast<int>(s.size()), nullptr, 0);
108 std::wstring w(static_cast<size_t>(n), L'\0');
109 MultiByteToWideChar(CP_UTF8, 0, s.data(), static_cast<int>(s.size()), w.data(), n);
110 return w;
111}
112
113std::string narrowUtf8(const wchar_t* w) {
114 if (!w || !*w) return {};
115 const int n = WideCharToMultiByte(CP_UTF8, 0, w, -1, nullptr, 0, nullptr, nullptr);
116 if (n <= 1) return {};
117 std::string s(static_cast<size_t>(n - 1), '\0');
118 WideCharToMultiByte(CP_UTF8, 0, w, -1, s.data(), n, nullptr, nullptr);
119 return s;
120}

130// The Vista+ common item dialog in folder mode: the same modern Explorer
131// window as Open/Save, Unicode throughout. (SHBrowseForFolderA was the XP-era
132// tree control and returned ANSI, mangling names outside the system codepage.)
133std::string PlatformUtilsWin32::selectFolderDialog(const std::string& title) const {
134 const HRESULT init = CoInitializeEx(nullptr, COINIT_APARTMENTTHREADED | COINIT_DISABLE_OLE1DDE);
135 const bool uninit = SUCCEEDED(init); // S_FALSE (already initialised) also needs balancing
136 if (init == RPC_E_CHANGED_MODE) {
137 // The thread is already MTA; the shell dialog wants STA and may not show.
138 Aestra::Log::warning("[Platform] Folder dialog on an MTA thread; the picker may fail");
139 }
140
141 std::string result;
142 IFileOpenDialog* dialog = nullptr;
143 const HRESULT created =
144 CoCreateInstance(CLSID_FileOpenDialog, nullptr, CLSCTX_INPROC_SERVER, IID_PPV_ARGS(&dialog));
145 if (FAILED(created)) {
146 // A failure must not read as the user cancelling: say so.
147 Aestra::Log::warning("[Platform] Folder dialog could not be created: " + hresultHex(created));
148 } else {
149 DWORD options = 0;
150 if (SUCCEEDED(dialog->GetOptions(&options))) {
151 dialog->SetOptions(options | FOS_PICKFOLDERS | FOS_FORCEFILESYSTEM | FOS_PATHMUSTEXIST);
152 }
153 if (!title.empty()) {
154 const std::wstring wideTitle = widenUtf8(title);
155 dialog->SetTitle(wideTitle.c_str());
156 }
157 const HRESULT shown = dialog->Show(GetActiveWindow());
158 if (FAILED(shown) && shown != HRESULT_FROM_WIN32(ERROR_CANCELLED)) {
159 Aestra::Log::warning("[Platform] Folder dialog failed: " + hresultHex(shown));
160 }
161 if (SUCCEEDED(shown)) {
162 IShellItem* item = nullptr;
163 if (SUCCEEDED(dialog->GetResult(&item))) {
164 PWSTR path = nullptr;
165 if (SUCCEEDED(item->GetDisplayName(SIGDN_FILESYSPATH, &path))) {
166 result = narrowUtf8(path);
167 CoTaskMemFree(path);
168 }
169 item->Release();
170 }
171 }
172 dialog->Release();
173 }
174
175 if (uninit) CoUninitialize();
176 return result;
177}

179std::string PlatformUtilsWin32::getKnownFolderPath(KnownFolder folder) const {
180 const KNOWNFOLDERID* id = nullptr;
181 switch (folder) {
182 case KnownFolder::Desktop: id = &FOLDERID_Desktop; break;
183 case KnownFolder::Downloads: id = &FOLDERID_Downloads; break;
184 case KnownFolder::Documents: id = &FOLDERID_Documents; break;
185 case KnownFolder::Music: id = &FOLDERID_Music; break;
186 }
187 if (!id) return {};
188 PWSTR path = nullptr;
189 std::string result;
190 if (SUCCEEDED(SHGetKnownFolderPath(*id, KF_FLAG_DEFAULT, nullptr, &path))) {
191 result = narrowUtf8(path);
192 }
193 CoTaskMemFree(path); // required even on failure
194 return result;
195}

201void PlatformUtilsWin32::setClipboardText(const std::string& text) const {
202 if (!OpenClipboard(nullptr)) {
203 return;
204 }
205
206 EmptyClipboard();
207
208 HGLOBAL hMem = GlobalAlloc(GMEM_MOVEABLE, text.size() + 1);
209 if (hMem) {
210 char* pMem = (char*)GlobalLock(hMem);
211 if (pMem) {
212 memcpy(pMem, text.c_str(), text.size() + 1);
213 GlobalUnlock(hMem);
214 SetClipboardData(CF_TEXT, hMem);
215 }
216 }
217
218 CloseClipboard();
219}

221std::string PlatformUtilsWin32::getClipboardText() const {
222 if (!OpenClipboard(nullptr)) {
223 return "";
224 }
225
226 std::string result;
227 HANDLE hData = GetClipboardData(CF_TEXT);
228 if (hData) {
229 char* pData = (char*)GlobalLock(hData);
230 if (pData) {
231 result = std::string(pData);
232 GlobalUnlock(hData);
233 }
234 }
235
236 CloseClipboard();
237 return result;
238}
