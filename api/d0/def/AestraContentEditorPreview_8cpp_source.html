3// The Sample Editor's one-off actions (normalize, reverse) and both editors' Preview. Preview in the Sample Editor
4// sounds the unit itself (AuditionUnit at its root note, for the trimmed sample's length), so trim, envelope and
5// pitch are all heard; in the Audio Clip Editor it plays the clip's part of its source through the preview engine.
6#include "../Panels/AudioClipEditorPanel.h"
7#include "../Panels/SampleEditorPanel.h"

10#include "Plugin/SamplerPlugin.h"
11#include "PreviewEngine.h"
12#include "TrackManager.h"

14void AestraContent::wireEditorActions() {
15 if (!m_sampleEditorPanel || !m_audioClipEditorPanel)
16 return; // called once both editors are built
17 m_sampleEditorPanel->onNormalizeRequested = [this]() {
18 if (!m_trackManager || !m_sampleEditorUnitId) {
19 return;
20 }
21 auto plugin = m_trackManager->getUnitManager().getUnitPlugin(m_sampleEditorUnitId);
22 auto sampler = std::dynamic_pointer_cast<Aestra::Audio::Plugins::SamplerPlugin>(plugin);
23 if (!sampler) {
24 return;
25 }
26 sampler->normalizeSample();
27 };
28 m_sampleEditorPanel->onReverseRequested = [this]() {
29 if (!m_trackManager || !m_sampleEditorUnitId) {
30 return;
31 }
32 auto plugin = m_trackManager->getUnitManager().getUnitPlugin(m_sampleEditorUnitId);
33 auto sampler = std::dynamic_pointer_cast<Aestra::Audio::Plugins::SamplerPlugin>(plugin);
34 if (!sampler) {
35 return;
36 }
37 sampler->reverseSample();
38 };
39 // Preview sounds the unit itself (AuditionUnit at its root note), so trim, envelope and pitch are all heard.
40 m_sampleEditorPanel->onPreviewRequested = [this](bool start, double seconds) {
41 if (!m_audioEngine || !m_sampleEditorUnitId)
42 return;
43 Aestra::Audio::AudioQueueCommand cmd;
44 cmd.type = Aestra::Audio::AudioQueueCommandType::AuditionUnit;
45 cmd.trackIndex = static_cast<uint32_t>(m_sampleEditorUnitId);
46 if (!start) {
47 cmd.value1 = -1.0f; // stop this unit's preview
48 } else {
49 cmd.value1 = static_cast<float>(m_sampleEditorPanel->getPitchTune().rootMidiNote);
50 cmd.value2 = 100.0f / 127.0f;
51 cmd.samplePos = static_cast<uint64_t>(seconds * static_cast<double>(m_audioEngine->getSampleRate()));
52 }
53 m_audioEngine->commandQueue().push(cmd);
54 };
55 // Clip preview: the clip's part of its source through the preview engine (the browser's), then seek to its start.
56 m_audioClipEditorPanel->setOnPreviewRequested(
57 [this](const std::string& path, double startSeconds, double lengthSeconds, float rate, float gainDb) {
58 auto* preview = getPreviewEngine();
59 if (!preview)
60 return;
61 if (path.empty()) {
62 preview->stop();
63 return;
64 }
65 preview->play(path, gainDb, lengthSeconds, rate);
66 if (startSeconds > 0.0)
67 preview->seek(startSeconds);
68 });
69}
