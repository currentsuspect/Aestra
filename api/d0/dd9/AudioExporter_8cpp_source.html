28AudioExporter::Result AudioExporter::bounceToWav(AudioEngine& engine, TrackManager& trackManager,
29 double startBeat, double endBeat,
30 const std::string& outputPath, int32_t trackId) {
31 if (trackId >= 0) {
32 AudioExporter::Result unsupported;
33 unsupported.outputPath = outputPath;
34 unsupported.errorMessage =
35 "Isolated-track bounce is not yet supported by AudioExporter::bounceToWav";
36 return unsupported;
37 }
38
39 // Validate beat range
40 if (!std::isfinite(startBeat) || !std::isfinite(endBeat)) {
41 AudioExporter::Result invalid;
42 invalid.outputPath = outputPath;
43 invalid.errorMessage = "Invalid beat range: non-finite beat input";
44 return invalid;
45 }
46 if (endBeat <= startBeat) {
47 AudioExporter::Result invalid;
48 invalid.outputPath = outputPath;
49 invalid.errorMessage = "Invalid beat range: endBeat must be greater than startBeat";
50 return invalid;
51 }
52
53 AudioExporter exporter(engine, trackManager);
54 Config config;
55 config.outputPath = outputPath;
56 config.startBeat = startBeat;
57 config.endBeat = endBeat;
58 // Use FullSong scope — computeRenderDurationBeats now handles beat ranges
59 // under FullSong. Selection scope reads startTimeSeconds/endTimeSeconds
60 // which bounceToWav does not set, causing zero-duration renders.
61 config.scope = RenderScope::FullSong;
62 config.sampleRate = engine.getSampleRate();
63 config.bitDepth = BitDepth::Float_32;
64 config.numChannels = 2;
65 config.tailSeconds = 0.0;
66 return exporter.render(config);
67}

69AudioExporter::Result AudioExporter::render(const Config& config) {
70 Result result;
71 result.outputPath = config.outputPath;
72
73 if (config.outputPath.empty()) {
74 result.errorMessage = "No output path specified";
75 return result;
76 }
77 if (config.sampleRate == 0) {
78 result.errorMessage = "Invalid sample rate";
79 return result;
80 }
81 if (config.numChannels == 0 || config.numChannels > 64) {
82 result.errorMessage = "Invalid channel count";
83 return result;
84 }
85
86 // Compute render duration from actual playlist
87 double startBeat = 0.0;
88 double durationBeats = computeRenderDurationBeats(config, startBeat);
89 if (durationBeats <= 0.0) {
90 switch (config.scope) {
91 case RenderScope::FullSong:
92 result.errorMessage = "Nothing to render (empty timeline)";
93 break;
94 case RenderScope::LoopRegion:
95 result.errorMessage = "Nothing to render (invalid or empty loop region)";
96 break;
97 case RenderScope::Selection:
98 result.errorMessage = "Nothing to render (no selection range)";
99 break;
100 }
101 return result;
102 }
103
104 // Add tail
105 double tailBeats = 0.0;
106 if (config.tailSeconds > 0.0) {
107 double bpm = m_engine.getBPM();
108 if (bpm > 0.0) {
109 double secondsPerBeat = 60.0 / bpm;
110 tailBeats = config.tailSeconds / secondsPerBeat;
111 }
112 }
113 durationBeats += tailBeats;
114
115 // Convert beats to samples
116 double bpm = m_engine.getBPM();
117 double sampleRate = static_cast<double>(config.sampleRate);
118 if (bpm <= 0.0) bpm = 120.0;
119 double samplesPerBeat = (sampleRate * 60.0) / bpm;
120 uint64_t startSample = static_cast<uint64_t>(startBeat * samplesPerBeat);
121 uint64_t totalFrames = static_cast<uint64_t>(durationBeats * samplesPerBeat);
122
123 if (totalFrames == 0) {
124 result.errorMessage = "Nothing to render (zero frames)";
125 return result;
126 }
127 int bitsPerSample = static_cast<int>(config.bitDepth);
128 uint64_t blockAlign64 = static_cast<uint64_t>(config.numChannels) * static_cast<uint64_t>(bitsPerSample / 8);
129 if (blockAlign64 == 0 || blockAlign64 > std::numeric_limits<uint16_t>::max() ||
130 totalFrames > std::numeric_limits<uint32_t>::max() / blockAlign64 ||
131 36ull + totalFrames * blockAlign64 > std::numeric_limits<uint32_t>::max()) {
132 result.errorMessage = "WAV export is too large for RIFF/WAV; use a shorter range or lower format settings";
133 return result;
134 }
135
136 // Open output file
137 std::ofstream file(config.outputPath, std::ios::binary);
138 if (!file) {
139 result.errorMessage = "Cannot open output file: " + config.outputPath;
140 return result;
141 }
142
143 // Write placeholder WAV header
144 if (!writeWavHeader(file, config, totalFrames)) {
145 result.errorMessage = "Failed to write WAV header";
146 return result;
147 }
148
149 // Setup render state
150 m_isRendering.store(true, std::memory_order_release);
151 m_cancelled.store(false, std::memory_order_release);
152 m_peakLevel.store(0.0f, std::memory_order_release);
153 m_lastProgressTime = std::chrono::steady_clock::now();
154 m_progressInterval = config.progressInterval;
155
156 struct RenderGuard {
157 AudioExporter& exporter;
158 explicit RenderGuard(AudioExporter& e) : exporter(e) {}
159 ~RenderGuard() { exporter.m_isRendering.store(false, std::memory_order_release); }
160 } guard(*this);
161
162 Log::info("[Export] Starting render: " + config.outputPath);
163 Log::info("[Export] Duration: " + std::to_string(durationBeats) + " beats (" +
164 std::to_string(static_cast<double>(totalFrames) / sampleRate) + "s), " +
165 "SampleRate: " + std::to_string(config.sampleRate) + ", " +
166 "BitDepth: " + bitDepthToString(config.bitDepth));
167
168 // Pre-allocate render buffer — always stereo (processBlock renders 2ch)
169 m_renderBufferF.resize(static_cast<size_t>(RENDER_BLOCK_FRAMES) * 2u);
170 m_exportDither.setSeed(static_cast<uint32_t>(startSample) ^
171 (static_cast<uint32_t>(config.sampleRate) * 0x9E3779B9u) ^
172 static_cast<uint32_t>(config.bitDepth));
173
174 // Save original engine state
175 uint32_t originalSampleRate = m_engine.getSampleRate();
176 bool wasPlaying = m_engine.isTransportPlaying();
177 uint64_t savedSamplePos = m_engine.getGlobalSamplePos();
178
179 // Stop transport for safe offline rendering
180 if (wasPlaying) {
181 m_engine.setTransportPlaying(false);
182 for (int i = 0; i < 20 && m_engine.isTransportPlaying(); ++i) {
183 std::this_thread::sleep_for(std::chrono::milliseconds(2));
184 }
185 }
186
187 // Set engine to export sample rate
188 m_engine.setSampleRate(config.sampleRate);
189 m_trackManager.setOutputSampleRate(sampleRate);
190
191 // Offline export should follow the exact live engine path to avoid render-path
192 // mismatches between playback and export.
193 const bool wasMetronomeEnabled = m_engine.isMetronomeEnabled();
194 const bool wasAuditionEnabled = m_engine.isAuditionModeEnabled();
195 m_engine.setMetronomeEnabled(false);
196 m_engine.setAuditionModeEnabled(false);
197 m_engine.setGlobalSamplePos(startSample);
198 m_engine.setTransportPlaying(true);
199
200 // Zero the render buffer before the first block so stale DSP state from
201 // prior playback cannot bleed into the export render.
202 std::fill(m_renderBufferF.begin(), m_renderBufferF.end(), 0.0f);
203
204 // Phase 2: clear the engine's true-peak max-hold so the export reading
205 // reflects only this render pass, not stale peaks from prior playback.
206 m_engine.clearTruePeakHold();
207
208 // Render loop using AudioRenderer::renderBlock (same path as bounceRangeToWav)
209 uint64_t framesRemaining = totalFrames;
210 result.framesRendered = 0;
211 while (framesRemaining > 0 && !shouldCancel()) {
212 uint32_t framesThisBlock = static_cast<uint32_t>(
213 std::min<uint64_t>(RENDER_BLOCK_FRAMES, framesRemaining));
214
215 m_engine.processBlock(m_renderBufferF.data(), nullptr, framesThisBlock, 0.0);
216 m_engine.performNonRealtimeMaintenance();
217
218 // Track peak level
219 float blockPeak = calculatePeakDb(m_renderBufferF.data(), framesThisBlock, config.numChannels);
220 m_peakLevel.store(std::max(m_peakLevel.load(std::memory_order_relaxed), blockPeak),
221 std::memory_order_relaxed);
222
223 // Write samples
224 bool writeOk = false;
225 switch (config.bitDepth) {
226 case BitDepth::PCM_16:
227 writeOk = writeSamples<int16_t>(file, m_renderBufferF.data(), framesThisBlock, config.numChannels);
228 break;
229 case BitDepth::PCM_24:
230 writeOk = writeSamples<int32_t>(file, m_renderBufferF.data(), framesThisBlock, config.numChannels);
231 break;
232 case BitDepth::Float_32:
233 writeOk = writeSamples<float>(file, m_renderBufferF.data(), framesThisBlock, config.numChannels);
234 break;
235 }
236
237 if (!writeOk) {
238 result.errorMessage = "Failed to write audio data";
239 break;
240 }
241
242 // Advance position
243 framesRemaining -= framesThisBlock;
244 result.framesRendered += framesThisBlock;
245
246 // Update progress
247 float progress = static_cast<float>(result.framesRendered) / static_cast<float>(totalFrames);
248 updateProgress(progress);
249 }
250
251 // Restore engine state
252 m_engine.setTransportPlaying(false);
253 m_engine.setSampleRate(originalSampleRate);
254 m_trackManager.setOutputSampleRate(static_cast<double>(originalSampleRate));
255 m_engine.setMetronomeEnabled(wasMetronomeEnabled);
256 m_engine.setAuditionModeEnabled(wasAuditionEnabled);
257 m_engine.setGlobalSamplePos(savedSamplePos);
258 if (wasPlaying) {
259 m_engine.setTransportPlaying(true);
260 }
261
262 // Check if render completed successfully
263 if (framesRemaining == 0 && result.framesRendered > 0) {
264 result.success = true;
265 }
266
267 if (!result.success) {
268 if (shouldCancel()) {
269 result.errorMessage = "Render cancelled by user";
270 } else {
271 result.errorMessage = "Render failed during audio processing";
272 }
273 }
274
275 if (result.errorMessage.empty()) {
276 // Rewrite WAV header with actual frame count
277 file.seekp(0, std::ios::beg);
278 writeWavHeader(file, config, result.framesRendered);
279 file.flush();
280 if (!file || !Aestra::fsyncPath(config.outputPath)) {
281 result.errorMessage = "Failed to sync exported WAV to disk";
282 result.success = false;
283 file.close();
284 std::remove(config.outputPath.c_str());
285 updateProgress(1.0f);
286 return result;
287 }
288 file.close();
289#ifndef _WIN32
290 if (!Aestra::fsyncParentDirectory(config.outputPath)) {
291 Log::warning("[Export] Failed to sync exported WAV directory; keeping completed export");
292 }
293#endif
294
295 result.success = true;
296 result.durationSeconds = static_cast<double>(result.framesRendered) / config.sampleRate;
297 float peak = m_peakLevel.load(std::memory_order_relaxed);
298 result.peakDb = peak > 0.0f ? 20.0 * std::log10(peak) : -96.0;
299
300 // Phase 2: capture true-peak metrics measured during the export
301 // (engine's meter ran throughout processBlock).
302 result.maxTruePeakdBTP = m_engine.getMaxTruePeakdBTP();
303
304 if (config.validateTruePeak && result.maxTruePeakdBTP > config.truePeakCeilingdBTP) {
305 result.truePeakCeilingExceeded = true;
306 const std::string msg =
307 "[Export] True peak " + std::to_string(result.maxTruePeakdBTP) +
308 " dBTP exceeds ceiling " + std::to_string(config.truePeakCeilingdBTP) + " dBTP";
309 if (config.failOnTruePeakExceeded) {
310 Log::error(msg + " \u2014 export aborted");
311 result.errorMessage = msg;
312 result.success = false;
313 std::remove(config.outputPath.c_str());
314 } else {
315 Log::warning(msg);
316 }
317 }
318
319 if (result.success) {
320 Log::info("[Export] Render complete: " + std::to_string(result.framesRendered) +
321 " frames, peak: " + std::to_string(result.peakDb) + " dB, true peak: " +
322 std::to_string(result.maxTruePeakdBTP) + " dBTP");
323 }
324 } else {
325 file.close();
326 std::remove(config.outputPath.c_str());
327 }
328
329 updateProgress(1.0f);
330 return result;
331}

333double AudioExporter::computeRenderDurationBeats(const Config& config, double& outStartBeat) {
334 auto& playlist = m_trackManager.getPlaylistModel();
335 double totalBeats = playlist.getTotalDurationBeats();
336
337 switch (config.scope) {
338 case RenderScope::FullSong:
339 if (config.endBeat > config.startBeat) {
340 outStartBeat = std::max(0.0, config.startBeat);
341 const double boundedEndBeat =
342 (totalBeats > 0.0) ? std::min(config.endBeat, totalBeats)
343 : config.endBeat;
344 const double duration = boundedEndBeat - outStartBeat;
345 return duration > 0.0 ? duration : 0.0;
346 }
347 outStartBeat = 0.0;
348 return totalBeats > 0.0 ? totalBeats : 0.0;
349
350 case RenderScope::LoopRegion: {
351 double loopStart = m_engine.getLoopStartBeat();
352 double loopEnd = m_engine.getLoopEndBeat();
353 if (loopEnd > loopStart) {
354 outStartBeat = loopStart;
355 return loopEnd - loopStart;
356 }
357 return 0.0;
358 }
359
360 case RenderScope::Selection: {
361 double bpm = m_engine.getBPM();
362 if (bpm <= 0.0) bpm = 120.0;
363 double secondsPerBeat = 60.0 / bpm;
364 outStartBeat = config.startTimeSeconds / secondsPerBeat;
365 double endBeat = config.endTimeSeconds / secondsPerBeat;
366 double dur = endBeat - outStartBeat;
367 return dur > 0.0 ? dur : 0.0;
368 }
369 }
370 return 0.0;
371}

373bool AudioExporter::writeWavHeader(std::ofstream& file, const Config& config, uint64_t totalFrames) {
374 if (!file) return false;
375
376 int bitsPerSample = static_cast<int>(config.bitDepth);
377 int bytesPerSample = bitsPerSample / 8;
378
379 uint64_t byteRate64 = static_cast<uint64_t>(config.sampleRate) * config.numChannels * bytesPerSample;
380 uint64_t blockAlign64 = static_cast<uint64_t>(config.numChannels) * bytesPerSample;
381 if (config.numChannels == 0 || blockAlign64 == 0 ||
382 blockAlign64 > std::numeric_limits<uint16_t>::max() ||
383 byteRate64 > std::numeric_limits<uint32_t>::max() ||
384 totalFrames > std::numeric_limits<uint64_t>::max() / blockAlign64) {
385 return false;
386 }
387
388 uint64_t dataSize64 = totalFrames * blockAlign64;
389 if (dataSize64 > std::numeric_limits<uint32_t>::max() ||
390 36ull + dataSize64 > std::numeric_limits<uint32_t>::max()) {
391 return false;
392 }
393
394 uint32_t byteRate = static_cast<uint32_t>(byteRate64);
395 uint16_t blockAlign = static_cast<uint16_t>(blockAlign64);
396 uint32_t dataSize = static_cast<uint32_t>(dataSize64);
397 uint32_t fileSize = 36 + dataSize;
398
399 uint16_t audioFormat = (config.bitDepth == BitDepth::Float_32) ? 3 : 1;
400
401 file.write("RIFF", 4);
402 file.write(reinterpret_cast<const char*>(&fileSize), 4);
403 file.write("WAVE", 4);
404
405 file.write("fmt ", 4);
406 uint32_t fmtSize = 16;
407 file.write(reinterpret_cast<const char*>(&fmtSize), 4);
408 file.write(reinterpret_cast<const char*>(&audioFormat), 2);
409
410 uint16_t numChannels = static_cast<uint16_t>(config.numChannels);
411 file.write(reinterpret_cast<const char*>(&numChannels), 2);
412 file.write(reinterpret_cast<const char*>(&config.sampleRate), 4);
413 file.write(reinterpret_cast<const char*>(&byteRate), 4);
414 file.write(reinterpret_cast<const char*>(&blockAlign), 2);
415 file.write(reinterpret_cast<const char*>(&bitsPerSample), 2);
416
417 file.write("data", 4);
418 file.write(reinterpret_cast<const char*>(&dataSize), 4);
419
420 return file.good();
421}

423template<typename SampleType>
424bool AudioExporter::writeSamples(std::ofstream& file, const float* buffer,
425 size_t frames, uint32_t channels) {
426 if constexpr (std::is_same_v<SampleType, float>) {
427 file.write(reinterpret_cast<const char*>(buffer),
428 frames * channels * sizeof(float));
429 } else if constexpr (std::is_same_v<SampleType, int16_t>) {
430 std::vector<int16_t> converted(frames * channels);
431 for (size_t i = 0; i < frames * channels; ++i) {
432 converted[i] = ExportQuantization::quantizePcm16Dithered(buffer[i], m_exportDither);
433 }
434 file.write(reinterpret_cast<const char*>(converted.data()),
435 converted.size() * sizeof(int16_t));
436 } else if constexpr (std::is_same_v<SampleType, int32_t>) {
437 std::vector<uint8_t> converted(frames * channels * 3);
438 for (size_t i = 0; i < frames * channels; ++i) {
439 const int32_t packed24 = ExportQuantization::quantizePcm24Dithered(buffer[i], m_exportDither);
440 ExportQuantization::storePcm24LittleEndian(packed24, converted.data() + i * 3);
441 }
442 file.write(reinterpret_cast<const char*>(converted.data()), converted.size());
443 }
444
445 return file.good();
446}

448void AudioExporter::updateProgress(float percent) {
449 auto now = std::chrono::steady_clock::now();
450 auto elapsed = now - m_lastProgressTime;
451 if (elapsed >= m_progressInterval) {
452 m_lastProgressTime = now;
453 if (m_progressCallback) {
454 m_progressCallback(percent);
455 }
456 }

466float AudioExporter::calculatePeakDb(const float* buffer, size_t frames, uint32_t channels) {
467 float peak = 0.0f;
468 for (size_t i = 0; i < frames * channels; ++i) {
469 peak = std::max(peak, std::abs(buffer[i]));
470 }
471 return peak;
472}
