25class LoudnessWindow {
26public:
27 static constexpr float kSilence = -144.0f;
28 static constexpr size_t kSubBlocksMomentary = 4;
29 static constexpr size_t kSubBlocksShortTerm = 30;
30
31 void prepare(double sampleRate) noexcept {
32 m_subLength.store(static_cast<uint32_t>(std::max(1.0, std::round(sampleRate * 0.1))), std::memory_order_relaxed);
33 requestReset();
34 }
35
37 void requestReset() noexcept {
38 m_resetRequested.store(true, std::memory_order_release);
39 m_momentary.store(kSilence, std::memory_order_relaxed);
40 m_shortTerm.store(kSilence, std::memory_order_relaxed);
41 }
42
44 inline void add(double kWeightedEnergy) noexcept {
45 m_sum += kWeightedEnergy;
46 if (++m_count >= m_subLength.load(std::memory_order_relaxed))
47 closeSubBlock();
48 }
49
50 float momentaryLufs() const noexcept { return m_momentary.load(std::memory_order_relaxed); }
51 float shortTermLufs() const noexcept { return m_shortTerm.load(std::memory_order_relaxed); }
52
53 static float energyToLufs(double meanEnergy) noexcept {
54 return meanEnergy > 1e-14 ? static_cast<float>(-0.691 + 10.0 * std::log10(meanEnergy)) : kSilence;
55 }
56
57private:
58 void closeSubBlock() noexcept {
59 if (m_resetRequested.exchange(false, std::memory_order_acquire)) {
60 m_ring.fill(0.0);
61 m_write = 0;
62 m_filled = 0;
63 }
64 m_ring[m_write] = m_sum / static_cast<double>(m_count);
65 m_write = (m_write + 1) % kSubBlocksShortTerm;
66 m_filled = std::min(m_filled + 1, kSubBlocksShortTerm);
67 m_sum = 0.0;
68 m_count = 0;
69
70 if (m_filled < kSubBlocksMomentary) {
71 m_momentary.store(kSilence, std::memory_order_relaxed);
72 m_shortTerm.store(kSilence, std::memory_order_relaxed);
73 return;
74 }
75 m_momentary.store(energyToLufs(meanOfLast(kSubBlocksMomentary)), std::memory_order_relaxed);
76 m_shortTerm.store(energyToLufs(meanOfLast(m_filled)), std::memory_order_relaxed);
77 }
78
79 double meanOfLast(size_t n) const noexcept {
80 double total = 0.0;
81 for (size_t i = 0; i < n; ++i)
82 total += m_ring[(m_write + kSubBlocksShortTerm - 1 - i) % kSubBlocksShortTerm];
83 return total / static_cast<double>(n);
84 }
85
86 // Audio thread only.
87 std::array<double, kSubBlocksShortTerm> m_ring{};
88 size_t m_write{0};
89 size_t m_filled{0};
90 double m_sum{0.0};
91 uint32_t m_count{0};
92
93 std::atomic<uint32_t> m_subLength{4800};
94 std::atomic<bool> m_resetRequested{false};
95 std::atomic<float> m_momentary{kSilence};
96 std::atomic<float> m_shortTerm{kSilence};
97};
