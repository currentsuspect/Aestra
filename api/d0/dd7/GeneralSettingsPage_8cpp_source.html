13// The folder the library opens at (Documents/Aestra, as FileBrowser makes it), with the home folder shown as ~.
14std::string aestraFolderForDisplay() {
15#if defined(_WIN32)
16 const char* home = std::getenv("USERPROFILE");
17#else
18 const char* home = std::getenv("HOME");
19#endif
20 if (!home || !*home) return "Documents/Aestra";
21 const std::string full = (std::filesystem::path(home) / "Documents" / "Aestra").string();
22#if defined(_WIN32)
23 return full;
24#else
25 return "~/Documents/Aestra";
26#endif
27}

34void GeneralSettingsPage::createUI() {
35 auto& theme = AestraUI::NUIThemeManager::getInstance();
36 m_projectsPathLabel = std::make_shared<AestraUI::NUILabel>();
37 m_projectsPathLabel->setText("Aestra folder");
38 m_projectsPathLabel->setFontSize(theme.getFontSize("m"));
39 addChild(m_projectsPathLabel);
40
41 m_projectsPathInput = std::make_shared<AestraUI::NUITextInput>();
42 // Shows where the library lives. Choosing another folder is not built yet, so the field is read-only and
43 // the Browse button stays hidden rather than doing nothing.
44 m_projectsPathInput->setText(aestraFolderForDisplay());
45 m_projectsPathInput->setReadOnly(true);
46 addChild(m_projectsPathInput);
47
48 m_browseButton = std::make_shared<AestraUI::NUIButton>();
49 m_browseButton->setText("Browse");
50 m_browseButton->setStyle(AestraUI::NUIButton::Style::Secondary);
51 m_browseButton->setVisible(false);
52 addChild(m_browseButton);
53
54 m_autoSaveLabel = std::make_shared<AestraUI::NUILabel>();
55 m_autoSaveLabel->setText("Auto-Save:");
56 m_autoSaveLabel->setFontSize(theme.getFontSize("m"));
57 addChild(m_autoSaveLabel);
58
59 // A switch, set from the stored preference (it used to start "on" whatever was saved).
60 m_autoSaveToggle = std::make_shared<AestraUI::NUIToggle>();
61 m_autoSaveToggle->setAnimated(false);
62 m_autoSaveToggle->setOn(Preferences::instance().autoSaveEnabled);
63 m_autoSaveToggle->setOnToggle([this](bool enabled) {
64 m_dirty = true;
65 syncAutoSaveLabel();
66 if (m_onAutoSaveToggled) m_onAutoSaveToggled(enabled);
67 });
68 m_autoSaveDetail = std::make_shared<AestraUI::NUILabel>();
69 m_autoSaveDetail->setFontSize(theme.getFontSize("m"));
70 m_autoSaveDetail->setTextColor(theme.getColor("textSecondary"));
71 addChild(m_autoSaveDetail);
72 syncAutoSaveLabel();
73 addChild(m_autoSaveToggle);
74}

76void GeneralSettingsPage::syncAutoSaveLabel() {
77 if (!m_autoSaveToggle || !m_autoSaveDetail) return;
78 if (!m_autoSaveToggle->isOn()) {
79 m_autoSaveDetail->setText("Off");
80 m_form.setHint(m_autoSaveToggle.get(), "Off");
81 return;
82 }
83 const int seconds = std::clamp(Preferences::instance().autoSaveIntervalSeconds, 10, 3600);
84 m_autoSaveDetail->setText(seconds % 60 == 0 ? "Every " + std::to_string(seconds / 60) + (seconds == 60 ? " minute" : " minutes")
85 : "Every " + std::to_string(seconds) + " seconds");
86 m_form.setHint(m_autoSaveToggle.get(), m_autoSaveDetail->getText());
87}

109void GeneralSettingsPage::layoutComponents() {
110 m_projectsPathLabel->setVisible(false);
111 m_autoSaveLabel->setVisible(false);
112 m_autoSaveDetail->setVisible(false); // the Auto-save row's hint
113 constexpr float kPathWidth = 300.0f;
114
115 m_form.clear();
116 m_form.section("PROJECTS");
117 m_form.row("Aestra folder", m_projectsPathInput, "", kPathWidth);
118 m_form.row("Auto-save", m_autoSaveToggle, m_autoSaveDetail->getText(), 40.0f);
119 m_form.layout(getBounds());
120
121 // Browse sits beside the folder path, in the same row.
122 const auto& path = m_projectsPathInput->getBounds();
123 m_browseButton->setBounds(AestraUI::NUIRect(path.right() + 8.0f, path.y, 84.0f, path.height));
124 m_browseButton->setVisible(m_projectsPathInput->isVisible());
125}
