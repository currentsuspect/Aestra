35inline std::string keepBeforeSave(const std::string& projectPath, int currentVersion) {
36 namespace fs = std::filesystem;
37 std::ifstream in(projectPath, std::ios::binary);
38 if (!in) return {};
39 std::ostringstream text;
40 text << in.rdbuf();
41 JSON root;
42 try {
43 root = JSON::parse(text.str());
44 } catch (...) {
45 return {};
46 }
47 if (!root.isObject() || !root.has("version") || !root["version"].isNumber() ||
48 !std::isfinite(root["version"].asNumber())) {
49 return {};
50 }
51 const int version = static_cast<int>(root["version"].asNumber());
52 if (version <= 0 || version >= currentVersion) return {};
53
54 const std::string copyPath = pathFor(projectPath, version);
55 std::error_code ec;
56 if (fs::exists(copyPath, ec)) return {};
57 fs::copy_file(projectPath, copyPath, fs::copy_options::none, ec);
58 if (ec) {
59 Log::warning("Pre-upgrade copy failed (non-fatal): " + ec.message());
60 return {};
61 }
62 Log::info("[ProjectSave] Kept the schema v" + std::to_string(version) + " file as " + copyPath +
63 " before saving as v" + std::to_string(currentVersion));
64 return copyPath;
65}
