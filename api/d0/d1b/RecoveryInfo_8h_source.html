33inline std::string gapWords(uint64_t fromMs, uint64_t toMs) {
34 const uint64_t ms = toMs > fromMs ? toMs - fromMs : 0;
35 const uint64_t minutes = (ms + 30000) / 60000;
36 if (minutes == 0) return "moments";
37 if (minutes < 120) return std::to_string(minutes) + (minutes == 1 ? " minute" : " minutes");
38 const uint64_t hours = (minutes + 30) / 60;
39 return std::to_string(hours) + (hours == 1 ? " hour" : " hours");
40}

42inline Info build(const TakeManager::Manifest& manifest, std::vector<uint64_t> autosaveTimesMs) {
43 Info info;
44 std::sort(autosaveTimesMs.begin(), autosaveTimesMs.end());
45 const auto* open = manifest.ok ? manifest.activeTake() : nullptr;
46 if (open) {
47 info.hasVersion = true;
48 info.lastSavedName = open->name.empty() ? open->id : open->name;
49 info.lastSavedMs = open->updatedAtMs ? open->updatedAtMs : open->createdAtMs;
50 const auto lanes = VersionLanes::names(manifest.takes);
51 const std::string branch = open->branch.empty() ? "Main" : open->branch;
52 info.lane = static_cast<int>(std::find(lanes.begin(), lanes.end(), branch) - lanes.begin());
53 info.branch = VersionLanes::label(branch);
54 }
55 for (const uint64_t t : autosaveTimesMs)
56 if (!info.hasVersion || t > info.lastSavedMs) info.autosaveMs.push_back(t);
57 if (info.autosaveMs.size() > kMaxDots) info.autosaveMs.erase(info.autosaveMs.begin(), info.autosaveMs.end() - kMaxDots);
58
59 const uint64_t newest = autosaveTimesMs.empty() ? 0 : autosaveTimesMs.back();
60 if (!info.hasVersion) info.detail = "No version of this project has been saved yet.";
61 else if (newest > info.lastSavedMs) info.detail = "The newest autosave is " + gapWords(info.lastSavedMs, newest) + " after your last saved version.";
62 else info.detail = "Nothing newer than your last saved version was autosaved.";
63 return info;
64}
