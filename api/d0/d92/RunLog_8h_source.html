4// #889: where Aestra writes its log. One file per run, in the platform's per-user state folder, so a crash report can
5// carry the run that crashed and nobody has to know where the app was launched from:
6//
7// Linux $XDG_STATE_HOME/aestra/, else ~/.local/state/aestra/
8// Windows %LOCALAPPDATA%\Aestra\logs\
9// macOS ~/Library/Logs/Aestra/
10//
11// When that folder cannot be created the working directory is used, and the caller says so. Old runs are pruned.

31template <class GetEnv> Location resolveDirectory(GetEnv&& getEnv) {
32 namespace fs = std::filesystem;
33 const auto var = [&](const char* name) -> std::string {
34 const char* v = getEnv(name);
35 return v && *v ? std::string(v) : std::string();
36 };
37 fs::path dir;
38#if defined(_WIN32)
39 if (const auto base = var("LOCALAPPDATA"); !base.empty()) dir = fs::path(base) / "Aestra" / "logs";
40#elif defined(__APPLE__)
41 if (const auto home = var("HOME"); !home.empty()) dir = fs::path(home) / "Library" / "Logs" / "Aestra";
42#else
43 if (const auto state = var("XDG_STATE_HOME"); !state.empty()) dir = fs::path(state) / "aestra";
44 else if (const auto home = var("HOME"); !home.empty()) dir = fs::path(home) / ".local" / "state" / "aestra";
45#endif
46 std::error_code ec;
47 if (!dir.empty()) {
48 fs::create_directories(dir, ec);
49 if (!ec && fs::is_directory(dir, ec)) return {dir, false};
50 }
51 return {fs::current_path(ec), true};
52}

55inline std::string fileNameFor(const std::tm& start) {
56 char buf[64];
57 std::snprintf(buf, sizeof(buf), "aestra-%04d%02d%02d-%02d%02d%02d.log", start.tm_year + 1900, start.tm_mon + 1,
58 start.tm_mday, start.tm_hour, start.tm_min, start.tm_sec);
59 return buf;
60}

63inline void pruneOldRuns(const std::filesystem::path& dir, size_t keep) {
64 namespace fs = std::filesystem;
65 std::error_code ec;
66 std::vector<fs::path> runs;
67 for (const auto& entry : fs::directory_iterator(dir, ec)) {
68 const std::string name = entry.path().filename().string();
69 if (name.rfind("aestra-", 0) == 0 && entry.path().extension() == ".log") runs.push_back(entry.path());
70 }
71 if (runs.size() <= keep) return;
72 std::sort(runs.begin(), runs.end());
73 for (size_t i = 0; i + keep < runs.size(); ++i) fs::remove(runs[i], ec);
74}

77inline const std::filesystem::path& path(bool* fellBack = nullptr) {
78 static bool fallback = false;
79 static const std::filesystem::path resolved = [] {
80 const Location where = resolveDirectory([](const char* n) { return std::getenv(n); });
81 fallback = where.fellBack;
82 const std::time_t now = std::time(nullptr);
83 std::tm local{};
84#if defined(_WIN32)
85 localtime_s(&local, &now);
86#else
87 localtime_r(&now, &local);
88#endif
89 pruneOldRuns(where.directory, 20);
90 return where.directory / fileNameFor(local);
91 }();
92 if (fellBack) *fellBack = fallback;
93 return resolved;
94}
