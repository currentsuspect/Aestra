11// Synchronization policy for this file (#391):
12// m_mutex guards all non-atomic manager state. Public methods lock and delegate
13// to *Locked() helpers (which assume the lock is held and compose without
14// re-locking). The health-monitor thread is only started/stopped (joined) at the
15// public boundary, never under the lock. The driver-mode-change callback is
16// captured under the lock and fired only after it is released. See the header
17// for the full policy and the invariants it maintains.

36bool AudioDeviceManager::initialize(bool registerPlatformDrivers) {
37 {
38 std::lock_guard<std::mutex> lock(m_mutex);
39 if (m_initialized) {
40 return true;
41 }
42 }
43
44 AESTRA_LOG_INFO("AestraAudio multi-tier driver system initializing...");
45
46 try {
47 // Register platform-specific drivers (Dependency Injection Point).
48 // NOT under m_mutex: RegisterPlatformDrivers calls back into addDriver(),
49 // which locks per push. Startup is single-threaded, so no other thread is
50 // racing this; we then lock briefly to publish the initialized state.
51 // Tests pass registerPlatformDrivers=false to run against only the drivers
52 // they injected via addDriver() (no real hardware).
53 if (registerPlatformDrivers) {
54 AESTRA_LOG_DEBUG("RegisterPlatformDrivers about to be called");
55 RegisterPlatformDrivers(*this);
56 }
57
58 std::lock_guard<std::mutex> lock(m_mutex);
59 AESTRA_LOG_DEBUG("RegisterPlatformDrivers returned, drivers count: " + std::to_string(m_drivers.size()));
60
61 if (m_drivers.empty()) {
62 AESTRA_LOG_WARNING("No audio drivers available");
63 } else {
64 for (const auto& driver : m_drivers) {
65 AESTRA_LOG_DEBUG(std::string(driver->isAvailable() ? "✓ " : "✗ ") + driver->getDisplayName() +
66 (driver->isAvailable() ? " available" : " unavailable"));
67 }
68 }
69
70 m_initialized = true;
71 AESTRA_LOG_INFO("Audio system initialized and ready");
72 return true;
73
74 } catch (const std::exception& e) {
75 AESTRA_LOG_ERROR(std::string("AudioDeviceManager::initialize exception: ") + e.what());
76 return false;
77 }
78}

80void AudioDeviceManager::shutdown() {
81 // Join the health monitor before taking the lock: the monitor thread acquires
82 // m_mutex in checkDriverHealth, so joining it under the lock would deadlock
83 // (#391 constraint 5). After this returns the monitor is not running.
84 stopHealthMonitor();
85
86 std::lock_guard<std::mutex> lock(m_mutex);
87 if (!m_initialized) {
88 return;
89 }
90 closeStreamLocked();
91 m_drivers.clear();
92 m_activeDriver = nullptr;
93 m_initialized = false;
94 AESTRA_LOG_DEBUG("Audio system shutdown complete");
95}

159bool AudioDeviceManager::tryDriver(IAudioDriver* driver, const AudioStreamConfig& config, AudioCallback callback,
160 void* userData) {
161 if (!driver || !driver->isAvailable()) {
162 return false;
163 }
164
165 AESTRA_LOG_DEBUG("Trying " + driver->getDisplayName() + "...");
166
167 if (driver->openStream(config, callback, userData)) {
168 AESTRA_LOG_DEBUG(driver->getDisplayName() + " opened successfully");
169 double lat = driver->getStreamLatency();
170 AESTRA_LOG_DEBUG(" Latency: " + std::to_string(lat * 1000.0) + "ms");
171 m_activeDriver = driver;
172 return true;
173 }
174
175 Aestra::Log::warning(driver->getDisplayName() + " failed: " + driver->getErrorMessage());
176 return false;
177}

179bool AudioDeviceManager::openStreamLocked(const AudioStreamConfig& config, AudioCallback callback, void* userData) {
180 AESTRA_LOG_DEBUG("[AudioDeviceManager] openStream called. Rate: " + std::to_string(config.sampleRate) +
181 "Hz, Output Device: " + std::to_string(config.deviceId) +
182 ", Input Device: " + std::to_string(config.inputDeviceId));
183
184 if (!m_initialized) {
185 Aestra::Log::error("[AudioDeviceManager] openStream failed: Not initialized");
186 return false;
187 }
188
189 if (!validateStreamConfigLocked(config)) {
190 Aestra::Log::error("[AudioDeviceManager] openStream failed: Invalid stream configuration");
191 return false;
192 }
193
194 m_currentConfig = config;
195 m_currentCallback = callback;
196 m_currentUserData = userData;
197
198 std::string firstFailureReason;
199 bool firstAttempt = true;
200
201 for (auto& driver : m_drivers) {
202 if (tryDriver(driver.get(), config, callback, userData)) {
203 if (m_activeDriver && m_activeDriver->getDriverType() != m_preferredDriverType) {
204 if (m_fallbackReason.empty()) {
205 m_fallbackReason = "Preferred driver (" + std::string(DriverTypeToString(m_preferredDriverType)) +
206 ") was unavailable, fell back to " + m_activeDriver->getDisplayName();
207 }
208 }
209 return true;
210 }
211 if (firstAttempt) {
212 firstFailureReason = driver->getErrorMessage();
213 firstAttempt = false;
214 }
215 }
216
217 if (!firstAttempt && !firstFailureReason.empty()) {
218 m_fallbackReason = "All drivers failed. Preferred (" + std::string(DriverTypeToString(m_preferredDriverType)) +
219 ") error: " + firstFailureReason;
220 }
221
222 Aestra::Log::error("[AudioDeviceManager] All drivers failed to open stream!");
223 return false;
224}

335// ---------------------------------------------------------------------------
336// Reconfiguration transactions
337//
338// Each of these is a single locked transaction: capture running state, stop and
339// close, apply the new config, and — on failure — roll back to the previous
340// working configuration before returning. The health monitor is joined up front
341// (outside the lock) and restarted afterward if a stream ends up running, so a
342// transition is never observable half-complete and never deadlocks on the join.
343// ---------------------------------------------------------------------------
344
345bool AudioDeviceManager::switchDevice(uint32_t deviceId) {
346 stopHealthMonitor();
347 bool result;
348 {
349 std::lock_guard<std::mutex> lock(m_mutex);
350 if (!m_initialized) {
351 result = false;
352 } else if (m_currentConfig.deviceId == deviceId) {
353 result = true;
354 } else if (!validateDeviceConfigLocked(deviceId, m_currentConfig.sampleRate)) {
355 result = false;
356 } else {
357 const uint32_t previousDevice = m_currentConfig.deviceId;
358 m_wasRunning = isStreamRunningLocked();
359 if (m_wasRunning) {
360 stopStreamLocked();
361 }
362 closeStreamLocked();
363
364 m_currentConfig.deviceId = deviceId;
365 if (!openStreamLocked(m_currentConfig, m_currentCallback, m_currentUserData)) {
366 // Roll back to the previous device so we do not leave a dead stream.
367 m_currentConfig.deviceId = previousDevice;
368 if (openStreamLocked(m_currentConfig, m_currentCallback, m_currentUserData) && m_wasRunning) {
369 startStreamLocked();
370 }
371 result = false;
372 } else {
373 result = (!m_wasRunning) || startStreamLocked();
374 }
375 }
376 }
377 if (isStreamRunning()) {
378 startHealthMonitorIfStopped();
379 }
380 return result;
381}

383bool AudioDeviceManager::switchInputDevice(uint32_t deviceId) {
384 stopHealthMonitor();
385 bool result;
386 {
387 std::lock_guard<std::mutex> lock(m_mutex);
388 if (!m_initialized) {
389 result = false;
390 } else if (m_currentConfig.inputDeviceId == deviceId) {
391 result = true;
392 } else {
393 const auto devices = getDevicesLocked();
394 const auto it = std::find_if(devices.begin(), devices.end(), [deviceId](const AudioDeviceInfo& device) {
395 return device.id == deviceId;
396 });
397 if (it == devices.end() || it->maxInputChannels == 0) {
398 result = false;
399 } else {
400 const uint32_t previousInput = m_currentConfig.inputDeviceId;
401 const uint32_t previousInChans = m_currentConfig.numInputChannels;
402 m_wasRunning = isStreamRunningLocked();
403 if (m_wasRunning) {
404 stopStreamLocked();
405 }
406 closeStreamLocked();
407
408 m_currentConfig.inputDeviceId = deviceId;
409 if (m_currentConfig.numInputChannels > it->maxInputChannels) {
410 m_currentConfig.numInputChannels = it->maxInputChannels;
411 }
412
413 if (!openStreamLocked(m_currentConfig, m_currentCallback, m_currentUserData)) {
414 m_currentConfig.inputDeviceId = previousInput;
415 m_currentConfig.numInputChannels = previousInChans;
416 if (openStreamLocked(m_currentConfig, m_currentCallback, m_currentUserData) && m_wasRunning) {
417 startStreamLocked();
418 }
419 result = false;
420 } else {
421 result = (!m_wasRunning) || startStreamLocked();
422 }
423 }
424 }
425 }
426 if (isStreamRunning()) {
427 startHealthMonitorIfStopped();
428 }
429 return result;
430}

432bool AudioDeviceManager::setSampleRate(uint32_t sampleRate) {
433 AESTRA_LOG_DEBUG("Request to set sample rate to: " + std::to_string(sampleRate));
434 stopHealthMonitor();
435 bool result;
436 {
437 std::lock_guard<std::mutex> lock(m_mutex);
438 if (!m_initialized) {
439 AESTRA_LOG_ERROR("setSampleRate failed: Not initialized");
440 result = false;
441 } else if (m_currentConfig.sampleRate == sampleRate) {
442 AESTRA_LOG_DEBUG("Sample rate unchanged (" + std::to_string(sampleRate) + " Hz), skipping reopen");
443 result = true;
444 } else if (sampleRate != 44100 && sampleRate != 48000 && sampleRate != 88200 && sampleRate != 96000 &&
445 sampleRate != 176400 && sampleRate != 192000) {
446 Aestra::Log::error("[AudioDeviceManager] setSampleRate failed: Invalid rate " + std::to_string(sampleRate));
447 result = false;
448 } else if (!validateDeviceConfigLocked(m_currentConfig.deviceId, sampleRate)) {
449 result = false;
450 } else {
451 const uint32_t previousSampleRate = m_currentConfig.sampleRate;
452 m_wasRunning = isStreamRunningLocked();
453 if (m_wasRunning) {
454 stopStreamLocked();
455 }
456 closeStreamLocked();
457
458 // Give the device time to fully release (avoids AUDCLNT_E_DEVICE_IN_USE).
459 std::this_thread::sleep_for(std::chrono::milliseconds(100));
460
461 m_currentConfig.sampleRate = sampleRate;
462 if (!openStreamLocked(m_currentConfig, m_currentCallback, m_currentUserData)) {
463 Aestra::Log::error("[AudioDeviceManager] Failed to reopen stream with sample rate " +
464 std::to_string(sampleRate) + ", rolling back to " +
465 std::to_string(previousSampleRate));
466 m_currentConfig.sampleRate = previousSampleRate;
467 if (!openStreamLocked(m_currentConfig, m_currentCallback, m_currentUserData)) {
468 Aestra::Log::error("[AudioDeviceManager] CRITICAL: Failed to restore previous sample rate!");
469 } else if (m_wasRunning) {
470 startStreamLocked();
471 }
472 result = false;
473 } else {
474 result = (!m_wasRunning) || startStreamLocked();
475 }
476 }
477 }
478 if (isStreamRunning()) {
479 startHealthMonitorIfStopped();
480 }
481 return result;
482}

531bool AudioDeviceManager::validateDeviceConfigLocked(uint32_t deviceId, uint32_t sampleRate) const {
532 if (!m_initialized) {
533 Aestra::Log::error("[AudioDeviceManager] validateDeviceConfig failed: Not initialized");
534 return false;
535 }
536
537 const auto devices = getDevicesLocked();
538 for (const auto& device : devices) {
539 if (device.id == deviceId) {
540 if (device.maxOutputChannels == 0) {
541 Aestra::Log::error("[AudioDeviceManager] validateDeviceConfig failed: Device has 0 output channels");
542 return false;
543 }
544 for (uint32_t supportedRate : device.supportedSampleRates) {
545 if (supportedRate == sampleRate) {
546 return true;
547 }
548 }
549 // Some drivers report empty/incorrect supported lists during enumeration
550 // but work when actually opened. Warn but allow; openStream is final judge.
551 Aestra::Log::warning("[AudioDeviceManager] Sample rate " + std::to_string(sampleRate) +
552 " not in supported list (Driver Issue?), attempting anyway...");
553 return true;
554 }
555 }
556
557 Aestra::Log::error("[AudioDeviceManager] validateDeviceConfig failed: Device ID " + std::to_string(deviceId) +
558 " not found");
559 return false;
560}

567bool AudioDeviceManager::validateStreamConfigLocked(const AudioStreamConfig& config) const {
568 if (!validateDeviceConfigLocked(config.deviceId, config.sampleRate)) {
569 return false;
570 }
571
572 if (config.numInputChannels == 0) {
573 return true;
574 }
575
576 const auto devices = getDevicesLocked();
577 const uint32_t inputDeviceId = (config.inputDeviceId != 0) ? config.inputDeviceId : config.deviceId;
578 for (const auto& device : devices) {
579 if (device.id != inputDeviceId) {
580 continue;
581 }
582 if (device.maxInputChannels == 0) {
583 Aestra::Log::error("[AudioDeviceManager] validateStreamConfig failed: Input device has 0 input channels");
584 return false;
585 }
586 if (config.numInputChannels > device.maxInputChannels) {
587 Aestra::Log::error(
588 "[AudioDeviceManager] validateStreamConfig failed: Requested input channels exceed device capacity");
589 return false;
590 }
591 return true;
592 }
593
594 Aestra::Log::error("[AudioDeviceManager] validateStreamConfig failed: Input device ID " +
595 std::to_string(inputDeviceId) + " not found");
596 return false;
597}

613bool AudioDeviceManager::setPreferredDriverType(AudioDriverType type) {
614 stopHealthMonitor();
615 bool result;
616 {
617 std::lock_guard<std::mutex> lock(m_mutex);
618 if (!m_initialized) {
619 AESTRA_LOG_ERROR("Cannot set driver type: not initialized");
620 result = false;
621 } else if (m_preferredDriverType == type) {
622 result = true; // no-op; avoids redundant reopen during UI population
623 } else {
624 AESTRA_LOG_DEBUG("Changing driver type");
625 m_preferredDriverType = type;
626
627 if (m_activeDriver && m_currentCallback) {
628 const bool wasRunning = isStreamRunningLocked();
629 auto savedCallback = m_currentCallback;
630 auto savedUserData = m_currentUserData;
631 auto savedConfig = m_currentConfig;
632
633 if (wasRunning) {
634 stopStreamLocked();
635 }
636 closeStreamLocked();
637
638 if (!openStreamLocked(savedConfig, savedCallback, savedUserData)) {
639 AESTRA_LOG_ERROR("Failed to reopen stream with any driver");
640 result = false;
641 } else if (wasRunning && !startStreamLocked()) {
642 AESTRA_LOG_ERROR("Failed to restart stream");
643 result = false;
644 } else {
645 result = true;
646 }
647 } else {
648 result = true;
649 }
650 }
651 }
652 if (isStreamRunning()) {
653 startHealthMonitorIfStopped();
654 }
655 return result;
656}

707void AudioDeviceManager::checkAndAutoScaleBuffer() {
708 // Reconfigures the stream when underruns spike, so — like the other
709 // transactions — it must not run while the health monitor might be joining or
710 // reconfiguring. Callers must invoke this from the app/main loop, never from
711 // the health-monitor thread. Join the monitor up front, work under the lock,
712 // then fire any captured callback and restart the monitor outside the lock.
713 stopHealthMonitor();
714
715 PendingModeChange pending;
716 {
717 std::lock_guard<std::mutex> lock(m_mutex);
718 checkDriverHealthLocked(pending);
719
720 if (m_autoBufferScalingEnabled && m_activeDriver && isStreamRunningLocked()) {
721 const auto now = std::chrono::steady_clock::now();
722 const auto elapsed = std::chrono::duration_cast<std::chrono::seconds>(now - m_lastUnderrunCheck);
723 if (elapsed.count() >= 60) {
724 const DriverStatistics stats = m_activeDriver->getStatistics();
725 const uint64_t newUnderruns = stats.underrunCount - m_lastUnderrunCount;
726
727 if (newUnderruns >= m_underrunThreshold) {
728 const uint32_t currentBuffer = m_currentConfig.bufferSize;
729 uint32_t newBuffer = currentBuffer;
730 if (currentBuffer < 128) {
731 newBuffer = 128;
732 } else if (currentBuffer < 256) {
733 newBuffer = 256;
734 } else if (currentBuffer < 512) {
735 newBuffer = 512;
736 } else if (currentBuffer < 1024) {
737 newBuffer = 1024;
738 }
739
740 if (newBuffer != currentBuffer) {
741 const bool wasRunning = isStreamRunningLocked();
742 if (wasRunning) {
743 stopStreamLocked();
744 }
745 closeStreamLocked();
746 // Roll back like setBufferSize: on reopen failure, restore the
747 // previous size so m_currentConfig never advertises a buffer
748 // with no live stream behind it (CR review, #391 coherence).
749 const uint32_t previousBuffer = m_currentConfig.bufferSize;
750 m_currentConfig.bufferSize = newBuffer;
751 if (openStreamLocked(m_currentConfig, m_currentCallback, m_currentUserData)) {
752 if (wasRunning) {
753 startStreamLocked();
754 }
755 } else {
756 m_currentConfig.bufferSize = previousBuffer;
757 if (openStreamLocked(m_currentConfig, m_currentCallback, m_currentUserData) &&
758 wasRunning) {
759 startStreamLocked();
760 }
761 }
762 }
763 }
764
765 m_lastUnderrunCheck = now;
766 m_lastUnderrunCount = stats.underrunCount;
767 }
768 }
769 }
770 fireModeChange(pending);
771 if (isStreamRunning()) {
772 startHealthMonitorIfStopped();
773 }
774}

776bool AudioDeviceManager::switchToSafetyDriverLocked(PendingModeChange& outPending) {
777 AESTRA_LOG_WARNING("Attempting emergency fallback to Dummy driver");
778
779 IAudioDriver* dummy = nullptr;
780 for (auto& driver : m_drivers) {
781 if (driver->getDriverType() == AudioDriverType::DUMMY) {
782 dummy = driver.get();
783 break;
784 }
785 }
786 if (!dummy) {
787 AESTRA_LOG_ERROR("[AudioDeviceManager] Critical failure: no dummy driver available for fallback");
788 return false;
789 }
790
791 if (m_activeDriver) {
792 try {
793 m_activeDriver->stopStream();
794 m_activeDriver->closeStream();
795 } catch (const std::exception& e) {
796 AESTRA_LOG_WARNING("Error closing audio driver: " + std::string(e.what()));
797 }
798 m_activeDriver = nullptr;
799 }
800
801 if (dummy->openStream(m_currentConfig, m_currentCallback, m_currentUserData)) {
802 if (dummy->startStream()) {
803 m_activeDriver = dummy;
804 AESTRA_LOG_WARNING("Safety fallback ACTIVE — audio engine still running");
805 // Capture the notification; the caller fires it after unlocking so the
806 // callback can safely re-enter a manager getter (#391 constraint 6).
807 outPending.valid = static_cast<bool>(m_driverModeChangeCallback);
808 outPending.preferred = m_preferredDriverType;
809 outPending.actual = AudioDriverType::DUMMY;
810 outPending.reason = "Hardware stall/disconnect: fallback to safety driver";
811 return true;
812 }
813 // Opened but could not start: close it and leave no active driver rather
814 // than a half-live dummy (CR review, #391 coherence). m_activeDriver stays
815 // null (cleared above), so no silent driver change is reported.
816 dummy->closeStream();
817 }
818 m_activeDriver = nullptr;
819 return false;
820}

833void AudioDeviceManager::checkDriverHealthLocked(PendingModeChange& outPending) {
834 if (!m_activeDriver || !isStreamRunningLocked() || m_activeDriver->getDriverType() == AudioDriverType::DUMMY) {
835 return;
836 }
837
838 const DriverStatistics stats = m_activeDriver->getStatistics();
839 const auto now = std::chrono::steady_clock::now();
840
841 if (!m_healthTrackingInitialized || stats.callbackCount != m_healthLastCallbackCount) {
842 m_healthLastCallbackCount = stats.callbackCount;
843 m_healthLastUpdateTime = now;
844 m_healthTrackingInitialized = true;
845 return;
846 }
847
848 const auto elapsed = std::chrono::duration_cast<std::chrono::milliseconds>(now - m_healthLastUpdateTime).count();
849 if (elapsed > 2000) {
850 Aestra::Log::error("[AudioDeviceManager] DRIVER STALL DETECTED (" + m_activeDriver->getDisplayName() +
851 "), switching to safety driver");
852 switchToSafetyDriverLocked(outPending);
853 m_healthLastUpdateTime = now;
854 }
855}

866void AudioDeviceManager::fireModeChange(const PendingModeChange& pending) const {
867 if (!pending.valid) {
868 return;
869 }
870 // m_driverModeChangeCallback is read under the lock in the capturing path;
871 // here we only fire what was captured. A concurrent setDriverModeChangeCallback
872 // cannot tear this call because we invoke the local target snapshot.
873 DriverModeChangeCallback cb;
874 {
875 std::lock_guard<std::mutex> lock(m_mutex);
876 cb = m_driverModeChangeCallback;
877 }
878 if (cb) {
879 cb(pending.preferred, pending.actual, pending.reason);
880 }

890void AudioDeviceManager::startHealthMonitorIfStopped() {
891 // Guarded by m_healthMonitorMutex (the monitor lifecycle lock), NOT m_mutex:
892 // this serializes concurrent start/stop from different transaction threads and
893 // protects the thread handle. The spawned loop acquires m_mutex itself.
894 std::lock_guard<std::mutex> lock(m_healthMonitorMutex);
895 if (m_healthMonitorRunning.load(std::memory_order_acquire)) {
896 return;
897 }
898 if (m_healthMonitorThread.joinable()) {
899 m_healthMonitorThread.join(); // reap a previously-stopped run
900 }
901 m_healthMonitorRunning.store(true, std::memory_order_release);
902 m_healthMonitorThread = std::thread(&AudioDeviceManager::healthMonitorLoop, this);
903}

909void AudioDeviceManager::stopHealthMonitor() {
910 // Lock-free w.r.t. m_mutex by contract (#391 constraint 5): never called while
911 // m_mutex is held. Uses only the lifecycle lock, so joining here cannot
912 // deadlock against the monitor thread (which only wants m_mutex).
913 std::lock_guard<std::mutex> lock(m_healthMonitorMutex);
914 m_healthMonitorRunning.store(false, std::memory_order_release);
915 if (m_healthMonitorThread.joinable()) {
916 m_healthMonitorThread.join();
917 }
