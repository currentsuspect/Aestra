92void BPMDisplay::onUpdate(double deltaTime) {
93 // Smooth scrolling animation towards target BPM
94 const float animSpeed = 8.0f; // Faster animation for responsiveness
95 float diff = m_targetBPM - m_displayBPM;
96
97 if (std::abs(diff) > 0.01f) {
98 m_displayBPM += diff * animSpeed * static_cast<float>(deltaTime);
99 } else {
100 m_displayBPM = m_targetBPM;
101 }
102
103 // Decay pulse animation
104 if (m_pulseAnimation > 0.0f) {
105 m_pulseAnimation -= static_cast<float>(deltaTime) * 4.0f; // Fast decay
106 if (m_pulseAnimation < 0.0f) m_pulseAnimation = 0.0f;
107 }
108
109 // Hold-to-repeat: continuously adjust BPM when arrow is held
110 if (m_upArrowPressed || m_downArrowPressed) {
111 m_holdDelay -= static_cast<float>(deltaTime);
112 if (m_holdDelay <= 0.0f) {
113 m_holdTimer += static_cast<float>(deltaTime);
114 // After 0.3s initial delay, repeat every 50ms
115 if (m_holdTimer >= 0.05f) {
116 m_holdTimer = 0.0f;
117 if (m_upArrowPressed) {
118 incrementBPM(1.0f);
119 } else {
120 decrementBPM(1.0f);
121 }
122 }
123 }
124 }
125
126 AestraUI::NUIComponent::onUpdate(deltaTime);
127}

143bool BPMDisplay::onMouseEvent(const AestraUI::NUIMouseEvent& event) {
144 AestraUI::NUIRect bounds = getBounds();
145 AestraUI::NUIRect upBounds = getUpArrowBounds();
146 AestraUI::NUIRect downBounds = getDownArrowBounds();
147
148 bool inBounds = bounds.contains(event.position);
149 bool inUp = upBounds.contains(event.position);
150 bool inDown = downBounds.contains(event.position);
151
152 // Track hover state for visual feedback
153 bool wasHovered = m_isHovered;
154 m_isHovered = inBounds;
155 m_upArrowHovered = inUp;
156 m_downArrowHovered = inDown;
157
158 // Handle mouse wheel for fine adjustment (anywhere on BPM display)
159 if (event.wheelDelta != 0.0f && inBounds) {
160 // Modifier keys: Shift = 5x faster, Ctrl = 0.1x for fine control
161 float increment = 1.0f;
162 if (event.modifiers & AestraUI::NUIModifiers::Shift) {
163 increment = 5.0f;
164 } else if (event.modifiers & AestraUI::NUIModifiers::Ctrl) {
165 increment = 0.1f;
166 }
167
168 if (event.wheelDelta > 0) {
169 incrementBPM(increment);
170 } else {
171 decrementBPM(increment);
172 }
173 m_pulseAnimation = 1.0f; // Trigger pulse
174 return true;
175 }
176
177 // Handle mouse button for arrow clicks
178 if (event.pressed && event.button == AestraUI::NUIMouseButton::Left) {
179 if (inUp) {
180 m_upArrowPressed = true;
181 m_holdDelay = 0.3f; // 300ms before repeat starts
182 m_holdTimer = 0.0f;
183 incrementBPM(1.0f);
184 m_pulseAnimation = 1.0f;
185 return true;
186 }
187 if (inDown) {
188 m_downArrowPressed = true;
189 m_holdDelay = 0.3f;
190 m_holdTimer = 0.0f;
191 decrementBPM(1.0f);
192 m_pulseAnimation = 1.0f;
193 return true;
194 }
195 }
196
197 // Handle mouse button release
198 if (event.released && event.button == AestraUI::NUIMouseButton::Left) {
199 m_upArrowPressed = false;
200 m_downArrowPressed = false;
201 }
202
203 // Capture hover changes for redraw
204 if (wasHovered != m_isHovered || inUp || inDown) {
205 if (inBounds) {
206 AestraUI::NUIComponent::showRemoteTooltip("BPM - scroll to adjust", event.position, this);
207 } else {
208 AestraUI::NUIComponent::hideRemoteTooltip(this);
209 }
210 return true; // Consume event to trigger redraw
211 }
212
213 return AestraUI::NUIComponent::onMouseEvent(event);
214}

231std::string TimerDisplay::formatTime(double seconds) const {
232 int totalSeconds = static_cast<int>(seconds);
233 int minutes = totalSeconds / 60;
234 int secs = totalSeconds % 60;
235 int millis = static_cast<int>((seconds - totalSeconds) * 100);
236
237 std::stringstream ss;
238 ss << minutes << ":" << std::setfill('0')
239 << std::setw(2) << secs << "."
240 << std::setw(2) << millis;
241 return ss.str();
242}

275void TimeSignatureDisplay::cycleNext() {
276 // Common time signatures: 2/4, 3/4, 4/4, 5/4, 6/8, 7/8
277 static const int signatures[] = {2, 3, 4, 5, 6, 7};
278 static const int numSignatures = 6;
279
280 int currentIndex = 0;
281 for (int i = 0; i < numSignatures; ++i) {
282 if (signatures[i] == m_beatsPerBar) {
283 currentIndex = i;
284 break;
285 }
286 }
287
288 m_beatsPerBar = signatures[(currentIndex + 1) % numSignatures];
289
290 if (m_onTimeSignatureChange) {
291 m_onTimeSignatureChange(m_beatsPerBar);
292 }
293
294 setDirty(true);
295}

362void TransportInfoContainer::layoutComponents() {
363 AestraUI::NUIRect bounds = getBounds();
364
365 // Expanded Layout: [TimeSig 50px] [BPM 90px] [Timer 80px] = ~220px
366
367 float padding = 4.0f; // Increased padding
368 float contentHeight = 28.0f;
369 float yPos = bounds.y + (bounds.height - contentHeight) / 2.0f;
370
371 float currentX = bounds.x;
372
373 // 1. Time Signature (Left)
374 float timeSigWidth = 50.0f; // Increased
375 if (m_timeSignatureDisplay) {
376 m_timeSignatureDisplay->setBounds(AestraUI::NUIRect(std::floor(currentX), std::floor(yPos), timeSigWidth, contentHeight));
377 }
378 currentX += timeSigWidth + padding;

380 // 2. BPM (Center/Next)
