133 // Stand in for the helper, so a test decides when each result comes back (#1229): set up the hand-off
134 // without a process, take the chunk process() sent, hand a result back. Same protocol as workerLoop().
135 void prepareHandOffForTest(double sampleRate, uint32_t maxBlockSize) { resetHandOff(sampleRate, maxBlockSize); }
136 bool takeChunkForTest(std::vector<float>& interleaved, uint32_t& frames) {
137 uint8_t posted = 2;
138 if (!m_pendingState.compare_exchange_strong(posted, 3, std::memory_order_acq_rel)) {
139 return false;
140 }
141 frames = m_pendingFrames.load(std::memory_order_acquire);
142 interleaved.assign(m_pendingInput.begin(), m_pendingInput.begin() + size_t(frames) * m_transportChannels);
143 m_workerStart = m_pendingStart;
144 m_pendingState.store(0, std::memory_order_release);
145 return true;
146 }
147 bool postResultForTest(const std::vector<float>& interleaved, uint32_t frames) {
148 uint8_t empty = 0;
149 if (!m_readyState.compare_exchange_strong(empty, 1, std::memory_order_acq_rel)) {
150 return false; // the audio thread has not taken the last one: the result is lost, as in workerLoop()
151 }
152 std::copy_n(interleaved.data(), size_t(frames) * m_transportChannels, m_readyOutput.data());
153 m_readyStart = m_workerStart;
154 m_readyFrames.store(frames, std::memory_order_release);
155 m_readyState.store(2, std::memory_order_release);
156 return true;
157 }

240 // Host->child parameter changes. setParameter() only pushes onto this
241 // lock-free queue (no IPC, no lock, no allocation, never blocks); the worker
242 // thread (single consumer) drains it and forwards each change to the child as
243 // an ordered SETPARAM command, keeping the realtime callback free of parameter
244 // IPC (#238). The ring is SPSC: setParameter must be called from one producer
245 // thread. Today that is the UI/message thread — automation gates plugin-
246 // parameter curves to Internal-format plugins in AudioEngine, so the render
247 // thread never reaches OOP setParameter. Because the push itself never blocks,
248 // a future single RT producer (e.g. third-party automation, #467) would remain
249 // realtime-safe; adding a *second* concurrent producer would require an MPSC
250 // queue instead.
251 struct ParamChange {
252 uint32_t id{0};
253 float value{0.0f};
254 };
255 // Sized to comfortably absorb a full-preset burst (hundreds of distinct
256 // parameters applied in a tight loop) before the worker drains it.
257 static constexpr size_t kParamQueueCapacity = 4096;
258 // Cap per worker pass so a large burst against a slow child cannot monopolize
259 // the worker and starve PROCESS (which would make the RT callback fall back to
260 // audible passThrough). The remainder stays queued for the next pass.
261 static constexpr size_t kMaxParamDrainPerPass = 64;
262 LockFreeRingBuffer<ParamChange, kParamQueueCapacity> m_paramQueue;
263 // Automation (V82-H5) has its own single producer, the render thread, so it gets its own ring; the worker drains
264 // both. A curve hands over its value every block, so only a value that changed is queued (recent[] remembers the
265 // last one sent per parameter; a miss in it just sends once more).
266 LockFreeRingBuffer<ParamChange, 1024> m_autoQueue;
267 struct RecentAutomation {
268 uint32_t id{0};
269 float value{0.0f};
270 bool valid{false};
