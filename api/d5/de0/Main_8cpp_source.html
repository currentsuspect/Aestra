33std::filesystem::path getExecutableDirectory() {
34#ifdef _WIN32
35 char exePath[MAX_PATH] = {};
36 const DWORD len = GetModuleFileNameA(nullptr, exePath, MAX_PATH);
37 if (len > 0 && len < MAX_PATH) {
38 return std::filesystem::path(std::string(exePath, len)).parent_path();
39 }
40#elif __APPLE__
41 char exePath[PATH_MAX] = {};
42 uint32_t size = sizeof(exePath);
43 if (_NSGetExecutablePath(exePath, &size) == 0) {
44 return std::filesystem::path(exePath).parent_path();
45 }
46#else
47 char exePath[4096] = {};
48 const ssize_t len = readlink("/proc/self/exe", exePath, sizeof(exePath) - 1);
49 if (len > 0) {
50 exePath[len] = '\0';
51 return std::filesystem::path(exePath).parent_path();
52 }
53#endif
54 return {};
55}

57std::filesystem::path findAestraFontDirectory() {
58 namespace fs = std::filesystem;
59 std::vector<fs::path> roots;
60 roots.push_back(fs::current_path());
61
62 const fs::path exeDir = getExecutableDirectory();
63 if (!exeDir.empty() && exeDir != roots.front()) {
64 roots.push_back(exeDir);
65 }
66
67 std::error_code ec;
68 for (const auto& root : roots) {
69 fs::path probe = root;
70 while (!probe.empty()) {
71 const fs::path candidate = probe / "AestraAssets" / "fonts";
72 const fs::path geistMedium = candidate / "Geist" / "Geist-Medium.ttf";
73 if (fs::exists(geistMedium, ec) && !ec) {
74 return candidate;
75 }
76
77 const fs::path parent = probe.parent_path();
78 if (parent == probe) {
79 break;
80 }
81 probe = parent;
82 }
83 }
84
85 return {};
86}

88void configureFontDirectoryEnv() {
89 if (std::getenv("AESTRA_FONT_DIR") != nullptr) {
90 Log::info("AESTRA_FONT_DIR already set externally");
91 return;
92 }
93
94 const auto fontDir = findAestraFontDirectory();

100#ifdef _WIN32

104#endif

107}
