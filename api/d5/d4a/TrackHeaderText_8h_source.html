3// The second line of a timeline track header (FD-50, board 8): what plays the track, and where it goes when that
4// is not Master: "Kick - Hard → Drum Bus", "Audio", "4 sounds". The board's "Arsenal ·" prefix is left out: next to
5// the M S R keys the line has about 100 px, and the prefix used most of it. Header-only and free of the
6// models, so TrackHeaderTextTest runs in the headless build; TrackUIComponentHeader.cpp gathers the facts.

25inline std::string sourceLine(const LaneSource& source, const std::string& outputName) {
26 std::string line;
27 if (source.unitNames.size() == 1) {
28 line = source.unitNames.front();
29 } else if (source.unitNames.size() > 1) {
30 line = std::to_string(source.unitNames.size()) + " sounds";
31 }
32 if (source.hasAudio) line = line.empty() ? "Audio" : line + " + audio";
33 if (!outputName.empty()) line = line.empty() ? "→ " + outputName : line + " → " + outputName;
34 return line;
35}
