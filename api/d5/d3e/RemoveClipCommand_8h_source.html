18 void execute() override {
19 if (m_executed)
20 return;
21
22 // Capture clip state before removal

24 if (!clip)
25 return;
26
27 m_clipState = *clip;

29

31 m_model.removeClip(m_clipId);
32
33 m_executed = true;
34 }

36 void undo() override {
37 if (!m_executed)
38 return;
39
40 // Re-add the clip with its original state
41 m_model.addClip(m_laneId, m_clipState);
42
43 m_executed = false;
44 }
45
46 void redo() override {

48 return;

51 m_executed = true;
52 }
