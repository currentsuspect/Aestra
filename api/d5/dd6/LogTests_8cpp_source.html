184

188 auto fileLogger = std::make_shared<FileLogger>(logFile, LogLevel::Debug);
189 Log::init(fileLogger);
190
191 const int numThreads = 4;
192 const int messagesPerThread = 100;
193 std::vector<std::thread> threads;
194
195 for (int t = 0; t < numThreads; ++t) {
196 threads.emplace_back([t, messagesPerThread]() {
197 for (int i = 0; i < messagesPerThread; ++i) {
198 AESTRA_LOG_STREAM_INFO << "Thread " << t << " message " << i;
199 }
200 });
201 }
202
203 for (auto& thread : threads) {
204 thread.join();
205 }
206
207 // Give file logger time to flush
208 std::this_thread::sleep_for(std::chrono::milliseconds(50));
209
210 // Verify file contains expected number of messages
211 std::string content = File::readAllText(logFile);
212 int messageCount = 0;
213 size_t pos = 0;
214 while ((pos = content.find("[INFO]", pos)) != std::string::npos) {
215 messageCount++;
216 pos++;
217 }
218
219 TEST_ASSERT(messageCount == numThreads * messagesPerThread, "Should have all messages from all threads");
220
221 // Cleanup
222 std::remove(logFile.c_str());
