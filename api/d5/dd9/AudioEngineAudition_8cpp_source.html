3// Auditioning a unit outside the transport: the piano roll's click (an eighth of a second) and the editors'
4// Preview (the sample's length, and a stop). Runs on the audio thread from the command drain; bounded, no
5// allocation, no locks.
6#include "Core/AudioEngine.h"
7
8#include <algorithm>

10namespace Aestra {
11namespace Audio {

13void AudioEngine::applyAuditionUnitRt(const AudioQueueCommand& cmd) {
14 // A negative note stops this unit's auditions (the Sample Editor's Preview, pressed again).
15 if (cmd.value1 < 0.0f) {
16 for (auto& s : m_unitAuditionStates) {
17 if (s.active && s.unitId == static_cast<UnitID>(cmd.trackIndex)) {
18 s.noteOffSamplesRemaining = std::min<uint32_t>(s.noteOffSamplesRemaining, 1);
19 }
20 }
21 return;
22 }
23 // Claim a slot from the pool: a free one when possible, otherwise
24 // the one closest to its note-off. A chord stamp sends one command
25 // per pitch, so simultaneous triad tones each get their own slot.
26 UnitAuditionState* slot = nullptr;
27 for (auto& s : m_unitAuditionStates) {
28 if (!s.active) {
29 slot = &s;
30 break;
31 }
32 }
33 if (!slot) {
34 slot = &m_unitAuditionStates[0];
35 for (auto& s : m_unitAuditionStates) {
36 if (s.noteOffSamplesRemaining < slot->noteOffSamplesRemaining) {
37 slot = &s;
38 }
39 }
40 }
41 slot->unitId = static_cast<UnitID>(cmd.trackIndex);
42 slot->note = static_cast<uint8_t>(std::clamp(static_cast<int>(cmd.value1), 0, 127));
43 slot->velocity = static_cast<uint8_t>(std::clamp(static_cast<int>(cmd.value2 * 127.0f), 1, 127));
44 // A length, when given (an editor previewing a whole sample), capped at a minute; else the click's eighth.
45 const uint32_t eighth = std::max<uint32_t>(1, m_sampleRate.load(std::memory_order_relaxed) / 8);
46 const uint64_t minute = static_cast<uint64_t>(m_sampleRate.load(std::memory_order_relaxed)) * 60u;
47 slot->noteOffSamplesRemaining =
48 cmd.samplePos > 0 ? static_cast<uint32_t>(std::min<uint64_t>(cmd.samplePos, minute)) : eighth;
49 slot->noteOnPending = true;
50 slot->active = true;
51 // Wake the master out of the post-stop Silent fast path so the
52 // audition is actually rendered. After the transport stops the fade
53 // settles to Silent, whose early-return drops everything before the
54 // unit-audition MIDI is injected — mirrors the metronome count-in
55 // recovery, and matches the pre-first-play (None) state.
56 if (m_fadeState.load(std::memory_order_relaxed) == FadeState::Silent) {
57 m_fadeState.store(FadeState::None, std::memory_order_relaxed);
58 }
59}
