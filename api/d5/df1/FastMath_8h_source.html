27// Fast Sine/Cosine (5th order polynomial, max error < 0.001)
28// =============================================================================

39inline float fastSin(float x) noexcept {
40 // Wrap to [-PI, PI] range
41 // x = x - TWO_PI * std::floor((x + PI) * INV_TWO_PI); // Standard wrapping
42 // Since this is "FastMath" and usually used for LFOs/Panning in known ranges,
43 // we can skip expensive wrapping if we assume inputs are vaguely sane or handle it via branching.
44 // But for safety and generic use, valid range handling is good.
45 // However, Bhaskara works for [0, PI].
46
47 // Quick wrap for standard audio ranges [-PI, PI] or [0, 2PI]
48 // Exploiting symmetry for Bhaskara:
49
50 if (x < -PI)
51 x += TWO_PI;
52 if (x > PI)
53 x -= TWO_PI;
54
55 // Parabolic approximation (Bhaskara I)
56 // sin(x) ≈ (16 * x * (π - x)) / (5 * π² - 4 * x * (π - x)) for x in [0, π]
57 // sin(-x) = -sin(x)
58
59 float sign = 1.0f;
60 if (x < 0) {
61 x = -x;
62 sign = -1.0f;
63 }
64
65 // If x > PI (after wrap? No, assuming input is somewhat near range),
66 // Map [PI, 2PI] -> [-PI, 0] logic handled by wrap?
67 // Let's assume input is in [-PI, PI] after wrapping logic above.
68 // Now x is in [0, PI].
69
70 // Bhaskara I:
71 // P = x * (PI - x)
72 // result = (16 * P) / (5 * PI*PI - 4 * P)
73
74 const float P = x * (PI - x);
75 const float Q = 49.3480220054f; // 5 * PI^2
76
77 return sign * (16.0f * P) / (Q - 4.0f * P);
78}

111inline void fastPan(float pan, float& leftGain, float& rightGain) noexcept {
112 // Convert pan [-1, 1] to angle [0, π/2]
113 float angle = (pan + 1.0f) * 0.25f * PI;
114
115 // Use fast sin/cos for gains
116 leftGain = fastCos(angle);

126inline void fastPanLinear(float pan, float& leftGain, float& rightGain) noexcept {
127 // Linear approximation of constant-power curve
128 // L = sqrt((1-pan)/2), R = sqrt((1+pan)/2)
129 // Approximated with: L ≈ (1-pan)*0.7071 + (1-pan²)*0.2929
130
131 float panAbs = pan * pan;
132 leftGain = 0.7071067811865f - 0.5f * pan + 0.2928932188f * (1.0f - panAbs);
133 rightGain = 0.7071067811865f + 0.5f * pan + 0.2928932188f * (1.0f - panAbs);
134}

146inline float fastDbToLinear(float dB) noexcept {
147 // Clamp to reasonable range
148 if (dB <= -96.0f)
149 return 0.0f;
150 if (dB >= 24.0f)
151 dB = 24.0f;
152
153 // 10^(dB/20) = e^(dB * ln(10)/20) = e^(dB * 0.11512925...)
154 // Use fast exp approximation
155 float x = dB * 0.11512925464970228f;
156
157 // Fast exp approximation: e^x ≈ (1 + x/n)^n with n=8
158 float e = 1.0f + x * 0.125f;
159 e = e * e; // ^2
160 e = e * e; // ^4
161 e = e * e; // ^8
162
163 return e;
164}

169inline float fastLinearToDb(float linear) noexcept {
170 if (linear <= 0.0f)
171 return -96.0f;
172
173 // 20 * log10(linear) = 20 / ln(10) * ln(linear) = 8.685889... * ln(linear)
174 // Use fast log approximation
175
176 // IEEE 754 floating point log trick
177 union {
178 float f;
179 uint32_t i;
180 } v = {linear};
181 float lnx = (float)(v.i - 1064866805) * 8.262958e-8f;
182
183 return lnx * 8.685889638065f;
184}

193inline float fastInvSqrt(float x) noexcept {
194 union {
195 float f;
196 uint32_t i;
197 } conv = {x};
198 conv.i = 0x5f3759df - (conv.i >> 1);
199 conv.f *= 1.5f - (x * 0.5f * conv.f * conv.f);
200 return conv.f;
201}

206inline float fastSqrt(float x) noexcept {
207 if (x <= 0.0f)
208 return 0.0f;
209 return x * fastInvSqrt(x);
