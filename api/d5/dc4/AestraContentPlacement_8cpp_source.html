20void AestraContent::placePatternInTimeline(PatternID pid) {
21 if (!m_trackManager || !pid.isValid()) return;
22 auto& tm = *m_trackManager;
23 auto& playlist = tm.getPlaylistModel();
24 const auto* pattern = tm.getPatternManager().getPattern(pid);
25 if (!pattern) return;
26
27 // Where the arrangement stands: every clip, with its lane's place in the list.
28 std::vector<PlacedClip> clips;
29 std::vector<PlaylistLaneID> laneIds;
30 playlist.withLanes([&](const auto& lanes) {
31 for (size_t i = 0; i < lanes.size(); ++i) {
32 laneIds.push_back(lanes[i].id);
33 for (const auto& c : lanes[i].clips) clips.push_back({static_cast<int>(i), c.startBeat, c.startBeat + c.durationBeats, c.patternId.value});
34 }
35 return 0;
36 });
37
38 // In the Arsenal the transport is the pattern loop, so the timeline's cursor is where it was left on the way in.
39 const double cursorSeconds = m_viewFocus == ViewFocus::Arsenal ? m_savedTimelinePosition : tm.getPosition();
40 const double beatsPerBar = static_cast<double>(tm.getTimelineClock().getBeatsPerBar());
41 const PlacementTarget target = planPlacement(clips, static_cast<int>(laneIds.size()), pid.value, pattern->lengthBeats,
42 playlist.secondsToBeats(std::max(0.0, cursorSeconds)), beatsPerBar);
43
44 // A new lane and its clip are one undo step, like a drop (the lane is part of the edit).
45 std::shared_ptr<CreateTrackWithLaneCommand> laneCommand;
46 PlaylistLaneID laneId;
47 if (target.lane >= 0) {
48 laneId = laneIds[static_cast<size_t>(target.lane)];
49 } else {
50 laneCommand = std::make_shared<CreateTrackWithLaneCommand>(tm, pattern->name.empty() ? "Pattern" : pattern->name);
51 laneCommand->execute();
52 laneId = laneCommand->getLaneId();
53 if (!laneId.isValid()) { showToast("Could not make a track for the pattern"); return; }
54 }
55
56 ClipInstance clip;
57 clip.id = ClipInstanceID::generate();
58 clip.startBeat = target.startBeat;
59 clip.durationBeats = pattern->lengthBeats;
60 clip.patternId = pid;
61 clip.sourceId = pid.value;
62 auto add = std::make_shared<AddClipCommand>(playlist, laneId, clip);
63 add->execute();
64 if (!playlist.getClip(clip.id)) {
65 if (laneCommand) laneCommand->undo();
66 showToast("Could not place the pattern");
67 return;
68 }
69 auto transaction = std::make_shared<CommandTransaction>("Place Pattern in Timeline");
70 if (laneCommand) transaction->add(laneCommand);
71 transaction->add(add);
72 transaction->markExecuted();
73 tm.getCommandHistory().pushExecuted(transaction);
74
75 tm.refreshTimelinePatternInstances(); // a clip added while playing must be scheduled
76 if (m_trackManagerUI) {
77 m_trackManagerUI->refreshTracks();
78 m_trackManagerUI->invalidateCache();
79 }
80 setDirty(true);
81 const int bar = static_cast<int>(std::floor(target.startBeat / std::max(beatsPerBar, 1.0))) + 1;
82 const int track = (target.lane >= 0 ? target.lane : static_cast<int>(laneIds.size())) + 1;
83 showToast("Placed " + (pattern->name.empty() ? std::string("pattern") : pattern->name) + " at bar " + std::to_string(bar) + ", track " + std::to_string(track));
84}
