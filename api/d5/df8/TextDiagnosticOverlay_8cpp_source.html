63void TextDiagnosticOverlay::sectionRule(NUIRenderer& renderer, const char* label, float& y) {
64 const float x = getBounds().x + PADDING;
65 y += 6.0f;
66 renderer.drawText(label, NUIPoint(x, y), kLabelSize, kHeading);
67 const float ruleY = y + 4.0f;
68 // Rule starts past the label rather than under it; measureText is the only
69 // honest way to know where that is, and it exercises the same metrics path
70 // the samples below use.
71 const float labelWidth = renderer.measureText(label, kLabelSize).width;
72 renderer.drawLine(NUIPoint(x + labelWidth + 8.0f, ruleY),
73 NUIPoint(getBounds().x + PANEL_WIDTH - PADDING, ruleY), 1.0f, kRule);
74 y += LINE + 2.0f;
75}

86void TextDiagnosticOverlay::renderState(NUIRenderer& renderer, const NUIRenderer::TextDiagnostics& d, float& y) {
87 sectionRule(renderer, "PIPELINE STATE", y);
88
89 keyValue(renderer, "rasteriser",
90 d.lcdSubpixel ? "FreeType · LCD subpixel (filtered)" : "FreeType · grayscale",
91 y);
92 // The shader collapses the LCD triplet to one scalar, so the filter's blur
93 // is paid for and its resolution discarded (F1, measured 2026-09-12).
94 if (d.lcdSubpixel) {
95 keyValue(renderer, "", "subpixel data is averaged away in the shader (F1)", y, true, true);
96 }
97 keyValue(renderer, "textContrast_", f(d.textContrast), y);
98 keyValue(renderer, "caller alpha",
99 d.alphaPreserved ? "preserved — multiplied, never reshaped"
100 : "RESHAPED — see section B",
101 y, true, !d.alphaPreserved);
102 keyValue(renderer, "blend space",
103 d.framebufferSRGB ? "LINEAR — split from the cache, see F3"
104 : "gamma (sRGB) — cached and direct agree",
105 y, true, d.framebufferSRGB);
106}

108void TextDiagnosticOverlay::renderTierTable(NUIRenderer& renderer, const NUIRenderer::TextDiagnostics& d, float& y) {
109 sectionRule(renderer, "RESOLVED UNIFORMS — what the shader will use", y);
110 const float x = getBounds().x + PADDING;
111 const float cols[] = {0.0f, 70.0f, 150.0f, 240.0f, 320.0f};
112
113 renderer.drawText("tier", NUIPoint(x + cols[0], y), kLabelSize, kLabel);
114 renderer.drawText("atlas", NUIPoint(x + cols[1], y), kLabelSize, kLabel);
115 renderer.drawText("serves", NUIPoint(x + cols[2], y), kLabelSize, kLabel);
116 renderer.drawText("gamma", NUIPoint(x + cols[3], y), kLabelSize, kLabel);
117 renderer.drawText("sharpen", NUIPoint(x + cols[4], y), kLabelSize, kLabel);
118 y += LINE;
119
120 for (int i = 0; i < d.tierCount; ++i) {
121 const auto& t = d.tiers[i];
122 std::ostringstream serves;
123 if (t.maxFontSize > 0.0f) serves << "<= " << f(t.maxFontSize, 2) << " px";
124 else serves << "above";
125
126 // The small tier is baked at 16 px and serves up to 17 px — it is
127 // magnified, not supersampled, yet still gets the aggressive tiny-atlas
128 // treatment (F5). Flagged where it is visible rather than in a comment.
129 const bool magnifies = (t.maxFontSize > 0.0f && t.atlasSize > 0 &&
130 t.maxFontSize > static_cast<float>(t.atlasSize));
131
132 renderer.drawText(t.name, NUIPoint(x + cols[0], y), kValueSize, magnifies ? kBad : kValue);
133 renderer.drawText(std::to_string(t.atlasSize) + " px", NUIPoint(x + cols[1], y), kValueSize, kValue);
134 renderer.drawText(serves.str(), NUIPoint(x + cols[2], y), kValueSize, magnifies ? kBad : kValue);
135 renderer.drawText(f(t.gamma), NUIPoint(x + cols[3], y), kValueSize, kValue);
136 renderer.drawText(f(t.sharpen, 2), NUIPoint(x + cols[4], y), kValueSize, kValue);
137 y += LINE;
138 }

154void TextDiagnosticOverlay::renderAlphaRamp(NUIRenderer& renderer, const NUIRenderer::TextDiagnostics& d, float& y) {
155 sectionRule(renderer, "B · ALPHA RAMP — caller alpha must survive the pipeline", y);
156 const float x = getBounds().x + PADDING;
157
158 // This section used to predict a distortion. It now asserts its absence,
159 // which is the shape a regression check wants: the renderer reports whether
160 // it reshapes caller alpha, and the ramp shows what that looks like.
161 //
162 // The old 0.35 light-theme lift is still computed, as the counter-example.
163 // Keeping it visible is the point — it is what a reintroduced compensation
164 // would do, and the "was" column is the only thing that makes "is" legible.
165 constexpr float kRemovedLift = 0.35f;
166 constexpr float kLightThemeContrast = 0.88f;
167
168 // Two rendered columns, not one. A single column proves nothing on a dark
169 // theme: textContrast_ is 1.0 there, the removed lift resolved to 1.0 too,
170 // and the old pipeline produced identity as well. The discriminating case
171 // needs alpha < 1 AND the light-theme contrast at the same time, which no
172 // other section produces — C varies contrast at full alpha, and this used to
173 // vary alpha at whatever contrast happened to be live.
174 //
175 // So the right column is drawn under textContrast_ = 0.88. Before the fix it
176 // rendered visibly heavier than the left at every step below 1.0; the two
177 // columns matching is the proof, and it holds on either theme.
178 // Offsets must match the sample draws below exactly — 48 / 190 / 410. A
179 // column heading over the wrong column is worse in a diagnostic than no
180 // heading at all: it produces a confident misattribution, which is the one
181 // failure this panel exists to prevent.
182 renderer.drawText("at live contrast", NUIPoint(x + 48.0f, y), kLabelSize, kLabel);
183 renderer.drawText("at 0.88 (light)", NUIPoint(x + 190.0f, y), kLabelSize, kLabel);
184 renderer.drawText("was", NUIPoint(x + 410.0f, y), kLabelSize, kLabel);
185 y += LINE;
186
187 const float alphas[] = {1.0f, 0.75f, 0.5f, 0.25f};
188 for (float a : alphas) {
189 renderer.drawText("a=" + f(a, 2), NUIPoint(x, y), kLabelSize, kLabel);
190
191 renderer.drawText(SAMPLE, NUIPoint(x + 48.0f, y), 11.0f, kSampleInk.withAlpha(a));
192 renderer.flush();
193 renderer.setTextContrast(kLightThemeContrast);
194 renderer.drawText(SAMPLE, NUIPoint(x + 190.0f, y), 11.0f, kSampleInk.withAlpha(a));
195 renderer.flush();
196 renderer.setTextContrast(d.textContrast);
197 renderer.flush();
198
199 renderer.drawText("(" + f(std::pow(a, kRemovedLift)) + ")",
200 NUIPoint(x + 410.0f, y), kLabelSize, kLabel);
201 y += LINE + 3.0f;
202 }
203
204 // The assertion, stated so a reader does not have to infer it from four rows
205 // of numbers agreeing.
206 // State the check rather than claim a verdict: the panel cannot read back
207 // its own pixels, so it says what a reader must look for. Both columns
208 // fading in step with a is the pass; either one staying flat is the F7
209 // regression. The columns are not expected to be identical — 0.88 contrast
210 // legitimately thickens strokes through uTextGamma — but their ramps must
211 // have the same shape.
212 renderer.drawText(d.alphaPreserved
213 ? "both columns must fade in step with a; a flat column = renderer overriding alpha"
214 : "FAIL — the renderer reports it is reshaping caller alpha (F7 regression)",
215 NUIPoint(x + 48.0f, y), kLabelSize, d.alphaPreserved ? kGood : kBad);
216 y += LINE;
217}

219void TextDiagnosticOverlay::renderPolarity(NUIRenderer& renderer, const NUIRenderer::TextDiagnostics& d, float& y) {
220 sectionRule(renderer, "C · POLARITY x COMPENSATION — 2x2, one variable per axis", y);
221 const float x = getBounds().x + PADDING;
222
223 // A 2x2 and not two rows, deliberately. Ground polarity and textContrast_
224 // are different mechanisms — one is how the glyph composites against what is
225 // behind it, the other is the compensation the frame loop applies for it —
226 // and varying both together produces a difference that cannot be assigned to
227 // either. The off-diagonal cells are the whole point: light ground at 1.00
228 // is the uncompensated case, dark ground at 0.88 is the compensation with
229 // nothing to compensate for.
230 const float colW = (PANEL_WIDTH - PADDING * 2.0f - 62.0f) * 0.5f;
231 const float rowH = 19.0f;
232
233 renderer.drawText("contrast 1.00", NUIPoint(x + 62.0f, y), kLabelSize, kLabel);
234 renderer.drawText("contrast 0.88", NUIPoint(x + 62.0f + colW, y), kLabelSize, kLabel);
235 y += LINE;
236
237 struct GroundRow { const char* label; NUIColor ground; NUIColor ink; };
238 const GroundRow grounds[] = {
239 {"on dark", kDarkGround, kSampleInk},
240 {"on light", kLightGround, kDarkInk},
241 };
242 const float contrasts[] = {1.0f, 0.88f};
243
244 for (const auto& g : grounds) {
245 renderer.drawText(g.label, NUIPoint(x, y + 4.0f), kLabelSize, kLabel);
246 for (int c = 0; c < 2; ++c) {
247 const float cx = x + 62.0f + colW * static_cast<float>(c);
248 renderer.fillRect(NUIRect(cx, y, colW - 4.0f, rowH), g.ground);
249 // textContrast_ resolves into uniforms uploaded with the batch, so
250 // the batch must be closed between cells or all four render with
251 // whichever value was set last. That is the only reason flush()
252 // appears in this file.
253 renderer.flush();
254 renderer.setTextContrast(contrasts[c]);
255 renderer.drawText(SAMPLE, NUIPoint(cx + 4.0f, y + 4.0f), 11.0f, g.ink);
256 renderer.flush();
257 }
258 y += rowH + 3.0f;
259 }
260
261 // Leave the renderer exactly as found: this panel reports, it does not
262 // configure. The frame loop sets this every frame anyway, but relying on
263 // that would make the overlay's correctness depend on a caller it does not
264 // own — and an overlay that leaves global state altered is a diagnostic that
265 // changes the thing it measures.
266 renderer.setTextContrast(d.textContrast);
267 renderer.flush();
268}

270void TextDiagnosticOverlay::renderCacheSplit(NUIRenderer& renderer, float& y) {
271 sectionRule(renderer, "D · RENDER CACHE — same blend space as direct since F3", y);
272 const float x = getBounds().x + PADDING;
273
274 renderer.drawText("direct", NUIPoint(x, y), kLabelSize, kLabel);
275 renderer.drawText(SAMPLE, NUIPoint(x + 60.0f, y), 12.0f, kSampleInk);
276 y += LINE + 2.0f;
277
278 auto* cache = renderer.getRenderCache();
279 if (cache == nullptr || !cache->isEnabled()) {
280 // Not a failure to report around: on Linux the cache is disabled
281 // outright (V8-C2), which is itself the answer for this row.
282 renderer.drawText("cached", NUIPoint(x, y), kLabelSize, kLabel);
283 renderer.drawText(cache == nullptr ? "no cache on this backend"
284 : "render cache disabled on this platform (V8-C2)",
285 NUIPoint(x + 60.0f, y), kValueSize, kBad);
286 y += LINE + 2.0f;
287 return;
288 }
289
290 renderer.drawText("cached", NUIPoint(x, y), kLabelSize, kLabel);
291 renderer.drawText("enabled — compare against 'direct' above", NUIPoint(x + 60.0f, y), kValueSize, kGood);
292 y += LINE + 2.0f;
293}

295void TextDiagnosticOverlay::onRender(NUIRenderer& renderer) {
296 if (!m_visible) return;
297
298 renderBackground(renderer);
299
300 float y = getBounds().y + PADDING;
301 const float x = getBounds().x + PADDING;
302
303 renderer.drawText("TEXT PIPELINE DIAGNOSTICS", NUIPoint(x, y), 11.0f, kHeading);
304 renderer.drawText("FD-21 · V8-C9 · reporting only", NUIPoint(x + 200.0f, y), kLabelSize, kLabel);
305 y += LINE + 4.0f;
306
307 NUIRenderer::TextDiagnostics d;
308 if (!renderer.getTextDiagnostics(d)) {
309 renderer.drawText("Text diagnostics unavailable: no font atlas is built.",
310 NUIPoint(x, y), kValueSize, kBad);
311 return;
312 }
313
314 renderState(renderer, d, y);
315 renderTierTable(renderer, d, y);
316 renderAtlasTierSamples(renderer, y);
317 renderAlphaRamp(renderer, d, y);
318 renderPolarity(renderer, d, y);
319 renderCacheSplit(renderer, y);
320
321 // What the next frame sizes the panel to. Taken from where the layout
322 // actually finished rather than from a constant restating it.
323 m_contentHeight = (y - getBounds().y) + PADDING;
324}
