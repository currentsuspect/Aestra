21bool PianoRollRuler::onMouseEvent(const NUIMouseEvent& event) {
22 const auto bounds = getBounds();
23 if (!bounds.contains(event.position) && !isScrubbing_ && !isDrawingZone_) return false;
24
25 // Zoom on Wheel
26 if (event.wheelDelta != 0.0f) {
27 if (onZoomRequested) {
28 float localX = event.position.x - bounds.x;
29 onZoomRequested(event.wheelDelta, localX);
30 return true;
31 }
32 }
33
34 auto scrubToPointer = [&]() {
35 const double localX = static_cast<double>(event.position.x - bounds.x + scrollX_);
36 const double beat = std::max(0.0, localX / static_cast<double>(pixelsPerBeat_));
37 playheadBeat_ = beat;
38 if (onPlayheadScrubbed) onPlayheadScrubbed(beat, true);
39 repaint();
40 };
41
42 // Loop zone (SPEC 3 §5.2): right-drag, or Ctrl+left-drag, like the timeline ruler. A plain
43 // left press still scrubs the playhead.
44 auto zoneBeatAtPointer = [&]() {
45 const double localX = static_cast<double>(event.position.x - bounds.x + scrollX_);
46 const double beat = std::max(0.0, localX / static_cast<double>(pixelsPerBeat_));
47 return snapBeat ? std::max(0.0, snapBeat(beat)) : beat;
48 };
49 const bool zonePress = event.pressed && (event.button == NUIMouseButton::Right ||
50 (event.button == NUIMouseButton::Left &&
51 (event.modifiers & NUIModifiers::Ctrl)));
52 if (zonePress && allowLoopZone_ && !isScrubbing_ && !isDrawingZone_) {
53 isDrawingZone_ = true;
54 zoneButton_ = event.button;
55 zoneAnchorBeat_ = zoneBeatAtPointer();
56 zoneDragBeat_ = zoneAnchorBeat_;
57 repaint();
58 return true;
59 }
60 if (isDrawingZone_) {
61 zoneDragBeat_ = zoneBeatAtPointer();
62 if (event.released && event.button == zoneButton_) {
63 isDrawingZone_ = false;
64 zoneButton_ = NUIMouseButton::None;
65 const double start = std::min(zoneAnchorBeat_, zoneDragBeat_);
66 const double end = std::max(zoneAnchorBeat_, zoneDragBeat_);
67 if (end - start > 1e-6 && onLoopZoneDrawn) {
68 onLoopZoneDrawn(start, end);
69 }
70 }
71 repaint();
72 return true;
73 }
74
75 if (event.pressed && event.button == NUIMouseButton::Left) {
76 isScrubbing_ = true;
77 scrubToPointer();
78 return true;
79 }
80 if (isScrubbing_) {
81 if (event.released && event.button == NUIMouseButton::Left) {
82 scrubToPointer();
83 isScrubbing_ = false;
84 if (onPlayheadScrubbed) onPlayheadScrubbed(playheadBeat_, false);
85 return true;
86 }
87 scrubToPointer();
88 return true;
89 }
90 return NUIComponent::onMouseEvent(event);
91}

93// Ruler Render: "Mature" Playlist Style
94void PianoRollRuler::onRender(NUIRenderer& renderer) {
95 if (!isVisible()) return;
96 const auto bounds = getBounds();
97 auto& theme = NUIThemeManager::getInstance();
98
99 const auto rulerBackground = NUIThemeManager::getInstance().getColor("backgroundPrimary");
100 const auto border = NUIColor::white().withAlpha(0.075f);
101 const auto highlight = NUIColor::white().withAlpha(0.014f);
102 const auto text = theme.getColor("textPrimary").withAlpha(0.86f);
103 const auto tick = NUIColor::fromHex(0x2e2e2e).withAlpha(0.82f);
104
105 renderer.fillRoundedRect(bounds, 3.0f, rulerBackground);
106 renderer.fillRect(NUIRect(bounds.x, bounds.y, bounds.width, 1.0f), highlight);
107 renderer.strokeRoundedRect(bounds, 3.0f, 1.0f, border);
108 renderer.drawLine(NUIPoint(bounds.x, bounds.bottom() - 1.0f),
109 NUIPoint(bounds.right(), bounds.bottom() - 1.0f),
110 1.0f,
111 border);
112
113 renderer.setClipRect(bounds);
114
115 // The loop zone band: the committed zone, or the one being drawn.
116 {
117 const bool drawing = isDrawingZone_;
118 const double zs = drawing ? std::min(zoneAnchorBeat_, zoneDragBeat_) : loopZoneStart_;
119 const double ze = drawing ? std::max(zoneAnchorBeat_, zoneDragBeat_) : loopZoneEnd_;
120 if ((drawing || loopZoneActive_) && ze > zs) {
121 const float x1 = bounds.x + static_cast<float>(zs * pixelsPerBeat_) - scrollX_;
122 const float x2 = bounds.x + static_cast<float>(ze * pixelsPerBeat_) - scrollX_;
123 const auto accent = theme.getColor("accentPrimary");
124 renderer.fillRect(NUIRect(x1, bounds.y, x2 - x1, bounds.height), accent.withAlpha(0.28f));
125 renderer.fillRect(NUIRect(x1, bounds.y, 1.5f, bounds.height), accent.withAlpha(0.9f));
126 renderer.fillRect(NUIRect(x2 - 1.5f, bounds.y, 1.5f, bounds.height), accent.withAlpha(0.9f));
127 }
128 }
129
130 const int beatsPerBar = std::max(1, beatsPerBar_);
131 const float pixelsPerBar = pixelsPerBeat_ * beatsPerBar;
132 int barStride = 1;
133 while ((pixelsPerBar * static_cast<float>(barStride)) < 28.0f && barStride < 128) {
134 barStride *= 2;
135 }
136
137 int startBar = static_cast<int>(std::floor(scrollX_ / pixelsPerBar));
138 startBar = static_cast<int>(std::floor(static_cast<double>(startBar) / barStride)) * barStride;
139 const int visibleBars = static_cast<int>(std::ceil((scrollX_ + bounds.width) / pixelsPerBar)) - startBar;
140 const int endBar = startBar + visibleBars + barStride;
141
142 for (int bar = startBar; bar <= endBar; bar += barStride) {
143 const float x = snapVerticalLineX(bounds.x + bar * pixelsPerBar - scrollX_);
144 const int barNumber = bar + 1;
145 const bool isMajorBar = barStride > 1 || barNumber == 1 || ((barNumber - 1) % 4 == 0);
146 const float fontSize = isMajorBar ? 11.0f : 10.0f;
147 const float textY = std::round(renderer.calculateTextY(bounds, fontSize));
148 renderer.drawText(std::to_string(barNumber),
149 NUIPoint(x + 4.0f, textY),
150 fontSize,
151 isMajorBar ? text : text.withAlpha(0.76f));
152
153 const float tickHeight = isMajorBar ? bounds.height * 0.58f : bounds.height * 0.24f;
154 renderer.drawLine(NUIPoint(x, bounds.bottom() - tickHeight),
155 NUIPoint(x, bounds.bottom()),
156 isMajorBar ? 1.15f : 1.0f,
157 isMajorBar ? tick : tick.withAlpha(0.58f));
158
159 if (pixelsPerBeat_ >= 10.0f && barStride == 1) {
160 for (int beat = 1; beat < beatsPerBar; ++beat) {
161 const float beatX = x + beat * pixelsPerBeat_;
162 renderer.drawLine(NUIPoint(beatX, bounds.bottom() - bounds.height * 0.22f),
163 NUIPoint(beatX, bounds.bottom()),
164 1.0f,
165 NUIColor::fromHex(0x262626).withAlpha(0.36f));
166 }
167 }
168 }
169
170 const float playheadX =
171 snapVerticalLineX(beatToScreenX(playheadBeat_, pixelsPerBeat_, scrollX_, bounds.x));
172 if (playheadX >= bounds.x && playheadX <= bounds.right()) {
173 // Same marker language as the Track Manager playhead: a small downward
174 // triangle at the ruler's bottom edge, tip pointing down at the grid line.
175 Playhead::drawHead(renderer, playheadX, bounds.bottom(), theme.getColor("accentPrimary"),
176 isScrubbing_ ? 1.25f : 1.0f);
177 }
178
179 renderer.clearClipRect();
180}
