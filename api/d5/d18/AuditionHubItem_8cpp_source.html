51void AuditionHub::ensureSide(const std::string& path) {
52 if (path.empty() || m_sides.count(path) || std::find(m_requested.begin(), m_requested.end(), path) != m_requested.end()) return;
53 m_requested.push_back(path);
54 auto shared = m_sideShared;
55 ++shared->working;
56 std::thread([shared, path]() {
57 std::vector<float> audio;
58 uint32_t rate = 0, channels = 0;
59 std::shared_ptr<const AuditionCompare::Side> side;
60 if (Audio::decodeAudioFile(path, audio, rate, channels) && channels > 0 && rate > 0 && !audio.empty())
61 side = std::make_shared<const AuditionCompare::Side>(AuditionCompare::measure(path, std::move(audio), channels, rate));
62 std::lock_guard<std::mutex> lock(shared->mutex);
63 if (side) shared->done.push_back(std::move(side));
64 else shared->failed.push_back(path);
65 --shared->working;
66 }).detach();
67}

69void AuditionHub::applySides() {
70 std::vector<std::shared_ptr<const AuditionCompare::Side>> done;
71 std::vector<std::string> failed;
72 {
73 std::lock_guard<std::mutex> lock(m_sideShared->mutex);
74 done.swap(m_sideShared->done);
75 failed.swap(m_sideShared->failed);
76 }
77 for (const auto& path : failed) {
78 m_requested.erase(std::remove(m_requested.begin(), m_requested.end(), path), m_requested.end());
79 Log::warning("[Audition] Could not read " + path);
80 }
81 if (done.empty() && failed.empty()) return;
82 for (auto& side : done) m_sides[side->path] = std::move(side);
83 rebuildResult();
84 setDirty(true);
85}

87void AuditionHub::rebuildResult() {
88 m_result.reset();
89 if (!m_hasItem || !m_ref) return;
90 const auto mix = m_sides.find(m_item.path), ref = m_sides.find(m_ref->path);
91 if (mix == m_sides.end() || ref == m_sides.end()) return;
92 auto result = std::make_shared<AuditionCompare::Result>(AuditionCompare::compare(*mix->second, *ref->second));
93 if (!result->ok) { Log::warning("[Audition] These two cannot be compared (one is silent)"); return; }
94 m_result = std::move(result);
95 applyToEngine();
96}

98void AuditionHub::applyToEngine() {
99 if (!m_engine || !m_result) return;
100 const double pos = m_engine->getPositionSeconds();
101 const bool playing = m_engine->isPlaying();
102 const auto source = [](const AuditionCompare::Side& side, const std::string& name) {
103 auto s = std::make_shared<Audio::ClipSource>(Audio::ClipSourceID{1}, name);
104 s->setBuffer(side.buffer);
105 return s;
106 };
107 m_engine->setMixSource(source(m_result->mix, m_item.title));
108 m_engine->setReferenceSource(source(m_result->ref, m_ref ? AuditionLibrary::displayName(*m_ref) : "reference"));
109 m_engine->setMixGainDb(0.0f);
110 m_engine->setReferenceGainDb(static_cast<float>(m_result->referenceGainDb + m_nudgeDb));
111 m_engine->setListenSide(Audio::AuditionEngine::ListenSide::Mix);
112 m_engine->seekSeconds(pos);
113 if (playing) m_engine->play();
114}

116void AuditionHub::nudgeReference(double db) {
117 m_nudgeDb = std::clamp(m_nudgeDb + db, -12.0, 12.0);
118 if (m_engine && m_result) m_engine->setReferenceGainDb(static_cast<float>(m_result->referenceGainDb + m_nudgeDb));
119 setDirty(true);

122void AuditionHub::openItem(const AuditionLibrary::Item& item) {
123 const bool sameItem = m_hasItem && m_item.path == item.path;
124 m_item = item;
125 m_hasItem = true;
126 if (!sameItem) { m_ref.reset(); m_result.reset(); m_nudgeDb = 0.0; if (m_engine) m_engine->clearReferenceSource(); }
127 ensureSide(item.path);
128 setPage(Page::Item);
129 if (!sameItem || !(m_engine && m_engine->isPlaying())) play(item);
130}

132void AuditionHub::chooseReference(const AuditionLibrary::Item& reference) {
133 if (!m_hasItem || reference.path == m_item.path) return;
134 m_ref = reference;
135 m_nudgeDb = 0.0;
136 m_result.reset();
137 ensureSide(reference.path);
138 rebuildResult(); // at once, if both are already measured
139 setDirty(true);
140}

150void AuditionHub::openReferenceMenu(const NUIPoint& at) {
151 if (m_menu) { if (auto* p = m_menu->getParent()) p->removeChild(m_menu); }
152 m_menu = std::make_shared<NUIContextMenu>();
153 const auto add = [&](const AuditionLibrary::Item& item) {
154 if (item.path == m_item.path) return;
155 auto entry = std::make_shared<NUIContextMenuItem>(AuditionLibrary::displayName(item));
156 entry->setShortcut(item.hasLufs ? fmt("%.1f LUFS", item.lufs) : std::string());
157 entry->setChecked(m_ref && m_ref->path == item.path);
158 entry->setOnClick([this, item]() { chooseReference(item); });
159 m_menu->addItem(entry);
160 };
161 for (const auto& r : m_references) add(r);
162 if (!m_references.empty() && !m_versions.empty()) m_menu->addSeparator();
163 for (const auto& v : m_versions) add(v);
164 if (m_ref) {
165 m_menu->addSeparator();
166 m_menu->addItem("Stop comparing", [this]() { clearReference(); });
167 }
168 if (m_menu->getItemCount() == 0) { m_menu.reset(); return; }
169 NUIComponent* root = this;
170 while (root->getParent()) root = root->getParent();
171 root->addChild(m_menu);
172 m_menu->showAt(at);
173 root->repaint();
174}

350bool AuditionHub::itemPageMouse(const NUIMouseEvent& event, const AuditionHubLayout::Frame& f) {
351 const auto p = AuditionHubLayout::itemFrame(f.content, m_result != nullptr);
352 const NUIPoint pos = event.position;
353 if (event.type == NUIMouseEventType::Move) {
354 const std::string before = m_hoverControl;
355 int tab = -1;
356 for (int i = 0; i < AuditionHubLayout::kTabs; ++i) if (p.tabs[static_cast<size_t>(i)].contains(pos)) tab = i;
357 if (p.compare.contains(pos)) m_hoverControl = "compare";
358 else if (tab >= 0) m_hoverControl = "tab" + std::to_string(tab);
359 else if (m_hoverControl == "compare" || m_hoverControl.rfind("tab", 0) == 0) m_hoverControl.clear();
360 if (before != m_hoverControl) setDirty(true);
361 }
362 if (event.wheelDelta != 0.0f && m_tab != ItemTab::Overview && p.body.contains(pos)) {
363 m_tabScroll = std::clamp(m_tabScroll - event.wheelDelta * 48.0f, 0.0f, std::max(0.0f, m_tabContentH - p.panel.height));
364 setDirty(true);
365 return true;
366 }
367 if (!(event.pressed && event.button == NUIMouseButton::Left) || !f.content.contains(pos)) return false;
368 if (p.play.contains(pos)) {
369 if (m_engine && m_engine->isPlaying() && m_hasNow && m_now.path == m_item.path) m_engine->pause();
370 else play(m_item);
371 setDirty(true);
372 return true;
373 }
374 if (p.compare.contains(pos)) { openReferenceMenu({p.compare.x, p.compare.bottom() + 4.0f}); return true; }
375 for (int i = 0; i < AuditionHubLayout::kTabs; ++i)
376 if (p.tabs[static_cast<size_t>(i)].contains(pos)) { setItemTab(static_cast<ItemTab>(i)); return true; }
377 if (m_tab != ItemTab::Overview && itemTabMouse(event, p)) return true;
378 if (m_tab == ItemTab::Overview && p.wave.contains(pos) && m_engine && m_hasNow && m_now.path == m_item.path) {
379 m_engine->seekNormalized(std::clamp((pos.x - p.wave.x) / p.wave.width, 0.0f, 1.0f));
380 return true;
381 }
