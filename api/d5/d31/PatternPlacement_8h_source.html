37inline PlacementTarget planPlacement(const std::vector<PlacedClip>& clips, int laneCount, uint64_t patternId,
38 double patternBeats, double cursorBeat, double beatsPerBar) {
39 const double length = std::max(patternBeats, 0.25);
40 const double bar = std::max(beatsPerBar, 1.0);
41 const PlacedClip* last = nullptr;
42 for (const auto& c : clips)
43 if (c.patternId == patternId && (!last || c.endBeat > last->endBeat)) last = &c;
44
45 PlacementTarget out;
46 if (last) {
47 out.startBeat = last->endBeat;
48 } else {
49 double songEnd = 0.0;
50 for (const auto& c : clips) songEnd = std::max(songEnd, c.endBeat);
51 const double endBar = std::ceil(songEnd / bar - 1e-9) * bar;
52 const double cursorBar = std::floor(std::max(0.0, cursorBeat) / bar) * bar;
53 out.startBeat = std::max(endBar, cursorBar);
54 }
55
56 const auto freeOn = [&](int lane) {
57 for (const auto& c : clips)
58 if (c.lane == lane && c.startBeat < out.startBeat + length - 1e-9 && c.endBeat > out.startBeat + 1e-9) return false;
59 return true;
60 };
61 if (last && last->lane >= 0 && last->lane < laneCount && freeOn(last->lane)) { out.lane = last->lane; return out; }
62 for (int lane = 0; lane < laneCount; ++lane)
63 if (freeOn(lane)) { out.lane = lane; return out; }
64 out.lane = -1;
65 return out;
66}
