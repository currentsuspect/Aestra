37 bool push(const Event& ev) noexcept {
38 const uint32_t head = m_head.load(std::memory_order_relaxed);
39 const uint32_t next = (head + 1) & kMask;
40 if (next == m_tail.load(std::memory_order_acquire)) {
41 return false;
42 }
43 m_events[head] = ev;
44 m_head.store(next, std::memory_order_release);
45 return true;
46 }

49 bool pop(Event& out) noexcept {
50 const uint32_t tail = m_tail.load(std::memory_order_relaxed);
51 if (tail == m_head.load(std::memory_order_acquire)) {
52 return false;
53 }
54 out = m_events[tail];
55 m_tail.store((tail + 1) & kMask, std::memory_order_release);
56 return true;
57 }

59 // Public so consumers can bound their drain loops: a producer pushing
60 // concurrently can keep pop() succeeding past the queue's snapshot size,
61 // so "drain until empty" is not inherently bounded work.
62 static constexpr uint32_t kCapacity = 512; // power of two; ~= 5s of frantic playing
63
64private:
