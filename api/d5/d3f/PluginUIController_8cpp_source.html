90void PluginUIController::startScan() {
91 if (!m_scanner || !m_browser) return;
92
93 // Show scanning state
94 m_browser->setScanning(true, 0.0f);
95 m_browser->setScanStatus("Starting scan...");
96
97 // Start async scan
98 m_scanner->scanAsync(
99 // Progress callback
100 [this](const std::string& path, int current, int total) {
101 if (m_browser) {
102 float progress = total > 0 ? static_cast<float>(current) / total : 0.0f;
103 m_browser->setScanning(true, progress);
104
105 // Extract filename for status
106 auto filename = std::filesystem::path(path).filename().string();
107 m_browser->setScanStatus(filename);
108 }
109 },
110 // Completion callback
111 [this](const std::vector<Aestra::Audio::PluginInfo>& plugins, bool success) {
112 if (m_browser) {
113 m_browser->setScanning(false, 1.0f);
114 if (success) {
115 refreshBrowserList();
116 }
117 }
118
119 if (m_onScanComplete) {
120 m_onScanComplete(static_cast<int>(plugins.size()));
121 }
122 }
123 );
124}

126void PluginUIController::bindEffectRack(EffectChainRack* rack,
127 Aestra::Audio::EffectChain* chain) {
128 if (!rack || !chain) return;
129
130 // Store binding
131 m_rackBindings.push_back({rack, chain});
132
133 // Wire slot clicks
134 rack->setOnSlotClicked([this, chain](int slot) {
135 auto instance = chain->getPlugin(slot);
136 if (instance && (instance->hasEditor() || instance->getParameterCount() > 0)) {
137 openPluginEditor(instance, nullptr);
138 }
139 });
140
141 rack->setOnAddPluginRequested([this, rack, chain](int slot) {
142 // Remove existing menu if any
143 if (m_activeMenu) {
144 if (auto parent = m_activeMenu->getParent()) {
145 parent->removeChild(m_activeMenu);
146 }
147 m_activeMenu.reset();
148 }
149
150 // Create new dropdown
151 m_activeMenu = std::make_shared<UIMixerPluginDropdown>();
152
153 // Add to popup layer if available, otherwise fallback to rack's parent
154 if (m_popupLayer) {
155 m_popupLayer->addChild(std::static_pointer_cast<NUIComponent>(m_activeMenu));
156 } else if (auto parent = rack->getParent()) {
157 parent->addChild(std::static_pointer_cast<NUIComponent>(m_activeMenu));
158 }
159
160 // Use the actual slot's rendered bounds as the anchor (not manual arithmetic).
161 NUIRect slotBounds = rack->getSlotBounds(slot);
162
163 constexpr float DROP_W = 240.0f;
164 float dropX = slotBounds.x;
165 // Inspector rack is on the right side — flip dropdown to its left so it
166 // never overlaps the inspector panel.
167 if (m_popupLayer) {
168 auto layerBounds = m_popupLayer->getBounds();
169 if (dropX > layerBounds.width * 0.5f) {
170 dropX = std::max(4.0f, slotBounds.x - DROP_W - 4.0f);
171 }
172 }
173
174 NUIRect triggerRect{dropX, slotBounds.y, DROP_W, slotBounds.height};
175
176 // Flip boundary = popup layer bottom (full window) so the dropdown always
177 // opens upward when it would otherwise run off-screen.
178 float flipBoundary = m_popupLayer ? m_popupLayer->getBounds().bottom()
179 : rack->getBounds().bottom();
180 m_activeMenu->showAt(triggerRect, flipBoundary);
181
182 // Handle selection
183 m_activeMenu->onPluginSelected = [this, chain, slot](const std::string& pluginId, const std::string&) {
184 loadPluginToSlot(pluginId, chain, slot);
185
186 // Close menu
187 if (m_activeMenu) {
188 if (auto parent = m_activeMenu->getParent()) {
189 parent->removeChild(m_activeMenu);
190 }
191 m_activeMenu.reset();
192 }
193 };
194
195 // Handle dismissal
196 m_activeMenu->onDismissed = [this]() {
197 if (m_activeMenu) {
198 if (auto parent = m_activeMenu->getParent()) {
199 parent->removeChild(m_activeMenu);
200 }
201 m_activeMenu.reset();
202 }
203 };
204 });
205
206 rack->setOnSlotBypassToggled([this, chain](int slot, bool bypassed) {
207 chain->setSlotBypassed(slot, bypassed);
208 if (!bypassed) {
209 if (auto plugin = chain->getPlugin(static_cast<size_t>(slot))) {
210 plugin->resetWatchdog();
211 if (!plugin->isActive()) {
212 plugin->activate();
213 }
214 }
215 }
216 effectChainChanged.emit();
217 });
218
219 // NOTE: Do NOT bind setOnSlotRemoveRequested here!
220 // UIMixerInspector already binds this callback and handles deletion via MixerViewModel.
221 // Binding it here would OVERWRITE that handler, causing the ViewModel to never know about deletions.
222 // rack->setOnSlotRemoveRequested([this, rack, chain](int slot) {
223 // removePluginFromSlot(chain, slot);
224 // refreshRackDisplay(rack);
225 // });
226
227 // Initial refresh
228 refreshRackDisplay(rack);
229}

231void PluginUIController::unbindEffectRack(EffectChainRack* rack) {
232 if (!rack) return;
233
234 rack->setOnSlotClicked(nullptr);
235 rack->setOnAddPluginRequested(nullptr);
236 rack->setOnSlotBypassToggled(nullptr);
237 rack->setOnSlotRemoveRequested(nullptr);
238
239 // Remove from bindings
240 m_rackBindings.erase(
241 std::remove_if(m_rackBindings.begin(), m_rackBindings.end(),
242 [rack](const RackBinding& b) { return b.rack == rack; }),
243 m_rackBindings.end());
244}

246void PluginUIController::refreshRackDisplay(EffectChainRack* rack) {
247 if (!rack) return;
248
249 // Find chain binding
250 Aestra::Audio::EffectChain* chain = nullptr;
251 for (const auto& binding : m_rackBindings) {
252 if (binding.rack == rack) {
253 chain = binding.chain;
254 break;
255 }
256 }
257
258 if (!chain) return;
259
260 // Update slot display
261 for (int i = 0; i < EffectChainRack::MAX_SLOTS; ++i) {
262 auto instance = chain->getPlugin(i);
263 EffectChainRack::EffectSlotInfo info;
264
265 if (instance) {
266 info.name = instance->getInfo().name;
267 info.isEmpty = false;
268 info.bypassed = chain->isSlotBypassed(i);
269 info.nonFiniteOutputFault = chain->isSlotBypassedByNonFiniteOutput(i);
270 } else {
271 info.name = "Empty";
272 info.isEmpty = true;
273 info.bypassed = false;
274 info.nonFiniteOutputFault = false;
275 }
276
277 rack->setSlot(i, info);
278 }

281bool PluginUIController::loadPluginToSlot(const std::string& pluginId,
282 Aestra::Audio::EffectChain* chain,
283 int slot) {
284 if (!m_manager || !chain) return false;
285
286 // Create instance via manager by ID
287 auto instance = m_manager->createInstanceById(pluginId);
288 if (!instance) return false;
289
290 // Initialize with current manager defaults.
291 double sampleRate = m_manager->getDefaultSampleRate();
292 if (sampleRate <= 0.0) sampleRate = 44100.0;
293 uint32_t blockSize = m_manager->getDefaultBlockSize();
294 if (blockSize == 0) blockSize = 512;
295 if (!instance->initialize(sampleRate, blockSize)) {
296 return false;
297 }
298
299 // CRITICAL: Activate plugin so it can process audio
300 // Without this, isActive() returns false and EffectChain skips it
301 instance->activate();
302
303 // Insert into chain
304 chain->insertPlugin(slot, instance);
305
306 pluginLoaded.emit({pluginId, slot});
307 effectChainChanged.emit();
308
309 // Refresh any bound rack
310 for (const auto& binding : m_rackBindings) {
311 if (binding.chain == chain) {
312 refreshRackDisplay(binding.rack);
313 }
314 }
315
316 return true;
317}

326void PluginUIController::openPluginEditor(
327 std::shared_ptr<Aestra::Audio::IPluginInstance> instance,
328 void* parentWindow) {
329
330 if (!instance) return;
331
332 std::shared_ptr<NUIComponent> editor;
333 const std::string& pluginId = instance->getInfo().id;
334
335 if (false) {

340 if (m_popupLayer) m_popupLayer->removeChild(ed);
341 m_activeEditors.erase(std::remove(m_activeEditors.begin(), m_activeEditors.end(), ed), m_activeEditors.end());
342 });
343 editor = ed;
344#endif
345 } else if (pluginId == "com.Aestrastudios.eq") {
346 auto ed = std::make_shared<AestraEQEditor>(instance);
347 ed->setOnClose([this, ed]() {
348 if (m_popupLayer) m_popupLayer->removeChild(ed);
349 m_activeEditors.erase(std::remove(m_activeEditors.begin(), m_activeEditors.end(), ed), m_activeEditors.end());
350 });
351 ed->setPlatformBridge(m_platformBridge);
352 editor = ed;
353 } else if (pluginId == "com.Aestrastudios.comp") {
354 auto ed = std::make_shared<AestraCompEditor>(instance);
355 ed->setOnClose([this, ed]() {
356 if (m_popupLayer) m_popupLayer->removeChild(ed);
357 m_activeEditors.erase(std::remove(m_activeEditors.begin(), m_activeEditors.end(), ed), m_activeEditors.end());
358 });
359 ed->setPlatformBridge(m_platformBridge);
360 editor = ed;
361 } else if (pluginId == "com.Aestrastudios.verb") {
362 auto ed = std::make_shared<AestraVerbEditor>(instance);
363 ed->setOnClose([this, ed]() {
364 if (m_popupLayer) m_popupLayer->removeChild(ed);
365 m_activeEditors.erase(std::remove(m_activeEditors.begin(), m_activeEditors.end(), ed), m_activeEditors.end());
366 });
367 ed->setPlatformBridge(m_platformBridge);
368 editor = ed;
369 } else if (pluginId == "com.Aestrastudios.delay") {
370 auto ed = std::make_shared<AestraDelayEditor>(instance);
371 ed->setOnClose([this, ed]() {
372 if (m_popupLayer) m_popupLayer->removeChild(ed);
373 m_activeEditors.erase(std::remove(m_activeEditors.begin(), m_activeEditors.end(), ed), m_activeEditors.end());
374 });
375 ed->setPlatformBridge(m_platformBridge);
376 editor = ed;
377 } else if (pluginId == "com.Aestrastudios.drift") {
378 auto ed = std::make_shared<AestraDriftEditor>(instance);
379 ed->setOnClose([this, ed]() {
380 if (m_popupLayer) m_popupLayer->removeChild(ed);
381 m_activeEditors.erase(std::remove(m_activeEditors.begin(), m_activeEditors.end(), ed), m_activeEditors.end());
382 });
383 ed->setPlatformBridge(m_platformBridge);
384 editor = ed;
385 } else {
386 auto ed = std::make_shared<GenericPluginEditor>(instance);
387 ed->setOnClose([this, ed]() {
388 if (m_popupLayer) m_popupLayer->removeChild(ed);
389 m_activeEditors.erase(std::remove(m_activeEditors.begin(), m_activeEditors.end(), ed), m_activeEditors.end());
390 });
391 editor = ed;
392 }
393
394 auto relayoutEditor = [&](const std::shared_ptr<NUIComponent>& editorComp,
395 float width,
396 float height) {
397 if (!editorComp) return;
398
399 if (auto delay = std::dynamic_pointer_cast<AestraDelayEditor>(editorComp)) {
400 delay->onResize();
401 } else if (auto drift = std::dynamic_pointer_cast<AestraDriftEditor>(editorComp)) {
402 drift->onResize();
403 } else if (auto eq = std::dynamic_pointer_cast<AestraEQEditor>(editorComp)) {
404 eq->onResize();
405 } else if (auto verb = std::dynamic_pointer_cast<AestraVerbEditor>(editorComp)) {
406 verb->onResize();
407 } else if (auto comp = std::dynamic_pointer_cast<AestraCompEditor>(editorComp)) {
408 comp->onResize();
409 } else if (auto generic = std::dynamic_pointer_cast<GenericPluginEditor>(editorComp)) {
410 generic->onResize();
411#ifdef AESTRAUI_ENABLE_PREMIUM_EDITORS
412 } else if (auto rumble = std::dynamic_pointer_cast<RumblePluginEditor>(editorComp)) {
413 rumble->onResize(static_cast<int>(width), static_cast<int>(height));
414#endif
415 }
416 };
417
418 // Position editor in center of popup layer
419 if (m_popupLayer && editor) {
420 auto layerBounds = m_popupLayer->getBounds();
421 auto editorBounds = editor->getBounds();
422 float editorWidth = editorBounds.width > 0.0f ? editorBounds.width : 0.0f;
423 float editorHeight = editorBounds.height > 0.0f ? editorBounds.height : 0.0f;
424
425 if (editorWidth <= 0.0f || editorHeight <= 0.0f) {
426 auto [preferredWidth, preferredHeight] = instance->getEditorSize();
427 editorWidth = preferredWidth > 0 ? static_cast<float>(preferredWidth) : 400.0f;
428 editorHeight = preferredHeight > 0 ? static_cast<float>(preferredHeight) : 400.0f;
429 }
430
431 editorWidth = std::min(editorWidth, layerBounds.width - 40.0f);
432 editorHeight = std::min(editorHeight, layerBounds.height - 40.0f);
433 float x = (layerBounds.width - editorWidth) * 0.5f;
434 float y = (layerBounds.height - editorHeight) * 0.5f;
435
436 editor->setBounds(x, y, editorWidth, editorHeight);
437 relayoutEditor(editor, editorWidth, editorHeight);
438
439 m_popupLayer->addChild(editor);
440 m_activeEditors.push_back(editor);
441 }
442}

454PluginListItem PluginUIController::convertToListItem(
455 const Aestra::Audio::PluginInfo& info) const {
456
457 PluginListItem item;
458 item.id = info.id;
459 item.name = info.name;
460 item.vendor = info.vendor;
461 item.version = info.version;
462 item.category = info.category;
463
464 // Format string
465 switch (info.format) {
466 case Aestra::Audio::PluginFormat::VST3:
467 item.formatStr = "VST3";
468 break;
469 case Aestra::Audio::PluginFormat::CLAP:
470 item.formatStr = "CLAP (Exp.)";
471 break;
472 default:
473 item.formatStr = "Int";
474 break;
475 }
476
477 // Type name
478 switch (info.type) {
479 case Aestra::Audio::PluginType::Effect:
480 item.typeName = "Effect";
481 break;
482 case Aestra::Audio::PluginType::Instrument:
483 item.typeName = "Instrument";
484 break;
485 case Aestra::Audio::PluginType::MidiEffect:
486 item.typeName = "MIDI";
487 break;
488 case Aestra::Audio::PluginType::Analyzer:
489 item.typeName = "Analyzer";
490 break;
491 }
492
493 item.isFavorite = false;
494 return item;
495}

542bool PluginEditorWindow::open(
543 std::shared_ptr<Aestra::Audio::IPluginInstance> instance,
544 const std::string& title) {
545
546 if (!instance || !instance->hasEditor()) {
547 return false;
548 }
549
550 if (m_isOpen) {
551 close();
552 }
553
554 m_instance = instance;
555
556 // Get preferred editor size
557 auto [width, height] = instance->getEditorSize();
558 if (width <= 0) width = 800;
559 if (height <= 0) height = 600;
560
561#ifdef _WIN32
562 // Register window class (once)
563 static bool classRegistered = false;
564 static const wchar_t* CLASS_NAME = L"AestraPluginEditor";
565
566 if (!classRegistered) {
567 WNDCLASSEXW wc = {};
568 wc.cbSize = sizeof(wc);
569 wc.style = CS_HREDRAW | CS_VREDRAW;
570 wc.lpfnWndProc = Impl::WindowProc;
571 wc.hInstance = GetModuleHandle(nullptr);
572 wc.hCursor = LoadCursor(nullptr, IDC_ARROW);
573 wc.hbrBackground = reinterpret_cast<HBRUSH>(COLOR_WINDOW + 1);
574 wc.lpszClassName = CLASS_NAME;
575 RegisterClassExW(&wc);
576 classRegistered = true;
577 }
578
579 // Convert title to wide string
580 int titleLen = MultiByteToWideChar(CP_UTF8, 0, title.c_str(), -1, nullptr, 0);
581 std::wstring wideTitle(titleLen, 0);
582 MultiByteToWideChar(CP_UTF8, 0, title.c_str(), -1, &wideTitle[0], titleLen);
583
584 // Create window
585 RECT rect = {0, 0, width, height};
586 AdjustWindowRect(&rect, WS_OVERLAPPEDWINDOW, FALSE);
587
588 m_impl->hwnd = CreateWindowExW(
589 0,
590 CLASS_NAME,
591 wideTitle.c_str(),
592 WS_OVERLAPPEDWINDOW,
593 m_impl->posX, m_impl->posY,
594 rect.right - rect.left, rect.bottom - rect.top,
595 nullptr,
596 nullptr,
597 GetModuleHandle(nullptr),
598 nullptr
599 );
600
601 if (!m_impl->hwnd) {
602 return false;
603 }
604
605 // Store this pointer
606 SetWindowLongPtr(m_impl->hwnd, GWLP_USERDATA, reinterpret_cast<LONG_PTR>(this));
607
608 // Open plugin editor into window
609 if (!instance->openEditor(m_impl->hwnd)) {
610 DestroyWindow(m_impl->hwnd);
611 m_impl->hwnd = nullptr;
612 return false;
613 }
614
615 // Show window
616 ShowWindow(m_impl->hwnd, SW_SHOW);
617 UpdateWindow(m_impl->hwnd);
618
619 m_isOpen = true;
620 return true;
621#else
622 // Non-Windows: stub implementation
623 m_isOpen = true;
624 return instance->openEditor(nullptr);
625#endif
626}
