31void writeFileAtomically(const std::string& path, const std::string& text) {
32 std::error_code ec;
33 const std::filesystem::path target(path);
34 std::filesystem::create_directories(target.parent_path(), ec);
35 const std::filesystem::path temp = target.string() + ".tmp";
36 {
37 std::ofstream out(temp, std::ios::binary | std::ios::trunc);
38 out << text;
39 }
40 std::filesystem::permissions(temp, std::filesystem::perms::owner_read | std::filesystem::perms::owner_write,
41 std::filesystem::perm_options::replace, ec);
42 std::filesystem::rename(temp, target, ec);
43 if (ec) Log::warning("[Update] could not save " + path + ": " + ec.message());
44}

91// What happened to the last update, told once, and the old version's backup cleared only at the start AFTER the new
92// version has run for a whole session.
93void UpdateNotifier::handleLastUpdate() {
94 namespace fs = std::filesystem;
95 const fs::path dir = dataDir();
96 if (dir.empty()) return;
97 const fs::path fresh = dir / "last-update.json";
98 const fs::path seen = dir / "last-update.seen.json";
99 std::error_code ec;
100 const std::string exe = currentExecutablePath();
101 const fs::path appDir = exe.empty() ? fs::path() : fs::path(exe).parent_path();
102
103 if (fs::exists(fresh, ec)) {
104 std::string text = readFile(fresh.string());
105 bool ok = false, rolledBack = false;
106 std::string to, message;
107 try {
108 const JSON j = JSON::parse(text);
109 ok = j["ok"].isBool() && j["ok"].asBool();
110 rolledBack = j["rolledBack"].isBool() && j["rolledBack"].asBool();
111 if (j["to"].isString()) to = j["to"].asString();
112 if (j["message"].isString()) message = j["message"].asString();
113 } catch (...) {
114 fs::remove(fresh, ec);
115 return;
116 }
117 Log::info(std::string("[Update] last update: ") + (ok ? "installed " : "did not install ") + to + " (" + message + ")");
118 if (m_hooks.showMessage) {
119 if (ok) m_hooks.showMessage("Aestra was updated to " + to + ".");
120 else m_hooks.showMessage("The update to " + to + " did not work" + (rolledBack ? " and was undone" : "") + ": " + message + ".");
121 }
122 fs::rename(fresh, seen, ec);
123 return;
124 }
125 if (fs::exists(seen, ec) && !appDir.empty()) {
126 // One whole session has passed since the update: the new version is the one to keep.
127 const std::string text = readFile(seen.string());
128 bool ok = false;
129 try {
130 const JSON j = JSON::parse(text);
131 ok = j["ok"].isBool() && j["ok"].asBool();
132 } catch (...) {
133 }
134 if (ok) fs::remove_all(appDir.string() + ".old", ec);
135 fs::remove(seen, ec);
136 }
137}

139UpdateLayout UpdateNotifier::makeLayout(std::string* error) const {
140 namespace fs = std::filesystem;
141 UpdateLayout layout;
142 const std::string exe = currentExecutablePath();
143 if (exe.empty() || dataDir().empty()) {
144 if (error) *error = "Aestra cannot tell where it is installed, so it cannot update itself.";
145 return layout;
146 }
147 const fs::path exePath(exe);
148 layout.appDir = exePath.parent_path().string();
149 layout.executableName = exePath.filename().string();
150 layout.updaterName = std::string("AestraUpdater") + (exePath.extension() == ".exe" ? ".exe" : "");
151 layout.downloadDir = (dataDir() / "updates-download").string();
152 return layout;
153}

155void UpdateNotifier::startInstall() {
156 const UpdateCheckResult result = lastResult();
157 if (result.status != UpdateCheckStatus::Available) return;
158 m_localFailure.clear();
159 std::string error;
160 const UpdateLayout layout = makeLayout(&error);
161 if (layout.appDir.empty()) {
162 m_localFailure = error;
163 return;
164 }
165 if (!m_session) m_session = std::make_unique<UpdateSession>(createDefaultHttpTransport());
166 m_layout = layout;
167 m_installingVersion = result.manifest.version;
168 m_restartRequested = false;
169 m_session->start(makeConfig(), result.manifest, layout);
170}

192void UpdateNotifier::applyPendingRestart() {
193 if (!m_restartRequested || !m_session || m_session->state().phase != InstallPhase::Ready) return;
194 m_restartRequested = false;
195 std::string error;
196 const std::filesystem::path result = dataDir() / "last-update.json";
197 if (!launchUpdater(m_layout, result.string(), currentVersion(), m_installingVersion, currentProcessId(), &error))
198 Log::error("[Update] could not hand over to the updater: " + error);
199 else
200 Log::info("[Update] handed over to the updater for " + m_installingVersion);
201}

219void UpdateNotifier::onResult(const UpdateCheckResult& result) {
220 // Never the key, never the feed body: the status and one line.
221 Log::info("[Update] " + std::string(result.status == UpdateCheckStatus::Available ? "available" : "no update") + ": " + result.detail);
222 if (!shouldAnnounceUpdate(result.status, result.manifest.version, settings()) || !m_hooks.postToMain) return;
223
224 const std::string version = result.manifest.version;
225 const std::weak_ptr<bool> alive = m_alive;
226 m_hooks.postToMain([this, version, alive]() {
227 if (alive.expired()) return;
228 UpdateSettings s = settings();
229 if (!shouldAnnounceUpdate(UpdateCheckStatus::Available, version, s)) return; // announced meanwhile
230 s.lastNotified = version;
231 setSettings(s);
232 if (m_hooks.showMessage) m_hooks.showMessage("Aestra " + version + " is available. Settings > Updates has the details.");
233 });
234}
