41AestraUI::DropFeedback TrackManagerUI::onDragEnter(const AestraUI::DragData& data, const AestraUI::NUIPoint& position) {
42 Log::info("[TrackManagerUI] Drag entered");
43
44 // Accept source content, patterns, and MIDI clips. Effects belong to the
45 // mixer and never imply a Playlist-lane/mixer-insert association. (Timeline clip
46 // moves are an internal mouse drag, not a DragData transfer — see
47 // m_draggedClipId.)
48 if (data.type != AestraUI::DragDataType::File && data.type != AestraUI::DragDataType::MidiClip &&
49 data.type != AestraUI::DragDataType::Pattern) {
50 return AestraUI::DropFeedback::Invalid;
51 }
52
53 // Early reject unsupported formats (cheap extension check; full validation happens on drop).
54 if (data.type == AestraUI::DragDataType::File && !AudioFileValidator::hasValidAudioExtension(data.filePath)) {
55 m_showDropPreview = false;
56 setDirty(true);
57 return AestraUI::DropFeedback::Invalid;
58 }
59
60 // Calculate target track and time
61 m_dropTargetTrack = getTrackAtPosition(position.y);
62 m_dropTargetTime = getTimeAtPosition(position.x);
63
64 // Allow dropping on existing tracks OR appending a new track
65 int trackCount = static_cast<int>(m_trackManager->getTrackCount());
66
67 // If dragging below last track, target the next available slot
68 if (m_dropTargetTrack >= trackCount) {
69 m_dropTargetTrack = trackCount;
70 }
71
72 if (m_dropTargetTrack >= 0 && m_dropTargetTrack <= trackCount) {
73 m_showDropPreview = true;
74 setDirty(true);
75 return AestraUI::DropFeedback::Copy;
76 }
77
78 return AestraUI::DropFeedback::Invalid;
79}

81AestraUI::DropFeedback TrackManagerUI::onDragOver(const AestraUI::DragData& data, const AestraUI::NUIPoint& position) {
82 // Same payload allowlist as onDragEnter. Without it, dragging an effect across
83 // the timeline showed Copy feedback for the whole traverse — onDragEnter refused
84 // it and onDrop refuses it, but the cursor said it would land.
85 if (data.type != AestraUI::DragDataType::File && data.type != AestraUI::DragDataType::MidiClip &&
86 data.type != AestraUI::DragDataType::Pattern) {
87 if (m_showDropPreview) {
88 m_showDropPreview = false;
89 setDirty(true);
90 }
91 return AestraUI::DropFeedback::Invalid;
92 }
93
94 // Keep feedback "Invalid" for unsupported formats while hovering.
95 if (data.type == AestraUI::DragDataType::File && !AudioFileValidator::hasValidAudioExtension(data.filePath)) {
96 if (m_showDropPreview) {
97 m_showDropPreview = false;
98 setDirty(true);
99 }
100 return AestraUI::DropFeedback::Invalid;
101 }
102
103 // Update target track and time as mouse moves
104 int newTrack = getTrackAtPosition(position.y);
105
106 // Explicit mapping: Workspace -> Grid -> Beat
107 auto& theme = AestraUI::NUIThemeManager::getInstance();
108 float controlWidth = theme.getLayoutDimensions().trackControlsWidth;
109 float gridStartX = getBounds().x + controlWidth + kTimelineGridInsetX;
110
111 // REJECTION: If dropping on the control area
112 if (position.x < gridStartX) {
113 if (m_showDropPreview) {
114 m_showDropPreview = false;
115 setDirty(true);
116 Log::info("[TrackManagerUI] Drag over rejected: Cursor in control area");
117 }
118 return AestraUI::DropFeedback::Invalid;
119 }
120
121 double rawTimeBeats = gridOffsetToBeat(position.x - gridStartX);
122 double snappedBeats = snapBeatToGrid(rawTimeBeats);
123 double newTime = m_trackManager->getPlaylistModel().beatToSeconds(snappedBeats);
124
125 int trackCount = static_cast<int>(m_trackManager->getTrackCount());
126
127 // If dragging below last track, target the next available slot
128 if (newTrack >= trackCount) {
129 newTrack = trackCount;
130 }
131
132 // Only update if changed (performance optimization)
133 if (newTrack != m_dropTargetTrack || std::abs(newTime - m_dropTargetTime) > 0.001) {
134 m_dropTargetTrack = newTrack;
135 m_dropTargetTime = std::max(0.0, newTime); // Don't allow negative time
136
137 if (m_dropTargetTrack >= 0 && m_dropTargetTrack <= trackCount) {
138 m_showDropPreview = true;
139 setDirty(true);
140 return AestraUI::DropFeedback::Copy;
141 } else {
142 m_showDropPreview = false;
143 setDirty(true);
144 return AestraUI::DropFeedback::Invalid;
145 }
146 }
147
148 // Return appropriate feedback based on preview state
149 if (m_showDropPreview) {
150 return AestraUI::DropFeedback::Copy;
151 }
152 return AestraUI::DropFeedback::Invalid;
153}

161AestraUI::DropResult TrackManagerUI::onDrop(const AestraUI::DragData& data, const AestraUI::NUIPoint& position) {
162 AestraUI::DropResult result;
163
164 // 1. Calculate drop location
165 int laneIndex = getTrackAtPosition(position.y);
166 double rawTimeSeconds = std::max(0.0, getTimeAtPosition(position.x));
167
168 // v3.0: We work strictly in beats for arrangement
169 double rawTimeBeats = m_trackManager->getPlaylistModel().secondsToBeats(rawTimeSeconds);
170
171 // Snap-to-grid logic (Canonical Beat-Space)
172 double timePositionBeats = snapBeatToGrid(rawTimeBeats);
173
174 auto& playlist = m_trackManager->getPlaylistModel();
175 size_t laneCount = playlist.getLaneCount();
176
177 Log::info("[TrackManagerUI] onDrop: position.y=" + std::to_string(position.y) +
178 ", laneIndex=" + std::to_string(laneIndex) + ", laneCount=" + std::to_string(laneCount));
179
180 if (laneIndex < 0 || laneIndex > static_cast<int>(laneCount)) {
181 result.accepted = false;
182 result.message = "Invalid track position";
183 clearDropPreview();
184 return result;
185 }
186
187 // Set target track index for logging
188 result.targetTrackIndex = laneIndex;
189
190 // Reject unsupported payloads before appending a lane.
191 if (data.type == AestraUI::DragDataType::Plugin) {
192 result.accepted = false;
193 result.message = "Drop effects on a mixer insert";
194 clearDropPreview();
195 return result;
196 }
197 if (data.type == AestraUI::DragDataType::MidiClip) {
198 result.accepted = false;
199 result.message = "MIDI track creation not yet implemented";
200 clearDropPreview();
201 return result;
202 }
203 if (data.type == AestraUI::DragDataType::File && !AudioFileValidator::isValidAudioFile(data.filePath)) {
204 result.accepted = false;
205 result.message = "Unsupported file format";
206 clearDropPreview();
207 return result;
208 }
209
210 // 2. Resolve target lane
211 //
212 // A drop that has to append a lane must undo as ONE step: the lane is part of
213 // the edit, not scaffolding around it. Creating it through CreateLaneCommand
214 // rather than calling playlist.createLane() directly is what lets it join the
215 // clip in a single transaction — otherwise Ctrl+Z removes the clip and leaves
216 // an empty orphan lane the user never asked for and cannot undo away.
217 PlaylistLaneID targetLaneId;
218 std::shared_ptr<CreateLaneCommand> laneCommand;
219 if (laneIndex == static_cast<int>(laneCount)) {
220 laneCommand = std::make_shared<CreateLaneCommand>(playlist, "Track " + std::to_string(laneIndex + 1));
221 laneCommand->execute();
222 targetLaneId = laneCommand->getLaneId();
223 if (!targetLaneId.isValid()) {
224 laneCommand.reset();
225 result.accepted = false;
226 result.message = "Could not create track";
227 clearDropPreview();
228 return result;
229 }
230
231 Log::info("[TrackManagerUI] Created new lane " + std::to_string(laneIndex) + " for drop.");
232 } else {
233 targetLaneId = playlist.getLaneId(laneIndex);
234 }
235
236 // Undo the appended lane without recording anything: a drop that failed is not
237 // history. Reused lanes are left alone.
238 auto rollbackLane = [&laneCommand]() {
239 if (laneCommand) {
240 laneCommand->undo();
241 }
242 };
243
244 // Adopt an already-executed clip command, plus the lane it needed, as one undo
245 // step. The members were executed stepwise — each validated against the state
246 // its predecessor produced — so the transaction is adopted with
247 // markExecuted()/pushExecuted() rather than replayed by the history.
248 auto commitDrop = [this, &laneCommand](const std::string& name,
249 const std::shared_ptr<ICommand>& clipCommand) {
250 auto transaction = std::make_shared<CommandTransaction>(name);
251 if (laneCommand) {
252 transaction->add(laneCommand);
253 }
254 transaction->add(clipCommand);
255 transaction->markExecuted();
256 return m_trackManager->getCommandHistory().pushExecuted(transaction);
257 };
258
259 // 3. Handle Pattern Drop
260 if (data.type == AestraUI::DragDataType::Pattern) {
261 PatternID pid;
262
263 // Try to extract PatternID from customData
264 try {
265 if (data.customData.has_value()) {
266 pid = std::any_cast<PatternID>(data.customData);
267 }
268 } catch (const std::bad_any_cast&) {
269 Log::error("[TrackManagerUI] Failed to cast pattern ID from drag data");
270 }
271
272 if (pid.isValid()) {
273 auto pattern = m_trackManager->getPatternManager().getPattern(pid);
274 if (pattern) {
275 double duration = pattern->lengthBeats;
276 // Create clip instance manually and use command for undo support
277 ClipInstance clip;
278 clip.id = ClipInstanceID::generate();
279 clip.startBeat = timePositionBeats;
280 clip.durationBeats = duration;
281 clip.patternId = pid;
282 clip.sourceId = pid.value;
283 if (pattern->isAudio()) {
284 clip.edits = ClipEdits::forNewAudioClip();
285 }
286
287 auto cmd = std::make_shared<AddClipCommand>(playlist, targetLaneId, clip);
288 cmd->execute();
289
290 if (!playlist.getClip(clip.id)) {
291 rollbackLane();
292 result.accepted = false;
293 result.message = "Could not place pattern";
294 clearDropPreview();
295 return result;
296 }
297
298 commitDrop("Add Pattern Clip", cmd);
299
300 result.accepted = true;
301 result.message = "Pattern added: " + pattern->name;
302 Log::info("[TrackManagerUI] Pattern added to timeline: " + pattern->name);
303
304 refreshTracks();
305 invalidateCache();
306 scheduleTimelineMinimapRebuild();
307 } else {
308 rollbackLane();
309 result.accepted = false;
310 result.message = "Pattern not found";
311 }
312 } else {
313 rollbackLane();
314 result.accepted = false;
315 result.message = "Invalid pattern ID";
316 }
317 clearDropPreview();
318 return result;
319 }
320
321 // 4. Handle File Drop (New Audio Content)
322 if (data.type == AestraUI::DragDataType::File) {
323 Log::info("[TrackManagerUI] File drop received: " + data.filePath);
324
325 // Register file with SourceManager
326 auto& sourceManager = m_trackManager->getSourceManager();
327 ClipSourceID sourceId = sourceManager.getOrCreateSource(data.filePath);
328 ClipSource* source = sourceManager.getSource(sourceId);
329
330 if (source) {
331 std::weak_ptr<AestraUI::NUIComponent> weakSelf = weak_from_this();
332
333 // Helper to create clip once source is ready. Async decode and queued tasks capture only weakSelf so
334 // they cannot call back into a destroyed TrackManagerUI.
335 //
336 // Returns whether a clip was actually placed. The already-decoded branch
337 // below calls this synchronously and used to report success no matter what
338 // happened inside — a file whose duration decoded as zero, or whose pattern
339 // could not be created, still told the user "Imported".
340 auto createClipFromSource = [weakSelf, sourceId, displayName = data.displayName, targetLaneId,
341 laneCommand, timePositionBeats]() -> bool {
342 auto self = std::dynamic_pointer_cast<TrackManagerUI>(weakSelf.lock());
343 if (!self || !self->m_trackManager)
344 return false;
345
346 auto& sourceManager = self->m_trackManager->getSourceManager();
347 ClipSource* source = sourceManager.getSource(sourceId);
348
349 // Remove from pending imports (animation cleanup)
350 auto it = std::remove_if(
351 self->m_pendingImports.begin(), self->m_pendingImports.end(), [&](const PendingImport& pi) {
352 return pi.laneId == targetLaneId && std::abs(pi.startBeat - timePositionBeats) < 0.001 &&
353 pi.displayName == displayName;
354 });
355 if (it != self->m_pendingImports.end()) {
356 self->m_pendingImports.erase(it, self->m_pendingImports.end());
357 self->setDirty(true);
358 }
359
360 if (self->m_onClipLibraryChanged) {
361 self->m_onClipLibraryChanged();
362 }
363
364 auto rollbackCreatedLane = [&]() {
365 if (!laneCommand) {
366 return;
367 }
368 laneCommand->undo();
369 self->refreshTracks();
370 self->invalidateCache();
371 self->scheduleTimelineMinimapRebuild();
372 };
373
374 // Same one-undo-step contract as the synchronous paths: the lane the
375 // import had to append belongs to the import, so it is adopted into the
376 // transaction alongside the clip rather than left outside history.
377 auto commitImport = [&](const std::shared_ptr<ICommand>& clipCommand) {
378 auto transaction = std::make_shared<CommandTransaction>("Import Audio Clip");
379 if (laneCommand) {
380 transaction->add(laneCommand);
381 }
382 transaction->add(clipCommand);
383 transaction->markExecuted();
384 self->m_trackManager->getCommandHistory().pushExecuted(transaction);
385 };
386
387 if (!source || !source->isReady()) {
388 rollbackCreatedLane();
389 return false;
390 }
391
392 double durationSeconds = source->getDurationSeconds();
393 double durationBeats = self->secondsToBeats(durationSeconds);
394 if (!std::isfinite(durationSeconds) || durationSeconds <= 0.0 ||
395 !std::isfinite(durationBeats) || durationBeats <= 0.0) {
396 rollbackCreatedLane();
397 Log::error("[TrackManagerUI] Invalid decoded duration for imported source");
398 return false;
399 }
400 Log::info("[TrackManagerUI] Duration: " + std::to_string(durationSeconds) +
401 "s, beats: " + std::to_string(durationBeats));
402
403 // Create Audio Pattern
404 AudioSlicePayload payload;
405 payload.audioSourceId = sourceId;
406 payload.durationSeconds = durationSeconds;
407 // Populate the sample-domain fields the serializer persists
408 // (startSamples/lengthSamples). The old {0.0, numFrames} form set
409 // startOffset/duration instead, so the slice saved as start:0 length:0.
410 AudioSlice fullSlice;
411 fullSlice.startSamples = 0.0;
412 fullSlice.lengthSamples = static_cast<double>(source->getNumFrames());
413 payload.slices.push_back(fullSlice);
414
415 auto& patternManager = self->m_trackManager->getPatternManager();
416 PatternID patternId = patternManager.createAudioPattern(displayName, durationBeats, payload);
417
418 if (patternId.isValid()) {
419 auto& playlist = self->m_trackManager->getPlaylistModel();
420
421 // Create clip instance manually and use command for undo support
422 ClipInstance clip;
423 clip.id = ClipInstanceID::generate();
424 clip.startBeat = timePositionBeats;
425 clip.durationBeats = durationBeats;
426 clip.durationSeconds = durationSeconds;
427 clip.patternId = patternId;
428 clip.sourceId = patternId.value;
429 clip.edits = ClipEdits::forNewAudioClip();
430
431 auto cmd = std::make_shared<AddClipCommand>(playlist, targetLaneId, clip);
432 cmd->execute();
433 if (!playlist.getClip(clip.id)) {
434 patternManager.removePattern(patternId);
435 rollbackCreatedLane();
436 Log::error("[TrackManagerUI] Failed to add imported clip to target lane; removed orphan audio pattern");
437 return false;
438 }
439
440 commitImport(cmd);
441
442 self->refreshTracks();
443 self->invalidateCache();
444 self->scheduleTimelineMinimapRebuild();
445 Log::info("[TrackManagerUI] Clip added successfully via command");
446 return true;
447 }
448
449 rollbackCreatedLane();
450 Log::error("[TrackManagerUI] PatternManager::createAudioPattern failed");
451 return false;
452 };
453
454 if (!source->isReady()) {
455 Log::info("[TrackManagerUI] Decoding new source (ASYNC): " + data.filePath);
456
457 // Add to pending imports for visualizer
458 PendingImport pending;
459 pending.displayName = data.displayName;
460 pending.laneId = targetLaneId;
461 pending.startBeat = timePositionBeats;
462 m_pendingImports.push_back(pending);
463 setDirty(true);
464
465 // Capture necessary data for async thread
466 std::string filePath = data.filePath;
467 std::string displayName = data.displayName;
468 std::shared_ptr<TrackManager> tm = m_trackManager;
469
470 // Launch decoding in background thread to prevent UI freeze
471 std::thread([weakSelf, tm, sourceId, filePath, displayName, createClipFromSource, targetLaneId,
472 timePositionBeats]() {
473 std::vector<float> decodedData;
474 uint32_t sampleRate = 0;
475 uint32_t numChannels = 0;
476
477 auto lastUpdate =
478 std::make_shared<std::chrono::steady_clock::time_point>(std::chrono::steady_clock::now());
479
480 // Capture by value for the progress callback to avoid reference lifetime issues
481 auto progressCb = [weakSelf, targetLaneId, timePositionBeats, displayName, lastUpdate](float p) {
482 auto now = std::chrono::steady_clock::now();
483 if (std::chrono::duration_cast<std::chrono::milliseconds>(now - *lastUpdate).count() < 16 &&
484 p < 1.0f)
485 return;
486 *lastUpdate = now;
487
488 auto self = std::dynamic_pointer_cast<TrackManagerUI>(weakSelf.lock());
489 if (!self)
490 return;
491
492 std::lock_guard<std::mutex> lock(self->m_pendingTasksMutex);
493 self->m_pendingTasks.push_back([weakSelf, targetLaneId, timePositionBeats, displayName, p]() {
494 auto self = std::dynamic_pointer_cast<TrackManagerUI>(weakSelf.lock());
495 if (!self)
496 return;
497
498 for (auto& pi : self->m_pendingImports) {
499 if (pi.laneId == targetLaneId && std::abs(pi.startBeat - timePositionBeats) < 0.001 &&
500 pi.displayName == displayName) {
501 pi.progress = p;
502 self->setDirty(true);
503 break;
504 }
505 }
506 });
507 };
508
509 // Heavy IO/Decoding operation
510 bool success = decodeAudioFile(filePath, decodedData, sampleRate, numChannels, progressCb);
511
512 // Post completion task to main thread
513 if (auto self = std::dynamic_pointer_cast<TrackManagerUI>(weakSelf.lock())) {
514 std::lock_guard<std::mutex> lock(self->m_pendingTasksMutex);
515 self->m_pendingTasks.push_back([weakSelf, tm, sourceId, success,
516 decodedData = std::move(decodedData), sampleRate, numChannels,
517 displayName, createClipFromSource, filePath]() mutable {
518 auto self = std::dynamic_pointer_cast<TrackManagerUI>(weakSelf.lock());
519 if (!self)
520 return;
521
522 auto& sm = tm->getSourceManager();
523 ClipSource* src = sm.getSource(sourceId);
524 if (!src)
525 return;
526
527 if (success) {
528 auto buffer = std::make_shared<AudioBufferData>();
529 buffer->interleavedData = std::move(decodedData);
530 buffer->sampleRate = sampleRate;
531 buffer->numChannels = numChannels;
532 buffer->numFrames = buffer->interleavedData.size() / numChannels;
533 src->setBuffer(buffer);
534 const uint64_t sourceRevision = src->getContentRevision();
535
536 Log::info("[TrackManagerUI] Async load complete for: " + filePath);
537
538 // Trigger waveform cache build
539 self->m_waveformBuilder.buildAsync(
540 *src, [weakSelf, sourceId,
541 sourceRevision](std::shared_ptr<Aestra::Audio::WaveformCache> cache) {
542 if (cache) {
543 auto self = std::dynamic_pointer_cast<TrackManagerUI>(weakSelf.lock());
544 if (!self)
545 return;
546
547 std::lock_guard<std::mutex> lock(self->m_pendingTasksMutex);
548 self->m_pendingTasks.push_back([weakSelf, sourceId, sourceRevision, cache]() {
549 auto self = std::dynamic_pointer_cast<TrackManagerUI>(weakSelf.lock());
550 if (!self)
551 return;
552 if (!self->m_trackManager)
553 return;
554
555 auto* src = self->m_trackManager->getSourceManager().getSource(sourceId);
556 if (!src)
557 return;
558 if (src->getContentRevision() != sourceRevision)
559 return;
560
561 src->setWaveformCache(cache);
562 Log::info("Waveform cache ready for: " + src->getName());
563 self->invalidateCache();
564 self->m_backgroundNeedsUpdate = true;
565 self->setDirty(true);
566 });
567 }
568 });
569
570 // Queue clip creation to main thread via pending tasks
571 // This ensures UI updates happen on the main thread, not the background decode thread
572 std::lock_guard<std::mutex> lock(self->m_pendingTasksMutex);
573 self->m_pendingTasks.push_back([createClipFromSource]() { createClipFromSource(); });
574
575 } else {
576 Log::error("[TrackManagerUI] Failed to decode file async: " + filePath);
577 // Queue cleanup to main thread
578 std::lock_guard<std::mutex> lock(self->m_pendingTasksMutex);
579 self->m_pendingTasks.push_back([createClipFromSource]() {
580 createClipFromSource(); // Cleanup UI
581 });
582 }
583 });
584 }
585 }).detach();
586
587 result.accepted = true;
588 result.message = "Importing...";
589 } else {
590 // Already loaded, proceed immediately — and report what actually
591 // happened. createClipFromSource rolls back the appended lane on every
592 // failure path, so a false here means nothing was left behind either.
593 if (createClipFromSource()) {
594 result.accepted = true;
595 result.message = "Imported: " + data.displayName;
596 } else {
597 result.accepted = false;
598 result.message = "Could not import " + data.displayName;
599 }
600 }
601 } else {
602 rollbackLane();
603 result.accepted = false;
604 result.message = "Failed to create source";
605 }
606
607 clearDropPreview();
608 return result;
609 }
610
611 result.accepted = false;
612 result.message = "Unknown drop type";
613 clearDropPreview();
614 return result;
615}

623double TrackManagerUI::snapBeatToGrid(double beat) const {
624 if (!m_snapEnabled || m_snapSetting == AestraUI::SnapGrid::None) {
625 return beat;
626 }
627
628 double grid = AestraUI::MusicTheory::getSnapDuration(m_snapSetting);
629 if (grid <= 0.00001)
630 return beat;
631
632 // Round to nearest grid line
633 double snappedBeats = std::round(beat / grid) * grid;
634
635 return std::max(0.0, snappedBeats);
636}
637
638double TrackManagerUI::snapBeatToGridForward(double beat) const {
639 if (!m_snapEnabled || m_snapSetting == AestraUI::SnapGrid::None) {
640 return beat;
641 }
642
643 double grid = AestraUI::MusicTheory::getSnapDuration(m_snapSetting);
644 if (grid <= 0.00001)
645 return beat;
646
647 // Snap forward to next grid line
648 double snappedBeats = std::floor(beat / grid) * grid + grid;
649
650 return std::max(0.0, snappedBeats);
651}

657int TrackManagerUI::getTrackAtPosition(float y) const {
658 AestraUI::NUIRect bounds = getBounds();
659 auto& themeManager = AestraUI::NUIThemeManager::getInstance();
660
661 // Get ruler height and track area start
662 // MUST match renderTrackManagerDirect layout exactly:
663 // header(38) + horizontalScrollbar(24) + ruler(28)
664 float headerHeight = kTimelineHeaderHeight;
665 float horizontalScrollbarHeight = kTimelineHorizontalScrollbarHeight;
666 float rulerHeight = kTimelineRulerHeight;
667 float trackAreaY = bounds.y + headerHeight + horizontalScrollbarHeight + rulerHeight;
668
669 // Relative Y position in track area
670 float relativeY = y - trackAreaY + m_scrollOffset;
671
672 if (relativeY < 0) {
673 return -1; // Above track area
674 }
675
676 // Calculate track index based on track height + spacing
677 int trackIndex = static_cast<int>(relativeY / (m_trackHeight + m_trackSpacing));
678
679 return trackIndex;

682double TrackManagerUI::getTimeAtPosition(float x) const {
683 AestraUI::NUIRect bounds = getBounds();
684 auto& themeManager = AestraUI::NUIThemeManager::getInstance();
685
686 // Get control area width (where track buttons are)
687 float controlAreaWidth = themeManager.getLayoutDimensions().trackControlsWidth;
688 float gridStartX = controlAreaWidth + kTimelineGridInsetX;
689
690 // Same conversion the drag and zoom paths use; this one then clamps and
691 // converts to seconds, which is why it could not simply be called from them.
692 const double beats = gridOffsetToBeat(x - bounds.x - gridStartX);
693
694 if (beats < 0.0) {
695 return 0.0; // Before grid start
696 }
697
698 // beats / beatsPerMinute * 60 = seconds
699 double bpm = std::max(m_trackManager->getPlaylistModel().getBPM(), 1.0);
700 double seconds = (beats / bpm) * 60.0;
701
702 return seconds;
703}

707 return;
708 }
709
710 // Issue #50: Don't show skeleton preview for existing clip drags
711 // Instant clip drag provides real-time visual feedback by moving the actual clip
712 if (m_isDraggingClipInstant) {

716 AestraUI::NUIRect bounds = getBounds();
717 auto& themeManager = AestraUI::NUIThemeManager::getInstance();
718
719 // Calculate grid area
720 float controlAreaWidth = themeManager.getLayoutDimensions().trackControlsWidth;
721 float gridStartX = bounds.x + controlAreaWidth + kTimelineGridInsetX;

724 // layoutTracks uses: headerHeight(38) + horizontalScrollbarHeight(24) + rulerHeight(28)

733 double beats = (m_dropTargetTime * bpm) / 60.0;

736 // Draw subtle track highlight (just a hint)
737 AestraUI::NUIRect trackHighlight(gridStartX, trackY, bounds.width - controlAreaWidth - 20,
738 static_cast<float>(m_trackHeight));

740 renderer.fillRect(trackHighlight, highlightColor);
741
742 // Draw clip preview using the same visual language as placed timeline clips.
743 if (timeX >= gridStartX && timeX <= bounds.right() - 20) {
744 float previewWidth = 150.0f; // Reasonable preview width
745
746 AestraUI::NUIRect clipPreview(timeX,
747 trackY + 2, // Same as real clip: bounds.y + 2
748 previewWidth,
749 static_cast<float>(m_trackHeight) - 4 // Same as real clip: bounds.height - 4
750 );
751
752 const float clipRadius = 6.0f;
753 const float labelBarHeight = 14.0f;

758 renderer.strokeRoundedRect(clipPreview, clipRadius, 1.0f, clipBorder);

761 clipColor.lightened(0.18f).withAlpha(0.95f));

764 const AestraUI::NUIRect headerRect(clipPreview.x + 1.0f, clipPreview.y + 1.0f,
765 std::max(0.0f, clipPreview.width - 2.0f), headerH);

774 displayName = displayName.substr(0, 15) + "...";

777 const float labelTextY = headerRect.y + std::max(0.0f, (headerRect.height - clipLabelFont) * 0.5f);
778 AestraUI::NUIPoint textPos(clipPreview.x + 6.0f, labelTextY);
779 renderer.drawText(displayName, textPos, clipLabelFont, AestraUI::NUIThemeManager::getInstance().getCurrentTheme().textPrimary);
780 }

784void TrackManagerUI::renderDeleteAnimations(AestraUI::NUIRenderer& renderer) {
785 if (m_deleteAnimations.empty()) {
786 return;
787 }
788
789 // Update and render each animation
790 auto it = m_deleteAnimations.begin();
791 while (it != m_deleteAnimations.end()) {
792 DeleteAnimation& anim = *it;
793
794 // Update progress (assume ~60fps, so ~16ms per frame)
795 anim.progress += (1.0f / 60.0f) / anim.duration;
796
797 if (anim.progress >= 1.0f) {
798 // Animation complete, remove it
799 it = m_deleteAnimations.erase(it);
800 continue;
801 }
802
803 // Subtle red ripple expanding from click point
804
805 // Calculate ripple radius (smaller, more subtle)
806 float maxRadius = 50.0f;
807 float currentRadius = anim.progress * maxRadius;
808
809 // Ripple alpha fades out as it expands
810 float rippleAlpha = (1.0f - anim.progress) * 0.4f;
811
812 // Draw single subtle expanding ring
813 if (currentRadius > 0) {
814 AestraUI::NUIColor ringColor(1.0f, 0.3f, 0.3f, rippleAlpha);
815
816 // Draw a circle using lines
817 const int segments = 24;
818 for (int i = 0; i < segments; i++) {
819 float angle1 = (float)i / segments * 2.0f * 3.14159f;
820 float angle2 = (float)(i + 1) / segments * 2.0f * 3.14159f;
821
822 AestraUI::NUIPoint p1(anim.rippleCenter.x + std::cos(angle1) * currentRadius,
823 anim.rippleCenter.y + std::sin(angle1) * currentRadius);
824 AestraUI::NUIPoint p2(anim.rippleCenter.x + std::cos(angle2) * currentRadius,
825 anim.rippleCenter.y + std::sin(angle2) * currentRadius);
826
827 renderer.drawLine(p1, p2, 1.5f, ringColor);
828 }
829 }
830
831 // Force continuous redraw during animation
832 invalidateCache();
833
834 ++it;
835 }
836}
