16// Parses "#RRGGBB" or "#RRGGBBAA" (leading '#' required). Returns false on
17// malformed input without touching `out`.
18bool parseHexColor(const std::string& text, NUIColor& out) {
19 if (text.empty() || text[0] != '#') {
20 return false;
21 }
22 const std::string hex = text.substr(1);
23 if (hex.size() != 6 && hex.size() != 8) {
24 return false;
25 }
26 for (char c : hex) {
27 if (!std::isxdigit(static_cast<unsigned char>(c))) {
28 return false;
29 }
30 }
31 const auto rgb = static_cast<uint32_t>(std::stoul(hex.substr(0, 6), nullptr, 16));
32 float alpha = 1.0f;
33 if (hex.size() == 8) {
34 alpha = static_cast<float>(std::stoul(hex.substr(6, 2), nullptr, 16)) / 255.0f;
35 }
36 out = NUIColor::fromHex(rgb, alpha);
37 return true;
38}

101std::shared_ptr<NUITheme> NUITheme::loadFromFile(const std::string& filepath) {
102 // Start from the default theme so every token missing from the file keeps
103 // its default value. On any failure this default theme is returned (never
104 // nullptr) and the failure is logged.
105 auto theme = createDefault();
106
107 std::ifstream file(filepath, std::ios::binary);
108 if (!file.is_open()) {
109 Aestra::Log::warning("[NUITheme] Cannot open theme file '" + filepath + "' — using default theme");
110 return theme;
111 }
112
113 // Theme files are tiny; reject pathological sizes before parsing
114 // (file parsing is security-sensitive input handling).
115 constexpr std::streamoff kMaxThemeFileBytes = 1 << 20; // 1 MiB
116 file.seekg(0, std::ios::end);
117 const std::streamoff fileSize = file.tellg();
118 if (fileSize < 0 || fileSize > kMaxThemeFileBytes) {
119 Aestra::Log::error("[NUITheme] Theme file '" + filepath + "' rejected (size " + std::to_string(fileSize) +
120 " bytes) — using default theme");
121 return theme;
122 }
123 file.seekg(0, std::ios::beg);
124 std::string content(static_cast<size_t>(fileSize), '\0');
125 file.read(content.data(), fileSize);
126
127 Aestra::JSON root = Aestra::JSON::parse(content);
128 if (!root.isObject()) {
129 Aestra::Log::error("[NUITheme] Theme file '" + filepath + "' is not a valid JSON object — using default theme");
130 return theme;
131 }
132
133 int applied = 0;
134 int skipped = 0;
135
136 // NOTE: Aestra::JSON's const asObject() intentionally returns an empty map
137 // (SEC-RTM-014); iterate via the non-const copy-returning overload.
138 auto rootObj = root.asObject();
139
140 auto forEachEntry = [&](const char* section, const std::function<bool(const std::string&, Aestra::JSON&)>& apply) {
141 auto it = rootObj.find(section);
142 if (it == rootObj.end()) {
143 return;
144 }
145 if (!it->second.isObject()) {
146 Aestra::Log::warning("[NUITheme] Section '" + std::string(section) + "' in '" + filepath +
147 "' is not an object — ignored");
148 return;
149 }
150 auto entries = it->second.asObject();
151 for (auto& [key, value] : entries) {
152 if (apply(key, value)) {
153 ++applied;
154 } else {
155 ++skipped;
156 Aestra::Log::warning("[NUITheme] Invalid value for '" + std::string(section) + "." + key + "' in '" +
157 filepath + "' — keeping default");
158 }
159 }
160 };
161
162 forEachEntry("colors", [&](const std::string& key, Aestra::JSON& value) {
163 NUIColor color;
164 if (!value.isString() || !parseHexColor(value.asString(), color)) {
165 return false;
166 }
167 theme->setColor(key, color);
168 return true;
169 });
170
171 auto numericEntry = [](Aestra::JSON& value, float& out, bool requirePositive) {
172 if (!value.isNumber() || !std::isfinite(value.asNumber())) {
173 return false;
174 }
175 if (requirePositive && value.asNumber() <= 0.0) {
176 return false;
177 }
178 out = static_cast<float>(value.asNumber());
179 return true;
180 };
181
182 forEachEntry("dimensions", [&](const std::string& key, Aestra::JSON& value) {
183 float v = 0.0f;
184 if (!numericEntry(value, v, false))
185 return false;
186 theme->setDimension(key, v);
187 return true;
188 });
189
190 forEachEntry("effects", [&](const std::string& key, Aestra::JSON& value) {
191 float v = 0.0f;
192 if (!numericEntry(value, v, false))
193 return false;
194 theme->setEffect(key, v);
195 return true;
196 });
197
198 forEachEntry("fontSizes", [&](const std::string& key, Aestra::JSON& value) {
199 float v = 0.0f;
200 if (!numericEntry(value, v, true))
201 return false; // font sizes must be > 0
202 theme->setFontSize(key, v);
203 return true;
204 });
205
206 Aestra::Log::info("[NUITheme] Loaded theme '" + filepath + "': " + std::to_string(applied) + " tokens applied, " +
207 std::to_string(skipped) + " invalid tokens kept as defaults");
208 return theme;
209}
