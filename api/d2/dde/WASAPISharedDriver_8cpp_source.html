50// K-002: IMMNotificationClient implementation for hot-plug detection
51class WASAPIDeviceNotifier : public IMMNotificationClient {
52public:
53 WASAPIDeviceNotifier(WASAPISharedDriver* driver) : m_refCount(1), m_driver(driver) {}
54
55 // IUnknown
56 STDMETHODIMP QueryInterface(REFIID riid, void** ppv) override {
57 if (riid == IID_IUnknown || riid == __uuidof(IMMNotificationClient)) {
58 *ppv = static_cast<IMMNotificationClient*>(this);
59 AddRef();
60 return S_OK;
61 }
62 *ppv = nullptr;
63 return E_NOINTERFACE;
64 }
65 STDMETHODIMP_(ULONG) AddRef() override { return InterlockedIncrement(&m_refCount); }
66 STDMETHODIMP_(ULONG) Release() override {
67 ULONG count = InterlockedDecrement(&m_refCount);
68 if (count == 0)
69 delete this;
70 return count;
71 }
72
73 // IMMNotificationClient
74 STDMETHODIMP OnDeviceStateChanged(LPCWSTR pwstrDeviceId, DWORD dwNewState) override {
75 (void)pwstrDeviceId;
76 (void)dwNewState;
77 // Device connected/disabled/removed — trigger re-enumeration
78 if (m_driver) {
79 m_driver->onDeviceStateChanged("", dwNewState);
80 }
81 return S_OK;
82 }
83 STDMETHODIMP OnDefaultDeviceChanged(EDataFlow flow, ERole role, LPCWSTR pwstrDefaultDeviceId) override {
84 (void)flow;
85 (void)role;
86 (void)pwstrDefaultDeviceId;
87 if (m_driver && flow == eRender && role == eConsole) {
88 m_driver->onDefaultDeviceChanged(true, "");
89 }
90 return S_OK;
91 }
92 STDMETHODIMP OnPropertyValueChanged(LPCWSTR pwstrDeviceId, const PROPERTYKEY key) override {
93 (void)pwstrDeviceId;
94 (void)key;
95 return S_OK;
96 }
97 STDMETHODIMP OnDeviceAdded(LPCWSTR pwstrDeviceId) override {
98 (void)pwstrDeviceId;
99 if (m_driver)
100 m_driver->onDeviceStateChanged("", DEVICE_STATE_ACTIVE);
101 return S_OK;
102 }
103 STDMETHODIMP OnDeviceRemoved(LPCWSTR pwstrDeviceId) override {
104 (void)pwstrDeviceId;
105 if (m_driver)
106 m_driver->onDeviceStateChanged("", DEVICE_STATE_NOTPRESENT);
107 return S_OK;
108 }
109
110private:
111 LONG m_refCount;
112 WASAPISharedDriver* m_driver;
113};

162bool WASAPISharedDriver::initializeCOM() {
163 HRESULT hr = CoInitializeEx(nullptr, COINIT_MULTITHREADED);
164 if (FAILED(hr) && hr != RPC_E_CHANGED_MODE) {
165 setError(DriverError::INITIALIZATION_FAILED, "COM initialization failed: " + HResultToString(hr));
166 return false;
167 }
168
169 hr = CoCreateInstance(CLSID_MMDeviceEnumerator, nullptr, CLSCTX_ALL, IID_IMMDeviceEnumerator,
170 (void**)&m_deviceEnumerator);
171
172 if (FAILED(hr)) {
173 setError(DriverError::INITIALIZATION_FAILED, "Failed to create device enumerator: " + HResultToString(hr));
174 CoUninitialize();
175 return false;
176 }
177
178 return true;
179}

201bool WASAPISharedDriver::enumerateDevices(std::vector<AudioDeviceInfo>& devices) {
202 IMMDeviceCollection* deviceCollection = nullptr;
203 UINT deviceIdCounter = 0;
204
205 // 1. Enumerate OUTPUT devices (eRender)
206 HRESULT hr = reinterpret_cast<IMMDeviceEnumerator*>(m_deviceEnumerator)
207 ->EnumAudioEndpoints(eRender, DEVICE_STATE_ACTIVE, &deviceCollection);
208
209 if (SUCCEEDED(hr)) {
210 UINT count = 0;
211 deviceCollection->GetCount(&count);
212
213 for (UINT i = 0; i < count; ++i) {
214 IMMDevice* device = nullptr;
215 if (SUCCEEDED(deviceCollection->Item(i, &device))) {
216 AudioDeviceInfo info;
217 info.id = deviceIdCounter++;
218 info.maxOutputChannels = 2; // Default to stereo
219 info.maxInputChannels = 0;
220 info.preferredSampleRate = 48000;
221 info.supportedSampleRates = {44100, 48000, 96000};
222 info.isDefaultOutput = (i == 0); // First device is default
223 info.isDefaultInput = false;
224
225 // Get device name
226 LPWSTR deviceId = nullptr;
227 if (SUCCEEDED(device->GetId(&deviceId))) {
228 IPropertyStore* propertyStore = nullptr;
229 if (SUCCEEDED(device->OpenPropertyStore(STGM_READ, &propertyStore))) {
230 PROPVARIANT varName;
231 PropVariantInit(&varName);
232 if (SUCCEEDED(propertyStore->GetValue(PKEY_Device_FriendlyName, &varName))) {
233 // Convert wide string to narrow
234 int len =
235 WideCharToMultiByte(CP_UTF8, 0, varName.pwszVal, -1, nullptr, 0, nullptr, nullptr);
236 if (len > 0) {
237 std::string name(len - 1, '\0');
238 WideCharToMultiByte(CP_UTF8, 0, varName.pwszVal, -1, &name[0], len, nullptr, nullptr);
239 info.name = name;
240 }
241 PropVariantClear(&varName);
242 }
243 propertyStore->Release();
244 }
245 CoTaskMemFree(deviceId);
246 }
247
248 devices.push_back(info);
249 device->Release();
250 }
251 }
252 deviceCollection->Release();
253 }
254
255 // 2. Enumerate INPUT devices (eCapture)
256 hr = reinterpret_cast<IMMDeviceEnumerator*>(m_deviceEnumerator)
257 ->EnumAudioEndpoints(eCapture, DEVICE_STATE_ACTIVE, &deviceCollection);
258
259 if (SUCCEEDED(hr)) {
260 UINT count = 0;
261 deviceCollection->GetCount(&count);
262
263 for (UINT i = 0; i < count; ++i) {
264 IMMDevice* device = nullptr;
265 if (SUCCEEDED(deviceCollection->Item(i, &device))) {
266 AudioDeviceInfo info;
267 info.id = deviceIdCounter++;
268 info.maxOutputChannels = 0;
269 info.maxInputChannels = 2; // Default to stereo input
270 info.preferredSampleRate = 48000;
271 info.supportedSampleRates = {44100, 48000, 96000};
272 info.isDefaultOutput = false;
273 info.isDefaultInput = (i == 0); // First capture device is default
274
275 // Get device name
276 LPWSTR deviceId = nullptr;
277 if (SUCCEEDED(device->GetId(&deviceId))) {
278 IPropertyStore* propertyStore = nullptr;
279 if (SUCCEEDED(device->OpenPropertyStore(STGM_READ, &propertyStore))) {
280 PROPVARIANT varName;
281 PropVariantInit(&varName);
282 if (SUCCEEDED(propertyStore->GetValue(PKEY_Device_FriendlyName, &varName))) {
283 // Convert wide string to narrow
284 int len =
285 WideCharToMultiByte(CP_UTF8, 0, varName.pwszVal, -1, nullptr, 0, nullptr, nullptr);
286 if (len > 0) {
287 std::string name(len - 1, '\0');
288 WideCharToMultiByte(CP_UTF8, 0, varName.pwszVal, -1, &name[0], len, nullptr, nullptr);
289 info.name = name;
290 }
291 PropVariantClear(&varName);
292 }
293 propertyStore->Release();
294 }
295 CoTaskMemFree(deviceId);
296 }
297
298 devices.push_back(info);
299 device->Release();
300 }
301 }
302 deviceCollection->Release();
303 }
304
305 return !devices.empty();
306}

316bool WASAPISharedDriver::openStream(const AudioStreamConfig& config, AudioCallback callback, void* userData) {
317 if (m_state == DriverState::STREAM_RUNNING) {
318 stopStream();
319 }
320 if (m_state == DriverState::STREAM_OPEN) {
321 closeStream();
322 }
323
324 m_config = config;
325 m_userCallback = callback;
326 m_userData = userData;
327 m_telemetry = config.telemetry;
328
329 if (!openDevice(config.deviceId)) {
330 return false;
331 }
332
333 if (!initializeAudioClient()) {
334 closeDevice();
335 return false;
336 }
337
338 m_state = DriverState::STREAM_OPEN;
339 Aestra::Log::info("[WASAPI Shared] Stream opened successfully");
340
341 // K-002: Register for hot-plug notifications
342 registerDeviceNotifier();
343
344 return true;
345}

402bool WASAPISharedDriver::initializeAudioClient() {
403 HRESULT hr =
404 reinterpret_cast<IMMDevice*>(m_device)->Activate(IID_IAudioClient, CLSCTX_ALL, nullptr, (void**)&m_audioClient);
405
406 if (FAILED(hr)) {
407 setError(DriverError::STREAM_OPEN_FAILED, "Failed to activate audio client: " + HResultToString(hr));
408 return false;
409 }
410
411 // Get the mix format
412 hr = reinterpret_cast<IAudioClient*>(m_audioClient)->GetMixFormat(reinterpret_cast<WAVEFORMATEX**>(&m_waveFormat));
413 if (FAILED(hr)) {
414 setError(DriverError::STREAM_OPEN_FAILED, "Failed to get mix format: " + HResultToString(hr));
415 return false;
416 }
417
418 // Log the format we got
419 WAVEFORMATEX* wf = reinterpret_cast<WAVEFORMATEX*>(m_waveFormat);
420 std::cout << "[WASAPI Shared] Mix format: ";
421 if (wf->wFormatTag == WAVE_FORMAT_IEEE_FLOAT) {
422 std::cout << "32-bit float";
423 } else if (wf->wFormatTag == WAVE_FORMAT_PCM) {
424 std::cout << wf->wBitsPerSample << "-bit PCM";
425 } else if (wf->wFormatTag == WAVE_FORMAT_EXTENSIBLE) {
426 WAVEFORMATEXTENSIBLE* wfex = (WAVEFORMATEXTENSIBLE*)wf;
427 if (wfex->SubFormat == KSDATAFORMAT_SUBTYPE_IEEE_FLOAT) {
428 std::cout << "32-bit float (extensible)";
429 } else if (wfex->SubFormat == KSDATAFORMAT_SUBTYPE_PCM) {
430 std::cout << wf->wBitsPerSample << "-bit PCM (extensible)";
431 } else {
432 std::cout << "UNKNOWN EXTENSIBLE SUBFORMAT";
433 }
434 } else {
435 std::cout << "UNKNOWN FORMAT TAG: " << wf->wFormatTag;
436 }
437 std::cout << " @ " << wf->nSamplesPerSec << " Hz, " << wf->nChannels << " channels" << std::endl;
438
439 // Try to use IAudioClient3 for low-latency shared mode (Windows 10+)
440 bool usingIAudioClient3 = false;
441 hr = reinterpret_cast<IAudioClient*>(m_audioClient)
442 ->QueryInterface(__uuidof(IAudioClient3), (void**)&m_audioClient3);
443 if (SUCCEEDED(hr) && m_audioClient3) {
444 std::cout << "[WASAPI Shared] IAudioClient3 available - attempting low-latency mode" << std::endl;
445
446 // Query supported engine periods
447 WAVEFORMATEX* wf = reinterpret_cast<WAVEFORMATEX*>(m_waveFormat);
448 UINT32 defaultPeriodFrames, fundamentalPeriodFrames, minPeriodFrames, maxPeriodFrames;
449 hr = reinterpret_cast<IAudioClient3*>(m_audioClient3)
450 ->GetSharedModeEnginePeriod(wf, &defaultPeriodFrames, &fundamentalPeriodFrames, &minPeriodFrames,
451 &maxPeriodFrames);
452
453 if (SUCCEEDED(hr)) {
454 std::cout << "[WASAPI Shared] Engine periods - Min: " << minPeriodFrames
455 << ", Default: " << defaultPeriodFrames << ", Max: " << maxPeriodFrames << " frames" << std::endl;
456
457 // Choose a safe period: honor requested buffer size when possible.
458 // Using absolute minPeriodFrames in shared mode is often too aggressive
459 // and can cause underruns on real projects.
460 UINT32 targetPeriodFrames = m_config.bufferSize;
461 if (targetPeriodFrames < minPeriodFrames)
462 targetPeriodFrames = minPeriodFrames;
463 if (targetPeriodFrames > maxPeriodFrames)
464 targetPeriodFrames = maxPeriodFrames;
465
466 // Align to the device's fundamental period if required.
467 if (fundamentalPeriodFrames > 0) {
468 UINT32 aligned = ((targetPeriodFrames + fundamentalPeriodFrames - 1) / fundamentalPeriodFrames) *
469 fundamentalPeriodFrames;
470 if (aligned <= maxPeriodFrames) {
471 targetPeriodFrames = aligned;
472 }
473 }
474
475 // Create event for audio thread synchronization
476 m_audioEvent = CreateEvent(nullptr, FALSE, FALSE, nullptr);
477 if (!m_audioEvent) {
478 setError(DriverError::STREAM_OPEN_FAILED, "Failed to create audio event");
479 return false;
480 }
481
482 // Initialize with low-latency shared mode
483 hr = reinterpret_cast<IAudioClient3*>(m_audioClient3)
484 ->InitializeSharedAudioStream(AUDCLNT_STREAMFLAGS_EVENTCALLBACK, targetPeriodFrames, wf, nullptr);
485
486 if (SUCCEEDED(hr)) {
487 usingIAudioClient3 = true;
488 m_bufferFrameCount = targetPeriodFrames;
489 std::cout << "[WASAPI Shared] Using IAudioClient3 low-latency mode: " << targetPeriodFrames << " frames"
490 << std::endl;
491
492 // Set event handle
493 hr = reinterpret_cast<IAudioClient3*>(m_audioClient3)
494 ->SetEventHandle(reinterpret_cast<HANDLE>(m_audioEvent));
495 if (FAILED(hr)) {
496 setError(DriverError::STREAM_OPEN_FAILED,
497 "Failed to set event handle (IAudioClient3): " + HResultToString(hr));
498 return false;
499 }
500 } else {
501 std::cout << "[WASAPI Shared] IAudioClient3 initialization failed, falling back to legacy mode"
502 << std::endl;
503 CloseHandle(m_audioEvent);
504 m_audioEvent = nullptr;
505 }
506 }
507 }
508
509 // Fallback to classic IAudioClient if IAudioClient3 failed or unavailable
510 if (!usingIAudioClient3) {
511 std::cout << "[WASAPI Shared] Using legacy IAudioClient mode" << std::endl;
512
513 // Convert buffer size from frames to duration (100ns units)
514 REFERENCE_TIME requestedDuration = (REFERENCE_TIME)(10000000.0 * m_config.bufferSize / m_config.sampleRate);
515
516 // Create event for audio thread synchronization
517 if (!m_audioEvent) {
518 m_audioEvent = CreateEvent(nullptr, FALSE, FALSE, nullptr);
519 if (!m_audioEvent) {
520 setError(DriverError::STREAM_OPEN_FAILED, "Failed to create audio event");
521 return false;
522 }
523 }
524
525 // Initialize audio client in shared mode with event-driven callback
526 hr = reinterpret_cast<IAudioClient*>(m_audioClient)
527 ->Initialize(AUDCLNT_SHAREMODE_SHARED, AUDCLNT_STREAMFLAGS_EVENTCALLBACK, requestedDuration, 0,
528 reinterpret_cast<const WAVEFORMATEX*>(m_waveFormat), nullptr);
529
530 if (FAILED(hr)) {
531 setError(DriverError::STREAM_OPEN_FAILED, "Failed to initialize audio client: " + HResultToString(hr));
532 return false;
533 }
534
535 // Set event handle
536 hr = reinterpret_cast<IAudioClient*>(m_audioClient)->SetEventHandle(reinterpret_cast<HANDLE>(m_audioEvent));
537 if (FAILED(hr)) {
538 setError(DriverError::STREAM_OPEN_FAILED, "Failed to set event handle: " + HResultToString(hr));
539 return false;
540 }
541
542 // Get buffer size
543 hr = reinterpret_cast<IAudioClient*>(m_audioClient)->GetBufferSize(&m_bufferFrameCount);
544 if (FAILED(hr)) {
545 setError(DriverError::STREAM_OPEN_FAILED, "Failed to get buffer size: " + HResultToString(hr));
546 return false;
547 }
548 }
549
550 // Get render client
551 hr = reinterpret_cast<IAudioClient*>(m_audioClient)->GetService(IID_IAudioRenderClient, (void**)&m_renderClient);
552 if (FAILED(hr)) {
553 setError(DriverError::STREAM_OPEN_FAILED, "Failed to get render client: " + HResultToString(hr));
554 return false;
555 }
556
557 // Calculate and report accurate latency metrics
558 wf = reinterpret_cast<WAVEFORMATEX*>(m_waveFormat);
559 AudioLatencyInfo latencyInfo = AudioLatencyInfo::calculate(m_bufferFrameCount, wf->nSamplesPerSec,
560 3.0 // Shared mode typically has ~3x RTL multiplier
561 );
562
563 std::cout << "[WASAPI Shared] Initialized - "
564 << "Sample Rate: " << wf->nSamplesPerSec << " Hz, "
565 << "Buffer: " << m_bufferFrameCount << " frames\n"
566 << " Buffer Period: " << std::fixed << std::setprecision(2) << latencyInfo.bufferPeriodMs
567 << "ms (one-way)\n"
568 << " Estimated RTL: " << latencyInfo.estimatedRTL_Ms << "ms (round-trip)" << std::endl;
569
570 return true;
571}

579bool WASAPISharedDriver::startStream() {
580 if (m_state != DriverState::STREAM_OPEN) {
581 setError(DriverError::STREAM_START_FAILED, "Stream not open");
582 return false;
583 }
584
585 m_shouldStop = false;
586 m_isRunning = true;
587
588 // Start audio client
589 HRESULT hr = reinterpret_cast<IAudioClient*>(m_audioClient)->Start();
590 if (FAILED(hr)) {
591 setError(DriverError::STREAM_START_FAILED, "Failed to start audio client: " + HResultToString(hr));
592 m_isRunning = false;
593 return false;
594 }
595
596 // Start audio thread
597 m_audioThread = std::thread(&WASAPISharedDriver::audioThreadProc, this);
598
599 m_state = DriverState::STREAM_RUNNING;
600 std::cout << "[WASAPI Shared] Stream started" << std::endl;
601 return true;
602}

604void WASAPISharedDriver::stopStream() {
605 if (!m_isRunning) {
606 return;
607 }
608
609 std::cout << "[WASAPI Shared] Stopping stream safely..." << std::endl;
610
611 m_shouldStop = true;
612
613 // Signal event to wake up thread immediately
614 if (m_audioEvent) {
615 SetEvent(m_audioEvent);
616 }
617
618 // Wait for thread to finish with proper shutdown sequence
619 if (m_audioThread.joinable()) {
620 // Wait a short period for thread to exit gracefully
621 if (m_audioThread.joinable()) {
622 m_audioThread.join(); // Try normal join first
623 }
624
625 // If still joinable after a brief wait, it may be hung - detach to prevent std::terminate
626 if (m_audioThread.joinable()) {
627 std::cerr << "[WASAPI Shared] Warning: Audio thread didn't stop gracefully, detaching" << std::endl;
628 m_audioThread.detach();
629 } else {
630 // Thread stopped successfully - safe to fill buffer
631 fillAudioBufferWithSilence();
632 }
633 } else {
634 // Thread was not joinable (already detached or not started)
635 fillAudioBufferWithSilence();
636 }
637
638 // Stop audio client
639 if (m_audioClient) {
640 reinterpret_cast<IAudioClient*>(m_audioClient)->Stop();
641 }
642
643 m_isRunning = false;
644 m_state = DriverState::STREAM_OPEN;
645 std::cout << "[WASAPI Shared] Stream stopped safely" << std::endl;
646}

648void WASAPISharedDriver::fillAudioBufferWithSilence() {
649 // Fill buffer with silence before stopping to prevent clicks
650 // Only call this after thread has stopped to avoid race conditions
651 if (m_audioClient && m_renderClient && m_bufferFrameCount > 0) {
652 UINT32 padding = 0;
653 HRESULT hr = reinterpret_cast<IAudioClient*>(m_audioClient)->GetCurrentPadding(&padding);
654 if (SUCCEEDED(hr)) {
655 UINT32 framesAvailable = m_bufferFrameCount - padding;
656 if (framesAvailable > 0) {
657 BYTE* data = nullptr;
658 hr = reinterpret_cast<IAudioRenderClient*>(m_renderClient)->GetBuffer(framesAvailable, &data);
659 if (SUCCEEDED(hr)) {
660 std::memset(reinterpret_cast<void*>(data), 0,
661 framesAvailable * reinterpret_cast<WAVEFORMATEX*>(m_waveFormat)->nBlockAlign);
662 reinterpret_cast<IAudioRenderClient*>(m_renderClient)
663 ->ReleaseBuffer(framesAvailable, AUDCLNT_BUFFERFLAGS_SILENT);
664 }
665 }
666 }
667 }

670double WASAPISharedDriver::getStreamLatency() const {
671 if (!m_audioClient || !m_waveFormat) {
672 return 0.0;
673 }
674
675 REFERENCE_TIME latency = 0;
676 HRESULT hr = reinterpret_cast<IAudioClient*>(m_audioClient)->GetStreamLatency(&latency);
677 if (SUCCEEDED(hr)) {
678 // Convert from 100ns units to seconds
679 return static_cast<double>(latency) / 10000000.0;
680 }
681
682 return 0.0;

685void WASAPISharedDriver::audioThreadProc() {
686 if (!setThreadPriority()) {
687 std::cerr << "[WASAPI Shared] Warning: Failed to set thread priority" << std::endl;
688 }
689
690 // Set MMCSS Pro Audio scheduling for stable performance
691 DWORD taskIndex = 0;
692 HANDLE avrtHandle = AvSetMmThreadCharacteristicsA("Pro Audio", &taskIndex);
693 if (!avrtHandle) {
694 std::cerr << "[WASAPI Shared] Warning: Failed to set MMCSS characteristics" << std::endl;
695 } else {
696 // Set thread priority (lower than Exclusive mode since shared mode has higher latency)
697 BOOL prioritySuccess = AvSetMmThreadPriority(avrtHandle, AVRT_PRIORITY_HIGH);
698 if (!prioritySuccess) {
699 std::cerr << "[WASAPI Shared] Warning: Failed to set MMCSS priority to HIGH" << std::endl;
700 } else {
701 std::cout << "[WASAPI Shared] MMCSS enabled: Pro Audio @ HIGH priority" << std::endl;
702 }
703 }
704
705 // Temporary buffer for user callback
706 std::vector<float> userBuffer(m_bufferFrameCount * m_config.numOutputChannels);
707
708 WAVEFORMATEX* wf = reinterpret_cast<WAVEFORMATEX*>(m_waveFormat);
709 std::cout << "[WASAPI Shared] Audio thread running with " << m_bufferFrameCount << " frames at "
710 << wf->nSamplesPerSec << " Hz" << std::endl;
711
712 while (!m_shouldStop) {
713 // Wait for buffer event
714 DWORD waitResult = WaitForSingleObject(m_audioEvent, 2000);
715
716 if (waitResult != WAIT_OBJECT_0) {
717 if (!m_shouldStop) {
718 m_statistics.underrunCount++;
719 if (m_telemetry) {
720 m_telemetry->incrementXruns();
721 m_telemetry->incrementUnderruns();
722 }
723 }
724 continue;
725 }
726
727 if (m_shouldStop) {
728 break;
729 }
730
731 LARGE_INTEGER startTime;
732 QueryPerformanceCounter(&startTime);
733
734 // Get padding (number of frames already in buffer)
735 UINT32 padding = 0;
736 HRESULT hr = reinterpret_cast<IAudioClient*>(m_audioClient)->GetCurrentPadding(&padding);
737 if (FAILED(hr)) {
738 continue;
739 }
740
741 // Calculate available frames
742 UINT32 availableFrames = m_bufferFrameCount - padding;
743 if (availableFrames == 0) {
744 continue;
745 }
746
747 // Get buffer from WASAPI
748 BYTE* data = nullptr;
749 hr = reinterpret_cast<IAudioRenderClient*>(m_renderClient)->GetBuffer(availableFrames, &data);
750 if (FAILED(hr)) {
751 m_statistics.underrunCount++;
752 if (m_telemetry) {
753 m_telemetry->incrementXruns();
754 m_telemetry->incrementUnderruns();
755 }
756 continue;
757 }
758
759 // Call user callback
760 WAVEFORMATEX* wf = reinterpret_cast<WAVEFORMATEX*>(m_waveFormat);
761 double streamTime = static_cast<double>(m_statistics.callbackCount * m_bufferFrameCount) / wf->nSamplesPerSec;
762
763 if (m_userCallback) {
764 m_userCallback(userBuffer.data(), nullptr, availableFrames, streamTime, m_userData);
765 } else {
766 // Silence
767 std::fill(userBuffer.begin(), userBuffer.end(), 0.0f);
768 }
769
770 m_statistics.callbackCount++;
771
772 // Convert float to WASAPI format
773 if (wf->wFormatTag == WAVE_FORMAT_IEEE_FLOAT ||
774 (wf->wFormatTag == WAVE_FORMAT_EXTENSIBLE &&
775 ((WAVEFORMATEXTENSIBLE*)wf)->SubFormat == KSDATAFORMAT_SUBTYPE_IEEE_FLOAT)) {
776 // 32-bit float - direct copy
777 std::memcpy(reinterpret_cast<void*>(data), userBuffer.data(), availableFrames * wf->nBlockAlign);
778 } else if (wf->wFormatTag == WAVE_FORMAT_PCM ||
779 (wf->wFormatTag == WAVE_FORMAT_EXTENSIBLE &&
780 ((WAVEFORMATEXTENSIBLE*)wf)->SubFormat == KSDATAFORMAT_SUBTYPE_PCM)) {
781 // PCM format - need to convert floats to integers
782 if (wf->wBitsPerSample == 16) {
783 // 16-bit PCM
784 int16_t* pcmData = reinterpret_cast<int16_t*>(data);
785 const float ditherScale = 1.0f / 32768.0f;
786
787 for (uint32_t i = 0; i < availableFrames * m_config.numOutputChannels; ++i) {
788 float sample = userBuffer[i];
789
790 // Apply TPDF Dither
791 if (m_ditherEnabled) {
792 sample += DitherUtils::generateTPDF(m_ditherState) * ditherScale;
793 }
794
795 if (sample > 1.0f)
796 sample = 1.0f;
797 if (sample < -1.0f)
798 sample = -1.0f;
799 pcmData[i] = static_cast<int16_t>(sample * 32767.0f);
800 }
801 } else if (wf->wBitsPerSample == 24) {
802 // 24-bit PCM
803 uint8_t* pcmData = data;
804 const float ditherScale = 1.0f / 8388608.0f;
805
806 for (uint32_t i = 0; i < availableFrames * m_config.numOutputChannels; ++i) {
807 float sample = userBuffer[i];
808
809 // Apply TPDF Dither
810 if (m_ditherEnabled.load(std::memory_order_relaxed)) {
811 sample += DitherUtils::generateTPDF(m_ditherState) * ditherScale;
812 }
813
814 if (sample > 1.0f)
815 sample = 1.0f;
816 if (sample < -1.0f)
817 sample = -1.0f;
818 int32_t pcmValue = static_cast<int32_t>(sample * 8388607.0f);
819 pcmData[i * 3 + 0] = static_cast<uint8_t>(pcmValue & 0xFF);
820 pcmData[i * 3 + 1] = static_cast<uint8_t>((pcmValue >> 8) & 0xFF);
821 pcmData[i * 3 + 2] = static_cast<uint8_t>((pcmValue >> 16) & 0xFF);
822 }
823 } else {
824 // Unknown bit depth - zero the buffer
825 std::memset(reinterpret_cast<void*>(data), 0, availableFrames * wf->nBlockAlign);
826 }
827 } else {
828 // Unknown format - zero the buffer
829 std::memset(reinterpret_cast<void*>(data), 0,
830 availableFrames * reinterpret_cast<WAVEFORMATEX*>(m_waveFormat)->nBlockAlign);
831 }
832
833 // Release buffer
834 hr = reinterpret_cast<IAudioRenderClient*>(m_renderClient)->ReleaseBuffer(availableFrames, 0);
835 if (FAILED(hr)) {
836 // RT-safe: no logging in audio thread
837 }
838
839 // Update statistics
840 LARGE_INTEGER endTime;
841 QueryPerformanceCounter(&endTime);
842 double callbackTimeUs =
843 static_cast<double>(endTime.QuadPart - startTime.QuadPart) * 1000000.0 / static_cast<double>(m_perfFreq);
844 updateStatistics(callbackTimeUs);
845
846 // Update telemetry (lock-free, RT-safe)
847 if (m_telemetry) {
848 uint64_t callbackNs = static_cast<uint64_t>(callbackTimeUs * 1000);
849 m_telemetry->updateLastCallbackNs(callbackNs);
850 m_telemetry->updateMaxCallbackNs(callbackNs);
851 m_telemetry->updateLastBufferFrames(availableFrames);
852 if (m_config.sampleRate > 0) {
853 m_telemetry->updateLastSampleRate(m_config.sampleRate);
854 }
855 }
856 }
857
858 if (avrtHandle) {
859 AvRevertMmThreadCharacteristics(avrtHandle);
860 }
861
862 std::cout << "[WASAPI Shared] Audio thread exiting" << std::endl;
863}

881void WASAPISharedDriver::updateStatistics(double callbackTimeUs) {
882 m_statistics.callbackCount++;
883
884 // Update average callback time (exponential moving average)
885 const double alpha = 0.1;
886 m_statistics.averageCallbackTimeUs = alpha * callbackTimeUs + (1.0 - alpha) * m_statistics.averageCallbackTimeUs;
887
888 // Update max callback time
889 if (callbackTimeUs > m_statistics.maxCallbackTimeUs) {
890 m_statistics.maxCallbackTimeUs = callbackTimeUs;
891 }
892
893 // Estimate CPU load
894 if (m_waveFormat && m_bufferFrameCount > 0) {
895 WAVEFORMATEX* wf = reinterpret_cast<WAVEFORMATEX*>(m_waveFormat);
896 double bufferDurationUs = static_cast<double>(m_bufferFrameCount) * 1000000.0 / wf->nSamplesPerSec;
897 m_statistics.cpuLoadPercent = (callbackTimeUs / bufferDurationUs) * 100.0;
898 }
899
900 // Avoid COM calls on the RT thread. Approximate latency from buffer period.
901 if (m_waveFormat && m_bufferFrameCount > 0) {
902 WAVEFORMATEX* wf = reinterpret_cast<WAVEFORMATEX*>(m_waveFormat);
903 m_statistics.actualLatencyMs = (static_cast<double>(m_bufferFrameCount) * 1000.0) / wf->nSamplesPerSec;
904 }
