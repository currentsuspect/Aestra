101 void incrementUnderruns() noexcept {
102 underruns.fetch_add(1, std::memory_order_relaxed);
103 uint32_t consecutive = consecutiveUnderruns.fetch_add(1, std::memory_order_relaxed) + 1;
104 stableFramesSinceRecovery.store(0, std::memory_order_relaxed);
105
106 // Trigger recovery mode if threshold exceeded
107 if (consecutive >= kRecoveryThreshold && !inRecoveryMode.load(std::memory_order_relaxed)) {
108 inRecoveryMode.store(true, std::memory_order_release);
109 recoveryModeActivations.fetch_add(1, std::memory_order_relaxed);
110 }
111 }

119 void recordStableBlock() noexcept {
120 consecutiveUnderruns.store(0, std::memory_order_relaxed);
121
122 if (inRecoveryMode.load(std::memory_order_relaxed)) {
123 uint32_t stable = stableFramesSinceRecovery.fetch_add(1, std::memory_order_relaxed) + 1;
124 if (stable >= kRecoveryFrames) {
125 inRecoveryMode.store(false, std::memory_order_release);
126 stableFramesSinceRecovery.store(0, std::memory_order_relaxed);
127 }
128 }
