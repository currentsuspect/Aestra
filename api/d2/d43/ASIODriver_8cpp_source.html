75std::vector<AudioDeviceInfo> ASIODriver::getDevices() {
76 std::vector<AudioDeviceInfo> devices;
77 m_knownDrivers.clear();
78
79 // Scan HKEY_LOCAL_MACHINE\SOFTWARE\ASIO
80 HKEY hKey;
81 if (RegOpenKeyExA(HKEY_LOCAL_MACHINE, "SOFTWARE\\ASIO", 0, KEY_READ, &hKey) == ERROR_SUCCESS) {
82 char keyName[256];
83 DWORD keyNameLen = 256;
84 DWORD index = 0;
85
86 while (RegEnumKeyExA(hKey, index, keyName, &keyNameLen, nullptr, nullptr, nullptr, nullptr) == ERROR_SUCCESS) {
87 HKEY hSubKey;
88 if (RegOpenKeyExA(hKey, keyName, 0, KEY_READ, &hSubKey) == ERROR_SUCCESS) {
89 char clsidStr[128];
90 char descStr[128];
91 DWORD clsidLen = sizeof(clsidStr);
92 DWORD descLen = sizeof(descStr);
93 DWORD type;
94
95 if (RegQueryValueExA(hSubKey, "CLSID", nullptr, &type, (LPBYTE)clsidStr, &clsidLen) == ERROR_SUCCESS) {
96 if (RegQueryValueExA(hSubKey, "Description", nullptr, &type, (LPBYTE)descStr, &descLen) !=
97 ERROR_SUCCESS) {
98 // [SEC-FIX] strcpy_s requires destination size; descStr is 128 bytes.
99 strcpy_s(descStr, sizeof(descStr), keyName);
100 }
101
102 // Add to internal registry
103 m_knownDrivers.push_back({std::string(descStr), std::string(clsidStr)});
104 uint32_t devId = (uint32_t)(m_knownDrivers.size() - 1);
105
106 devices.push_back({devId,
107 std::string(descStr),
108 2,
109 2, // Default capability guess (refined on open)
110 {44100, 48000, 88200, 96000}, // Supported rates guess
111 44100,
112 false,
113 false});
114 }
115 RegCloseKey(hSubKey);
116 }
117 index++;
118 keyNameLen = 256;
119 }
120 RegCloseKey(hKey);
121 }
122
123 return devices;
124}

126bool ASIODriver::loadDriver(uint32_t deviceIndex) {
127 if (deviceIndex >= m_knownDrivers.size()) {
128 m_lastError = "Invalid device index";
129 return false;
130 }
131
132 // Ensure COM is initialized (STA required for ASIO)
133 HRESULT hrInit = CoInitializeEx(nullptr, COINIT_APARTMENTTHREADED);
134 if (hrInit == RPC_E_CHANGED_MODE) {
135 // Already initialized as MTA - This is suboptimal but we proceed.
136 }
137
138 if (m_asio) {
139 m_asio->Release();
140 m_asio = nullptr;
141 }
142
143 const std::string& argClsid = m_knownDrivers[deviceIndex].clsid;
144 m_driverName = m_knownDrivers[deviceIndex].name;
145
146 // Convert string CLSID to GUID
147 CLSID clsid;
148 std::wstring wClsid(argClsid.begin(), argClsid.end());
149 if (CLSIDFromString(wClsid.c_str(), &clsid) != S_OK) {
150 m_lastError = "Invalid CLSID format";
151 return false;
152 }
153
154 // Load Driver via CoCreateInstance
155 IUnknown* unk = nullptr;
156 HRESULT hrCreate = CoCreateInstance(clsid, nullptr, CLSCTX_INPROC_SERVER, __uuidof(IUnknown), (void**)&unk);
157 if (hrCreate != S_OK) {
158 m_lastError = "CoCreateInstance failed: " + std::to_string(hrCreate);
159 return false;
160 }
161
162 // Query for IASIO interface
163 // Standard IID_IASIO: {47d6d9d4-e949-11d1-86ba-006008caaa53}
164 static const GUID IID_IASIO = {0x47d6d9d4, 0xe949, 0x11d1, {0x86, 0xba, 0x00, 0x60, 0x08, 0xca, 0xaa, 0x53}};
165
166 HRESULT hrQI = unk->QueryInterface(IID_IASIO, (void**)&m_asio);
167
168 // Fallback: Some legacy drivers use their own CLSID as the Interface ID
169 if (hrQI != S_OK) {
170 hrQI = unk->QueryInterface(clsid, (void**)&m_asio);
171 }
172
173 unk->Release(); // Always release the IUnknown wrapper
174
175 if (hrQI != S_OK || !m_asio) {
176 m_lastError = "QueryInterface(IASIO) failed. HRESULT: " + std::to_string(hrQI);
177 return false;
178 }
179
180 // Initialize Driver
181 // ASIO init returns 1 (ASIOTrue) on success, 0 on failure.
182 // NOTE: It does NOT return ASE_SUCCESS (0x3f4847a0) or ASE_OK (0).
183 // Use GetForegroundWindow() or GetConsoleWindow() as ASIO4ALL and others prefer an owned window handle.
184 HWND hwnd = GetForegroundWindow();
185 if (!hwnd)
186 hwnd = GetDesktopWindow();
187
188 if (m_asio->init(hwnd) != 0)
189 return true;
190 if (m_asio->init(nullptr) != 0)
191 return true;
192
193 m_lastError = "ASIO Init failed";
194 m_asio->Release();
195 m_asio = nullptr;
196 return false;
197}

199bool ASIODriver::openStream(const AudioStreamConfig& config, AudioCallback callback, void* userData) {
200 if (m_isRunning)
201 return false;
202
203 // Load Driver if not loaded or different
204 // Since we rely on index, we can't easily check 'if different' unless we store current index.
205 // For now, always reload to be safe.
206 if (!loadDriver(config.deviceId)) {
207 // Attempt to rescan?
208 getDevices(); // refresh list
209 if (!loadDriver(config.deviceId))
210 return false;
211 }
212
213 // 2. Query Capabilities
214 long inputCh = 0, outputCh = 0;
215 m_asio->getChannels(&inputCh, &outputCh);
216
217 // 3. Setup Buffers
218 long minSize = 0, maxSize = 0, prefSize = 0, gran = 0;
219 ASIO::ASIOError errBuffer = (ASIO::ASIOError)m_asio->getBufferSize(&minSize, &maxSize, &prefSize, &gran);
220 if (errBuffer != ASIO::ASE_OK) {
221 m_lastError = "getBufferSize failed";
222 return false;
223 }
224
225 m_bufferSize = config.bufferSize;
226 if (m_bufferSize < minSize)
227 m_bufferSize = minSize; // Clamp to valid range
228 if (m_bufferSize > maxSize)
229 m_bufferSize = maxSize;
230
231 // 4. Create Buffers
232 m_bufferInfos.clear();
233 int activeIn = std::min((int)config.numInputChannels, (int)inputCh);
234 int activeOut = std::min((int)config.numOutputChannels, (int)outputCh);
235
236 if (activeIn + activeOut == 0) {
237 m_lastError = "No channels to open";
238 return false;
239 }
240
241 // Input Buffers
242 for (int i = 0; i < activeIn; ++i) {
243 ASIO::ASIOBufferInfo info;
244 memset(&info, 0, sizeof(info));
245 info.isInput = 1;
246 info.channelNum = i;
247 info.buffers[0] = info.buffers[1] = nullptr;
248 m_bufferInfos.push_back(info);
249 }
250
251 // Output Buffers
252 for (int i = 0; i < activeOut; ++i) {
253 ASIO::ASIOBufferInfo info;
254 memset(&info, 0, sizeof(info));
255 info.isInput = 0;
256 info.channelNum = i;
257 info.buffers[0] = info.buffers[1] = nullptr;
258 m_bufferInfos.push_back(info);
259 }
260
261 ASIO::ASIOError err = (ASIO::ASIOError)m_asio->createBuffers(m_bufferInfos.data(), (long)m_bufferInfos.size(),
262 m_bufferSize, &s_asioCallbacks);
263 if (err != ASIO::ASE_OK && err != ASIO::ASE_SUCCESS) {
264 m_lastError = "CreateBuffers failed: " + std::to_string(err);
265 return false;
266 }
267
268 // Get Channel Infos
269 m_channelInfos.resize(m_bufferInfos.size());
270 for (size_t i = 0; i < m_bufferInfos.size(); ++i) {
271 m_channelInfos[i].channel = m_bufferInfos[i].channelNum;
272 m_channelInfos[i].isInput = m_bufferInfos[i].isInput;
273 m_asio->getChannelInfo(&m_channelInfos[i]);
274 }
275
276 m_callback = callback;
277 m_callbackUserData = userData;
278 m_inputChannels = activeIn;
279 m_outputChannels = activeOut;
280 m_configuredInputChannels = config.numInputChannels;
281 m_configuredOutputChannels = config.numOutputChannels;
282
283 m_interleavedInput.resize(m_bufferSize * m_configuredInputChannels);
284 m_interleavedOutput.resize(m_bufferSize * m_configuredOutputChannels);
285
286 // Set Sample Rate
287 double currentRate = 0;
288 m_asio->getSampleRate(&currentRate);
289 if (abs(currentRate - (double)config.sampleRate) > 1.0) {
290 if (m_asio->canSampleRate((double)config.sampleRate) == ASIO::ASE_OK) {
291 m_asio->setSampleRate((double)config.sampleRate);
292 m_sampleRate = (double)config.sampleRate;
293 } else {
294 m_sampleRate = currentRate;
295 // Sample rate mismatch but continuing with driver's rate
296 }
297 } else {
298 m_sampleRate = currentRate;
299 }
300
301 s_instance = this;
302 return true;
303}

359ASIO::ASIOTime* ASIODriver::asioBufferSwitch(long doubleBufferIndex, long directProcess) {
360 if (!m_isRunning || !m_callback)
361 return nullptr;
362
363 // "Black Magic": Ensure ASIO Driver thread runs at Real-Time Priority
364 // This is safe to call repeatedly but checking a static flag is faster
365 static thread_local bool s_threadPrioritySet = false;
366 if (!s_threadPrioritySet) {
367 MMCSS::setProAudio();
368 s_threadPrioritySet = true;
369 }
370
371 // 1. De-interleave Input (ASIO -> Interleaved Float)
372 // Zero out the entire buffer first (handles channels that are in config but not in device)
373 if (!m_interleavedInput.empty()) {
374 memset(m_interleavedInput.data(), 0, m_interleavedInput.size() * sizeof(float));
375 }
376
377 for (int i = 0; i < m_inputChannels; ++i) {
378 ASIO::ASIOChannelInfo& chInfo = m_channelInfos[i];
379 void* src = m_bufferInfos[i].buffers[doubleBufferIndex];
380
381 switch (chInfo.type) {
382 case ASIO::ASIOSTInt32LSB: {
383 int32_t* src32 = static_cast<int32_t*>(src);
384 for (long f = 0; f < m_bufferSize; ++f) {
385 m_interleavedInput[f * m_configuredInputChannels + i] =
386 static_cast<float>(src32[f]) * (1.0f / 2147483648.0f);
387 }
388 break;
389 }
390 case ASIO::ASIOSTInt16LSB: {
391 int16_t* src16 = static_cast<int16_t*>(src);
392 for (long f = 0; f < m_bufferSize; ++f) {
393 m_interleavedInput[f * m_configuredInputChannels + i] =
394 static_cast<float>(src16[f]) * (1.0f / 32768.0f);
395 }
396 break;
397 }
398 case ASIO::ASIOSTFloat32LSB: {
399 float* srcFloat = static_cast<float*>(src);
400 for (long f = 0; f < m_bufferSize; ++f) {
401 m_interleavedInput[f * m_configuredInputChannels + i] = srcFloat[f];
402 }
403 break;
404 }
405 default:
406 // Buffer already zeroed
407 break;
408 }
409 }
410
411 // 2. Call Engine
412 m_callback(m_interleavedOutput.data(), m_interleavedInput.data(), m_bufferSize, 0.0, m_callbackUserData);
413 m_stats.callbackCount++;
414
415 // 3. Interleave Output (Interleaved Float -> ASIO)
416 int outStart = m_inputChannels;
417 for (int i = 0; i < m_outputChannels; ++i) {
418 ASIO::ASIOChannelInfo& chInfo = m_channelInfos[outStart + i];
419 void* dest = m_bufferInfos[outStart + i].buffers[doubleBufferIndex];
420
421 switch (chInfo.type) {
422 case ASIO::ASIOSTInt32LSB: {
423 int32_t* dest32 = static_cast<int32_t*>(dest);
424 for (long f = 0; f < m_bufferSize; ++f) {
425 float v = m_interleavedOutput[f * m_configuredOutputChannels + i];
426 if (v > 1.0f)
427 v = 1.0f;
428 else if (v < -1.0f)
429 v = -1.0f;
430 dest32[f] = static_cast<int32_t>(v * 2147483647.0f);
431 }
432 break;
433 }
434 case ASIO::ASIOSTInt16LSB: {
435 int16_t* dest16 = static_cast<int16_t*>(dest);
436 for (long f = 0; f < m_bufferSize; ++f) {
437 float v = m_interleavedOutput[f * m_configuredOutputChannels + i];
438 if (v > 1.0f)
439 v = 1.0f;
440 else if (v < -1.0f)
441 v = -1.0f;
442 dest16[f] = static_cast<int16_t>(v * 32767.0f);
443 }
444 break;
445 }
446 case ASIO::ASIOSTFloat32LSB: {
447 float* destFloat = static_cast<float*>(dest);
448 for (long f = 0; f < m_bufferSize; ++f) {
449 destFloat[f] = m_interleavedOutput[f * m_configuredOutputChannels + i];
450 }
451 break;
452 }
453 default:
454 memset(dest, 0, m_bufferSize * 4);
455 break;
456 }
457 }
458
459 m_asio->outputReady();
460 return nullptr;
461}
