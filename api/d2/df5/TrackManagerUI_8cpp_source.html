146void TrackManagerUI::refreshTracks() {
147 m_needsTrackRefresh = false; // Clear lazy-init flag if we got here directly
148
149 if (!m_trackManager) {
150 Log::error("refreshTracks: m_trackManager is null!");
151 return;
152 }
153
154 Log::info("refreshTracks: starting, laneCount=" +
155 std::to_string(m_trackManager->getPlaylistModel().getLaneCount()));
156
157 // v3.0 logic: iterate over PlaylistModel lanes instead of Mixer channels
158 auto& playlist = m_trackManager->getPlaylistModel();
159 size_t laneCount = playlist.getLaneCount();
160 Log::info("refreshTracks: looping over " + std::to_string(laneCount) + " lanes");
161
162 // Stable lane IDs own selection. Raw pointers only describe the current
163 // widget generation, so discard that view before destroying any rows.
164 m_selectedTracks.clear();
165 for (auto& trackUI : m_trackUIComponents) {
166 removeChild(trackUI);
167 }
168 m_trackUIComponents.clear();
169 for (size_t i = 0; i < laneCount; ++i) {
170 auto laneId = playlist.getLaneId(i);
171 auto lane = playlist.getLane(laneId);
172 if (!lane) {
173 Log::warning("refreshTracks: lane " + std::to_string(i) + " is null!");
174 continue;
175 }
176
177 // Playlist lanes are arrangement-only; mixer inserts are managed in
178 // the mixer and sources keep their own stable destinations.
179 auto trackUI = std::make_shared<TrackUIComponent>(laneId, nullptr, m_trackManager.get());
180
181 // Register callbacks
182 trackUI->setOnSoloToggled([this](TrackUIComponent* soloedTrack) { this->onTrackSoloToggled(soloedTrack); });
183
184 trackUI->setOnCacheInvalidationNeeded([this]() { this->invalidateCache(); });
185
186 trackUI->setOnClipDeleted(
187 [this](TrackUIComponent* trackComp, ClipInstanceID clipId, AestraUI::NUIPoint ripplePos) {
188 this->onClipDeleted(trackComp, clipId, ripplePos);
189 });
190
191 trackUI->setIsSplitToolActive([this]() { return this->m_currentTool == PlaylistTool::Split; });
192
193 trackUI->setOnSplitRequested(
194 [this](TrackUIComponent* trackComp, double splitTime) { this->onSplitRequested(trackComp, splitTime); });
195
196 trackUI->setOnClipSelected([this](TrackUIComponent*, ClipInstanceID clipId) {
197 selectClip(clipId);
198 if (clipId.isValid()) {
199 Log::info("TrackManagerUI: Clip selected " + clipId.toString());
200 // Auto-Picking: Selecting a clip automatically loads it into the clipboard/brush
201 copySelectedClip();
202 }
203 });
204
205 trackUI->setOnPatternClipOpenRequested([this](PatternID patternId) {
206 if (m_onOpenPatternInPianoRoll) {
207 m_onOpenPatternInPianoRoll(patternId);
208 }
209 });
210 trackUI->setOnAudioClipOpenRequested([this](ClipInstanceID clipId) {
211 if (m_onOpenAudioClipEditor) {
212 m_onOpenAudioClipEditor(clipId);
213 }
214 });
215 trackUI->setOnPatternClipDragStarted([this](PatternID patternId) {
216 if (m_onPreviewPatternClip && patternId.isValid()) {
217 m_onPreviewPatternClip(patternId);
218 m_dragPatternPreviewActive = true;
219 }
220 });
221
222 trackUI->setOnTrackSelected([this](TrackUIComponent* trackComp, TrackSelectionIntent intent) {
223 this->selectTrack(trackComp, intent);
224 });
225
226 trackUI->setOnSendToAudition([this, i, lane]() {
227 if (this->m_onSendToAudition) {
228 // Resolve the lane to a mixer channel position: lanes and mixer
229 // channels are separate domains (a lane can be arrangement-only
230 // or its clips routed to any channel), so the lane index is not
231 // a valid channel position (#761 review). Use the first clip's
232 // mixer channel; fall back to the lane index when unresolved.
233 uint32_t channelIndex = static_cast<uint32_t>(i);
234 for (const auto& clip : lane->clips) {
235 if (!clip.patternId.isValid())
236 continue;
237 auto* pattern = this->m_trackManager->getPatternManager().getPattern(clip.patternId);
238 if (!pattern)
239 continue;
240 const uint32_t channelId = pattern->getMixerChannelId();
241 for (size_t c = 0; c < this->m_trackManager->getChannelCount(); ++c) {
242 if (const auto* channel = this->m_trackManager->getChannel(c)) {
243 if (channel->getChannelId() == channelId) {
244 channelIndex = static_cast<uint32_t>(c);
245 break;
246 }
247 }
248 }
249 break;
250 }
251 this->m_onSendToAudition(channelIndex, lane->name);
252 }
253 });
254
255 // Sync zoom/scroll settings
256 trackUI->setPixelsPerBeat(m_pixelsPerBeat);
257 trackUI->setBeatsPerBar(m_beatsPerBar);
258 trackUI->setTimelineScrollOffset(m_timelineScrollOffset);
259 trackUI->setSnapSetting(m_snapSetting); // Sync snap setting for resize
260
261 // Pass platform bridge for cursor capture (volume knob)
262 trackUI->setPlatformBridge(m_window);
263
264 m_trackUIComponents.push_back(trackUI);
265 addChild(trackUI);
266 } // Close lane loop
267
268 std::vector<PlaylistLaneID> validLaneIds;
269 validLaneIds.reserve(m_trackUIComponents.size());
270 for (const auto& trackUI : m_trackUIComponents) {
271 if (trackUI) {
272 validLaneIds.push_back(trackUI->getLaneId());
273 }
274 }
275 m_trackSelection.retainOnly(validLaneIds);
276 syncTrackSelectionView();
277
278 if (m_selectedClipId.isValid() && !playlist.getClip(m_selectedClipId)) {
279 m_selectedClipId = ClipInstanceID{};
280 }
281 selectClip(m_selectedClipId);
282
283 layoutTracks();
284
285 // Mixer strips are now refreshed by AestraContent when syncing state
286
287 // Update scrollbar after tracks are refreshed (fixes initial glitch)
288 scheduleTimelineMinimapRebuild();
289 updateTimelineMinimap(0.0);
290
291 invalidateCache(); // Invalidate cache when tracks refreshed
292
293 Log::info("refreshTracks: completed, created " + std::to_string(m_trackUIComponents.size()) + " TrackUIs");
294}

311void TrackManagerUI::onClipDeleted(TrackUIComponent* trackComp, ClipInstanceID clipId,
312 const AestraUI::NUIPoint& rippleCenter) {
313 if (!trackComp || !m_trackManager || !clipId.isValid())
314 return;
315
316 auto& playlist = m_trackManager->getPlaylistModel();
317 const auto* clip = playlist.getClip(clipId);
318 if (!clip)
319 return;
320
321 // Get clip bounds for animation before we delete
322 AestraUI::NUIRect clipBounds = trackComp->getBounds();
323
324 // Start delete animation
325 DeleteAnimation anim;
326 anim.laneId = trackComp->getLaneId();
327 anim.clipId = clipId;
328 anim.rippleCenter = rippleCenter;
329 anim.clipBounds = clipBounds;
330 anim.progress = 0.0f;
331 anim.duration = 0.25f;
332 m_deleteAnimations.push_back(anim);
333
334 // Core deletion: remove from PlaylistModel using command for undo support
335 auto cmd = std::make_shared<RemoveClipCommand>(playlist, clipId);
336 m_trackManager->getCommandHistory().pushAndExecute(cmd);
337 if (m_selectedClipId == clipId) {
338 m_selectedClipId = ClipInstanceID{};
339 }
340
341 // FL-style transport behavior: if we just cleared the last clip while playing,
342 // snap back to bar 1.
343 if (m_trackManager->isPlaying()) {
344 if (playlist.getTotalDurationBeats() <= 1e-6) {
345 m_trackManager->setPosition(0.0);
346 }
347 }
348
349 refreshTracks();
350 invalidateCache();
351 scheduleTimelineMinimapRebuild();
352
353 Log::info("[TrackManagerUI] Clip deleted via PlaylistModel: " + clipId.toString());
354}

356void TrackManagerUI::onSplitRequested(TrackUIComponent* trackComp, double splitBeat) {
357 if (!trackComp || !m_trackManager)
358 return;
359
360 // Find which clip is at this beat position on this lane
361 auto& playlist = m_trackManager->getPlaylistModel();
362 PlaylistLaneID laneId = trackComp->getLaneId();
363 auto lane = playlist.getLane(laneId);
364 if (!lane)
365 return;
366
367 // Snap split position to grid
368 double snappedBeat = snapBeatToGrid(splitBeat);
369
370 ClipInstanceID targetClipId;
371 for (const auto& clip : lane->clips) {
372 if (snappedBeat > clip.startBeat && snappedBeat < clip.startBeat + clip.durationBeats) {
373 targetClipId = clip.id;
374 break;
375 }
376 }
377
378 if (targetClipId.isValid()) {
379 auto cmd = std::make_shared<Aestra::Audio::SplitClipCommand>(playlist, targetClipId, snappedBeat);
380 m_trackManager->getCommandHistory().pushAndExecute(cmd);
381
382 refreshTracks();
383 invalidateCache();
384 scheduleTimelineMinimapRebuild();
385 Log::info("[TrackManagerUI] Clip split via Command at beat " + std::to_string(snappedBeat));
386 }
387}

399void TrackManagerUI::layoutTracks() {
400 AestraUI::NUIRect bounds = getBounds();
401
402 // Get layout dimensions from theme
403 auto& themeManager = AestraUI::NUIThemeManager::getInstance();
404 const auto& layout = themeManager.getLayoutDimensions();
405
406 float headerHeight = 40.0f;
407 float scrollbarWidth = kTimelineScrollbarWidth;
408 float horizontalScrollbarHeight = kTimelineHorizontalScrollbarHeight;
409 float rulerHeight = kTimelineRulerHeight;
410
411 float viewportHeight = std::max(0.0f, bounds.height - headerHeight - horizontalScrollbarHeight - rulerHeight);
412
413 // In v3.1, panels are floating overlays and do not affect workspace viewport directly.
414 // If we wanted docking, we'd subtract their space here based on external state pointers.
415
416 // Layout timeline minimap (top, right after header, before ruler)
417 if (m_timelineMinimap) {
418 float minimapWidth = std::max(0.0f, bounds.width - scrollbarWidth);
419 float minimapY = headerHeight;
420 m_timelineMinimap->setBounds(
421 AestraUI::NUIAbsolute(bounds, 0, minimapY, minimapWidth, horizontalScrollbarHeight));
422 updateTimelineMinimap(0.0);
423 }
424
425 // Layout vertical scrollbar (right side, below header, horizontal scrollbar, and ruler)
426 if (m_scrollbar) {
427 float scrollbarY = headerHeight + horizontalScrollbarHeight + rulerHeight;
428 float scrollbarX = std::max(0.0f, bounds.width - scrollbarWidth);
429 m_scrollbar->setBounds(AestraUI::NUIAbsolute(bounds, scrollbarX, scrollbarY, scrollbarWidth, viewportHeight));
430 updateScrollbar();
431 }
432
433 float controlAreaWidth = layout.trackControlsWidth;
434 float gridStartX = bounds.x + std::max(0.0f, controlAreaWidth + kTimelineGridInsetX);
435 float trackAreaTop = bounds.y + std::max(0.0f, headerHeight + horizontalScrollbarHeight + rulerHeight);
436
437 // === V3.0 LANE LAYOUT (Two-Rect Model) ===
438 for (size_t i = 0; i < m_trackUIComponents.size(); ++i) {
439 auto trackUI = m_trackUIComponents[i];
440 if (!trackUI)
441 continue;
442
443 float yPos = trackAreaTop + (i * (m_trackHeight + m_trackSpacing)) - m_scrollOffset;
444
445 // Fix: Use absolute coordinates (bounds.x, yPos).
446 // AestraUI components use absolute screen coordinates.
447 float trackWidth = std::max(0.0f, bounds.width - scrollbarWidth - 5.0f);
448 trackUI->setBounds(bounds.x, yPos, trackWidth, m_trackHeight);
449 trackUI->setVisible(m_playlistVisible);
450
451 // Zebra Striping: Ensure index is set during layout (critical for refresh persistence)
452 trackUI->setRowIndex(static_cast<int>(i));
453 }
454
455 // Panels (Mixer, Piano Roll, Sequencer) now live in OverlayLayer
456 // and handle their own layout reacting to visibility changes.
457}

575// Selection query for looping
576std::pair<double, double> TrackManagerUI::getSelectionBeatRange() const {
577 // Priority 1: Ruler selection (for looping)
578 if (m_hasRulerSelection) {
579 double start = std::min(m_rulerSelectionStartBeat, m_rulerSelectionEndBeat);
580 double end = std::max(m_rulerSelectionStartBeat, m_rulerSelectionEndBeat);
581 return {start, end};
582 }
583
584 // Priority 2: Single selected clip
585 if (m_selectedClipId.isValid() && m_trackManager) {
586 const auto* clip = m_trackManager->getPlaylistModel().getClip(m_selectedClipId);
587 if (clip) {
588 return {clip->startBeat, clip->startBeat + clip->durationBeats};
589 }
590 }
591
592 // Priority 3: Selection box / Multi-selection (future)
593 // For now, if no clip is selected, return invalid range
594
595 return {0.0, 0.0};
596}

598void TrackManagerUI::openTrackContextMenu(const ::AestraUI::NUIPoint& position,
599 std::function<void()> onSendToAudition) {
600 if (m_activeContextMenu) {
601 detachContextMenu(m_activeContextMenu);
602 }
603
604 m_activeContextMenu = std::make_shared<AestraUI::NUIContextMenu>();
605 auto menu = m_activeContextMenu;
606
607 TrackUIComponent* selectedTrack = getSelectedTrackUI();
608 auto* lane = selectedTrack && m_trackManager
609 ? m_trackManager->getPlaylistModel().getLane(selectedTrack->getLaneId())
610 : nullptr;
611 if (lane && selectedTrack) {
612 const PlaylistLaneID laneId = lane->id;
613 menu->addCheckbox("Mute Track", lane->muted, [this, laneId](bool muted) {
614 if (auto* target = m_trackManager->getPlaylistModel().getLane(laneId)) {
615 target->muted = muted;
616 m_trackManager->requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged);
617 m_trackManager->markModified();
618 for (const auto& trackUI : m_trackUIComponents) {
619 if (trackUI && trackUI->getLaneId() == laneId) trackUI->updateUI();
620 }
621 invalidateCache();
622 }
623 });
624 menu->addCheckbox("Solo Track", lane->solo, [this, laneId](bool soloed) {
625 if (auto* target = m_trackManager->getPlaylistModel().getLane(laneId)) {
626 target->solo = soloed;
627 m_trackManager->requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged);
628 m_trackManager->markModified();
629 for (const auto& trackUI : m_trackUIComponents) {
630 if (trackUI && trackUI->getLaneId() == laneId) {
631 onTrackSoloToggled(trackUI.get());
632 break;
633 }
634 }
635 }
636 });
637 menu->addSeparator();
638 }
639
640 menu->addItem("Send Track to Audition", [onSendToAudition]() {
641 if (onSendToAudition)
642 onSendToAudition();
