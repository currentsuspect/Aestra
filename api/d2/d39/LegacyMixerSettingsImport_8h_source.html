69inline std::optional<bool> readLegacyInspectorExpanded(const std::string& legacyPath) {
70 std::optional<JSON> root = readJSONStrict(legacyPath);
71 if (!root.has_value() || !root->isObject()) {
72 return std::nullopt;
73 }
74 if (!root->has("inspectorExpanded") || !(*root)["inspectorExpanded"].isBool()) {
75 return std::nullopt;
76 }
77 return (*root)["inspectorExpanded"].asBool();
78}

90inline LegacyImportResult importLegacyMixerInspectorPreference(UISurfaceStoreFile& store,
91 const std::string& legacyPath) {
92 if (store.boolPreference(UISurfaceKeys::kMixerInspectorExpanded).has_value()) {
93 return LegacyImportResult::AlreadyInStore;
94 }
95 const std::optional<bool> legacy = readLegacyInspectorExpanded(legacyPath);
96 if (!legacy.has_value()) {
97 return LegacyImportResult::NothingToImport;
98 }
99 return store.setBoolPreference(UISurfaceKeys::kMixerInspectorExpanded, *legacy) ? LegacyImportResult::Imported
100 : LegacyImportResult::SaveFailed;
101}
