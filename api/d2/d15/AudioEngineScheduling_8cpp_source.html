3// Note scheduling glue between the audio thread and the pattern scheduler.
4//
5// The scheduler (PatternPlaybackEngine::refillWindow) runs on the main loop and
6// queues a quarter second of notes ahead of the playhead. Two things keep that
7// sample-accurate across a loop wrap and a seek:
8// - every loop, pattern or timeline, is planned in a monotonic frame domain
9// (ScheduleLoop): the next pass is queued before the wrap, so the loop top
10// never waits for a UI tick;
11// - the audio thread publishes {queue epoch, frame, loop} as one tuple each
12// block, and the refill plans from exactly that tuple, so a rewind between
13// reading the playhead and stamping events cannot revive stale notes.

24void AudioEngine::refillNoteSchedule() {
25 auto* scheduler = m_patternEngine.load(std::memory_order_acquire);
26 if (!scheduler || !m_transportPlaying.load(std::memory_order_relaxed)) {
27 return;
28 }
29 const uint32_t sampleRate = m_sampleRate.load(std::memory_order_relaxed);
30 if (sampleRate == 0) {
31 return;
32 }
33 const int lookahead = std::max(4096, static_cast<int>(sampleRate / 4)); // 250 ms: see refillWindow() docs
34
35 ScheduleSnapshot snapshot;
36 if (!m_schedulePublication.read(snapshot)) {
37 if (m_schedulePublication.hasPublished()) {
38 return; // the audio thread kept publishing over the read; try next tick
39 }
40 // No block has played yet: plan linearly from the cued playhead. Ticket first,
41 // playhead second (see refillWindow).
42 snapshot = ScheduleSnapshot{};
43 snapshot.epoch = scheduler->refillTicket();
44 snapshot.frame = m_globalSamplePos.load(std::memory_order_relaxed);
45 }
46
47 if (snapshot.epoch == m_lastRefillEpoch && snapshot.loop != m_lastRefillLoop) {
48 // Same queue, new domain: tempo, sample rate, pattern length or the loop braces
49 // changed, or a loop engaged or let go. Everything queued is timed in the old
50 // domain, so rewind; the audio thread republishes under the new epoch next block.
51 scheduler->rewindScheduledInstances();
52 m_lastRefillLoop = snapshot.loop;
53 return;
54 }
55 m_lastRefillLoop = snapshot.loop;
56 m_lastRefillEpoch = snapshot.epoch;
57 scheduler->refillWindow(snapshot.epoch, snapshot.frame, static_cast<int>(sampleRate), lookahead, snapshot.loop);
58}

60ScheduleLoop AudioEngine::resolveScheduleLoopRt(bool patternMode, uint64_t loopStart, uint64_t loopEnd,
61 uint64_t position, bool transportRestart) {
62 ScheduleLoop loop;
63 const bool valid = loopEnd > loopStart;
64 if (patternMode) {
65 // The pattern is the whole world: [0, length), wrapped regardless of where playback began.
66 if (valid) {
67 loop.start = loopStart;
68 loop.length = loopEnd - loopStart;
69 }
70 return loop;
71 }
72
73 uint64_t pass = m_patternLoopIteration.load(std::memory_order_relaxed);
74 if (transportRestart && pass != 0) {
75 // A play or seek placed the playhead absolutely; the pass count starts over.
76 pass = 0;
77 m_patternLoopIteration.store(0, std::memory_order_relaxed);
78 }
79 // Mirrors the wrap test in processBlock: the loop catches a playhead that reaches its
80 // end from inside. Exactly AT the end counts when the playhead got there by advancing.
81 const auto engagedIn = [&](uint64_t p) {
82 if (!valid) {
83 return false;
84 }
85 const bool beforeEnd = position < loopEnd || (position == loopEnd && (p > 0 || !transportRestart));
86 return p == 0 ? beforeEnd : (position >= loopStart && beforeEnd);
87 };
88 if (pass != 0 && !engagedIn(pass)) {
89 // The loop was switched off, or moved off the playhead, mid-pass: the pass count no
90 // longer describes where the playhead is. Back to the linear domain, and drop what
91 // was queued in the old one.
92 pass = 0;
93 m_patternLoopIteration.store(0, std::memory_order_relaxed);
94 if (auto* scheduler = m_patternEngine.load(std::memory_order_relaxed)) {
95 scheduler->rewindScheduledInstances();
96 }
97 }
98 if (!engagedIn(pass)) {
99 return loop; // playing linearly (started past the loop end, or no loop)
100 }
101 loop.start = loopStart;
102 loop.length = loopEnd - loopStart;
103 loop.timeline = true;
104 return loop;
105}

107// Tempo is a timing-domain change too: every position the engine holds is in samples, so
108// switching tempo without rescaling them moved the playhead to a different bar (bar 9 at
109// 120 BPM became bar ~10.3 at 140). While playing, the rescale and the tempo switch land in
110// the same audio block; stopped, nothing advances the playhead, so it is done here.
111double AudioEngine::setTempoKeepingBeat(float bpm) {
112 if (!std::isfinite(bpm) || bpm <= 0.0f) {
113 return 1.0;
114 }
115 // While playing, the audio thread applies tempo changes in order, so each one is measured from the
116 // tempo last SENT, not the one the metronome shows: it may not have taken the previous command yet.
117 // Measured from getBPM(), two edits in one block both rescaled from the same base, and a quick
118 // 120 -> 140 -> 120 hit the "unchanged" shortcut and left the engine at 140.
119 float& requested = m_transportStart.requestedBpm;
120 const bool playing = isTransportPlaying();
121 if (!playing) {
122 requested = 0.0f; // nothing in flight: the metronome is current
123 }
124 const float previous = requested > 0.0f ? requested : getBPM();
125 if (!(previous > 0.0f) || previous == bpm) {
126 if (!playing) {
127 setBPM(bpm);
128 }
129 return 1.0;
130 }
131 const double factor = static_cast<double>(previous) / static_cast<double>(bpm);
132 if (playing) {
133 AudioQueueCommand cmd;
134 cmd.type = AudioQueueCommandType::SetTempoKeepingBeat;
135 cmd.value1 = bpm;
136 cmd.value2 = previous;
137 if (m_commandQueue.pushReliable(cmd)) {
138 requested = bpm;
139 return factor;
140 }
141 Aestra::Log::error("[AudioEngine] tempo change could not reach the audio thread; applying it directly");
142 }
143 const uint64_t position = m_globalSamplePos.load(std::memory_order_relaxed);
144 m_globalSamplePos.store(static_cast<uint64_t>(std::llround(static_cast<double>(position) * factor)),
145 std::memory_order_relaxed);
146 setBPM(bpm);
147 if (auto* scheduler = m_patternEngine.load(std::memory_order_acquire)) {
148 scheduler->rewindScheduledInstances();
149 }
150 return factor;
151}

153uint64_t AudioEngine::applyTempoKeepingBeatRt(float bpm, float previousBpm, uint64_t position) {
154 if (!std::isfinite(bpm) || bpm <= 0.0f || !std::isfinite(previousBpm) || previousBpm <= 0.0f) {
155 return position;
156 }
157 const uint64_t rescaled = static_cast<uint64_t>(
158 std::llround(static_cast<double>(position) * static_cast<double>(previousBpm) / static_cast<double>(bpm)));
159 m_globalSamplePos.store(rescaled, std::memory_order_relaxed);
160 m_metronomeEngine.setBPM(bpm);
161 m_metronomeEngine.reset(rescaled, m_sampleRate.load(std::memory_order_relaxed));
162 if (auto* scheduler = m_patternEngine.load(std::memory_order_relaxed)) {
163 scheduler->rewindScheduledInstances(); // queued notes are timed at the old tempo
164 }
165 return rescaled;
166}

168// Pressing play: the scheduler runs on the main loop, so without this the queue was empty when
169// the first block played and notes at the play position came out a UI frame late (or never,
170// if short). Instead the start is parked: the block plays silence, the playhead stays put and
171// {epoch, frame, loop} is published, so the next refill plans from exactly the start. Playback
172// begins on the first block after that refill lands, bounded at 50 ms so a stalled main loop
173// cannot hold the transport. Only with the live app's opt-in and something scheduled; offline
174// renders and manual renders never park (their timing must stay frame-exact).
175bool AudioEngine::holdStartForScheduleRt(bool transportRestart, float* output, uint32_t numFrames,
176 uint32_t channels) {
177 TransportStartState& start = m_transportStart;
178 auto* scheduler = m_patternEngine.load(std::memory_order_relaxed);
179 const uint32_t sampleRate = m_sampleRate.load(std::memory_order_relaxed);
180 const bool patternMode = m_patternPlaybackMode.load(std::memory_order_relaxed);
181 const uint64_t position = m_globalSamplePos.load(std::memory_order_relaxed);
182 const double samplesPerBeat =
183 (static_cast<double>(sampleRate) * 60.0) / std::max(static_cast<double>(m_metronomeEngine.getBPM()), 1.0);
184 const double loopStartBeat = patternMode ? 0.0 : m_loopStartBeat.load(std::memory_order_relaxed);
185 const double loopEndBeat = patternMode ? m_patternLengthBeats.load(std::memory_order_relaxed)
186 : m_loopEndBeat.load(std::memory_order_relaxed);
187 const uint64_t loopStart = static_cast<uint64_t>(loopStartBeat * samplesPerBeat);
188 const uint64_t loopEnd = static_cast<uint64_t>(loopEndBeat * samplesPerBeat);
189 const bool loopOn = patternMode || m_loopEnabled.load(std::memory_order_relaxed);
190
191 if (transportRestart) {
192 const bool live = start.holdForSchedule.load(std::memory_order_relaxed) &&
193 !m_offlineRenderActive.load(std::memory_order_relaxed) &&
194 m_manualRenderOwner.load(std::memory_order_relaxed) == 0;
195 // A play placed exactly on a timeline loop's end is linear (processBlock's wrap test),
196 // and only the restart block can tell; it is not parked.
197 const bool onLoopEnd = !patternMode && loopOn && position == loopEnd;
198 start.holdFramesLeft = (live && !onLoopEnd && scheduler && scheduler->hasScheduledInstances() && sampleRate > 0)
199 ? sampleRate / 20
200 : 0;
201 if (start.holdFramesLeft > 0 && !patternMode) {
202 m_patternLoopIteration.store(0, std::memory_order_relaxed); // the pass count starts over
203 }
204 }
205 if (start.holdFramesLeft == 0 || !scheduler || !m_transportPlaying.load(std::memory_order_relaxed)) {
206 start.holdFramesLeft = 0;
207 return false;
208 }
209 const uint16_t epoch = scheduler->refillTicket();
210 if (scheduler->plannedEpoch() == epoch) {
211 start.holdFramesLeft = 0; // the queue holds this start's notes: go
212 return false;
213 }
214
215 ScheduleSnapshot snapshot;
216 snapshot.epoch = epoch;
217 if (loopOn && loopEnd > loopStart && (patternMode || position < loopEnd)) {
218 snapshot.loop.start = loopStart;
219 snapshot.loop.length = loopEnd - loopStart;
220 snapshot.loop.timeline = !patternMode;
221 }
222 snapshot.frame = snapshot.loop.active()
223 ? m_patternLoopIteration.load(std::memory_order_relaxed) * snapshot.loop.length + position
224 : position;
225 m_schedulePublication.publish(snapshot);
226
227 if (output && numFrames > 0) {
228 std::memset(output, 0, static_cast<size_t>(numFrames) * std::max<uint32_t>(1, channels) * sizeof(float));
229 }
230 start.holdFramesLeft = numFrames >= start.holdFramesLeft ? 0 : start.holdFramesLeft - numFrames;
231 return true;
232}
