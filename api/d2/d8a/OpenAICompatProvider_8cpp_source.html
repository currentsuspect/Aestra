9ModelResponse OpenAICompatProvider::complete(const ModelRequest& request) {
10 ModelResponse response;
11
12 JSON body = JSON::object();
13 body.set("model", JSON(request.model.empty() ? m_model : request.model));
14 body.set("max_tokens", JSON(static_cast<double>(request.maxTokens)));
15
16 if (!request.tools.empty()) {
17 JSON tools = JSON::array();
18 for (const auto& tool : request.tools) {
19 JSON function = JSON::object();
20 function.set("name", JSON(tool.name));
21 function.set("description", JSON(tool.description));
22 function.set("parameters", tool.parametersSchema);
23 JSON entry = JSON::object();
24 entry.set("type", JSON("function"));
25 entry.set("function", function);
26 tools.push(entry);
27 }
28 body.set("tools", tools);
29 }
30
31 JSON messages = JSON::array();
32 if (!request.systemPrompt.empty()) {
33 JSON systemEntry = JSON::object();
34 systemEntry.set("role", JSON("system"));
35 systemEntry.set("content", JSON(request.systemPrompt));
36 messages.push(systemEntry);
37 }
38 for (const auto& message : request.messages) {
39 JSON entry = JSON::object();
40 if (message.role == "tool") {
41 entry.set("role", JSON("tool"));
42 entry.set("tool_call_id", JSON(message.toolCallId));
43 entry.set("content", JSON(message.text));
44 } else if (message.role == "assistant" && !message.toolCalls.empty()) {
45 entry.set("role", JSON("assistant"));
46 entry.set("content", JSON(message.text));
47 JSON calls = JSON::array();
48 for (const auto& call : message.toolCalls) {
49 JSON function = JSON::object();
50 function.set("name", JSON(call.name));
51 // OpenAI wire format carries arguments as a JSON *string*.
52 JSON arguments = call.arguments;
53 function.set("arguments", JSON(arguments.toString()));
54 JSON callEntry = JSON::object();
55 callEntry.set("id", JSON(call.id));
56 callEntry.set("type", JSON("function"));
57 callEntry.set("function", function);
58 calls.push(callEntry);
59 }
60 entry.set("tool_calls", calls);
61 } else {
62 entry.set("role", JSON(message.role));
63 entry.set("content", JSON(message.text));
64 }
65 messages.push(entry);
66 }
67 body.set("messages", messages);
68
69 std::vector<std::pair<std::string, std::string>> headers;
70 if (!m_apiKey.empty()) {
71 headers.emplace_back("Authorization", "Bearer " + m_apiKey);
72 }
73
74 const HttpResponse http = httpPostJson(m_baseUrl + "/v1/chat/completions", headers,
75 body.toString(), request.timeoutSeconds);
76
77 if (!http.error.empty()) {
78 response.error = "transport: " + http.error;
79 return response;
80 }
81
82 JSON parsed;
83 try {
84 parsed = JSON::parse(http.body);
85 } catch (const std::exception& e) {
86 response.error = "unparseable response: " + std::string(e.what());
87 return response;
88 }
89
90 if (http.status != 200) {
91 std::string detail = http.body;
92 if (parsed.has("error") && parsed["error"].has("message") &&
93 parsed["error"]["message"].isString()) {
94 detail = parsed["error"]["message"].asString();
95 }
96 response.error = "HTTP " + std::to_string(http.status) + ": " + detail;
97 return response;
98 }
99
100 if (!parsed.has("choices") || parsed["choices"].size() == 0) {
101 response.error = "response carried no choices";
102 return response;
103 }
104 JSON& choice = parsed["choices"][0];
105 if (!choice.has("message")) {
106 response.error = "malformed choice (missing \"message\")";
107 return response;
108 }
109 JSON& message = choice["message"];
110 if (message.has("content") && message["content"].isString()) {
111 response.text = message["content"].asString();
112 }
113 if (message.has("tool_calls")) {
114 JSON& calls = message["tool_calls"];
115 for (size_t i = 0; i < calls.size(); ++i) {
116 JSON& callEntry = calls[i];
117 ToolCall call;
118 call.id = callEntry.has("id") && callEntry["id"].isString()
119 ? callEntry["id"].asString()
120 : "call_" + std::to_string(i);
121 if (!callEntry.has("function") || !callEntry["function"].has("name") ||
122 !callEntry["function"]["name"].isString()) {
123 response.error = "malformed tool call (missing function name)";
124 response.stopReason = StopReason::Error;
125 return response;
126 }
127 JSON& function = callEntry["function"];
128 call.name = function["name"].asString();
129 // Arguments arrive as a JSON *string*. Anything else — missing,
130 // non-string, or unparseable — must not silently become an
131 // empty-argument call; reject the response so the loop reports
132 // the provider, not a phantom Muse edit.
133 if (!function.has("arguments") || !function["arguments"].isString()) {
134 response.error = "tool call \"" + call.name +
135 "\" carried missing or non-string arguments";
136 response.stopReason = StopReason::Error;
137 return response;
138 }
139 try {
140 call.arguments = JSON::parse(function["arguments"].asString());
141 } catch (const std::exception& e) {
142 response.error = "tool call \"" + call.name +
143 "\" carried unparseable arguments: " + e.what();
144 response.stopReason = StopReason::Error;
145 return response;
146 }
147 response.toolCalls.push_back(call);
148 }
149 }
150
151 if (parsed.has("usage")) {
152 JSON& usage = parsed["usage"];
153 if (usage.has("prompt_tokens") && usage["prompt_tokens"].isNumber())
154 response.usage.inputTokens =
155 static_cast<long long>(usage["prompt_tokens"].asNumber());
156 if (usage.has("completion_tokens") && usage["completion_tokens"].isNumber())
157 response.usage.outputTokens =
158 static_cast<long long>(usage["completion_tokens"].asNumber());
159 }
160
161 const std::string finish = choice.has("finish_reason") && choice["finish_reason"].isString()
162 ? choice["finish_reason"].asString()
163 : "stop";
164 if (finish == "tool_calls" || !response.toolCalls.empty()) {
165 response.stopReason = StopReason::ToolUse;
166 } else if (finish == "length") {
167 response.stopReason = StopReason::MaxTokens;
168 } else {
169 response.stopReason = StopReason::EndTurn;
170 }
171 return response;
172}
