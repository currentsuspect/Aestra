82 static AudioLatencyInfo calculate(uint32_t bufferFrames, uint32_t sampleRate, double rtlMultiplier = 3.0) {
83 AudioLatencyInfo info;
84 info.actualBufferFrames = bufferFrames;
85 info.sampleRate = sampleRate;
86 info.bufferPeriodMs = (1000.0 * bufferFrames) / sampleRate;
87 info.estimatedRTL_Ms = info.bufferPeriodMs * rtlMultiplier;
88 return info;
89 }
