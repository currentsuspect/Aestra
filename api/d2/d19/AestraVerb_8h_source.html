50#define AESTRA_DIAG_PRECLAMP(l, r) \

52 const float _absL = std::abs(l); \
53 const float _absR = std::abs(r); \
54 const float _pk = _absL > _absR ? _absL : _absR; \
55 if (_pk > m_diagPreClampPeak) m_diagPreClampPeak = _pk; \
56 } while(0)
57#define AESTRA_DIAG_POSTCLAMP(l, r) \

60 const float _absR = std::abs(r); \
61 const float _pk = _absL > _absR ? _absL : _absR; \
62 if (_pk > m_diagPostClampPeak) m_diagPostClampPeak = _pk; \
63 if (std::abs(l) >= 0.9999f || std::abs(r) >= 0.9999f) m_diagClampSampleCount += 2; \
64 m_diagTotalSamples += 2; \
65 } while(0)
66#define AESTRA_DIAG_WETPRE(l, r) \
67 do { \
68 const float _absL = std::abs(l); \
69 const float _absR = std::abs(r); \
70 const float _pk = _absL > _absR ? _absL : _absR; \
71 if (_pk > m_diagWetPreClampPeak) m_diagWetPreClampPeak = _pk; \
72 } while(0)

148 static uint32_t nextPowerOfTwo(uint32_t v) {
149 v--;
150 v |= v >> 1;
151 v |= v >> 2;
152 v |= v >> 4;
153 v |= v >> 8;
154 v |= v >> 16;
155 v++;
156 return v;
157 }

185 void process(const float* const* inputs, float** outputs,
186 uint32_t numInputChannels, uint32_t numOutputChannels,
187 uint32_t numFrames, const MidiBuffer* midiInput = nullptr,
188 MidiBuffer* midiOutput = nullptr) override {
189 (void)midiInput;
190 (void)midiOutput;
191
192 if (!m_active.load(std::memory_order_acquire) ||
193 m_params[kBypass].load(std::memory_order_relaxed) > 0.5f) {
194 copyDry(inputs, outputs, numInputChannels, numOutputChannels, numFrames);
195 return;
196 }
197
198 // Buffer storage is rebuilt only while inactive. Runtime control changes are atomic scalars.
199
200 if (m_predelayL.empty() || m_delayLines[0].empty()) {
201 copyDry(inputs, outputs, numInputChannels, numOutputChannels, numFrames);
202 return;
203 }
204
205 const Mode mode = currentMode();
206 const ModeConstants constants = constantsForMode(mode);
207 const float sampleScale = static_cast<float>(m_sampleRate) / kReferenceSampleRate;
208 const float blockSmoothingCoeff = 1.0f - std::exp(-4.0f / std::max(1.0f, static_cast<float>(m_sampleRate) * 0.015f));
209 static constexpr uint32_t kSmoothBlock = 4;
210 const float lowDampCoeff = 1.0f - std::exp(-kTwoPi * 110.0f / static_cast<float>(m_sampleRate));
211 // Session 006: restore random modulation smoothing coefficient.
212 const float randomCoeff = 1.0f - std::exp(-1.0f / std::max(1.0f, static_cast<float>(m_sampleRate) * 0.22f));
213 const int preMask = m_predelayMask;
214
215 int predelayPos = m_predelayPos;
216 int earlyPos = m_earlyPos;
217 auto delayPos = m_delayPos;
218 auto diffuserPos = m_diffuserPos;
219 auto platePostPos = m_platePostPos;
220 auto dampingState = m_dampingState;
221 auto lowDampState = m_lowDampState;
222 auto lfoPhase = m_lfoPhase;
223 auto lfoPhase2 = m_lfoPhase2;
224 auto lfoSin = m_lfoSin;
225 auto lfoCos = m_lfoCos;
226 auto lfoSin2 = m_lfoSin2;
227 auto lfoCos2 = m_lfoCos2;
228 auto randomMod = m_randomMod;
229 auto randomTarget = m_randomTarget;
230 auto randomCounter = m_randomCounter;
231 auto randomState = m_randomState;
232 auto smoothedParams = m_smoothedParams;
233
234 predelayPos &= m_predelayMask;
235 earlyPos &= m_earlyMask;
236 for (size_t line = 0; line < kFDNLineCount; ++line) {
237 delayPos[line] &= m_delayLineMasks[line];
238 }
239 for (size_t stage = 0; stage < kDiffuserCount; ++stage) {
240 diffuserPos[stage] &= m_diffuserMasks[stage];
241 }
242 for (size_t stage = 0; stage < kPlatePostAllpassCount; ++stage) {
243 platePostPos[stage] &= m_platePostMasks[stage];
244 }
245
246 std::array<float*, kFDNLineCount> delayPtrs{};
247 std::array<int, kFDNLineCount> delayMasks{};
248 for (size_t line = 0; line < kFDNLineCount; ++line) {
249 delayPtrs[line] = m_delayLines[line].data();
250 delayMasks[line] = m_delayLineMasks[line];
251 }
252
253 std::array<float*, kDiffuserCount> diffuserPtrsL{};
254 std::array<float*, kDiffuserCount> diffuserPtrsR{};
255 std::array<int, kDiffuserCount> diffuserMasks{};
256 for (size_t stage = 0; stage < kDiffuserCount; ++stage) {
257 diffuserPtrsL[stage] = m_diffuserL[stage].data();
258 diffuserPtrsR[stage] = m_diffuserR[stage].data();
259 diffuserMasks[stage] = m_diffuserMasks[stage];
260 }
261
262 std::array<float, kFDNLineCount> lineOut{};
263 ControlCache control{};
264 uint32_t controlCountdown = 0;
265
266 uint32_t smoothCountdown = 0;
267 for (uint32_t i = 0; i < numFrames; ++i) {
268 AESTRA_PROFILE_STAGE(kParamSmooth);
269 if (smoothCountdown == 0) {
270 for (uint32_t p = 0; p < kParamCount; ++p) {
271 if (p == kBypass || p == kMode) continue;
272 const float target = m_params[p].load(std::memory_order_relaxed);
273 smoothedParams[p] += (target - smoothedParams[p]) * blockSmoothingCoeff;
274 }
275 smoothCountdown = kSmoothBlock;
276 }
277 --smoothCountdown;
278 AESTRA_PROFILE_STAGE_END(kParamSmooth);
279
280 AESTRA_PROFILE_STAGE(kLFOControl);
281 if (controlCountdown == 0) {
282#ifdef AESTRA_REVERB_HAS_AVX2
283 static const bool useAVX2 =
284 Aestra::Core::CPUDetection::get().hasAVX2() && Aestra::Core::CPUDetection::get().hasFMA();
285 if (useAVX2) {
286 DSP::ReverbSIMD::normalizeLFOsAVX2(lfoSin.data(), lfoCos.data());
287 DSP::ReverbSIMD::normalizeLFOsAVX2(lfoSin2.data(), lfoCos2.data());
288 } else
289#endif
290 {
291 for (size_t line = 0; line < kFDNLineCount; ++line) {
292 normalizeOscillator(lfoSin[line], lfoCos[line]);
293 normalizeOscillator(lfoSin2[line], lfoCos2[line]);
294 }
295 }
296 updateControlCache(control, smoothedParams, constants, mode, sampleScale);
297 controlCountdown = kControlInterval;
298 }
299 --controlCountdown;
300 AESTRA_PROFILE_STAGE_END(kLFOControl);
301
302 AESTRA_PROFILE_STAGE(kInputPrep);
303 const float inL = (numInputChannels > 0 && inputs[0]) ? inputs[0][i] : 0.0f;
304 const float inR = (numInputChannels > 1 && inputs[1]) ? inputs[1][i] : inL;
305 const float dryL = inL;
306 const float dryR = inR;
307
308 m_predelayL[predelayPos] = inL;
309 m_predelayR[predelayPos] = inR;
310 int predelayRead = (predelayPos - control.predelaySamples) & preMask;
311 float delayedL = m_predelayL[predelayRead];
312 float delayedR = m_predelayR[predelayRead];
313 predelayPos = (predelayPos + 1) & preMask;
314 AESTRA_PROFILE_STAGE_END(kInputPrep);
315
316 AESTRA_PROFILE_STAGE(kEarlyReflections);
317 float earlyL = 0.0f;
318 float earlyR = 0.0f;
319 processEarlyReflections(delayedL, delayedR, control, earlyL, earlyR, earlyPos);
320 AESTRA_PROFILE_STAGE_END(kEarlyReflections);
321
322 AESTRA_PROFILE_STAGE(kDiffuser);
323 if (control.diffusionEnabled) {
324#ifdef AESTRA_REVERB_HAS_SSE
325 static const bool useSSE = Aestra::Core::CPUDetection::get().hasSSE41();
326 if (useSSE) {
327 DSP::ReverbSIMD::processDiffusersSSE(delayedL, delayedR, control.diffusionG,
328 diffuserPtrsL.data(), diffuserPtrsR.data(),
329 diffuserPos.data(), diffuserMasks.data(),
330 control.diffuserLengths.data(), kDiffuserCount);
331 } else
332#endif
333 {
334 processDiffusers(delayedL, delayedR, control, diffuserPos);
335 }
336 }
337
338 AESTRA_PROFILE_STAGE_END(kDiffuser);
339
340 delayedL += earlyL * 0.22f;
341 delayedR += earlyR * 0.22f;
342
343 AESTRA_PROFILE_STAGE(kModulationLFO);
344 // Vectorized LFO updates (sin/cos quadrature oscillators)
345 if (control.modulationEnabled) {
346#ifdef AESTRA_REVERB_HAS_AVX2
347 static const bool useAVX2 =
348 Aestra::Core::CPUDetection::get().hasAVX2() && Aestra::Core::CPUDetection::get().hasFMA();
349 if (useAVX2) {
350 DSP::ReverbSIMD::updateLFOsAVX2(
351 lfoSin.data(), lfoCos.data(),
352 lfoSin2.data(), lfoCos2.data(),
353 control.lfoCosInc.data(), control.lfoSinInc.data(),
354 control.lfoCosInc2.data(), control.lfoSinInc2.data());
355 } else
356#endif
357#ifdef AESTRA_REVERB_HAS_SSE
358 {
359 DSP::ReverbSIMD::updateLFOsSSE(lfoSin.data(), lfoCos.data(),
360 control.lfoCosInc.data(), control.lfoSinInc.data());
361 DSP::ReverbSIMD::updateLFOsSSE(&lfoSin[4], &lfoCos[4],
362 &control.lfoCosInc[4], &control.lfoSinInc[4]);
363 DSP::ReverbSIMD::updateLFOsSSE(lfoSin2.data(), lfoCos2.data(),
364 control.lfoCosInc2.data(), control.lfoSinInc2.data());
365 DSP::ReverbSIMD::updateLFOsSSE(&lfoSin2[4], &lfoCos2[4],
366 &control.lfoCosInc2[4], &control.lfoSinInc2[4]);
367 }
368#else
369 {
370 for (size_t line = 0; line < kFDNLineCount; ++line) {
371 const float nextSin = lfoSin[line] * control.lfoCosInc[line] + lfoCos[line] * control.lfoSinInc[line];
372 const float nextCos = lfoCos[line] * control.lfoCosInc[line] - lfoSin[line] * control.lfoSinInc[line];
373 lfoSin[line] = nextSin;
374 lfoCos[line] = nextCos;
375 const float nextSin2 = lfoSin2[line] * control.lfoCosInc2[line] + lfoCos2[line] * control.lfoSinInc2[line];
376 const float nextCos2 = lfoCos2[line] * control.lfoCosInc2[line] - lfoSin2[line] * control.lfoSinInc2[line];
377 lfoSin2[line] = nextSin2;
378 lfoCos2[line] = nextCos2;
379 }
380 }
381#endif
382 }
383
384 AESTRA_PROFILE_STAGE_END(kModulationLFO);
385
386 AESTRA_PROFILE_STAGE(kFDNDelayRead);
387 for (size_t line = 0; line < kFDNLineCount; ++line) {
388 float lfoOffset = 0.0f;
389 if (control.modulationEnabled) {
390 // Session 006: restore random/aperiodic modulation component.
391 // Blend: 0.40 primary LFO + 0.48 secondary LFO + 0.12 random.
392 // The random component provides organic, non-periodic motion.
393 if (--randomCounter[line] <= 0) {
394 randomTarget[line] = nextRandomBipolar(randomState[line]);
395 randomCounter[line] = std::max(64, static_cast<int>((0.075f + 0.013f * static_cast<float>(line)) * static_cast<float>(m_sampleRate)));
396 }
397 randomMod[line] += (randomTarget[line] - randomMod[line]) * randomCoeff;
398 const float multi = lfoSin[line] * 0.40f +
399 lfoSin2[line] * 0.48f +
400 randomMod[line] * 0.12f;
401 lfoOffset = multi * control.modDepthSamples;
402
403 lfoPhase[line] += control.lfoPhaseInc[line];
404 lfoPhase2[line] += control.lfoPhaseInc2[line];
405 if (lfoPhase[line] >= kTwoPi) lfoPhase[line] -= kTwoPi;
406 if (lfoPhase2[line] >= kTwoPi) lfoPhase2[line] -= kTwoPi;
407 }
408 lineOut[line] = readDelayLine(line, delayPos[line], static_cast<float>(control.lineLengths[line]) + lfoOffset);
409 }
410 AESTRA_PROFILE_STAGE_END(kFDNDelayRead);
411
412 AESTRA_PROFILE_STAGE(kFDNFeedbackMatrix);
413 float wetL = 0.0f;
414 float wetR = 0.0f;
415
416 // SIMD-accelerated FDN Householder matrix + feedback + output mixing
417 // Falls back to scalar path on non-SIMD platforms.
418 {
419 DSP::ReverbSIMD::processFDNSample(
420 lineOut.data(), delayedL, delayedR,
421 dampingState.data(), lowDampState.data(),
422 delayPtrs.data(), delayPos.data(), delayMasks.data(),
423 control.feedbackGains.data(),
424 kInjectL.data(), kInjectR.data(),
425 kOutputL.data(), kOutputR.data(),
426 control.dampingCoeff, lowDampCoeff, control.lowDampAmount,
427 wetL, wetR);
428 }
429 AESTRA_PROFILE_STAGE_END(kFDNFeedbackMatrix);
430
431 AESTRA_PROFILE_STAGE(kOutputMix);
432 wetL += earlyL * 0.12f;
433 wetR += earlyR * 0.12f;
434
435 wetL *= kWetMakeupGain;
436 wetR *= kWetMakeupGain;
437
438 // Session 004: per-mode wet compensation for box-cut/air energy loss
439 {
440 const float compGain = (mode == Mode::Room) ? kRoomWetCompGain
441 : (mode == Mode::Hall) ? kHallWetCompGain
442 : kPlateWetCompGain;
443 wetL *= compGain;
444 wetR *= compGain;
445 }
446
447 // Session 004: mud cleanup — gentle one-pole HP filter, mode-dependent
448 {
449 const int modeIdx = static_cast<int>(mode);
450 const float alpha = m_mudHpCoeff[static_cast<size_t>(modeIdx)];
451 const float blend = kMudHpBlend[static_cast<size_t>(modeIdx)];
452 // One-pole HP: y[n] = x[n] - state; state += (1-alpha)*y[n]
453 const float hpL = wetL - m_mudHpStateL;
454 const float hpR = wetR - m_mudHpStateR;
455 m_mudHpStateL += (1.0f - alpha) * hpL;
456 m_mudHpStateR += (1.0f - alpha) * hpR;
457 wetL = wetL + (hpL - wetL) * blend;
458 wetR = wetR + (hpR - wetR) * blend;
459 }
460
461 if (mode == Mode::Plate) {
462 AESTRA_PROFILE_STAGE(kPlatePostAllpass);
463 processPlatePostAllpass(wetL, wetR, platePostPos);
464 AESTRA_PROFILE_STAGE_END(kPlatePostAllpass);
465
466 // Session 012: gentle peaking cut at ~562 Hz to tame metallic ringing.
467 // Direct Form 2 (transposed) biquad — 2 states per channel.
468 const auto& c = m_platePeakCoeff;
469 const float inL = wetL, inR = wetR;
470 wetL = c[0] * inL + m_platePeakZ1L;
471 m_platePeakZ1L = c[1] * inL - c[3] * wetL + m_platePeakZ2L;
472 m_platePeakZ2L = c[2] * inL - c[4] * wetL;
473 wetR = c[0] * inR + m_platePeakZ1R;
474 m_platePeakZ1R = c[1] * inR - c[3] * wetR + m_platePeakZ2R;
475 m_platePeakZ2R = c[2] * inR - c[4] * wetR;
476 wetL = sanitize(wetL);
477 wetR = sanitize(wetR);
478 }
479
480 processWetVoicing(wetL, wetR, mode, control);
481
482 // Late-tail stereo decorrelation (Session 011).
483 // Anti-correlated diff boost: wetL' = wetL + k*(wetL-wetR), wetR' = wetR - k*(wetL-wetR).
484 // Preserves mono fold-down exactly: (wetL+diff) + (wetR-diff) = wetL + wetR.
485 float kDecorr = 0.0f;
486 if (mode == Mode::Room) kDecorr = 0.22f;
487 else if (mode == Mode::Plate) kDecorr = 0.26f;
488 if (kDecorr > 0.0f) {
489 const float diff = (wetL - wetR) * kDecorr;
490 wetL += diff;
491 wetR -= diff;
492 }
493
494 const float widthL = wetL * control.widthMain + wetR * control.widthCross;
495 const float widthR = wetR * control.widthMain + wetL * control.widthCross;
496 wetL = widthL * 0.5f;
497 wetR = widthR * 0.5f;
498
499 wetL = sanitize(wetL);
500 wetR = sanitize(wetR);
501
502 AESTRA_DIAG_WETPRE(wetL, wetR);
503
504 const float sumL = dryL * control.dryGain + wetL * control.wetGain;
505 const float sumR = dryR * control.dryGain + wetR * control.wetGain;
506
507 AESTRA_DIAG_PRECLAMP(sumL, sumR);
508
509 const float outL = sanitize(sumL);
510 const float outR = sanitize(sumR);
511
512 AESTRA_DIAG_POSTCLAMP(outL, outR);
513
514 if (numOutputChannels > 0 && outputs[0]) {
515 outputs[0][i] = outL;
516 }
517 if (numOutputChannels > 1 && outputs[1]) {
518 outputs[1][i] = outR;
519 }
520 AESTRA_PROFILE_STAGE_END(kOutputMix);
521 }

817 void updateControlCache(ControlCache& cache,
818 const std::array<float, kParamCount>& smoothedParams,
819 const ModeConstants& constants,
820 Mode mode,
821 float sampleScale) const {
822 static constexpr std::array<float, kEarlyTapCount> tapMs = {
823 5.1f, 7.9f, 11.3f, 13.7f, 17.1f, 19.9f, 23.5f, 29.7f, 34.1f, 41.9f, 53.3f, 67.7f
824 };
825 static constexpr std::array<float, kEarlyTapCount> tapGain = {
826 0.34f, -0.24f, 0.21f, 0.18f, -0.16f, 0.14f, -0.12f, 0.105f, 0.092f, -0.078f, 0.062f, -0.052f
827 };
828
829 const float sr = static_cast<float>(m_sampleRate);
830 const float size = 0.1f + std::clamp(smoothedParams[kSize], 0.0f, 1.0f) * 1.9f;
831 const float decayTime = (0.3f + smoothedParams[kDecay] * 9.7f) * constants.decayScalar;
832 const float dampingParam = std::clamp(smoothedParams[kDamping], 0.0f, 1.0f);
833 const float damping = std::clamp(dampingParam * constants.dampingScalar, 0.0f, 0.98f);
834 cache.dampingCoeff = 1.0f - damping;
835 cache.lowDampAmount = mode == Mode::Hall ? 0.43f : (mode == Mode::Plate ? 0.42f : 0.48f);
836
837 // Keep Hall darker than Plate, but leave enough top-end through for air after the low-mid cleanup.
838 const float toneCutHz = mode == Mode::Hall
839 ? 8600.0f - dampingParam * 2300.0f
840 : (mode == Mode::Plate ? 9300.0f - dampingParam * 1800.0f : 9600.0f - dampingParam * 3000.0f);
841 cache.wetToneCoeff = 1.0f - std::exp(-kTwoPi * std::clamp(toneCutHz, 4800.0f, 14000.0f) / sr);
842 // Session 006: restore air blend — Session 004 values were too low (~6%),
843 // removing most high-frequency content above the tone cutoff.
844 cache.wetAirBlend = mode == Mode::Hall ? 0.21f : (mode == Mode::Plate ? 0.18f : 0.15f);
845
846 const float predelayMs = std::clamp(smoothedParams[kPredelayMs], 0.0f, 1.0f) * kMaxPredelayMs;
847 cache.predelaySamples = std::clamp(static_cast<int>(std::round((predelayMs / 1000.0f) * sr)),
848 0, std::max(0, m_maxPredelaySamples));
849
850 cache.diffusionG = std::clamp(smoothedParams[kDiffusion] * 0.74f * constants.diffusionScalar, 0.0f, 0.78f);
851 cache.diffusionEnabled = cache.diffusionG > 0.0001f;
852
853 const float modRateScalar = smoothedParams[kModRate] * 2.0f;
854 // Session 006: increase mod depth multiplier from 5.0 to 7.0.
855 // Effective depth at default (0.14): Room 0.14×7×0.45=0.44 smp, Hall 0.14×7×0.64=0.63 smp,
856 // Plate 0.14×7×0.50=0.49 smp. Subtle but audible — avoids the flangey extremes.
857 cache.modDepthSamples = smoothedParams[kModDepth] * 7.0f * constants.modDepthScalar;
858 cache.modulationEnabled = cache.modDepthSamples > 0.0001f && modRateScalar > 0.0001f;
859
860 const float width = std::clamp(smoothedParams[kWidth], 0.0f, 1.0f);
861 cache.widthMain = 1.0f + width;
862 cache.widthCross = 1.0f - width;
863
864 const float mix = std::clamp(smoothedParams[kMix], 0.0f, 1.0f);
865 const float mixAngle = mix * (kTwoPi * 0.25f);
866 cache.dryGain = std::cos(mixAngle);
867 cache.wetGain = std::sin(mixAngle);
868
869 for (size_t line = 0; line < kFDNLineCount; ++line) {
870 cache.lineLengths[line] = logicalFdnLength(line, constants, sampleScale, size);
871 const float delaySeconds = static_cast<float>(cache.lineLengths[line]) / sr;
872 cache.feedbackGains[line] = std::pow(10.0f, -3.0f * delaySeconds / std::max(0.1f, decayTime));
873
874 const float inc = (kTwoPi * kLfoBaseRates[line] * modRateScalar) / sr;
875 cache.lfoPhaseInc[line] = inc;
876 cache.lfoSinInc[line] = std::sin(inc);
877 cache.lfoCosInc[line] = std::cos(inc);
878
879 const float inc2 = (kTwoPi * kLfoSecondaryRates[line] * modRateScalar) / sr;
880 cache.lfoPhaseInc2[line] = inc2;
881 cache.lfoSinInc2[line] = std::sin(inc2);
882 cache.lfoCosInc2[line] = std::cos(inc2);
883 }
884
885 for (size_t stage = 0; stage < kDiffuserCount; ++stage) {
886 cache.diffuserLengths[stage] = logicalDiffuserLength(stage, sampleScale, size);
887 }
888
889 // Hall wants a dense front edge, not an obvious second echo before the tail blooms.
890 const float modeSpread = mode == Mode::Hall ? 1.14f : (mode == Mode::Plate ? 0.55f : 0.82f);
891 const float modeLevel = mode == Mode::Hall ? 0.42f : (mode == Mode::Plate ? 0.40f : 0.68f);
892 const float sizeTerm = std::sqrt(std::max(0.1f, size));
893 const int ringSize = static_cast<int>(m_earlyL.size());
894 const int maxDelay = std::max(1, ringSize - 1);
895 for (size_t tap = 0; tap < kEarlyTapCount; ++tap) {
896 const int delay = std::clamp(
897 static_cast<int>(std::round(tapMs[tap] * modeSpread * sizeTerm * sr / 1000.0f)),
898 1,
899 maxDelay
900 );
901 const int decorrelate = static_cast<int>((tap % 3U) + 1U);
902 cache.earlyDelayL[tap] = delay;
903 cache.earlyDelayR[tap] = std::min(maxDelay, delay + decorrelate);
904 cache.earlyGains[tap] = tapGain[tap] * modeLevel;
905 }
906 }

908 void prepareDelayLines(bool randomizeLfos) {
909 const float sampleScale = static_cast<float>(m_sampleRate) / kReferenceSampleRate;
910
911 m_maxPredelaySamples = std::max(1, static_cast<int>(std::ceil(m_sampleRate * 0.5)));
912 const uint32_t predelayPow2 = nextPowerOfTwo(static_cast<uint32_t>(m_maxPredelaySamples) + 1);

914 m_predelayR.assign(static_cast<size_t>(predelayPow2), 0.0f);
915 m_predelayMask = static_cast<int>(predelayPow2) - 1;
916
917 for (size_t line = 0; line < kFDNLineCount; ++line) {
918 const int maxLength = std::max(
919 128,
920 static_cast<int>(std::ceil(maxFdnBaseForLine(line) * sampleScale * 2.0f)) + 64
921 );

924 m_delayLineMasks[line] = static_cast<int>(pow2Size) - 1;

926 }

930 2,
931 static_cast<int>(std::ceil(kBaseDiffuserLengths[stage] * sampleScale * 2.0f)) + 8
932 );
933 const uint32_t pow2Size = nextPowerOfTwo(static_cast<uint32_t>(maxLength));
934 m_diffuserLengths[stage] = maxLength / 2;
935 m_diffuserL[stage].assign(static_cast<size_t>(pow2Size), 0.0f);
936 m_diffuserR[stage].assign(static_cast<size_t>(pow2Size), 0.0f);
937 m_diffuserMasks[stage] = static_cast<int>(pow2Size) - 1;
938 }

940 for (size_t stage = 0; stage < kPlatePostAllpassCount; ++stage) {
941 const int maxLength = std::max(
942 2,
943 static_cast<int>(std::ceil(kPlatePostAllpassLengths[stage] * sampleScale)) + 4
944 );
945 const uint32_t pow2Size = nextPowerOfTwo(static_cast<uint32_t>(maxLength));
946 m_platePostL[stage].assign(static_cast<size_t>(pow2Size), 0.0f);
947 m_platePostR[stage].assign(static_cast<size_t>(pow2Size), 0.0f);
948 m_platePostDelayLengths[stage] = maxLength;
949 m_platePostMasks[stage] = static_cast<int>(pow2Size) - 1;
950 }

952 const int earlyMaxSamples = std::max(2, static_cast<int>(std::ceil(m_sampleRate * 0.14)));
953 const uint32_t earlyPow2 = nextPowerOfTwo(static_cast<uint32_t>(earlyMaxSamples));
954 m_earlyL.assign(static_cast<size_t>(earlyPow2), 0.0f);
955 m_earlyR.assign(static_cast<size_t>(earlyPow2), 0.0f);
956 m_earlyMask = static_cast<int>(earlyPow2) - 1;
957
958 const float sr = static_cast<float>(m_sampleRate);
959 m_platePeakCoeff = peakingCoefficients(-5.5f, 620.0f, sr, 1.15f);
960 // Session 006: reduce box-cut depth — Session 004 values removed too much body.
961 // Room: -4.6→-3.3 dB, Hall: -3.0→-2.8 dB, Plate: -4.2→-3.3 dB
962 m_boxCutCoeff[0] = peakingCoefficients(-3.3f, 430.0f, sr, 0.85f);
963 m_boxCutCoeff[1] = peakingCoefficients(-2.8f, 520.0f, sr, 0.75f);
964 m_boxCutCoeff[2] = peakingCoefficients(-3.3f, 1250.0f, sr, 1.10f);
965
966 // Session 004: precompute per-mode mud HP filter coefficients (one-pole)
967 for (size_t m = 0; m < 3; ++m) {
968 const float omega = kTwoPi * kMudHpCutoffHz[m] / sr;
969 m_mudHpCoeff[m] = std::exp(-omega);
970 }

979 for (auto& b : m_platePostL) std::fill(b.begin(), b.end(), 0.0f);
980 for (auto& b : m_platePostR) std::fill(b.begin(), b.end(), 0.0f);
981 std::fill(m_earlyL.begin(), m_earlyL.end(), 0.0f);
982 std::fill(m_earlyR.begin(), m_earlyR.end(), 0.0f);
983 std::fill(m_predelayL.begin(), m_predelayL.end(), 0.0f);
984 std::fill(m_predelayR.begin(), m_predelayR.end(), 0.0f);

986 m_diffuserPos.fill(0);
987 m_platePostPos.fill(0);
988 m_dampingState.fill(0.0f);
989 m_lowDampState.fill(0.0f);
990 m_predelayPos = 0;
991 m_earlyPos = 0;
992
993 if (randomizeLfos) {
994 std::mt19937 rng{0xAE57A000u}; // deterministic seed for reproducible tests
995 std::uniform_real_distribution<float> dist(0.0f, kTwoPi);
996 for (auto& phase : m_lfoPhase) phase = dist(rng);
