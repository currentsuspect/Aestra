321 void process(const float* const* inputs, float** outputs,
322 uint32_t numInputChannels, uint32_t numOutputChannels,
323 uint32_t numFrames, const MidiBuffer* midiInput = nullptr,
324 MidiBuffer* midiOutput = nullptr) override {
325 (void)midiInput;
326 (void)midiOutput;
327
328 if (!m_active.load(std::memory_order_acquire) ||
329 m_params[kBypass].load(std::memory_order_relaxed) > 0.5f) {
330 copyDry(inputs, outputs, numInputChannels, numOutputChannels, numFrames);
331 return;
332 }
333
334 // Buffer storage is rebuilt only while inactive. Runtime control changes are atomic scalars.
335
336 if (m_predelayL.empty() || m_delayLines[0].empty()) {
337 copyDry(inputs, outputs, numInputChannels, numOutputChannels, numFrames);
338 return;
339 }
340
341 // F8 tail dormancy: when the whole input block is silent, the wet tail
342 // has already decayed below audibility, and freeze is not holding a tail,
343 // skip the entire wet pipeline for this block and emit only the dry path
344 // (the wet contribution would be inaudible anyway). The FDN state is left
345 // untouched — it is near zero, so processing resumes seamlessly on the
346 // first non-silent block. This makes idle reverb instances nearly free,
347 // which matters most when many are inserted on a low-core target.
348 bool fadeToSleep = false;
349 {
350 bool inputSilent = true;
351 for (uint32_t ch = 0; ch < numInputChannels && inputSilent; ++ch) {
352 if (!inputs[ch]) continue;
353 for (uint32_t i = 0; i < numFrames; ++i) {
354 if (std::abs(inputs[ch][i]) > kSilenceThreshold) { inputSilent = false; break; }
355 }
356 }
357 if (inputSilent) {
358 m_silentSamples = (m_silentSamples > 0xFFFFFFFFu - numFrames) ? 0xFFFFFFFFu
359 : m_silentSamples + numFrames;
360 } else {
361 m_silentSamples = 0;
362 }
363
364 if (!inputSilent) m_sleepFaded = false;
365
366 const bool freezeHeld = m_params[kFreeze].load(std::memory_order_relaxed) > 0.5f;
367 const uint32_t guardSamples = static_cast<uint32_t>(0.75f * static_cast<float>(m_sampleRate));
368 const bool wantSleep = inputSilent && !freezeHeld && m_tailEnv < kDormantThreshold &&
369 m_silentSamples > guardSamples;
370 if (wantSleep && m_sleepFaded) {
371 // Keep the smoothed control state current while asleep. The wet
372 // pipeline is skipped, so its smoothing never runs; without this
373 // an automation move during dormancy (e.g. Mix, Decay, Size) would
374 // leave stale targets and the first wake block would ramp from the
375 // pre-sleep value. Snapping is inaudible here because the output is
376 // silent, and it makes the wake start from the latest automation.
377 for (uint32_t p = 0; p < kParamCount; ++p) {
378 m_smoothedParams[p] = m_params[p].load(std::memory_order_relaxed);
379 }
380
381 // Fully dormant: emit only the dry path (silent input -> silence),
382 // skipping the entire wet pipeline. The FDN state is untouched and
383 // near zero, so the first non-silent block resumes seamlessly.
384 const float mix = std::clamp(m_params[kMix].load(std::memory_order_relaxed), 0.0f, 1.0f);
385 const float dryGain = std::cos(mix * (kTwoPi * 0.25f));
386 for (uint32_t i = 0; i < numFrames; ++i) {
387 const float inL = (numInputChannels > 0 && inputs[0]) ? inputs[0][i] : 0.0f;
388 const float inR = (numInputChannels > 1 && inputs[1]) ? inputs[1][i] : inL;
389 if (numOutputChannels > 0 && outputs[0]) outputs[0][i] = inL * dryGain;
390 if (numOutputChannels > 1 && outputs[1]) outputs[1][i] = inR * dryGain;
391 }
392 return;
393 }
394 // If sleep is wanted but not yet faded, process this block normally
395 // but ramp the wet to zero across it, then sleep next block.
396 fadeToSleep = wantSleep && !m_sleepFaded;
397 }

463 uint32_t controlCountdown = 0;

466 const int modeIdx = static_cast<int>(mode);
467 const float mudAlpha = m_mudHpCoeff[static_cast<size_t>(modeIdx)];
468 const float mudBlend = kMudHpBlend[static_cast<size_t>(modeIdx)];
469 const bool isPlateMode = (mode == Mode::Plate || mode == Mode::SmoothPlate);

472 case Mode::Room: kDecorr = 0.22f; break;
473 case Mode::Plate: kDecorr = 0.26f; break;
474 case Mode::Cathedral: kDecorr = 0.20f; break;

483 uint32_t smoothCountdown = 0;
484 for (uint32_t i = 0; i < numFrames; ++i) {
485 AESTRA_PROFILE_STAGE(kParamSmooth);
486 if (smoothCountdown == 0) {
487 for (uint32_t p = 0; p < kParamCount; ++p) {
488 if (p == kBypass || p == kMode || p == kFreeze) {
489 smoothedParams[p] = m_params[p].load(std::memory_order_relaxed);

493 smoothedParams[p] += (target - smoothedParams[p]) * blockSmoothingCoeff;
494 }

496 }
497 --smoothCountdown;
498 AESTRA_PROFILE_STAGE_END(kParamSmooth);
499
500 AESTRA_PROFILE_STAGE(kLFOControl);
501 if (controlCountdown == 0) {
502#ifdef AESTRA_REVERB_HAS_AVX2
503 static const bool useAVX2 =
504 Aestra::Core::CPUDetection::get().hasAVX2() && Aestra::Core::CPUDetection::get().hasFMA();

506 DSP::ReverbSIMD::normalizeLFOsAVX2(lfoSin.data(), lfoCos.data());
507 DSP::ReverbSIMD::normalizeLFOsAVX2(lfoSin2.data(), lfoCos2.data());
508 } else
509#endif
510 {
511 for (size_t line = 0; line < kFDNLineCount; ++line) {
512 normalizeOscillator(lfoSin[line], lfoCos[line]);
513 normalizeOscillator(lfoSin2[line], lfoCos2[line]);
514 }
515 }
516 updateControlCache(control, smoothedParams, constants, mode, sampleScale);
517 controlCountdown = kControlInterval;
518 }
519 --controlCountdown;
520 AESTRA_PROFILE_STAGE_END(kLFOControl);
521
522 AESTRA_PROFILE_STAGE(kInputPrep);
523 const float inL = (numInputChannels > 0 && inputs[0]) ? inputs[0][i] : 0.0f;
524 const float inR = (numInputChannels > 1 && inputs[1]) ? inputs[1][i] : inL;
525 const float dryL = inL;
526 const float dryR = inR;
527
528 // Freeze: hard-mute the input. The frozen loop is lossless, so any
529 // residual injection (the old 0.001 scale) accumulates without
530 // bound; capture-and-hold is the predictable behavior.
531 const bool freezeActive = smoothedParams[kFreeze] > 0.5f;
532 const float inputScale = freezeActive ? 0.0f : 1.0f;
533
534 m_predelayL[predelayPos] = inL * inputScale;
535 m_predelayR[predelayPos] = inR * inputScale;
536
537 // Predelay: the effective length (manual or BPM-synced) is resolved
538 // at control rate in updateControlCache; just read it here.
539 int predelayRead = (predelayPos - control.predelaySamples) & preMask;
540 float delayedL = m_predelayL[predelayRead];
541 float delayedR = m_predelayR[predelayRead];
542 predelayPos = (predelayPos + 1) & preMask;
543 AESTRA_PROFILE_STAGE_END(kInputPrep);
544
545 AESTRA_PROFILE_STAGE(kEarlyReflections);
546 float earlyL = 0.0f;
547 float earlyR = 0.0f;
548 processEarlyReflections(delayedL, delayedR, control, earlyL, earlyR, earlyPos);
549 AESTRA_PROFILE_STAGE_END(kEarlyReflections);
550
551 AESTRA_PROFILE_STAGE(kDiffuser);
552 if (control.diffusionEnabled) {
553#ifdef AESTRA_REVERB_HAS_SSE
554 static const bool useSSE = Aestra::Core::CPUDetection::get().hasSSE41();
555 if (useSSE) {
556 DSP::ReverbSIMD::processDiffusersSSE(delayedL, delayedR, control.diffusionG,
557 diffuserPtrsL.data(), diffuserPtrsR.data(),
558 diffuserPos.data(), diffuserMasks.data(),
559 control.diffuserLengths.data(), kDiffuserCount);
560 } else
561#endif
562 {
563 processDiffusers(delayedL, delayedR, control, diffuserPos);
564 }
565 }
566
567 AESTRA_PROFILE_STAGE_END(kDiffuser);
568
569 // Early-to-late blend: attack=0 → mostly late (0.08), attack=1 → mostly early (0.44)
570 const float earlyInjectGain = 0.08f + control.earlyBlend * 0.36f;
571 delayedL += earlyL * earlyInjectGain;
572 delayedR += earlyR * earlyInjectGain;
573
574 AESTRA_PROFILE_STAGE(kModulationLFO);
575 // Vectorized LFO updates (sin/cos quadrature oscillators)
576 if (control.modulationEnabled) {
577#ifdef AESTRA_REVERB_HAS_AVX2
578 static const bool useAVX2 =
579 Aestra::Core::CPUDetection::get().hasAVX2() && Aestra::Core::CPUDetection::get().hasFMA();
580 if (useAVX2) {
581 // NOTE argument order: the kernels take (sinInc, cosInc).
582 // These were passed swapped, which rotated every LFO by
583 // ~pi/2 per sample: the "0.3 Hz" modulators actually ran
584 // at sr/4 (12 kHz at 48k), FM-ing all delay lines at audio
585 // rate and imaging the tail's low-mid modes to 11-13 kHz
586 // at ~-30 dB — the audible metallic "top ring".
587 DSP::ReverbSIMD::updateLFOsAVX2(
588 lfoSin.data(), lfoCos.data(),
589 lfoSin2.data(), lfoCos2.data(),
590 control.lfoSinInc.data(), control.lfoCosInc.data(),
591 control.lfoSinInc2.data(), control.lfoCosInc2.data());
592 } else
593#endif
594#ifdef AESTRA_REVERB_HAS_SSE
595 {
596 // Same (sinInc, cosInc) order as above.
597 DSP::ReverbSIMD::updateLFOsSSE(lfoSin.data(), lfoCos.data(),
598 control.lfoSinInc.data(), control.lfoCosInc.data());
599 DSP::ReverbSIMD::updateLFOsSSE(&lfoSin[4], &lfoCos[4],
600 &control.lfoSinInc[4], &control.lfoCosInc[4]);
601 DSP::ReverbSIMD::updateLFOsSSE(lfoSin2.data(), lfoCos2.data(),
602 control.lfoSinInc2.data(), control.lfoCosInc2.data());
603 DSP::ReverbSIMD::updateLFOsSSE(&lfoSin2[4], &lfoCos2[4],
604 &control.lfoSinInc2[4], &control.lfoCosInc2[4]);
605 }
606#else
607 {
608 for (size_t line = 0; line < kFDNLineCount; ++line) {
609 const float nextSin = lfoSin[line] * control.lfoCosInc[line] + lfoCos[line] * control.lfoSinInc[line];
610 const float nextCos = lfoCos[line] * control.lfoCosInc[line] - lfoSin[line] * control.lfoSinInc[line];
611 lfoSin[line] = nextSin;
612 lfoCos[line] = nextCos;
613 const float nextSin2 = lfoSin2[line] * control.lfoCosInc2[line] + lfoCos2[line] * control.lfoSinInc2[line];
614 const float nextCos2 = lfoCos2[line] * control.lfoCosInc2[line] - lfoSin2[line] * control.lfoSinInc2[line];
615 lfoSin2[line] = nextSin2;
616 lfoCos2[line] = nextCos2;
617 }
618 }
619#endif
620 // Advance the chaotic-character oscillator ONCE per sample (not
621 // once per line as the old shared phase did). Three incommensurate
622 // low-frequency accumulators; the per-line decorrelation comes from
623 // fixed phase offsets at read time.
624 chaoticPhaseA += control.chaoticIncA;
625 chaoticPhaseB += control.chaoticIncB;
626 chaoticPhaseC += control.chaoticIncC;
627 if (chaoticPhaseA >= kTwoPi) chaoticPhaseA -= kTwoPi;
628 if (chaoticPhaseB >= kTwoPi) chaoticPhaseB -= kTwoPi;
629 if (chaoticPhaseC >= kTwoPi) chaoticPhaseC -= kTwoPi;
630 }
631
632 AESTRA_PROFILE_STAGE_END(kModulationLFO);
633
634 AESTRA_PROFILE_STAGE(kFDNDelayRead);
635 for (size_t line = 0; line < kFDNLineCount; ++line) {
636 float lfoOffset = 0.0f;
637 if (control.modulationEnabled) {
638 // Session 006: restore random/aperiodic modulation component.
639 // Blend: 0.40 primary LFO + 0.48 secondary LFO + 0.12 random.
640 // The random component provides organic, non-periodic motion.
641 if (--randomCounter[line] <= 0) {
642 randomTarget[line] = nextRandomBipolar(randomState[line]);
643 randomCounter[line] = std::max(64, static_cast<int>((0.075f + 0.013f * static_cast<float>(line)) * srFloat));
644 }
645 randomMod[line] += (randomTarget[line] - randomMod[line]) * randomCoeff;
646
647 // Mod character: 0=Random, 1=Chorus, 2=Chaotic
648 float multi;
649 if (control.modCharacter == 1) {
650 // Chorus: more LFO-driven, pitched modulation
651 multi = lfoSin[line] * 0.62f + lfoSin2[line] * 0.30f + randomMod[line] * 0.08f;
652 } else if (control.modCharacter == 2) {
653 // Chaotic (wow & flutter): read the shared deterministic
654 // oscillator (advanced once per sample above) at this line's
655 // fixed phase offsets, so all eight lines wander below ~2 Hz
656 // but decorrelated. No shared audio-rate phase.
657 //
658 // fastSin only wraps its argument once (valid ~[-PI, 3PI]),
659 // so the phase+offset sum (both in [0, 2PI) → up to 4PI) must
660 // be reduced into [0, 2PI) first; otherwise the polynomial is
661 // evaluated out of domain and returns |value| > 1, blowing the
662 // excursion past its intended bound.
663 float aA = chaoticPhaseA + kChaoticOffsetA[line];
664 float aB = chaoticPhaseB + kChaoticOffsetB[line];
665 float aC = chaoticPhaseC + kChaoticOffsetC[line];
666 if (aA >= kTwoPi) aA -= kTwoPi;
667 if (aB >= kTwoPi) aB -= kTwoPi;
668 if (aC >= kTwoPi) aC -= kTwoPi;
669 multi = Aestra::Audio::FastMath::fastSin(aA) * 0.50f +
670 Aestra::Audio::FastMath::fastSin(aB) * 0.30f +
671 Aestra::Audio::FastMath::fastSin(aC) * 0.20f;
672 } else {
673 // Random: smooth random modulation (default)
674 multi = lfoSin[line] * 0.40f + lfoSin2[line] * 0.48f + randomMod[line] * 0.12f;
675 }
676 lfoOffset = multi * control.modDepthSamples;
677 }
678 // Interleaved by line: [f0l0..f0l7, f1l0..f1l7, ...]
679 AESTRA_MOD_TRACE(lfoOffset);
680 lineOut[line] = readDelayLine(delayPtrs[line], delayMasks[line], delayPos[line], static_cast<float>(control.lineLengths[line]) + lfoOffset);
681 }
682 AESTRA_PROFILE_STAGE_END(kFDNDelayRead);
683
684 AESTRA_PROFILE_STAGE(kFDNFeedbackMatrix);
685 float wetL = 0.0f;
686 float wetR = 0.0f;
687
688 // SIMD-accelerated FDN Householder matrix + feedback + output mixing
689 // Falls back to scalar path on non-SIMD platforms.
690 {
691 DSP::ReverbSIMD::processFDNSample(
692 lineOut.data(), delayedL, delayedR,
693 dampingState.data(), lowDampState.data(),
694 delayPtrs.data(), delayPos.data(), delayMasks.data(),
695 control.feedbackGains.data(),
696 kInjectL.data(), kInjectR.data(),
697 kOutputL.data(), kOutputR.data(),
698 control.dampingCoeff, lowDampCoeff, control.lowDampAmount,
699 wetL, wetR);
700 }
701 AESTRA_PROFILE_STAGE_END(kFDNFeedbackMatrix);
702
703 AESTRA_PROFILE_STAGE(kOutputMix);
704 // Shape: higher attack → sharper early reflections in output
705 const float earlyOutputGain = 0.12f + control.attackShape * 0.15f;
706 wetL += earlyL * earlyOutputGain;
707 wetR += earlyR * earlyOutputGain;
708
709 wetL *= kWetMakeupGain;
710 wetR *= kWetMakeupGain;
711
712 // Session 004: per-mode wet compensation for box-cut/air energy loss
713 {
714 const int modeIdx = static_cast<int>(mode);
715 wetL *= kWetCompGain[modeIdx];
716 wetR *= kWetCompGain[modeIdx];
717 }
718
719 // Session 004: mud cleanup — gentle one-pole HP filter, mode-dependent
720 {
721 // One-pole HP: y[n] = x[n] - state; state += (1-alpha)*y[n]
722 const float hpL = wetL - m_mudHpStateL;
723 const float hpR = wetR - m_mudHpStateR;
724 m_mudHpStateL += (1.0f - mudAlpha) * hpL;
725 m_mudHpStateR += (1.0f - mudAlpha) * hpR;
726 wetL = wetL + (hpL - wetL) * mudBlend;
727 wetR = wetR + (hpR - wetR) * mudBlend;
728 }
729
730 if (isPlateMode) {
731 AESTRA_PROFILE_STAGE(kPlatePostAllpass);
732 processPlatePostAllpass(wetL, wetR, platePostPos);
733 AESTRA_PROFILE_STAGE_END(kPlatePostAllpass);
734
735 // Session 012: gentle peaking cut at ~562 Hz to tame metallic ringing.
736 // Direct Form 2 (transposed) biquad — 2 states per channel.
737 const auto& c = m_platePeakCoeff;
738 const float inL = wetL, inR = wetR;
739 wetL = c[0] * inL + m_platePeakZ1L;
740 m_platePeakZ1L = c[1] * inL - c[3] * wetL + m_platePeakZ2L;
741 m_platePeakZ2L = c[2] * inL - c[4] * wetL;
742 wetR = c[0] * inR + m_platePeakZ1R;
743 m_platePeakZ1R = c[1] * inR - c[3] * wetR + m_platePeakZ2R;
744 m_platePeakZ2R = c[2] * inR - c[4] * wetR;
745 wetL = sanitize(wetL);
746 wetR = sanitize(wetR);
747 }
748
749 processWetVoicing(wetL, wetR, mode, control);
750
751 // Late-tail stereo decorrelation (Session 011).
752 // Anti-correlated diff boost: wetL' = wetL + k*(wetL-wetR), wetR' = wetR - k*(wetL-wetR).
753 // Preserves mono fold-down exactly: (wetL+diff) + (wetR-diff) = wetL + wetR.
754 if (kDecorr > 0.0f) {
755 const float diff = (wetL - wetR) * kDecorr;
756 wetL += diff;
757 wetR -= diff;
758 }
759
760 const float widthL = wetL * control.widthMain + wetR * control.widthCross;
761 const float widthR = wetR * control.widthMain + wetL * control.widthCross;
762 wetL = widthL * 0.5f;
763 wetR = widthR * 0.5f;
764
765 // Post-reverb EQ: Low Cut (one-pole HP) + High Cut (one-pole LP).
766 // Textbook one-pole forms with the alpha from the control cache;
767 // no output blending. (The previous code fed the HP integrator
768 // with exp(-w) instead of alpha — an inverted coefficient — and
769 // crossfaded both filters against the dry wet signal with ad-hoc
770 // alpha-derived amounts, which made the High Cut a dead knob above
771 // ~1.8 kHz and the Low Cut response neither the displayed cutoff
772 // nor a clean high-pass shape.)
773 // The filter states always track the signal, so toggling the
774 // enabled flags at the range extremes is click-free.
775 {
776 // Low Cut one-pole HP: y[n] = x[n] - lp[n]; lp[n] += alpha*y[n]
777 const float hpL = wetL - m_lowCutStateL;
778 const float hpR = wetR - m_lowCutStateR;
779 m_lowCutStateL += control.lowCutCoeff * hpL;
780 m_lowCutStateR += control.lowCutCoeff * hpR;
781 if (control.lowCutEnabled) {
782 wetL = hpL;
783 wetR = hpR;
784 }
785
786 // High Cut one-pole LP: y[n] += alpha * (x[n] - y[n])
787 m_highCutStateL += control.highCutCoeff * (wetL - m_highCutStateL);
788 m_highCutStateR += control.highCutCoeff * (wetR - m_highCutStateR);
789 if (control.highCutEnabled) {
790 wetL = m_highCutStateL;
791 wetR = m_highCutStateR;
792 }
793 }
794
795 wetL = sanitize(wetL);
796 wetR = sanitize(wetR);
797
798 // F8: on the transition-to-dormant block, ramp the wet linearly to
799 // zero across the block so the ~-100 dBFS residual reaches exactly
800 // zero with no step before the pipeline is skipped next block.
801 if (fadeToSleep) {
802 const float g = 1.0f - static_cast<float>(i) / static_cast<float>(numFrames);
803 wetL *= g;
804 wetR *= g;
805 }
806
807 // F8: the post-EQ, pre-mix wet is the tail's actual contribution.
808 blockWetPeak = std::max(blockWetPeak, std::max(std::abs(wetL), std::abs(wetR)));
809
810 AESTRA_DIAG_WET_OUTPUT(wetL, wetR);
811
812 const float sumL = dryL * control.dryGain + wetL * control.wetGain;
813 const float sumR = dryR * control.dryGain + wetR * control.wetGain;
814
815 AESTRA_DIAG_PRE_SANITIZE(sumL, sumR);
816
817 const float outL = sanitize(sumL);
818 const float outR = sanitize(sumR);
819
820 AESTRA_DIAG_POST_SANITIZE(outL, outR);
821
822 if (numOutputChannels > 0 && outputs[0]) {
823 outputs[0][i] = outL;
824 }
825 if (numOutputChannels > 1 && outputs[1]) {
826 outputs[1][i] = outR;
827 }
828 AESTRA_PROFILE_STAGE_END(kOutputMix);
829 }
830
831 m_predelayPos = predelayPos;
832 m_earlyPos = earlyPos;
833 m_delayPos = delayPos;
834 m_diffuserPos = diffuserPos;
835 m_platePostPos = platePostPos;
836 m_dampingState = dampingState;
837 m_lowDampState = lowDampState;
838 m_lfoSin = lfoSin;
839 m_lfoCos = lfoCos;
840 m_lfoSin2 = lfoSin2;
841 m_lfoCos2 = lfoCos2;
842 m_randomMod = randomMod;
843 m_randomTarget = randomTarget;
844 m_randomCounter = randomCounter;
845 m_randomState = randomState;
846 m_smoothedParams = smoothedParams;
847 m_chaoticPhaseA = chaoticPhaseA;
848 m_chaoticPhaseB = chaoticPhaseB;
849 m_chaoticPhaseC = chaoticPhaseC;
850 // F8: remember this block's wet peak; the next block sleeps if this
851 // decayed below the dormant floor (and the input is silent).
852 m_tailEnv = blockWetPeak;
853 if (fadeToSleep) m_sleepFaded = true;
854 }

894 std::string getParameterDisplay(uint32_t id) const override {
895 if (id >= kParamCount) return "";
896 const float v = getParameter(id);
897 switch (id) {
898 case kDecay: {
899 // Truthful display: include the current mode's decay scalar
900 // (0.90-1.60) — the DSP uses (0.3 + v*9.7) * decayScalar, so a
901 // Cathedral "7.1s" was actually 11.4s before this was applied.
902 const float scalar = constantsForMode(currentMode()).decayScalar;
903 return formatCompactFloat(static_cast<int>((0.3f + v * 9.7f) * scalar * 10.0f) / 10.0f, 1) + "s";
904 }
905 case kDamping: return std::to_string(static_cast<int>(v * 100)) + "%";
906 case kPredelayMs: return std::to_string(static_cast<int>(v * kMaxPredelayMs)) + "ms";
907 case kWidth: return std::to_string(static_cast<int>(v * 100)) + "%";
908 case kMix: return std::to_string(static_cast<int>(v * 100)) + "%";
909 case kBypass: return v > 0.5f ? "ON" : "OFF";
910 case kSize: {
911 const float size = 0.1f + v * 1.9f;
912 return formatCompactFloat(static_cast<int>(size * 100.0f) / 100.0f, 2) + "x";
913 }
914 case kDiffusion: return std::to_string(static_cast<int>(v * 100)) + "%";
915 case kModRate: return std::to_string(static_cast<int>(v * 200)) + "%";
916 case kModDepth: {
917 // Truthful display: mirror the DSP's compressive depth curve and
918 // per-mode scalar (updateControlCache). The old "v * 8 smp" text
919 // matched neither the previous x7 mapping nor the current one.
920 const float scalar = constantsForMode(currentMode()).modDepthScalar;
921 const float smp = (v <= 0.0f ? 0.0f : std::pow(v, 0.6f)) * 12.5f * scalar;
922 return formatCompactFloat(static_cast<int>(smp * 10.0f) / 10.0f, 1) + " smp";
923 }
924 case kMode: {
925 static const char* modeNames[] = {
926 "Room", "Hall", "Plate", "Cathedral", "Chamber",
927 "Bright Hall", "Ambience", "Scoring", "Smooth Plate"

931 }
932 case kLowCut: {
933 const float hz = v < 0.001f ? 20.0f : 20.0f * std::pow(100.0f, v);
934 return std::to_string(static_cast<int>(hz)) + "Hz";
935 }
936 case kHighCut: {
937 const float hz = v > 0.999f ? 20000.0f : 200.0f * std::pow(100.0f, v);
938 return std::to_string(static_cast<int>(hz)) + "Hz";
939 }
940 case kFreeze: return v > 0.5f ? "ON" : "OFF";
941 case kAttack: return std::to_string(static_cast<int>(v * 100)) + "%";
942 case kShape: return std::to_string(static_cast<int>(v * 100)) + "%";
943 case kPredelaySync: {
944 static const char* syncNames[] = { "OFF", "1/16", "1/8", "1/4", "1/2", "1 bar", "2 bars" };
945 const int idx = std::clamp(static_cast<int>(std::round(v * static_cast<float>(kPredelaySyncCount - 1))), 0, kPredelaySyncCount - 1);
946 if (idx == 0) return syncNames[0];
947 // Truthful display: the predelay buffer caps at m_maxPredelaySamples
948 // (~500 ms), so long note values at slow tempos can't be realized.
949 // Show the actual resulting time and flag when it is capped, rather
950 // than a nominal division the engine silently clamps. (N3.)
951 const float bpm = std::clamp(m_bpm.load(std::memory_order_relaxed), 20.0f, 999.0f);
952 const float divRatio = static_cast<float>(1 << (idx - 1)) / 4.0f; // beats
953 const float wantMs = (60.0f / bpm) * divRatio * 1000.0f;
954 const float sr = static_cast<float>(m_sampleRate > 0 ? m_sampleRate : 48000);
955 const float maxMs = (m_maxPredelaySamples > 1) ? (static_cast<float>(m_maxPredelaySamples) / sr * 1000.0f)
956 : (kMaxPredelayMs);
957 std::string label = syncNames[idx];
958 if (wantMs > maxMs + 0.5f) {
959 label += " (" + std::to_string(static_cast<int>(std::round(maxMs))) + "ms max)";
960 } else {
961 label += " (" + std::to_string(static_cast<int>(std::round(wantMs))) + "ms)";
962 }
963 return label;
964 }
965 case kModCharacter: {
966 static const char* charNames[] = { "Random", "Chorus", "Chaotic" };
967 const int idx = std::clamp(static_cast<int>(std::round(v * static_cast<float>(kModCharacterCount - 1))), 0, kModCharacterCount - 1);
968 return charNames[idx];
969 }
970 default: return "";
971 }

974 static std::string formatCompactFloat(float value, int precision) {
975 std::ostringstream stream;
976 stream << std::fixed << std::setprecision(precision) << value;
977 std::string text = stream.str();
978 if (text.find('.') != std::string::npos) {

1271 5.1f, 7.9f, 11.3f, 13.7f, 17.1f, 19.9f, 23.5f, 29.7f, 34.1f, 41.9f, 53.3f, 67.7f
1272 };

1274 0.34f, -0.24f, 0.21f, 0.18f, -0.16f, 0.14f, -0.12f, 0.105f, 0.092f, -0.078f, 0.062f, -0.052f
1275 };

1277 const float sr = static_cast<float>(m_sampleRate);
1278 const float size = 0.1f + std::clamp(smoothedParams[kSize], 0.0f, 1.0f) * 1.9f;
1279 const float decayTime = (0.3f + smoothedParams[kDecay] * 9.7f) * constants.decayScalar;
1280 const float dampingParam = std::clamp(smoothedParams[kDamping], 0.0f, 1.0f);
1281 const float damping = std::clamp(dampingParam * constants.dampingScalar, 0.0f, 0.98f);
1282 // 'damping' is the one-pole pole position at the 44.1k reference rate.
1283 // Re-pitch the pole for the actual sample rate (pole' = pole^(refSR/sr))
1284 // so the damping filter's cutoff stays fixed in Hz: without this the
1285 // cutoff scales with the sample rate and T60 drifts ~25% at 96k for
1286 // identical parameters. Identity at the reference rate by construction.
1287 const float dampingPole =
1288 damping <= 0.0f ? 0.0f : std::pow(damping, kReferenceSampleRate / sr);
1289 cache.dampingCoeff = 1.0f - dampingPole;
1290
1291 // Per-mode low damp amount
1292 switch (mode) {
1293 case Mode::Hall: cache.lowDampAmount = 0.43f; break;
1294 case Mode::Plate: cache.lowDampAmount = 0.42f; break;
1295 case Mode::Cathedral: cache.lowDampAmount = 0.44f; break;
1296 case Mode::Chamber: cache.lowDampAmount = 0.46f; break;
1297 case Mode::BrightHall: cache.lowDampAmount = 0.40f; break;
1298 case Mode::Ambience: cache.lowDampAmount = 0.50f; break;
1299 case Mode::Scoring: cache.lowDampAmount = 0.42f; break;
1300 case Mode::SmoothPlate:cache.lowDampAmount = 0.41f; break;
1301 default: cache.lowDampAmount = 0.48f; break;
1302 }
1303
1304 // Per-mode tone cutoff
1305 float toneCutHz;
1306 switch (mode) {
1307 case Mode::Hall: toneCutHz = 8600.0f - dampingParam * 2300.0f; break;
1308 case Mode::Plate: toneCutHz = 9300.0f - dampingParam * 1800.0f; break;
1309 case Mode::Cathedral: toneCutHz = 8200.0f - dampingParam * 2500.0f; break;
1310 case Mode::Chamber: toneCutHz = 9000.0f - dampingParam * 2100.0f; break;
1311 case Mode::BrightHall: toneCutHz = 10500.0f - dampingParam * 1800.0f; break;
1312 case Mode::Ambience: toneCutHz = 10000.0f - dampingParam * 2000.0f; break;
1313 case Mode::Scoring: toneCutHz = 8400.0f - dampingParam * 2400.0f; break;
1314 case Mode::SmoothPlate:toneCutHz = 9100.0f - dampingParam * 1900.0f; break;
1315 default: toneCutHz = 9600.0f - dampingParam * 3000.0f; break;
1316 }
1317 cache.wetToneCoeff = 1.0f - std::exp(-kTwoPi * std::clamp(toneCutHz, 4800.0f, 14000.0f) / sr);
1318
1319 // Per-mode air blend
1320 switch (mode) {
1321 case Mode::Hall: cache.wetAirBlend = 0.21f; break;
1322 case Mode::Plate: cache.wetAirBlend = 0.18f; break;
1323 case Mode::Cathedral: cache.wetAirBlend = 0.19f; break;
1324 case Mode::Chamber: cache.wetAirBlend = 0.17f; break;
1325 case Mode::BrightHall: cache.wetAirBlend = 0.24f; break;
1326 case Mode::Ambience: cache.wetAirBlend = 0.16f; break;
1327 case Mode::Scoring: cache.wetAirBlend = 0.20f; break;

1330 }
1331
1332 const float predelayMs = std::clamp(smoothedParams[kPredelayMs], 0.0f, 1.0f) * kMaxPredelayMs;
1333 cache.predelaySamples = std::clamp(static_cast<int>(std::round((predelayMs / 1000.0f) * sr)),
1334 0, std::max(0, m_maxPredelaySamples));
1335
1336 cache.diffusionG = std::clamp(smoothedParams[kDiffusion] * 0.74f * constants.diffusionScalar, 0.0f, 0.78f);
1337 cache.diffusionEnabled = cache.diffusionG > 0.0001f;
1338
1339 const float modRateScalar = smoothedParams[kModRate] * 2.0f;
1340 // Depth re-range after the SIMD LFO argument-swap fix. The historical
1341 // linear multipliers (Session 006: 5.0 -> 7.0) were tuned against the
1342 // broken modulator, which ran at sr/4 instead of sub-Hz — what was
1343 // audible then was the FM artifact, not chorusing. With the real
1344 // modulator, x7 measured 0.18 cents of pitch motion at the default
1345 // depth and only 0.9 cents at MAX (audibility ~2-5 cents), and gave no
1346 // small-room mode smearing (47 dB -> 46 dB ring prominence).
1347 // The measured wobble-vs-depth response is strongly superlinear, so a
1348 // linear multiplier cannot make the default audible without pushing
1349 // max into flange territory (x24: max 10.2 cents). A compressive knob
1350 // curve (depth^0.6 x 12.5) lands: default (0.14) 0.46 cents (present,
1351 // subtle), 0.6 depth ~1.9 cents, max 3.9 cents — classic hall chorus,
1352 // not flange (all measured, 1 kHz zero-crossing wobble p5..p95).
1353 // Max read offset is 12.5 x 0.70 (largest mode scalar) = +/-8.75 smp;
1354 // the logicalFdnLength margin (28) covers it with room to spare, and
1355 // the minimum length (100) keeps the shortest modulated delay clear of
1356 // the write head. pow() runs at control rate, not per sample.
1357 // Default depth retuned 0.14 -> 0.07 (owner-confirmed by ear). The
1358 // audible "tremolo/sheet" was the FDN's narrow peaks/notches being MOVED
1359 // across sustained source harmonics by the LFO (individual harmonics
1360 // swing even though the broadband tail loudness barely changes). Measured
1361 // per-harmonic swing dropped from ~2.7 dB (Plate) / ~2.3 dB (Room) at 0.14
1362 // to ~2.0 / ~1.7 dB at 0.07 — below the tremolo audibility knee — while
1363 // still smearing the static modes (swing at depth 0 collapses to ~0.1 dB,
1364 // re-exposing the ring). See labs/reverb/tremolo/. Higher settings remain
1365 // available for anyone who wants overt motion.
1366 const float depthParam = std::clamp(smoothedParams[kModDepth], 0.0f, 1.0f);
1367 const float depthCurve = depthParam <= 0.0f ? 0.0f : std::pow(depthParam, 0.6f);
1368 cache.modDepthSamples = depthCurve * 12.5f * constants.modDepthScalar;
1369 cache.modulationEnabled = cache.modDepthSamples > 0.0001f && modRateScalar > 0.0001f;
1370
1371 const float width = std::clamp(smoothedParams[kWidth], 0.0f, 1.0f);
1372 cache.widthMain = 1.0f + width;
1373 cache.widthCross = 1.0f - width;
1374
1375 const float mix = std::clamp(smoothedParams[kMix], 0.0f, 1.0f);
1376 const float mixAngle = mix * (kTwoPi * 0.25f);
1377 cache.dryGain = std::cos(mixAngle);
1378 cache.wetGain = std::sin(mixAngle);
1379
1380 for (size_t line = 0; line < kFDNLineCount; ++line) {
1381 cache.lineLengths[line] = logicalFdnLength(line, constants, sampleScale, size);
1382 const float delaySeconds = static_cast<float>(cache.lineLengths[line]) / sr;
1383 cache.feedbackGains[line] = std::pow(10.0f, -3.0f * delaySeconds / std::max(0.1f, decayTime));
1384
1385 const float inc = (kTwoPi * kLfoBaseRates[line] * modRateScalar) / sr;
1386 cache.lfoSinInc[line] = std::sin(inc);
1387 cache.lfoCosInc[line] = std::cos(inc);
1388
1389 const float inc2 = (kTwoPi * kLfoSecondaryRates[line] * modRateScalar) / sr;
1390 cache.lfoSinInc2[line] = std::sin(inc2);
1391 cache.lfoCosInc2[line] = std::cos(inc2);
1392 }
1393
1394 // Chaotic-character oscillator: three incommensurate low-frequency
1395 // components, base rate scaled by Mod Rate and normalized by sample rate.
1396 // Advanced once per sample (not once per line) so the frequency is
1397 // sample-rate-independent and stays well below 10 Hz.
1398 const float chaoticHz = kChaoticBaseHz * modRateScalar;
1399 cache.chaoticIncA = (kTwoPi * chaoticHz) / sr;
1400 cache.chaoticIncB = (kTwoPi * chaoticHz * kChaoticRatioB) / sr;
1401 cache.chaoticIncC = (kTwoPi * chaoticHz * kChaoticRatioC) / sr;
1402
1403 // Freeze: make the feedback loop lossless so the tail sustains
1404 // indefinitely. Unity feedback gains alone are NOT enough — the
1405 // in-loop damping low-pass and the low-damp shelf subtraction each
1406 // bleed energy every pass (measured ~22 dB decay over 6 s). Damping
1407 // coeff 1.0 makes the LP transparent (state snaps to input) and
1408 // lowDampAmount 0 disables the shelf, leaving only the orthogonal
1409 // Householder matrix in the loop.
1410 cache.freezeActive = smoothedParams[kFreeze] > 0.5f;
1411 if (cache.freezeActive) {
1412 cache.feedbackGains.fill(1.0f);
1413 cache.dampingCoeff = 1.0f;
1414 cache.lowDampAmount = 0.0f;
1415 // Freeze the modulation as well: with modulation active the
1416 // fractional cubic-Hermite delay reads are not energy-exact and
1417 // the unity loop slowly GROWS (~+1 dB/s measured). Zero modulation
1418 // makes every read integer-exact, so the loop is truly lossless
1419 // and the frozen tail holds level indefinitely.
1420 cache.modulationEnabled = false;
1421 }
1422
1423 for (size_t stage = 0; stage < kDiffuserCount; ++stage) {
1424 cache.diffuserLengths[stage] = logicalDiffuserLength(stage, sampleScale, size);
1425 }

1427 // Hall wants a dense front edge, not an obvious second echo before the tail blooms.
1428 // Per-mode early reflection spread and level
1429 float modeSpread;
1430 float modeLevel;
1431 switch (mode) {
1432 case Mode::Hall: modeSpread = 1.14f; modeLevel = 0.42f; break;
1433 case Mode::Plate: modeSpread = 0.55f; modeLevel = 0.40f; break;
1434 case Mode::Cathedral: modeSpread = 1.28f; modeLevel = 0.38f; break;
1435 case Mode::Chamber: modeSpread = 0.90f; modeLevel = 0.52f; break;
1436 case Mode::BrightHall: modeSpread = 1.10f; modeLevel = 0.44f; break;
1437 case Mode::Ambience: modeSpread = 0.65f; modeLevel = 0.72f; break;
1438 case Mode::Scoring: modeSpread = 1.22f; modeLevel = 0.40f; break;
1439 case Mode::SmoothPlate:modeSpread = 0.58f; modeLevel = 0.42f; break;
1440 default: modeSpread = 0.82f; modeLevel = 0.68f; break;
1441 }

1682 const int i2 = (readPosI + 1) & mask;

1720 pos &= mask;
1721
1722 m_earlyL[static_cast<size_t>(pos)] = left;
1723 m_earlyR[static_cast<size_t>(pos)] = right;

1729 earlyL += m_earlyL[static_cast<size_t>(readL)] * gain +

1841 // Compute peaking EQ coefficients (b0,b1,b2,a1,a2 with a0 normalized to 1).
1842 // dbGain: positive = boost, negative = cut. Q: bandwidth at -3dB (or -dBgain/2 for peaking).
1843 // Uses standard RBJ Audio EQ Cookbook formula (same equation works for boost and cut).
1844 static std::array<float, 5> peakingCoefficients(float dbGain, float freqHz, float sampleRate, float Q) {
1845 const float A = std::sqrt(std::pow(10.0f, dbGain / 20.0f));
1846 const float w0 = 2.0f * static_cast<float>(M_PI) * freqHz / sampleRate;
1847 const float cosw0 = std::cos(w0);
1848 const float sinw0 = std::sin(w0);
1849 const float alpha = sinw0 / (2.0f * Q);
1850
1851 const float b0 = 1.0f + alpha * A;
1852 const float b1 = -2.0f * cosw0;
1853 const float b2 = 1.0f - alpha * A;
1854 const float a0 = 1.0f + alpha / A;
1855 const float a1 = -2.0f * cosw0;
1856 const float a2 = 1.0f - alpha / A;
1857
1858 const float invA0 = 1.0f / a0;
1859 return {b0 * invA0, b1 * invA0, b2 * invA0, a1 * invA0, a2 * invA0};
1860 }
