26inline std::vector<Audio::TakesPanel::StepEntry> versionStepEntries(const Audio::CommandHistory& history) {
27 std::vector<Audio::TakesPanel::StepEntry> out;
28 const auto name = [](const std::shared_ptr<Audio::ICommand>& c) {
29 std::string n = c ? c->getName() : std::string();
30 return n.empty() ? std::string("(unnamed)") : n;
31 };
32 const auto& redo = history.getRedoStack();
33 const auto& undo = history.getUndoStack();
34 for (int i = 0; i < static_cast<int>(redo.size()); ++i) out.push_back({name(redo[static_cast<size_t>(i)]), true, i, redo[static_cast<size_t>(i)]->createdAtMs()});
35 for (int i = static_cast<int>(undo.size()) - 1; i >= 0; --i) out.push_back({name(undo[static_cast<size_t>(i)]), false, i, undo[static_cast<size_t>(i)]->createdAtMs()});
36 return out;
37}

40inline void goToVersionStep(Audio::CommandHistory& history, const Audio::TakesPanel::StepEntry& step) {
41 if (step.undone) history.redoTo(step.stackIndex);
42 else history.undoTo(step.stackIndex + 1);
43}

46inline int unsavedStepCount(const Audio::CommandHistory& history, const TakeManager::Manifest& manifest) {
47 const auto* open = manifest.ok ? manifest.activeTake() : nullptr;
48 const int recorded = open ? static_cast<int>(open->steps.size()) : 0;
49 return std::max(0, static_cast<int>(history.getUndoStack().size()) - recorded);
50}

53inline void wireVersionsDock(Audio::TakesPanel& panel, std::function<Audio::CommandHistory&()> history,
54 std::function<std::string()> projectPath, std::function<std::string()> projectName,
55 std::function<void()> afterStep = nullptr) {
56 panel.setProjectNameProvider(std::move(projectName));
57 panel.setStepsProvider([history]() { return versionStepEntries(history()); });
58 panel.setOnStepActivated([history, afterStep](const Audio::TakesPanel::StepEntry& e) { goToVersionStep(history(), e); if (afterStep) afterStep(); });
59 panel.setUnsavedProvider([history, projectPath]() {
60 const std::string path = projectPath();
61 return path.empty() ? 0 : unsavedStepCount(history(), TakeManager::loadManifest(path));
62 });
63}

91inline void wireBringOver(Audio::TakesPanel& panel, std::function<std::string()> projectPath,
92 std::function<Audio::TrackManager*()> currentProject, std::function<bool(const std::string&)> afterBringOver) {
93 struct Session { std::shared_ptr<Audio::TrackManager> other; };
94 auto session = std::make_shared<Session>();
95 panel.setBringOverProvider([=](const std::string& takeId) -> std::optional<BringOverCard> {
96 const std::string path = projectPath();
97 Audio::TrackManager* current = currentProject();
98 const auto manifest = TakeManager::loadManifest(path);
99 const auto* take = manifest.ok ? manifest.findTake(takeId) : nullptr;
100 const auto* open = manifest.ok ? manifest.activeTake() : nullptr;
101 if (!take || !open || !current) return std::nullopt;
102 const std::string snapshot = TakeManager::resolveSnapshotPath(path, *take);
103 auto scratch = std::make_shared<Audio::TrackManager>();
104 scratch->getPlaylistModel().setPatternManager(&scratch->getPatternManager());
105 if (snapshot.empty() || !ProjectSerializer::load(snapshot, scratch, path).ok) return std::nullopt;
106 const std::string from = take->name.empty() ? take->id : take->name;
107 BringOverCard card;
108 card.from = from;
109 card.into = open->name.empty() ? open->id : open->name;
110 const std::string bounce = TakeManager::resolveBouncePath(path, *take);
111 card.canListen = !bounce.empty() && std::filesystem::exists(bounce);
112 card.listenPath = bounce;
113 const auto plan = BringOver::plan(*scratch, *current, from);
114 const auto add = [&](const BringOver::Item& item, bool pattern) {
115 BringOverCard::Row row;
116 row.name = item.name;
117 row.detail = item.detail;
118 row.status = static_cast<BringOverCard::Status>(item.status);
119 row.checked = BringOver::defaultChecked(item);
120 row.pattern = pattern;
121 row.id = item.id;
122 card.rows.push_back(std::move(row));
123 };
124 for (const auto& item : plan.tracks) add(item, false);
125 for (const auto& item : plan.patterns) add(item, true);
126 session->other = std::move(scratch);
127 return card;
128 });
129 panel.setOnBringOver([=](const std::string&, const std::vector<BringOverCard::Row>& chosen) {
130 Audio::TrackManager* current = currentProject();
131 if (!session->other || !current) return false;
132 std::vector<uint64_t> tracks, patterns;
133 for (const auto& row : chosen) (row.pattern ? patterns : tracks).push_back(row.id);
134 auto command = std::make_shared<BringOver::Command>(*current, *session->other, tracks, patterns);
135 current->getCommandHistory().pushAndExecute(command);
136 session->other.reset();
137 return afterBringOver ? afterBringOver("") : true;
138 });
139}
