2// AestraLimit V1 — musical brickwall limiter with density-driven auto-release.
3//
4// OPTFLOOR: AestraLimit 2026-06-11
5// CPU: ~100 cycles/sample @ 48 kHz, 512-sample buffer
6// Breakdown: log10 ~20c, exp (gain) ~20c, exp (release) ~20c, upsample 4x ~16c, misc ~24c
7// Memory: 3.5 KB hot working set per instance (840 bytes L1-active at 48 kHz)
8// Cache: fits in L1 at any instance count; fits in L2 up to 300+ instances
9// Transcendental floor: log10 and exp are accuracy-critical for gain computation;
10// replacing with approximations introduces > 0.1 dB error — unacceptable for brickwall limiting.
11// SIMD: N/A — lookahead is a circular buffer (write 1, read 1), not a batch scan.
12// 4-sample peak detection loop is below SIMD threshold.
13// Denormals: handled by engine-level FTZ+DAZ. EMA state zeroed on reset/bypass.
14// Next review: when sample rate exceeds 192 kHz or lookahead exceeds 2 ms.

83 void process(const float* const* inputs, float** outputs,
84 uint32_t numInputChannels, uint32_t numOutputChannels,
85 uint32_t numFrames, const MidiBuffer* midiInput = nullptr,
86 MidiBuffer* midiOutput = nullptr) override {
87 (void)midiInput;
88 (void)midiOutput;
89
90 if (!m_active.load(std::memory_order_relaxed) ||
91 m_params[kBypass].load(std::memory_order_relaxed) > 0.5f) {
92 copyOrClear(inputs, outputs, numInputChannels, numOutputChannels, numFrames);
93 return;
94 }
95
96 const uint32_t channels = std::min<uint32_t>(2, numOutputChannels);
97 const bool stereo = channels >= 2;
98
99 // Cache per-block constants (avoids repeated atomic loads and transcendental calls)
100 const bool autoRelease = m_params[kReleaseMode].load(std::memory_order_relaxed) <= 0.5f;
101 const float clipLevel = m_clipLevel;
102
103 float blockInputPeak = 0.0f;
104 float blockOutputPeak = 0.0f;
105
106 for (uint32_t i = 0; i < numFrames; ++i) {
107 const float inL = sanitizeSample(readInput(inputs, numInputChannels, 0, i));
108 const float inR = stereo ? sanitizeSample(readInput(inputs, numInputChannels, 1, i)) : inL;
109
110 // Update smoothed params
111 const float targetCeil = getParameter(kCeiling);
112 const float targetRel = getParameter(kRelease);
113 m_ceilingSmoothed += (targetCeil - m_ceilingSmoothed) * m_smoothCoeff;
114 m_releaseSmoothed += (targetRel - m_releaseSmoothed) * m_smoothCoeff;
115 if (autoRelease)
116 computeAutoReleaseMs();
117
118 // --- True Peak Detection (per channel, 4x oversample) ---
119 float osValsL[kOsRatio];
120 upsampleOs(m_tpUpsampleL, inL, osValsL);
121 float peakL = 0.0f;
122 for (uint32_t p = 0; p < kOsRatio; ++p)
123 peakL = std::max(peakL, std::abs(osValsL[p]));
124
125 float blockPeak = peakL;
126 if (stereo) {
127 float osValsR[kOsRatio];
128 upsampleOs(m_tpUpsampleR, inR, osValsR);
129 float peakR = 0.0f;
130 for (uint32_t p = 0; p < kOsRatio; ++p)
131 peakR = std::max(peakR, std::abs(osValsR[p]));
132 blockPeak = std::max(peakL, peakR);
133 }
134
135 // --- Density Analysis (mono sum) ---
136 const float monoSum = (inL + inR) * 0.5f;
137 updateDensity(monoSum);
138
139 // --- Lookahead Write ---
140 m_lookaheadBuf[0][m_writeCursor] = inL;
141 if (stereo)
142 m_lookaheadBuf[1][m_writeCursor] = inR;
143 m_writeCursor = (m_writeCursor + 1) % m_lookaheadSize;
144
145 // --- Gain Computer ---
146 const float peakDbTP = linearToDb(blockPeak);
147 const float userCeilingDb = ceilingDbFromNorm(m_ceilingSmoothed);
148 const float internalCeilingDb = userCeilingDb - kCeilingMarginDb;
149
150 float targetGainDb;
151 if (peakDbTP > userCeilingDb) {
152 targetGainDb = internalCeilingDb - peakDbTP;
153 } else {
154 targetGainDb = 0.0f;
155 }
156
157 // --- Release Smoother ---
158 if (targetGainDb < m_appliedGainDb) {
159 m_appliedGainDb = targetGainDb;
160 } else {
161 const float releaseMs = autoRelease
162 ? m_autoReleaseMs
163 : releaseMsFromNorm(m_releaseSmoothed);
164 const float releaseCoeff = std::exp(-m_oneOverSampleRateMs / releaseMs);
165 m_appliedGainDb = releaseCoeff * m_appliedGainDb
166 + (1.0f - releaseCoeff) * targetGainDb;
167 }
168
169 // --- Lookahead Read ---
170 const uint32_t readOffset = (m_writeCursor + m_lookaheadSize - m_lookaheadSamples) % m_lookaheadSize;
171 const float delayedL = m_lookaheadBuf[0][readOffset];
172 const float delayedR = stereo ? m_lookaheadBuf[1][readOffset] : delayedL;
173
174 // --- VCA ---
175 const float gainLinear = dbToLinear(m_appliedGainDb);
176 const float vcaOutL = delayedL * gainLinear;
177 const float vcaOutR = delayedR * gainLinear;
178
179 // --- Output Safety Clip ---
180 const float outL = std::clamp(vcaOutL, -clipLevel, clipLevel);
181 const float outR = stereo ? std::clamp(vcaOutR, -clipLevel, clipLevel) : outL;
182
183 blockInputPeak = std::max(blockInputPeak, blockPeak);
184 blockOutputPeak = std::max(blockOutputPeak, std::max(std::abs(outL), std::abs(outR)));
185
186 if (numOutputChannels > 0 && outputs[0]) outputs[0][i] = outL;
187 if (numOutputChannels > 1 && outputs[1]) outputs[1][i] = outR;
188 for (uint32_t ch = 2; ch < numOutputChannels; ++ch) {
189 if (outputs[ch]) outputs[ch][i] = 0.0f;
190 }
191 }
192
193 m_inputLevel.store(blockInputPeak, std::memory_order_relaxed);
194 m_outputLevel.store(blockOutputPeak, std::memory_order_relaxed);
195 m_gainReduction.store(std::max(0.0f, -m_appliedGainDb), std::memory_order_relaxed);
196 }

232 struct Blob {

236 } blob;

244 bool loadState(const std::vector<uint8_t>& state) override {
245 if (state.size() < sizeof(uint32_t) * 2) return false;
246 uint32_t magic = 0;
247 std::memcpy(&magic, state.data(), sizeof(magic));
248 if (magic != kStateMagic) return false;
249 if (state.size() < sizeof(uint32_t) * 2 + sizeof(float) * kParamCount) return false;
250 struct Blob {
251 uint32_t magic;
252 uint32_t version;
253 float params[kParamCount];
254 };
255 Blob blob{};
256 std::memcpy(&blob, state.data(), sizeof(blob));
257 if (blob.version < 1 || blob.version > 1) return false;
258 for (uint32_t i = 0; i < kParamCount; ++i) {
259 setParameter(i, blob.params[i]);
260 }
261 return true;
262 }

288private:
289 // ── OS Coefficients: 4-point Lagrange interpolation at 4x ──
290 static constexpr float kOsCoeffs[kOsRatio * kOsTapsPerPhase] = {
291 // Phase 0 (f=0.0)
292 0.0f, 1.0f, 0.0f, 0.0f,
293 // Phase 1 (f=0.25)
294 -0.0546875f, 0.8203125f, 0.2734375f, -0.0390625f,
295 // Phase 2 (f=0.5)
296 -0.0625f, 0.5625f, 0.5625f, -0.0625f,
297 // Phase 3 (f=0.75)
298 -0.0390625f, 0.2734375f, 0.8203125f, -0.0546875f,
299 };
300
301 struct UpsampleState {
302 float hist[kOsTapsPerPhase]{};
303 };
304
305 static void upsampleOs(UpsampleState& st, float input, float* output) {
306 for (int32_t i = kOsTapsPerPhase - 1; i > 0; --i)
307 st.hist[i] = st.hist[i - 1];
308 st.hist[0] = input;
309
310 for (uint32_t p = 0; p < kOsRatio; ++p) {
311 float sum = 0.0f;
312 const float* phaseCoeffs = &kOsCoeffs[p * kOsTapsPerPhase];
313 for (uint32_t t = 0; t < kOsTapsPerPhase; ++t) {
314 sum += phaseCoeffs[t] * st.hist[t];
315 }
316 output[p] = sum;
317 }
318 }

320 void updateDensity(float monoSum) {
321 const float inputSq = monoSum * monoSum;
322 m_shortEma = m_alphaShort * inputSq + (1.0f - m_alphaShort) * m_shortEma;
323 m_longEma = m_alphaLong * inputSq + (1.0f - m_alphaLong) * m_longEma;
324 }
325
326 void computeAutoReleaseMs() {
327 const float density = (m_longEma > 1e-10f) ? (m_shortEma / m_longEma) : 0.0f;
328
329 static constexpr float kDenseThresh = 0.7f;
330 static constexpr float kSparseThresh = 0.3f;
331 static constexpr float kReleaseMaxMs = 300.0f;
332 static constexpr float kReleaseMinMs = 60.0f;
333
334 if (density >= kDenseThresh) {
335 m_autoReleaseMs = kReleaseMaxMs;
336 } else if (density <= kSparseThresh) {
337 m_autoReleaseMs = kReleaseMinMs;
338 } else {
339 const float t = (density - kSparseThresh) / (kDenseThresh - kSparseThresh);
340 m_autoReleaseMs = kReleaseMinMs + t * (kReleaseMaxMs - kReleaseMinMs);
341 }
342 }
343
344 void updateLookaheadSize() {
345 const uint32_t newSize = std::max(1u, std::min(kMaxLookaheadSamples,
346 static_cast<uint32_t>(std::ceil(kLookaheadMs * 0.001 * m_sampleRate))));
347 if (newSize != m_lookaheadSize) {
348 m_lookaheadSize = newSize;
349 m_lookaheadSamples = newSize;
350 std::fill_n(m_lookaheadBuf[0].begin(), m_lookaheadSize, 0.0f);
351 std::fill_n(m_lookaheadBuf[1].begin(), m_lookaheadSize, 0.0f);
352 m_writeCursor = 0;
353 }
354 }
355
356 void resetRuntimeState() {
357 m_tpUpsampleL = {};
358 m_tpUpsampleR = {};
359 m_shortEma = 0.0f;
360 m_longEma = 0.0f;
361 const float sr = static_cast<float>(m_sampleRate);
362 m_alphaShort = 1.0f - std::exp(-1.0f / (0.050f * sr));
363 m_alphaLong = 1.0f - std::exp(-1.0f / (0.300f * sr));
364 m_smoothCoeff = 1.0f - std::exp(-1.0f / (0.002f * sr));
365 m_oneOverSampleRateMs = 1.0f / (0.001f * sr);
366 m_clipLevel = dbToLinear(-kCeilingMarginDb);
367 m_appliedGainDb = 0.0f;
368 m_autoReleaseMs = 100.0f;
369 m_writeCursor = 0;
370 std::fill_n(m_lookaheadBuf[0].begin(), m_lookaheadSize, 0.0f);
371 std::fill_n(m_lookaheadBuf[1].begin(), m_lookaheadSize, 0.0f);
372 m_gainReduction.store(0.0f, std::memory_order_relaxed);
373 m_inputLevel.store(0.0f, std::memory_order_relaxed);
374 m_outputLevel.store(0.0f, std::memory_order_relaxed);
375 }

391 // ── Utility ──
392 static void copyOrClear(const float* const* inputs, float** outputs,
393 uint32_t numInputChannels, uint32_t numOutputChannels,
394 uint32_t numFrames) {
395 for (uint32_t ch = 0; ch < numOutputChannels; ++ch) {
396 if (outputs[ch] && ch < numInputChannels && inputs[ch]) {
397 std::memcpy(outputs[ch], inputs[ch], numFrames * sizeof(float));
398 } else if (outputs[ch]) {
399 std::memset(outputs[ch], 0, numFrames * sizeof(float));
400 }
401 }
402 }

404 static float readInput(const float* const* inputs, uint32_t channels, uint32_t channel, uint32_t frame) {
405 if (channel >= channels || !inputs[channel]) return 0.0f;
406 return inputs[channel][frame];

409 static float sanitizeSample(float sample) {
410 if (!std::isfinite(sample)) return 0.0f;
411 return std::clamp(sample, -16.0f, 16.0f);
412 }

414 static float linearToDb(float linear) {
415 if (!std::isfinite(linear) || linear <= 1.0e-12f) return -120.0f;
416 return std::max(-120.0f, 20.0f * std::log10(linear));
417 }
418
419 static float dbToLinear(float db) {
420 return std::exp(db * 0.11512925464970229f);
421 }

423 static std::string formatDb(float db) {
424 const int rounded = static_cast<int>(std::round(db * 10.0f));
425 if (rounded >= 0) return "0.0dB";
426 return "-" + std::to_string((-rounded) / 10) + "." + std::to_string((-rounded) % 10) + "dB";
427 }

430 return std::to_string(static_cast<int>(std::round(ms))) + "ms";
