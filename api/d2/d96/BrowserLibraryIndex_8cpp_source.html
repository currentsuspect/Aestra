24bool isSameOrInside(const std::filesystem::path& inner, const std::filesystem::path& outer) {
25 auto o = outer.begin();
26 auto i = inner.begin();
27 for (; o != outer.end(); ++o, ++i) {
28 if (o->empty())
29 break; // trailing separator
30 if (i == inner.end() || *i != *o)
31 return false;
32 }
33 return true;
34}

36// Overlapping roots (one inside another, or two spellings of one folder) are
37// resolved ONCE, here, instead of canonicalising every file found: realpath is
38// several syscalls, and a full library is 100k files. Directory symlinks are not
39// followed during the crawl, so two disjoint canonical roots cannot reach the
40// same file.
41std::vector<std::filesystem::path> distinctRoots(const std::vector<std::string>& roots) {
42 std::vector<std::filesystem::path> canon;
43 for (const auto& r : roots) {
44 std::error_code ec;
45 if (r.empty() || !std::filesystem::is_directory(r, ec))
46 continue;
47 canon.push_back(canonicalOrNormal(r));
48 }
49 std::vector<std::filesystem::path> out;
50 for (size_t i = 0; i < canon.size(); ++i) {
51 bool covered = false;
52 for (size_t j = 0; j < canon.size() && !covered; ++j) {
53 if (i == j)
54 continue;
55 // Inside another root, or an exact duplicate that a lower index keeps.
56 const bool inside = isSameOrInside(canon[i], canon[j]);
57 covered = inside && (canon[i] != canon[j] || j < i);
58 }
59 if (!covered)
60 out.push_back(canon[i]);
61 }
62 return out;
63}

70void fillAudioFacts(FileItem& item) {
71 if (item.isDirectory || !isAudioType(item.type)) return;
72 BrowserLibrary::AudioInfo info;
73 if (BrowserLibrary::readAudioInfo(item.path, info)) {
74 item.durationSec = info.durationSec;
75 item.sampleRate = info.sampleRate;
76 item.channels = info.channels;
77 }
78 item.detectedBpm = BrowserLibrary::parseBpmFromFilename(item.name);
79 if (item.detectedBpm == 0 && info.tempo > 0.0f) item.detectedBpm = static_cast<int>(info.tempo + 0.5f);
80 item.musicalKey = BrowserLibrary::parseKeyFromFilename(item.name);
81}

88void BrowserLibraryIndex::rebuild(std::vector<std::string> roots) {
89 auto stop = std::make_shared<std::atomic<bool>>(false);
90 {
91 std::lock_guard<std::mutex> lock(m_shared->mutex);
92 if (m_shared->currentStop) m_shared->currentStop->store(true);
93 m_shared->currentStop = stop;
94 m_shared->crawling.store(true);
95 }
96 std::thread([shared = m_shared, stop, roots = std::move(roots)]() {
97 // An exception escaping a thread is std::terminate. One unrepresentable
98 // file name (std::system_error on Windows) must not end the app.
99 try {
100 crawl(shared, stop, roots);
101 } catch (...) {
102 std::lock_guard<std::mutex> lock(shared->mutex);
103 if (!stop->load()) shared->crawling.store(false);
104 }
105 }).detach();
106}

121void BrowserLibraryIndex::crawl(const std::shared_ptr<Shared>& shared, const std::shared_ptr<std::atomic<bool>>& stop,
122 const std::vector<std::string>& roots) {
123 namespace fs = std::filesystem;
124 std::vector<FileItem> entries;
125 size_t sincePublish = 0;
126
127 const auto publish = [&](bool final) {
128 auto snap = std::make_shared<const std::vector<FileItem>>(entries);
129 std::lock_guard<std::mutex> lock(shared->mutex);
130 if (stop->load()) return; // superseded: never overwrite the newer crawl
131 shared->published = std::move(snap);
132 shared->revision.fetch_add(1);
133 if (final) shared->crawling.store(false);
134 };
135
136 for (const auto& root : distinctRoots(roots)) {
137 if (stop->load() || entries.size() >= kMaxEntries) break;
138 std::error_code ec;
139
140 // Default options: directory symlinks are not followed (no loops).
141 fs::recursive_directory_iterator it(root, fs::directory_options::skip_permission_denied, ec);
142 const fs::recursive_directory_iterator end;
143 for (; !ec && it != end; it.increment(ec)) {
144 if (stop->load() || entries.size() >= kMaxEntries) break;
145 const fs::directory_entry& entry = *it;
146 const std::string name = entry.path().filename().string();
147 std::error_code typeEc;
148 const bool isDir = entry.is_directory(typeEc);
149 if (!name.empty() && name[0] == '.') {
150 if (isDir) it.disable_recursion_pending();
151 continue;
152 }
153 if (isDir) {
154 if (it.depth() >= kMaxDepth) it.disable_recursion_pending();
155 continue;
156 }
157 if (typeEc || !entry.is_regular_file(typeEc)) continue;
158 const std::string path = entry.path().string();
159 // getType by extension, not isAllowed(): the entry is already known
160 // to be a regular file, and isAllowed() stats the path again to ask
161 // whether it is a directory -- one wasted syscall per file, 100k on
162 // a full library. For a file the two agree exactly: anything
163 // isAllowed() accepts maps to a known type.
164 const FileType type = FileFilter::getType(path, false);
165 if (type == FileType::Unknown) continue;
166
167 std::error_code sizeEc;
168 const auto size = entry.file_size(sizeEc);
169 FileItem item(name, path, type, false, sizeEc ? 0 : static_cast<size_t>(size));
170 std::error_code timeEc;
171 const auto modified = entry.last_write_time(timeEc);
172 if (!timeEc) item.modifiedTime = static_cast<int64_t>(modified.time_since_epoch().count());
173 fillAudioFacts(item);
174 entries.push_back(std::move(item));
175
176 if (++sincePublish >= kPublishEvery) {
177 sincePublish = 0;
178 publish(false);
179 }
180 }
181 }
182 publish(true);
183}
