2// AestraSat V1 — oversampled waveshaping saturator (tape / tube / hard clip).
3//
4// Signal path (per channel, per sample):
5// in -> drive gain -> 4x oversample -> waveshaper -> downsample
6// -> DC blocker -> tone low-pass -> output trim -> mix with latency-aligned dry
7//
8// The nonlinear stage always runs at 4x via DSP::Oversampler (47/29-tap halfband
9// cascade), so latency is the oversampler's reported constant. The dry path is
10// delayed by the same amount so the Mix control never comb-filters.
11

49 bool initialize(double sampleRate, uint32_t maxBlockSize) override {
50 (void)maxBlockSize;

52 // Seed parameter defaults only on the first initialization of a fresh

63 }
64 resetRuntimeState();
65 snapSmoothedParams();
66 return true;
67 }

80 void process(const float* const* inputs, float** outputs, uint32_t numInputChannels, uint32_t numOutputChannels,
81 uint32_t numFrames, const MidiBuffer* midiInput = nullptr, MidiBuffer* midiOutput = nullptr) override {
82 (void)midiInput;
83 (void)midiOutput;
84
85 if (!m_active.load(std::memory_order_relaxed) || m_params[kBypass].load(std::memory_order_relaxed) > 0.5f) {
86 // The plugin always reports the oversampler latency, so internal
87 // bypass must delay by the same amount (through the same ring,
88 // which keeps advancing) or this slot lands ~30 samples early
89 // relative to PDC-compensated tracks.
90 copyDelayed(inputs, outputs, numInputChannels, numOutputChannels, numFrames);
91 m_wasBypassed = true;
92 return;
93 }
94 if (m_wasBypassed) {
95 // Flush stale wet-path state from before the bypass so reactivation
96 // does not emit it; the dry ring stays hot for latency alignment.
97 for (auto& os : m_oversampler) {
98 os.reset();
99 }
100 m_dcX1[0] = m_dcX1[1] = 0.0f;
101 m_dcY1[0] = m_dcY1[1] = 0.0f;
102 m_toneState[0] = m_toneState[1] = 0.0f;
103 snapSmoothedParams();
104 m_wasBypassed = false;
105 }
106
107 const uint32_t channels = std::min<uint32_t>(2, numOutputChannels);
108 const bool stereo = channels >= 2;
109 const auto mode = static_cast<Mode>(std::min<uint32_t>(
110 kModeCount - 1,
111 static_cast<uint32_t>(m_params[kMode].load(std::memory_order_relaxed) * static_cast<float>(kModeCount - 1) +
112 0.5f)));
113
114 float blockInputPeak = 0.0f;
115 float blockOutputPeak = 0.0f;
116
117 // Parameters are read and smoothed once per control block; the
118 // per-sample loop only does the oversampled shaper, DC blocker, tone
119 // one-pole and mix — no atomics or transcendentals in the hot path.
120 for (uint32_t blockStart = 0; blockStart < numFrames; blockStart += kCtrlBlock) {
121 const uint32_t blockEnd = std::min(blockStart + kCtrlBlock, numFrames);
122
123 m_driveSmoothed += (getParameter(kDrive) - m_driveSmoothed) * m_smoothCoeff;
124 m_toneSmoothed += (getParameter(kTone) - m_toneSmoothed) * m_smoothCoeff;
125 m_outputSmoothed += (getParameter(kOutput) - m_outputSmoothed) * m_smoothCoeff;
126 m_mixSmoothed += (getParameter(kMix) - m_mixSmoothed) * m_smoothCoeff;
127
128 const float driveGain = dbToLinear(driveDbFromNorm(m_driveSmoothed));
129 const float outGain = dbToLinear(outputDbFromNorm(m_outputSmoothed));
130 const float wet = m_mixSmoothed;
131 const float dry = 1.0f - wet;
132 const float toneCoeff = toneCoeffFromNorm(m_toneSmoothed);
133
134 for (uint32_t i = blockStart; i < blockEnd; ++i) {
135 for (uint32_t ch = 0; ch < channels; ++ch) {
136 const float in = sanitizeSample(readInput(inputs, numInputChannels, ch, i));
137 blockInputPeak = std::max(blockInputPeak, std::abs(in));
138
139 // Latency-aligned dry tap
140 const float delayedDry = m_dryDelay[ch][m_dryPos];
141 m_dryDelay[ch][m_dryPos] = in;
142
143 // Nonlinear stage at 4x
144 float osBuf[kOsFactor];
145 m_oversampler[ch].upsample(in * driveGain, osBuf);
146 for (uint32_t p = 0; p < kOsFactor; ++p) {
147 osBuf[p] = shape(osBuf[p], mode);
148 }
149 float sat = m_oversampler[ch].downsample(osBuf);
150
151 // DC blocker (~10 Hz high-pass) — tube asymmetry adds DC
152 const float dcOut = sat - m_dcX1[ch] + m_dcCoeff * m_dcY1[ch];
153 m_dcX1[ch] = sat;
154 m_dcY1[ch] = dcOut;
155
156 // Tone: one-pole low-pass on the wet path
157 m_toneState[ch] += (dcOut - m_toneState[ch]) * toneCoeff;
158
159 const float out = delayedDry * dry + m_toneState[ch] * outGain * wet;
160 blockOutputPeak = std::max(blockOutputPeak, std::abs(out));
161
162 if (outputs[ch])
163 outputs[ch][i] = out;
164 }
165
166 m_dryPos = (m_dryPos + 1u) % kDryDelayLen;
167
168 if (!stereo && numOutputChannels > 1 && outputs[1] && outputs[0]) {
169 outputs[1][i] = outputs[0][i];
170 }
171 for (uint32_t ch = 2; ch < numOutputChannels; ++ch) {
172 if (outputs[ch])
173 outputs[ch][i] = 0.0f;
174 }
175 }
176 }
177
178 m_inputLevel.store(blockInputPeak, std::memory_order_relaxed);
179 m_outputLevel.store(blockOutputPeak, std::memory_order_relaxed);
180 }

209 std::string getParameterDisplay(uint32_t id) const override {
210 if (id >= kParamCount)
211 return "";
212 const float v = getParameter(id);
213 char buf[32];
214 switch (id) {
215 case kDrive:
216 std::snprintf(buf, sizeof(buf), "%.1f dB", driveDbFromNorm(v));
217 return buf;
218 case kMode:
219 switch (modeFromNorm(v)) {
220 case kModeTape:
221 return "Tape";
222 case kModeTube:
223 return "Tube";
224 default:
225 return "Hard";
226 }
227 case kTone: {
228 const float hz = toneHzFromNorm(v);
229 if (hz >= 10000.0f) {
230 std::snprintf(buf, sizeof(buf), "%.1f kHz", hz / 1000.0f);
231 } else if (hz >= 1000.0f) {
232 std::snprintf(buf, sizeof(buf), "%.2f kHz", hz / 1000.0f);
233 } else {
234 std::snprintf(buf, sizeof(buf), "%.0f Hz", hz);
235 }
236 return buf;
237 }
238 case kOutput:
239 std::snprintf(buf, sizeof(buf), "%+.1f dB", outputDbFromNorm(v));
240 return buf;
241 case kMix:
242 return std::to_string(static_cast<int>(std::round(v * 100.0f))) + "%";
243 case kBypass:
244 return v > 0.5f ? "ON" : "OFF";
245 default:
246 return "";
247 }

250 std::vector<uint8_t> saveState() const override {
251 struct Blob {
252 uint32_t magic = kStateMagic;
253 uint32_t version = 1;
254 float params[kParamCount] = {};
255 } blob;
256 for (uint32_t i = 0; i < kParamCount; ++i) {
257 blob.params[i] = getParameter(i);
258 }
259 const auto* data = reinterpret_cast<const uint8_t*>(&blob);
260 return {data, data + sizeof(blob)};
261 }

263 bool loadState(const std::vector<uint8_t>& state) override {
264 if (state.size() < sizeof(uint32_t) * 2)
265 return false;
266 uint32_t magic = 0;
267 std::memcpy(&magic, state.data(), sizeof(magic));
268 if (magic != kStateMagic)
269 return false;
270 struct Blob {
271 uint32_t magic;
272 uint32_t version;
273 float params[kParamCount];
274 };
275 if (state.size() < sizeof(Blob))
276 return false;
277 Blob blob{};
278 std::memcpy(&blob, state.data(), sizeof(blob));
279 if (blob.version < 1 || blob.version > 1)
280 return false;
281 for (uint32_t i = 0; i < kParamCount; ++i) {
282 setParameter(i, blob.params[i]);
283 }
284 return true;
285 }

308 // ── Parameter mapping (public: shared with editor/tests) ──
309 static float driveDbFromNorm(float value) { return std::clamp(value, 0.0f, 1.0f) * 36.0f; }
310
311 static float outputDbFromNorm(float value) { return -12.0f + std::clamp(value, 0.0f, 1.0f) * 24.0f; }
312
313 static float toneHzFromNorm(float value) {
314 // 1 kHz .. 20 kHz, log
315 return 1000.0f * std::pow(20.0f, std::clamp(value, 0.0f, 1.0f));
316 }
317
318 static Mode modeFromNorm(float value) {
319 return static_cast<Mode>(std::min<uint32_t>(
320 kModeCount - 1,
321 static_cast<uint32_t>(std::clamp(value, 0.0f, 1.0f) * static_cast<float>(kModeCount - 1) + 0.5f)));
322 }
323
324 static float normFromMode(Mode mode) { return static_cast<float>(mode) / static_cast<float>(kModeCount - 1); }

326private:
327 // Read-before-write circular buffer of length N delays by exactly N —
328 // sized to match the oversampler round-trip latency.
329 static constexpr uint32_t kDryDelayLen = DSP::Oversampler::kReportedLatency;
330 static constexpr uint32_t kCtrlBlock = 16; // control-rate interval for param/coefficient updates
331
335 void copyDelayed(const float* const* inputs, float** outputs, uint32_t numInputChannels, uint32_t numOutputChannels,
336 uint32_t numFrames) {
337 for (uint32_t i = 0; i < numFrames; ++i) {
338 for (uint32_t ch = 0; ch < 2; ++ch) {
339 const float in = sanitizeSample(readInput(inputs, numInputChannels, ch, i));
340 const float delayed = m_dryDelay[ch][m_dryPos];
341 m_dryDelay[ch][m_dryPos] = in;
342 if (ch < numOutputChannels && outputs[ch])
343 outputs[ch][i] = delayed;
344 }
345 m_dryPos = (m_dryPos + 1u) % kDryDelayLen;
346 }
347 for (uint32_t ch = 2; ch < numOutputChannels; ++ch) {
348 if (outputs[ch] && ch < numInputChannels && inputs[ch]) {
349 std::memcpy(outputs[ch], inputs[ch], numFrames * sizeof(float));
350 } else if (outputs[ch]) {
351 std::memset(outputs[ch], 0, numFrames * sizeof(float));
352 }
353 }
354 }

356 static float shape(float x, Mode mode) {
357 switch (mode) {
358 case kModeTape:
359 return std::tanh(x);
360 case kModeTube:
361 // Asymmetric: the negative lobe saturates earlier (limit -1/1.5)
362 // which generates even harmonics. Both branches have unity slope
363 // at 0, so the curve is C1-continuous.
364 return x >= 0.0f ? std::tanh(x) : std::tanh(1.5f * x) * (1.0f / 1.5f);
365 default:
366 return std::clamp(x, -1.0f, 1.0f);
367 }
368 }
369
370 float toneCoeffFromNorm(float norm) const {
371 const float fc = toneHzFromNorm(norm);
372 const float x = 2.0f * 3.14159265358979323846f * fc / static_cast<float>(m_sampleRate);
373 return 1.0f - std::exp(-x);
374 }

376 void resetRuntimeState() {
377 for (auto& os : m_oversampler) {
378 os.reset();
379 }
380 for (auto& d : m_dryDelay) {
381 d.fill(0.0f);
382 }
383 m_dryPos = 0;
384 m_dcX1[0] = m_dcX1[1] = 0.0f;
385 m_dcY1[0] = m_dcY1[1] = 0.0f;
386 m_toneState[0] = m_toneState[1] = 0.0f;
387 m_wasBypassed = false;
388 const float sr = static_cast<float>(m_sampleRate);
389 // Smoothing runs once per control block, so the coefficient is scaled
390 // to the block interval to keep the 2 ms time constant.
391 m_smoothCoeff = 1.0f - std::exp(-static_cast<float>(kCtrlBlock) / (0.002f * sr));
392 m_dcCoeff = 1.0f - 2.0f * 3.14159265358979323846f * 10.0f / sr;
393 m_inputLevel.store(0.0f, std::memory_order_relaxed);
394 m_outputLevel.store(0.0f, std::memory_order_relaxed);
395 }

416 static float readInput(const float* const* inputs, uint32_t channels, uint32_t channel, uint32_t frame) {
417 if (channel >= channels || !inputs[channel]) {
418 return (channel == 1 && channels > 0 && inputs[0]) ? inputs[0][frame] : 0.0f;
419 }
420 return inputs[channel][frame];
