33std::optional<std::string> handleHistoryVerbs(const RequestContext& ctx, const ResponseEnvelope& env) {
34 // Bound to the names the moved bodies already use, so no verb body had to be
35 // edited to make the move.
36 const double id = ctx.id;
37 const std::string& verb = ctx.verb;
38 JSON& request = *ctx.request;
39 TrackManager* const m_trackManager = ctx.trackManager;
40 AudioEngine* const m_engine = ctx.engine;
41 // A reference, not a pointer: get_project_load_report tests this for a value
42 // and then dereferences it, so it has to carry std::optional's own semantics.
43 // A pointer would be truthy whenever the address is non-null — including when
44 // the optional it points at is empty — which is not the same question.
45 const std::optional<JSON>& m_projectLoadReport = *ctx.projectLoadReport;
46
47 const auto makeOk = [&env]() { return env.ok(); };
48 const auto finish = [&env](JSON& response) { return env.finish(response); };
49
50 // ------------------------------------------------------------------
51 // undo / redo — drive the same history the UI's Ctrl+Z drives.
52 //
53 // Every mutation verb and every batch already lands there as one step;
54 // without these the surface could build that history but never walk it,
55 // so the only way back from a mistake was to hand-write the inverse
56 // edit. Against the live app these move the user's undo stack, which is
57 // the point: agent edits and hand edits are the same edits.
58 // ------------------------------------------------------------------
59 if (verb == "undo" || verb == "redo") {
60 if (!m_trackManager) {
61 return makeError(id, "execution_error", "no track manager", verb).toString();
62 }
63 if (request.has("args") && request["args"].isObject() &&
64 !request["args"].asObject().empty()) {
65 return makeError(id, "validation_error", verb + " takes no args", verb).toString();
66 }
67
68 CommandHistory& history = m_trackManager->getCommandHistory();
69 const bool undoing = (verb == "undo");
70
71 // An empty history is a refusal, not a silent success: a caller that
72 // gets ok back would reasonably believe an edit was reverted.
73 if (undoing ? !history.canUndo() : !history.canRedo()) {
74 return makeError(id, "execution_error",
75 undoing ? "nothing to undo" : "nothing to redo", verb)
76 .toString();
77 }
78 if (!(undoing ? history.undo() : history.redo())) {
79 return makeError(id, "execution_error",
80 std::string(undoing ? "undo" : "redo") + " failed", verb)
81 .toString();
82 }
83
84 // Report the new ends of the stack so a caller can walk it without
85 // guessing when to stop.
86 JSON result = JSON::object();
87 result.set("canUndo", JSON(history.canUndo()));
88 result.set("canRedo", JSON(history.canRedo()));
89 JSON response = makeOk();
90 response.set("result", result);
91 return finish(response);
92 }
93
94 if (verb == "batch") {
95 if (!m_trackManager) {
96 return makeError(id, "execution_error", "no track manager", verb).toString();
97 }
98 if (!request.has("args") || !request["args"].isObject()) {
99 return makeError(id, "validation_error",
100 "batch requires args: {\"commands\": [{\"verb\": ..., \"args\": "
101 "...}, ...]}",
102 verb)
103 .toString();
104 }
105 JSON& args = request["args"];
106 for (auto& entry : args.asObject()) {
107 if (entry.first != "commands") {
108 return makeError(id, "validation_error",
109 "unknown arg for batch: " + entry.first, verb)
110 .toString();
111 }
112 }
113 if (!args.has("commands") || !args["commands"].isArray()) {
114 return makeError(id, "validation_error", "arg 'commands' must be an array", verb)
115 .toString();
116 }
117 JSON& commands = args["commands"];
118 const size_t count = commands.size();
119 constexpr size_t kMaxBatchCommands = 64;
120 if (count == 0) {
121 return makeError(id, "validation_error", "batch must contain at least one command",
122 verb)
123 .toString();
124 }
125 if (count > kMaxBatchCommands) {
126 return makeError(id, "validation_error",
127 "batch too large: " + std::to_string(count) + " commands (max " +
128 std::to_string(kMaxBatchCommands) + ")",
129 verb)
130 .toString();
131 }
132
133 // All-or-nothing, stepwise: each member is validated, built, and
134 // executed against the state its predecessors produced — so a
135 // batch can set the pan of a track it just added. On any failure
136 // the executed prefix is undone in reverse and nothing is
137 // recorded. On success the whole group lands in history as one
138 // already-executed CommandTransaction: a single undo step.
139 CommandParser parser;
140 auto transaction = std::make_shared<CommandTransaction>("Muse Batch");
141 std::vector<std::shared_ptr<ICommand>> executed;
142 executed.reserve(count);
143 JSON createdList = JSON::array();
144 // Unwind the executed prefix in reverse and produce the final
145 // error response. A rollback failure must not hide behind the
146 // member error: the response then reports execution_error and
147 // says the session may be inconsistent.
148 const auto failBatch = [&](const char* errorStatus,
149 const std::string& message) -> std::string {
150 bool rollbackClean = true;
151 for (auto it = executed.rbegin(); it != executed.rend(); ++it) {
152 try {
153 (*it)->undo();
154 } catch (...) {
155 // Keep unwinding the rest of the prefix, but remember.
156 rollbackClean = false;
157 }
158 }
159 if (!rollbackClean) {
160 return makeError(id, "execution_error",
161 message +
162 " (rollback also failed; session state may be "
163 "inconsistent)",
164 verb)
165 .toString();
166 }
167 return makeError(id, errorStatus, message, verb).toString();
168 };
169
170 for (size_t i = 0; i < count; ++i) {
171 const std::string prefix = "commands[" + std::to_string(i) + "]: ";
172 JSON& item = commands[i];
173 if (!item.isObject() || !item.has("verb") || !item["verb"].isString()) {
174 return failBatch("validation_error",
175 prefix + "must be an object with a string verb");
176 }
177 // Refuse unknown member keys by name, the way every other args
178 // object here does. Accepting them silently is how a member
179 // written with "flags" instead of "args" runs with NO args and
180 // still reports ok — an add_track that quietly takes its default
181 // name rather than the one asked for.
182 for (auto& memberEntry : item.asObject()) {
183 if (memberEntry.first != "verb" && memberEntry.first != "args") {
184 return failBatch("validation_error",
185 prefix + "unknown key: " + memberEntry.first +
186 " (a member is {\"verb\": ..., \"args\": ...})");
187 }
188 }
189 const std::string subVerb = item["verb"].asString();
190 if (isQueryVerb(subVerb) || isActionVerb(subVerb)) {
191 return failBatch("validation_error",
192 prefix + "only mutation verbs are allowed in a batch");
193 }
194
195 std::unordered_map<std::string, std::string> flags;
196 if (item.has("args")) {
197 if (!item["args"].isObject()) {
198 return failBatch("validation_error", prefix + "args must be an object");
199 }
200 std::string convertError;
201 if (!jsonArgsToFlags(item["args"], flags, convertError)) {
202 return failBatch("validation_error", prefix + convertError);
203 }
204 }
205
206 CommandStatus buildStatus = CommandStatus::Success;
207 std::string buildMessage;
208 const CommandContext ctx{m_engine, m_trackManager};
209 auto built = parser.buildValidated(subVerb, flags, buildStatus, buildMessage, ctx);
210 if (!built) {
211 return failBatch(statusName(buildStatus), prefix + buildMessage);
212 }
213
214 std::shared_ptr<ICommand> cmd(std::move(built));
215 try {
216 cmd->execute();
217 } catch (const std::exception& e) {
218 return failBatch("execution_error", prefix + e.what());
219 } catch (...) {
220 return failBatch("execution_error", prefix + "command threw");
221 }
222 CommandResult member;
223 if (!CommandParser::describeCreated(*cmd, member)) {
224 executed.push_back(cmd); // rolled back with the rest
225 return failBatch("execution_error", prefix + member.message);
226 }
227 if (member.hasCreatedId) {
228 JSON entry = JSON::object();
229 entry.set("index", JSON(static_cast<double>(i)));
230 entry.set("verb", JSON(subVerb));
231 if (!member.createdIdText.empty()) {
232 entry.set("createdId", JSON(member.createdIdText));
233 } else {
234 entry.set("createdId", JSON(static_cast<double>(member.createdId)));
235 }
236 entry.set("createdKind", JSON(member.createdKind));
237 createdList.push(entry);
238 }
239 executed.push_back(cmd);
240 transaction->add(cmd);
241 }
242
243 transaction->markExecuted();
244 if (!m_trackManager->getCommandHistory().pushExecuted(transaction)) {
245 // Executed but not recorded (e.g. a deferred transaction is
246 // active on this history). "ok, undoable" would be a lie —
247 // keep all-or-nothing honest by rolling the batch back.
248 return failBatch("execution_error",
249 "batch could not be recorded in history; rolled back");
250 }
251
252 JSON result = JSON::object();
253 result.set("count", JSON(static_cast<double>(count)));
254 // What each member created, in member order, so a batch that builds a
255 // unit and then needs its id later is not followed by a list diff.
256 result.set("created", createdList);
257 JSON response = makeOk();
258 response.set("result", result);
259 response.set("undoable", JSON(true));
260 return finish(response);
261 }
262 // No verb matched. In the single-file version this fell out of the if-chain
263 // into the next family's checks and eventually the mutation path, which is
264 // exactly what nullopt asks handleRequest to do.
265 return std::nullopt;
266}
