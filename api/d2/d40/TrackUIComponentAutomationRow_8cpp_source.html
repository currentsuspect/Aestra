2// V82-T1 (FD-50 board 8): automation as a lane of its own under its track. "Automate Volume" opens one row under
3// that track; other tracks and their clips stay as they are. The row is a TrackUIComponent for the same playlist
4// lane in automation mode: it draws no clips, its curve is the track's colour with small square points, and its
5// header names the parameter (▾ switches it), shows the value at the playhead, and × hides the row without
6// deleting the curve. A track whose automation row is hidden but holds points shows a small curve mark next to
7// its name; clicking it brings the row back. Kept out of TrackUIComponent.cpp, which is at its size pin.

56void TrackUIComponent::renderAutomationRowHeader(AestraUI::NUIRenderer& renderer,
57 const AestraUI::NUIRect& controlArea) {
58 auto& theme = AestraUI::NUIThemeManager::getInstance();
59 const PlaylistLane* lane = m_trackManager ? m_trackManager->getPlaylistModel().getLane(m_laneId) : nullptr;
60 if (!lane) return;
61 const AestraUI::NUIColor ink = ClipPaint::identityFromARGB(lane->colorRGBA);
62
63 // One step lighter than the track's chrome, with the track's colour bar, so it reads as part of the track.
64 renderer.fillRect(controlArea, theme.getColor("trackChrome"));
65 renderer.fillRect(controlArea, AestraUI::NUIColor::white().withAlpha(0.03f));
66 renderer.fillRect({controlArea.x, controlArea.y, 3.0f, controlArea.height}, ink);
67 renderer.drawLine({controlArea.right(), controlArea.y}, {controlArea.right(), controlArea.bottom()}, 1.0f,
68 theme.getColor("borderSubtle").withAlpha(0.24f));
69
70 // "↳ Volume ▾" — the parameter, and the switch. Arrow and caret are drawn: the UI font has neither glyph.
71 const float labelSize = theme.getFontSize("s");
72 const std::string label = automationTargetLabel(m_editedAutomationTarget);
73 const AestraUI::NUIColor text = theme.getColor("textPrimary").withAlpha(0.86f);
74 const float cy = std::round(controlArea.y + controlArea.height * 0.5f);
75 const float arrowX = controlArea.x + 18.0f;
76 renderer.fillRect({arrowX, cy - 6.0f, 1.0f, 6.0f}, text.withAlpha(0.5f)); // ↳: down, then right to an arrowhead
77 renderer.fillRect({arrowX, cy, 7.0f, 1.0f}, text.withAlpha(0.5f));
78 renderer.fillRect({arrowX + 5.0f, cy - 2.0f, 1.0f, 5.0f}, text.withAlpha(0.5f));
79 const float labelX = arrowX + 12.0f;
80 const float labelWidth = renderer.measureText(label, labelSize).width;
81 renderer.drawText(label, {labelX, std::round(renderer.calculateTextY(controlArea, labelSize))}, labelSize, text);
82 const float caretX = labelX + labelWidth + 7.0f;
83 for (int i = 0; i < 3; ++i) { // ▾, as the toolbar's SNAP caret
84 renderer.fillRect({caretX - 2.0f + static_cast<float>(i), cy - 1.0f + static_cast<float>(i),
85 5.0f - 2.0f * static_cast<float>(i), 1.0f},
86 text.withAlpha(0.6f));
87 }
88 m_automationTargetRect = {arrowX - 4.0f, controlArea.y + 4.0f, caretX + 7.0f - (arrowX - 4.0f), controlArea.height - 8.0f};
89
90 // × at the right edge, then the value at the playhead to its left, in the track's colour.
91 m_automationHideRect = {controlArea.right() - kHideBoxSize - 6.0f,
92 std::round(controlArea.y + (controlArea.height - kHideBoxSize) * 0.5f), kHideBoxSize,
93 kHideBoxSize};
94 renderer.drawTextCentered("\xC3\x97", m_automationHideRect, labelSize, theme.getColor("textSecondary").withAlpha(0.6f));
95
96 const int curveIndex = automationCurveIndexFor(lane->automationCurves, m_editedAutomationTarget);
97 const double beat = m_trackManager->getUIPosition() * std::max(1.0, m_trackManager->getPlaylistModel().getBPM()) / 60.0;
98 // A selected node's level when one is picked (a click says "where is THIS one"), else the level at the playhead.
99 const AutomationHeaderReadout readout = automationHeaderReadout(
100 curveIndex >= 0 ? &lane->automationCurves[static_cast<size_t>(curveIndex)] : nullptr,
101 makeAutomationCurve(m_editedAutomationTarget).getDefaultValue(), beat);
102 const std::string valueText = automationValueLabel(m_editedAutomationTarget, readout.value);
103 const float valueSize = theme.getFontSize("xs");
104 const float markerWidth = readout.isNode ? 9.0f : 0.0f; // a node's value wears the node's square
105 // With a neighbour to compare against, the node's level sits above and how far it is from that neighbour below;
106 // otherwise the single readout stays centred, as at the playhead.
107 const bool twoLines = readout.isNode && !readout.delta.empty();
108 const float deltaWidth = twoLines ? renderer.measureText(readout.delta, valueSize).width : 0.0f;
109 const float valueWidth = std::max(renderer.measureText(valueText, valueSize).width + markerWidth, deltaWidth);
110 const float valueX = std::max(m_automationTargetRect.right() + 6.0f, m_automationHideRect.x - 6.0f - valueWidth);
111 const float lineShift = twoLines ? 6.0f : 0.0f;
112 const float valueY = std::round(renderer.calculateTextY(controlArea, valueSize)) - lineShift;
113 if (readout.isNode) {
114 renderer.fillRect({valueX, std::round(cy - lineShift) - 2.5f, 5.0f, 5.0f}, AestraUI::NUIColor::white());
115 }
116 renderer.drawText(valueText, {valueX + markerWidth, valueY}, valueSize, ink);
117 if (twoLines) renderer.drawText(readout.delta, {valueX, valueY + 13.0f}, valueSize, ink.withAlpha(0.62f));
118}

120bool TrackUIComponent::handleAutomationRowHeaderMouse(const AestraUI::NUIMouseEvent& event) {
121 const bool leftPress = event.pressed && event.button == AestraUI::NUIMouseButton::Left;
122 if (!m_isAutomationRow) {
123 // A track row's mark for hidden automation brings its row back.
124 if (leftPress && m_hasHiddenAutomation && m_onShowAutomation &&
125 m_hiddenAutomationMarkRect.contains(event.position)) {
126 m_onShowAutomation();
127 return true;
128 }
129 return false;
130 }
131 const auto& layout = AestraUI::NUIThemeManager::getInstance().getLayoutDimensions();
132 const AestraUI::NUIRect bounds = getBounds();
133 // The header ends kAutomationEdgeSlop short of the lane: a point at beat 0 sits ON the lane's left edge, and its
134 // left half belongs to the lane (it used to be swallowed here, so the first point could not be picked up).
135 const AestraUI::NUIRect controlArea(bounds.x, bounds.y,
136 std::max(0.0f, std::min(layout.trackControlsWidth, bounds.width) - kAutomationEdgeSlop),
137 bounds.height);
138 if (!controlArea.contains(event.position)) return false;
139 if (leftPress && m_automationHideRect.contains(event.position)) {
140 if (m_onAutomationRowHide) m_onAutomationRowHide();
141 return true;
142 }
143 if (leftPress && m_automationTargetRect.contains(event.position)) {
144 showAutomationTargetMenu({m_automationTargetRect.x, m_automationTargetRect.bottom()});
145 return true;
146 }
147 return event.pressed || event.released; // the header is the row's own: no track selection or M/S/R under it
148}

150void TrackUIComponent::showAutomationTargetMenu(const AestraUI::NUIPoint& anchor) {
151 if (m_automationTargetMenu) {
152 if (auto* parent = m_automationTargetMenu->getParent()) parent->removeChild(m_automationTargetMenu);
153 }
154 m_automationTargetMenu = std::make_shared<AestraUI::NUIContextMenu>();
155 for (const AutomationTarget target : {AutomationTarget::Volume, AutomationTarget::Pan}) {
156 m_automationTargetMenu->addRadioItem(automationTargetLabel(target), "automationRowTarget",
157 target == m_editedAutomationTarget,
158 [this, weakSelf = weak_from_this(), target]() {
159 if (weakSelf.expired()) return;
160 setEditedAutomationTarget(target);
161 if (m_onAutomationRowSwitch) m_onAutomationRowSwitch(target);
162 });
163 }
164 AestraUI::NUIComponent* root = rootOf(this);
165 if (!root) return;
166 root->addChild(m_automationTargetMenu);
167 m_automationTargetMenu->setOwner(weak_from_this());
168 m_automationTargetMenu->showAt(anchor);
169 root->repaint();
170}

172void TrackUIComponent::renderAutomationLayer(AestraUI::NUIRenderer& renderer, const AestraUI::NUIRect& bounds, float gridStartX) {
173 auto& playlist = m_trackManager->getPlaylistModel();
174 auto lane = playlist.getLane(m_laneId);
175 if (!lane) return;
176
177 auto& theme = AestraUI::NUIThemeManager::getInstance();
178 // FD-50 board 8: the curve is the track's colour, points small squares.
179 const AestraUI::NUIColor ink = ClipPaint::identityFromARGB(lane->colorRGBA);
180
181 // Automation Area bounds (exclude controls)
182 AestraUI::NUIRect gridArea(gridStartX, bounds.y, bounds.width - (gridStartX - bounds.x), bounds.height);
183
184 // V8-A2: the edited curve draws at full strength, the others recede, so the
185 // lane shows which curve a click edits while keeping every curve visible.
186 int editedIndex = automationCurveIndexFor(lane->automationCurves, m_editedAutomationTarget);
187 if (editedIndex < 0 && !m_automationTargetChosen && !lane->automationCurves.empty()) editedIndex = 0;
188 renderAutomationClips(renderer, bounds, gridStartX, editedIndex);
189 for (size_t ci = 0; ci < lane->automationCurves.size(); ++ci) {
190 const auto& curve = lane->automationCurves[ci];
191 if (!curve.isVisible()) continue;
192 const bool edited = static_cast<int>(ci) == editedIndex;
193 const AutomationTarget target = curve.getAutomationTarget();
194 auto yOf = [&](double v) {
195 return gridArea.y + (1.0f - static_cast<float>(automationHeightOfValue(target, v))) * gridArea.height;
196 };
197
198 const auto& points = curve.getPoints();
199 if (points.empty()) {
200 // Draw a flat line at default value if no points
201 float y = yOf(curve.getDefaultValue());
202 renderer.drawLine(AestraUI::NUIPoint(gridArea.x, y),
203 AestraUI::NUIPoint(gridArea.right(), y),
204 1.5f, ink.withAlpha(edited ? 0.55f : 0.2f));
205 continue;
206 }
207
208 double bpm = m_trackManager ? m_trackManager->getPlaylistModel().getBPM() : 120.0;
209 double sampleRate = m_trackManager ? m_trackManager->getPlaylistModel().getProjectSampleRate() : 48000.0;
210 double samplesPerBeat = (sampleRate * 60.0) / std::max(bpm, 1.0);
211
212 AestraUI::NUIColor curveColor = ink.withAlpha(edited ? 1.0f : 0.35f);
213
214 // Draw lines between points
215 std::vector<AestraUI::NUIPoint> polyPoints;
216 polyPoints.reserve(1024);
217
218 for (size_t i = 1; i < points.size(); ++i) {
219 const auto& p1 = points[i-1];
220 const auto& p2 = points[i];
221
222 const float sx1 = gridStartX + (static_cast<float>(p1.beat) * m_pixelsPerBeat) - m_timelineScrollOffset;
223 const float sy1 = yOf(p1.value);
224 const float sx2 = gridStartX + (static_cast<float>(p2.beat) * m_pixelsPerBeat) - m_timelineScrollOffset;
225 const float sy2 = yOf(p2.value);
226
227 // Only the part of the segment inside the lane is drawn, sampled at full density (visibleSegmentSpan).
228 const SegmentSpan span = visibleSegmentSpan(sx1, sx2, gridArea.x, gridArea.right());
229 if (span.empty()) continue; // wholly left or right of the lane
230 const float t0 = span.t0, t1 = span.t1;
231
232 // Adaptive subdivision based on the visible screen length: one vertex per pixel.
233 const float dist = std::hypot((sx2 - sx1) * (t1 - t0), (sy2 - sy1) * (t1 - t0));
234 const int steps = std::clamp(static_cast<int>(dist), 4, 512);
235
236 for (int s = 0; s <= steps; ++s) {
237 const double t = static_cast<double>(t0) + static_cast<double>(t1 - t0) * (static_cast<double>(s) / static_cast<double>(steps));
238 const double beat = p1.beat + (p2.beat - p1.beat) * t;
239 const double val = curve.getValueAtBeat(beat, samplesPerBeat);
240
241 const float x = gridStartX + (static_cast<float>(beat) * m_pixelsPerBeat) - m_timelineScrollOffset;
242 const float y = yOf(val);
243
244 polyPoints.emplace_back(x, y);
245 }
246 }
247
248 if (polyPoints.size() >= 2) {
249 // Use thick 2px capsules - high subdivision minimizes angle between segments so joints are invisible
250 renderer.drawPolyline(polyPoints.data(), static_cast<int>(polyPoints.size()), 2.0f, curveColor);
251 }
252
253
254 // Draw endpoints before/after first/last point if they are within view
255 if (!points.empty()) {
256 const auto& first = points.front();
257 float fx = gridStartX + (static_cast<float>(first.beat) * m_pixelsPerBeat) - m_timelineScrollOffset;
258 float fy = yOf(first.value);
259 if (fx > gridArea.x) {
260 renderer.drawLine(AestraUI::NUIPoint(gridArea.x, fy), AestraUI::NUIPoint(fx, fy), 1.5f, curveColor.withAlpha(0.5f));
261 }
262
263 const auto& last = points.back();
264 float lx = gridStartX + (static_cast<float>(last.beat) * m_pixelsPerBeat) - m_timelineScrollOffset;
265 float ly = yOf(last.value);
266 if (lx < gridArea.right()) {
267 renderer.drawLine(AestraUI::NUIPoint(lx, ly), AestraUI::NUIPoint(gridArea.right(), ly), 1.5f, curveColor.withAlpha(0.5f));
268 }
269 }
270
271 // Draw point handles
272 for (const auto& p : points) {
273 float x = gridStartX + (static_cast<float>(p.beat) * m_pixelsPerBeat) - m_timelineScrollOffset;
274 float y = yOf(p.value);
275
276 if (x < gridArea.x || x > gridArea.right()) continue;
277
278 const AestraUI::NUIColor ptColor = p.selected ? AestraUI::NUIColor::white() : curveColor;
279 renderer.fillRect(AestraUI::NUIRect(std::round(x) - 2.5f, std::round(y) - 2.5f, 5.0f, 5.0f), ptColor);
280 }
281 }
