27struct ArsenalKit {
28 static constexpr size_t kMaxUnits = 16;
29
31 static std::vector<std::string> filesIn(const std::string& dir) {
32 std::vector<std::string> out;
33 std::error_code ec;
34 if (dir.empty() || !std::filesystem::is_directory(dir, ec)) return out;
35 for (const auto& entry : std::filesystem::directory_iterator(dir, std::filesystem::directory_options::skip_permission_denied, ec)) {
36 std::error_code e2;
37 if (entry.is_regular_file(e2) && ArsenalSoundStrip::isAudioFile(entry.path().filename().string())) out.push_back(entry.path().string());
38 }
39 return out;
40 }
41
44 static std::vector<KitSlot> plan(std::vector<std::string> files) {
45 files.erase(std::remove_if(files.begin(), files.end(), [](const std::string& f) { return !ArsenalSoundStrip::isAudioFile(std::filesystem::path(f).filename().string()); }), files.end());
46 std::sort(files.begin(), files.end(), [](const std::string& a, const std::string& b) { return lower(a) < lower(b); });
47 if (files.size() > kMaxUnits) files.resize(kMaxUnits);
48 std::vector<KitSlot> out;
49 for (const auto& f : files) {
50 std::string base = ArsenalSoundStrip::displayName(f), name = base;
51 for (int n = 2; std::any_of(out.begin(), out.end(), [&](const KitSlot& s) { return s.name == name; }); ++n) name = base + " " + std::to_string(n);
52 out.push_back({name, f});
53 }
54 return out;
55 }
56
57private:
58 static std::string lower(std::string s) {
59 for (auto& c : s) c = static_cast<char>(std::tolower(static_cast<unsigned char>(c)));
60 return s;
61 }
62};
