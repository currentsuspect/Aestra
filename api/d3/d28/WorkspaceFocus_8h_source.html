103enum class WorkspaceTransitionKind {
104 PlaybackHotSwap, // Arsenal<->Timeline: re-arm pattern/arrangement transport (stop + play)
105 Audition, // enters or exits Audition: full teardown + audition engine
106 RoutingMap, // enters or exits RoutingMap: overlay only
107 Ordinary // everything else: pure visibility, zero transport/engine mutation
108};

111inline WorkspaceTransitionKind classifyTransition(ViewFocus current, ViewFocus previous) {
112 if (current == ViewFocus::Audition || previous == ViewFocus::Audition) {
113 return WorkspaceTransitionKind::Audition;
114 }
115 if (current == ViewFocus::RoutingMap || previous == ViewFocus::RoutingMap) {
116 return WorkspaceTransitionKind::RoutingMap;
117 }
118 const bool currentIsDaw = (current == ViewFocus::Arsenal || current == ViewFocus::Timeline);
119 const bool previousIsDaw = (previous == ViewFocus::Arsenal || previous == ViewFocus::Timeline);
120 if (current != previous && currentIsDaw && previousIsDaw) {
121 return WorkspaceTransitionKind::PlaybackHotSwap;
122 }
123 return WorkspaceTransitionKind::Ordinary;
124}
