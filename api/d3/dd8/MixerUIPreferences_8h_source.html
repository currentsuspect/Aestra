83 static MixerUIPreferences load(const std::string& path)
84 {
85 MixerUIPreferences prefs;
86 if (path.empty()) {
87 return prefs;
88 }
89
90 std::ifstream in(path);
91 if (!in) {
92 return prefs; // First run.
93 }
94
95 const std::string contents((std::istreambuf_iterator<char>(in)),
96 std::istreambuf_iterator<char>());
97 if (contents.empty()) {
98 return prefs;
99 }
100
101 bool consumedAll = false;
102 const JSON root = JSON::parseStrict(contents, consumedAll);
103 if (!consumedAll) {
104 return prefs;
105 }
106 // Type-checked, not merely present. JSON::asBool() returns its default
107 // false for any non-Boolean node, so {"inspectorExpanded": 1} would
108 // otherwise pass the has() check and collapse the inspector — the same
109 // failure the absent-key guard exists to prevent, arriving by a
110 // different route.
111 if (root.has("inspectorExpanded") && root["inspectorExpanded"].isBool()) {
112 prefs.inspectorExpanded = root["inspectorExpanded"].asBool();
113 }
114 return prefs;
115 }
116
118 bool save(const std::string& path) const
119 {
120 if (path.empty()) {
121 return false;
122 }
123
124 JSON root = JSON::object();
125 root.set("version", JSON(1.0));
126 // Only the explicit preference is written. The width-derived collapse is
127 // recomputed per layout and must never reach disk, or a session that
128 // happened to end in a narrow window would rewrite the choice.
129 root.set("inspectorExpanded", JSON(inspectorExpanded));
130
131 std::ofstream out(path, std::ios::trunc);
132 if (!out) {
133 return false;
134 }
135 out << root.toString(2);
136 // Closed explicitly before the status check: the destructor would close
137 // too, but by then the flush result is unobservable — a settings file
138 // truncated by a full disk would report success.
139 out.close();
140 return out.good();
141 }
