17uint8_t toMidiVelocity(float velocity) {
18 const float clamped = std::max(0.0f, velocity);
19 if (clamped <= 1.0f) {
20 return static_cast<uint8_t>(std::clamp<int>(static_cast<int>(std::lround(clamped * 127.0f)), 0, 127));
21 }
22 return static_cast<uint8_t>(std::clamp<int>(static_cast<int>(std::lround(clamped)), 0, 127));
23}

25bool eventComesBefore(const ScheduledEvent& a, const ScheduledEvent& b) {
26 if (a.sampleFrame != b.sampleFrame) {
27 return a.sampleFrame < b.sampleFrame;
28 }
29 if (a.priority != b.priority) {
30 return a.priority < b.priority;
31 }
32 if (a.instanceId != b.instanceId) {
33 return a.instanceId < b.instanceId;
34 }
35 if (a.statusByte != b.statusByte) {
36 return a.statusByte < b.statusByte;
37 }
38 if (a.data1 != b.data1) {
39 return a.data1 < b.data1;
40 }
41 return a.data2 < b.data2;
42}

44constexpr double kPitchedSamplerStepBeats = 0.25;

46const MidiNote* findNextActiveStepForUnit(const MidiPayload& midi, const MidiNote& currentNote) {
47 const double nextStepBeat = currentNote.startBeat + kPitchedSamplerStepBeats;
48 for (const auto& candidate : midi.notes) {
49 if (candidate.unitId != currentNote.unitId || candidate.durationBeats <= 0.0) {
50 continue;
51 }
52 if (std::abs(candidate.startBeat - nextStepBeat) <= 0.001) {
53 return &candidate;
54 }
55 }
56 return nullptr;
57}

59int resolveSamplerRootMidiNote(const UnitInfo* unit) {
60 if (!unit) {
61 return 60;
62 }
63 auto sampler = std::dynamic_pointer_cast<Plugins::SamplerPlugin>(unit->plugin);
64 return sampler ? sampler->getRootMidiNote() : 60;
65}

67int resolvePitchedSamplerMidiNote(const MidiNote& note, int rootMidiNote) {
68 if (note.pitchOffset == 0 && note.pitch > 0 && note.pitch != rootMidiNote) {
69 return std::clamp(note.pitch, 0, 127);
70 }
71 return std::clamp(rootMidiNote + static_cast<int>(note.pitchOffset), 0, 127);
72}

85void PatternPlaybackEngine::schedulePatternInstance(PatternID pid, double startBeat, uint32_t instanceId,
86 double sourceStartBeat, double durationBeats) {
87 if (instanceId >= 256) {
88 Aestra::Log::error("[PatternPlayback] Instance ID must be < 256");
89 return;
90 }
91
92 // Reset cancellation flag
93 m_instanceCancelled[instanceId].store(false, std::memory_order_release);
94
95 {
96 std::lock_guard<std::mutex> lock(m_mutex);
97 // Add to active instances
98 PatternInstance inst;
99 inst.patternId = pid;
100 inst.startBeat = startBeat;
101 inst.instanceId = instanceId;
102 inst.sourceStartBeat = std::max(0.0, sourceStartBeat);
103 inst.sourceEndBeat =
104 durationBeats > 0.0 ? (inst.sourceStartBeat + durationBeats) : std::numeric_limits<double>::infinity();
105 inst.scheduledThroughFrame = 0;
106
107 m_activeInstances.push_back(inst);
108 }
109
110 Aestra::Log::info("[PatternPlayback] Scheduled instance " + std::to_string(instanceId) + " pattern " +
111 std::to_string(pid.value) + " at beat " + std::to_string(startBeat) +
112 " sourceStart=" + std::to_string(sourceStartBeat) +
113 " duration=" + std::to_string(durationBeats));
114}

116void PatternPlaybackEngine::cancelPatternInstance(uint32_t instanceId) {
117 if (instanceId >= 256)
118 return;
119
120 // Set cancellation flag (RT-safe)
121 m_instanceCancelled[instanceId].store(true, std::memory_order_release);
122
123 {
124 std::lock_guard<std::mutex> lock(m_mutex);
125 // Remove from active instances (non-RT)
126 m_activeInstances.erase(
127 std::remove_if(m_activeInstances.begin(), m_activeInstances.end(),
128 [instanceId](const PatternInstance& inst) { return inst.instanceId == instanceId; }),
129 m_activeInstances.end());
130 }

141void PatternPlaybackEngine::refillWindow(uint64_t currentFrame, int sampleRate, int lookaheadSamples) {
142 // RT safety: this method must NOT be called from the audio callback.
143 // It acquires a mutex, allocates from scratch buffer, and calls pattern lookup.
144 // See performNonRealtimeMaintenance() for the correct call site.
145 if (reportRealtimeMisuse("PatternPlaybackEngine::refillWindow")) {
146 return;
147 }
148
149 uint64_t windowEnd = currentFrame + lookaheadSamples;
150
151 std::lock_guard<std::mutex> lock(m_mutex);
152
153 // Process deferred flush from audio thread
154 if (m_flushRequested.exchange(false, std::memory_order_acq_rel)) {
155 for (auto& inst : m_activeInstances) {
156 inst.scheduledThroughFrame = 0;
157 }
158 m_lastRefillFrame = 0;
159 }
160
161 if (currentFrame < m_lastRefillFrame) {
162 for (auto& inst : m_activeInstances) {
163 inst.scheduledThroughFrame = 0;
164 }
165 }
166 m_lastRefillFrame = currentFrame;
167
168 m_scratchEvents.clear();
169
170 for (auto& inst : m_activeInstances) {
171 // Skip cancelled instances
172 if (m_instanceCancelled[inst.instanceId].load(std::memory_order_acquire)) {
173 continue;
174 }
175
176 // Get pattern events
177 auto* pattern = m_patternManager->getPattern(inst.patternId);
178 if (!pattern) {
179 continue;
180 }
181 if (!pattern->isMidi()) {
182 continue;
183 }
184
185 auto& midi = std::get<MidiPayload>(pattern->payload);
186
187 const uint64_t previousScheduledThroughFrame = inst.scheduledThroughFrame;
188 const uint64_t scheduleFromFrame = std::max(currentFrame, previousScheduledThroughFrame);
189 if (scheduleFromFrame >= windowEnd) {
190 continue;
191 }
192
193 for (const auto& note : midi.notes) {
194 const double noteStartInSource = note.startBeat;
195 const double noteEndInSource = note.startBeat + note.durationBeats;
196 if (noteEndInSource <= inst.sourceStartBeat || noteStartInSource >= inst.sourceEndBeat) {
197 continue;
198 }
199
200 const UnitInfo* unit = m_unitManager->getUnit(note.unitId);
201 const bool isPitchedSampler = unit && unit->type == UnitType::PitchedSampler;
202 const int resolvedMidiNote = isPitchedSampler
203 ? resolvePitchedSamplerMidiNote(note, resolveSamplerRootMidiNote(unit))
204 : std::clamp(note.pitch, 0, 127);
205 double noteBeat = inst.startBeat + note.startBeat;
206 uint64_t noteFrame = m_clock->sampleFrameAtBeat(noteBeat, sampleRate);
207 double offBeat = std::min(noteBeat + note.durationBeats, inst.startBeat + inst.sourceEndBeat);
208 bool suppressNoteOff = false;
209
210 if (isPitchedSampler) {
211 const double gate = std::clamp(static_cast<double>(note.gate), 0.1, 2.0);
212 offBeat = std::min(noteBeat + gate * kPitchedSamplerStepBeats, inst.startBeat + inst.sourceEndBeat);
213
214 if (const auto* nextStep = findNextActiveStepForUnit(midi, note)) {
215 const double nextStepBeat = inst.startBeat + nextStep->startBeat;
216 if (note.slide) {
217 suppressNoteOff = true;
218 } else {
219 offBeat = std::min(offBeat, nextStepBeat);
220 }
221 }
222 }
223
224 uint64_t offFrame = m_clock->sampleFrameAtBeat(offBeat, sampleRate);
225
226 uint16_t channelIdx = getChannelForUnit(note.unitId);
227
228 // Off-centre notes ship their pan as a polyphonic-aftertouch event
229 // (0xA0, note, pan) at the same frame with priority 1, so it lands
230 // after note-offs but before every note-on of that frame. Keyed by
231 // note number, simultaneous chord notes each latch their own pan.
232 // Centred notes emit nothing — the sampler defaults to centre.
233 // Encoded 1..127 (64 = centre); 0 is the sampler's "no pan" slot.
234 const bool hasPan = std::abs(note.pan) > 0.004f;
235 const uint8_t panByte =
236 static_cast<uint8_t>(std::clamp<long>(64 + std::lround(note.pan * 64.0f), 1, 127));
237
238 if (noteFrame >= scheduleFromFrame && noteFrame < windowEnd) {
239 if (hasPan) {
240 ScheduledEvent panEvent{};
241 panEvent.sampleFrame = noteFrame;
242 panEvent.instanceId = inst.instanceId;
243 panEvent.unitId = note.unitId;
244 panEvent.channelIdx = channelIdx;
245 panEvent.statusByte = 0xA0;
246 panEvent.data1 = static_cast<uint8_t>(resolvedMidiNote);
247 panEvent.data2 = panByte;
248 panEvent.priority = 1;
249 m_scratchEvents.push_back(panEvent);
250 }
251 ScheduledEvent onEvent{};
252 onEvent.sampleFrame = noteFrame;
253 onEvent.instanceId = inst.instanceId;
254 onEvent.unitId = note.unitId;
255 onEvent.channelIdx = channelIdx;
256 onEvent.statusByte = 0x90;
257 onEvent.data1 = static_cast<uint8_t>(resolvedMidiNote);
258 onEvent.data2 = toMidiVelocity(note.velocity);
259 onEvent.priority = 2;
260 m_scratchEvents.push_back(onEvent);
261 } else if (noteFrame < scheduleFromFrame && offFrame > scheduleFromFrame &&
262 noteFrame >= previousScheduledThroughFrame) {
263 // Playback entered while this note was already active; start it immediately at the buffer edge once.
264 if (hasPan) {
265 ScheduledEvent panEvent{};
266 panEvent.sampleFrame = scheduleFromFrame;
267 panEvent.instanceId = inst.instanceId;
268 panEvent.unitId = note.unitId;
269 panEvent.channelIdx = channelIdx;
270 panEvent.statusByte = 0xA0;
271 panEvent.data1 = static_cast<uint8_t>(resolvedMidiNote);
272 panEvent.data2 = panByte;
273 panEvent.priority = 1;
274 m_scratchEvents.push_back(panEvent);
275 }
276 ScheduledEvent resumeOnEvent{};
277 resumeOnEvent.sampleFrame = scheduleFromFrame;
278 resumeOnEvent.instanceId = inst.instanceId;
279 resumeOnEvent.unitId = note.unitId;
280 resumeOnEvent.channelIdx = channelIdx;
281 resumeOnEvent.statusByte = 0x90;
282 resumeOnEvent.data1 = static_cast<uint8_t>(resolvedMidiNote);
283 resumeOnEvent.data2 = toMidiVelocity(note.velocity);
284 resumeOnEvent.priority = 2;
285 m_scratchEvents.push_back(resumeOnEvent);
286 }
287
288 if (!suppressNoteOff && offFrame >= scheduleFromFrame && offFrame < windowEnd) {
289 ScheduledEvent offEvent{};
290 offEvent.sampleFrame = offFrame;
291 offEvent.instanceId = inst.instanceId;
292 offEvent.unitId = note.unitId;
293 offEvent.channelIdx = channelIdx;
294 offEvent.statusByte = 0x80;
295 offEvent.data1 = static_cast<uint8_t>(resolvedMidiNote);
296 offEvent.data2 = 0;
297 offEvent.priority = 0;
298 m_scratchEvents.push_back(offEvent);
299 }
300 }
301
302 inst.scheduledThroughFrame = windowEnd;
303 }
304
305 std::sort(m_scratchEvents.begin(), m_scratchEvents.end(), eventComesBefore);
306 for (const auto& event : m_scratchEvents) {
307 if (!m_rtQueue.push(event)) {
308 m_overflowCounter.fetch_add(1, std::memory_order_relaxed);
309 }
310 }
311}

313void PatternPlaybackEngine::processAudio(uint64_t currentFrame, int bufferSize, const UnitMidiRoute* routes,
314 size_t routeCount) noexcept {
315 if (!routes || routeCount == 0 || bufferSize <= 0) {
316 return;
317 }
318
319 ScheduledEvent ev;
320
321 while (m_rtQueue.peek(ev)) {
322 // Check if event is in current buffer
323 if (ev.sampleFrame >= currentFrame + static_cast<uint64_t>(bufferSize)) {
324 break; // Event is in the future
325 }
326
327 // Check cancellation flag (RT-safe)
328 if (ev.instanceId < 256 && m_instanceCancelled[ev.instanceId].load(std::memory_order_acquire)) {
329 m_rtQueue.pop();
330 continue;
331 }
332
333 // Calculate offset in buffer
334 int offset = static_cast<int>(ev.sampleFrame - currentFrame);
335 offset = std::max(0, std::min(offset, bufferSize - 1));
336
337 // Route to Unit
338 if (ev.unitId != 0) {
339 MidiBuffer* target = nullptr;
340 for (size_t i = 0; i < routeCount; ++i) {
341 if (routes[i].unitId == ev.unitId) {
342 target = routes[i].midiBuffer;
343 break;
344 }
345 }
346
347 if (target) {
348 uint8_t data[3] = {ev.statusByte, ev.data1, ev.data2};
349 target->addEvent(static_cast<uint32_t>(offset), data, 3);
350 m_processedCounter.fetch_add(1, std::memory_order_relaxed);
351 }
352 }
353
354 m_rtQueue.pop();
355 }

358void PatternPlaybackEngine::flush() {
359 // RT-safe: set atomic flag for deferred processing by non-RT maintenance.
360 // The SPSC queue and m_activeInstances are only safely mutable from the
361 // control thread, so actual drain/reset happens in refillWindow().
362 m_flushRequested.store(true, std::memory_order_release);
363
364 // Drain the RT queue lock-free from the producer side.
365 // Resetting m_head to m_tail is safe because the consumer (audio thread)
366 // only reads m_tail, and any events pushed after this reset are legitimate.
367 m_rtQueue.forceDrain();
368}
