19uint8_t toMidiVelocity(float velocity) {
20 const float clamped = std::max(0.0f, velocity);
21 if (clamped <= 1.0f) {
22 return static_cast<uint8_t>(std::clamp<int>(static_cast<int>(std::lround(clamped * 127.0f)), 0, 127));
23 }
24 return static_cast<uint8_t>(std::clamp<int>(static_cast<int>(std::lround(clamped)), 0, 127));
25}

27// One id per note occurrence: its instance, its ON and OFF frames (distinct per loop pass; the
28// off frame tells apart stacked same-pitch notes of different lengths) and its resolved pitch.
29// The note-off is often scheduled in a later window than the note-on, so the id must be
30// recomputable from the occurrence alone. Never 0 (0 means "no id"). An edit that moves a
31// sounding note changes this id; the gate record keeps the voice's id (see refillWindow).
32uint32_t noteOccurrenceId(uint32_t instanceId, uint64_t noteFrame, uint64_t offFrame, uint8_t midiNote) {
33 uint64_t h = 0x9E3779B97F4A7C15ull ^ (static_cast<uint64_t>(instanceId) << 32) ^ midiNote;
34 h ^= noteFrame + 0x9E3779B97F4A7C15ull + (h << 6) + (h >> 2);
35 h ^= offFrame + 0x9E3779B97F4A7C15ull + (h << 6) + (h >> 2);
36 h ^= h >> 33;
37 h *= 0xff51afd7ed558ccdull;
38 h ^= h >> 33;
39 const uint32_t id = static_cast<uint32_t>(h ^ (h >> 32));
40 return id != 0 ? id : 1u;
41}

43bool eventComesBefore(const ScheduledEvent& a, const ScheduledEvent& b) {
44 if (a.sampleFrame != b.sampleFrame) {
45 return a.sampleFrame < b.sampleFrame;
46 }
47 if (a.priority != b.priority) {
48 return a.priority < b.priority;
49 }
50 if (a.instanceId != b.instanceId) {
51 return a.instanceId < b.instanceId;
52 }
53 if (a.statusByte != b.statusByte) {
54 return a.statusByte < b.statusByte;
55 }
56 if (a.data1 != b.data1) {
57 return a.data1 < b.data1;
58 }
59 return a.data2 < b.data2;
60}

62constexpr double kPitchedSamplerStepBeats = 0.25;
63
64const MidiNote* findNextActiveStepForUnit(const MidiPayload& midi, const MidiNote& currentNote) {
65 const double nextStepBeat = currentNote.startBeat + kPitchedSamplerStepBeats;
66 for (const auto& candidate : midi.notes) {
67 if (candidate.unitId != currentNote.unitId || candidate.durationBeats <= 0.0) {
68 continue;
69 }
70 if (std::abs(candidate.startBeat - nextStepBeat) <= 0.001) {
71 return &candidate;
72 }
73 }
74 return nullptr;
75}

77// When a scheduled note ends, and whether it ends at all. ONE definition, used by the refill
78// that queues the note-off and by the live-edit reconciliation that decides whether a sounding
79// note still exists: a pitched-sampler gate can outlast the note's duration, so the two
80// computing it differently released notes early (or kept stale ones).
81struct NoteOffTiming {
82 double offBeat;
83 bool suppressNoteOff; // pitched-sampler slide into the next step holds the voice
84};
85
86// How far past its own start one repeat of a pattern can still ring: its longest note, plus a beat for swing
87// delay and pitched-sampler gates. Bounds the repeats a window walks (PatternRepeat::inWindow).
88double repeatReachBeats(const std::vector<MidiNote>& notes, double patternLengthBeats) {
89 double reach = patternLengthBeats;
90 for (const auto& n : notes) reach = std::max(reach, n.startBeat + n.durationBeats);
91 return reach + 1.0;
92}

94NoteOffTiming calculateNoteOffTiming(const MidiPayload& midi, const MidiNote& note, double instanceStartBeat,
95 double sourceEndBeatAbsolute, bool isPitchedSampler) {
96 const double noteBeat = instanceStartBeat + note.startBeat;
97 NoteOffTiming timing{std::min(noteBeat + note.durationBeats, sourceEndBeatAbsolute), false};
98 if (isPitchedSampler) {
99 const double gate = std::clamp(static_cast<double>(note.gate), 0.1, 2.0);
100 timing.offBeat = std::min(noteBeat + gate * kPitchedSamplerStepBeats, sourceEndBeatAbsolute);
101 if (const auto* nextStep = findNextActiveStepForUnit(midi, note)) {
102 if (note.slide) {
103 timing.suppressNoteOff = true;
104 } else {
105 timing.offBeat = std::min(timing.offBeat, instanceStartBeat + nextStep->startBeat);
106 }
107 }
108 }
109 return timing;
110}

112int resolvePitchedSamplerMidiNote(const MidiNote& note, int rootMidiNote) {
113 if (note.pitchOffset == 0 && note.pitch > 0 && note.pitch != rootMidiNote) {
114 return std::clamp(note.pitch, 0, 127);
115 }
116 return std::clamp(rootMidiNote + static_cast<int>(note.pitchOffset), 0, 127);
117}

130void PatternPlaybackEngine::schedulePatternInstance(PatternID pid, double startBeat, uint32_t instanceId,
131 double sourceStartBeat, double durationBeats, bool repeatPattern) {
132 if (instanceId >= 256) {
133 Aestra::Log::error("[PatternPlayback] Instance ID must be < 256");
134 return;
135 }
136
137 // Reset cancellation flag
138 m_instanceCancelled[instanceId].store(false, std::memory_order_release);
139
140 {
141 std::lock_guard<std::mutex> lock(m_mutex);
142 PatternInstance inst;
143 inst.patternId = pid;
144 inst.startBeat = startBeat;
145 inst.instanceId = instanceId;
146 inst.sourceStartBeat = std::max(0.0, sourceStartBeat);
147 inst.sourceEndBeat =
148 durationBeats > 0.0 ? (inst.sourceStartBeat + durationBeats) : std::numeric_limits<double>::infinity();
149 inst.repeatPattern = repeatPattern;
150 inst.scheduledThroughFrame = 0;
151
152 // An instanceId identifies a SLOT, not an occurrence — cancellation is keyed on
153 // it (m_instanceCancelled is indexed by id), so two live entries sharing an id
154 // were never individually addressable anyway. This used to push_back
155 // unconditionally: the Arsenal preview always re-arms slot 1, and nothing ever
156 // erased entries (rewindScheduledInstances() only rewinds scheduledThroughFrame),
157 // so a session accumulated dozens of overlapping copies — 31 observed in one
158 // sitting — each emitting its own events into the same units. Re-arming a slot
159 // must replace it.
160 auto existing = std::find_if(m_activeInstances.begin(), m_activeInstances.end(),
161 [instanceId](const PatternInstance& e) { return e.instanceId == instanceId; });
162 if (existing != m_activeInstances.end()) {
163 *existing = inst;
164 } else {
165 m_activeInstances.push_back(inst);
166 }
167 m_instanceCount.store(m_activeInstances.size(), std::memory_order_release);
168 }
169
170 Aestra::Log::info("[PatternPlayback] Scheduled instance " + std::to_string(instanceId) + " pattern " +
171 std::to_string(pid.value) + " at beat " + std::to_string(startBeat) +
172 " sourceStart=" + std::to_string(sourceStartBeat) +
173 " duration=" + std::to_string(durationBeats) + (repeatPattern ? " repeats" : ""));
174}

176void PatternPlaybackEngine::updatePatternInstance(PatternID pid, double startBeat, uint32_t instanceId,
177 double sourceStartBeat, double durationBeats, bool repeatPattern) {
178 if (instanceId >= 256) {
179 Aestra::Log::error("[PatternPlayback] Instance ID must be < 256");
180 return;
181 }
182 {
183 std::lock_guard<std::mutex> lock(m_mutex);
184 auto existing = std::find_if(m_activeInstances.begin(), m_activeInstances.end(),
185 [instanceId](const PatternInstance& e) { return e.instanceId == instanceId; });
186 if (existing != m_activeInstances.end()) {
187 existing->patternId = pid;
188 existing->startBeat = startBeat;
189 existing->sourceStartBeat = std::max(0.0, sourceStartBeat);
190 existing->sourceEndBeat = durationBeats > 0.0 ? (existing->sourceStartBeat + durationBeats)
191 : std::numeric_limits<double>::infinity();
192 existing->repeatPattern = repeatPattern;
193 // scheduledThroughFrame deliberately kept: see the header.
194 return;
195 }
196 }
197 schedulePatternInstance(pid, startBeat, instanceId, sourceStartBeat, durationBeats, repeatPattern);
198}

244void PatternPlaybackEngine::refillWindow(uint16_t ticket, uint64_t currentFrame, int sampleRate,
245 int lookaheadSamples, const ScheduleLoop& loop) {
246 // RT safety: this method must NOT be called from the audio callback.
247 // It acquires a mutex, allocates from scratch buffer, and calls pattern lookup.
248 // See performNonRealtimeMaintenance() for the correct call site.
249 if (reportRealtimeMisuse("PatternPlaybackEngine::refillWindow")) {
250 return;
251 }
252
253 uint64_t windowEnd = currentFrame + lookaheadSamples;
254
255 std::lock_guard<std::mutex> lock(m_mutex);
256
257 // A rewind requested since the last refill (from the audio thread or the UI) bumped the
258 // queue epoch; everything queued under the old one is already dead to processAudio().
259 // The ticket predates the caller's playhead read, so a rewind after that read leaves
260 // this refill stamping an epoch that is already stale: its events drop unplayed and
261 // the next refill replans from the new position.
262 if (ticket != m_scheduledEpoch) {
263 for (auto& inst : m_activeInstances) {
264 inst.scheduledThroughFrame = 0;
265 }
266 m_lastRefillFrame = 0;
267 m_scheduledEpoch = ticket;
268 }
269
270 if (currentFrame < m_lastRefillFrame) {
271 for (auto& inst : m_activeInstances) {
272 inst.scheduledThroughFrame = 0;
273 }
274 // The frame domain jumped backwards (bounce restart, stop→locate):
275 // events already queued for the old, higher frames — including the
276 // next loop iteration pre-scheduled before a wrap — would sit at the
277 // queue head and block everything queued below them. Discard them; the
278 // window below re-queues whatever is actually due.
279 invalidateQueuedEventsLocked();
280 }
281 m_lastRefillFrame = currentFrame;
282
283 // Gate records whose note-off already fired are dead weight.
284 m_gatedNotes.erase(std::remove_if(m_gatedNotes.begin(), m_gatedNotes.end(),
285 [currentFrame](const GatedNote& g) { return g.offFrame < currentFrame; }),
286 m_gatedNotes.end());
287
288 // Content edit while playing: pull each instance's frontier back to the
289 // playhead so this pass re-queues from live pattern data. Deletions drop
290 // out of the queue immediately and additions ahead of the playhead enter
291 // at their exact frame. The frontier stays > 0, so the entry catch-up in
292 // the loop below stays dormant — a full rewind here re-fired every
293 // currently-sounding note (flam on each grid edit) and made fresh
294 // placements sound before the playhead reached them (#823).
295 // Edits announce themselves through patternContentEdited() from the Arsenal grid, but the
296 // Piano Roll, Muse note verbs, undo and a dozen direct note writes do not. With the longer
297 // lookahead (performNonRealtimeMaintenance) an unannounced edit would wait up to a quarter
298 // second to be heard, so every active pattern is fingerprinted on each pass and any change is
299 // treated exactly like an announced edit. First sight of an instance only records its print.
300 bool contentEdited = m_contentEditRequested.exchange(false, std::memory_order_acq_rel);
301 for (auto& inst : m_activeInstances) {
302 const PatternSource* pattern = m_patternManager ? m_patternManager->getPattern(inst.patternId) : nullptr;
303 const uint64_t print = pattern ? contentFingerprint(*pattern) : 0;
304 if (inst.contentFingerprint != 0 && print != inst.contentFingerprint) {
305 contentEdited = true;
306 }
307 inst.contentFingerprint = print;
308 }
309 if (contentEdited) {
310 // A note deleted while still inside its gate leaves a hung voice: its
311 // ON already fired but its queued OFF dies with the drain below, and
312 // the note is gone from the pattern so the refill loop cannot re-emit
313 // it. Dispatch immediate OFFs for tracked gates whose note no longer
314 // exists; survivors are untouched — their OFFs are re-queued normally
315 // by the loop below.
316 auto findInstance = [this](uint32_t id) -> PatternInstance* {
317 for (auto& inst : m_activeInstances)
318 if (inst.instanceId == id)
319 return &inst;
320 return nullptr;
321 };
322
323 std::vector<ScheduledEvent> deletedOffs;
324 const uint64_t playheadPass = loop.passOf(currentFrame);
325 const uint64_t loopBase = loop.baseOf(playheadPass);
326 for (const auto& gated : m_gatedNotes)
327 {
328 continue;
329 }
330 // Gate records released here, or already past, are dropped after the pass. A surviving
331 // record takes its note's CURRENT off frame: an edit that lengthened a sounding note left
332 // the old one, the record was purged once that passed while the note still sounded, and a
333 // later edit deleting the note then had nothing to release it with (review, #956).
334 std::vector<size_t> consumedGates;
335 // Each gate follows ITS note: first the note whose id is unchanged (an unedited note,
336 // even among overlapping same-pitch notes); only for gates left over, a same-pitch note
337 // still spanning the playhead that no other gate claimed (the note was moved or resized
338 // while sounding; the gate adopts its new id and keeps its voice's). Exact matches run
339 // for every gate before any fallback, so a fallback can't take another gate's note.
340 struct GateMatch {
341 bool found = false;
342 uint64_t offFrame = 0;
343 uint32_t currentId = 0;
344 };
345 std::vector<GateMatch> matches(m_gatedNotes.size());
346 std::vector<uint32_t> claimedIds;
347 const auto forEachSpanningNote = [&](const GatedNote& gated, auto&& visit) {
348 PatternInstance* instPtr = findInstance(gated.instanceId);
349 // A removed slot (a clip deleted, muted or moved out of the timeline while its
350 // note sounds) can re-emit nothing, so its held note is released below or it hangs.
351 auto* pattern = instPtr ? m_patternManager->getPattern(instPtr->patternId) : nullptr;
352 if (!instPtr || !pattern || !pattern->isMidi()) return;
353 const auto& patternNotes = std::get<MidiPayload>(pattern->payload).notes;
354 const bool expressive = needsExpression(patternNotes, pattern->swing);
355 // The repeats that can be sounding at the playhead (V82-T3); just repeat 0 when the clip doesn't repeat.
356 const double repeatLength = instPtr->repeatPattern ? pattern->lengthBeats : 0.0;
357 const double playheadBeat =
358 m_clock->beatAtSampleFrame(currentFrame > loopBase ? currentFrame - loopBase : 0, sampleRate);
359 const PatternRepeat::Range repeatsHere = PatternRepeat::inWindow(
360 repeatLength, instPtr->sourceStartBeat, instPtr->sourceEndBeat, instPtr->startBeat, playheadBeat,
361 playheadBeat, instPtr->repeatPattern ? repeatReachBeats(patternNotes, repeatLength) : 0.0);
362 for (int64_t repeat = repeatsHere.first; repeat <= repeatsHere.last; ++repeat) {
363 const double repeatStartBeat = instPtr->startBeat + static_cast<double>(repeat) * repeatLength;
364 const std::vector<MidiNote> expandedNotes =
365 expressive ? expandForPlayback(patternNotes, pattern->swing, pattern->id.value, instPtr->instanceId,
366 PatternRepeat::passSeed(
367 playheadPass, repeat))
368 : std::vector<MidiNote>{};
369 for (const auto& note : (expressive ? expandedNotes : patternNotes)) {
370 if (note.unitId != gated.unitId || (repeatLength > 0.0 && note.startBeat >= repeatLength))
371 continue;
372 const UnitInfo* unit = m_unitManager->getUnit(note.unitId);
373 const bool isPitchedSampler = unit && unit->type == UnitType::PitchedSampler;
374 int resolved = std::clamp(note.pitch, 0, 127);
375 if (isPitchedSampler) {
376 const auto sampler =
377 unit ? std::dynamic_pointer_cast<Plugins::SamplerPlugin>(unit->plugin) : nullptr;
378 resolved = resolvePitchedSamplerMidiNote(note, sampler ? sampler->getRootMidiNote() : 60);
379 }
380 if (resolved != static_cast<int>(gated.noteNumber))
381 continue;
382 const double onBeat = repeatStartBeat + note.startBeat;
383 uint64_t onFrame = loopBase + m_clock->sampleFrameAtBeat(onBeat, sampleRate);
384 // The note must still SPAN the playhead at its current placement. Started
385 // before the playhead is not enough: a clip moved (or a note shortened) so the
386 // note has already ended there gets no re-queued off, so the old voice would hang.
387 const NoteOffTiming offTiming =
388 calculateNoteOffTiming(std::get<MidiPayload>(pattern->payload), note, repeatStartBeat,
389 instPtr->startBeat + instPtr->sourceEndBeat, isPitchedSampler);
390 uint64_t offFrame = loopBase + m_clock->sampleFrameAtBeat(offTiming.offBeat, sampleRate);
391 if (!loop.placeInPass(playheadPass, onFrame, offFrame))
392 continue;
393 if (onFrame < currentFrame && (offTiming.suppressNoteOff || offFrame > currentFrame)) {
394 const uint32_t id =
395 noteOccurrenceId(gated.instanceId, onFrame, offFrame, static_cast<uint8_t>(resolved));
396 if (visit(id, offFrame)) return;
397 }
398 }
399 } // repeat
400 };
401 const auto claimed = [&](uint32_t id) {
402 return std::find(claimedIds.begin(), claimedIds.end(), id) != claimedIds.end();
403 };
404 for (size_t i = 0; i < m_gatedNotes.size(); ++i) { // exact: the unedited note
405 const auto& gated = m_gatedNotes[i];
406 if (gated.offFrame <= currentFrame) continue;
407 forEachSpanningNote(gated, [&](uint32_t id, uint64_t offFrame) {
408 if (id != gated.currentId || claimed(id)) return false;
409 matches[i] = {true, offFrame, id};
410 claimedIds.push_back(id);
411 return true;
412 });
413 }
414 for (size_t i = 0; i < m_gatedNotes.size(); ++i) { // fallback: the note was edited
415 const auto& gated = m_gatedNotes[i];
416 if (gated.offFrame <= currentFrame || matches[i].found) continue;
417 forEachSpanningNote(gated, [&](uint32_t id, uint64_t offFrame) {
418 if (claimed(id)) return false;
419 matches[i] = {true, offFrame, id};
420 claimedIds.push_back(id);
421 return true;
422 });
423 }
424 for (size_t gateIndex = 0; gateIndex < m_gatedNotes.size(); ++gateIndex) {
425 auto& gated = m_gatedNotes[gateIndex];
426 if (gated.offFrame <= currentFrame) {
427 consumedGates.push_back(gateIndex);
428 continue;
429 }
430 if (matches[gateIndex].found) {
431 gated.offFrame = matches[gateIndex].offFrame;
432 gated.currentId = matches[gateIndex].currentId;
433 continue;
434 }
435 ScheduledEvent off{};
436 off.sampleFrame = currentFrame;
437 off.instanceId = gated.instanceId;
438 off.unitId = gated.unitId;
439 off.channelIdx = gated.channelIdx;
440 off.statusByte = 0x80;
441 off.data1 = gated.noteNumber;
442 off.data2 = 0;
443 off.priority = 0;
444 off.noteId = gated.noteId;
445 deletedOffs.push_back(off);
446 consumedGates.push_back(gateIndex);
447 }
448 // Drop consumed gate records in the same pass (indices ascend; erase from the back).
449 for (auto it = consumedGates.rbegin(); it != consumedGates.rend(); ++it) {
450 m_gatedNotes.erase(m_gatedNotes.begin() + static_cast<std::ptrdiff_t>(*it));
451 }
452 for (auto& inst : m_activeInstances) {
453 inst.scheduledThroughFrame = std::min(inst.scheduledThroughFrame, currentFrame);
454 }
455 invalidateQueuedEventsLocked();
456 for (const auto& ev : deletedOffs) {
457 pushScheduledLocked(ev);
458 }
459 }
460
461 m_scratchEvents.clear();
462
463 bool anyLate = false;
464 uint64_t maxLate = 0;
465 uint64_t missed = 0;
466 for (auto& inst : m_activeInstances) {
467 // Skip cancelled instances
468 if (m_instanceCancelled[inst.instanceId].load(std::memory_order_acquire)) {
469 continue;
470 }
471
472 // Get pattern events
473 auto* pattern = m_patternManager->getPattern(inst.patternId);
474 if (!pattern) {
475 continue;
476 }
477 if (!pattern->isMidi()) {
478 continue;
479 }
480
481 auto& midi = std::get<MidiPayload>(pattern->payload);
482
483 const uint64_t previousScheduledThroughFrame = inst.scheduledThroughFrame;
484 const uint64_t scheduleFromFrame = std::max(currentFrame, previousScheduledThroughFrame);
485 // The playhead passed the last queued frame: this refill came later than the lookahead
486 // (a main-loop stall). Hits in [frontier, playhead) are gone; count them below.
487 const bool lateRefill = previousScheduledThroughFrame > 0 && currentFrame > previousScheduledThroughFrame;
488 if (lateRefill) {
489 anyLate = true;
490 maxLate = std::max(maxLate, currentFrame - previousScheduledThroughFrame);
491 }
492 if (scheduleFromFrame >= windowEnd) {
493 continue;
494 }
495
496 // Looped pattern mode works in a monotonic frame domain: iterate every
497 // loop pass that intersects the window so the next iteration's events
498 // (downbeat included) are queued BEFORE the wrap, sample-accurately.
499 uint64_t iterBegin = 0;
500 uint64_t iterEnd = 0;
501 if (loop.active()) {
502 iterBegin = loop.passOf(scheduleFromFrame);
503 iterEnd = loop.passOf(windowEnd - 1);
504 }
505
506 // A clip longer than its pattern plays it again every pattern length (V82-T3). Only the repeats that
507 // can sound in this window are walked; repeat r sits r * length later, in source and on the timeline.
508 // Without repeats the range is the single repeat 0 and nothing below moves.
509 const double repeatLength = inst.repeatPattern ? pattern->lengthBeats : 0.0;
510 const double repeatReach = inst.repeatPattern ? repeatReachBeats(midi.notes, repeatLength) : 0.0;
511 for (uint64_t loopIter = iterBegin; loopIter <= iterEnd; ++loopIter) {
512 const uint64_t loopBase = loop.baseOf(loopIter);
513 const PatternRepeat::Range repeatsHere = PatternRepeat::inWindow(
514 repeatLength, inst.sourceStartBeat, inst.sourceEndBeat, inst.startBeat,
515 m_clock->beatAtSampleFrame(scheduleFromFrame > loopBase ? scheduleFromFrame - loopBase : 0, sampleRate),
516 m_clock->beatAtSampleFrame(windowEnd - loopBase, sampleRate), repeatReach);
517 for (int64_t repeat = repeatsHere.first; repeat <= repeatsHere.last; ++repeat) {
518 const double repeatOffset = static_cast<double>(repeat) * repeatLength;
519 const double repeatStartBeat = inst.startBeat + repeatOffset;
520 std::vector<MidiNote> expandedNotes;
521 const bool expressive = needsExpression(midi.notes, pattern->swing);
522 if (expressive) expandedNotes = expandForPlayback(midi.notes, pattern->swing, pattern->id.value, inst.instanceId, PatternRepeat::passSeed(loopIter, repeat));
523 for (const auto& note : (expressive ? expandedNotes : midi.notes)) {
524 if (repeatLength > 0.0 && note.startBeat >= repeatLength) {
525 continue; // past the pattern's end: a repeating clip wraps there, so the note never plays
526 }
527 const double noteStartInSource = repeatOffset + note.startBeat;
528 const double noteEndInSource = noteStartInSource + note.durationBeats;
529 if (noteEndInSource <= inst.sourceStartBeat || noteStartInSource >= inst.sourceEndBeat) {
530 continue;
531 }
532
533 const UnitInfo* unit = m_unitManager->getUnit(note.unitId);
534 const bool isPitchedSampler = unit && unit->type == UnitType::PitchedSampler;
535 const auto sampler = unit ? std::dynamic_pointer_cast<Plugins::SamplerPlugin>(unit->plugin) : nullptr;
536 const int resolvedMidiNote = isPitchedSampler
537 ? resolvePitchedSamplerMidiNote(note,
538 sampler ? sampler->getRootMidiNote() : 60)
539 : std::clamp(note.pitch, 0, 127);
540 double noteBeat = repeatStartBeat + note.startBeat;
541 uint64_t noteFrame = loopBase + m_clock->sampleFrameAtBeat(noteBeat, sampleRate);
542 // A one-shot sample may continue to its natural endpoint when no
543 // note gate is present, but a Piano Roll note still owns an ADSR
544 // gate. Always emit its note-off so the sampler can enter release
545 // when the note ends. Pitched-sampler slides are the one explicit
546 // exception because they intentionally hold the voice across the
547 // next step.
548 const NoteOffTiming offTiming =
549 calculateNoteOffTiming(midi, note, repeatStartBeat, inst.startBeat + inst.sourceEndBeat, isPitchedSampler);
550 const double offBeat = offTiming.offBeat;
551 const bool suppressNoteOff = offTiming.suppressNoteOff;
552
553 uint64_t offFrame = loopBase + m_clock->sampleFrameAtBeat(offBeat, sampleRate);
554 if (!loop.placeInPass(loopIter, noteFrame, offFrame)) {
555 continue; // a timeline loop pass this note does not sound in
556 }
557 const uint32_t noteId =
558 noteOccurrenceId(inst.instanceId, noteFrame, offFrame, static_cast<uint8_t>(resolvedMidiNote));
559
560 uint16_t channelIdx = getChannelForUnit(note.unitId);
561
562 // Off-centre notes ship their pan as a polyphonic-aftertouch event
563 // (0xA0, note, pan) at the same frame with priority 1, so it lands
564 // after note-offs but before every note-on of that frame. Keyed by
565 // note number, simultaneous chord notes each latch their own pan.
566 // Centred notes emit nothing — the sampler defaults to centre.
567 // Encoded 1..127 (64 = centre); 0 is the sampler's "no pan" slot.
568 const bool hasPan = std::abs(note.pan) > 0.004f;
569 const uint8_t panByte =
570 static_cast<uint8_t>(std::clamp<long>(64 + std::lround(note.pan * 64.0f), 1, 127));
571
572 if (noteFrame >= scheduleFromFrame && noteFrame < windowEnd) {
573 if (hasPan) {
574 ScheduledEvent panEvent{};
575 panEvent.sampleFrame = noteFrame;
576 panEvent.instanceId = inst.instanceId;
577 panEvent.unitId = note.unitId;
578 panEvent.channelIdx = channelIdx;
579 panEvent.statusByte = 0xA0;
580 panEvent.data1 = static_cast<uint8_t>(resolvedMidiNote);
581 panEvent.data2 = panByte;
582 panEvent.priority = 1;
583 m_scratchEvents.push_back(panEvent);
584 }
585 ScheduledEvent onEvent{};
586 onEvent.sampleFrame = noteFrame;
587 onEvent.instanceId = inst.instanceId;
588 onEvent.unitId = note.unitId;
589 onEvent.channelIdx = channelIdx;
590 onEvent.statusByte = 0x90;
591 onEvent.data1 = static_cast<uint8_t>(resolvedMidiNote);
592 onEvent.data2 = toMidiVelocity(note.velocity);
593 onEvent.priority = 2;
594 onEvent.noteId = noteId;
595 m_scratchEvents.push_back(onEvent);
596 if (!suppressNoteOff) {
597 m_gatedNotes.push_back({inst.instanceId, note.unitId,
598 static_cast<uint8_t>(resolvedMidiNote), channelIdx, offFrame, noteId, noteId});
599 }
600 } else if (lateRefill && noteFrame >= previousScheduledThroughFrame && noteFrame < scheduleFromFrame &&
601 offFrame <= scheduleFromFrame) {
602 ++missed; // fell in the gap and already over (the branch below restarts the rest): never sounded
603 } else if (noteFrame < scheduleFromFrame && offFrame > scheduleFromFrame &&
604 noteFrame >= previousScheduledThroughFrame) {
605 // Playback entered while this note was already active; start it immediately at the buffer edge once.
606 if (hasPan) {
607 ScheduledEvent panEvent{};
608 panEvent.sampleFrame = scheduleFromFrame;
609 panEvent.instanceId = inst.instanceId;
610 panEvent.unitId = note.unitId;
611 panEvent.channelIdx = channelIdx;
612 panEvent.statusByte = 0xA0;
613 panEvent.data1 = static_cast<uint8_t>(resolvedMidiNote);
614 panEvent.data2 = panByte;
615 panEvent.priority = 1;
616 m_scratchEvents.push_back(panEvent);
617 }
618 ScheduledEvent resumeOnEvent{};
619 resumeOnEvent.sampleFrame = scheduleFromFrame;
620 resumeOnEvent.instanceId = inst.instanceId;
621 resumeOnEvent.unitId = note.unitId;
622 resumeOnEvent.channelIdx = channelIdx;
623 resumeOnEvent.statusByte = 0x90;
624 resumeOnEvent.data1 = static_cast<uint8_t>(resolvedMidiNote);
625 resumeOnEvent.data2 = toMidiVelocity(note.velocity);
626 resumeOnEvent.priority = 2;
627 resumeOnEvent.noteId = noteId;
628 m_scratchEvents.push_back(resumeOnEvent);
629 if (!suppressNoteOff) {
630 m_gatedNotes.push_back({inst.instanceId, note.unitId,
631 static_cast<uint8_t>(resolvedMidiNote), channelIdx, offFrame, noteId, noteId});
632 }
633 }
634
635 if (!suppressNoteOff && offFrame >= scheduleFromFrame && offFrame < windowEnd) {
636 ScheduledEvent offEvent{};
637 offEvent.sampleFrame = offFrame;
638 offEvent.instanceId = inst.instanceId;
639 offEvent.unitId = note.unitId;
640 offEvent.channelIdx = channelIdx;
641 offEvent.statusByte = 0x80;
642 offEvent.data1 = static_cast<uint8_t>(resolvedMidiNote);
643 offEvent.data2 = 0;
644 offEvent.priority = 0;
645 // An edited sounding note: its gate keeps the id its voice was started with.
646 offEvent.noteId = noteId;
647 for (const auto& gated : m_gatedNotes) {
648 if (gated.currentId == noteId && gated.instanceId == inst.instanceId &&
649 gated.unitId == note.unitId && gated.noteNumber == static_cast<uint8_t>(resolvedMidiNote)) {
650 offEvent.noteId = gated.noteId;
651 break;
652 }
653 }
654 m_scratchEvents.push_back(offEvent);
655 }
656 }
657 } // repeat
658 } // loopIter
659
660 inst.scheduledThroughFrame = windowEnd;
661 }
662 if (anyLate) {
663 m_lateRefills.fetch_add(1, std::memory_order_relaxed);
664 m_missedNoteOns.fetch_add(missed, std::memory_order_relaxed);
665 uint64_t prev = m_maxLateFrames.load(std::memory_order_relaxed);
666 while (maxLate > prev && !m_maxLateFrames.compare_exchange_weak(prev, maxLate, std::memory_order_relaxed)) {
667 }
668 }
669
670 std::sort(m_scratchEvents.begin(), m_scratchEvents.end(), eventComesBefore);
671 for (const auto& event : m_scratchEvents) {
672 pushScheduledLocked(event);
673 }
674 // Published last: once this equals the live epoch, the queue holds this playhead's notes.
675 m_plannedEpoch.store(m_scheduledEpoch, std::memory_order_release);
676}

678void PatternPlaybackEngine::pushScheduledLocked(ScheduledEvent event) {
679 event.epoch = m_scheduledEpoch;
680 if (!m_rtQueue.push(event)) {
681 m_overflowCounter.fetch_add(1, std::memory_order_relaxed);
682 }
683}

685void PatternPlaybackEngine::processAudio(uint64_t currentFrame, int bufferSize, const UnitMidiRoute* routes,
686 size_t routeCount) noexcept {
687 if (!routes || routeCount == 0 || bufferSize <= 0) {
688 return;
689 }
690
691 ScheduledEvent ev;
692 uint16_t epoch = m_queueEpoch.load(std::memory_order_acquire);
693
694 while (m_rtQueue.peek(ev)) {
695 // Queued before the last rewind: drop it unplayed. A mismatch re-reads the epoch
696 // once, because an event published after a bump on another thread can be newer
697 // than the value loaded above.
698 if (ev.epoch != epoch) {
699 epoch = m_queueEpoch.load(std::memory_order_acquire);
700 if (ev.epoch != epoch) {
701 m_rtQueue.pop();
702 continue;
703 }
704 }
705
706 // Check if event is in current buffer
707 if (ev.sampleFrame >= currentFrame + static_cast<uint64_t>(bufferSize)) {
708 break; // Event is in the future
709 }
710
711 // Check cancellation flag (RT-safe)
712 if (ev.instanceId < 256 && m_instanceCancelled[ev.instanceId].load(std::memory_order_acquire)) {
713 m_rtQueue.pop();
714 continue;
715 }
716
717 // Calculate offset in buffer
718 int offset = static_cast<int>(ev.sampleFrame - currentFrame);
719 offset = std::max(0, std::min(offset, bufferSize - 1));
720
721 // Route to Unit
722 if (ev.unitId != 0) {
723 MidiBuffer* target = nullptr;
724 for (size_t i = 0; i < routeCount; ++i) {
725 if (routes[i].unitId == ev.unitId) {
726 target = routes[i].midiBuffer;
727 break;
728 }
729 }
730
731 if (target) {
732 uint8_t data[3] = {ev.statusByte, ev.data1, ev.data2};
733 target->addEvent(static_cast<uint32_t>(offset), data, 3, ev.noteId);
734 m_processedCounter.fetch_add(1, std::memory_order_relaxed);
735 }
736 }
737
738 m_rtQueue.pop();
739 }
740}

752void PatternPlaybackEngine::clearScheduledInstances() {
753 // Control thread only — takes the scheduler mutex. processAudio() consumes m_rtQueue
754 // and never reads m_activeInstances, so erasing here cannot race the audio thread.
755 if (reportRealtimeMisuse("PatternPlaybackEngine::clearScheduledInstances")) {
756 return;
757 }
758
759 {
760 std::lock_guard<std::mutex> lock(m_mutex);
761 m_activeInstances.clear();
762 m_instanceCount.store(0, std::memory_order_release);
763 m_lastRefillFrame = 0;
764 // Discards every queued event; adopting the new epoch also retires any pending rewind.
765 invalidateQueuedEventsLocked();
766 }
767 m_gatedNotes.clear();
768
769 for (auto& flag : m_instanceCancelled) {
770 flag.store(false, std::memory_order_release);
771 }

782uint64_t PatternPlaybackEngine::contentFingerprint(const PatternSource& pattern) {
783 // FNV-1a over the fields that change what plays. Never returns 0 (0 = "not seen yet").
784 uint64_t h = 1469598103934665603ull;
785 const auto mix = [&h](const void* data, size_t size) {
786 const auto* bytes = static_cast<const unsigned char*>(data);
787 for (size_t i = 0; i < size; ++i) {
788 h ^= bytes[i];
789 h *= 1099511628211ull;
790 }
791 };
792 mix(&pattern.lengthBeats, sizeof(pattern.lengthBeats));
793 mix(&pattern.swing, sizeof(pattern.swing));
794 if (pattern.isMidi()) {
795 for (const auto& n : std::get<MidiPayload>(pattern.payload).notes) {
796 mix(&n.pitch, sizeof(n.pitch));
797 mix(&n.startBeat, sizeof(n.startBeat));
798 mix(&n.durationBeats, sizeof(n.durationBeats));
799 mix(&n.velocity, sizeof(n.velocity));
800 mix(&n.pan, sizeof(n.pan));
801 mix(&n.unitId, sizeof(n.unitId));
802 mix(&n.pitchOffset, sizeof(n.pitchOffset));
803 mix(&n.gate, sizeof(n.gate));
804 mix(&n.slide, sizeof(n.slide));
805 mix(&n.chance, sizeof(n.chance));
806 mix(&n.ratchet, sizeof(n.ratchet));
807 }
808 }
809 return h == 0 ? 1 : h;
810}

817void PatternPlaybackEngine::rewindScheduledInstances() {
818 // Callable from any thread, the audio thread included. Bumping the epoch both
819 // discards every queued event (processAudio drops older epochs unplayed) and asks
820 // the next refillWindow() to rewind the scheduling frontiers. It used to reset the
821 // queue's write index from here, which made the audio thread a second writer of an
822 // index the scheduler owns: a reset racing a push resurrected events it meant to drop.
823 m_queueEpoch.fetch_add(1, std::memory_order_acq_rel);
824}
