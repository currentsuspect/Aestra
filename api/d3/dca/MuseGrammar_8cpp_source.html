353// Descriptions are prose and may quote things; the schema is hand-built JSON text, so escape them.
354std::string jsonEscaped(const std::string& text) {
355 std::string out;
356 out.reserve(text.size());
357 for (char c : text) {
358 if (c == '"' || c == '\\')
359 out.push_back('\\');
360 out.push_back(c == '\n' ? ' ' : c);
361 }
362 return out;
363}
