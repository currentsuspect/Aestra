69 size_t getSizeInBytes() const override {
70 // Counted honestly for CommandHistory's LRU budget: the curves carry
71 // their own point vectors and names, so sizeof() alone under-reports.
72 return sizeof(*this) + m_name.capacity() + curvesBytes(m_before) + curvesBytes(m_after);
73 }
74
75private:
76 static size_t curvesBytes(const std::vector<AutomationCurve>& curves) {
77 size_t bytes = curves.capacity() * sizeof(AutomationCurve);
78 for (const auto& curve : curves) {
79 bytes += curve.points.capacity() * sizeof(AutomationPoint);
80 bytes += curve.name.capacity();
81 }
82 return bytes;
83 }
84
85 void apply(const std::vector<AutomationCurve>& curves) {
86 auto* lane = m_manager.getPlaylistModel().getLane(m_laneId);
87 if (!lane)
88 return;
89 lane->automationCurves = curves;
90
91 // Automation reaches the engine only through the runtime snapshot, so
92 // an undo that does not request a rebuild stays inaudible until some
93 // later edit happens to trigger one.
94 m_manager.requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged);
95 m_manager.markModified();
96 }
