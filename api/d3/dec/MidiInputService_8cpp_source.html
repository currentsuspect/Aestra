18struct MidiInputService::Impl {
19 struct Port {
20 std::unique_ptr<RtMidiIn> in;
21 std::string name;
22 };
23 std::vector<Port> ports;
24 MidiInputService* owner{nullptr};
25 // Each open port is its own RtMidiIn with its own callback thread, so with
26 // multiple ports rtCallback runs concurrently. The engine's hardware queue
27 // is SPSC; this mutex serializes sink invocations so the queue only ever
28 // sees one producer at a time. MIDI callback threads only — the audio
29 // thread never touches this lock.
30 std::mutex sinkMutex;

32 static void rtCallback(double /*deltaTime*/, std::vector<unsigned char>* message, void* userData) {
33 auto* self = static_cast<MidiInputService*>(userData);
34 if (self == nullptr || message == nullptr) {
35 return;
36 }
37 const uint64_t unit = self->m_targetUnit.load(std::memory_order_relaxed);
38 if (unit == 0 || !self->m_sink) {
39 return; // no target instrument — drop silently
40 }
41 const auto translated = MidiInputService::translateMessage(message->data(), message->size());
42 if (translated.has_value()) {
43 const std::lock_guard<std::mutex> lock(self->m_impl->sinkMutex);
44 self->m_sink(unit, translated->status, translated->data1, translated->data2);
45 }
46 }

48 size_t openAllPorts(MidiInputService* service) {
49 size_t opened = 0;
50 try {
51 RtMidiIn probe;
52 const unsigned int count = probe.getPortCount();
53 for (unsigned int i = 0; i < count; ++i) {
54 std::string name;
55 try {
56 name = probe.getPortName(i);
57 } catch (const RtMidiError&) {
58 continue;
59 }
60 // Skip ports we already hold open.
61 bool alreadyOpen = false;
62 for (const auto& p : ports) {
63 if (p.name == name) {
64 alreadyOpen = true;
65 break;
66 }
67 }
68 if (alreadyOpen) {
69 ++opened;
70 continue;
71 }
72 auto in = std::make_unique<RtMidiIn>();
73 try {
74 in->openPort(i, "Aestra In");
75 in->ignoreTypes(true, true, true); // sysex, timecode, active sensing
76 in->setCallback(&rtCallback, service);
77 } catch (const RtMidiError& e) {
78 Log::warning("[MidiInput] Failed to open port '" + name + "': " + e.getMessage());
79 continue;
80 }
81 Log::info("[MidiInput] Opened MIDI input: " + name);
82 ports.push_back(Port{std::move(in), name});
83 ++opened;
84 }
85 } catch (const RtMidiError& e) {
86 Log::warning(std::string("[MidiInput] MIDI backend unavailable: ") + e.getMessage());
87 }
88 return opened;
89 }
90
91 void closeAll() {
92 for (auto& p : ports) {
93 if (p.in) {
94 p.in->cancelCallback();
95 p.in->closePort();
96 }
97 }
98 ports.clear();
99 }
100};
