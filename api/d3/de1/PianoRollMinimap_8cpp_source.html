34void PianoRollMinimap::setView(double start, double duration, bool preserveEdge) {
35 if (isDragging_) return;
36 if (preserveEdge) {
37 const double previousStart = startBeat_;
38 const double previousEnd = startBeat_ + viewDuration_;
39 viewDuration_ = std::clamp(duration, 0.25, totalDuration_);
40 if (std::abs(previousStart) <= 1e-9) {
41 startBeat_ = 0.0;
42 } else if (std::abs(previousEnd - totalDuration_) <= 1e-6) {
43 startBeat_ = std::max(0.0, totalDuration_ - viewDuration_);
44 } else {
45 startBeat_ = std::clamp(start, 0.0, std::max(0.0, totalDuration_ - viewDuration_));
46 }
47 } else {
48 viewDuration_ = std::clamp(duration, 0.25, totalDuration_);
49 startBeat_ = std::clamp(start, 0.0, std::max(0.0, totalDuration_ - viewDuration_));
50 }
51 repaint();
52}

54void PianoRollMinimap::setTotalDuration(double total) {
55 const double previousTotal = totalDuration_;
56 const double previousStart = startBeat_;
57 const double previousEnd = startBeat_ + viewDuration_;
58 totalDuration_ = std::max(1.0, total);
59 viewDuration_ = std::clamp(viewDuration_, 0.25, totalDuration_);
60 if (std::abs(previousStart) <= 1e-9) {
61 startBeat_ = 0.0;
62 } else if (std::abs(previousEnd - previousTotal) <= 1e-6) {
63 startBeat_ = std::max(0.0, totalDuration_ - viewDuration_);
64 } else {
65 startBeat_ = std::clamp(startBeat_, 0.0, std::max(0.0, totalDuration_ - viewDuration_));
66 }
67 repaint();
68}

80void PianoRollMinimap::onRender(NUIRenderer& renderer) {
81 if (!isVisible()) return;
82 auto b = getBounds();
83 auto& theme = NUIThemeManager::getInstance();
84
85 const auto panelBg = theme.getColor("backgroundPrimary").darkened(0.03f);
86 const auto border = theme.getColor("border").withAlpha(0.48f);
87 renderer.fillRoundedRect(NUIRect(b.x + 4.0f, b.y + 3.0f, b.width - 8.0f, b.height - 6.0f), 5.0f, panelBg);
88 renderer.strokeRoundedRect(NUIRect(b.x + 4.0f, b.y + 3.0f, b.width - 8.0f, b.height - 6.0f), 5.0f, 1.0f, border);
89
90 // View Rect
91 float x1 = b.x + beatToX(startBeat_);
92 float w = beatToX(viewDuration_);
93 NUIRect viewRect(x1, b.y + 2, w, b.height - 4);
94
95 auto thumbCol = theme.getColor("accentPrimary").withAlpha(0.13f);
96 auto borderCol = theme.getColor("accentPrimary").withAlpha(0.72f);
97
98 renderer.fillRoundedRect(viewRect, 5.0f, thumbCol);
99 renderer.strokeRoundedRect(viewRect, 5.0f, 1.0f, borderCol);
100
101 renderer.setClipRect(NUIRect(b.x + 2.0f, b.y + 2.0f, b.width - 4.0f, b.height - 4.0f));
102 int minPitch = 127;
103 int maxPitch = 0;
104 for (const auto& note : notes_) {
105 if (note.isDeleted || note.durationBeats <= 0.0) continue;
106 minPitch = std::min(minPitch, std::clamp(note.pitch, 0, 127));
107 maxPitch = std::max(maxPitch, std::clamp(note.pitch, 0, 127));
108 }
109 if (minPitch > maxPitch) {
110 minPitch = 48;
111 maxPitch = 72;
112 } else {
113 minPitch = std::max(0, minPitch - 2);
114 maxPitch = std::min(127, maxPitch + 2);
115 }
116
117 const float noteAreaY = b.y + 4.0f;
118 const float noteAreaH = std::max(8.0f, b.height - 8.0f);
119 const int pitchSpan = std::max(1, maxPitch - minPitch + 1);
120 const float laneHeight = noteAreaH / static_cast<float>(pitchSpan);
121 const float visibleNoteH = std::max(3.0f, std::min(5.0f, laneHeight * 0.95f));
122 const auto noteFill = theme.getColor("accentPrimary").lightened(0.08f).withAlpha(0.94f);
123 const auto noteEdge = theme.getColor("accentSecondary").withAlpha(0.55f);
124
125 for (const auto& note : notes_) {
126 if (note.isDeleted || note.durationBeats <= 0.0) continue;
127 const int clampedPitch = std::clamp(note.pitch, minPitch, maxPitch);
128 const int relativePitch = maxPitch - clampedPitch;
129 const float x = std::round(b.x + beatToX(note.startBeat));
130 const float wNote = std::max(4.0f, std::round(beatToX(std::max(0.05, note.durationBeats))));
131 const float laneY = noteAreaY + static_cast<float>(relativePitch) * laneHeight;
132 const float y = std::round(laneY + std::max(0.0f, (laneHeight - visibleNoteH) * 0.5f));
133 const float maxW = std::max(0.0f, (b.x + b.width - 4.0f) - x);
134 if (maxW <= 0.0f) continue;
135 NUIRect noteRect(x, y, std::min(wNote, maxW), visibleNoteH);
136 renderer.fillRoundedRect(noteRect, std::min(2.0f, visibleNoteH * 0.45f), noteFill);
137 renderer.strokeRoundedRect(noteRect, std::min(2.0f, visibleNoteH * 0.45f), 1.0f, noteEdge);
138 }
139 renderer.clearClipRect();
140
141 const auto playheadColor = NUIThemeManager::getInstance().getColor("accentPrimary");
142 const float playheadX = b.x + beatToX(playheadBeat_);
143 if (playheadX >= b.x && playheadX <= b.x + b.width) {
144 renderer.drawLine(NUIPoint(playheadX, b.y + 1.0f),
145 NUIPoint(playheadX, b.y + b.height - 1.0f),
146 2.0f,
147 playheadColor.withAlpha(0.9f));
148 }
149
150 // Handles (Visual only, logic in mouse)
151 const float handleW = 4.0f;
152 const float handleH = std::max(8.0f, b.height - 8.0f);
153 const auto handleFill = borderCol.withAlpha(0.5f);
154 NUIRect leftHandle(x1 + 2.0f, b.y + 4.0f, handleW, handleH);
155 NUIRect rightHandle(x1 + w - 6.0f, b.y + 4.0f, handleW, handleH);
156 renderer.fillRoundedRect(leftHandle, 2.0f, handleFill);
157 renderer.fillRoundedRect(rightHandle, 2.0f, handleFill);
158
159 // Bottom border to anchor the overview visually to the ruler beneath it.
160 // Color-matched to the ruler tick marks (use the theme border with stronger alpha).
161 renderer.drawLine(NUIPoint(b.x + 2.0f, b.bottom() - 1.0f),
162 NUIPoint(b.right() - 2.0f, b.bottom() - 1.0f),
163 1.0f,
164 theme.getColor("border").withAlpha(0.82f));
165}

167bool PianoRollMinimap::onMouseEvent(const NUIMouseEvent& event) {
168 if (!getBounds().contains(event.position) && !isDragging_) return false;
169
170 auto b = getBounds();
171 float localX = event.position.x - b.x;
172 float localY = event.position.y - b.y;
173
174 float x1 = beatToX(startBeat_);
175 float w = beatToX(viewDuration_);
176 float x2 = x1 + w;
177 const float handleW = 8.0f;
178 const float handleH = std::max(8.0f, b.height - 8.0f);
179 const NUIRect leftHandleRect(x1 + 2.0f, 4.0f, handleW, handleH);
180 const NUIRect rightHandleRect(x1 + w - handleW - 2.0f, 4.0f, handleW, handleH);
181 const NUIRect viewportRect(x1, 2.0f, w, b.height - 4.0f);
182 const NUIPoint localPos(localX, localY);
183
184 if (event.pressed && event.button == NUIMouseButton::Left) {
185 // Hit Test
186 if (leftHandleRect.contains(localPos)) {
187 isDragging_ = true;
188 isResizingL_ = true;
189 dragStartPos_ = event.position;
190 dragStartStart_ = startBeat_;
191 dragStartDuration_ = viewDuration_;
192 } else if (rightHandleRect.contains(localPos)) {
193 isDragging_ = true;
194 isResizingR_ = true;
195 dragStartPos_ = event.position;
196 dragStartStart_ = startBeat_;
197 dragStartDuration_ = viewDuration_;
198 } else if (viewportRect.contains(localPos)) {
199 isDragging_ = true;
200 dragStartPos_ = event.position;
201 dragStartStart_ = startBeat_;
202 dragStartDuration_ = viewDuration_;
203 } else {
204 double beat = xToBeat(localX);
205 startBeat_ = std::clamp(beat - viewDuration_ * 0.5, 0.0, std::max(0.0, totalDuration_ - viewDuration_));
206 isDragging_ = false;
207 isResizingL_ = false;
208 isResizingR_ = false;
209 if (onViewChanged) onViewChanged(startBeat_, viewDuration_);
210 repaint();
211 return true;
212 }
213 return true;
214 }
215 else if (event.released && event.button == NUIMouseButton::Left) {
216 isDragging_ = false;
217 isResizingL_ = false;
218 isResizingR_ = false;
219 return true;
220 }
221 else if (!event.pressed && isDragging_) {
222 float dx = event.position.x - dragStartPos_.x;
223 double db = dx / b.width * totalDuration_;
224
225 if (isResizingL_) {
226 double newStart = dragStartStart_ + db;
227 double newDur = dragStartDuration_ - db;
228
229 if (newDur < 0.25) { newStart -= (0.25 - newDur); newDur = 0.25; }
230
231 viewDuration_ = std::clamp(newDur, 0.25, totalDuration_);
232 startBeat_ = std::clamp(newStart, 0.0, std::max(0.0, totalDuration_ - viewDuration_));
233 }
234 else if (isResizingR_) {
235 double newDur = dragStartDuration_ + db;
236 viewDuration_ = std::clamp(newDur, 0.25, totalDuration_);
237 startBeat_ = std::clamp(startBeat_, 0.0, std::max(0.0, totalDuration_ - viewDuration_));
238 }
239 else {
240 startBeat_ = std::clamp(dragStartStart_ + db, 0.0, std::max(0.0, totalDuration_ - viewDuration_));
241 }
242
243 if (onViewChanged) onViewChanged(startBeat_, viewDuration_);
244 repaint();
245 return true;
246 }
247
248 return NUIComponent::onMouseEvent(event);
249}
