108 // RT-safe: requests an immediate voice reset on the next process() call.
109 void requestHardResetVoices() noexcept { m_resetVoicesRequested.store(true, std::memory_order_release); }
110
111 // RT-safe: asks every sounding voice to enter its ADSR release on the next process() call,
112 // BEFORE that block's MIDI is handled (so a note that starts in the same block still sounds).
113 // This is what a pause, a stop and a timeline loop wrap need: the queued note-offs for notes
114 // still held at that moment are discarded with the transport jump, and without this the voices
115 // sustain until the sample runs out. A tight release stays tight.
116 void requestReleaseAllVoices() noexcept { m_releaseVoicesRequested.store(true, std::memory_order_release); }

163 // Voice Architecture

166 struct Voice {
167 bool active = false;
168 int note = 0;
169 float velocity = 0.0f;
170 float gainL = 1.0f; // Per-note pan, latched at note-on (unity centre)
171 float gainR = 1.0f;
172 double position = 0.0; // Sample index
173 double playbackRate = 1.0; // sample index increment/frame
174 double targetPlaybackRate = 1.0;
175 int playbackDirection = 1;
176 bool glideActive = false;
177 EnvStage stage = EnvStage::Off;
178 double stageTime = 0.0; // Seconds in current stage
179 float currentGain = 0.0f;
180 float releaseGain = 0.0f; // Gain at start of release
181 uint64_t order = 0; // Trigger order: an id-less note-off ends the oldest held voice of its pitch
182 uint32_t noteId = 0; // The scheduler's note identity (0 = none, e.g. live MIDI)
183 // Keep-length voices play their own stretched copy; null plays the shared sample.
184 std::shared_ptr<const KeepLengthCache::Buffer> ownData;
185 // A sample swapped out while this voice sounded: the voice finishes on the sample it started on. Set by
186 // the audio thread (a refcount copy); the plugin's retired list keeps the buffer alive past the voice, so
187 // dropping this reference is never the last one on the audio thread.
188 std::shared_ptr<SampleData> held;
189 };

191 static constexpr int kMaxVoices = 32;
192 std::array<Voice, kMaxVoices> m_voices;

194 // Per-note pan latch: the pattern scheduler sends pan as polyphonic
195 // aftertouch (0xA0, note, 0..127) just before the matching note-on, which
196 // consumes and re-centres its slot. Audio-thread only — no atomics needed.
197 std::array<uint8_t, 128> m_pendingNotePan{};
198
199 // Note ownership. A scheduled note-off carries its note's id and ends exactly the
200 // voice that note started; if that voice is already gone (choked, stolen, taken over
201 // in mono, or its sample ran out) the note-off does nothing. Live MIDI has no id:
202 // it ends the oldest held voice of its pitch (exact, since a key can't be pressed
203 // twice without being released). Audio-thread only.
204 uint64_t m_nextVoiceOrder = 1;
