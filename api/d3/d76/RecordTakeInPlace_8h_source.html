4// Record where the user armed: a finished take goes onto the track's own lane, as a clip you can see land under the
5// playhead. A take lane (a new row under the track) used to be made every time, so the recording left the lane it was
6// drawn on and the view followed it. A take lane is now the fallback, for when the lane already has something where the
7// take goes: nothing is overwritten or hidden underneath.

23inline bool takeFitsOnLane(const PlaylistLane& lane, const ClipInstance& clip) {
24 const double end = clip.startBeat + clip.durationBeats;
25 return std::none_of(lane.clips.begin(), lane.clips.end(), [&](const ClipInstance& other) {
26 return other.startBeat < end - 1e-6 && other.startBeat + other.durationBeats > clip.startBeat + 1e-6;
27 });
28}

32inline bool recordTakeInPlace(PlaylistModel& playlist, CommandHistory& history, const Track& track, const ClipInstance& clip) {
33 const PlaylistLane* lane = track.activeLaneId.isValid() ? playlist.getLane(track.activeLaneId) : nullptr;
34 if (!lane || lane->trackId != track.trackId || !takeFitsOnLane(*lane, clip)) return false;
35 auto add = std::make_shared<AddClipCommand>(playlist, track.activeLaneId, clip);
36 add->execute();
37 if (!playlist.getClip(clip.id)) return false;
38 auto transaction = std::make_shared<CommandTransaction>("Record Take");
39 transaction->add(add);
40 transaction->markExecuted();
41 if (history.pushExecuted(transaction)) return true;
42 add->undo(); // history refused it: take it back out
43 return false;
44}
