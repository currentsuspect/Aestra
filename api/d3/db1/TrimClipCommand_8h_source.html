55 void execute() override {
56 if (m_executed)
57 return;
58
59 auto* clip = m_model.getClip(m_clipId);
60 if (!clip)
61 return;
62
63 if (!m_originalCaptured) {
64 m_originalStartBeat = clip->startBeat;
65 m_originalDuration = clip->durationBeats;
66 m_originalDurationSeconds = clip->durationSeconds;
67 m_originalSourceOffsetSeconds = clip->sourceOffsetSeconds;
68 m_originalCaptured = true;
69 }
70 if (!m_wasAudioDetermined) {
71 // Audio-ness cannot change mid-gesture; capture once, reuse for undo.
72 m_originalWasAudio = m_model.isAudioClip(*clip);
73 m_wasAudioDetermined = true;
74 }
75
76 double newStart = (m_newStartBeat < 0) ? clip->startBeat : m_newStartBeat;
77 double newEnd = (m_newEndBeat < 0) ? (clip->startBeat + clip->durationBeats) : m_newEndBeat;
78
79 // Validate: end must be after start
80 if (newEnd <= newStart) {
81 return; // Invalid trim range - don't apply
82 }
83
84 if (m_originalWasAudio) {
85 // The source read offset must never go negative: clamp the start to
86 // the source-start beat (start - offset), same formula the UI drag
87 // uses, so the model can never hold an unrenderable offset. The
88 // combined Speed/Pitch varispeed is the render authority (#746).
89 const double secondsPerBeat = m_model.beatToSeconds(1.0);
90 const double varispeed = static_cast<double>(clip->edits.effectiveVarispeed());
91 const double minStart =
92 m_originalStartBeat - m_originalSourceOffsetSeconds / (secondsPerBeat * varispeed);
93 if (newStart < minStart) {
94 newStart = minStart;
95 }
96 if (newEnd <= newStart) {
97 return; // Clamped start swallowed the trim range - don't apply
98 }
99
100 clip->durationSeconds = m_model.beatToSeconds(newEnd - newStart) / varispeed;
101 if (newStart != m_originalStartBeat) {
102 // Left-edge trim: advance the source read position by the same
103 // musical amount (scaled by varispeed so the right edge stays
104 // pinned under Speed/Pitch). Right-edge trims leave the
105 // source offset untouched.
106 clip->sourceOffsetSeconds =
107 m_originalSourceOffsetSeconds + (newStart - m_originalStartBeat) * secondsPerBeat * varispeed;
108 }
109 }
110
111 clip->startBeat = newStart;
112 clip->durationBeats = newEnd - newStart;
113
114 m_executed = true;
115 }
116
117 void undo() override {
118 if (!m_executed)
119 return;
120
121 auto* clip = m_model.getClip(m_clipId);
122 if (!clip)
123 return;
124
125 clip->startBeat = m_originalStartBeat;
126 clip->durationBeats = m_originalDuration;
127 if (m_originalWasAudio) {
128 clip->durationSeconds = m_originalDurationSeconds;
129 clip->sourceOffsetSeconds = m_originalSourceOffsetSeconds;
130 }
131
132 m_executed = false;
133 }
134
135 void redo() override { execute(); }
136
137 std::string getName() const override { return "Trim Clip"; }
138 size_t getSizeInBytes() const override { return sizeof(*this); }
139 bool changesProjectState() const override { return true; }
