82static uint64_t fnv1a64_append_u32(uint64_t h, uint32_t v) {
83 constexpr uint64_t prime = 1099511628211ull;
84 for (int i = 0; i < 4; ++i) {
85 const uint8_t b = static_cast<uint8_t>((v >> (i * 8)) & 0xFFu);
86 h ^= static_cast<uint64_t>(b);
87 h *= prime;
88 }
89 return h;
90}

111static bool jsonGetU32(const JSON& obj, const char* key, uint32_t& out) {
112 if (!obj.isObject() || !obj.has(key)) return false;
113 const auto& v = obj[key];
114 if (!v.isNumber()) return false;
115 const double n = v.asNumber();
116 if (n < 0.0) return false;
117 out = static_cast<uint32_t>(n);
118 return true;

146static std::vector<Scenario> loadScenarioFile(const std::string& path) {
147 std::vector<Scenario> scenarios;
148 std::string text;
149 if (!readTextFile(path, text)) return scenarios;
150
151 JSON root = JSON::parse(text);
152 if (!root.isObject() || !root.has("scenarios") || !root["scenarios"].isArray()) return scenarios;
153
154 const auto& arr = root["scenarios"].asArray();
155 scenarios.reserve(arr.size());
156
157 for (const auto& item : arr) {
158 if (!item.isObject()) continue;
159
160 Scenario sc;
161 jsonGetString(item, "name", sc.name);
162
163 std::string typeStr;
164 if (jsonGetString(item, "type", typeStr)) {
165 bool ok = false;
166 sc.type = parseScenarioType(typeStr, ok);
167 if (!ok) continue;
168 }
169
170 jsonGetU32(item, "sr", sc.sampleRate);
171 jsonGetU32(item, "frames", sc.bufferFrames);
172 jsonGetU32(item, "seconds", sc.seconds);
173
174 double minPeak = static_cast<double>(sc.minPeak);
175 if (jsonGetDouble(item, "minPeak", minPeak)) sc.minPeak = static_cast<float>(minPeak);
176
177 // Project
178 jsonGetString(item, "project", sc.projectPath);
179
180 // Tone
181 jsonGetDouble(item, "frequencyHz", sc.frequencyHz);
182 jsonGetDouble(item, "amplitude", sc.amplitude);
183
184 // Stress
185 jsonGetU32(item, "tracks", sc.tracks);
186 jsonGetU32(item, "timelineSeconds", sc.timelineSeconds);
187 jsonGetU32(item, "commandHz", sc.commandHz);
188 jsonGetU32(item, "graphSwapHz", sc.graphSwapHz);
189
190 if (sc.name.empty()) continue;
191 scenarios.push_back(std::move(sc));
192 }
193
194 return scenarios;
195}

197static std::shared_ptr<AudioBuffer> makeSineBuffer(uint32_t sampleRate, uint32_t seconds, double frequencyHz, double amplitude) {
198 auto buffer = std::make_shared<AudioBuffer>();
199 buffer->channels = 2;
200 buffer->sampleRate = sampleRate;
201 buffer->numFrames = static_cast<uint64_t>(sampleRate) * seconds;
202 buffer->data.resize(static_cast<size_t>(buffer->numFrames) * buffer->channels);
203
204 const double twoPi = 6.28318530717958647693;
205 const double invSR = 1.0 / static_cast<double>(sampleRate);
206
207 for (uint64_t i = 0; i < buffer->numFrames; ++i) {
208 const double t = static_cast<double>(i) * invSR;
209 const float s = static_cast<float>(amplitude * std::sin(twoPi * frequencyHz * t));
210 buffer->data[static_cast<size_t>(i) * 2] = s;
211 buffer->data[static_cast<size_t>(i) * 2 + 1] = s;
212 }
213
214 return buffer;
215}

217static AudioGraph buildLoopGraph(const std::shared_ptr<AudioBuffer>& source,
218 uint32_t engineSampleRate,
219 uint32_t tracks,
220 uint32_t timelineSeconds) {
221 AudioGraph graph;
222 graph.tracks.reserve(tracks);
223
224 const uint64_t timelineEnd = static_cast<uint64_t>(engineSampleRate) * timelineSeconds;
225 graph.timelineEndSample = timelineEnd;
226
227 for (uint32_t i = 0; i < tracks; ++i) {
228 TrackRenderState tr;
229 tr.trackId = i + 1;
230 tr.trackIndex = i;
231 tr.volume = 1.0f;
232 tr.pan = 0.0f;
233 tr.mute = false;
234 tr.solo = false;
235
236 ClipRenderState clip;
237 clip.buffer = source;
238 clip.audioData = source->data.data();
239 clip.startSample = 0;
240 clip.endSample = timelineEnd;
241 clip.sampleOffset = 0;
242 clip.totalFrames = source->numFrames;
243 clip.sourceSampleRate = static_cast<double>(source->sampleRate);
244 clip.channels = static_cast<uint32_t>(source->channels);
245 clip.gain = 1.0f;
246 clip.pan = 0.0f;
247
248 tr.clips.push_back(std::move(clip));
249 graph.tracks.push_back(std::move(tr));
250 }
251
252 return graph;
253}

255static RunMetrics renderEngine(AudioEngine& engine,
256 uint32_t sampleRate,
257 uint32_t bufferFrames,
258 uint32_t seconds,
259 float minPeak,
260 std::function<void(uint64_t /*blockIndex*/)> onBlock = {}) {
261 RunMetrics m;
262 m.audioSeconds = static_cast<double>(seconds);
263
264 const uint64_t totalFrames = static_cast<uint64_t>(sampleRate) * seconds;
265 uint64_t framesProcessed = 0;
266 uint64_t blockIndex = 0;
267
268 std::vector<float> out(static_cast<size_t>(bufferFrames) * kChannels);
269
270 float maxPeak = 0.0f;
271 long double sumSquares = 0.0L;
272 uint64_t sampleCount = 0;
273 uint64_t nanCount = 0;
274
275 uint64_t hash = fnv1a64_init();
276
277 const auto start = std::chrono::steady_clock::now();
278 double streamTime = 0.0;
279
280 while (framesProcessed < totalFrames) {
281 if (onBlock) onBlock(blockIndex);
282
283 const uint32_t framesThisBlock = static_cast<uint32_t>(
284 std::min<uint64_t>(bufferFrames, totalFrames - framesProcessed));
285
286 engine.processBlock(out.data(), nullptr, framesThisBlock, streamTime);
287 engine.performNonRealtimeMaintenance();
288
289 for (uint32_t i = 0; i < framesThisBlock * kChannels; ++i) {
290 const float s = out[i];
291 if (isBadSample(s)) {
292 nanCount++;
293 continue;
294 }
295
296 maxPeak = std::max(maxPeak, std::fabs(s));
297 const long double sd = static_cast<long double>(s);
298 sumSquares += sd * sd;
299 sampleCount++;
300
301 uint32_t bits = 0;
302 static_assert(sizeof(float) == sizeof(uint32_t));
303 std::memcpy(&bits, &s, sizeof(bits));
304 hash = fnv1a64_append_u32(hash, bits);
305 }
306
307 framesProcessed += framesThisBlock;
308 streamTime += static_cast<double>(framesThisBlock) / static_cast<double>(sampleRate);
309 blockIndex++;
310 }
311
312 const auto end = std::chrono::steady_clock::now();
313 const std::chrono::duration<double, std::milli> durMs = end - start;
314 m.renderMs = durMs.count();
315 m.cpuRatio = (m.audioSeconds > 0.0) ? (m.renderMs / (m.audioSeconds * 1000.0)) : 0.0;
316 m.maxPeak = maxPeak;
317 m.nanOrInfCount = nanCount;
318 m.hash64 = hash;
319 m.rms = (sampleCount > 0) ? static_cast<float>(std::sqrt(static_cast<double>(sumSquares / static_cast<long double>(sampleCount)))) : 0.0f;
320
321 if (nanCount > 0) {
322 m.ok = false;
323 m.failure = "nan_or_inf";
324 return m;
325 }
326 if (isBadSample(maxPeak)) {
327 m.ok = false;
328 m.failure = "invalid_peak";
329 return m;
330 }
331 if (maxPeak < minPeak) {
332 m.ok = false;
333 m.failure = "near_silence";
334 return m;
335 }
336
337 m.ok = true;
338 return m;
339}

341static RunMetrics runScenarioProject(const Scenario& sc) {
342 const std::string projectPath = resolvePathRelativeToCwd(sc.projectPath);
343 if (projectPath.empty()) {
344 RunMetrics m;
345 m.ok = false;
346 m.failure = "missing_project";
347 return m;
348 }
349
350 auto trackManager = std::make_shared<TrackManager>();
351 trackManager->setOutputSampleRate(static_cast<double>(sc.sampleRate));
352 trackManager->setInputSampleRate(static_cast<double>(sc.sampleRate));
353 trackManager->setInputChannelCount(0);
354
355 ProjectSerializer::LoadResult loadRes = ProjectSerializer::load(projectPath, trackManager);
356 if (!loadRes.ok) {
357 RunMetrics m;
358 m.ok = false;
359 m.failure = "project_load_failed";
360 return m;
361 }
362
363 AudioEngine engine;
364 engine.setSampleRate(sc.sampleRate);
365 engine.setBufferConfig(sc.bufferFrames, kChannels);
366
367 // Mirror app wiring for metering + routing.
368 auto meterBuffer = std::make_shared<MeterSnapshotBuffer>();
369 engine.setMeterSnapshots(meterBuffer);
370 trackManager->setMeterSnapshots(meterBuffer);
371 if (auto slotMap = trackManager->getChannelSlotMapShared()) {
372 engine.setChannelSlotMap(slotMap);
373 }
374
375 engine.setBPM(static_cast<float>(loadRes.tempo));
376 trackManager->setPosition(loadRes.playhead);
377 engine.setGlobalSamplePos(static_cast<uint64_t>(loadRes.playhead * static_cast<double>(sc.sampleRate)));
378
379 auto graph = AudioGraphBuilder::buildFromTrackManager(*trackManager);
380 engine.setGraph(graph);
381
382 engine.setTransportPlaying(true);
383 {
384 AudioQueueCommand cmd{};
385 cmd.type = AudioQueueCommandType::SetTransportState;
386 cmd.value1 = 1.0f;
387 cmd.samplePos = engine.getGlobalSamplePos();
388 engine.commandQueue().push(cmd);
389 }
390
391 return renderEngine(engine, sc.sampleRate, sc.bufferFrames, sc.seconds, sc.minPeak);
392}

395 AudioEngine engine;

407 cmd.samplePos = 0;
408 engine.commandQueue().push(cmd);
409 }
410
411 return renderEngine(engine, sc.sampleRate, sc.bufferFrames, sc.seconds, sc.minPeak);
412}

414static RunMetrics runScenarioStress(const Scenario& sc) {
415 AudioEngine engine;
416 engine.setSampleRate(sc.sampleRate);
417 engine.setBufferConfig(sc.bufferFrames, kChannels);
418
419 // Force some SRC activity by using mismatched source SR.
420 auto source = makeSineBuffer(44100, sc.timelineSeconds, 997.0, 0.2);
421 auto graph = buildLoopGraph(source, sc.sampleRate, std::max<uint32_t>(1, sc.tracks), sc.timelineSeconds);
422 engine.setGraph(graph);
423
424 engine.setTransportPlaying(true);
425 {
426 AudioQueueCommand cmd{};
427 cmd.type = AudioQueueCommandType::SetTransportState;
428 cmd.value1 = 1.0f;
429 cmd.samplePos = 0;
430 engine.commandQueue().push(cmd);
431 }
432
433 // Deterministic interleaving: push commands / swap graph based on block index.
434 const uint64_t blocksPerSecond = (sc.sampleRate > 0 && sc.bufferFrames > 0)
435 ? static_cast<uint64_t>(std::max<double>(1.0, static_cast<double>(sc.sampleRate) / static_cast<double>(sc.bufferFrames)))
436 : 1;
437
438 const uint64_t cmdEvery = (sc.commandHz > 0)
439 ? std::max<uint64_t>(1, blocksPerSecond / sc.commandHz)
440 : std::numeric_limits<uint64_t>::max();
441
442 const uint64_t graphEvery = (sc.graphSwapHz > 0)
443 ? std::max<uint64_t>(1, blocksPerSecond / sc.graphSwapHz)
444 : std::numeric_limits<uint64_t>::max();
445
446 std::mt19937 rng(0xC0FFEEu);
447 std::uniform_int_distribution<uint32_t> trackDist(0, std::max<uint32_t>(1, sc.tracks) - 1);
448 std::uniform_real_distribution<float> volDist(0.0f, 1.2f);
449 std::uniform_real_distribution<float> panDist(-1.0f, 1.0f);
450 std::uniform_int_distribution<int> boolDist(0, 1);
451
452 auto onBlock = [&](uint64_t blockIndex) {
453 if (blockIndex % cmdEvery == 0) {
454 AudioQueueCommand cmd{};
455 const uint32_t trackIndex = trackDist(rng);
456 cmd.trackIndex = trackIndex;
457 const int which = static_cast<int>(rng() % 4u);
458 switch (which) {
459 case 0:
460 cmd.type = AudioQueueCommandType::SetTrackVolume;
461 cmd.value1 = volDist(rng);
462 break;
463 case 1:
464 cmd.type = AudioQueueCommandType::SetTrackPan;
465 cmd.value1 = panDist(rng);
466 break;
467 case 2:
468 cmd.type = AudioQueueCommandType::SetTrackMute;
469 cmd.value1 = boolDist(rng) ? 1.0f : 0.0f;
470 break;
471 default:
472 cmd.type = AudioQueueCommandType::SetTrackSolo;
473 cmd.value1 = boolDist(rng) ? 1.0f : 0.0f;
474 break;
475 }
476 engine.commandQueue().push(cmd);
477 }
478
479 if (blockIndex % graphEvery == 0) {
480 // Mutate a few clip gains and swap.
481 for (uint32_t i = 0; i < std::min<uint32_t>(sc.tracks, 8); ++i) {
482 const uint32_t trackIndex = trackDist(rng);
483 if (trackIndex >= graph.tracks.size()) continue;
484 if (graph.tracks[trackIndex].clips.empty()) continue;
485 graph.tracks[trackIndex].clips[0].gain = 0.6f + 0.4f * volDist(rng);
486 }
487 engine.setGraph(graph);
488 }
489 };
490
491 return renderEngine(engine, sc.sampleRate, sc.bufferFrames, sc.seconds, sc.minPeak, onBlock);
492}

494static RunMetrics runScenario(const Scenario& sc) {
495 switch (sc.type) {
496 case ScenarioType::Project: return runScenarioProject(sc);
497 case ScenarioType::Tone: return runScenarioTone(sc);
498 case ScenarioType::Stress: return runScenarioStress(sc);
499 }
500 RunMetrics m;
501 m.ok = false;
502 m.failure = "unknown_type";
503 return m;
504}

506static JSON makeReportJson(const Scenario& sc, const RunMetrics& m) {
507 auto toHex64 = [](uint64_t v) {
508 std::ostringstream oss;
509 oss << "0x" << std::hex << v;
510 return oss.str();
511 };
512
513 JSON r = JSON::object();
514 r.set("name", JSON(sc.name));
515 r.set("type", JSON(sc.type == ScenarioType::Project ? "project" : sc.type == ScenarioType::Tone ? "tone" : "stress"));
516 r.set("ok", JSON(m.ok));
517 if (!m.failure.empty()) r.set("failure", JSON(m.failure));
518
519 JSON cfg = JSON::object();
520 cfg.set("sr", JSON(static_cast<double>(sc.sampleRate)));
521 cfg.set("frames", JSON(static_cast<double>(sc.bufferFrames)));
522 cfg.set("seconds", JSON(static_cast<double>(sc.seconds)));
523 cfg.set("minPeak", JSON(static_cast<double>(sc.minPeak)));
524 if (sc.type == ScenarioType::Project) cfg.set("project", JSON(sc.projectPath));
525 if (sc.type == ScenarioType::Tone) {
526 cfg.set("frequencyHz", JSON(sc.frequencyHz));
527 cfg.set("amplitude", JSON(sc.amplitude));
528 }
529 if (sc.type == ScenarioType::Stress) {
530 cfg.set("tracks", JSON(static_cast<double>(sc.tracks)));
531 cfg.set("timelineSeconds", JSON(static_cast<double>(sc.timelineSeconds)));
532 cfg.set("commandHz", JSON(static_cast<double>(sc.commandHz)));
533 cfg.set("graphSwapHz", JSON(static_cast<double>(sc.graphSwapHz)));
534 }
535 r.set("config", cfg);
536
537 JSON metrics = JSON::object();
538 metrics.set("maxPeak", JSON(static_cast<double>(m.maxPeak)));
539 metrics.set("rms", JSON(static_cast<double>(m.rms)));
540 metrics.set("nanOrInfCount", JSON(static_cast<double>(m.nanOrInfCount)));
541 metrics.set("hash64", JSON(toHex64(m.hash64)));
542 metrics.set("renderMs", JSON(m.renderMs));
543 metrics.set("audioSeconds", JSON(m.audioSeconds));
544 metrics.set("cpuRatio", JSON(m.cpuRatio));
545 r.set("metrics", metrics);
546
547 return r;
548}

585static Cli parseCli(int argc, char** argv) {
586 Cli cli;
587
588 for (int i = 1; i < argc; ++i) {
589 const char* a = argv[i];
590 auto nextStr = [&](std::string& dst) {
591 if (i + 1 >= argc) return;
592 dst = argv[++i];
593 };
594 auto nextU32 = [&](uint32_t& dst) {
595 if (i + 1 >= argc) return;
596 dst = static_cast<uint32_t>(std::strtoul(argv[++i], nullptr, 10));
597 };
598
599 if (std::strcmp(a, "--scenario-file") == 0) nextStr(cli.scenarioFile);
600 else if (std::strcmp(a, "--scenario") == 0) nextStr(cli.scenarioName);
601 else if (std::strcmp(a, "--report") == 0) nextStr(cli.reportPath);
602 else if (std::strcmp(a, "--list-scenarios") == 0) cli.listScenarios = true;
603 else if (std::strcmp(a, "--human") == 0) cli.human = true;
604
605 else if (std::strcmp(a, "--project") == 0) nextStr(cli.projectPath);
606 else if (std::strcmp(a, "--sr") == 0) nextU32(cli.sr);
607 else if (std::strcmp(a, "--frames") == 0) nextU32(cli.frames);
608 else if (std::strcmp(a, "--seconds") == 0) nextU32(cli.seconds);
609 else if (std::strcmp(a, "--min-peak") == 0) {
610 if (i + 1 < argc) cli.minPeak = static_cast<float>(std::strtod(argv[++i], nullptr));
611 }
612
613 else if (std::strcmp(a, "--help") == 0 || std::strcmp(a, "-h") == 0) {
614 printHelp();
615 std::exit(0);
616 }
617 }
618
619 if (cli.scenarioFile.empty()) {
620 namespace fs = std::filesystem;
621 const fs::path defaultPath = fs::current_path() / "Tests" / "headless_scenarios.json";
622 std::error_code ec;
623 if (fs::exists(defaultPath, ec) && !ec) {
624 cli.scenarioFile = defaultPath.string();
625 }
626 }
627
628 if (const char* env = std::getenv("AESTRA_MIN_PEAK")) {
629 // [SEC-RTM-016] Use endptr to detect malformed values — silently accepting
630 // 0.0 from a bad env var is a CI supply-chain risk.
631 char* end = nullptr;
632 double val = std::strtod(env, &end);
633 if (end == env || *end != '\0' || !std::isfinite(val)) {
634 Log::warning("[Headless] Ignoring malformed AESTRA_MIN_PEAK: '" + std::string(env) + "'");
635 } else {
636 cli.minPeak = static_cast<float>(val);
637 }
638 }
639
640 return cli;
641}
