22void PianoRollGrid::onRender(NUIRenderer& renderer) {
23 if (!isVisible()) return;
24 const auto bounds = getBounds();
25 auto& theme = NUIThemeManager::getInstance();
26
27 renderer.setClipRect(bounds);
28 renderer.fillRect(bounds, theme.getColor("timelineBed"));
29
30 const auto gridInk = theme.getCurrentTheme().textPrimary;
31 const auto accidentalRow = gridInk.withAlpha(0.032f);
32 const auto rootRow = theme.getColor("accentPrimary").withAlpha(0.055f);
33 // Out-of-scale rows recede toward the bed's own polarity, not toward black.
34 const auto outOfScaleRow = gridInk.withAlpha(0.10f);
35 const auto rowLine = gridInk.withAlpha(0.045f);
36
37 int startPitch = 127 - static_cast<int>(scrollY_ / keyHeight_);
38 int endPitch = 127 - static_cast<int>((scrollY_ + bounds.height) / keyHeight_);
39 startPitch = std::clamp(startPitch + 2, 0, 127);
40 endPitch = std::clamp(endPitch - 2, 0, 127);
41
42 for (int pitch = startPitch; pitch >= endPitch; --pitch) {
43 const float y = bounds.y + (127 - pitch) * keyHeight_ - scrollY_;
44 const NUIRect row(bounds.x, y, bounds.width, keyHeight_);
45 const int pitchClass = ((pitch % 12) + 12) % 12;
46 const bool isRoot = pitchClass == rootKey_;
47 const bool isInScale =
48 scaleType_ == ScaleType::Chromatic || MusicTheory::isNoteInScale(pitch, rootKey_, scaleType_);
49
50 if (isRoot) {
51 renderer.fillRect(row, rootRow);
52 } else if (isBlackKey(pitch)) {
53 renderer.fillRect(row, accidentalRow);
54 }
55 if (!isInScale) renderer.fillRect(row, outOfScaleRow);
56 renderer.drawLine(NUIPoint(bounds.x, y), NUIPoint(bounds.right(), y), 1.0f, rowLine);
57 }
58
59 renderTimelineGrid(
60 renderer, bounds, bounds.x, bounds.right(), scrollX_, pixelsPerBeat_, beatsPerBar_, gridInk,
61 {}, getSnapSubdivisionBeats());
62
63 if (hoveredPitch_ >= 0 && hoveredPitch_ <= 127) {
64 const float hoverY = bounds.y + (127 - hoveredPitch_) * keyHeight_ - scrollY_;
65 const NUIRect hoverRow(bounds.x, hoverY, bounds.width, keyHeight_);
66 const auto hoverColor = theme.getColor("accentPrimary").withAlpha(0.075f);
67 renderer.fillRect(hoverRow, hoverColor);
68 renderer.drawLine(NUIPoint(bounds.x, hoverRow.y),
69 NUIPoint(bounds.right(), hoverRow.y),
70 1.0f,
71 theme.getColor("accentPrimary").withAlpha(0.28f));
72 renderer.drawLine(NUIPoint(bounds.x, hoverRow.bottom()),
73 NUIPoint(bounds.right(), hoverRow.bottom()),
74 1.0f,
75 theme.getColor("accentPrimary").withAlpha(0.18f));
76 }
77
78 const float patternEndX =
79 beatToScreenX(totalDurationBeats_, pixelsPerBeat_, scrollX_, bounds.x);
80 if (patternEndX < bounds.right()) {
81 const float shadeX = std::max(bounds.x, patternEndX);
82 renderer.fillRect(NUIRect(shadeX, bounds.y, bounds.right() - shadeX, bounds.height),
83 NUIThemeManager::getInstance().getCurrentTheme().textPrimary.withAlpha(0.14f));
84 }
85 if (patternEndX >= bounds.x && patternEndX <= bounds.right()) {
86 renderer.drawLine(NUIPoint(patternEndX, bounds.y),
87 NUIPoint(patternEndX, bounds.bottom()),
88 2.0f,
89 theme.getColor("accentPrimary").withAlpha(0.58f));
90 }
91
92 renderer.clearClipRect();
93}
