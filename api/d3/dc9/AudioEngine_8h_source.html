384 // Build complete snapshot (manager + cache) before publishing
385 // so audio thread never sees mixed state
386 auto* snapshot = new UnitManagerSnapshot();
387 snapshot->manager = mgr;

393 delete old;
394 // Also update legacy fields for backward compatibility during transition
395 m_unitManager.store(mgr, std::memory_order_release);
396 auto cache = snapshot->cache;
397 m_cachedSamplers = cache.samplers;
398 m_cachedSamplerOwners = cache.owners;

400
401 // Update new thread-safe cache as well

403 if (mgr) {
404 refreshSamplerCacheToSnapshot(*mgr, *newCache);
405 }
406 auto retired = std::move(m_samplerCacheOwned);
407 m_samplerCacheOwned = std::move(newCache);
408 m_samplerCacheRaw.store(m_samplerCacheOwned.get(), std::memory_order_release);
409 GarbageCollector::instance().release(std::move(retired), "AudioEngine::SamplerCache");

559 void setTrackManager(std::shared_ptr<TrackManager> trackManager) {
560 if (auto previous = m_trackManager.lock()) {
561 previous->setChannelPrepareCallback(nullptr);
562 }
563 m_trackManager = std::move(trackManager);
564 if (auto current = m_trackManager.lock()) {
565 auto prepareConfig = m_channelPrepareConfig;
566 auto prepareChannel = [prepareConfig](MixerChannel& channel) {
567 if (!prepareConfig) {
568 return;
569 }
570 const uint32_t maxBlockSize = prepareConfig->maxBlockSize.load(std::memory_order_relaxed);
571 if (maxBlockSize == 0) {
572 return;
573 }
574 const double sampleRate =
575 static_cast<double>(prepareConfig->sampleRate.load(std::memory_order_relaxed));
576 channel.prepareProcessingBuffers(maxBlockSize);
577 channel.getEffectChain().prepare(sampleRate, maxBlockSize);
578 };
579 current->setChannelPrepareCallback(prepareChannel);
580 const size_t channelCount = current->getChannelCount();
581 for (size_t i = 0; i < channelCount; ++i) {
582 if (auto* channel = current->getChannel(i)) {
583 prepareChannel(*channel);
584 channel->setEffectChainLatencyCallback([this]() { calculateLatencyCompensation(); });
585 }
586 }
587 calculateLatencyCompensation();
588 }

1031 // Lock-free queue for passing block energies to worker
1032 // Fixed size ring buffer, power of 2 for fast wrapping
1033 static constexpr size_t kLoudnessQueueSize = 1024;
1034 struct BlockEnergyQueue {
1035 std::atomic<size_t> writeIdx{0};
1036 std::atomic<size_t> readIdx{0};
1037 std::array<double, kLoudnessQueueSize> data;
1038
1039 bool try_push(double val) {
1040 size_t currentWrite = writeIdx.load(std::memory_order_relaxed);
1041 size_t nextWrite = (currentWrite + 1) & (kLoudnessQueueSize - 1);
1042 if (nextWrite == readIdx.load(std::memory_order_acquire)) {
1043 return false; // Full
1044 }
1045 data[currentWrite] = val;
1046 writeIdx.store(nextWrite, std::memory_order_release);
1047 return true;
1048 }
1049
1050 bool pop(double& val) {
1051 size_t currentRead = readIdx.load(std::memory_order_relaxed);
1052 if (currentRead == writeIdx.load(std::memory_order_acquire)) {
1053 return false; // Empty
1054 }
1055 val = data[currentRead];
1056 readIdx.store((currentRead + 1) & (kLoudnessQueueSize - 1), std::memory_order_release);
1057 return true;
1058 }
1059 };
