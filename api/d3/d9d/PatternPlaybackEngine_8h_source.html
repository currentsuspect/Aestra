53template <typename T, size_t Capacity> class LockFreeSPSCQueue {
54public:
58 LockFreeSPSCQueue() : m_head(0), m_tail(0) {}
59
65 bool push(const T& item) {
66 uint32_t head = m_head.load(std::memory_order_relaxed);
67 uint32_t nextHead = (head + 1) % Capacity;
68
69 if (nextHead == m_tail.load(std::memory_order_acquire)) {
70 return false; // Queue full
71 }
72
73 m_buffer[head] = item;
74 m_head.store(nextHead, std::memory_order_release);
75 return true;
76 }
77
83 bool peek(T& item) const {
84 uint32_t tail = m_tail.load(std::memory_order_relaxed);
85 if (tail == m_head.load(std::memory_order_acquire)) {
86 return false; // Queue empty
87 }
88 item = m_buffer[tail];
89 return true;
90 }
91
95 void pop() {
96 uint32_t tail = m_tail.load(std::memory_order_relaxed);
97 m_tail.store((tail + 1) % Capacity, std::memory_order_release);
98 }
99
107 void forceDrain() {
108 uint32_t tail = m_tail.load(std::memory_order_acquire);
109 m_head.store(tail, std::memory_order_release);
110 }
111
116 size_t size() const {
117 uint32_t head = m_head.load(std::memory_order_acquire);
118 uint32_t tail = m_tail.load(std::memory_order_acquire);
119 return (head >= tail) ? (head - tail) : (Capacity - tail + head);
120 }
121
122private:
123 std::array<T, Capacity> m_buffer;
124 std::atomic<uint32_t> m_head;
125 std::atomic<uint32_t> m_tail;
126};
