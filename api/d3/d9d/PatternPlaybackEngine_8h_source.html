56template <typename T, size_t Capacity> class LockFreeSPSCQueue {
57public:
61 LockFreeSPSCQueue() : m_head(0), m_tail(0) {}
62
68 bool push(const T& item) {
69 uint32_t head = m_head.load(std::memory_order_relaxed);
70 uint32_t nextHead = (head + 1) % Capacity;
71
72 if (nextHead == m_tail.load(std::memory_order_acquire)) {
73 return false; // Queue full
74 }
75
76 m_buffer[head] = item;
77 m_head.store(nextHead, std::memory_order_release);
78 return true;
79 }
80
86 bool peek(T& item) const {
87 uint32_t tail = m_tail.load(std::memory_order_relaxed);
88 if (tail == m_head.load(std::memory_order_acquire)) {
89 return false; // Queue empty
90 }
91 item = m_buffer[tail];
92 return true;
93 }
94
98 void pop() {
99 uint32_t tail = m_tail.load(std::memory_order_relaxed);
100 m_tail.store((tail + 1) % Capacity, std::memory_order_release);
101 }
102
107 size_t size() const {
108 uint32_t head = m_head.load(std::memory_order_acquire);
109 uint32_t tail = m_tail.load(std::memory_order_acquire);
110 return (head >= tail) ? (head - tail) : (Capacity - tail + head);
111 }
112
113private:
114 std::array<T, Capacity> m_buffer;
115 std::atomic<uint32_t> m_head;
116 std::atomic<uint32_t> m_tail;
117};

359 // Queue epoch. Discarding queued events is a counter bump, never an index
360 // reset: each SPSC index keeps exactly one writer (head: the scheduler,
361 // tail: the audio thread). Any thread may bump it (rewindScheduledInstances);
362 // processAudio() drops events stamped with an older epoch, and refillWindow()
363 // treats a bump it did not make as a rewind request.
364 std::atomic<uint16_t> m_queueEpoch{0};
365 // The epoch the scheduler stamps onto new events (scheduler thread, under m_mutex).
366 uint16_t m_scheduledEpoch{0};
367 std::atomic<uint16_t> m_plannedEpoch{0xFFFF}; // see plannedEpoch(); no refill yet
368 std::atomic<size_t> m_instanceCount{0}; // m_activeInstances.size(), for hasScheduledInstances()
369 // Scheduler side: discard everything queued so far and stamp from a fresh epoch.
370 // Bumps only from the epoch this refill planned under: if a rewind landed since, this
371 // refill's events are already dead, and adopting the bumped epoch would revive them.
372 void invalidateQueuedEventsLocked() {
373 uint16_t expected = m_scheduledEpoch;
374 if (m_queueEpoch.compare_exchange_strong(expected, static_cast<uint16_t>(expected + 1),
375 std::memory_order_acq_rel)) {
376 m_scheduledEpoch = static_cast<uint16_t>(expected + 1);
377 }
