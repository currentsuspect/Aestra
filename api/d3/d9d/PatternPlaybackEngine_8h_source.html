52template <typename T, size_t Capacity> class LockFreeSPSCQueue {
53public:
57 LockFreeSPSCQueue() : m_head(0), m_tail(0) {}
58
64 bool push(const T& item) {
65 uint32_t head = m_head.load(std::memory_order_relaxed);
66 uint32_t nextHead = (head + 1) % Capacity;
67
68 if (nextHead == m_tail.load(std::memory_order_acquire)) {
69 return false; // Queue full
70 }
71
72 m_buffer[head] = item;
73 m_head.store(nextHead, std::memory_order_release);
74 return true;
75 }
76
82 bool peek(T& item) const {
83 uint32_t tail = m_tail.load(std::memory_order_relaxed);
84 if (tail == m_head.load(std::memory_order_acquire)) {
85 return false; // Queue empty
86 }
87 item = m_buffer[tail];
88 return true;
89 }
90
94 void pop() {
95 uint32_t tail = m_tail.load(std::memory_order_relaxed);
96 m_tail.store((tail + 1) % Capacity, std::memory_order_release);
97 }
98
106 void forceDrain() {
107 uint32_t tail = m_tail.load(std::memory_order_acquire);
108 m_head.store(tail, std::memory_order_release);
109 }
110
115 size_t size() const {
116 uint32_t head = m_head.load(std::memory_order_acquire);
117 uint32_t tail = m_tail.load(std::memory_order_acquire);
118 return (head >= tail) ? (head - tail) : (Capacity - tail + head);
119 }
120
121private:
122 std::array<T, Capacity> m_buffer;
123 std::atomic<uint32_t> m_head;
124 std::atomic<uint32_t> m_tail;
125};
