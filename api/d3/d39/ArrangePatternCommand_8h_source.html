30class ArrangePatternCommand : public ICommand {
31public:
32 ArrangePatternCommand(TrackManager& trackManager, PatternID patternId, size_t trackIndex,
33 double startBeat)
34 : m_trackManager(trackManager), m_patternId(patternId), m_trackIndex(trackIndex),
35 m_startBeat(startBeat) {}
36
37 void execute() override {
38 if (m_executed) return;
39 m_createdLanes.clear();
40 m_previousUnitRoutes.clear();
41
42 auto& playlist = m_trackManager.getPlaylistModel();
43 auto& unitManager = m_trackManager.getUnitManager();
44 const PatternSource* pattern = m_trackManager.getPatternManager().getPattern(m_patternId);
45 if (!pattern || !pattern->isMidi()) return;
46
47 // A failure after partial mutation must not leave lanes or routes
48 // behind: unwind everything staged so far before returning without
49 // m_executed set.
50 const auto rollbackStaged = [&]() {
51 for (auto it = m_previousUnitRoutes.rbegin(); it != m_previousUnitRoutes.rend();
52 ++it) {
53 unitManager.assignUnitToTimelineLane(it->first, it->second);
54 }
55 for (auto it = m_createdLanes.rbegin(); it != m_createdLanes.rend(); ++it) {
56 playlist.removeLane(*it);
57 }
58 m_previousUnitRoutes.clear();
59 m_createdLanes.clear();
60 };
61
62 while (playlist.getLaneCount() <= m_trackIndex) {
63 PlaylistLaneID laneId =
64 playlist.createLane("Lane " + std::to_string(playlist.getLaneCount() + 1));
65 if (!laneId.isValid()) {
66 rollbackStaged();
67 return;
68 }
69 m_createdLanes.push_back(laneId);
70 }
71 const PlaylistLaneID laneId = playlist.getLaneId(m_trackIndex);
72 if (!laneId.isValid()) {
73 rollbackStaged();
74 return;
75 }
76
77 std::unordered_set<UnitID> routed;
78 for (const MidiNote& note : std::get<MidiPayload>(pattern->payload).notes) {
79 if (note.unitId == 0 || !routed.insert(note.unitId).second) continue;
80 if (!unitManager.getUnit(note.unitId)) continue;
81 m_previousUnitRoutes.emplace_back(note.unitId,
82 unitManager.getUnitTimelineLane(note.unitId));
83 unitManager.assignUnitToTimelineLane(note.unitId, static_cast<int>(m_trackIndex));
84 }
85
86 ClipInstance clip;
87 clip.id = ClipInstanceID::generate();
88 clip.name = pattern->name;
89 clip.startBeat = m_startBeat;
90 clip.durationBeats = pattern->lengthBeats;
91 clip.patternId = m_patternId;
92 clip.sourceId = m_patternId.value;
93
94 m_clipId = playlist.addClip(laneId, clip);
95 if (!m_clipId.isValid()) {
96 rollbackStaged();
97 return;
98 }
99 m_executed = true;
100 }
101
102 void undo() override {
103 if (!m_executed) return;
104 auto& playlist = m_trackManager.getPlaylistModel();
105 auto& unitManager = m_trackManager.getUnitManager();
106
107 playlist.removeClip(m_clipId);
108 for (auto it = m_previousUnitRoutes.rbegin(); it != m_previousUnitRoutes.rend(); ++it) {
109 unitManager.assignUnitToTimelineLane(it->first, it->second);
110 }
111 for (auto it = m_createdLanes.rbegin(); it != m_createdLanes.rend(); ++it) {
112 playlist.removeLane(*it);
113 }
114 m_executed = false;
115 }
116
117 void redo() override {
118 if (m_executed) return;
119 execute();
120 }
121
122 std::string getName() const override { return "Arrange Pattern"; }
123 size_t getSizeInBytes() const override { return sizeof(*this); }
124 bool changesProjectState() const override { return true; }
125
126private:
127 TrackManager& m_trackManager;
128 PatternID m_patternId;
129 size_t m_trackIndex;
130 double m_startBeat;
131
132 // Recorded by execute() for undo
133 std::vector<PlaylistLaneID> m_createdLanes;
134 std::vector<std::pair<UnitID, int>> m_previousUnitRoutes;
135 ClipInstanceID m_clipId;
136 bool m_executed = false;
137};
