25class ArrangePatternCommand : public ICommand {
26public:
27 ArrangePatternCommand(TrackManager& trackManager, PatternID patternId, size_t trackIndex,
28 double startBeat)
29 : m_trackManager(trackManager), m_patternId(patternId), m_trackIndex(trackIndex),
30 m_startBeat(startBeat) {}
31
32 void execute() override {
33 if (m_executed) return;
34 m_createdLanes.clear();
35
36 auto& playlist = m_trackManager.getPlaylistModel();
37 const PatternSource* pattern = m_trackManager.getPatternManager().getPattern(m_patternId);
38 if (!pattern || !pattern->isMidi()) return;
39
40 // A failure after partial mutation must not leave lanes behind.
41 const auto rollbackStaged = [&]() {
42 for (auto it = m_createdLanes.rbegin(); it != m_createdLanes.rend(); ++it) {
43 playlist.removeLane(*it);
44 }
45 m_createdLanes.clear();
46 };
47
48 while (playlist.getLaneCount() <= m_trackIndex) {
49 PlaylistLaneID laneId =
50 playlist.createLane("Lane " + std::to_string(playlist.getLaneCount() + 1));
51 if (!laneId.isValid()) {
52 rollbackStaged();
53 return;
54 }
55 m_createdLanes.push_back(laneId);
56 }
57 const PlaylistLaneID laneId = playlist.getLaneId(m_trackIndex);
58 if (!laneId.isValid()) {
59 rollbackStaged();
60 return;
61 }
62
63 ClipInstance clip;
64 clip.id = ClipInstanceID::generate();
65 clip.name = pattern->name;
66 clip.startBeat = m_startBeat;
67 clip.durationBeats = pattern->lengthBeats;
68 clip.patternId = m_patternId;
69 clip.sourceId = m_patternId.value;
70
71 m_clipId = playlist.addClip(laneId, clip);
72 if (!m_clipId.isValid()) {
73 rollbackStaged();
74 return;
75 }
76 m_executed = true;
77 }
78
79 void undo() override {
80 if (!m_executed) return;
81 auto& playlist = m_trackManager.getPlaylistModel();
82
83 playlist.removeClip(m_clipId);
84 for (auto it = m_createdLanes.rbegin(); it != m_createdLanes.rend(); ++it) {
85 playlist.removeLane(*it);
86 }
87 m_executed = false;
88 }
89
90 void redo() override {
91 if (m_executed) return;
92 execute();
93 }
94
95 std::string getName() const override { return "Arrange Pattern"; }
96 size_t getSizeInBytes() const override { return sizeof(*this); }
97 bool changesProjectState() const override { return true; }
98
99private:

101 PatternID m_patternId;

103 double m_startBeat;
104

106 std::vector<PlaylistLaneID> m_createdLanes;

108 bool m_executed = false;
