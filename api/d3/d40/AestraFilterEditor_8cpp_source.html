125void AestraFilterEditor::drawResponseDisplay(NUIRenderer& renderer) {
126 auto& theme = NUIThemeManager::getInstance();
127 renderer.fillRoundedRect(m_responseRect, 12.0f, insetSurface());
128 renderer.strokeRoundedRect(m_responseRect, 12.0f, 1.0f, accent().withAlpha(0.22f));
129
130 renderer.drawText("FILTER RESPONSE", {m_responseRect.x + 14.0f, m_responseRect.y + 10.0f}, 9.0f,
131 theme.getColor("textSecondary").withAlpha(0.68f));
132
133 const auto filter = std::dynamic_pointer_cast<AestraFilter>(m_instance);
134 const float envelope = filter ? std::clamp(filter->getEnvelopeLevel(), 0.0f, 1.0f) : 0.0f;
135 const float cutoffNorm = m_instance->getParameter(AestraFilter::kCutoff);
136 const float baseCutoff = AestraFilter::cutoffHzFromNorm(cutoffNorm);
137 const float envAmount = 2.0f * m_instance->getParameter(AestraFilter::kEnvAmount) - 1.0f;
138 const float modCutoff =
139 std::clamp(baseCutoff * std::exp2(envAmount * AestraFilter::kEnvModOctaves * envelope), 20.0f, 20000.0f);
140 const float q = AestraFilter::qFromNorm(m_instance->getParameter(AestraFilter::kReso));
141 const auto type = AestraFilter::typeFromNorm(m_instance->getParameter(AestraFilter::kType));
142
143 const int envPct = static_cast<int>(std::round(envelope * 100.0f));
144 const std::string envLabel = "ENV " + std::to_string(envPct) + "%";
145 const float envLabelW = renderer.measureText(envLabel, 9.0f).width;
146 renderer.drawText(envLabel, {m_responseRect.right() - envLabelW - 14.0f, m_responseRect.y + 10.0f}, 9.0f,
147 accent().withAlpha(0.82f));
148
149 // Keep a dedicated scale gutter below the plot so frequency labels never
150 // sit on the response border or compete with the live envelope meter.
151 const NUIRect graph{m_responseRect.x + 14.0f, m_responseRect.y + 31.0f, m_responseRect.width - 28.0f,
152 m_responseRect.height - 65.0f};
153 renderer.fillRoundedRect(graph, 7.0f, editorNeutral(NUIColor(0.006f, 0.010f, 0.011f, 0.98f)));
154
155 auto xForHz = [&graph](float hz) {
156 const float norm = std::log10(std::clamp(hz, 20.0f, 20000.0f) / 20.0f) / 3.0f;
157 return graph.x + norm * graph.width;
158 };
159 auto yForDb = [&graph](float db) {
160 constexpr float kTopDb = 18.0f;
161 constexpr float kBottomDb = -36.0f;
162 return graph.y + (kTopDb - std::clamp(db, kBottomDb, kTopDb)) / (kTopDb - kBottomDb) * graph.height;
163 };
164
165 for (float db : {-24.0f, -12.0f, 0.0f, 12.0f}) {
166 const float y = yForDb(db);
167 renderer.drawLine({graph.x, y}, {graph.right(), y}, db == 0.0f ? 1.2f : 1.0f,
168 gridColor().withAlpha(db == 0.0f ? 0.22f : 0.12f));
169 }
170
171 struct FrequencyGuide {
172 float hz;
173 const char* label;
174 };
175 static constexpr FrequencyGuide kGuides[] = {
176 {20.0f, "20"}, {100.0f, "100"}, {1000.0f, "1k"}, {10000.0f, "10k"}, {20000.0f, "20k"},
177 };
178 for (const auto& guide : kGuides) {
179 const float x = xForHz(guide.hz);
180 renderer.drawLine({x, graph.y}, {x, graph.bottom()}, 1.0f, gridColor());
181 const float labelW = renderer.measureText(guide.label, 8.0f).width;
182 renderer.drawText(guide.label,
183 {std::clamp(x - labelW * 0.5f, graph.x, graph.right() - labelW), graph.bottom() + 5.0f},
184 8.0f, theme.getColor("textSecondary").withAlpha(0.58f));
185 }
186
187 auto drawResponse = [&](float cutoff, NUIColor color, float thickness) {
188 std::array<NUIPoint, 129> points{};
189 for (size_t i = 0; i < points.size(); ++i) {
190 const float t = static_cast<float>(i) / static_cast<float>(points.size() - 1);
191 const float freq = 20.0f * std::pow(1000.0f, t);
192 const float ratio = freq / std::max(20.0f, cutoff);
193 const float ratio2 = ratio * ratio;
194 const float damping = ratio / q;
195 const float denominator = std::sqrt((1.0f - ratio2) * (1.0f - ratio2) + damping * damping);
196 float magnitude = 1.0f / std::max(1.0e-5f, denominator);
197 if (type == AestraFilter::kTypeBandPass)
198 magnitude *= ratio;
199 else if (type == AestraFilter::kTypeHighPass)
200 magnitude *= ratio2;
201 const float db = 20.0f * std::log10(std::max(1.0e-5f, magnitude));
202 points[i] = {graph.x + t * graph.width, yForDb(db)};
203 }
204 renderer.drawPolyline(points.data(), static_cast<int>(points.size()), thickness, color);
205 };
206
207 renderer.setClipRect(graph);
208 const bool modulationActive = std::abs(envAmount) > 0.005f;
209 if (modulationActive) {
210 drawResponse(baseCutoff, theme.getColor("textSecondary").withAlpha(0.28f), 1.5f);
211 renderer.drawLine({xForHz(baseCutoff), graph.y}, {xForHz(baseCutoff), graph.bottom()}, 1.0f,
212 theme.getColor("textSecondary").withAlpha(0.20f));
213 }
214 drawResponse(modCutoff, accent().withAlpha(0.96f), 2.4f);
215 renderer.drawLine({xForHz(modCutoff), graph.y}, {xForHz(modCutoff), graph.bottom()}, 1.2f,
216 accent().withAlpha(0.34f));
217 renderer.clearClipRect();
218
219 const NUIRect envTrack{graph.x, m_responseRect.bottom() - 8.0f, graph.width, 3.0f};
220 renderer.fillRoundedRect(envTrack, 1.5f, editorInk(0.06f));
221 renderer.fillRoundedRect({envTrack.x, envTrack.y, envTrack.width * envelope, envTrack.height}, 1.5f,
222 accent().withAlpha(0.72f));
223}

297bool AestraFilterEditor::onMouseEvent(const NUIMouseEvent& event) {
298 if (!isVisible())
299 return false;
300 if (AestraPanelWindow::onMouseEvent(event))
301 return true;
302 if (!m_instance)
303 return false;
304
305 const float mx = event.position.x;
306
307 // Bypass pill
308 if (m_bypassRect.contains(event.position) && event.pressed && event.button == NUIMouseButton::Left) {
309 const float cur = m_instance->getParameter(AestraFilter::kBypass);
310 m_instance->setParameter(AestraFilter::kBypass, cur > 0.5f ? 0.0f : 1.0f);
311 setDirty();
312 return true;
313 }
314 m_bypassHovered = m_bypassRect.contains(event.position);
315
316 // Type segments
317 if (event.pressed && event.button == NUIMouseButton::Left) {
318 for (size_t i = 0; i < m_typeRects.size(); ++i) {
319 if (m_typeRects[i].contains(event.position)) {
320 m_instance->setParameter(AestraFilter::kType,
321 AestraFilter::normFromType(static_cast<AestraFilter::Type>(i)));
322 setDirty();
323 return true;
324 }
325 }
326 }
327
328 // Knob vertical drag (rotary -> cursor capture): AestraPanelWindow::handleKnobDrag.
329 const KnobTarget knobs[] = {
330 {m_cutoffRect, AestraFilter::kCutoff}, {m_resoRect, AestraFilter::kReso},
331 {m_driveRect, AestraFilter::kDrive}, {m_envRect, AestraFilter::kEnvAmount},
332 {m_attackRect, AestraFilter::kEnvAttack}, {m_releaseRect, AestraFilter::kEnvRelease},
333 };
334 const auto get = [this](uint32_t id) { return m_instance->getParameter(id); };
335 const auto set = [this](uint32_t id, float v) { m_instance->setParameter(id, v); };
336 if (handleKnobDrag(event, knobs, std::size(knobs), kDragRangePx, get, set))
337 return true;
338
339 // Mix slider drag
340 const NUIRect mixTrack = mixTrackRect();
341 if (m_draggingMix) {
342 if (event.released) {
343 m_draggingMix = false;
344 return true;
345 }
346 if (event.button == NUIMouseButton::None) {
347 const float t = std::clamp((mx - mixTrack.x) / mixTrack.width, 0.0f, 1.0f);
348 m_instance->setParameter(AestraFilter::kMix, t);
349 setDirty();
350 return true;
351 }
352 }
353 if (event.pressed && event.button == NUIMouseButton::Left && mixTrack.contains(event.position)) {
354 m_draggingMix = true;
355 const float t = std::clamp((mx - mixTrack.x) / mixTrack.width, 0.0f, 1.0f);
356 m_instance->setParameter(AestraFilter::kMix, t);
357 setDirty();
358 return true;
359 }
360
361 return consumeInsideBounds(event);
362}
