56 void draw(AestraUI::NUIRenderer& r, const ArsenalRackLayout& rack, const std::vector<RowPtr>& rows, Audio::TrackManager* tm,
57 Audio::PatternID pattern, const AestraUI::NUIRect& viewport) const {
58 if (!tm || viewport.height <= 0.0f) return;
59 auto& theme = AestraUI::NUIThemeManager::getInstance();
60 r.setClipRect(viewport);
61 for (const auto& item : rack.items) {
62 if (item.kind != ArsenalRackItem::Kind::Row || !rows[item.rowIndex]) continue;
63 const auto& row = *rows[item.rowIndex];
64 const bool open = m_open.count(row.getUnitId()) > 0;
65 const auto c = caretRect(row);
66 const float cx = c.x + c.width * 0.5f, cy = c.y + c.height * 0.5f;
67 const auto col = theme.getColor("textSecondary").withAlpha(open ? 1.0f : 0.8f);
68 r.strokeRoundedRect(c, theme.getRadius("xs"), 1.0f, theme.getColor("border").withAlpha(open ? 0.9f : 0.6f));
69 const float d = open ? -2.5f : 2.5f; // open points up
70 r.drawLine({cx - 4.0f, cy - d}, {cx, cy + d}, 1.5f, col);
71 r.drawLine({cx, cy + d}, {cx + 4.0f, cy - d}, 1.5f, col);
72 if (open && item.extraH > 0.0f) {
73 auto ctx = contextFor(*tm, pattern, row, item.extraH);
74 ctx.clip = viewport; // nested clips stay inside the list, and hand it back when done
75 if (hasSound(*tm, row.getUnitId())) {
76 auto& strip = syncStrip(*tm, row.getUnitId());
77 strip.draw(r, soundRect(row), viewport);
78 if (m_drawerUnit == row.getUnitId())
79 ArsenalSoundDrawer::draw(r, drawerRect(row), m_drawer, strip.list(), tm->getUnitManager().getUnit(row.getUnitId())->audioClipPath,
80 row.channelColor(), strip.setName(), m_missing[row.getUnitId()], viewport);
81 }
82 m_lanes.draw(r, ctx);
83 }
84 }
85 r.clearClipRect();
86 }

89 bool onMouse(const AestraUI::NUIMouseEvent& e, const ArsenalRackLayout& rack, const std::vector<RowPtr>& rows,
90 Audio::TrackManager* tm, Audio::PatternID pattern, const AestraUI::NUIRect& viewport, bool& relayout, bool& edited) {
91 if (!tm) return false;
92 if (m_drawerUnit != 0 && e.wheelDelta != 0.0f) {
93 for (const auto& item : rack.items)
94 if (item.kind == ArsenalRackItem::Kind::Row && rows[item.rowIndex] && rows[item.rowIndex]->getUnitId() == m_drawerUnit &&
95 drawerRect(*rows[item.rowIndex]).contains(e.position)) {
96 ArsenalSoundDrawer::scrollBy(m_drawer, ArsenalSoundDrawer::filtered(m_sounds[m_drawerUnit].list(), m_drawer.filter).size(), e.wheelDelta > 0.0f ? -2 : 2);
97 return true;
98 }
99 }
100 if (m_lanes.dragging()) {
101 if (const auto* item = itemFor(rack, m_dragRow); item && rows[m_dragRow]) {
102 const auto ctx = contextFor(*tm, pattern, *rows[m_dragRow], item->extraH);
103 if (!e.pressed && e.button == AestraUI::NUIMouseButton::Left) m_lanes.release(ctx);
104 else m_lanes.drag(ctx, e.position);
105 edited = m_lanes.takeEdited();
106 return true;
107 }
108 m_lanes.release({});
109 return false;
110 }
111 if (!e.pressed || e.button != AestraUI::NUIMouseButton::Left) return false;
112 if (viewport.height > 0.0f && !viewport.contains(e.position)) return false;
113 if (m_drawerUnit != 0) { // a press anywhere outside the drawer and its strip closes it, keeping the sound it is on
114 bool inside = false;
115 for (const auto& item : rack.items)
116 if (item.kind == ArsenalRackItem::Kind::Row && rows[item.rowIndex] && rows[item.rowIndex]->getUnitId() == m_drawerUnit)
117 inside = drawerRect(*rows[item.rowIndex]).contains(e.position) || soundRect(*rows[item.rowIndex]).contains(e.position);
118 if (!inside) { m_drawerUnit = 0; relayout = true; }
119 }
120 for (const auto& item : rack.items) {
121 if (item.kind != ArsenalRackItem::Kind::Row || !rows[item.rowIndex]) continue;
122 const auto& row = *rows[item.rowIndex];
123 if (caretRect(row).contains(e.position)) {
124 const auto id = row.getUnitId();
125 if (!m_open.erase(id)) m_open.insert(id);
126 relayout = true;
127 return true;
128 }
129 if (m_drawerUnit == row.getUnitId() && item.extraH > 0.0f && drawerRect(row).contains(e.position)) {
130 auto& strip = m_sounds[m_drawerUnit];
131 const auto idx = ArsenalSoundDrawer::filtered(strip.list(), m_drawer.filter);
132 if (ArsenalSoundDrawer::chipRect(drawerRect(row)).contains(e.position)) { // the chip moves to the next sound set (Folder, then each set)
133 std::vector<std::string> choices{std::string()};
134 if (m_setNames) for (const auto& n : m_setNames()) choices.push_back(n);
135 const auto* unit = tm->getUnitManager().getUnit(m_drawerUnit);
136 const auto at = unit ? std::find(choices.begin(), choices.end(), unit->soundSet) : choices.end();
137 const std::string next = choices[(static_cast<size_t>(at == choices.end() ? 0 : at - choices.begin()) + 1) % choices.size()];
138 if (m_onSetChosen) m_onSetChosen(m_drawerUnit, next);
139 m_stale.insert(m_drawerUnit);
140 auto& fresh = syncStrip(*tm, m_drawerUnit);
141 m_drawer = ArsenalSoundDrawer::open(fresh.list(), unit ? unit->audioClipPath : std::string());
142 return true;
143 }
144 const int hit = ArsenalSoundDrawer::rowAt(m_drawer, idx.size(), drawerRect(row), e.position);
145 if (hit >= 0) {
146 m_drawer.selected = hit;
147 if (m_onSwap) m_onSwap(m_drawerUnit, strip.list()[static_cast<size_t>(idx[static_cast<size_t>(hit)])]); // click auditions in the loop
148 }
149 return true;
150 }
151 if (item.extraH > 0.0f && m_open.count(row.getUnitId()) && hasSound(*tm, row.getUnitId()) && soundRect(row).contains(e.position)) {
152 auto& strip = m_sounds[row.getUnitId()];
153 const auto hit = strip.hitTest(soundRect(row), e.position);
154 if (hit == ArsenalSoundStrip::Hit::None) { // the name opens (or, again, closes keeping) the drawer
155 if (m_drawerUnit == row.getUnitId()) m_drawerUnit = 0;
156 else if (const auto* unit = tm->getUnitManager().getUnit(row.getUnitId())) {
157 m_stale.insert(row.getUnitId()); // pick up changes made to the library since the row was last drawn
158 m_drawerUnit = row.getUnitId();
159 m_drawer = ArsenalSoundDrawer::open(syncStrip(*tm, row.getUnitId()).list(), unit->audioClipPath);
160 }
161 relayout = true;
162 return true;
163 }
164 const std::string next = hit == ArsenalSoundStrip::Hit::None ? std::string() : strip.neighbour(hit == ArsenalSoundStrip::Hit::Next ? 1 : -1);
165 if (!next.empty() && m_onSwap) m_onSwap(row.getUnitId(), next);
166 return true;
167 }
168 if (item.extraH > 0.0f && m_open.count(row.getUnitId()) &&
169 m_lanes.press(contextFor(*tm, pattern, row, item.extraH), e.position)) {
170 m_dragRow = item.rowIndex;
171 edited = m_lanes.takeEdited();
172 return true;
173 }
174 }
175 return false;
176 }

180 bool onKey(const AestraUI::NUIKeyEvent& e, Audio::TrackManager* tm, bool& relayout) {
181 if (m_drawerUnit == 0 || !tm || !e.pressed) return false;
182 auto& strip = m_sounds[m_drawerUnit];
183 const auto out = ArsenalSoundDrawer::onKey(m_drawer, strip.list(), e);
184 if (out.result == ArsenalSoundDrawer::Result::Ignored) return false;
185 const auto* unit = tm->getUnitManager().getUnit(m_drawerUnit);
186 const std::string current = unit ? unit->audioClipPath : std::string();
187 if (!out.path.empty() && out.path != current && m_onSwap && out.result != ArsenalSoundDrawer::Result::Consumed) m_onSwap(m_drawerUnit, out.path);
188 if (out.result == ArsenalSoundDrawer::Result::Keep || out.result == ArsenalSoundDrawer::Result::Revert) { m_drawerUnit = 0; relayout = true; }
189 return true;
190 }
191 bool drawerOpen() const { return m_drawerUnit != 0; }

195 ArsenalSoundStrip& syncStrip(Audio::TrackManager& tm, Audio::UnitID id) const {
196 auto& strip = m_sounds[id];
197 const auto* unit = tm.getUnitManager().getUnit(id);
198 if (!unit) return strip;
199 if (!unit->soundSet.empty() && m_setMembers) {
200 if (strip.setName() != unit->soundSet || !strip.isSet() || m_stale.count(id)) {
201 const auto resolved = ArsenalSoundSet::resolve(m_setMembers(unit->soundSet));
202 strip.useFolder();
203 strip.follow(unit->audioClipPath);
204 strip.useList(unit->soundSet, resolved.files);
205 m_missing[id] = resolved.missing.size();
206 m_stale.erase(id);
207 }
208 } else if (strip.isSet() || m_stale.count(id)) {
209 strip.useFolder();
210 m_missing[id] = 0;
211 m_stale.erase(id);
212 }
213 strip.follow(unit->audioClipPath);
214 return strip;
215 }

243 ArsenalStepLanes::Context contextFor(Audio::TrackManager& tm, Audio::PatternID pattern, const AestraUI::UnitRow& row, float extraH) const {
244 ArsenalStepLanes::Context c;
245 const auto b = row.getBounds();
246 c.tm = &tm;
247 c.pattern = pattern;
248 c.unit = row.getUnitId();
249 if (const auto* u = tm.getUnitManager().getUnit(c.unit)) c.type = u->type;
250 c.root = tm.getUnitManager().getUnitRootMidiNote(c.unit);
251 c.steps = row.getStepCount();
252 const float off = soundOffset(c.type) + drawerExtra(c.unit);
253 c.area = {b.x, b.bottom() + 2.0f + off, b.width, std::max(0.0f, extraH - 2.0f - off)};
254 c.strip = row.stepStrip();
255 c.stepW = row.stepPadWidth();
256 c.groupGap = AestraUI::UnitRow::stepGroupGap();
257 c.scrollX = row.getScrollX();
258 c.accent = row.channelColor();
259 return c;
260 }
