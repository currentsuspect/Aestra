14void AestraContent::bounceClipsInPlace(const std::vector<Aestra::Audio::ClipInstanceID>& clipIds) {
15 if (!m_audioEngine || !m_trackManager) {
16 return;
17 }
18 // The render drives the engine by hand, as Export does: the live callback must not be
19 // running it at the same time.
20 auto* deviceManager = Aestra::ServiceLocator::get<Aestra::Audio::AudioDeviceManager>();
21 const bool resume = deviceManager && deviceManager->isStreamRunning();
22 if (resume) {
23 deviceManager->stopStream();
24 }
25 const auto result = Aestra::Audio::bounceClipsInPlace(*m_audioEngine, *m_trackManager, clipIds);
26 if (resume) {
27 deviceManager->startStream();
28 }
29 showToast(result.ok ? "Bounced in place. Undo puts the clips and effects back." : result.error);
30}

32bool AestraContent::renderMixToBuffer(std::vector<float>& interleaved, uint32_t& channels, uint32_t& sampleRate) {
33 if (!m_audioEngine || !m_trackManager) return false;
34 const double beats = m_trackManager->getPlaylistModel().getTotalDurationBeats();
35 if (beats < 0.1) {
36 AESTRA_LOG_INFO("[Versions] Nothing on the timeline to bounce");
37 return false;
38 }
39 auto* deviceManager = Aestra::ServiceLocator::get<Aestra::Audio::AudioDeviceManager>();
40 const bool resume = deviceManager && deviceManager->isStreamRunning();
41 if (resume) deviceManager->stopStream();
42 Aestra::Audio::AudioExporter exporter(*m_audioEngine, *m_trackManager);
43 Aestra::Audio::AudioExporter::Config config;
44 config.scope = Aestra::Audio::AudioExporter::RenderScope::FullSong;
45 config.startBeat = 0.0;
46 config.endBeat = beats;
47 config.sampleRate = 44100;
48 config.tailSeconds = 1.0;
49 Aestra::Audio::AudioExporter::RenderedAudio rendered;
50 // Pattern and MIDI clips reach their units through the pattern-playback engine, which only holds what play
51 // scheduled: put the timeline in for the render, and take it out again so playback starts clean.
52 auto& patterns = m_trackManager->getPatternPlaybackEngine();
53 patterns.clearScheduledInstances();
54 m_trackManager->scheduleTimelineForOfflineRender(0.0);
55 const auto result = exporter.renderToBuffer(config, rendered);
56 patterns.clearScheduledInstances();
57 if (resume) deviceManager->startStream();
58 if (!result.ok() || rendered.interleaved.empty()) {
59 AESTRA_LOG_WARNING("[Versions] Bounce render failed: " + result.errorMessage);
60 return false;
61 }
62 interleaved = std::move(rendered.interleaved);
63 channels = rendered.numChannels;
64 sampleRate = rendered.sampleRate;
65 return true;
66}

74bool AestraContent::togglePreviewFile(const std::string& path) {
75 if (path.empty()) return false;
76 if (isPreviewFilePlaying(path)) {
77 m_auditionEngine->stop();
78 return false;
79 }
80 if (!m_auditionEngine) {
81 m_auditionEngine = std::make_shared<Aestra::Audio::AuditionEngine>();
82 if (m_audioEngine) {
83 m_audioEngine->setAuditionEngine(m_auditionEngine.get());
84 m_auditionEngine->setSampleRate(static_cast<double>(m_audioEngine->getSampleRate()));
85 }
86 }
87 m_auditionEngine->addToQueue(path);
88 m_auditionEngine->jumpToTrack(m_auditionEngine->getQueue().size() - 1);
89 m_auditionEngine->play();
90 return true;
91}

110void AestraContent::setAuditionActive(bool active) {
111 if (!m_auditionPanel) return;
