29 static Resolved resolve(const std::vector<std::string>& members) {
30 namespace fs = std::filesystem;
31 Resolved out;
32 for (const auto& member : members) {
33 std::error_code ec;
34 if (fs::is_directory(member, ec)) {
35 for (const auto& f : ArsenalSoundStrip::siblings((fs::path(member) / "x").string())) out.files.push_back(f); // the folder's audio, sorted
36 } else if (fs::is_regular_file(member, ec)) {
37 if (ArsenalSoundStrip::isAudioFile(fs::path(member).filename().string())) out.files.push_back(member);
38 } else {
39 out.missing.push_back(member);
40 }
41 }
42 const auto lower = [](std::string s) { for (auto& c : s) c = static_cast<char>(std::tolower(static_cast<unsigned char>(c))); return s; };
43 std::sort(out.files.begin(), out.files.end(), [&](const std::string& a, const std::string& b) {
44 const auto na = lower(fs::path(a).filename().string()), nb = lower(fs::path(b).filename().string());
45 return na != nb ? na < nb : a < b;
46 });
47 out.files.erase(std::unique(out.files.begin(), out.files.end()), out.files.end());
48 if (out.files.size() > kMaxSounds) out.files.resize(kMaxSounds);
49 return out;
50 }
51};
