93inline bool move(AutomationCurve& curve, uint64_t id, double deltaBeats) {
94 materialize(curve);
95 auto it = std::find_if(curve.regions.begin(), curve.regions.end(), [id](const auto& r) { return r.id == id; });
96 if (it == curve.regions.end())
97 return false;
98 if (std::abs(deltaBeats) < kEps)
99 return true;
100 const AutomationRegion src = *it;
101 const double start = src.startBeat + deltaBeats;
102 const double end = src.endBeat + deltaBeats;
103 if (start < -kEps || overlapsAnother(curve, id, start, end))
104 return false;
105
106 std::vector<AutomationPoint> carried;
107 for (const auto& p : curve.points)
108 if (inside(src, p.beat))
109 carried.push_back(p);
110 auto& pts = curve.points;
111 pts.erase(std::remove_if(pts.begin(), pts.end(), [&](const AutomationPoint& p) { return inside(src, p.beat); }),
112 pts.end());
113 clearSpan(curve, start, end);
114 for (auto p : carried) {
115 p.beat += deltaBeats;
116 p.sample = 0; // legacy tempo cache, never read
117 pts.push_back(p);
118 }
119 curve.sortPoints();
120 it = std::find_if(curve.regions.begin(), curve.regions.end(), [id](const auto& r) { return r.id == id; });
121 it->startBeat = start;
122 it->endBeat = end;
123 sortRegions(curve);
124 return true;
125}

128inline std::optional<uint64_t> copy(AutomationCurve& curve, uint64_t id, double deltaBeats) {
129 materialize(curve);
130 const AutomationRegion* found = find(curve, id);
131 if (!found)
132 return std::nullopt;
133 const AutomationRegion src = *found;
134 const double start = src.startBeat + deltaBeats;
135 const double end = src.endBeat + deltaBeats;
136 if (start < -kEps || overlapsAnother(curve, 0, start, end))
137 return std::nullopt;
138
139 std::vector<AutomationPoint> carried;
140 for (const auto& p : curve.points)
141 if (inside(src, p.beat)) {
142 auto c = p;
143 c.beat += deltaBeats;
144 c.sample = 0;
145 c.selected = false;
146 carried.push_back(c);
147 }
148 clearSpan(curve, start, end);
149 for (const auto& p : carried)
150 curve.points.push_back(p);
151 curve.sortPoints();
152 const uint64_t newId = nextId(curve);
153 curve.regions.push_back({newId, start, end});
154 sortRegions(curve);
155 return newId;
156}

161inline std::optional<uint64_t> create(AutomationCurve& curve, double start, double end, float value) {
162 if (start < -kEps || end <= start + kEps)
163 return std::nullopt;
164 materialize(curve); // an implicit clip over all the points becomes a stored one, so the overlap test sees it
165 if (overlapsAnother(curve, 0, start, end))
166 return std::nullopt;
167 clearSpan(curve, start, end);
168 for (const double beat : {start, end}) {
169 AutomationPoint p;
170 p.beat = beat;
171 p.value = value;
172 curve.points.push_back(p);
173 }
174 curve.sortPoints();
175 const uint64_t newId = nextId(curve);
176 curve.regions.push_back({newId, start, end});
177 sortRegions(curve);
178 return newId;
179}

182inline std::optional<uint64_t> split(AutomationCurve& curve, uint64_t id, double beat) {
183 materialize(curve);
184 auto it = std::find_if(curve.regions.begin(), curve.regions.end(), [id](const auto& r) { return r.id == id; });
185 if (it == curve.regions.end() || !(beat > it->startBeat + kEps) || !(beat <= it->endBeat + kEps))
186 return std::nullopt;
187 // The last point strictly before the beat ends the left part; the first at or after it starts the right.
188 const AutomationPoint* left = nullptr;
189 const AutomationPoint* right = nullptr;
190 for (const auto& p : curve.points) {
191 if (!inside(*it, p.beat))
192 continue;
193 if (p.beat < beat - kEps)
194 left = &p;
195 else if (!right)
196 right = &p;
197 }
198 if (!left || !right)
199 return std::nullopt;
200 const double leftEnd = left->beat;
201 const double rightStart = right->beat;
202 const uint64_t newId = nextId(curve);
203 const double oldEnd = it->endBeat;
204 it->endBeat = leftEnd;
205 curve.regions.push_back({newId, rightStart, oldEnd});
206 sortRegions(curve);
207 return newId;
208}

211inline bool remove(AutomationCurve& curve, uint64_t id) {
212 materialize(curve);

222inline void retighten(AutomationCurve& curve) {
223 for (auto& r : curve.regions) {
224 double lo = 0.0, hi = 0.0;
225 bool any = false;
226 for (const auto& p : curve.points)
227 if (inside(r, p.beat)) {
228 lo = any ? std::min(lo, p.beat) : p.beat;
229 hi = any ? std::max(hi, p.beat) : p.beat;
230 any = true;
231 }
232 r.startBeat = lo;
233 r.endBeat = any ? hi : -1.0;
234 }
235 curve.regions.erase(std::remove_if(curve.regions.begin(), curve.regions.end(),
236 [](const AutomationRegion& r) { return r.endBeat < r.startBeat; }),
237 curve.regions.end());
238}
