94const DBusApi* dbusApi() {
95 static DBusApi api;
96 static bool ok = false;
97 static std::once_flag once;
98 std::call_once(once, []() {

101 ok = resolve(lib, "dbus_error_init", api.error_init) && resolve(lib, "dbus_error_free", api.error_free) &&
102 resolve(lib, "dbus_error_is_set", api.error_is_set) &&
103 resolve(lib, "dbus_threads_init_default", api.threads_init_default) &&
104 resolve(lib, "dbus_bus_get_private", api.bus_get_private) &&
105 resolve(lib, "dbus_connection_set_exit_on_disconnect", api.connection_set_exit_on_disconnect) &&
106 resolve(lib, "dbus_connection_close", api.connection_close) &&
107 resolve(lib, "dbus_connection_unref", api.connection_unref) &&
108 resolve(lib, "dbus_bus_get_unique_name", api.bus_get_unique_name) &&

133// One private session-bus connection per request: SDL shares the default
134// connection for its own use, and a private one lets us close it cleanly.
135class Connection {
136public:
137 explicit Connection(const DBusApi& api) : m_api(api) {
138 DBusError err{};
139 m_api.error_init(&err);
140 m_conn = m_api.bus_get_private(kBusSession, &err);
141 if (m_api.error_is_set(&err)) m_api.error_free(&err);
142 // libdbus's default is to _exit() the process when the bus goes away.
143 if (m_conn) m_api.connection_set_exit_on_disconnect(m_conn, 0);
144 }
145 ~Connection() {
146 if (!m_conn) return;
147 m_api.connection_close(m_conn);
148 m_api.connection_unref(m_conn);
149 }
150 Connection(const Connection&) = delete;
151 Connection& operator=(const Connection&) = delete;
152 DBusConnection* get() const { return m_conn; }
153
154private:
155 const DBusApi& m_api;
156 DBusConnection* m_conn = nullptr;
157};

171 OptionsWriter(const DBusApi& api, DBusMessageIter* parent) : m_api(api), m_parent(parent) {

176 void string(const char* key, const std::string& value) {
177 entry(key, "s", [&](DBusMessageIter* v) {
178 const char* s = value.c_str();
179 m_api.message_iter_append_basic(v, kTypeString, &s);
180 });
181 }
182 void boolean(const char* key, bool value) {
183 entry(key, "b", [&](DBusMessageIter* v) {
184 const dbus_bool_t b = value ? 1 : 0;
185 m_api.message_iter_append_basic(v, kTypeBoolean, &b);
186 });
187 }
188 // Portal paths travel as NUL-terminated byte arrays.
189 void bytePath(const char* key, const std::string& path) {
190 entry(key, "ay", [&](DBusMessageIter* v) {
191 DBusMessageIter arr{};
192 m_api.message_iter_open_container(v, kTypeArray, "y", &arr);
193 const char* data = path.c_str();
194 m_api.message_iter_append_fixed_array(&arr, kTypeByte, &data, static_cast<int>(path.size() + 1));
195 m_api.message_iter_close_container(v, &arr);
196 });
197 }
198 void filters(const std::vector<Filter>& filters) {
199 if (filters.empty()) return;
200 entry("filters", "a(sa(us))", [&](DBusMessageIter* v) {
201 DBusMessageIter list{};
202 m_api.message_iter_open_container(v, kTypeArray, "(sa(us))", &list);
203 for (const auto& f : filters) {
204 DBusMessageIter item{};
205 m_api.message_iter_open_container(&list, kTypeStruct, nullptr, &item);
206 const char* name = f.name.c_str();
207 m_api.message_iter_append_basic(&item, kTypeString, &name);
208 DBusMessageIter pats{};
209 m_api.message_iter_open_container(&item, kTypeArray, "(us)", &pats);
210 for (const auto& p : f.patterns) {
211 DBusMessageIter pat{};
212 m_api.message_iter_open_container(&pats, kTypeStruct, nullptr, &pat);
213 const uint32_t glob = 0;
214 m_api.message_iter_append_basic(&pat, kTypeUint32, &glob);
215 const char* ps = p.c_str();
216 m_api.message_iter_append_basic(&pat, kTypeString, &ps);
217 m_api.message_iter_close_container(&pats, &pat);
218 }
219 m_api.message_iter_close_container(&item, &pats);
220 m_api.message_iter_close_container(&list, &item);
221 }
222 m_api.message_iter_close_container(v, &list);
223 });
224 }

244// FileChooser interface version, or 0 when the portal cannot be reached.
245uint32_t fileChooserVersion(const DBusApi& api, DBusConnection* conn) {
246 MessageRef call{api, api.message_new_method_call(kPortalBus, kPortalPath, "org.freedesktop.DBus.Properties", "Get")};
247 if (!call.msg) return 0;
248 DBusMessageIter args{};
249 api.message_iter_init_append(call.msg, &args);
250 const char* iface = kFileChooser;
251 const char* prop = "version";
252 api.message_iter_append_basic(&args, kTypeString, &iface);
253 api.message_iter_append_basic(&args, kTypeString, &prop);
254
255 DBusError err{};
256 api.error_init(&err);
257 // Generous: the portal is bus-activated and may be starting up.
258 MessageRef reply{api, api.connection_send_with_reply_and_block(conn, call.msg, 25000, &err)};
259 if (api.error_is_set(&err)) {
260 api.error_free(&err);
261 return 0;
262 }
263 if (!reply.msg) return 0;
264
265 DBusMessageIter it{};
266 DBusMessageIter variant{};
267 if (!api.message_iter_init(reply.msg, &it) || api.message_iter_get_arg_type(&it) != kTypeVariant) return 0;
268 api.message_iter_recurse(&it, &variant);
269 if (api.message_iter_get_arg_type(&variant) != kTypeUint32) return 0;
270 uint32_t version = 0;
271 api.message_iter_get_basic(&variant, &version);
272 return version;
273}

275// First entry of results["uris"], or "" if absent.
276std::string firstUri(const DBusApi& api, DBusMessageIter* results) {
277 DBusMessageIter dict{};
278 api.message_iter_recurse(results, &dict);
279 while (api.message_iter_get_arg_type(&dict) == kTypeDictEntry) {
280 DBusMessageIter entry{};
281 api.message_iter_recurse(&dict, &entry);
282 const char* key = nullptr;
283 if (api.message_iter_get_arg_type(&entry) == kTypeString) api.message_iter_get_basic(&entry, &key);
284 if (key && std::string(key) == "uris" && api.message_iter_next(&entry) &&
285 api.message_iter_get_arg_type(&entry) == kTypeVariant) {
286 DBusMessageIter variant{};
287 api.message_iter_recurse(&entry, &variant);
288 if (api.message_iter_get_arg_type(&variant) == kTypeArray) {
289 DBusMessageIter uris{};
290 api.message_iter_recurse(&variant, &uris);
291 if (api.message_iter_get_arg_type(&uris) == kTypeString) {
292 const char* uri = nullptr;
293 api.message_iter_get_basic(&uris, &uri);
294 return uri ? std::string(uri) : std::string();
295 }
296 }
297 }
298 api.message_iter_next(&dict);
299 }
300 return {};
301}

308// Portal request object path for (unique name, token): ":1.42" -> "1_42".
309std::string requestPath(const std::string& uniqueName, const std::string& token) {
310 std::string sender = uniqueName;
311 if (!sender.empty() && sender[0] == ':') sender.erase(0, 1);
312 for (char& c : sender) {
313 if (c == '.') c = '_';
314 }
315 return std::string(kPortalPath) + "/request/" + sender + "/" + token;
316}

328std::optional<std::string> run(const Request& req) {
329 const DBusApi* api = dbusApi();
330 if (!api) return std::nullopt;
331
332 Connection conn(*api);
333 if (!conn.get()) return std::nullopt;
334
335 const uint32_t version = fileChooserVersion(*api, conn.get());
336 if (version == 0) return std::nullopt;
337 if (req.directory && version < 3) return std::nullopt; // "directory" arrived in v3
338
339 const char* unique = api->bus_get_unique_name(conn.get());
340 if (!unique) return std::nullopt;
341 const std::string token = requestToken();
342 const std::string expectedPath = requestPath(unique, token);
343
344 // Subscribe BEFORE the call so a fast answer cannot be missed. The private
345 // connection only carries this request's traffic.
346 {
347 DBusError err{};
348 api->error_init(&err);
349 const std::string rule = std::string("type='signal',sender='") + kPortalBus + "',interface='" + kRequest +
350 "',member='Response'";
351 api->bus_add_match(conn.get(), rule.c_str(), &err);
352 if (api->error_is_set(&err)) {
353 api->error_free(&err);
354 return std::nullopt;
355 }
356 }
357
358 MessageRef call{*api, api->message_new_method_call(kPortalBus, kPortalPath, kFileChooser,
359 req.method == Method::Save ? "SaveFile" : "OpenFile")};
360 if (!call.msg) return std::nullopt;
361 {
362 DBusMessageIter args{};
363 api->message_iter_init_append(call.msg, &args);
364 const char* parentWindow = "";
365 const char* title = req.title.c_str();
366 api->message_iter_append_basic(&args, kTypeString, &parentWindow);
367 api->message_iter_append_basic(&args, kTypeString, &title);
368 OptionsWriter options(*api, &args);
369 options.string("handle_token", token);
370 options.boolean("modal", true);
371 if (req.method == Method::Open) {
372 options.boolean("multiple", false);
373 if (req.directory) options.boolean("directory", true);
374 } else if (!req.suggestedPath.empty()) {
375 const std::filesystem::path suggested(req.suggestedPath);
376 if (suggested.has_filename()) options.string("current_name", suggested.filename().string());
377 std::error_code ec;
378 if (suggested.has_parent_path() && std::filesystem::is_directory(suggested.parent_path(), ec)) {
379 options.bytePath("current_folder", suggested.parent_path().string());
380 }
381 }
382 options.filters(req.filters);
383 }
384
385 std::string handlePath;
386 {
387 DBusError err{};
388 api->error_init(&err);
389 MessageRef reply{*api, api->connection_send_with_reply_and_block(conn.get(), call.msg, 25000, &err)};
390 if (api->error_is_set(&err)) {
391 std::cerr << "[Portal] FileChooser call failed: " << (err.message ? err.message : "?") << std::endl;
392 api->error_free(&err);
393 return std::nullopt;
394 }
395 if (!reply.msg) return std::nullopt;
396 DBusMessageIter it{};
397 if (api->message_iter_init(reply.msg, &it) && api->message_iter_get_arg_type(&it) == 'o') {
398 const char* handle = nullptr;
399 api->message_iter_get_basic(&it, &handle);
400 if (handle) handlePath = handle;
401 }
402 // A reply with no request object confirms no dialog, and a Response on a
403 // path we cannot predict would never match: fall back instead of waiting.
404 if (handlePath.empty()) return std::nullopt;
405 }
406
407 // From here the dialog is on screen: every outcome is final (a result or a
408 // cancel), never "unavailable" — falling back now would stack a second picker.
409 while (true) {
410 if (!api->connection_read_write(conn.get(), 250)) {
411 return std::string(); // bus went away mid-dialog
412 }
413 while (DBusMessage* raw = api->connection_pop_message(conn.get())) {
414 MessageRef msg{*api, raw};
415 if (!api->message_is_signal(msg.msg, kRequest, "Response")) continue;
416 const char* path = api->message_get_path(msg.msg);
417 // Pre-0.9 portals ignored handle_token and returned their own path.
418 if (!path || (expectedPath != path && handlePath != path)) continue;
419
420 DBusMessageIter it{};
421 if (!api->message_iter_init(msg.msg, &it) || api->message_iter_get_arg_type(&it) != kTypeUint32) {
422 return std::string();
423 }
424 uint32_t response = 2;
425 api->message_iter_get_basic(&it, &response);
426 if (response != 0) return std::string(); // 1 = cancelled, 2 = ended otherwise
427 if (!api->message_iter_next(&it) || api->message_iter_get_arg_type(&it) != kTypeArray) {
428 return std::string();
429 }
430 return pathFromFileUri(firstUri(*api, &it));
431 }
432 }

435int hexValue(char c) {
436 if (c >= '0' && c <= '9') return c - '0';
437 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
438 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
439 return -1;
440}

444std::string pathFromFileUri(const std::string& uri) {
445 static const std::string kScheme = "file://";
446 if (uri.compare(0, kScheme.size(), kScheme) != 0) return {};
447 std::string rest = uri.substr(kScheme.size());
448 // file:///abs or file://localhost/abs; any other host is not local.
449 if (rest.compare(0, 9, "localhost") == 0) rest.erase(0, 9);
450 if (rest.empty() || rest[0] != '/') return {};
451
452 std::string path;
453 path.reserve(rest.size());
454 for (size_t i = 0; i < rest.size(); ++i) {
455 if (rest[i] == '%' && i + 2 < rest.size()) {
456 const int hi = hexValue(rest[i + 1]);
457 const int lo = hexValue(rest[i + 2]);
458 if (hi >= 0 && lo >= 0) {
459 path.push_back(static_cast<char>(hi * 16 + lo));
460 i += 2;
461 continue;
462 }
463 }
464 path.push_back(rest[i]);
465 }
466 return path;
467}

469std::vector<Filter> parseWin32Filter(const std::string& filter) {
470 std::vector<Filter> out;
471 std::vector<std::string> parts;
472 size_t start = 0;
473 for (size_t i = 0; i < filter.size(); ++i) {
474 if (filter[i] == '\0') {
475 parts.push_back(filter.substr(start, i - start));
476 start = i + 1;
477 }
478 }
479 if (start < filter.size()) parts.push_back(filter.substr(start));
480
481 for (size_t i = 0; i + 1 < parts.size(); i += 2) {
482 if (parts[i].empty()) break;
483 Filter f;
484 f.name = parts[i];
485 size_t p = 0;
486 const std::string& spec = parts[i + 1];
487 while (p <= spec.size()) {
488 const size_t semi = spec.find(';', p);
489 std::string pattern = spec.substr(p, semi == std::string::npos ? std::string::npos : semi - p);
490 if (pattern == "*.*") pattern = "*";
491 if (!pattern.empty()) f.patterns.push_back(pattern);
492 if (semi == std::string::npos) break;
493 p = semi + 1;
494 }
495 if (!f.patterns.empty()) out.push_back(std::move(f));
496 }
497 return out;
498}
