28BounceInPlaceResult bounceClipsInPlace(AudioEngine& engine, TrackManager& trackManager,
29 const std::vector<ClipInstanceID>& clipIds) {
30 auto& playlist = trackManager.getPlaylistModel();
31 if (clipIds.empty())
32 return refuse("Select the audio clips to bounce first.");
33
34 uint32_t channelId = 0;
35 double spanStart = 0.0;
36 double spanEnd = 0.0;
37 bool first = true;
38 for (const auto& id : clipIds) {
39 const ClipInstance* clip = playlist.getClip(id);
40 const PatternSource* pattern = clip ? trackManager.getPatternManager().getPattern(clip->patternId) : nullptr;
41 if (!clip || !pattern || !pattern->isAudio())
42 return refuse("Bounce in Place works on audio clips.");
43 const uint32_t clipChannel = pattern->getMixerChannelId();
44 if (clipChannel == 0)
45 return refuse("This clip goes straight to the master. Route it to a track first, then bounce.");
46 if (!first && clipChannel != channelId)
47 return refuse("The selected clips are on different tracks. Bounce one track at a time.");
48 channelId = clipChannel;
49 spanStart = first ? clip->startBeat : std::min(spanStart, clip->startBeat);
50 spanEnd = first ? clip->endBeat() : std::max(spanEnd, clip->endBeat());
51 first = false;
52 }
53 if (!(spanEnd > spanStart))
54 return refuse("The selected clips have no length to bounce.");
55 if (!trackManager.getChannelById(channelId))
56 return refuse("The clips' track no longer exists.");
57
58 // Anything else this track plays across the span would be baked into the bounce and then
59 // play again from its own source.
60 const auto isSelected = [&clipIds](const ClipInstanceID& id) {
61 return std::find(clipIds.begin(), clipIds.end(), id) != clipIds.end();
62 };
63 for (size_t l = 0; l < playlist.getLaneCount(); ++l) {
64 const PlaylistLane* lane = playlist.getLane(playlist.getLaneId(l));
65 if (!lane)
66 continue;
67 for (const auto& other : lane->clips) {
68 if (isSelected(other.id) || !overlaps(spanStart, spanEnd, other.startBeat, other.endBeat()))
69 continue;
70 const PatternSource* otherPattern = trackManager.getPatternManager().getPattern(other.patternId);
71 if (otherPattern && otherPattern->getMixerChannelId() == channelId)
72 return refuse("Other clips on this track play during the same stretch. Select them too, or "
73 "move them away, so nothing plays twice.");
74 }
75 }
76 const auto& units = trackManager.getUnitManager();
77 for (const UnitID unitId : units.getAllUnitIDs()) {
78 const auto* unit = units.getUnit(unitId);
79 if (unit && unit->targetMixerChannelId == channelId)
80 return refuse("This track has an instrument. Bounce for instrument tracks is not built yet.");
81 }
82
83 AudioExporter::Config config;
84 config.scope = AudioExporter::RenderScope::FullSong;
85 config.startBeat = spanStart;
86 config.endBeat = spanEnd;
87 config.sampleRate = static_cast<uint32_t>(std::max(1.0, playlist.getProjectSampleRate()));
88 config.numChannels = 2;
89 config.tailSeconds = 0.0; // the unit is the clips' span (owner ruling 2026-10-11)
90 config.captureMixerChannelId = channelId;
91
92 AudioExporter exporter(engine, trackManager);
93 AudioExporter::RenderedAudio rendered;
94 const auto result = exporter.renderToBuffer(config, rendered);
95 if (!result.ok() || rendered.frames == 0)
96 return refuse(result.errorMessage.empty() ? "The render produced no audio." : result.errorMessage);
97
98 auto buffer = std::make_shared<AudioBufferData>();
99 buffer->interleavedData = std::move(rendered.interleaved);
100 buffer->sampleRate = rendered.sampleRate;
101 buffer->numChannels = rendered.numChannels;
102 buffer->numFrames = rendered.frames;
103
104 auto command = std::make_shared<BounceInPlaceCommand>(trackManager, clipIds, channelId, std::move(buffer),
105 spanStart, spanEnd - spanStart);
106 trackManager.getCommandHistory().pushAndExecute(command);
107 if (!command->isUndoable())
108 return refuse("The bounced audio could not be saved.");
109
110 BounceInPlaceResult done;
111 done.ok = true;
112 done.clip = command->bouncedClipId();
113 return done;
114}
