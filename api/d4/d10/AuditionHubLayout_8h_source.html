101 static int columns(float width) {
102 const float usable = width - kPad * 2.0f;
103 return std::max(1, static_cast<int>(std::floor((usable + kGap) / (kMinTile + kGap))));
104 }
105 static float tileSize(float width) {
106 const int cols = columns(width);
107 const float usable = width - kPad * 2.0f;
108 return std::min(kTargetTile * 1.25f, (usable - kGap * static_cast<float>(cols - 1)) / static_cast<float>(cols));
109 }

118 static std::vector<Shelf> shelves(const AestraUI::NUIRect& content, float top, const std::vector<size_t>& counts) {
119 std::vector<Shelf> out;
120 const int cols = columns(content.width);
121 const float ts = tileSize(content.width);
122 float y = top;
123 for (const size_t n : counts) {
124 Shelf s;
125 s.header = {content.x + kPad, y, content.width - kPad * 2.0f, kShelfHeaderH};
126 y += kShelfHeaderH;
127 for (size_t i = 0; i < n; ++i) {
128 const int row = static_cast<int>(i) / cols, col = static_cast<int>(i) % cols;
129 s.tiles.push_back({content.x + kPad + static_cast<float>(col) * (ts + kGap), y + static_cast<float>(row) * (ts + kCaptionH + kGap * 0.5f), ts, ts + kCaptionH});
130 }
131 const int rows = n ? (static_cast<int>(n) + cols - 1) / cols : 0;
132 y += static_cast<float>(rows) * (ts + kCaptionH + kGap * 0.5f) + 24.0f;
133 out.push_back(std::move(s));
134 }
135 return out;
136 }

138 static float contentHeight(const std::vector<Shelf>& shelves, float top) {
139 if (shelves.empty()) return top;
140 const auto& s = shelves.back();
141 const float bottom = s.tiles.empty() ? s.header.bottom() : s.tiles.back().bottom();
142 return bottom + 32.0f - top;
143 }
