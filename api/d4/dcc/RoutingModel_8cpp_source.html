22RoutingModel RoutingModel::capture(const TrackManager& manager) {
23 RoutingModel model;
24 const size_t count = manager.getChannelCount();
25 model.m_channels.reserve(count);
26 for (size_t i = 0; i < count; ++i) {
27 const MixerChannel* ch = manager.getChannel(i);
28 if (!ch) continue;
29
30 Channel info;
31 info.id = ch->getChannelId();
32 info.name = ch->getName();
33 info.mainOutputId = toModelSpace(ch->getMainOutputId());
34 model.m_channels.push_back(info);
35
36 Edge main;
37 main.sourceId = info.id;
38 main.targetId = info.mainOutputId;
39 model.m_edges.push_back(main);
40
41 for (const AudioRoute& send : ch->getSends()) {
42 Edge e;
43 e.sourceId = info.id;
44 e.targetId = toModelSpace(send.targetChannelId);
45 e.kind = EdgeKind::Send;
46 e.sendId = send.sendId;
47 e.gain = send.gain;
48 e.postFader = send.postFader;
49 e.muted = send.mute;
50 e.sidechainOnly = send.sidechainOnly;
51 model.m_edges.push_back(e);
52 }
53 }
54 return model;
55}

65std::vector<RoutingModel::Edge> RoutingModel::feeds(uint32_t id) const {
66 std::vector<Edge> out;
67 for (const Edge& e : m_edges) {
68 if (e.targetId == id) out.push_back(e);
69 }
70 std::stable_partition(out.begin(), out.end(), [](const Edge& e) { return e.kind == EdgeKind::Main; });
71 return out;
72}

91std::vector<RoutingModel::OutputGroup> RoutingModel::groupByOutput() const {
92 std::vector<OutputGroup> groups;
93 auto groupFor = [&](uint32_t target) -> OutputGroup& {
94 for (OutputGroup& g : groups) {
95 if (g.targetId == target) return g;
96 }
97 OutputGroup g;
98 g.targetId = target;
99 g.targetName = nameOf(target);
100 groups.push_back(std::move(g));
101 return groups.back();
102 };
103 for (const Channel& ch : m_channels) {
104 groupFor(ch.mainOutputId).sourceIds.push_back(ch.id);
105 }
106 std::stable_sort(groups.begin(), groups.end(), [](const OutputGroup& a, const OutputGroup& b) {
107 return (a.targetId == kMaster) > (b.targetId == kMaster);
108 });
109 return groups;
110}
