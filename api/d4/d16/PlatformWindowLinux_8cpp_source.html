12bool getX11WindowInfo(SDL_Window* window, Display*& display, ::Window& x11Window) {
13 display = nullptr;
14 x11Window = 0;
15 if (!window) return false;
16
17 SDL_SysWMinfo info;
18 SDL_VERSION(&info.version);
19 if (!SDL_GetWindowWMInfo(window, &info)) return false;
20 if (info.subsystem != SDL_SYSWM_X11) return false;
21
22 display = info.info.x11.display;
23 x11Window = info.info.x11.window;
24 return (display != nullptr && x11Window != 0);
25}

36bool PlatformWindowLinux::create(const WindowDesc& desc) {
37 // Convert flags
38 Uint32 flags = SDL_WINDOW_OPENGL | SDL_WINDOW_SHOWN | SDL_WINDOW_ALLOW_HIGHDPI;
39 if (desc.resizable)
40 flags |= SDL_WINDOW_RESIZABLE;
41 if (desc.startMaximized)
42 flags |= SDL_WINDOW_MAXIMIZED;
43 if (desc.startFullscreen)
44 flags |= SDL_WINDOW_FULLSCREEN_DESKTOP;
45 if (!desc.decorated)
46 flags |= SDL_WINDOW_BORDERLESS;
47
48 // Position
49 int x = (desc.x == -1) ? SDL_WINDOWPOS_CENTERED : desc.x;
50 int y = (desc.y == -1) ? SDL_WINDOWPOS_CENTERED : desc.y;
51
52 // GL Attributes - Request generic 3.3 Core, can be upgraded by user config if needed
53 SDL_GL_SetAttribute(SDL_GL_CONTEXT_MAJOR_VERSION, 3);
54 SDL_GL_SetAttribute(SDL_GL_CONTEXT_MINOR_VERSION, 3);
55 SDL_GL_SetAttribute(SDL_GL_CONTEXT_PROFILE_MASK, SDL_GL_CONTEXT_PROFILE_CORE);
56 SDL_GL_SetAttribute(SDL_GL_DOUBLEBUFFER, 1);
57 SDL_GL_SetAttribute(SDL_GL_FRAMEBUFFER_SRGB_CAPABLE, 1);
58 SDL_GL_SetAttribute(SDL_GL_MULTISAMPLEBUFFERS, 1);
59 SDL_GL_SetAttribute(SDL_GL_MULTISAMPLESAMPLES, 4);
60
61 m_window = SDL_CreateWindow(desc.title.c_str(), x, y, desc.width, desc.height, flags);
62
63 if (!m_window) {
64 std::cerr << "SDL_CreateWindow Error: " << SDL_GetError() << std::endl;
65 return false;
66 }
67
68 m_isFullscreen = desc.startFullscreen;
69 m_isWindowVisible = true;
70 m_isWindowMapped = true;
71 m_maximized = desc.startMaximized;
72
73 // Seed the restore geometry from what was requested. Without this, a session
74 // that starts maximized and is never un-maximized has no restore bounds to
75 // save, and the maximized size would win by default (#655).
76 m_restoreX = (x == SDL_WINDOWPOS_CENTERED) ? 0 : x;
77 m_restoreY = (y == SDL_WINDOWPOS_CENTERED) ? 0 : y;
78 m_restoreW = desc.width;
79 m_restoreH = desc.height;
80 m_hasRestoreBounds = desc.width > 0 && desc.height > 0;
81
82 // Initial DPI check
83 int dw, dh;
84 SDL_GL_GetDrawableSize(m_window, &dw, &dh);
85 int w, h;
86 SDL_GetWindowSize(m_window, &w, &h);
87 if (w > 0) {
88 m_dpiScale = (float)dw / w;
89 }
90
91 return true;
92}

105bool PlatformWindowLinux::pollEvents() {
106 SDL_Event e;
107 while (SDL_PollEvent(&e)) {
108 switch (e.type) {
109 case SDL_QUIT:
110 if (m_closeCallback) {
111 m_closeCallback();
112 return true;
113 }
114 return false;
115
116 case SDL_WINDOWEVENT:
117 if (e.window.windowID == SDL_GetWindowID(m_window)) {
118 switch (e.window.event) {
119 case SDL_WINDOWEVENT_MAXIMIZED:
120 m_maximized = true;
121 break;
122 case SDL_WINDOWEVENT_MOVED:
123 rememberRestoreBoundsIfNormal();
124 break;
125 case SDL_WINDOWEVENT_RESIZED:
126 case SDL_WINDOWEVENT_SIZE_CHANGED:
127 rememberRestoreBoundsIfNormal();
128 if (m_resizeCallback)
129 m_resizeCallback(e.window.data1, e.window.data2);
130 {
131 // Check DPI change
132 int dw, dh;
133 SDL_GL_GetDrawableSize(m_window, &dw, &dh);
134 float newScale = (float)dw / e.window.data1;
135 if (abs(newScale - m_dpiScale) > 0.01f) {
136 m_dpiScale = newScale;
137 if (m_dpiChangeCallback)
138 m_dpiChangeCallback(m_dpiScale);
139 }
140 }
141 break;
142 case SDL_WINDOWEVENT_CLOSE:
143 if (m_closeCallback)
144 m_closeCallback();
145 break;
146 case SDL_WINDOWEVENT_FOCUS_GAINED:
147 if (m_focusCallback)
148 m_focusCallback(true);
149 break;
150 case SDL_WINDOWEVENT_FOCUS_LOST:
151 if (m_focusCallback)
152 m_focusCallback(false);
153 break;
154 case SDL_WINDOWEVENT_SHOWN:
155 m_isWindowVisible = true;
156 m_isWindowMapped = true;
157 break;
158 case SDL_WINDOWEVENT_HIDDEN:
159 m_isWindowVisible = false;
160 m_isWindowMapped = false;
161 break;
162 case SDL_WINDOWEVENT_MINIMIZED:
163 m_isWindowMapped = false;
164 break;
165 case SDL_WINDOWEVENT_RESTORED:
166 m_maximized = false;
167 if (m_isWindowVisible) {
168 m_isWindowMapped = true;
169 }
170 rememberRestoreBoundsIfNormal();
171 break;
172 }
173 }
174 break;
175
176 case SDL_MOUSEMOTION:
177 if (m_mouseMoveCallback)
178 m_mouseMoveCallback(e.motion.x, e.motion.y);
179 break;
180
181 case SDL_MOUSEBUTTONDOWN:
182 case SDL_MOUSEBUTTONUP:
183 if (m_mouseButtonCallback) {
184 MouseButton btn = MouseButton::Left;
185 if (e.button.button == SDL_BUTTON_RIGHT)
186 btn = MouseButton::Right;
187 else if (e.button.button == SDL_BUTTON_MIDDLE)
188 btn = MouseButton::Middle;
189 m_mouseButtonCallback(btn, e.type == SDL_MOUSEBUTTONDOWN, e.button.x, e.button.y);
190 }
191 break;
192
193 case SDL_MOUSEWHEEL:
194 if (m_mouseWheelCallback) {
195 // Preserve high-resolution touchpad deltas. SDL's integer x/y
196 // fields may round a fractional gesture to zero, which leaves
197 // drag scrollbars working while two-finger scrolling appears dead.
198#if SDL_VERSION_ATLEAST(2, 0, 18)
199 const float verticalDelta = e.wheel.preciseY;
200 const float horizontalDelta = e.wheel.preciseX;
201#else
202 const float verticalDelta = static_cast<float>(e.wheel.y);
203 const float horizontalDelta = static_cast<float>(e.wheel.x);
204#endif
205 float delta = verticalDelta;
206 // Use magnitude comparison to detect horizontal vs vertical gestures
207 // This handles touchpads that may report both axes for the same gesture
208 const bool horizontalGesture = (std::abs(horizontalDelta) > std::abs(verticalDelta));
209 if (horizontalGesture) {
210 // SDL wheel.x > 0 means scroll right, map to negative delta so consumers
211 // that do target -= wheelDelta move view content to the right.
212 delta = -horizontalDelta;
213 m_syntheticShiftForHorizontalWheel = true;
214 }
215 if (e.wheel.direction == SDL_MOUSEWHEEL_FLIPPED)
216 delta = -delta;
217 m_mouseWheelCallback(delta);
218 m_syntheticShiftForHorizontalWheel = false;
219 }
220 break;
221
222 case SDL_KEYDOWN:
223 case SDL_KEYUP:
224 if (m_keyCallback) {
225 KeyCode key = translateKey(e.key.keysym.sym);
226 KeyModifiers mods = getModifiers(e.key.keysym.mod);
227 m_keyCallback(key, e.type == SDL_KEYDOWN, mods);
228 }
229 break;
230
231 case SDL_TEXTINPUT:
232 if (m_charCallback) {
233 // Primitive UTF-8 decoding for single codepoint (simplified)
234 // In real app, might need full UTF8 decoder, but SDL gives char array
235 // For now pass the first byte if ASCII, or handle full utf8
236 // Aestras might expect unicode. e.text.text is char[32] utf8
237 // We'll just cast first byte for now if we don't have a decoder handy,
238 // or implement a quick one.
239 unsigned char* p = (unsigned char*)e.text.text;
240 if (*p < 0x80)
241 m_charCallback(*p);
242 // TODO: Proper UTF8 decoding
243 }
244 break;
245 }
246 }
247 return true;
248}

347bool PlatformWindowLinux::isMaximized() const {
348 if (!m_window)
349 return false;
350 // Union of the tracked state and the SDL flag. The flag alone missed
351 // WM-initiated maximize (#655): a window with _NET_WM_STATE_MAXIMIZED_VERT
352 // and _HORZ set reported false, so shutdown persisted "not maximized"
353 // together with the maximized dimensions.
354 if (m_maximized)
355 return true;
356 Uint32 flags = SDL_GetWindowFlags(m_window);
357 return (flags & SDL_WINDOW_MAXIMIZED) != 0;
358}

370void PlatformWindowLinux::rememberRestoreBoundsIfNormal() {
371 if (!m_window || isMaximized() || m_isFullscreen)
372 return;
373 int w = 0, h = 0, x = 0, y = 0;
374 SDL_GetWindowSize(m_window, &w, &h);
375 SDL_GetWindowPosition(m_window, &x, &y);
376 if (w <= 0 || h <= 0)
377 return;
378 m_restoreX = x;
379 m_restoreY = y;
380 m_restoreW = w;
381 m_restoreH = h;
382 m_hasRestoreBounds = true;
383}

508KeyCode PlatformWindowLinux::translateKey(SDL_Keycode key) {
509 // Map SDL keys to Aestra KeyCode
510 if (key >= 'a' && key <= 'z')
511 return (KeyCode)((int)KeyCode::A + (key - 'a'));
512 if (key >= '0' && key <= '9')
513 return (KeyCode)((int)KeyCode::Num0 + (key - '0'));
514
515 switch (key) {
516 case SDLK_ESCAPE:
517 return KeyCode::Escape;
518 case SDLK_TAB:
519 return KeyCode::Tab;
520 case SDLK_CAPSLOCK:
521 return KeyCode::CapsLock;
522 case SDLK_SPACE:
523 return KeyCode::Space;
524 case SDLK_RETURN:
525 return KeyCode::Enter;
526 case SDLK_BACKSPACE:
527 return KeyCode::Backspace;
528 case SDLK_DELETE:
529 return KeyCode::Delete;
530 case SDLK_INSERT:
531 return KeyCode::Insert;
532 case SDLK_HOME:
533 return KeyCode::Home;
534 case SDLK_END:
535 return KeyCode::End;
536 case SDLK_PAGEUP:
537 return KeyCode::PageUp;
538 case SDLK_PAGEDOWN:
539 return KeyCode::PageDown;
540 case SDLK_UP:
541 return KeyCode::Up;
542 case SDLK_DOWN:
543 return KeyCode::Down;
544 case SDLK_LEFT:
545 return KeyCode::Left;
546 case SDLK_RIGHT:
547 return KeyCode::Right;
548 case SDLK_LSHIFT:
549 case SDLK_RSHIFT:
550 return KeyCode::Shift;
551 case SDLK_LCTRL:
552 case SDLK_RCTRL:
553 return KeyCode::Control;
554 case SDLK_LALT:
555 case SDLK_RALT:
556 return KeyCode::Alt;
557 case SDLK_a: case SDLK_b: case SDLK_c: case SDLK_d: case SDLK_e:
558 case SDLK_f: case SDLK_g: case SDLK_h: case SDLK_i: case SDLK_j:
559 case SDLK_k: case SDLK_l: case SDLK_m: case SDLK_n: case SDLK_o:
560 case SDLK_p: case SDLK_q: case SDLK_r: case SDLK_s: case SDLK_t:
561 case SDLK_u: case SDLK_v: case SDLK_w: case SDLK_x: case SDLK_y:
562 case SDLK_z:
563 return static_cast<KeyCode>(static_cast<int>(KeyCode::A) + (key - SDLK_a));
564 case SDLK_0: case SDLK_1: case SDLK_2: case SDLK_3: case SDLK_4:
565 case SDLK_5: case SDLK_6: case SDLK_7: case SDLK_8: case SDLK_9:
566 return static_cast<KeyCode>(static_cast<int>(KeyCode::Num0) + (key - SDLK_0));
567 case SDLK_F1: case SDLK_F2: case SDLK_F3: case SDLK_F4:
568 case SDLK_F5: case SDLK_F6: case SDLK_F7: case SDLK_F8:
569 case SDLK_F9: case SDLK_F10: case SDLK_F11: case SDLK_F12:
570 return static_cast<KeyCode>(static_cast<int>(KeyCode::F1) + (key - SDLK_F1));
571 default:
572 return KeyCode::Unknown;
573 }
574}
