32NUIColor UIMixerHeader::colorFromARGB(uint32_t argb)
33{
34 const float a = ((argb >> 24) & 0xFF) / 255.0f;
35 const float r = ((argb >> 16) & 0xFF) / 255.0f;
36 const float g = ((argb >> 8) & 0xFF) / 255.0f;
37 const float b = (argb & 0xFF) / 255.0f;
38 return {r, g, b, a};
39}

83void UIMixerHeader::onRender(NUIRenderer& renderer)
84{
85 auto bounds = getBounds();
86
87 // Safety check for invalid bounds
88 if (bounds.width <= 0 || bounds.height <= 0) return;
89
90 // Selection belongs to the strip, not this label. Keeping it out of the
91 // header prevents the selected channel from acquiring a second card and a
92 // second border inside the selected strip.
93
94 // Track colour is identity, not surface decoration. A narrow marker gives
95 // the eye a fast way to scan channels without turning every nameplate into
96 // a coloured button.
97 constexpr float TOP_BAR_H = 3.0f;
98
99 NUIRect topBar{
100 std::floor(bounds.x),
101 std::floor(bounds.y),
102 std::ceil(bounds.width),
103 TOP_BAR_H
104 };
105
106 NUIColor trackColor = (m_isMaster && m_trackColorArgb == 0)
107 ? NUIThemeManager::getInstance().getColor("primary")
108 : colorFromARGB(m_trackColorArgb);
109 trackColor = trackColor.withAlpha(1.0f);
110
111 renderer.fillRect(topBar, trackColor);
112
113 // Text area (below the top bar)
114 NUIRect textRect{
115 bounds.x + PAD_X,
116 bounds.y + TOP_BAR_H + 3.0f,
117 bounds.width - (PAD_X * 2),
118 bounds.height - (TOP_BAR_H + 3.0f)
119 };
120
121 const float nameFont = m_isMaster ? 12.0f : 11.5f;
122 const float routeFont = 8.5f;
123
124 if (m_isMaster) {
125 // Width and placement already identify the master strip. Avoid a
126 // redundant MAIN BUS / MASTER / Output text stack.
127 constexpr float MASTER_NAME_H = 15.0f;
128 constexpr float MASTER_ROUTE_H = 10.0f;
129 constexpr float STACK_GAP = 1.0f;
130 const float stackH = MASTER_NAME_H +
131 (m_route.empty() ? 0.0f : STACK_GAP + MASTER_ROUTE_H);
132 const float stackY = textRect.y + std::max(0.0f, (textRect.height - stackH) * 0.5f);
133 renderer.drawTextCentered(m_name,
134 {textRect.x, stackY, textRect.width, MASTER_NAME_H},
135 nameFont,
136 m_text);
137 if (!m_route.empty()) {
138 NUIRect routeRect{textRect.x,
139 stackY + MASTER_NAME_H + STACK_GAP,
140 textRect.width, MASTER_ROUTE_H};
141 renderer.drawTextCentered(m_route, routeRect, routeFont,
142 m_textSecondary.withAlpha(0.62f));
143 }
144 return;
145 }
146
147 constexpr float TRACK_NAME_H = 13.0f;
148 constexpr float TRACK_ROUTE_H = 9.0f;
149 constexpr float TRACK_GAP = 1.5f;
150 const float stackH = m_route.empty()
151 ? TRACK_NAME_H
152 : (TRACK_NAME_H + TRACK_GAP + TRACK_ROUTE_H);
153 const float stackY = textRect.y + std::max(0.0f, (textRect.height - stackH) * 0.5f) - 0.5f;
154
155 NUIRect nameRect{textRect.x, stackY, textRect.width, TRACK_NAME_H};
156 renderer.drawTextCentered(m_name, nameRect, nameFont, m_text);
157
158 if (!m_route.empty()) {
159 NUIRect routeRect{textRect.x, stackY + TRACK_NAME_H + TRACK_GAP, textRect.width, TRACK_ROUTE_H};
160 renderer.drawTextCentered(m_route,
161 routeRect,
162 routeFont,
163 m_textSecondary.withAlpha(0.64f));
164 }
165}

167bool UIMixerHeader::onMouseEvent(const NUIMouseEvent& event)
168{
169 if (!isVisible() || !isEnabled()) return false;
170 if (!getBounds().contains(event.position)) return false;
171
172 if (event.pressed && event.button == NUIMouseButton::Right) {
173 if (!m_colorMenu) {
174 m_colorMenu = std::make_shared<NUIContextMenu>();
175 m_colorMenu->setCloseOnSelection(true);
176 }
177 m_colorMenu->clear();
178
179 for (int i = 0; i < PALETTE_SIZE; ++i) {
180 const bool selected = (i == m_colorIndex);
181 m_colorMenu->addRadioItem(PALETTE_NAMES[i], "color", selected, [this, i]() {
182 m_colorIndex = i;
183 m_trackColorArgb = paletteIndexToARGB(i);
184 repaint();
185 if (onColorChanged) onColorChanged(i);
186 });
187 }
188
189 NUIComponent* root = this;
190 while (root->getParent()) root = root->getParent();
191 root->addChild(m_colorMenu);
192 m_colorMenu->showAt(event.position);
193 root->repaint();
194 return true;
195 }
196
197 return false;
198}
