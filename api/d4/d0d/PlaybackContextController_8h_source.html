29struct PlaybackMirrors {
30 bool enginePattern = false;
31 bool trackManagerPattern = false;
32 bool loopOverride = false;
33 bool uiPattern = false;

36 bool anyPatternMirror() const { return enginePattern || trackManagerPattern || loopOverride || uiPattern; }
37 bool operator==(const PlaybackMirrors& o) const {
38 return enginePattern == o.enginePattern && trackManagerPattern == o.trackManagerPattern &&
39 loopOverride == o.loopOverride && uiPattern == o.uiPattern && audition == o.audition;
40 }
41 bool operator!=(const PlaybackMirrors& o) const { return !(*this == o); }

123 void startArsenalPlayback(PatternID pattern, double patternLengthBeats, double startSeconds = -1.0) {
124 // Published before play() pushes its transport command, so both land in one drain.
125 // A piano-roll zone on this pattern shortens the loop to the zone (TrackManager::arsenalLoopLengthBeats).
126 m_arsenalPattern = pattern;
127 const double loopBeats = m_trackManager.arsenalLoopLengthBeats(pattern, patternLengthBeats);
128 publishEngineContext(true, loopBeats, m_engineAudition);
129 m_trackManager.setPatternLoopOverride(0.0, loopBeats, true);
130 m_trackManager.playPatternInArsenal(pattern, startSeconds, /*useLoopZone=*/true);
131 m_context = PlaybackContext::Arsenal;
132 }

143 void switchArsenalPattern(PatternID pattern, double patternLengthBeats) {
144 if (!pattern.isValid() || pattern == m_arsenalPattern) {
145 return;
146 }
147 const bool playing = m_context == PlaybackContext::Arsenal && m_trackManager.isPlaying() &&
148 m_trackManager.isPatternMode() && m_arsenalPattern.isValid();
149 m_arsenalPattern = pattern;
150 if (!playing) {
151 return;
152 }
153 const double loopBeats = m_trackManager.arsenalLoopLengthBeats(pattern, patternLengthBeats);
154 const double bpm = std::max(1.0, m_trackManager.getTimelineClock().getCurrentTempo());
155 double beat = std::fmod(std::max(0.0, m_trackManager.getPosition()) * bpm / 60.0, std::max(1e-6, loopBeats));
156 if (m_trackManager.activeArsenalLoopZone(pattern)) {
157 beat = 0.0;
158 }
159 publishEngineContext(true, loopBeats, m_engineAudition);
160 m_trackManager.setPatternLoopOverride(0.0, loopBeats, true);
161 m_trackManager.playPatternInArsenal(pattern, beat * 60.0 / bpm, /*useLoopZone=*/true);
162 }

205 void setArsenalLoopZone(PatternID pattern, std::optional<TrackManager::ArsenalLoopZone> zone,
206 double patternLengthBeats) {
207 if (zone) {
208 zone->pattern = pattern;
209 }
210 m_trackManager.setArsenalLoopZone(zone);
211 if (!pattern.isValid()) {
212 return;
213 }
214 if (pattern == m_arsenalPattern) {
215 reapplyArsenalLoop(patternLengthBeats);
216 return;
217 }
218 // Another pattern is (or was last) Arsenal's. A zone drawn on this one becomes Arsenal's
219 // only when nothing is playing: never reschedule, or publish this pattern's loop length
220 // over, a different pattern that is playing. Clearing another pattern's zone changes
221 // nothing that is playing, so it never takes the Arsenal pattern over either (#970).
222 const bool otherPlaying = m_trackManager.isPlaying() && m_trackManager.isPatternMode() &&
223 m_arsenalPattern.isValid();
224 if (zone && !otherPlaying) {
225 m_arsenalPattern = pattern;
226 reapplyArsenalLoop(patternLengthBeats);
227 }
228 }

294 setUi(true);
295 publishEngineContext(m_enginePattern, m_engineLength, false);
296 m_trackManager.preparePatternForArsenal(pattern);
297 // Clip preview intentionally starts from the top. Every other play path resumes from the cued position.

300 // a ClipPreview context would outlive it and gate off Arsenal loop-length updates,

335 void reapplyArsenalLoop(double patternLengthBeats, bool keepPatternPosition = false, bool zoneAllowed = true) {
336 if (m_context != PlaybackContext::Arsenal) {
337 return;
338 }
339 const auto wanted = zoneAllowed ? m_trackManager.activeArsenalLoopZone(m_arsenalPattern)
340 : std::optional<TrackManager::ArsenalLoopZone>{};
341 const double loopBeats = wanted ? wanted->endBeat - wanted->startBeat : patternLengthBeats;
342 publishEngineContext(true, loopBeats, m_engineAudition);
343 m_trackManager.setPatternLoopOverride(0.0, loopBeats, true);
344 if (!m_trackManager.isPlaying() || !m_trackManager.isPatternMode() || !m_arsenalPattern.isValid()) {
345 return;
346 }
347 const auto scheduled = m_trackManager.scheduledArsenalLoopZone();
348 const bool same = scheduled.has_value() == wanted.has_value() &&
349 (!scheduled || (scheduled->pattern == wanted->pattern &&
350 scheduled->startBeat == wanted->startBeat &&
351 scheduled->endBeat == wanted->endBeat));
352 if (same) {
353 return;
354 }
355 double restartSeconds = 0.0;
356 if (keepPatternPosition) {
357 const double bpm = std::max(1.0, m_trackManager.getTimelineClock().getCurrentTempo());
358 const double oldStart = scheduled ? scheduled->startBeat : 0.0;
359 const double oldLength =
360 std::max(1e-6, scheduled ? scheduled->endBeat - scheduled->startBeat : patternLengthBeats);
361 double transportBeat = std::fmod(std::max(0.0, m_trackManager.getPosition()) * bpm / 60.0, oldLength);
362 const double patternBeat = transportBeat + oldStart;
363 const double newStart = wanted ? wanted->startBeat : 0.0;
364 transportBeat = patternBeat - newStart;
365 if (transportBeat < 0.0 || transportBeat >= loopBeats) {
366 transportBeat = 0.0; // outside the new loop: start at its top
367 }
368 restartSeconds = transportBeat * 60.0 / bpm;
369 }
370 m_trackManager.playPatternInArsenal(m_arsenalPattern, restartSeconds, zoneAllowed);
371 }

373 // The engine's neutral pattern length when pattern mode is off, as every call site passed.
374 static constexpr double kTimelineLengthBeats = 4.0;
375
376 // One request carries the whole engine-side context (pattern mode, loop length, audition),
377 // so the audio thread applies it atomically in queue order (PR 3). The requested values are
378 // kept here because the engine's own flags only change when the audio thread applies them.
379 void publishEngineContext(bool pattern, double lengthBeats, bool audition) {
380 m_enginePattern = pattern;
381 m_engineLength = lengthBeats;
382 m_engineAudition = audition;
383 if (m_engine) {
384 m_engine->requestPlaybackContext(pattern, lengthBeats, audition);
385 }
386 }
387 void setUi(bool patternMode) {
388 m_uiPattern = patternMode;
389 if (m_uiMirror) {
390 m_uiMirror(patternMode);
391 }
392 }

395 AudioEngine* m_engine = nullptr;
396 std::function<void(bool)> m_uiMirror;
397 bool m_uiPattern = false;
398 bool m_enginePattern = false;
399 double m_engineLength = kTimelineLengthBeats;
400 bool m_engineAudition = false;
401 // The pattern Arsenal plays (last started, or the piano roll's when it set a zone with nothing
402 // else playing). Loop-zone lookups use it, so a zone drawn on one pattern never shapes another.
403 PatternID m_arsenalPattern{};
404 PlaybackContext m_context = PlaybackContext::Timeline;
