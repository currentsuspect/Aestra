85inline std::vector<MidiClipPlaybackInstance>
86selectScheduledTimelineInstances(const std::vector<MidiClipPlaybackInstance>& instances, double playStartBeat,
87 bool* truncated = nullptr) {
88 std::vector<MidiClipPlaybackInstance> selected;
89 if (truncated) {
90 *truncated = false;
91 }
92 for (const auto& instance : instances) {
93 if (!instance.isValid() || instance.endBeat() <= playStartBeat) {
94 continue;
95 }
96 if (selected.size() >= kMaxScheduledTimelineMidiInstances) {

98 *truncated = true;
99 }
100 break;
101 }
102 selected.push_back(instance);
103 }

118inline std::optional<double> patternLocalBeatAt(const std::vector<MidiClipPlaybackInstance>& instances,
119 PatternID pattern, double arrangementBeat) {
120 for (const auto& instance : instances) {
121 if (instance.patternId == pattern && instance.isValid() && arrangementBeat >= instance.startBeat &&
122 arrangementBeat < instance.endBeat()) {
123 const double local = arrangementBeat - instance.patternStartBeat();
124 return instance.patternLengthBeats > 0.0 ? std::fmod(local, instance.patternLengthBeats) : local;
125 }
126 }
127 return std::nullopt;

143inline bool prunePatternToRegion(PatternManager* patternManager, PatternID patternId, double regionStart,
144 double regionBeats) {
145 if (!patternManager) {
146 return false;
147 }
148 auto* pattern = patternManager->getPattern(patternId);
149 if (!pattern || !pattern->isMidi() || regionBeats <= 0.0) {
150 return false;
151 }
152 const double regionEnd = regionStart + regionBeats;
153 auto& notes = pattern->getMidiNotes();
154 std::vector<MidiNote> kept;
155 kept.reserve(notes.size());
156 for (auto& note : notes) {
157 const double noteEnd = note.startBeat + note.durationBeats;
158 if (noteEnd <= regionStart || note.startBeat >= regionEnd) {
159 continue;
160 }
161 if (note.startBeat < regionStart) {
162 note.startBeat = regionStart;
163 note.durationBeats = noteEnd - regionStart;
164 }
165 if (note.startBeat + note.durationBeats > regionEnd) {
166 note.durationBeats = regionEnd - note.startBeat;
167 }
168 note.startBeat -= regionStart;
169 kept.push_back(note);
170 }
171 notes = std::move(kept);
172 pattern->lengthBeats = regionBeats;
173 return true;
174}

345 bool setClipEdits(const ClipInstanceID& clipId, const ClipEdits& edits) {
346 {
347 std::unique_lock<std::shared_mutex> lock(m_mutex);
348 auto* clip = getClipInternal(clipId);

353 // #746 canonical invariant: durationSeconds == beatToSeconds(
354 // durationBeats) / effectiveVarispeed. A rate/pitch edit that
355 // skips this leaves the serializer's canonical field stale and

358 // the field is audio-specific.
359 if (isAudioClipUnlocked(*clip)) {

460 void moveClip(const ClipInstanceID& clipId, double newStartBeat,
461 const PlaylistLaneID& newLaneId = PlaylistLaneID()) {
462 std::unique_lock<std::shared_mutex> lock(m_mutex);
463
464 auto* clip = getClipInternal(clipId);
465 if (!clip)
466 return;
467
468 // Store clip data
469 ClipInstance clipCopy = *clip;
470 clipCopy.startBeat = newStartBeat;
471
472 // Check if we need to move to a different lane
473 if (newLaneId.isValid()) {
474 auto oldLaneIt = m_clipLaneMap.find(clipId);
475 if (oldLaneIt != m_clipLaneMap.end() && oldLaneIt->second != newLaneId) {
476 // Remove from old lane
477 auto oldLaneIdxIt = m_laneMap.find(oldLaneIt->second);
478 if (oldLaneIdxIt != m_laneMap.end()) {
479 auto& oldClips = m_lanes[oldLaneIdxIt->second].clips;
480 oldClips.erase(std::remove_if(oldClips.begin(), oldClips.end(),
481 [&clipId](const ClipInstance& c) { return c.id == clipId; }),
482 oldClips.end());
483 }
484
485 // Add to new lane
486 auto newLaneIdxIt = m_laneMap.find(newLaneId);
487 if (newLaneIdxIt != m_laneMap.end()) {
488 m_lanes[newLaneIdxIt->second].clips.push_back(clipCopy);
489 m_clipLaneMap[clipId] = newLaneId;
490 }
491
492 notifyClipChanged(clipId);
493 return;
494 }
495 }
496
497 // Just update position
498 clip->startBeat = newStartBeat;
499 notifyClipChanged(clipId);
500 }

508 ClipInstanceID splitClip(const ClipInstanceID& clipId, double splitBeat) {
509 std::unique_lock<std::shared_mutex> lock(m_mutex);
510
511 auto* clip = getClipInternal(clipId);
512 if (!clip)
513 return ClipInstanceID();
514
515 if (splitBeat <= clip->startBeat || splitBeat >= clip->endBeat()) {
516 return ClipInstanceID(); // Invalid split position
517 }
518
519 auto laneIt = m_clipLaneMap.find(clipId);
520 if (laneIt == m_clipLaneMap.end())
521 return ClipInstanceID();
522
523 auto laneIdxIt = m_laneMap.find(laneIt->second);
524 if (laneIdxIt == m_laneMap.end())
525 return ClipInstanceID();
526
527 // Create new clip for second part
528 ClipInstance newClip;
529 newClip.id = ClipInstanceID::generate();
530 newClip.name = clip->name;
531 newClip.startBeat = splitBeat;
532 newClip.durationBeats = clip->endBeat() - splitBeat;
533 // Clone the pattern so split halves are independent (not instanced)
534 // Per design philosophy: "Everything is independent by default"
535 bool rebased = false;
536 if (m_patternManager) {
537 PatternID clonedId = m_patternManager->clonePattern(clip->patternId);
538 if (clonedId.isValid()) {
539 newClip.patternId = clonedId;
540 // Prune + rebase the clone to its region so the second half is
541 // a truthful, self-contained pattern (#786): notes fully
542 // outside the region are dropped (no ghost units), straddling
543 // notes re-anchor at the cut and are clamped to the region
544 // end, and the pattern starts at its own origin. Without this
545 // the clone carries the whole original pattern and only the
546 // engine's region filter keeps playback honest — the editor
547 // shows content the scheduler can never play.
548 rebased = prunePatternToRegion(m_patternManager, clonedId, splitBeat - clip->startBeat,
549 newClip.durationBeats);
550 } else {
551 newClip.patternId = clip->patternId; // fallback: share if clone fails
552 }
553 } else {
554 newClip.patternId = clip->patternId;
555 }
556 newClip.colorRGBA = clip->colorRGBA;
557 newClip.sourceId = newClip.patternId.value; // legacy mirror of the referent; never read (F8/F9)
558 // A rebased pattern starts at its own origin; the shared-pattern
559 // fallback keeps the region-shifted offset (engine filter semantics).
560 newClip.sourceOffset = rebased ? 0.0 : (clip->sourceOffset + (splitBeat - clip->startBeat));
561 newClip.sourceOffsetSeconds = clip->sourceOffsetSeconds;
562 newClip.edits = clip->edits;
563 newClip.edits.fadeInBeats = 0.0f; // Clear fades at split point
564
565 // Trim original clip
566 clip->durationBeats = splitBeat - clip->startBeat;
567 clip->edits.fadeOutBeats = 0.0f;
568 if (isAudioClipUnlocked(*clip)) {
569 // Source-domain invariants under varispeed (#746): the second half
570 // must start at splitOffsetSeconds * varispeed into the source and
571 // both halves keep durationSeconds == beatToSeconds(durationBeats)
572 // / varispeed, matching what the renderer reads.
573 const double splitOffsetSeconds = beatToSeconds(splitBeat - clip->startBeat);
574 const double varispeed = static_cast<double>(clip->edits.effectiveVarispeed());
575 newClip.sourceOffsetSeconds = clip->sourceOffsetSeconds + splitOffsetSeconds * varispeed;
576 clip->durationSeconds = beatToSeconds(clip->durationBeats) / varispeed;
577 newClip.durationSeconds = beatToSeconds(newClip.durationBeats) / varispeed;
578 }
579
580 // Add new clip
581 m_lanes[laneIdxIt->second].clips.push_back(newClip);
582 m_clipLaneMap[newClip.id] = laneIt->second;
583
584 notifyClipChanged(clipId);
585 notifyClipChanged(newClip.id);
586
587 return newClip.id;
588 }

602 std::unique_ptr<PlaylistRuntimeSnapshot> buildRuntimeSnapshot(const PatternManager& patterns,
603 const SourceManager& sources) const {
604 // Use shared_lock for readers (audio thread) - allows concurrent reads
605 std::shared_lock<std::shared_mutex> lock(m_mutex);
606
607 if (m_lanes.empty()) {
608 return nullptr;
609 }
610
611 auto snapshot = std::make_unique<PlaylistRuntimeSnapshot>();
612 snapshot->lanes.reserve(m_lanes.size());
613 snapshot->bpm = std::max(m_bpm, 1.0);
614 snapshot->projectSampleRate = m_projectSampleRate;
615
616 const double samplesPerBeat = (m_projectSampleRate * 60.0) / snapshot->bpm;
617
618 const bool anyLaneSolo = std::any_of(m_lanes.begin(), m_lanes.end(),
619 [](const PlaylistLane& lane) { return lane.solo && !lane.muted; });
620
621 for (const auto& lane : m_lanes) {
622 LaneRuntimeInfo laneInfo;
623 laneInfo.clips.reserve(lane.clips.size());
624
625 const bool laneAudible = !lane.muted && (!anyLaneSolo || lane.solo);
626
627 for (const auto& clip : lane.clips) {
628 ClipRuntimeInfo clipInfo;
629 clipInfo.startTime = static_cast<uint64_t>(clip.startBeat * samplesPerBeat);
630 clipInfo.duration = static_cast<uint64_t>(clip.durationBeats * samplesPerBeat);
631 clipInfo.sourceStart = getClipSourceStartSamples(clip);
632 clipInfo.gainLinear = std::isfinite(clip.edits.gainLinear) ? clip.edits.gainLinear : 1.0f;
633 clipInfo.pan =
634 std::isfinite(clip.edits.pan) ? std::clamp(clip.edits.pan, -1.0f, 1.0f) : 0.0f;
635 clipInfo.playbackRate =
636 std::isfinite(clip.edits.playbackRate) ? std::clamp(clip.edits.playbackRate, 0.25f, 4.0f) : 1.0f;
637 clipInfo.pitchSemitones =
638 std::isfinite(clip.edits.pitchSemitones)
639 ? std::clamp(clip.edits.pitchSemitones, ClipEdits::kMinPitchSemitones,
640 ClipEdits::kMaxPitchSemitones)
641 : 0.0f;
642 const double fadeInBeats =
643 std::isfinite(clip.edits.fadeInBeats) ? std::max(0.0, static_cast<double>(clip.edits.fadeInBeats))
644 : 0.0;
645 const double fadeOutBeats =
646 std::isfinite(clip.edits.fadeOutBeats)
647 ? std::max(0.0, static_cast<double>(clip.edits.fadeOutBeats))
648 : 0.0;
649 clipInfo.fadeInSamples = static_cast<uint64_t>(fadeInBeats * samplesPerBeat);
650 clipInfo.fadeOutSamples = static_cast<uint64_t>(fadeOutBeats * samplesPerBeat);
651 clipInfo.isAudioClip = true;
652
653 if (!laneAudible || clip.edits.muted) {
654 continue;
655 }
656
657 // Resolve audio data through pattern → AudioSlicePayload → source
658 if (clip.patternId.isValid()) {
659 auto* pattern = patterns.getPattern(clip.patternId);
660 if (pattern && pattern->isAudio()) {
661 clipInfo.mixerChannelId = pattern->getMixerChannelId();
662 auto& audioPayload = std::get<AudioSlicePayload>(pattern->payload);
663 if (auto* source = sources.getSource(audioPayload.audioSourceId)) {
664 // Phase 4 (F1): prefer the anti-aliased filtered copy when
665 // this clip is DOWNSAMPLED into the session and the copy is
666 // ready; otherwise the original buffer (pre-Phase-4 path).
667 // The filtered copy has its own sample rate (the source's, or the
668 // project's when stored decimated), read from the buffer just below,
669 // so the timing math is identical either way.
670 auto renderBuffer = source->getFilteredBufferFor(
671 static_cast<uint32_t>(m_projectSampleRate));
672 if (!renderBuffer) {
673 renderBuffer = source->getSharedBuffer();
674 }
675 clipInfo.audioData = const_cast<AudioBufferData*>(renderBuffer.get());
676 clipInfo.sharedAudioData = renderBuffer;
677 if (clipInfo.audioData) {
678 clipInfo.sourceSampleRate = clipInfo.audioData->sampleRate;
679 clipInfo.sourceChannels = clipInfo.audioData->numChannels;
680 }
681 }
682 } else if (pattern && pattern->isMidi()) {
683 clipInfo.isAudioClip = false;
684 }
685 }
686
687 laneInfo.clips.push_back(clipInfo);
688 }
689
690 // Automation rides the same snapshot as clips — this copy is what
691 // connects drawn/loaded curves to the render graph. It was missing
692 // entirely, so automation persisted and displayed but never played.
693 laneInfo.automationCurves = lane.automationCurves;
694
695 snapshot->lanes.push_back(std::move(laneInfo));
696 }
697
698 return snapshot;
699 }

716 uint64_t getClipSourceStartSamples(const ClipInstance& clip) const {
717 constexpr uint64_t kMaxSampleOffset = std::numeric_limits<uint64_t>::max();
718 const auto toSampleOffset = [kMaxSampleOffset](double offset, double scale) -> uint64_t {
719 if (!std::isfinite(offset) || offset <= 0.0 || !std::isfinite(scale) || scale <= 0.0) {
720 return 0;
721 }
722
723 const long double scaled = static_cast<long double>(offset) * static_cast<long double>(scale);
724 if (scaled >= static_cast<long double>(kMaxSampleOffset)) {
725 return kMaxSampleOffset;
726 }
727 return static_cast<uint64_t>(scaled);
728 };
729
730 const double bpm = std::max(m_bpm, 1.0);
731 const double samplesPerBeat = (m_projectSampleRate * 60.0) / bpm;
732 const uint64_t canonicalSourceStart =
733 clip.durationSeconds > 0.0 ? toSampleOffset(clip.sourceOffsetSeconds, m_projectSampleRate)
734 : toSampleOffset(clip.sourceOffset, samplesPerBeat);
735 const uint64_t instanceSourceStart = toSampleOffset(clip.edits.sourceStart, 1.0);
736 return instanceSourceStart > kMaxSampleOffset - canonicalSourceStart
737 ? kMaxSampleOffset
738 : canonicalSourceStart + instanceSourceStart;
739 }

751 void setBPM(double bpm) {
752 if (!std::isfinite(bpm) || bpm <= 0.0) {
753 return;

756 std::vector<ClipInstanceID> changedClips;
757 {
758 std::unique_lock<std::shared_mutex> lock(m_mutex);
759 m_bpm = bpm;
760 for (auto& lane : m_lanes) {
761 for (auto& clip : lane.clips) {
762 if (!isAudioClipUnlocked(clip) || clip.durationSeconds <= 0.0) {
763 continue;
764 }
765 const double newDurationBeats = secondsToBeatsUnlocked(clip.durationSeconds);
766 const double newSourceOffsetBeats = secondsToBeatsUnlocked(clip.sourceOffsetSeconds);
767 if (std::abs(newDurationBeats - clip.durationBeats) > 1.0e-9 ||
768 std::abs(newSourceOffsetBeats - clip.sourceOffset) > 1.0e-9) {
769 clip.durationBeats = newDurationBeats;
770 clip.sourceOffset = newSourceOffsetBeats;
771 changedClips.push_back(clip.id);
772 }
773 }
774 }
775 }
776

778 notifyClipChanged(clipId);
779 }

833 ClipInstanceID addClipFromPattern(const PlaylistLaneID& laneId, PatternID patternId, double startBeat,
834 double durationBeats) {
835 std::unique_lock<std::shared_mutex> lock(m_mutex);
836
837 auto it = m_laneMap.find(laneId);
838 if (it == m_laneMap.end())
839 return ClipInstanceID();
840
841 ClipInstance clip;
842 clip.id = ClipInstanceID::generate();
843 clip.startBeat = startBeat;
844 clip.durationBeats = durationBeats;
845 clip.patternId = patternId;
846 clip.sourceId = patternId.value; // Store pattern ID in sourceId for now
847 if (m_patternManager) {
848 const auto* pattern = m_patternManager->getPattern(patternId);
849 if (pattern && pattern->isAudio()) {
850 clip.edits = ClipEdits::forNewAudioClip();
851 const auto& payload = std::get<AudioSlicePayload>(pattern->payload);
852 if (payload.durationSeconds > 0.0) {
853 clip.durationSeconds = payload.durationSeconds;
854 } else {
855 clip.durationSeconds = beatToSeconds(durationBeats);
856 }
857 }
858 }
859
860 m_lanes[it->second].clips.push_back(clip);
861 m_clipLaneMap[clip.id] = laneId;
862
863 notifyClipChanged(clip.id);
864 return clip.id;
865 }

881 std::shared_lock<std::shared_mutex> lock(m_mutex);
882 std::vector<MidiClipPlaybackInstance> instances;
883 const bool anyLaneSolo = std::any_of(m_lanes.begin(), m_lanes.end(),
884 [](const PlaylistLane& lane) { return lane.solo && !lane.muted; });
885 for (const auto& lane : m_lanes) {

887 continue;
888 }
889 for (const auto& clip : lane.clips) {
890 if (clip.edits.muted) {
891 continue;
892 }
893 if (!clip.patternId.isValid()) {
894 continue;
895 }

897 if (!pattern || !pattern->isMidi()) {
898 continue;
899 }

902 instance.clipId = clip.id;
903 instance.patternId = clip.patternId;
904 instance.startBeat = clip.startBeat;
905 instance.sourceOffsetBeats = clip.sourceOffset;
906 instance.durationBeats = clip.durationBeats;
907 instance.patternLengthBeats = pattern->lengthBeats;
908 if (instance.isValid()) {
909 instances.push_back(instance);
910 }
911 }
912 }

930 bool removeLane(const PlaylistLaneID& laneId) {
931 std::unique_lock<std::shared_mutex> lock(m_mutex);
932
933 auto it = m_laneMap.find(laneId);
934 if (it == m_laneMap.end())
935 return false;
936
937 size_t laneIdx = it->second;
938
939 // Remove all clips in this lane from the clip lane map
940 for (const auto& clip : m_lanes[laneIdx].clips) {
941 m_clipLaneMap.erase(clip.id);
942 }
943
944 // Remove the lane
945 m_lanes.erase(m_lanes.begin() + laneIdx);
946
947 // Rebuild lane map with correct indices. PlaylistLane::index is
948 // display metadata consumed by the track-number marker and the
949 // automation-curve default channel pairing; after an erasure it must
950 // stay positional or a later lane reads a stale index (wrong channel,
951 // colliding numbers — #816 follow-up).
952 m_laneMap.clear();
953 for (size_t i = 0; i < m_lanes.size(); ++i) {
954 m_lanes[i].index = static_cast<int>(i);
955 m_laneMap[m_lanes[i].id] = i;
956 }
957
958 return true;
959 }

979 bool moveLaneToIndex(const PlaylistLaneID& laneId, size_t targetIndex) {
980 std::unique_lock<std::shared_mutex> lock(m_mutex);
981 auto it = m_laneMap.find(laneId);
982 if (it == m_laneMap.end() || m_lanes.empty()) {
983 return false;
984 }
985 const size_t from = it->second;
986 targetIndex = std::min(targetIndex, m_lanes.size() - 1);
987 if (from == targetIndex) {
988 return true;
989 }
990
991 PlaylistLane lane = std::move(m_lanes[from]);
992 m_lanes.erase(m_lanes.begin() + from);
993 m_lanes.insert(m_lanes.begin() + targetIndex, std::move(lane));
994
995 // Keep .index truthful across the span the lane traveled.
996 const size_t lo = std::min(from, targetIndex);
997 const size_t hi = std::max(from, targetIndex);
998 for (size_t i = lo; i <= hi; ++i) {
999 m_lanes[i].index = static_cast<int>(i);
1000 }
1001
1002 m_laneMap.clear();
1003 for (size_t i = 0; i < m_lanes.size(); ++i) {
1004 m_laneMap[m_lanes[i].id] = i;
1005 }
1006 return true;
1007 }
