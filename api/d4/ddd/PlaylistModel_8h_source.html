168 auto laneIt = m_clipLaneMap.find(clipId);

283 void moveClip(const ClipInstanceID& clipId, double newStartBeat,
284 const PlaylistLaneID& newLaneId = PlaylistLaneID()) {
285 std::unique_lock<std::shared_mutex> lock(m_mutex);
286
287 auto* clip = getClipInternal(clipId);
288 if (!clip)
289 return;
290
291 // Store clip data
292 ClipInstance clipCopy = *clip;
293 clipCopy.startBeat = newStartBeat;
294
295 // Check if we need to move to a different lane
296 if (newLaneId.isValid()) {
297 auto oldLaneIt = m_clipLaneMap.find(clipId);
298 if (oldLaneIt != m_clipLaneMap.end() && oldLaneIt->second != newLaneId) {
299 // Remove from old lane
300 auto oldLaneIdxIt = m_laneMap.find(oldLaneIt->second);
301 if (oldLaneIdxIt != m_laneMap.end()) {
302 auto& oldClips = m_lanes[oldLaneIdxIt->second].clips;
303 oldClips.erase(std::remove_if(oldClips.begin(), oldClips.end(),
304 [&clipId](const ClipInstance& c) { return c.id == clipId; }),
305 oldClips.end());
306 }
307
308 // Add to new lane
309 auto newLaneIdxIt = m_laneMap.find(newLaneId);
310 if (newLaneIdxIt != m_laneMap.end()) {
311 m_lanes[newLaneIdxIt->second].clips.push_back(clipCopy);
312 m_clipLaneMap[clipId] = newLaneId;
313 }
314
315 notifyClipChanged(clipId);
316 return;
317 }
318 }
319
320 // Just update position
321 clip->startBeat = newStartBeat;
322 notifyClipChanged(clipId);
323 }

331 ClipInstanceID splitClip(const ClipInstanceID& clipId, double splitBeat) {
332 std::unique_lock<std::shared_mutex> lock(m_mutex);
333
334 auto* clip = getClipInternal(clipId);
335 if (!clip)
336 return ClipInstanceID();
337
338 if (splitBeat <= clip->startBeat || splitBeat >= clip->endBeat()) {
339 return ClipInstanceID(); // Invalid split position
340 }
341
342 auto laneIt = m_clipLaneMap.find(clipId);
343 if (laneIt == m_clipLaneMap.end())
344 return ClipInstanceID();
345
346 auto laneIdxIt = m_laneMap.find(laneIt->second);
347 if (laneIdxIt == m_laneMap.end())
348 return ClipInstanceID();
349
350 // Create new clip for second part
351 ClipInstance newClip;
352 newClip.id = ClipInstanceID::generate();
353 newClip.name = clip->name;
354 // Clone the pattern so split halves are independent (not instanced)
355 // Per design philosophy: "Everything is independent by default"
356 if (m_patternManager) {
357 PatternID clonedId = m_patternManager->clonePattern(clip->patternId);
358 if (clonedId.isValid()) {
359 newClip.patternId = clonedId;
360 } else {
361 newClip.patternId = clip->patternId; // fallback: share if clone fails
362 }
363 } else {
364 newClip.patternId = clip->patternId;
365 }
366 newClip.colorRGBA = clip->colorRGBA;
367 newClip.startBeat = splitBeat;
368 newClip.durationBeats = clip->endBeat() - splitBeat;
369 newClip.sourceId = clip->sourceId;
370 newClip.sourceOffset = clip->sourceOffset + (splitBeat - clip->startBeat);
371 newClip.sourceOffsetSeconds = clip->sourceOffsetSeconds;
372 newClip.edits = clip->edits;
373 newClip.edits.fadeInBeats = 0.0f; // Clear fades at split point
374
375 // Trim original clip
376 clip->durationBeats = splitBeat - clip->startBeat;
377 clip->edits.fadeOutBeats = 0.0f;
378 if (isAudioClipUnlocked(*clip)) {
379 const double splitOffsetSeconds = beatToSeconds(splitBeat - clip->startBeat);
380 newClip.sourceOffsetSeconds = clip->sourceOffsetSeconds + splitOffsetSeconds;
381 clip->durationSeconds = beatToSeconds(clip->durationBeats);
382 newClip.durationSeconds = beatToSeconds(newClip.durationBeats);
383 }
384
385 // Add new clip
386 m_lanes[laneIdxIt->second].clips.push_back(newClip);
387 m_clipLaneMap[newClip.id] = laneIt->second;
388
389 notifyClipChanged(clipId);
390 notifyClipChanged(newClip.id);
391
392 return newClip.id;
393 }

407 std::unique_ptr<PlaylistRuntimeSnapshot> buildRuntimeSnapshot(const PatternManager& patterns,
408 const SourceManager& sources) const {
409 // Use shared_lock for readers (audio thread) - allows concurrent reads
410 std::shared_lock<std::shared_mutex> lock(m_mutex);
411
412 if (m_lanes.empty()) {
413 return nullptr;
414 }
415
416 auto snapshot = std::make_unique<PlaylistRuntimeSnapshot>();
417 snapshot->lanes.reserve(m_lanes.size());
418 snapshot->bpm = std::max(m_bpm, 1.0);
419 snapshot->projectSampleRate = m_projectSampleRate;
420
421 const double samplesPerBeat = (m_projectSampleRate * 60.0) / snapshot->bpm;
422
423 for (const auto& lane : m_lanes) {
424 LaneRuntimeInfo laneInfo;
425 laneInfo.clips.reserve(lane.clips.size());
426
427 for (const auto& clip : lane.clips) {
428 ClipRuntimeInfo clipInfo;
429 clipInfo.startTime = static_cast<uint64_t>(clip.startBeat * samplesPerBeat);
430 clipInfo.duration = static_cast<uint64_t>(clip.durationBeats * samplesPerBeat);
431 clipInfo.sourceStart = static_cast<uint64_t>(clip.sourceOffset * samplesPerBeat);
432 clipInfo.gainLinear = clip.edits.gain;
433 clipInfo.isAudioClip = true;
434
435 // Resolve audio data through pattern → AudioSlicePayload → source
436 if (clip.patternId.isValid()) {
437 auto* pattern = patterns.getPattern(clip.patternId);
438 if (pattern && pattern->isAudio()) {
439 auto& audioPayload = std::get<AudioSlicePayload>(pattern->payload);
440 if (auto* source = sources.getSource(audioPayload.audioSourceId)) {
441 // Phase 4 (F1): prefer the anti-aliased filtered copy when
442 // this clip is DOWNSAMPLED into the session and the copy is
443 // ready; otherwise the original buffer (pre-Phase-4 path).
444 // The filtered copy keeps the SOURCE sample rate, so all
445 // timing math below is identical either way.
446 auto renderBuffer = source->getFilteredBufferFor(
447 static_cast<uint32_t>(m_projectSampleRate));
448 if (!renderBuffer) {
449 renderBuffer = source->getSharedBuffer();
450 }
451 clipInfo.audioData = const_cast<AudioBufferData*>(renderBuffer.get());
452 clipInfo.sharedAudioData = renderBuffer;
453 if (clipInfo.audioData) {
454 clipInfo.sourceSampleRate = clipInfo.audioData->sampleRate;
455 clipInfo.sourceChannels = clipInfo.audioData->numChannels;
456 }
457 }
458 } else if (pattern && pattern->isMidi()) {
459 clipInfo.isAudioClip = false;
460 }
461 }
462
463 laneInfo.clips.push_back(clipInfo);
464 }
465
466 // Automation rides the same snapshot as clips — this copy is what
467 // connects drawn/loaded curves to the render graph. It was missing
468 // entirely, so automation persisted and displayed but never played.
469 laneInfo.automationCurves = lane.automationCurves;
470
471 snapshot->lanes.push_back(std::move(laneInfo));
472 }
473
474 return snapshot;
475 }

494 void setBPM(double bpm) {
495 if (!std::isfinite(bpm) || bpm <= 0.0) {
496 return;
497 }
498
499 std::vector<ClipInstanceID> changedClips;
500 {
501 std::unique_lock<std::shared_mutex> lock(m_mutex);
502 m_bpm = bpm;
503 for (auto& lane : m_lanes) {
504 for (auto& clip : lane.clips) {
505 if (!isAudioClipUnlocked(clip) || clip.durationSeconds <= 0.0) {
506 continue;
507 }
508 const double newDurationBeats = secondsToBeatsUnlocked(clip.durationSeconds);
509 const double newSourceOffsetBeats = secondsToBeatsUnlocked(clip.sourceOffsetSeconds);
510 if (std::abs(newDurationBeats - clip.durationBeats) > 1.0e-9 ||
511 std::abs(newSourceOffsetBeats - clip.sourceOffset) > 1.0e-9) {
512 clip.durationBeats = newDurationBeats;
513 clip.sourceOffset = newSourceOffsetBeats;
514 changedClips.push_back(clip.id);
515 }
516 }
517 }
518 }
519
520 for (const auto& clipId : changedClips) {
521 notifyClipChanged(clipId);
522 }
523 }

576 ClipInstanceID addClipFromPattern(const PlaylistLaneID& laneId, PatternID patternId, double startBeat,
577 double durationBeats) {
578 std::unique_lock<std::shared_mutex> lock(m_mutex);
579
580 auto it = m_laneMap.find(laneId);
581 if (it == m_laneMap.end())
582 return ClipInstanceID();
583
584 ClipInstance clip;
585 clip.id = ClipInstanceID::generate();
586 clip.startBeat = startBeat;
587 clip.durationBeats = durationBeats;
588 clip.patternId = patternId;
589 clip.sourceId = patternId.value; // Store pattern ID in sourceId for now
590 if (m_patternManager) {
591 const auto* pattern = m_patternManager->getPattern(patternId);
592 if (pattern && pattern->isAudio()) {
593 const auto& payload = std::get<AudioSlicePayload>(pattern->payload);
594 if (payload.durationSeconds > 0.0) {
595 clip.durationSeconds = payload.durationSeconds;
596 } else {
597 clip.durationSeconds = beatToSeconds(durationBeats);
598 }
599 }
600 }
601
602 m_lanes[it->second].clips.push_back(clip);
603 m_clipLaneMap[clip.id] = laneId;
604
605 notifyClipChanged(clip.id);
606 return clip.id;
607 }

662 bool removeLane(const PlaylistLaneID& laneId) {
663 std::unique_lock<std::shared_mutex> lock(m_mutex);
664
665 auto it = m_laneMap.find(laneId);
666 if (it == m_laneMap.end())
667 return false;
668
669 size_t laneIdx = it->second;
670
671 // Remove all clips in this lane from the clip lane map
672 for (const auto& clip : m_lanes[laneIdx].clips) {
673 m_clipLaneMap.erase(clip.id);
674 }
675
676 // Remove the lane
677 m_lanes.erase(m_lanes.begin() + laneIdx);
678
679 // Rebuild lane map with correct indices
680 m_laneMap.clear();
681 for (size_t i = 0; i < m_lanes.size(); ++i) {
682 m_laneMap[m_lanes[i].id] = i;
683 }
684
685 return true;
686 }
