13inline void drawArsenalGroupHeader(AestraUI::NUIRenderer& renderer, const AestraUI::NUIRect& rect, const char* label, int count) {
14 auto& theme = AestraUI::NUIThemeManager::getInstance();
15 const auto muted = theme.getColor("textSecondary");
16 std::string text = label;
17 for (auto& c : text) c = static_cast<char>(std::toupper(static_cast<unsigned char>(c)));
18 const float y = rect.y + rect.height - 20.0f;
19 renderer.drawText(text, {rect.x + 8.0f, y}, 10.0f, muted.withAlpha(0.8f));
20 const float tw = renderer.measureText(text, 10.0f).width;
21 const std::string n = std::to_string(count);
22 renderer.drawText(n, {rect.x + 8.0f + tw + 10.0f, y}, 10.0f, muted.withAlpha(0.5f));
23 const float lx = rect.x + 8.0f + tw + 10.0f + renderer.measureText(n, 10.0f).width + 10.0f;
24 renderer.fillRect(AestraUI::NUIRect{lx, y + 6.0f, std::max(0.0f, rect.right() - 8.0f - lx), 1.0f}, theme.getColor("border").withAlpha(0.35f));
25}

28inline void drawArsenalRackHeaders(AestraUI::NUIRenderer& renderer, const ArsenalRackLayout& rack,
29 const AestraUI::NUIRect& viewport, float rackTop) {
30 if (viewport.height <= 0.0f) return;
31 renderer.setClipRect(viewport);
32 for (const auto& item : rack.items) {
33 if (item.kind == ArsenalRackItem::Kind::Header) {
34 drawArsenalGroupHeader(renderer, AestraUI::NUIRect(viewport.x, rackTop + item.y, viewport.width, item.h),
35 ArsenalRackLayout::groupLabel(item.group), item.count);
36 }
37 }
38 renderer.clearClipRect();
39}

42inline void drawArsenalFitToggle(AestraUI::NUIRenderer& renderer, const AestraUI::NUIRect& toggle, const AestraUI::NUIRect& fitRect,
43 const AestraUI::NUIRect& scrollRect, bool fitToWidth) {
44 using namespace AestraUI;
45 if (toggle.width <= 0.0f) return;
46 auto& theme = NUIThemeManager::getInstance();
47 const auto& tp = theme.getCurrentTheme();
48 renderer.fillRoundedRect(toggle, tp.radiusS, theme.getColor("surfaceTertiary").withAlpha(0.85f));
49 renderer.strokeRoundedRect(toggle, tp.radiusS, 1.0f, theme.getColor("borderSubtle").withAlpha(0.5f));
50 const NUIColor activeFill = theme.getColor("accentPrimary").withAlpha(0.16f);
51 const NUIColor activeText = theme.getColor("accentPrimary").withAlpha(0.95f);
52 const NUIColor idleText = theme.getColor("textSecondary").withAlpha(0.85f);
53 if (fitRect.width > 0.0f) {
54 if (fitToWidth) renderer.fillRoundedRect(fitRect, tp.radiusXS, activeFill);
55 renderer.drawTextCentered("FIT", fitRect, tp.fontSizeMicro, fitToWidth ? activeText : idleText);
56 }
57 if (scrollRect.width > 0.0f) {
58 if (!fitToWidth) renderer.fillRoundedRect(scrollRect, tp.radiusXS, activeFill);
59 renderer.drawTextCentered("SCROLL", scrollRect, tp.fontSizeMicro, fitToWidth ? idleText : activeText);
60 }
61 const float dividerX = fitRect.width > 0.0f ? fitRect.right() : toggle.x + toggle.width * 0.5f;
62 renderer.drawLine(NUIPoint(dividerX, toggle.y + 5.0f), NUIPoint(dividerX, toggle.bottom() - 5.0f), 1.0f,
63 theme.getColor("borderSubtle").withAlpha(0.6f));
64}

81inline void drawArsenalUnitTypePicker(AestraUI::NUIRenderer& renderer, const AestraUI::NUIRect& card) {
82 using namespace AestraUI;
83 using Audio::UnitType;
84 auto& theme = NUIThemeManager::getInstance();
85 const auto& tp = theme.getCurrentTheme();
86 const float radius = std::max(tp.radiusL - 2.0f, 0.0f);
87 renderer.fillRoundedRect(card, radius, theme.getColor("backgroundSecondary").withAlpha(0.98f));
88 renderer.strokeRoundedRect(card, radius, 1.0f, theme.getColor("borderSubtle").withAlpha(0.6f));
89 const float padding = 10.0f, gap = 8.0f;
90 const float cellW = (card.width - padding * 2.0f - gap) * 0.5f, cellH = (card.height - padding * 2.0f - gap) * 0.5f;
91 struct Kind { UnitType type; const char* icon; const char* name; const char* text; };
92 static const Kind kinds[4] = {{UnitType::Sampler, "▦", "Sampler", "Classic step sequencer"},
93 {UnitType::PitchedSampler, "≋", "808", "808s, bass, slides"},
94 {UnitType::Instrument, "♫", "MIDI", "Notes, chords, instruments"},
95 {UnitType::Audio, "◫", "Audio", "Drop a clip directly"}};
96 for (int i = 0; i < 4; ++i) {
97 const NUIRect cell(card.x + padding + (i % 2) * (cellW + gap), card.y + padding + (i / 2) * (cellH + gap), cellW, cellH);
98 renderer.fillRoundedRect(cell, tp.radiusM, theme.getColor("surfaceTertiary").withAlpha(0.8f));
99 renderer.strokeRoundedRect(cell, tp.radiusM, 1.0f, theme.getColor("borderSubtle").withAlpha(0.5f));
100 renderer.drawText(kinds[i].icon, NUIPoint(cell.x + 10.0f, cell.y + 11.0f), tp.fontSizeS - 1.0f, theme.getColor("accentPrimary").withAlpha(0.9f));
101 renderer.drawText(kinds[i].name, NUIPoint(cell.x + 32.0f, cell.y + 10.0f), tp.fontSizeXS - 1.0f, theme.getColor("textPrimary").withAlpha(0.95f));
102 renderer.drawText(kinds[i].text, NUIPoint(cell.x + 10.0f, cell.y + 28.0f), tp.fontSizeMicro, theme.getColor("textSecondary").withAlpha(0.68f));
103 }
104}

107inline void drawArsenalDropGhost(AestraUI::NUIRenderer& renderer, const AestraUI::NUIRect& ghost, const std::string& label) {
108 using namespace AestraUI;
109 auto& theme = NUIThemeManager::getInstance();
110 const auto outline = theme.getColor("accentPrimary").withAlpha(0.85f);
111 constexpr float kDash = 7.0f, kGap = 5.0f;
112 for (float x = ghost.x; x < ghost.right(); x += kDash + kGap) {
113 const float x1 = std::min(x + kDash, ghost.right());
114 renderer.drawLine(NUIPoint(x, ghost.y), NUIPoint(x1, ghost.y), 1.6f, outline);
115 renderer.drawLine(NUIPoint(x, ghost.bottom()), NUIPoint(x1, ghost.bottom()), 1.6f, outline);
116 }
117 for (float y = ghost.y; y < ghost.bottom(); y += kDash + kGap) {
118 const float y1 = std::min(y + kDash, ghost.bottom());
119 renderer.drawLine(NUIPoint(ghost.x, y), NUIPoint(ghost.x, y1), 1.6f, outline);
120 renderer.drawLine(NUIPoint(ghost.right(), y), NUIPoint(ghost.right(), y1), 1.6f, outline);
121 }
122 const auto textSize = renderer.measureText(label, 10.5f);
123 renderer.drawText(label, NUIPoint(ghost.x + (ghost.width - textSize.width) * 0.5f, ghost.y + (ghost.height - textSize.height) * 0.5f), 10.5f,
124 theme.getColor("textPrimary").withAlpha(0.9f));
125}
