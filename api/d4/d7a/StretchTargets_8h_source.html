4// Stretch-to-Target: which clip lengths are worth offering, and in what order (SPEC-001, V8-W1).
5//
6// Two independent decisions that used to be one fixed 1 / 2 / 4 / 8 menu:
7// 1. the GENERATOR: which target lengths make musical sense for a clip of L bars
8// (engine-agnostic musical truth);
9// 2. the RANKER: which of those is the best trade given what the engine can do
10// (engine-aware cost).
11// The generator never changes when the engine does; only distanceCost() and the label wording do.
12//
13// Pure functions of the clip length (and, for the "match arrangement" row, the lengths of other
14// clips). No model, no UI, no engine: callable and testable anywhere.
15//
16// Where SPEC-001 disagrees with itself, the choice made here is written next to the code:
17// - window: the spec writes [L/4, L*2] ∩ [4, 256], which leaves a 1-bar clip with no target
18// at all, while §5 says the menu is "the only affordance that works for a 1-bar clip".
19// The floor here is 1 bar.
20// - distance cost: §4's prose says the varispeed era is quadratic, but its formula and every
21// worked example are linear. The formula as written is implemented (see distanceCost()).

62constexpr int kPhrase4[] = {4, 8, 16, 32, 64, 128};
63constexpr int kPhrase3[] = {12, 24, 48, 96};
64constexpr int kStructural[] = {6, 10, 20, 30};
65constexpr double kMinBars = 1.0; // a target is at least one bar
66constexpr double kMaxBars = 256.0; // and no more than the kernel's practical ceiling
67constexpr double kMinRatio = 0.25; // the engine's varispeed envelope, from #747
68constexpr double kMaxRatio = 4.0;

71constexpr double bonus(StretchTargetClass c) {
72 switch (c) {
73 case StretchTargetClass::Phrase4: return 2.5;
74 case StretchTargetClass::Phrase3: return 2.0;
75 case StretchTargetClass::Structural: return 1.5;
76 case StretchTargetClass::Fractional: return 1.0;
77 case StretchTargetClass::Echo: return 0.5;
78 case StretchTargetClass::Integer: return 0.0;
79 }
80 return 0.0;
81}

95inline std::vector<StretchTarget> generateStretchTargets(double lengthBars) {
96 using namespace stretch_detail;
97 std::vector<StretchTarget> out;
98 if (!(lengthBars > 0.0) || !std::isfinite(lengthBars)) {
99 return out;
100 }
101 const double lo = std::max(lengthBars / 4.0, kMinBars);
102 const double hi = std::min(lengthBars * 2.0, kMaxBars);
103
104 auto add = [&](double bars, StretchTargetClass cls) {
105 if (!isWholeBar(bars) || bars < lo - 1e-9 || bars > hi + 1e-9) {
106 return;
107 }
108 const int whole = static_cast<int>(std::lround(bars));
109 if (std::fabs(static_cast<double>(whole) - lengthBars) < 1e-9) {
110 return; // T == L: a no-op suggestion is noise
111 }
112 for (auto& existing : out) {
113 if (existing.bars == whole) {
114 if (bonus(cls) > bonus(existing.cls)) {
115 existing.cls = cls; // the strongest family it belongs to
116 }
117 return;
118 }
119 }
120 StretchTarget t;
121 t.bars = whole;
122 t.cls = cls;
123 t.speed = lengthBars / static_cast<double>(whole);
124 const double ratio = static_cast<double>(whole) / lengthBars;
125 t.approximate = ratio < kMinRatio - 1e-9 || ratio > kMaxRatio + 1e-9;
126 out.push_back(t);
127 };
128
129 for (int b : kPhrase4) add(b, StretchTargetClass::Phrase4);
130 for (int b : kPhrase3) add(b, StretchTargetClass::Phrase3);
131 for (int b : kStructural) add(b, StretchTargetClass::Structural);
132 add(lengthBars / 2.0, StretchTargetClass::Fractional);
133 add(lengthBars / 3.0, StretchTargetClass::Fractional);
134 add(lengthBars / 4.0, StretchTargetClass::Fractional);
135 add(lengthBars * 2.0, StretchTargetClass::Fractional);
136 add(lengthBars * 1.5, StretchTargetClass::Fractional);
137 add(std::round(lengthBars), StretchTargetClass::Integer); // the honest floor
138 return out;
139}

145inline StretchTargetList rankStretchTargets(double lengthBars, const std::vector<int>& echoLengthsBars = {},
146 StretchEngine engine = StretchEngine::Varispeed,
147 size_t maxRanked = 5) {
148 using namespace stretch_detail;
149 StretchTargetList list;
150 list.ranked = generateStretchTargets(lengthBars);
151 if (list.ranked.empty()) {
152 return list;
153 }
154 for (auto& t : list.ranked) {
155 t.score = distanceCost(engine, lengthBars - static_cast<double>(t.bars)) - bonus(t.cls);
156 }
157 std::sort(list.ranked.begin(), list.ranked.end(), [&](const StretchTarget& a, const StretchTarget& b) {
158 if (a.score != b.score) return a.score < b.score;
159 const double da = std::fabs(lengthBars - a.bars), db = std::fabs(lengthBars - b.bars);
160 if (da != db) return da < db;
161 return a.bars < b.bars;
162 });
163 if (list.ranked.size() > maxRanked) {
164 list.ranked.resize(maxRanked);
165 }
166
167 std::vector<int> echoes;
168 for (int e : echoLengthsBars) {
169 if (e >= 1 && static_cast<double>(e) >= std::max(lengthBars / 4.0, kMinBars) - 1e-9 &&
170 static_cast<double>(e) <= std::min(lengthBars * 2.0, kMaxBars) + 1e-9 &&
171 std::fabs(static_cast<double>(e) - lengthBars) > 1e-9 && std::find(echoes.begin(), echoes.end(), e) == echoes.end()) {
172 echoes.push_back(e);
173 }
174 }
175 std::sort(echoes.begin(), echoes.end(), [&](int a, int b) {
176 const double da = std::fabs(lengthBars - a), db = std::fabs(lengthBars - b);
177 return da != db ? da < db : a < b;
178 });
179 for (int e : echoes) {
180 StretchTarget t;
181 t.bars = e;
182 t.cls = StretchTargetClass::Echo;
183 t.speed = lengthBars / static_cast<double>(e);
184 t.score = distanceCost(engine, lengthBars - e) - bonus(StretchTargetClass::Echo);
185 const double ratio = static_cast<double>(e) / lengthBars;
186 t.approximate = ratio < kMinRatio - 1e-9 || ratio > kMaxRatio + 1e-9;
187 list.arrangement.push_back(t);
188 }
189 return list;
190}

195inline std::string formatStretchTarget(const StretchTarget& t, StretchEngine engine = StretchEngine::Varispeed) {
196 (void)engine; // varispeed is the only engine; its wording is the only wording
197 char speed[32];
198 std::snprintf(speed, sizeof(speed), "%.2f", t.speed);
199 std::string s = std::to_string(t.bars) + (t.bars == 1 ? " bar" : " bars") + " \xC2\xB7 \xC3\x97" + speed +
200 " speed \xC2\xB7 pitch follows";
201 if (t.approximate) {
202 s += " \xC2\xB7 approximate";
203 }
204 return s;
205}
