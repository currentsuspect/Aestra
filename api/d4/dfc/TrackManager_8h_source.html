157 if (reportRealtimeMisuse("TrackManager::addChannel")) {
158 return nullptr;
159 }
160 // IDs start at 1 to avoid collision with Master (ID 0). A duplicate or
161 // invalid persisted ID is replaced instead of aliasing two live routes.
162 constexpr uint32_t invalidChannelId = std::numeric_limits<uint32_t>::max();
163 uint32_t channelId = requestedId;
164 if (channelId == 0 || channelId == invalidChannelId || getChannelById(channelId) != nullptr) {
165 while (m_nextChannelId != invalidChannelId && getChannelById(m_nextChannelId) != nullptr) {
166 ++m_nextChannelId;
167 }
168 if (m_nextChannelId == invalidChannelId) {
169 return nullptr;
170 }
171 channelId = m_nextChannelId++;
172 } else {
173 m_nextChannelId = std::max(m_nextChannelId, channelId + 1);
174 }
175 auto channel = std::make_unique<MixerChannel>(
176 // "Insert" is reserved for effect slots — a mixer strip is a Channel.
177 name.empty() ? "Channel " + std::to_string(channelId) : name, channelId);
178 channel->setCommandSink(m_commandSink);
179 channel->setInputMonitoringStateChangedCallback([this]() { publishInputMonitoringSnapshot(); });
180 if (m_channelPrepareCallback) {

185 requestAudioGraphRebuild(GraphDirtyReason::TrackStructureChanged);
186 m_modified.store(true, std::memory_order_relaxed);

574 bool canRouteTo(uint32_t sourceId, uint32_t targetId) const {
575 // Master has two spellings: user/model space (0, MASTER_MIXER_CHANNEL_ID)
576 // and engine space (0xFFFFFFFF). Both are the same terminal sink.
577 constexpr uint32_t kEngineMaster = 0xFFFFFFFFu;
578 if (sourceId == MASTER_MIXER_CHANNEL_ID || sourceId == kEngineMaster) {
579 return false;
580 }
581 if (targetId == MASTER_MIXER_CHANNEL_ID || targetId == kEngineMaster) {
582 return true;
583 }
584 if (sourceId == targetId) {
585 return false;
586 }
587 if (!getChannelById(sourceId) || !getChannelById(targetId)) {
588 return false;
589 }
590
591 std::vector<uint32_t> stack{targetId};
592 std::unordered_set<uint32_t> visited;
593 while (!stack.empty()) {
594 const uint32_t current = stack.back();
595 stack.pop_back();
596 if (current == sourceId) {
597 return false;
598 }
599 if (!visited.insert(current).second) {
600 continue;
601 }
602 const MixerChannel* channel = getChannelById(current);
603 if (!channel) {
604 continue;
605 }
606 const uint32_t mainOutput = channel->getMainOutputId();
607 if (mainOutput != kEngineMaster) {
608 stack.push_back(mainOutput);
609 }
610 for (const auto& send : channel->getSends()) {
611 if (send.mute || send.sidechainOnly) {
612 continue;
613 }
614 if (send.targetChannelId != kEngineMaster) {
615 stack.push_back(send.targetChannelId);
616 }
617 }
618 }
619 return true;
620 }

798 bool getRecordingDataPeaks(uint64_t trackId, uint32_t maxPoints, std::vector<float>& peaksOut,
799 double& startBeat, size_t& ringSamplesOut) {
800 if (maxPoints == 0) {
801 return false;
802 }
803 std::lock_guard<std::mutex> lock(m_recordingMutex);
804 auto it = m_recordingCaptures.find(trackId);
805 if (it == m_recordingCaptures.end() || !it->second) {
806 return false;
807 }
808 RecordingCapture* capture = it->second.get();
809 const size_t size = capture->size.load(std::memory_order_acquire);
810 if (size == 0) {
811 return false;
812 }
813 startBeat = capture->startBeat.load(std::memory_order_acquire);
814 ringSamplesOut = size;
815
816 peaksOut.clear();
817 peaksOut.reserve(static_cast<size_t>(maxPoints) * 2);
818 const size_t head = capture->headIndex.load(std::memory_order_relaxed);
819 // Ceiling division: every sample lands in a bucket so the ring tail
820 // can never be dropped by rounding.
821 const size_t stride = (size + maxPoints - 1) / maxPoints;
822 if (stride == 0) {
823 return false;
824 }
825 for (size_t offset = 0; offset < size && peaksOut.size() < static_cast<size_t>(maxPoints) * 2;
826 offset += stride) {
827 const size_t span = std::min(stride, size - offset);
828 // Seed from the first sample: zero-init would report a false bound
829 // for positive-only or negative-only buckets. Non-finite input
830 // samples (driver glitches) are skipped rather than poisoning the
831 // pair — they are a render/capture concern, not signal.
832 bool seeded = false;
833 float lo = 0.0f;
834 float hi = 0.0f;
835 for (size_t k = 0; k < span; ++k) {
836 const float v =
837 capture->samples[(head + offset + k) % capture->capacity].load(std::memory_order_relaxed);
838 if (!std::isfinite(v)) {
839 continue;
840 }
841 if (!seeded) {
842 lo = v;
843 hi = v;
844 seeded = true;
845 continue;
846 }
847 lo = std::min(lo, v);
848 hi = std::max(hi, v);
849 }
850 if (!seeded) {
851 // All-non-finite span: emit a NaN pair as an explicit skipped
852 // slot. Dropping the pair entirely would shift every later
853 // bucket's screen position left (#854 round 2) — the renderer
854 // ignores non-finite pairs, so timing stays correct.
855 constexpr float nanPair = std::numeric_limits<float>::quiet_NaN();
856 peaksOut.push_back(nanPair);
857 peaksOut.push_back(nanPair);
858 continue;
859 }
860 peaksOut.push_back(lo);
861 peaksOut.push_back(hi);
862 }
863 return !peaksOut.empty();
864 }

1004 void processInput(const float* input, uint32_t frames, AudioTelemetry* telemetry = nullptr,
1005 uint64_t transportFrame = kUnknownTransportFrame) {
1006 if (!m_isCapturing.load(std::memory_order_relaxed) || !input || frames == 0 || m_inputChannelCount <= 0) {
1007 return;
1008 }
1009
1010 // Increment writer count BEFORE reading snapshot index so
1011 // finalizeCaptureSession() cannot destroy the snapshot while we read it.
1012 m_recordingWriters.fetch_add(1, std::memory_order_acq_rel);
1013
1014 // RAII guard: ensures writer count is always decremented on any exit path.
1015 struct WriterGuard {
1016 std::atomic<uint32_t>& writers;
1017 ~WriterGuard() { writers.fetch_sub(1, std::memory_order_acq_rel); }
1018 } writerGuard{m_recordingWriters};
1019
1020 // Read the immutable recording capture snapshot — avoids iterating the
1021 // live m_channels / m_recordingCaptures containers on the RT thread.
1022 const uint32_t snapIdx = m_activeRecordingCaptureSnapshot.load(std::memory_order_acquire);
1023 const uint32_t routeCount = m_recordingCaptureRouteCounts[snapIdx].load(std::memory_order_acquire);
1024 if (routeCount == 0) {
1025 return;
1026 }
1027
1028 const auto& routes = m_recordingCaptureSnapshots[snapIdx];
1029
1030 const double bpm = std::max(1.0, m_playlistModel.getBPM());
1031 const double sampleRate = std::max(1.0, m_inputSampleRate > 0.0 ? m_inputSampleRate : m_outputSampleRate);
1032 // T-6: prefer the engine's authoritative frame over the UI-cached
1033 // position for capture placement; the latter lags by audio buffers.
1034 const double captureBeat =
1035 (transportFrame != kUnknownTransportFrame)
1036 ? (static_cast<double>(transportFrame) / sampleRate) * bpm / 60.0
1037 : getCurrentTransportBeat();
1038 const bool hasDeferredStart = m_hasDeferredRecordingStart.load(std::memory_order_acquire);
1039 const double deferredStartBeat = m_deferredRecordingStartBeat.load(std::memory_order_relaxed);
1040 bool capturedAnyChannel = false;
1041 bool startedDeferredCapture = false;
1042
1043 for (uint32_t r = 0; r < routeCount; ++r) {
1044 RecordingCapture* capture = routes[r].capture;
1045 if (!capture) {
1046 continue;
1047 }
1048 const int requestedInput = routes[r].inputIndex;
1049 if (requestedInput < 0 && requestedInput != -2) {
1050 continue;
1051 }
1052
1053 const size_t capacity = capture->capacity;
1054 if (capacity == 0 || !capture->samples) {
1055 continue;
1056 }
1057
1058 size_t head = capture->headIndex.load(std::memory_order_relaxed);
1059 size_t size = capture->size.load(std::memory_order_relaxed);
1060 uint32_t startFrame = 0;
1061 double effectiveCaptureBeat = captureBeat;
1062
1063 if (hasDeferredStart) {
1064 const double beatsUntilCapture = deferredStartBeat - captureBeat;
1065 if (beatsUntilCapture > 0.0) {
1066 const double secondsUntilCapture = beatsUntilCapture * 60.0 / bpm;
1067 const uint32_t skippedFrames = static_cast<uint32_t>(std::ceil(secondsUntilCapture * sampleRate));
1068 if (skippedFrames >= frames) {
1069 continue;
1070 }
1071 startFrame = skippedFrames;
1072 effectiveCaptureBeat = deferredStartBeat;
1073 } else {
1074 effectiveCaptureBeat = deferredStartBeat;
1075 }
1076 }
1077
1078 auto appendSample = [&](float sample) {
1079 if (!std::isfinite(sample)) {
1080 sample = 0.0f;
1081 }
1082 size_t writeIndex = (head + size) % capacity;
1083 capture->samples[writeIndex].store(sample, std::memory_order_relaxed);
1084 if (size < capacity) {
1085 ++size;
1086 } else {
1087 // Wrapping is normal: the take stream has already drained
1088 // these frames to memory and disk (RecordingTakeStream).
1089 head = (head + 1) % capacity;
1090 }
1091 };
1092
1093 if (requestedInput == -2) {
1094 const float channelScale = 1.0f / static_cast<float>(m_inputChannelCount);
1095 for (uint32_t frame = startFrame; frame < frames; ++frame) {
1096 const size_t baseIndex = static_cast<size_t>(frame) * static_cast<size_t>(m_inputChannelCount);
1097 float mixedSample = 0.0f;
1098 for (int ch = 0; ch < m_inputChannelCount; ++ch) {
1099 mixedSample += input[baseIndex + static_cast<size_t>(ch)];
1100 }
1101 appendSample(mixedSample * channelScale);
1102 }
1103 } else {
1104 const int inputIndex = requestedInput;
1105 if (inputIndex < 0 || inputIndex >= m_inputChannelCount) {
1106 continue;
1107 }
1108
1109 for (uint32_t frame = startFrame; frame < frames; ++frame) {
1110 const size_t sampleIndex = static_cast<size_t>(frame) * static_cast<size_t>(m_inputChannelCount) +
1111 static_cast<size_t>(inputIndex);
1112 appendSample(input[sampleIndex]);
1113 }
1114 }
1115 bool expectedStart = false;
1116 if (capture->hasStarted.compare_exchange_strong(expectedStart, true, std::memory_order_acq_rel)) {
1117 capture->startBeat.store(effectiveCaptureBeat, std::memory_order_release);
1118 if (hasDeferredStart) {
1119 startedDeferredCapture = true;
1120 }
1121 }
1122 capture->headIndex.store(head, std::memory_order_release);
1123 capture->size.store(size, std::memory_order_release);
1124 // Release: the take stream reads the samples after acquiring this count.
1125 capture->totalCapturedFrames.fetch_add(frames - startFrame, std::memory_order_release);
1126 capturedAnyChannel = true;
1127 }
1128
1129 if (startedDeferredCapture) {
1130 clearDeferredRecordingStartBeat();
1131 }
1132
1133 if (!capturedAnyChannel && !m_recordingNoArmLogged) {
1134 (void)telemetry;
1135 m_recordingNoArmLogged = true;
1136 }

1214 void mixInputMonitoring(const float* input, float* output, uint32_t frames, uint32_t outputChannels) const {

1216 return;
1217 }
1218
1219 const uint32_t snapshotIndex = m_activeInputMonitorSnapshot.load(std::memory_order_acquire);
1220 const uint32_t monitoredCount = m_inputMonitorRouteCounts[snapshotIndex].load(std::memory_order_acquire);
1221 if (monitoredCount == 0) {
1222 return;
1223 }

1226 // the stereo-balance law since the strip pan-law fix (2026-08-14), and
1227 // the audition/preview parity contract keeps every listening surface
1228 // at the same reference level. Only the N-input normalization remains.
1229 const float monitorMixScale = 1.0f / static_cast<float>(monitoredCount);
1230 for (uint32_t frame = 0; frame < frames; ++frame) {
1231 const size_t inputBaseIndex = static_cast<size_t>(frame) * static_cast<size_t>(m_inputChannelCount);
1232 float monitoredSample = 0.0f;

1236 const int requestedInput = routes[routeIndex].inputIndex;

1238
1239 if (requestedInput == -2) {
1240 const float channelScale = 1.0f / static_cast<float>(m_inputChannelCount);
1241 for (int ch = 0; ch < m_inputChannelCount; ++ch) {
1242 sample += input[inputBaseIndex + static_cast<size_t>(ch)];

1245 } else if (requestedInput >= 0 && requestedInput < m_inputChannelCount) {
1246 sample = input[inputBaseIndex + static_cast<size_t>(requestedInput)];
1247 }
1248

1250 }

1256 output[outputBaseIndex + 1] += monitoredSample;

1274 void refreshTimelinePatternInstances() {
1275 if (!m_isPlaying.load(std::memory_order_relaxed) || m_patternMode.load(std::memory_order_relaxed)) {
1276 return;
1277 }
1278 // Incremental, NOT clear-and-reschedule: a clear dropped every sounding note's gate and queued note-off
1279 // and re-queued from the UI's lagging playhead (moving one clip cut other clips' notes).
1280 // Each clip keeps its slot (held-note tracking is per slot), a slot keeps its
1281 // frontier, and the content-edit refill re-queues from the engine's real playhead.
1282 const auto instances = m_playlistModel.collectMidiClipInstances(m_patternManager);
1283 std::unordered_map<ClipInstanceID, uint32_t> nextSlots;
1284 std::array<bool, 256> used{};
1285 used[0] = used[1] = true; // 0 unused; 1 is Arsenal's single-pattern slot
1286 for (const auto& [clip, slot] : m_timelineSlots) {
1287 (void)clip;
1288 used[slot] = true; // survivors keep theirs; removed ones are free from the NEXT refresh
1289 }
1290 std::vector<const MidiClipPlaybackInstance*> fresh;
1291 for (const auto& instance : instances) {
1292 if (!instance.isValid()) {
1293 continue;
1294 }
1295 auto it = m_timelineSlots.find(instance.clipId);
1296 if (it == m_timelineSlots.end()) {
1297 fresh.push_back(&instance);
1298 continue;
1299 }
1300 nextSlots[instance.clipId] = it->second;
1301 m_patternPlaybackEngine.updatePatternInstance(instance.patternId, instance.patternStartBeat(), it->second,
1302 instance.sourceOffsetBeats, instance.durationBeats, true);
1303 }
1304 for (const auto& [clip, slot] : m_timelineSlots) {
1305 if (nextSlots.find(clip) == nextSlots.end()) {
1306 // Stays marked used for THIS refresh: the content-edit refill still has to
1307 // release the removed clip's held notes by slot, and a new clip reusing the
1308 // slot now would be mistaken for them.
1309 m_patternPlaybackEngine.removePatternInstance(slot);
1310 }
1311 }
1312 uint32_t cursor = 2;
1313 for (const auto* instance : fresh) {
1314 while (cursor < used.size() && used[cursor]) {
1315 ++cursor;
1316 }
1317 if (cursor >= used.size()) {
1318 Log::warning("[TrackManager] Too many timeline MIDI clip instances to schedule; truncating at 254");
1319 break;
1320 }
1321 used[cursor] = true;
1322 nextSlots[instance->clipId] = cursor;
1323 m_patternPlaybackEngine.updatePatternInstance(instance->patternId, instance->patternStartBeat(), cursor,
1324 instance->sourceOffsetBeats, instance->durationBeats, true);
1325 }
1326 m_timelineSlots = std::move(nextSlots);
1327 // Record what the slots now play, with the timing just handed to the scheduler.
1328 m_scheduledTimelineInstances.clear();
1329 for (const auto& instance : instances) {
1330 if (m_timelineSlots.find(instance.clipId) != m_timelineSlots.end()) {
1331 m_scheduledTimelineInstances.push_back(instance);
1332 }
1333 }
1334 m_patternPlaybackEngine.patternContentEdited();
1335 m_timelineRescheduleCount.fetch_add(1, std::memory_order_relaxed);
1336 }

2521 void conditionRecordedTakeSamples(std::vector<float>& samples) const {
2522 if (samples.empty()) {
2523 return;
2524 }
2525
2526 // Remove DC bias first so the recorded waveform sits more naturally around zero.
2527 double mean = 0.0;
2528 for (float sample : samples) {
2529 mean += sample;
2530 }
2531 mean /= static_cast<double>(samples.size());
2532 for (float& sample : samples) {
2533 sample = static_cast<float>(sample - mean);
2534 }

2536 // If the capture came in too hot, scale it before it ever hits clip playback.
2537 float peak = analyzePeak(samples);
2538 constexpr float targetPeak = 0.85f;
2539 if (peak > targetPeak && peak > 0.0f) {
2540 const float scale = targetPeak / peak;
