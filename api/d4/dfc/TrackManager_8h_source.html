372 void processInput(const float* input, uint32_t frames, AudioTelemetry* telemetry = nullptr) {
373 if (!m_isCapturing.load(std::memory_order_relaxed) || !input || frames == 0 || m_inputChannelCount <= 0) {
374 return;
375 }
376
377 // Increment writer count BEFORE reading snapshot index so
378 // finalizeCaptureSession() cannot destroy the snapshot while we read it.
379 m_recordingWriters.fetch_add(1, std::memory_order_acq_rel);
380
381 // RAII guard: ensures writer count is always decremented on any exit path.
382 struct WriterGuard {
383 std::atomic<uint32_t>& writers;
384 ~WriterGuard() { writers.fetch_sub(1, std::memory_order_acq_rel); }
385 } writerGuard{m_recordingWriters};
386
387 // Read the immutable recording capture snapshot — avoids iterating the
388 // live m_channels / m_recordingCaptures containers on the RT thread.
389 const uint32_t snapIdx = m_activeRecordingCaptureSnapshot.load(std::memory_order_acquire);
390 const uint32_t routeCount = m_recordingCaptureRouteCounts[snapIdx].load(std::memory_order_acquire);
391 if (routeCount == 0) {
392 return;
393 }
394
395 const auto& routes = m_recordingCaptureSnapshots[snapIdx];
396
397 const double captureBeat = getCurrentTransportBeat();
398 const bool hasDeferredStart = m_hasDeferredRecordingStart.load(std::memory_order_acquire);
399 const double deferredStartBeat = m_deferredRecordingStartBeat.load(std::memory_order_relaxed);
400 const double bpm = std::max(1.0, m_playlistModel.getBPM());
401 const double sampleRate = std::max(1.0, m_inputSampleRate > 0.0 ? m_inputSampleRate : m_outputSampleRate);
402 bool capturedAnyChannel = false;
403 bool startedDeferredCapture = false;
404
405 for (uint32_t r = 0; r < routeCount; ++r) {
406 RecordingCapture* capture = routes[r].capture;
407 if (!capture) {
408 continue;
409 }
410 const int requestedInput = routes[r].inputIndex;
411 if (requestedInput < 0 && requestedInput != -2) {
412 continue;
413 }
414
415 const size_t capacity = capture->capacity;
416 if (capacity == 0 || !capture->samples) {
417 continue;
418 }
419
420 size_t head = capture->headIndex.load(std::memory_order_relaxed);
421 size_t size = capture->size.load(std::memory_order_relaxed);
422 double startBeat = capture->startBeat.load(std::memory_order_relaxed);
423 uint64_t droppedFrames = 0;
424 uint32_t startFrame = 0;
425 double effectiveCaptureBeat = captureBeat;
426
427 if (hasDeferredStart) {
428 const double beatsUntilCapture = deferredStartBeat - captureBeat;
429 if (beatsUntilCapture > 0.0) {
430 const double secondsUntilCapture = beatsUntilCapture * 60.0 / bpm;
431 const uint32_t skippedFrames = static_cast<uint32_t>(std::ceil(secondsUntilCapture * sampleRate));
432 if (skippedFrames >= frames) {
433 continue;
434 }
435 startFrame = skippedFrames;
436 effectiveCaptureBeat = deferredStartBeat;
437 } else {
438 effectiveCaptureBeat = deferredStartBeat;
439 }
440 }
441
442 auto appendSample = [&](float sample) {
443 size_t writeIndex = (head + size) % capacity;
444 capture->samples[writeIndex].store(sample, std::memory_order_relaxed);
445 if (size < capacity) {
446 ++size;
447 } else {
448 head = (head + 1) % capacity;
449 ++droppedFrames;
450 }
451 };
452
453 if (requestedInput == -2) {
454 const float channelScale = 1.0f / static_cast<float>(m_inputChannelCount);
455 for (uint32_t frame = startFrame; frame < frames; ++frame) {
456 const size_t baseIndex = static_cast<size_t>(frame) * static_cast<size_t>(m_inputChannelCount);
457 float mixedSample = 0.0f;
458 for (int ch = 0; ch < m_inputChannelCount; ++ch) {
459 mixedSample += input[baseIndex + static_cast<size_t>(ch)];
460 }
461 appendSample(mixedSample * channelScale);
462 }
463 } else {
464 const int inputIndex = requestedInput;
465 if (inputIndex < 0 || inputIndex >= m_inputChannelCount) {
466 continue;
467 }
468
469 for (uint32_t frame = startFrame; frame < frames; ++frame) {
470 const size_t sampleIndex = static_cast<size_t>(frame) * static_cast<size_t>(m_inputChannelCount) +
471 static_cast<size_t>(inputIndex);
472 appendSample(input[sampleIndex]);
473 }
474 }
475 bool expectedStart = false;
476 if (capture->hasStarted.compare_exchange_strong(expectedStart, true, std::memory_order_acq_rel)) {
477 capture->startBeat.store(effectiveCaptureBeat, std::memory_order_release);
478 if (hasDeferredStart) {
479 startedDeferredCapture = true;
480 }
481 }
482 if (droppedFrames > 0) {
483 startBeat += framesToBeats(static_cast<double>(droppedFrames));
484 capture->startBeat.store(startBeat, std::memory_order_release);
485 }
486 capture->headIndex.store(head, std::memory_order_release);
487 capture->size.store(size, std::memory_order_release);
488 capture->totalCapturedFrames.fetch_add(frames - startFrame, std::memory_order_relaxed);
489 capturedAnyChannel = true;
490 }
491
492 if (startedDeferredCapture) {
493 clearDeferredRecordingStartBeat();
494 }
495
496 if (!capturedAnyChannel && !m_recordingNoArmLogged) {
497 (void)telemetry;
498 m_recordingNoArmLogged = true;
499 }
500 // writerGuard destructor handles m_recordingWriters.fetch_sub
501 }

503 void publishInputMonitoringSnapshot() {
504 std::array<RtInputMonitorRoute, 32> routes{};
505 uint32_t count = 0;
506 for (const auto& channel : m_channels) {
507 if (!channel || !channel->isArmed() || !channel->isMonitoringEnabled()) {
508 continue;
509 }
510 const int requestedInput = channel->getInputChannelIndex();
511 if (requestedInput == -1) {
512 continue;
513 }
514 if (count >= routes.size()) {
515 break;
516 }
517 routes[count++] = RtInputMonitorRoute{requestedInput, channel->getVolume()};
518 }
519
520 const uint32_t inactive = 1u - m_activeInputMonitorSnapshot.load(std::memory_order_relaxed);
521 m_inputMonitorSnapshots[inactive] = routes;
522 m_inputMonitorRouteCounts[inactive].store(count, std::memory_order_release);
523 m_activeInputMonitorSnapshot.store(inactive, std::memory_order_release);
524 }

531 void updateInputDiagnostics(const float* input, uint32_t frames) {
532 if (!input || frames == 0 || m_inputChannelCount <= 0) {
533 return;
534 }
535
536 const int maxTrackedInputs = static_cast<int>(m_inputPeaks.size());
537 const int trackedInputs = std::min(m_inputChannelCount, maxTrackedInputs);
538
539 for (int ch = 0; ch < trackedInputs; ++ch) {
540 float peak = 0.0f;
541 for (uint32_t frame = 0; frame < frames; ++frame) {
542 const size_t sampleIndex =
543 static_cast<size_t>(frame) * static_cast<size_t>(m_inputChannelCount) + static_cast<size_t>(ch);
544 peak = std::max(peak, std::abs(input[sampleIndex]));
545 }
546 m_inputPeaks[static_cast<size_t>(ch)].store(peak, std::memory_order_relaxed);
547 }
548
549 for (int ch = trackedInputs; ch < maxTrackedInputs; ++ch) {
550 m_inputPeaks[static_cast<size_t>(ch)].store(0.0f, std::memory_order_relaxed);
551 }
552 }

577 void mixInputMonitoring(const float* input, float* output, uint32_t frames, uint32_t outputChannels) const {
578 if (!input || !output || frames == 0 || outputChannels == 0 || m_inputChannelCount <= 0) {
579 return;
580 }
581
582 const uint32_t snapshotIndex = m_activeInputMonitorSnapshot.load(std::memory_order_acquire);
583 const uint32_t monitoredCount = m_inputMonitorRouteCounts[snapshotIndex].load(std::memory_order_acquire);
584 if (monitoredCount == 0) {
585 return;
586 }
587
588 const float monitorMixScale = 0.85f / static_cast<float>(monitoredCount);
589 for (uint32_t frame = 0; frame < frames; ++frame) {
590 const size_t inputBaseIndex = static_cast<size_t>(frame) * static_cast<size_t>(m_inputChannelCount);
591 float monitoredSample = 0.0f;
592
593 const auto& routes = m_inputMonitorSnapshots[snapshotIndex];
594 for (uint32_t routeIndex = 0; routeIndex < monitoredCount && routeIndex < routes.size(); ++routeIndex) {
595 const int requestedInput = routes[routeIndex].inputIndex;
596 float sample = 0.0f;
597
598 if (requestedInput == -2) {
599 const float channelScale = 1.0f / static_cast<float>(m_inputChannelCount);
600 for (int ch = 0; ch < m_inputChannelCount; ++ch) {
601 sample += input[inputBaseIndex + static_cast<size_t>(ch)];
602 }
603 sample *= channelScale;
604 } else if (requestedInput >= 0 && requestedInput < m_inputChannelCount) {
605 sample = input[inputBaseIndex + static_cast<size_t>(requestedInput)];
606 }
607
608 monitoredSample += sample * routes[routeIndex].volume;
609 }
610
611 monitoredSample *= monitorMixScale;
612 const size_t outputBaseIndex = static_cast<size_t>(frame) * static_cast<size_t>(outputChannels);
613 output[outputBaseIndex] += monitoredSample;
614 if (outputChannels > 1) {
615 output[outputBaseIndex + 1] += monitoredSample;
616 }
617 for (uint32_t ch = 2; ch < outputChannels; ++ch) {
618 output[outputBaseIndex + static_cast<size_t>(ch)] += monitoredSample;
619 }
620 }

1110 void finalizeCaptureSession() {
1111 m_recordingCaptureAccepting.store(false, std::memory_order_release);
1112 const auto waitDeadline = std::chrono::steady_clock::now() + std::chrono::milliseconds(250);
1113 bool writersDrained = m_recordingWriters.load(std::memory_order_acquire) == 0;
1114 while (!writersDrained && std::chrono::steady_clock::now() < waitDeadline) {
1115 std::this_thread::sleep_for(std::chrono::milliseconds(1));
1116 writersDrained = m_recordingWriters.load(std::memory_order_acquire) == 0;
1117 }
1118 if (!writersDrained) {
1119 Log::warning("[TrackManager] Timed out waiting for recording writers to drain before finalizing capture. "
1120 "Deferring teardown — retry on next disarm.");
1121 // Leave m_isCapturing, snapshots, captures, and session fields intact.
1122 // processInput() callers still in-flight continue to hold valid pointers.
1123 m_recordingCaptureAccepting.store(true, std::memory_order_release);
1124 return;
1125 }
1126
1127 m_isCapturing.store(false, std::memory_order_relaxed);
1128
1129 // Publish empty snapshot so RT thread stops referencing captures.
1130 m_recordingCaptureRouteCounts[0].store(0, std::memory_order_release);
1131 m_recordingCaptureRouteCounts[1].store(0, std::memory_order_release);
1132 m_activeRecordingCaptureSnapshot.store(0, std::memory_order_release);

1136 bool sessionUsesPlacementOverride = false;
1137 {
1138 std::lock_guard<std::mutex> lock(m_recordingMutex);
1139 captures = std::move(m_recordingCaptures);
1140 sessionStartBeat = m_recordingSessionStartBeat;
1141 sessionUsesPlacementOverride = m_recordingSessionUsesPlacementOverride;

1159 void commitRecordingTake(uint32_t channelId, const RecordingCapture& capture, double fallbackStartBeat,
1160 bool forcePlacementStartBeat) {
1161 std::vector<float> capturedSamples = copyCaptureSamples(capture);
1162 if (capturedSamples.empty()) {
1163 return;
1164 }
1165
1166 const size_t channelIndex = findChannelIndexById(channelId);
1167 if (channelIndex == static_cast<size_t>(-1)) {
1168 Log::warning("[TrackManager] Could not resolve lane for recorded track " + std::to_string(channelId));
1169 return;
1170 }
1171
1172 PlaylistLaneID laneId = m_playlistModel.getLaneId(channelIndex);
1173 if (!laneId.isValid()) {
1174 Log::warning("[TrackManager] Invalid lane target for recorded track " + std::to_string(channelId));
1175 return;
1176 }
1177
1178 std::vector<float> conditionedSamples = std::move(capturedSamples);
1179 const float rawPeak = analyzePeak(conditionedSamples);
1180 conditionRecordedTakeSamples(conditionedSamples);
1181 const float conditionedPeak = analyzePeak(conditionedSamples);
1182
1183 const double captureStartBeat = capture.startBeat.load(std::memory_order_acquire);
1184 const double startBeat = forcePlacementStartBeat
1185 ? fallbackStartBeat
1186 : (captureStartBeat > 0.0 ? captureStartBeat : fallbackStartBeat);
1187 const double durationBeats = framesToBeats(static_cast<double>(conditionedSamples.size()));
1188 if (durationBeats <= 0.0) {
1189 return;
1190 }
1191 const double durationSeconds =
1192 static_cast<double>(conditionedSamples.size()) / std::max(1.0, m_inputSampleRate);
1193 const float playbackGain = computeRecordedTakeGain(conditionedSamples);
1194
1195 auto buffer = std::make_shared<AudioBufferData>();
1196 buffer->sampleRate = static_cast<uint32_t>(std::max(1.0, m_inputSampleRate));
1197 buffer->numChannels = 2;
1198 buffer->numFrames = conditionedSamples.size();
1199 buffer->interleavedData.resize(conditionedSamples.size() * 2);
1200 for (size_t i = 0; i < conditionedSamples.size(); ++i) {
1201 const float sample = conditionedSamples[i];
1202 buffer->interleavedData[i * 2] = sample;
1203 buffer->interleavedData[i * 2 + 1] = sample;
1204 }
1205
1206 const std::string takePath = buildRecordingTakePath(channelId);
1207 if (!writeRecordedTakeWav(takePath, *buffer)) {
1208 Log::error("[TrackManager] Failed to write recorded take: " + takePath);
1209 return;
1210 }
1211
1212 const std::string takeName = std::filesystem::path(takePath).stem().string();
1213 ClipSourceID sourceId = m_sourceManager.createRecordedSource(takePath, takeName, buffer);
1214 if (!sourceId.isValid()) {
1215 Log::error("[TrackManager] Failed to register recorded source for: " + takePath);
1216 return;
1217 }
1218
1219 AudioSlicePayload payload;
1220 payload.audioSourceId = sourceId;
1221 payload.durationSeconds = durationSeconds;
1222 payload.slices.push_back({0.0, static_cast<double>(buffer->numFrames)});
1223
1224 PatternID patternId = m_patternManager.createAudioPattern(takeName, durationBeats, payload);
1225 if (!patternId.isValid()) {
1226 Log::error("[TrackManager] Failed to create audio pattern for recorded take.");
1227 return;
1228 }
1229
1230 ClipInstance clip;
1231 clip.id = ClipInstanceID::generate();
1232 clip.name = takeName;
1233 clip.startBeat = startBeat;
1234 clip.durationBeats = durationBeats;
1235 clip.durationSeconds = durationSeconds;
1236 clip.patternId = patternId;
1237 clip.sourceId = patternId.value;
1238 clip.edits.gain = playbackGain;
1239 clip.edits.gainLinear = playbackGain;
1240
1241 auto cmd = std::make_shared<AddClipCommand>(m_playlistModel, laneId, clip);
1242 m_commandHistory.pushAndExecute(cmd);
1243 requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged);
1244 m_modified.store(true, std::memory_order_relaxed);
1245
1246 Log::info("[TrackManager] Recorded take committed: " + takePath + " on track " + std::to_string(channelId) +
1247 " at beat " + std::to_string(startBeat) + " with raw peak " + std::to_string(rawPeak) +
1248 ", conditioned peak " + std::to_string(conditionedPeak) + ", clip gain " +
1249 std::to_string(playbackGain));
1250 }

1290 bool writeRecordedTakeWav(const std::string& path, const AudioBufferData& buffer) const {
1291 std::ofstream file(path, std::ios::binary | std::ios::trunc);
1292 if (!file.is_open()) {
1293 return false;
1294 }
1295
1296 const uint16_t audioFormat = 3; // IEEE float
1297 if (buffer.numChannels == 0 || buffer.numChannels > std::numeric_limits<uint16_t>::max() ||
1298 buffer.sampleRate == 0 ||
1299 buffer.interleavedData.size() > (std::numeric_limits<uint32_t>::max() / sizeof(float)) ||
1300 buffer.numFrames > std::numeric_limits<uint32_t>::max()) {
1301 return false;
1302 }
1303 const uint16_t numChannels = static_cast<uint16_t>(buffer.numChannels);
1304 const uint32_t sampleRate = buffer.sampleRate;
1305 const uint16_t bitsPerSample = 32;
1306 const uint16_t blockAlign = static_cast<uint16_t>(numChannels * (bitsPerSample / 8));
1307 if (sampleRate > std::numeric_limits<uint32_t>::max() / blockAlign) {
1308 return false;
1309 }
1310 const uint32_t byteRate = sampleRate * blockAlign;
1311 const uint32_t dataSize = static_cast<uint32_t>(buffer.interleavedData.size() * sizeof(float));
1312 const uint32_t sampleCount = static_cast<uint32_t>(buffer.numFrames);
1313 const uint32_t factChunkSize = 4;
1314 if (dataSize > std::numeric_limits<uint32_t>::max() - 48u) {
1315 return false;
1316 }
1317 const uint32_t riffChunkSize = 48u + dataSize;
1318
1319 file.write("RIFF", 4);
1320 file.write(reinterpret_cast<const char*>(&riffChunkSize), sizeof(riffChunkSize));
1321 file.write("WAVE", 4);
1322 file.write("fmt ", 4);
1323
1324 const uint32_t fmtChunkSize = 16;
1325 file.write(reinterpret_cast<const char*>(&fmtChunkSize), sizeof(fmtChunkSize));
1326 file.write(reinterpret_cast<const char*>(&audioFormat), sizeof(audioFormat));
1327 file.write(reinterpret_cast<const char*>(&numChannels), sizeof(numChannels));
1328 file.write(reinterpret_cast<const char*>(&sampleRate), sizeof(sampleRate));
1329 file.write(reinterpret_cast<const char*>(&byteRate), sizeof(byteRate));
1330 file.write(reinterpret_cast<const char*>(&blockAlign), sizeof(blockAlign));
1331 file.write(reinterpret_cast<const char*>(&bitsPerSample), sizeof(bitsPerSample));
1332
1333 file.write("fact", 4);
1334 file.write(reinterpret_cast<const char*>(&factChunkSize), sizeof(factChunkSize));
1335 file.write(reinterpret_cast<const char*>(&sampleCount), sizeof(sampleCount));
1336
1337 file.write("data", 4);
1338 file.write(reinterpret_cast<const char*>(&dataSize), sizeof(dataSize));
1339 file.write(reinterpret_cast<const char*>(buffer.interleavedData.data()), dataSize);
1340 return file.good();
1341 }

1343 float computeRecordedTakeGain(const std::vector<float>& samples) const {
1344 float peak = analyzePeak(samples);
1345 if (peak <= 0.0f) {
1346 return 1.0f;
1347 }
1348
1349 constexpr float targetPeak = 0.35f;
1350 return std::clamp(targetPeak / peak, 0.18f, 1.0f);
1351 }

1353 float analyzePeak(const std::vector<float>& samples) const {
1354 float peak = 0.0f;
1355 for (float sample : samples) {
1356 peak = std::max(peak, std::abs(sample));
1357 }
1358 return peak;
1359 }

1361 void conditionRecordedTakeSamples(std::vector<float>& samples) const {
1362 if (samples.empty()) {
1363 return;
1364 }
1365
1366 // Remove DC bias first so the recorded waveform sits more naturally around zero.
1367 double mean = 0.0;
1368 for (float sample : samples) {
1369 mean += sample;
1370 }
1371 mean /= static_cast<double>(samples.size());
1372 for (float& sample : samples) {
1373 sample = static_cast<float>(sample - mean);
1374 }
1375
1376 // If the capture came in too hot, scale it before it ever hits clip playback.
1377 float peak = analyzePeak(samples);
1378 constexpr float targetPeak = 0.85f;
1379 if (peak > targetPeak && peak > 0.0f) {
1380 const float scale = targetPeak / peak;
1381 for (float& sample : samples) {
1382 sample *= scale;
1383 }
1384 }
1385 }
