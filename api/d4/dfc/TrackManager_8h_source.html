568 bool canRouteTo(uint32_t sourceId, uint32_t targetId) const {
569 // Master has two spellings: user/model space (0, MASTER_MIXER_CHANNEL_ID)
570 // and engine space (0xFFFFFFFF). Both are the same terminal sink.
571 constexpr uint32_t kEngineMaster = 0xFFFFFFFFu;
572 if (sourceId == MASTER_MIXER_CHANNEL_ID || sourceId == kEngineMaster) {
573 return false;
574 }
575 if (targetId == MASTER_MIXER_CHANNEL_ID || targetId == kEngineMaster) {
576 return true;
577 }
578 if (sourceId == targetId) {
579 return false;
580 }
581 if (!getChannelById(sourceId) || !getChannelById(targetId)) {
582 return false;
583 }
584
585 std::vector<uint32_t> stack{targetId};
586 std::unordered_set<uint32_t> visited;
587 while (!stack.empty()) {
588 const uint32_t current = stack.back();
589 stack.pop_back();
590 if (current == sourceId) {
591 return false;
592 }
593 if (!visited.insert(current).second) {
594 continue;
595 }
596 const MixerChannel* channel = getChannelById(current);
597 if (!channel) {
598 continue;
599 }
600 const uint32_t mainOutput = channel->getMainOutputId();
601 if (mainOutput != kEngineMaster) {
602 stack.push_back(mainOutput);
603 }
604 for (const auto& send : channel->getSends()) {
605 if (send.mute || send.sidechainOnly) {
606 continue;
607 }
608 if (send.targetChannelId != kEngineMaster) {
609 stack.push_back(send.targetChannelId);
610 }
611 }
612 }
613 return true;
614 }

783 bool getRecordingDataPeaks(uint64_t trackId, uint32_t maxPoints, std::vector<float>& peaksOut,
784 double& startBeat, size_t& ringSamplesOut) {
785 if (maxPoints == 0) {
786 return false;
787 }
788 std::lock_guard<std::mutex> lock(m_recordingMutex);
789 auto it = m_recordingCaptures.find(trackId);
790 if (it == m_recordingCaptures.end() || !it->second) {
791 return false;
792 }
793 RecordingCapture* capture = it->second.get();
794 const size_t size = capture->size.load(std::memory_order_acquire);
795 if (size == 0) {
796 return false;
797 }
798 startBeat = capture->startBeat.load(std::memory_order_acquire);
799 ringSamplesOut = size;
800
801 peaksOut.clear();
802 peaksOut.reserve(static_cast<size_t>(maxPoints) * 2);
803 const size_t head = capture->headIndex.load(std::memory_order_relaxed);
804 // Ceiling division: every sample lands in a bucket so the ring tail
805 // can never be dropped by rounding.
806 const size_t stride = (size + maxPoints - 1) / maxPoints;
807 if (stride == 0) {
808 return false;
809 }
810 for (size_t offset = 0; offset < size && peaksOut.size() < static_cast<size_t>(maxPoints) * 2;
811 offset += stride) {
812 const size_t span = std::min(stride, size - offset);
813 // Seed from the first sample: zero-init would report a false bound
814 // for positive-only or negative-only buckets. Non-finite input
815 // samples (driver glitches) are skipped rather than poisoning the
816 // pair — they are a render/capture concern, not signal.
817 bool seeded = false;
818 float lo = 0.0f;
819 float hi = 0.0f;
820 for (size_t k = 0; k < span; ++k) {
821 const float v =
822 capture->samples[(head + offset + k) % capture->capacity].load(std::memory_order_relaxed);
823 if (!std::isfinite(v)) {
824 continue;
825 }
826 if (!seeded) {
827 lo = v;
828 hi = v;
829 seeded = true;
830 continue;
831 }
832 lo = std::min(lo, v);
833 hi = std::max(hi, v);
834 }
835 if (!seeded) {
836 // All-non-finite span: emit a NaN pair as an explicit skipped
837 // slot. Dropping the pair entirely would shift every later
838 // bucket's screen position left (#854 round 2) — the renderer
839 // ignores non-finite pairs, so timing stays correct.
840 constexpr float nanPair = std::numeric_limits<float>::quiet_NaN();
841 peaksOut.push_back(nanPair);
842 peaksOut.push_back(nanPair);
843 continue;
844 }
845 peaksOut.push_back(lo);
846 peaksOut.push_back(hi);
847 }
848 return !peaksOut.empty();
849 }

988 static constexpr uint64_t kUnknownTransportFrame = UINT64_MAX;
989 void processInput(const float* input, uint32_t frames, AudioTelemetry* telemetry = nullptr,
990 uint64_t transportFrame = kUnknownTransportFrame) {
991 if (!m_isCapturing.load(std::memory_order_relaxed) || !input || frames == 0 || m_inputChannelCount <= 0) {
992 return;
993 }
994
995 // Increment writer count BEFORE reading snapshot index so
996 // finalizeCaptureSession() cannot destroy the snapshot while we read it.
997 m_recordingWriters.fetch_add(1, std::memory_order_acq_rel);
998
999 // RAII guard: ensures writer count is always decremented on any exit path.
1000 struct WriterGuard {
1001 std::atomic<uint32_t>& writers;
1002 ~WriterGuard() { writers.fetch_sub(1, std::memory_order_acq_rel); }
1003 } writerGuard{m_recordingWriters};
1004
1005 // Read the immutable recording capture snapshot — avoids iterating the
1006 // live m_channels / m_recordingCaptures containers on the RT thread.
1007 const uint32_t snapIdx = m_activeRecordingCaptureSnapshot.load(std::memory_order_acquire);
1008 const uint32_t routeCount = m_recordingCaptureRouteCounts[snapIdx].load(std::memory_order_acquire);
1009 if (routeCount == 0) {
1010 return;
1011 }
1012
1013 const auto& routes = m_recordingCaptureSnapshots[snapIdx];
1014
1015 const double bpm = std::max(1.0, m_playlistModel.getBPM());
1016 const double sampleRate = std::max(1.0, m_inputSampleRate > 0.0 ? m_inputSampleRate : m_outputSampleRate);
1017 // T-6: prefer the engine's authoritative frame over the UI-cached
1018 // position for capture placement; the latter lags by audio buffers.
1019 const double captureBeat =
1020 (transportFrame != kUnknownTransportFrame)
1021 ? (static_cast<double>(transportFrame) / sampleRate) * bpm / 60.0
1022 : getCurrentTransportBeat();
1023 const bool hasDeferredStart = m_hasDeferredRecordingStart.load(std::memory_order_acquire);
1024 const double deferredStartBeat = m_deferredRecordingStartBeat.load(std::memory_order_relaxed);
1025 bool capturedAnyChannel = false;
1026 bool startedDeferredCapture = false;
1027
1028 for (uint32_t r = 0; r < routeCount; ++r) {
1029 RecordingCapture* capture = routes[r].capture;
1030 if (!capture) {
1031 continue;
1032 }
1033 const int requestedInput = routes[r].inputIndex;
1034 if (requestedInput < 0 && requestedInput != -2) {
1035 continue;
1036 }
1037
1038 const size_t capacity = capture->capacity;
1039 if (capacity == 0 || !capture->samples) {
1040 continue;
1041 }
1042
1043 size_t head = capture->headIndex.load(std::memory_order_relaxed);
1044 size_t size = capture->size.load(std::memory_order_relaxed);
1045 double startBeat = capture->startBeat.load(std::memory_order_relaxed);
1046 uint64_t droppedFrames = 0;
1047 uint32_t startFrame = 0;
1048 double effectiveCaptureBeat = captureBeat;
1049
1050 if (hasDeferredStart) {
1051 const double beatsUntilCapture = deferredStartBeat - captureBeat;
1052 if (beatsUntilCapture > 0.0) {
1053 const double secondsUntilCapture = beatsUntilCapture * 60.0 / bpm;
1054 const uint32_t skippedFrames = static_cast<uint32_t>(std::ceil(secondsUntilCapture * sampleRate));
1055 if (skippedFrames >= frames) {
1056 continue;
1057 }
1058 startFrame = skippedFrames;
1059 effectiveCaptureBeat = deferredStartBeat;
1060 } else {
1061 effectiveCaptureBeat = deferredStartBeat;
1062 }
1063 }
1064
1065 auto appendSample = [&](float sample) {
1066 if (!std::isfinite(sample)) {
1067 sample = 0.0f;
1068 }
1069 size_t writeIndex = (head + size) % capacity;
1070 capture->samples[writeIndex].store(sample, std::memory_order_relaxed);
1071 if (size < capacity) {
1072 ++size;
1073 } else {
1074 head = (head + 1) % capacity;
1075 ++droppedFrames;
1076 }
1077 };
1078
1079 if (requestedInput == -2) {
1080 const float channelScale = 1.0f / static_cast<float>(m_inputChannelCount);
1081 for (uint32_t frame = startFrame; frame < frames; ++frame) {
1082 const size_t baseIndex = static_cast<size_t>(frame) * static_cast<size_t>(m_inputChannelCount);
1083 float mixedSample = 0.0f;
1084 for (int ch = 0; ch < m_inputChannelCount; ++ch) {
1085 mixedSample += input[baseIndex + static_cast<size_t>(ch)];
1086 }
1087 appendSample(mixedSample * channelScale);
1088 }
1089 } else {
1090 const int inputIndex = requestedInput;
1091 if (inputIndex < 0 || inputIndex >= m_inputChannelCount) {
1092 continue;
1093 }
1094
1095 for (uint32_t frame = startFrame; frame < frames; ++frame) {
1096 const size_t sampleIndex = static_cast<size_t>(frame) * static_cast<size_t>(m_inputChannelCount) +
1097 static_cast<size_t>(inputIndex);
1098 appendSample(input[sampleIndex]);
1099 }
1100 }
1101 bool expectedStart = false;
1102 if (capture->hasStarted.compare_exchange_strong(expectedStart, true, std::memory_order_acq_rel)) {
1103 capture->startBeat.store(effectiveCaptureBeat, std::memory_order_release);
1104 if (hasDeferredStart) {
1105 startedDeferredCapture = true;
1106 }
1107 }
1108 if (droppedFrames > 0) {
1109 startBeat += framesToBeats(static_cast<double>(droppedFrames));
1110 capture->startBeat.store(startBeat, std::memory_order_release);
1111 }
1112 capture->headIndex.store(head, std::memory_order_release);
1113 capture->size.store(size, std::memory_order_release);
1114 capture->totalCapturedFrames.fetch_add(frames - startFrame, std::memory_order_relaxed);
1115 capturedAnyChannel = true;
1116 }
1117
1118 if (startedDeferredCapture) {
1119 clearDeferredRecordingStartBeat();
1120 }
1121
1122 if (!capturedAnyChannel && !m_recordingNoArmLogged) {
1123 (void)telemetry;
1124 m_recordingNoArmLogged = true;
1125 }

1127 }

1203 void mixInputMonitoring(const float* input, float* output, uint32_t frames, uint32_t outputChannels) const {
1204 if (!input || !output || frames == 0 || outputChannels == 0 || m_inputChannelCount <= 0) {
1205 return;
1206 }
1207
1208 const uint32_t snapshotIndex = m_activeInputMonitorSnapshot.load(std::memory_order_acquire);
1209 const uint32_t monitoredCount = m_inputMonitorRouteCounts[snapshotIndex].load(std::memory_order_acquire);
1210 if (monitoredCount == 0) {
1211 return;
1212 }
1213
1214 // Unity centre gain per monitored input: the main path (strips) uses
1215 // the stereo-balance law since the strip pan-law fix (2026-08-14), and
1216 // the audition/preview parity contract keeps every listening surface
1217 // at the same reference level. Only the N-input normalization remains.
1218 const float monitorMixScale = 1.0f / static_cast<float>(monitoredCount);
1219 for (uint32_t frame = 0; frame < frames; ++frame) {
1220 const size_t inputBaseIndex = static_cast<size_t>(frame) * static_cast<size_t>(m_inputChannelCount);
1221 float monitoredSample = 0.0f;
1222
1223 const auto& routes = m_inputMonitorSnapshots[snapshotIndex];
1224 for (uint32_t routeIndex = 0; routeIndex < monitoredCount && routeIndex < routes.size(); ++routeIndex) {
1225 const int requestedInput = routes[routeIndex].inputIndex;
1226 float sample = 0.0f;
1227
1228 if (requestedInput == -2) {
1229 const float channelScale = 1.0f / static_cast<float>(m_inputChannelCount);
1230 for (int ch = 0; ch < m_inputChannelCount; ++ch) {
1231 sample += input[inputBaseIndex + static_cast<size_t>(ch)];
1232 }
1233 sample *= channelScale;
1234 } else if (requestedInput >= 0 && requestedInput < m_inputChannelCount) {
1235 sample = input[inputBaseIndex + static_cast<size_t>(requestedInput)];
1236 }
1237
1238 monitoredSample += sample * routes[routeIndex].volume;
1239 }
1240
1241 monitoredSample *= monitorMixScale;
1242 const size_t outputBaseIndex = static_cast<size_t>(frame) * static_cast<size_t>(outputChannels);
1243 output[outputBaseIndex] += monitoredSample;
1244 if (outputChannels > 1) {
1245 output[outputBaseIndex + 1] += monitoredSample;
1246 }
1247 for (uint32_t ch = 2; ch < outputChannels; ++ch) {
1248 output[outputBaseIndex + static_cast<size_t>(ch)] += monitoredSample;
1249 }
1250 }

1263 void refreshTimelinePatternInstances() {
1264 if (!m_isPlaying.load(std::memory_order_relaxed) || m_patternMode.load(std::memory_order_relaxed)) {
1265 return;
1266 }
1267 // Incremental, NOT clear-and-reschedule: a clear dropped every sounding note's gate and queued note-off
1268 // and re-queued from the UI's lagging playhead (moving one clip cut other clips' notes).
1269 // Each clip keeps its slot (held-note tracking is per slot), a slot keeps its
1270 // frontier, and the content-edit refill re-queues from the engine's real playhead.
1271 const auto instances = m_playlistModel.collectMidiClipInstances(m_patternManager);
1272 std::unordered_map<ClipInstanceID, uint32_t> nextSlots;
1273 std::array<bool, 256> used{};
1274 used[0] = used[1] = true; // 0 unused; 1 is Arsenal's single-pattern slot
1275 for (const auto& [clip, slot] : m_timelineSlots) {
1276 (void)clip;
1277 used[slot] = true; // survivors keep theirs; removed ones are free from the NEXT refresh
1278 }
1279 std::vector<const MidiClipPlaybackInstance*> fresh;
1280 for (const auto& instance : instances) {
1281 if (!instance.isValid()) {
1282 continue;
1283 }
1284 auto it = m_timelineSlots.find(instance.clipId);
1285 if (it == m_timelineSlots.end()) {
1286 fresh.push_back(&instance);
1287 continue;
1288 }
1289 nextSlots[instance.clipId] = it->second;
1290 m_patternPlaybackEngine.updatePatternInstance(instance.patternId, instance.patternStartBeat(), it->second,
1291 instance.sourceOffsetBeats, instance.durationBeats);
1292 }
1293 for (const auto& [clip, slot] : m_timelineSlots) {
1294 if (nextSlots.find(clip) == nextSlots.end()) {
1295 // Stays marked used for THIS refresh: the content-edit refill still has to
1296 // release the removed clip's held notes by slot, and a new clip reusing the
1297 // slot now would be mistaken for them.
1298 m_patternPlaybackEngine.removePatternInstance(slot);
1299 }
1300 }
1301 uint32_t cursor = 2;
1302 for (const auto* instance : fresh) {
1303 while (cursor < used.size() && used[cursor]) {
1304 ++cursor;
1305 }
1306 if (cursor >= used.size()) {
1307 Log::warning("[TrackManager] Too many timeline MIDI clip instances to schedule; truncating at 254");
1308 break;
1309 }
1310 used[cursor] = true;
1311 nextSlots[instance->clipId] = cursor;
1312 m_patternPlaybackEngine.updatePatternInstance(instance->patternId, instance->patternStartBeat(), cursor,
1313 instance->sourceOffsetBeats, instance->durationBeats);
1314 }
1315 m_timelineSlots = std::move(nextSlots);
1316 // Record what the slots now play, with the timing just handed to the scheduler.
1317 m_scheduledTimelineInstances.clear();
1318 for (const auto& instance : instances) {
1319 if (m_timelineSlots.find(instance.clipId) != m_timelineSlots.end()) {
1320 m_scheduledTimelineInstances.push_back(instance);
1321 }
1322 }
1323 m_patternPlaybackEngine.patternContentEdited();
1324 m_timelineRescheduleCount.fetch_add(1, std::memory_order_relaxed);
1325 }
