127 MixerChannel* addChannelWithId(const std::string& name, uint32_t requestedId) {
128 if (reportRealtimeMisuse("TrackManager::addChannel")) {
129 return nullptr;
130 }
131 // IDs start at 1 to avoid collision with Master (ID 0). A duplicate or
132 // invalid persisted ID is replaced instead of aliasing two live routes.
133 constexpr uint32_t invalidChannelId = std::numeric_limits<uint32_t>::max();
134 uint32_t channelId = requestedId;
135 if (channelId == 0 || channelId == invalidChannelId || getChannelById(channelId) != nullptr) {
136 while (m_nextChannelId != invalidChannelId && getChannelById(m_nextChannelId) != nullptr) {
137 ++m_nextChannelId;
138 }
139 if (m_nextChannelId == invalidChannelId) {
140 return nullptr;
141 }
142 channelId = m_nextChannelId++;
143 } else {
144 m_nextChannelId = std::max(m_nextChannelId, channelId + 1);
145 }
146 auto channel = std::make_unique<MixerChannel>(
147 // "Insert" is reserved for effect slots — a mixer strip is a Channel.
148 name.empty() ? "Channel " + std::to_string(channelId) : name, channelId);
149 channel->setCommandSink(m_commandSink);
150 channel->setInputMonitoringStateChangedCallback([this]() { publishInputMonitoringSnapshot(); });
151 if (m_channelPrepareCallback) {
152 m_channelPrepareCallback(*channel);
153 }
154 auto* raw = channel.get();
155 m_channels.push_back(std::move(channel));
156 requestAudioGraphRebuild(GraphDirtyReason::TrackStructureChanged);
157 m_modified.store(true, std::memory_order_relaxed);
158
159 // Rebuild channel slot map
160 if (!m_channelSlotMap) {
161 m_channelSlotMap = std::make_shared<ChannelSlotMap>();
162 }
163 m_channelSlotMap->rebuild(m_channels);
164 publishInputMonitoringSnapshot();
165
166 return raw;
167 }

592 void processInput(const float* input, uint32_t frames, AudioTelemetry* telemetry = nullptr) {
593 if (!m_isCapturing.load(std::memory_order_relaxed) || !input || frames == 0 || m_inputChannelCount <= 0) {
594 return;
595 }
596
597 // Increment writer count BEFORE reading snapshot index so
598 // finalizeCaptureSession() cannot destroy the snapshot while we read it.
599 m_recordingWriters.fetch_add(1, std::memory_order_acq_rel);
600
601 // RAII guard: ensures writer count is always decremented on any exit path.
602 struct WriterGuard {
603 std::atomic<uint32_t>& writers;
604 ~WriterGuard() { writers.fetch_sub(1, std::memory_order_acq_rel); }
605 } writerGuard{m_recordingWriters};
606
607 // Read the immutable recording capture snapshot — avoids iterating the
608 // live m_channels / m_recordingCaptures containers on the RT thread.
609 const uint32_t snapIdx = m_activeRecordingCaptureSnapshot.load(std::memory_order_acquire);
610 const uint32_t routeCount = m_recordingCaptureRouteCounts[snapIdx].load(std::memory_order_acquire);
611 if (routeCount == 0) {
612 return;
613 }
614
615 const auto& routes = m_recordingCaptureSnapshots[snapIdx];
616
617 const double captureBeat = getCurrentTransportBeat();
618 const bool hasDeferredStart = m_hasDeferredRecordingStart.load(std::memory_order_acquire);
619 const double deferredStartBeat = m_deferredRecordingStartBeat.load(std::memory_order_relaxed);
620 const double bpm = std::max(1.0, m_playlistModel.getBPM());
621 const double sampleRate = std::max(1.0, m_inputSampleRate > 0.0 ? m_inputSampleRate : m_outputSampleRate);
622 bool capturedAnyChannel = false;
623 bool startedDeferredCapture = false;
624
625 for (uint32_t r = 0; r < routeCount; ++r) {
626 RecordingCapture* capture = routes[r].capture;
627 if (!capture) {
628 continue;
629 }
630 const int requestedInput = routes[r].inputIndex;
631 if (requestedInput < 0 && requestedInput != -2) {
632 continue;
633 }
634
635 const size_t capacity = capture->capacity;
636 if (capacity == 0 || !capture->samples) {
637 continue;
638 }
639
640 size_t head = capture->headIndex.load(std::memory_order_relaxed);
641 size_t size = capture->size.load(std::memory_order_relaxed);
642 double startBeat = capture->startBeat.load(std::memory_order_relaxed);
643 uint64_t droppedFrames = 0;
644 uint32_t startFrame = 0;
645 double effectiveCaptureBeat = captureBeat;
646
647 if (hasDeferredStart) {
648 const double beatsUntilCapture = deferredStartBeat - captureBeat;
649 if (beatsUntilCapture > 0.0) {
650 const double secondsUntilCapture = beatsUntilCapture * 60.0 / bpm;
651 const uint32_t skippedFrames = static_cast<uint32_t>(std::ceil(secondsUntilCapture * sampleRate));
652 if (skippedFrames >= frames) {
653 continue;
654 }
655 startFrame = skippedFrames;
656 effectiveCaptureBeat = deferredStartBeat;
657 } else {
658 effectiveCaptureBeat = deferredStartBeat;
659 }
660 }
661
662 auto appendSample = [&](float sample) {
663 size_t writeIndex = (head + size) % capacity;
664 capture->samples[writeIndex].store(sample, std::memory_order_relaxed);
665 if (size < capacity) {
666 ++size;
667 } else {
668 head = (head + 1) % capacity;
669 ++droppedFrames;
670 }
671 };
672
673 if (requestedInput == -2) {
674 const float channelScale = 1.0f / static_cast<float>(m_inputChannelCount);
675 for (uint32_t frame = startFrame; frame < frames; ++frame) {
676 const size_t baseIndex = static_cast<size_t>(frame) * static_cast<size_t>(m_inputChannelCount);
677 float mixedSample = 0.0f;
678 for (int ch = 0; ch < m_inputChannelCount; ++ch) {
679 mixedSample += input[baseIndex + static_cast<size_t>(ch)];
680 }
681 appendSample(mixedSample * channelScale);
682 }
683 } else {
684 const int inputIndex = requestedInput;
685 if (inputIndex < 0 || inputIndex >= m_inputChannelCount) {
686 continue;
687 }
688
689 for (uint32_t frame = startFrame; frame < frames; ++frame) {
690 const size_t sampleIndex = static_cast<size_t>(frame) * static_cast<size_t>(m_inputChannelCount) +
691 static_cast<size_t>(inputIndex);
692 appendSample(input[sampleIndex]);
693 }
694 }
695 bool expectedStart = false;
696 if (capture->hasStarted.compare_exchange_strong(expectedStart, true, std::memory_order_acq_rel)) {
697 capture->startBeat.store(effectiveCaptureBeat, std::memory_order_release);
698 if (hasDeferredStart) {
699 startedDeferredCapture = true;
700 }
701 }
702 if (droppedFrames > 0) {
703 startBeat += framesToBeats(static_cast<double>(droppedFrames));
704 capture->startBeat.store(startBeat, std::memory_order_release);
705 }
706 capture->headIndex.store(head, std::memory_order_release);
707 capture->size.store(size, std::memory_order_release);
708 capture->totalCapturedFrames.fetch_add(frames - startFrame, std::memory_order_relaxed);
709 capturedAnyChannel = true;
710 }
711
712 if (startedDeferredCapture) {
713 clearDeferredRecordingStartBeat();
714 }
715
716 if (!capturedAnyChannel && !m_recordingNoArmLogged) {
717 (void)telemetry;
718 m_recordingNoArmLogged = true;
719 }
720 // writerGuard destructor handles m_recordingWriters.fetch_sub
721 }

723 void publishInputMonitoringSnapshot() {
724 std::array<RtInputMonitorRoute, 32> routes{};
725 uint32_t count = 0;
726 for (const auto& channel : m_channels) {
727 if (!channel || !channel->isArmed() || !channel->isMonitoringEnabled()) {
728 continue;
729 }
730 const int requestedInput = channel->getInputChannelIndex();
731 if (requestedInput == -1) {
732 continue;
733 }
734 if (count >= routes.size()) {
735 break;
736 }
737 routes[count++] = RtInputMonitorRoute{requestedInput, channel->getVolume()};
738 }
739
740 const uint32_t inactive = 1u - m_activeInputMonitorSnapshot.load(std::memory_order_relaxed);
741 m_inputMonitorSnapshots[inactive] = routes;
742 m_inputMonitorRouteCounts[inactive].store(count, std::memory_order_release);
743 m_activeInputMonitorSnapshot.store(inactive, std::memory_order_release);
744 }

751 void updateInputDiagnostics(const float* input, uint32_t frames) {
752 if (!input || frames == 0 || m_inputChannelCount <= 0) {
753 return;
754 }
755
756 const int maxTrackedInputs = static_cast<int>(m_inputPeaks.size());
757 const int trackedInputs = std::min(m_inputChannelCount, maxTrackedInputs);
758
759 for (int ch = 0; ch < trackedInputs; ++ch) {
760 float peak = 0.0f;
761 for (uint32_t frame = 0; frame < frames; ++frame) {
762 const size_t sampleIndex =
763 static_cast<size_t>(frame) * static_cast<size_t>(m_inputChannelCount) + static_cast<size_t>(ch);
764 peak = std::max(peak, std::abs(input[sampleIndex]));
765 }
766 m_inputPeaks[static_cast<size_t>(ch)].store(peak, std::memory_order_relaxed);
767 }
768
769 for (int ch = trackedInputs; ch < maxTrackedInputs; ++ch) {
770 m_inputPeaks[static_cast<size_t>(ch)].store(0.0f, std::memory_order_relaxed);
771 }
772 }

797 void mixInputMonitoring(const float* input, float* output, uint32_t frames, uint32_t outputChannels) const {
798 if (!input || !output || frames == 0 || outputChannels == 0 || m_inputChannelCount <= 0) {
799 return;
800 }
801
802 const uint32_t snapshotIndex = m_activeInputMonitorSnapshot.load(std::memory_order_acquire);
803 const uint32_t monitoredCount = m_inputMonitorRouteCounts[snapshotIndex].load(std::memory_order_acquire);
804 if (monitoredCount == 0) {
805 return;
806 }
807
808 const float monitorMixScale = PanLaw::kEqualPowerCenterGain / static_cast<float>(monitoredCount);
809 for (uint32_t frame = 0; frame < frames; ++frame) {
810 const size_t inputBaseIndex = static_cast<size_t>(frame) * static_cast<size_t>(m_inputChannelCount);
811 float monitoredSample = 0.0f;
812
813 const auto& routes = m_inputMonitorSnapshots[snapshotIndex];
814 for (uint32_t routeIndex = 0; routeIndex < monitoredCount && routeIndex < routes.size(); ++routeIndex) {
815 const int requestedInput = routes[routeIndex].inputIndex;
816 float sample = 0.0f;
817
818 if (requestedInput == -2) {
819 const float channelScale = 1.0f / static_cast<float>(m_inputChannelCount);
820 for (int ch = 0; ch < m_inputChannelCount; ++ch) {
821 sample += input[inputBaseIndex + static_cast<size_t>(ch)];
822 }
823 sample *= channelScale;
824 } else if (requestedInput >= 0 && requestedInput < m_inputChannelCount) {
825 sample = input[inputBaseIndex + static_cast<size_t>(requestedInput)];
826 }
827
828 monitoredSample += sample * routes[routeIndex].volume;
829 }
830
831 monitoredSample *= monitorMixScale;
832 const size_t outputBaseIndex = static_cast<size_t>(frame) * static_cast<size_t>(outputChannels);
833 output[outputBaseIndex] += monitoredSample;
834 if (outputChannels > 1) {
835 output[outputBaseIndex + 1] += monitoredSample;
836 }
837 for (uint32_t ch = 2; ch < outputChannels; ++ch) {
838 output[outputBaseIndex + static_cast<size_t>(ch)] += monitoredSample;
839 }
840 }

1264 void scheduleTimelinePatternInstances(double playStartPositionSeconds) {
1265 const double bpm = std::max(1.0, m_playlistModel.getBPM());
1266 const double playStartBeat = playStartPositionSeconds * bpm / 60.0;
1267 const auto instances = m_playlistModel.collectMidiClipInstances(m_patternManager);
1268
1269 Log::info("[TimelinePattern] scheduling " + std::to_string(instances.size()) +
1270 " MIDI clip instances from beat " + std::to_string(playStartBeat));
1271
1272 uint32_t instanceId = 2; // Reserve 1 for Arsenal-focused single-pattern playback.
1273 for (const auto& instance : instances) {
1274 if (!instance.isValid()) {
1275 continue;
1276 }
1277 if (instance.endBeat() <= playStartBeat) {
1278 continue;
1279 }
1280 if (instanceId >= 256) {
1281 Log::warning("[TrackManager] Too many timeline MIDI clip instances to schedule; truncating at 254");
1282 break;
1283 }
1284
1285 std::string routeSummary = "no-pattern";
1286 if (auto* pattern = m_patternManager.getPattern(instance.patternId); pattern && pattern->isMidi()) {
1287 const auto& midi = std::get<MidiPayload>(pattern->payload);
1288 size_t noteCount = midi.notes.size();
1289 UnitID firstUnitId = noteCount > 0 ? midi.notes.front().unitId : 0;
1290 uint32_t firstMixerChannelId = MASTER_MIXER_CHANNEL_ID;
1291 if (firstUnitId != 0) {
1292 if (auto* unit = m_unitManager.getUnit(firstUnitId)) {
1293 firstMixerChannelId = unit->targetMixerChannelId;
1294 }
1295 }
1296 routeSummary = "notes=" + std::to_string(noteCount) + " firstUnit=" + std::to_string(firstUnitId) +
1297 " firstMixerChannelId=" + std::to_string(firstMixerChannelId);
1298 }
1299
1300 Log::info("[TimelinePattern] pattern=" + std::to_string(instance.patternId.value) +
1301 " clipStart=" + std::to_string(instance.startBeat) +
1302 " sourceOffset=" + std::to_string(instance.sourceOffsetBeats) +
1303 " schedStart=" + std::to_string(instance.patternStartBeat()) + " " + routeSummary);
1304 m_patternPlaybackEngine.schedulePatternInstance(instance.patternId, instance.patternStartBeat(),
1305 instanceId++, instance.sourceOffsetBeats,
1306 instance.durationBeats);
1307 }

1366 void finalizeCaptureSession() {
1367 m_recordingCaptureAccepting.store(false, std::memory_order_release);
1368 const auto waitDeadline = std::chrono::steady_clock::now() + std::chrono::milliseconds(250);
1369 bool writersDrained = m_recordingWriters.load(std::memory_order_acquire) == 0;
1370 while (!writersDrained && std::chrono::steady_clock::now() < waitDeadline) {
1371 std::this_thread::sleep_for(std::chrono::milliseconds(1));
1372 writersDrained = m_recordingWriters.load(std::memory_order_acquire) == 0;
1373 }
1374 if (!writersDrained) {
1375 Log::warning("[TrackManager] Timed out waiting for recording writers to drain before finalizing capture. "
1376 "Deferring teardown — retry on next disarm.");
1377 // Leave m_isCapturing, snapshots, captures, and session fields intact.
1378 // processInput() callers still in-flight continue to hold valid pointers.
1379 m_recordingCaptureAccepting.store(true, std::memory_order_release);
1380 return;
1381 }
1382
1383 m_isCapturing.store(false, std::memory_order_relaxed);
1384
1385 // Publish empty snapshot so RT thread stops referencing captures.
1386 m_recordingCaptureRouteCounts[0].store(0, std::memory_order_release);
1387 m_recordingCaptureRouteCounts[1].store(0, std::memory_order_release);
1388 m_activeRecordingCaptureSnapshot.store(0, std::memory_order_release);

1390 std::unordered_map<uint32_t, std::unique_ptr<RecordingCapture>> captures;
1391 double sessionStartBeat = 0.0;

1394 std::lock_guard<std::mutex> lock(m_recordingMutex);
1395 captures = std::move(m_recordingCaptures);
1396 sessionStartBeat = m_recordingSessionStartBeat;
1397 sessionUsesPlacementOverride = m_recordingSessionUsesPlacementOverride;
1398 m_recordingCaptures.clear();
1399 m_recordingSessionStartBeat = 0.0;
1400 m_recordingSessionUsesPlacementOverride = false;
1401 m_recordingNoArmLogged = false;
1402 }

1407
1408 for (const auto& [channelId, capture] : captures) {
1409 if (capture) {

1415 void commitRecordingTake(uint32_t channelId, const RecordingCapture& capture, double fallbackStartBeat,
1416 bool forcePlacementStartBeat) {
1417 std::vector<float> capturedSamples = copyCaptureSamples(capture);
1418 if (capturedSamples.empty()) {
1419 return;
1420 }
1421
1422 if (!getChannelById(channelId)) {
1423 Log::warning("[TrackManager] Could not resolve mixer insert for recorded take " +
1424 std::to_string(channelId));
1425 return;
1426 }
1427
1428 std::vector<float> conditionedSamples = std::move(capturedSamples);
1429 const float rawPeak = analyzePeak(conditionedSamples);
1430 conditionRecordedTakeSamples(conditionedSamples);
1431 const float conditionedPeak = analyzePeak(conditionedSamples);
1432
1433 const double captureStartBeat = capture.startBeat.load(std::memory_order_acquire);
1434 const double startBeat = forcePlacementStartBeat
1435 ? fallbackStartBeat
1436 : (captureStartBeat > 0.0 ? captureStartBeat : fallbackStartBeat);
1437 const double durationBeats = framesToBeats(static_cast<double>(conditionedSamples.size()));
1438 if (durationBeats <= 0.0) {
1439 return;
1440 }
1441 const double durationSeconds =
1442 static_cast<double>(conditionedSamples.size()) / std::max(1.0, m_inputSampleRate);
1443 const float playbackGain = computeRecordedTakeGain(conditionedSamples);
1444
1445 auto buffer = std::make_shared<AudioBufferData>();
1446 buffer->sampleRate = static_cast<uint32_t>(std::max(1.0, m_inputSampleRate));
1447 buffer->numChannels = 2;
1448 buffer->numFrames = conditionedSamples.size();
1449 buffer->interleavedData.resize(conditionedSamples.size() * 2);
1450 for (size_t i = 0; i < conditionedSamples.size(); ++i) {
1451 const float sample = conditionedSamples[i];
1452 buffer->interleavedData[i * 2] = sample;
1453 buffer->interleavedData[i * 2 + 1] = sample;
1454 }
1455
1456 const std::string takePath = buildRecordingTakePath(channelId);
1457 if (!writeRecordedTakeWav(takePath, *buffer)) {
1458 Log::error("[TrackManager] Failed to write recorded take: " + takePath);
1459 return;
1460 }
1461
1462 const std::string takeName = std::filesystem::path(takePath).stem().string();
1463 ClipSourceID sourceId = m_sourceManager.createRecordedSource(takePath, takeName, buffer);
1464 if (!sourceId.isValid()) {
1465 Log::error("[TrackManager] Failed to register recorded source for: " + takePath);
1466 return;
1467 }
1468
1469 AudioSlicePayload payload;
1470 payload.audioSourceId = sourceId;
1471 payload.durationSeconds = durationSeconds;
1472 // Populate the sample-domain fields the serializer persists
1473 // (startSamples/lengthSamples). The old {0.0, numFrames} form set
1474 // startOffset/duration instead, so the slice saved as start:0 length:0.
1475 AudioSlice fullSlice;
1476 fullSlice.startSamples = 0.0;
1477 fullSlice.lengthSamples = static_cast<double>(buffer->numFrames);
1478 payload.slices.push_back(fullSlice);
1479
1480 PatternID patternId = m_patternManager.createAudioPattern(takeName, durationBeats, payload);
1481 if (!patternId.isValid()) {
1482 Log::error("[TrackManager] Failed to create audio pattern for recorded take.");
1483 return;
1484 }
1485 m_patternManager.setPatternMixerChannel(patternId, channelId);
1486
1487 ClipInstance clip;
1488 clip.id = ClipInstanceID::generate();
1489 clip.name = takeName;
1490 clip.startBeat = startBeat;
1491 clip.durationBeats = durationBeats;
1492 clip.durationSeconds = durationSeconds;
1493 clip.patternId = patternId;
1494 clip.sourceId = patternId.value;
1495 clip.edits.gainLinear = playbackGain;
1496
1497 // Recording destination and Playlist placement are separate. Create a
1498 // lane for the take while the PatternSource retains the armed insert.
1499 auto createLane = std::make_shared<CreateLaneCommand>(m_playlistModel, takeName);
1500 createLane->execute();
1501 const PlaylistLaneID laneId = createLane->getLaneId();
1502 if (!laneId.isValid()) {
1503 m_patternManager.removePattern(patternId);
1504 Log::error("[TrackManager] Failed to create a Playlist lane for recorded take.");
1505 return;
1506 }
1507
1508 auto addClip = std::make_shared<AddClipCommand>(m_playlistModel, laneId, clip);
1509 addClip->execute();
1510 if (!m_playlistModel.getClip(clip.id)) {
1511 createLane->undo();
1512 m_patternManager.removePattern(patternId);
1513 Log::error("[TrackManager] Failed to place recorded take on its Playlist lane.");
1514 return;
1515 }
1516
1517 auto transaction = std::make_shared<CommandTransaction>("Record Take");
1518 transaction->add(createLane);
1519 transaction->add(addClip);
1520 transaction->markExecuted();
1521 if (!m_commandHistory.pushExecuted(transaction)) {
1522 addClip->undo();
1523 createLane->undo();
1524 m_patternManager.removePattern(patternId);
1525 Log::error("[TrackManager] Failed to add recorded take to command history.");
1526 return;
1527 }
1528 requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged);
1529 m_modified.store(true, std::memory_order_relaxed);
1530
1531 Log::info("[TrackManager] Recorded take committed: " + takePath + " to mixer insert " +
1532 std::to_string(channelId) + " at beat " + std::to_string(startBeat) + " with raw peak " +
1533 std::to_string(rawPeak) + ", conditioned peak " + std::to_string(conditionedPeak) + ", clip gain " +
1534 std::to_string(playbackGain));
1535 }

1646 void conditionRecordedTakeSamples(std::vector<float>& samples) const {
1647 if (samples.empty()) {
1648 return;
1649 }
1650
1651 // Remove DC bias first so the recorded waveform sits more naturally around zero.
1652 double mean = 0.0;
1653 for (float sample : samples) {
1654 mean += sample;
1655 }
1656 mean /= static_cast<double>(samples.size());
1657 for (float& sample : samples) {
1658 sample = static_cast<float>(sample - mean);
1659 }
1660
1661 // If the capture came in too hot, scale it before it ever hits clip playback.
1662 float peak = analyzePeak(samples);
1663 constexpr float targetPeak = 0.85f;
1664 if (peak > targetPeak && peak > 0.0f) {
1665 const float scale = targetPeak / peak;
1666 for (float& sample : samples) {
1667 sample *= scale;
1668 }
1669 }
