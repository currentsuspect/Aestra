253 // Pad width for the current fit mode: fit shrinks to show every step,
254 // scroll mode keeps a readable minimum and lets m_scrollX page the grid.
255 // The 4-step group gap is carved out of the available width so fit mode
256 // still shows the whole loop exactly (content = stepWidth * N + gaps).
257 float gridStepWidth(float availWidth) const {
258 const float groupTotal = static_cast<float>((m_stepCount + 3) / 4) * kStepGroupGap;
259 const float w = std::max(0.0f, availWidth - groupTotal) / static_cast<float>(std::max(1, m_stepCount));
260 return m_fitToWidth ? std::max(w, 4.0f) : std::max(w, PAD_MIN_SIZE);
261 }
