20class MakeClipPatternUniqueCommand final : public ICommand {
21public:
22 MakeClipPatternUniqueCommand(TrackManager& manager, ClipInstanceID clipId)
23 : m_manager(manager), m_clipId(clipId) {}
24
25 void execute() override {
26 if (m_executed)
27 return;
28 const auto* clip = m_manager.getPlaylistModel().getClip(m_clipId);
29 if (!clip || !clip->patternId.isValid())
30 return;
31 const auto* sourcePattern = m_manager.getPatternManager().getPattern(clip->patternId);
32 if (!sourcePattern)
33 return;
34
35 // Capture before cloning: the source pointer can be invalidated by the
36 // clone's store mutation.
37 const uint32_t sourceMixerChannelId = sourcePattern->getMixerChannelId();
38
39 if (!m_originalPatternId.isValid()) {
40 m_originalPatternId = clip->patternId;
41 }
42 if (m_detachedPattern) {
43 if (!m_manager.getPatternManager().reinsertPattern(m_detachedPattern))
44 return;
45 } else {
46 m_uniquePatternId = m_manager.getPatternManager().clonePattern(m_originalPatternId);
47 if (!m_uniquePatternId.isValid())
48 return;
49 }
50 if (!m_manager.getPlaylistModel().setClipPattern(m_clipId, m_uniquePatternId)) {
51 m_detachedPattern = m_manager.getPatternManager().detachPattern(m_uniquePatternId);
52 return;
53 }
54
55 // Preserve per-pattern mixer routing when the source had one.
56 if (sourceMixerChannelId != 0) {
57 m_manager.getPatternManager().setPatternMixerChannel(m_uniquePatternId, sourceMixerChannelId);
58 }
59 m_manager.requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged);
60 // Live playback: scheduled instances still hold the old PatternID —
61 // refresh them so the clip plays its own (unique) pattern immediately.
62 m_manager.refreshTimelinePatternInstances();
63 m_executed = true;
64 }
65
66 void undo() override {
67 if (!m_executed)
68 return;
69 auto* clip = m_manager.getPlaylistModel().getClip(m_clipId);
70 if (clip && clip->patternId == m_uniquePatternId) {
71 if (!m_manager.getPlaylistModel().setClipPattern(m_clipId, m_originalPatternId)) {
72 return;
73 }
74 m_detachedPattern = m_manager.getPatternManager().detachPattern(m_uniquePatternId);
75 }
76 m_manager.requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged);
77 m_manager.refreshTimelinePatternInstances();
78 m_executed = false;
79 }
80
81 void redo() override {
82 // execute() owns the executed-state transition; a failed redo (clip
83 // gone, pattern missing) must not mark the command executed.
84 execute();
85 }
86
87 std::string getName() const override { return "Make Pattern Unique"; }
88 size_t getSizeInBytes() const override { return sizeof(*this); }
89 bool changesProjectState() const override { return true; }
90
92 PatternID uniquePatternId() const { return m_uniquePatternId; }
93
94private:
95 TrackManager& m_manager;
96 ClipInstanceID m_clipId;
97 PatternID m_originalPatternId;
98 PatternID m_uniquePatternId;
99 std::unique_ptr<PatternSource> m_detachedPattern;
100 bool m_executed{false};
