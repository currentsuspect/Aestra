38inline std::vector<float> peaks(const std::vector<float>& interleaved, uint32_t channels, int bars = kBars) {
39 std::vector<float> out(static_cast<size_t>(std::max(bars, 1)), 0.0f);
40 if (channels == 0 || interleaved.empty()) return out;
41 const size_t frames = interleaved.size() / channels;
42 for (size_t f = 0; f < frames; ++f) {
43 const size_t bar = std::min(out.size() - 1, f * out.size() / frames);
44 for (uint32_t c = 0; c < channels; ++c) out[bar] = std::max(out[bar], std::fabs(interleaved[f * channels + c]));
45 }
46 const float top = *std::max_element(out.begin(), out.end());
47 if (top > 0.0f) for (auto& p : out) p /= top;
48 return out;
49}

52inline bool writeWav16(const std::string& path, const std::vector<float>& interleaved, uint32_t channels, uint32_t rate) {
53 std::ofstream out(path, std::ios::binary | std::ios::trunc);
54 if (!out || channels == 0) return false;
55 const uint32_t dataBytes = static_cast<uint32_t>(interleaved.size() * 2);
56 const auto u32 = [&](uint32_t v) { for (int i = 0; i < 4; ++i) out.put(static_cast<char>((v >> (8 * i)) & 0xff)); };
57 const auto u16 = [&](uint16_t v) { out.put(static_cast<char>(v & 0xff)); out.put(static_cast<char>((v >> 8) & 0xff)); };
58 out.write("RIFF", 4); u32(36 + dataBytes); out.write("WAVEfmt ", 8);
59 u32(16); u16(1); u16(static_cast<uint16_t>(channels)); u32(rate); u32(rate * channels * 2); u16(static_cast<uint16_t>(channels * 2)); u16(16);
60 out.write("data", 4); u32(dataBytes);
61 for (float s : interleaved) {
62 const int v = static_cast<int>(std::lround(std::clamp(s, -1.0f, 1.0f) * 32767.0f));
63 u16(static_cast<uint16_t>(static_cast<int16_t>(v)));
64 }
65 return static_cast<bool>(out);
66}

71inline bool writeInfo(const std::string& wavPath, const VersionBounceInfo& info) {
72 std::ofstream out(infoPathFor(wavPath), std::ios::trunc);
73 if (!out) return false;
74 out << info.seconds << ' ' << info.lufs << '\n';
75 for (float p : info.peaks) out << p << ' ';
76 out << '\n';
77 return static_cast<bool>(out);
78}
79
80inline bool readInfo(const std::string& wavPath, VersionBounceInfo& info) {
81 std::ifstream in(infoPathFor(wavPath));
82 if (!in) return false;
83 VersionBounceInfo read;
84 if (!(in >> read.seconds >> read.lufs)) return false;
85 for (float p; in >> p;) read.peaks.push_back(std::clamp(p, 0.0f, 1.0f));
86 if (read.peaks.empty()) return false;
87 info = std::move(read);
88 return true;
89}

100using Render = std::function<bool(std::vector<float>& interleaved, uint32_t& channels, uint32_t& rate)>;
102using Measure = std::function<double(const std::vector<float>& interleaved, uint32_t channels, uint32_t rate)>;

106inline bool bounceVersion(const std::string& projectPath, const std::string& takeId, const Render& render, const Measure& measure) {
107 std::vector<float> audio;
108 uint32_t channels = 0, rate = 0;
109 if (!render || !render(audio, channels, rate) || audio.empty() || channels == 0 || rate == 0) return false;
110 const double seconds = static_cast<double>(audio.size() / channels) / rate;
111 if (seconds > kMaxSeconds) return false;
112
113 const std::string name = fileNameFor(takeId);
114 const std::filesystem::path wav = std::filesystem::path(TakeManager::getTakesDirectory(projectPath)) / name;
115 std::error_code ec;
116 std::filesystem::create_directories(wav.parent_path(), ec);
117 if (!writeWav16(wav.string(), audio, channels, rate)) return false;
118 VersionBounceInfo info;
119 info.seconds = seconds;
120 info.lufs = measure ? measure(audio, channels, rate) : -70.0;
121 info.peaks = peaks(audio, channels);
122 if (!writeInfo(wav.string(), info)) return false;
123 return TakeManager::setVersionBounce(projectPath, takeId, name).ok;
124}
