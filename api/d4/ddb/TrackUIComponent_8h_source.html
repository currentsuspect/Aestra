299 // V8-W3 — timeline fade handles. The model, serialization, offline render,
300 // live playback (ClipRenderKernel) and the AudioClipEditorPanel sliders all
301 // already handle fades; this is only the timeline gesture.
302 //
303 // Handles live in the WAVEFORM BODY, not at the clip's top corners.
304 //
305 // The top-left carries the burger menu — a 21x18 hit box whose handler runs
306 // before any fade check, so a handle up there is unreachable however the
307 // hit-test is ordered (confirmed by driving, not assumed). Owner's call:
308 // put them where the waveform is. That also makes them a bigger grab
309 // target, and it yields one explainable rule:
310 //
311 // clip HEADER = menu and trim · clip BODY = fades
312 //
313 // FADE_BODY_TOP_INSET clears the burger box (17px) with a pixel to spare.
314 static constexpr float FADE_BODY_TOP_INSET = 18.0f; // Where the fade band begins
315 static constexpr float FADE_HANDLE_GRAB = 9.0f; // Tolerance once a fade is visible
316 // An unset fade has no drawn handle to aim at, so its corner claims a
317 // wider inward zone. Without this the gesture only landed by luck.
318 static constexpr float FADE_CORNER_REACH = 26.0f;
319 static constexpr float FADE_HANDLE_SIZE = 6.0f; // Drawn marker size
