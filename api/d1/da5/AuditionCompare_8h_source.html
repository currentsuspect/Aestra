46inline std::vector<float> columnPeaks(const std::vector<float>& interleaved, uint32_t channels, size_t columns = kWaveColumns) {
47 std::vector<float> out(columns, 0.0f);
48 if (channels == 0 || interleaved.empty() || columns == 0) return out;
49 const size_t frames = interleaved.size() / channels;
50 for (size_t f = 0; f < frames; ++f) {
51 const size_t col = std::min(columns - 1, f * columns / frames);
52 for (uint32_t c = 0; c < channels; ++c) out[col] = std::max(out[col], std::fabs(interleaved[f * channels + c]));
53 }
54 return out;
55}

57inline Side measure(const std::string& path, std::vector<float> samples, uint32_t channels, uint32_t rate) {
58 Side s;
59 s.path = path;
60 s.peaks = columnPeaks(samples, channels);
61 s.analysis = Audio::analyzeAudio(samples, channels, rate);
62 const uint64_t frames = channels ? samples.size() / channels : 0;
63 s.stereo = Audio::analyzeStereoDetail(samples.data(), frames, channels, rate);
64 s.sections = Audio::detectSections(samples.data(), frames, channels, rate);
65 s.spectrum = Audio::analyzeSpectrum(samples.data(), frames, channels, rate);
66 s.seconds = rate ? static_cast<double>(frames) / rate : 0.0;
67 auto buffer = std::make_shared<Audio::AudioBufferData>();
68 buffer->sampleRate = rate;
69 buffer->numChannels = channels;
70 buffer->numFrames = frames;
71 buffer->interleavedData = std::move(samples);
72 s.buffer = std::move(buffer);
73 return s;
74}

77inline Result compare(Side mix, Side ref) {
78 Result r;
79 r.mix = std::move(mix);
80 r.ref = std::move(ref);
81 r.ok = r.mix.buffer && r.ref.buffer && r.mix.analysis.integratedLufs > -150.0 && r.ref.analysis.integratedLufs > -150.0;
82 // levelMatchGainDb is the gain that brings the mix to the reference; the reference moves the other way.
83 r.referenceGainDb = r.ok ? -Audio::levelMatchGainDb(r.mix.analysis.integratedLufs, r.ref.analysis.integratedLufs) : 0.0;
84 r.tonalDeltaDb = Audio::tonalBalanceDeltaDb(r.mix.analysis, r.ref.analysis);
85 // Both waveforms are drawn at the level they are heard at: the reference after its match.
86 if (r.ok) {
87 const float g = static_cast<float>(std::pow(10.0, r.referenceGainDb / 20.0));
88 for (auto& p : r.ref.peaks) p = std::min(1.0f, p * g);
89 }
90 return r;
91}
