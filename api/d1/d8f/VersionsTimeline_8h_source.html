47 static Result layout(const std::vector<TakeManager::TakeEntry>& versions, const AestraUI::NUIRect& area, float scrollX = 0.0f) {
48 Result r;
49 for (const auto& v : versions) {
50 const std::string lane = v.branch.empty() ? "Main" : v.branch;
51 if (std::find(r.lanes.begin(), r.lanes.end(), lane) == r.lanes.end()) r.lanes.push_back(lane);
52 }
53 const float top = area.y + 84.0f;
54 const float step = r.lanes.size() > 1
55 ? std::clamp((area.height - 84.0f - 90.0f) / static_cast<float>(r.lanes.size() - 1), 56.0f, 120.0f)
56 : 0.0f;
57 for (size_t i = 0; i < r.lanes.size(); ++i) r.laneY.push_back(top + step * static_cast<float>(i));
58
59 // Time order decides the left-to-right rank; ties keep the manifest's order.
60 std::vector<int> order(versions.size());
61 for (size_t i = 0; i < order.size(); ++i) order[i] = static_cast<int>(i);
62 std::stable_sort(order.begin(), order.end(), [&](int a, int b) { return versions[static_cast<size_t>(a)].createdAtMs < versions[static_cast<size_t>(b)].createdAtMs; });
63 const float left = area.x + kLabelColumn + 40.0f, right = std::max(left + 1.0f, area.right() - kTailRoom);
64 // A saved version takes a full slot along the axis; an autosave takes half, so the labels stay readable.
65 float total = 0.0f;
66 std::vector<float> slotAt(order.size(), 0.0f);
67 for (size_t rank = 0; rank < order.size(); ++rank) {
68 slotAt[rank] = total;
69 total += versions[static_cast<size_t>(order[rank])].kind == TakeManager::TakeEntry::Kind::Autosave ? 0.5f : 1.0f;
70 }
71 const float lastSlot = order.empty() ? 0.0f : slotAt.back();
72 const float stepX = lastSlot > 0.0f ? std::max(kMinStep, (right - left) / lastSlot) : 0.0f;
73 r.maxScroll = std::max(0.0f, stepX * lastSlot - (right - left));
74 scrollX = std::clamp(scrollX, 0.0f, r.maxScroll);
75
76 r.nodes.resize(versions.size());
77 for (size_t rank = 0; rank < order.size(); ++rank) {
78 const int i = order[rank];
79 const auto& v = versions[static_cast<size_t>(i)];
80 auto& n = r.nodes[static_cast<size_t>(i)];
81 n.version = i;
82 const std::string lane = v.branch.empty() ? "Main" : v.branch;
83 n.lane = static_cast<int>(std::find(r.lanes.begin(), r.lanes.end(), lane) - r.lanes.begin());
84 n.x = left + stepX * slotAt[rank] - scrollX;
85 n.y = r.laneY[static_cast<size_t>(n.lane)];
86 n.autosave = v.kind == TakeManager::TakeEntry::Kind::Autosave;
87 n.open = v.active;
88 }
89 for (size_t i = 0; i < versions.size(); ++i) {
90 if (versions[i].parentId.empty()) continue;
91 for (size_t j = 0; j < versions.size(); ++j)
92 if (versions[j].id == versions[i].parentId) r.links.push_back({static_cast<int>(j), static_cast<int>(i)});
93 }
94 return r;
95 }

98 static int nodeAt(const Result& r, const AestraUI::NUIPoint& p, float radius = 12.0f) {
99 for (int i = static_cast<int>(r.nodes.size()) - 1; i >= 0; --i) {
100 const auto& n = r.nodes[static_cast<size_t>(i)];
101 if (std::fabs(n.x - p.x) <= radius && std::fabs(n.y - p.y) <= radius) return i;
102 }
103 return -1;
104 }
