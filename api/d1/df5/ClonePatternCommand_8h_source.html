19class ClonePatternCommand : public ICommand {
20public:
21 ClonePatternCommand(PatternManager& patternManager, PatternID sourceId)
22 : m_patternManager(patternManager), m_sourceId(sourceId) {}
23
24 void execute() override {
25 if (m_executed) return;
26 m_clonedId = m_patternManager.clonePattern(m_sourceId);
27 if (!m_clonedId.isValid()) {
28 // Loud, not silent: a failed clone must never be recorded as an
29 // executed command or reported as success.
30 throw std::runtime_error("failed to clone pattern " +
31 std::to_string(m_sourceId.value));
32 }
33 m_executed = true;
34 }
35
36 void undo() override {
37 if (!m_executed) return;
38 m_patternManager.removePattern(m_clonedId);
39 m_executed = false;
40 }
41
42 void redo() override {
43 if (m_executed) return;
44 execute();
45 }
46
47 std::string getName() const override { return "Clone Pattern"; }
48 size_t getSizeInBytes() const override { return sizeof(*this); }
49 bool changesProjectState() const override { return true; }
50
51 PatternID getClonedPatternId() const { return m_clonedId; }
52
53private:
54 PatternManager& m_patternManager;
55 PatternID m_sourceId;
