14std::string UIState::getUIStatePath() {
15 IPlatformUtils* utils = Platform::getUtils();
16 if (!utils) {
17 return (std::filesystem::current_path() / "ui_state.json").string();
18 }
19 std::string appDataDir = utils->getAppDataPath("Aestra");
20 std::error_code ec;

28void UIState::load() {
29 std::string path = getUIStatePath();
30
31 if (!std::filesystem::exists(path)) {
32 Log::info("[UIState] No UI state file found, using defaults");
33 return;
34 }
35
36 std::ifstream file(path);
37 if (!file) {
38 Log::warning("[UIState] Could not open UI state file");
39 return;
40 }
41
42 std::string jsonStr((std::istreambuf_iterator<char>(file)),
43 std::istreambuf_iterator<char>());
44
45 try {
46 JSON json = JSON::parse(jsonStr);
47 fromJson(json);
48 Log::info("[UIState] Loaded from: " + path);
49 } catch (const std::exception& e) {
50 Log::warning("[UIState] Failed to parse UI state file: " + std::string(e.what()));
51 }
52}

54void UIState::save() {
55 std::string path = getUIStatePath();
56 std::string jsonStr = toJson().toString(2);
57
58 std::error_code ec;
59 std::filesystem::path tmpPath = path + ".tmp";
60
61 {
62 std::ofstream out(tmpPath, std::ios::binary | std::ios::trunc);
63 if (!out) {
64 Log::warning("[UIState] Failed to write UI state file");
65 return;
66 }
67 out.write(jsonStr.data(), static_cast<std::streamsize>(jsonStr.size()));
68 if (!Aestra::syncOfstream(out, tmpPath.string())) {
69 Log::warning("[UIState] Failed to sync UI state file");
70 out.close();
71 std::filesystem::remove(tmpPath, ec);
72 return;
73 }
74 }
75
76 // Replace existing file
77#ifdef _WIN32
78 if (std::filesystem::exists(path, ec)) {
79 std::filesystem::remove(path, ec);
80 }
81#endif
82 ec.clear();
83 std::filesystem::rename(tmpPath, path, ec);
84 if (ec) {
85 Log::warning("[UIState] Failed to replace UI state file: " + ec.message());
86 std::filesystem::remove(tmpPath, ec);
87 return;
88 }
89
90#ifndef _WIN32
91 if (!Aestra::fsyncParentDirectory(path)) {
92 Log::warning("[UIState] Failed to sync UI state directory");
93 return;
94 }
95#endif
96
97 Log::info("[UIState] Saved to: " + path);
98}

103JSON UIState::toJson() const {
104 JSON root = JSON::object();
105
106 // Window
107 JSON window = JSON::object();
108 window.set("x", JSON(static_cast<double>(windowX)));
109 window.set("y", JSON(static_cast<double>(windowY)));
110 window.set("width", JSON(static_cast<double>(windowWidth)));
111 window.set("height", JSON(static_cast<double>(windowHeight)));
112 window.set("maximized", JSON(maximized));
113 root.set("window", window);
114
115 // Panels
116 JSON panels = JSON::object();
117 panels.set("browserWidth", JSON(static_cast<double>(browserWidth)));
118 panels.set("mixerHeight", JSON(static_cast<double>(mixerHeight)));
119 panels.set("browserVisible", JSON(browserVisible));
120 panels.set("mixerVisible", JSON(mixerVisible));
121 root.set("panels", panels);
122
123 // File browser
124 JSON fileBrowser = JSON::object();
125 JSON expanded = JSON::array();
126 for (const auto& folder : expandedFolders) {
127 expanded.push(JSON(folder));
128 }
129 fileBrowser.set("expandedFolders", expanded);
130 fileBrowser.set("lastPath", JSON(lastBrowsedPath));
131 root.set("fileBrowser", fileBrowser);
132
133 // Track view
134 JSON trackView = JSON::object();
135 trackView.set("horizontalZoom", JSON(static_cast<double>(horizontalZoom)));
136 trackView.set("verticalZoom", JSON(static_cast<double>(verticalZoom)));
137 trackView.set("scrollX", JSON(static_cast<double>(scrollPositionX)));
138 trackView.set("scrollY", JSON(static_cast<double>(scrollPositionY)));
139 root.set("trackView", trackView);
140
141 return root;
142}

144void UIState::fromJson(const JSON& json) {
145 // Window
146 if (json.has("window") && json["window"].isObject()) {
147 const JSON& window = json["window"];
148 if (window.has("x") && window["x"].isNumber()) {
149 windowX = static_cast<int>(window["x"].asNumber());
150 }
151 if (window.has("y") && window["y"].isNumber()) {
152 windowY = static_cast<int>(window["y"].asNumber());
153 }
154 if (window.has("width") && window["width"].isNumber()) {
155 windowWidth = static_cast<int>(window["width"].asNumber());
156 }
157 if (window.has("height") && window["height"].isNumber()) {
158 windowHeight = static_cast<int>(window["height"].asNumber());
159 }
160 if (window.has("maximized") && window["maximized"].isBool()) {
161 maximized = window["maximized"].asBool();
162 }
163 }
164
165 // Panels
166 if (json.has("panels") && json["panels"].isObject()) {
167 const JSON& panels = json["panels"];
168 if (panels.has("browserWidth") && panels["browserWidth"].isNumber()) {
169 browserWidth = static_cast<float>(panels["browserWidth"].asNumber());
170 }
171 if (panels.has("mixerHeight") && panels["mixerHeight"].isNumber()) {
172 mixerHeight = static_cast<float>(panels["mixerHeight"].asNumber());
173 }
174 if (panels.has("browserVisible") && panels["browserVisible"].isBool()) {
175 browserVisible = panels["browserVisible"].asBool();
176 }
177 if (panels.has("mixerVisible") && panels["mixerVisible"].isBool()) {
178 mixerVisible = panels["mixerVisible"].asBool();
179 }
180 }
181
182 // File browser
183 if (json.has("fileBrowser") && json["fileBrowser"].isObject()) {
184 const JSON& fileBrowser = json["fileBrowser"];
185 if (fileBrowser.has("expandedFolders") && fileBrowser["expandedFolders"].isArray()) {
186 expandedFolders.clear();
187 for (size_t i = 0; i < fileBrowser["expandedFolders"].size(); ++i) {
188 const JSON& item = fileBrowser["expandedFolders"][i];
189 if (item.isString()) {
190 expandedFolders.insert(item.asString());
191 }
192 }
193 }
194 if (fileBrowser.has("lastPath") && fileBrowser["lastPath"].isString()) {
195 lastBrowsedPath = fileBrowser["lastPath"].asString();
196 }
197 }
198
199 // Track view
200 if (json.has("trackView") && json["trackView"].isObject()) {
201 const JSON& trackView = json["trackView"];
202 if (trackView.has("horizontalZoom") && trackView["horizontalZoom"].isNumber()) {
203 horizontalZoom = static_cast<float>(trackView["horizontalZoom"].asNumber());
204 }
205 if (trackView.has("verticalZoom") && trackView["verticalZoom"].isNumber()) {
206 verticalZoom = static_cast<float>(trackView["verticalZoom"].asNumber());
207 }
208 if (trackView.has("scrollX") && trackView["scrollX"].isNumber()) {
209 scrollPositionX = static_cast<float>(trackView["scrollX"].asNumber());
210 }
211 if (trackView.has("scrollY") && trackView["scrollY"].isNumber()) {
212 scrollPositionY = static_cast<float>(trackView["scrollY"].asNumber());
213 }
214 }
