26void TrackManager::ensureClipPrefilters() {
27 // Graph-build thread only (see the ClipSource threading contract).
28 const double projectRate = m_playlistModel.getProjectSampleRate();
29 const uint32_t targetRate = projectRate > 0.0 ? static_cast<uint32_t>(projectRate) : 0;
30
31 // 1. Apply finished results whose key still matches the live state.
32 if (m_clipPrefilterService) {
33 for (auto& result : m_clipPrefilterService->drainCompleted()) {
34 ClipSource* source = m_sourceManager.getSource(ClipSourceID{result.sourceId});
35 if (source == nullptr || !result.filtered) {
36 continue;
37 }
38 if (source->getContentRevision() != result.contentRevision ||
39 result.targetRate != targetRate ||
40 result.specVersion != ClipPrefilter::kSpecVersion) {
41 continue; // stale (buffer replaced / rate changed since enqueue) — discard
42 }
43 source->setFilteredVariant(std::move(result.filtered), result.targetRate,
44 result.contentRevision, result.specVersion);
45 }
46 }
47
48 if (targetRate == 0) {
49 return;
50 }
51
52 // 2. Clear stale variants; queue missing work for downsampled sources THAT A CLIP PLAYS. A filtered
53 // copy is a full float32 copy of the source, so one made for a source nothing plays (a browser preview,
54 // a deleted clip's leftover) is memory for nothing.
55 std::unordered_set<uint64_t> playedSources;
56 std::unordered_set<uint64_t> variedSources; // played by some clip at a speed or pitch other than 1
57 for (size_t l = 0; l < m_playlistModel.getLaneCount(); ++l) {
58 const PlaylistLane* lane = m_playlistModel.getLane(m_playlistModel.getLaneId(l));
59 if (lane == nullptr) {
60 continue;
61 }
62 for (const ClipInstance& clip : lane->clips) {
63 const PatternSource* pattern = m_patternManager.getPattern(clip.patternId);
64 if (pattern != nullptr && pattern->isAudio()) {
65 const uint64_t sourceKey = std::get<AudioSlicePayload>(pattern->payload).audioSourceId.value;
66 playedSources.insert(sourceKey);
67 if (std::fabs(static_cast<double>(clip.edits.effectiveVarispeed()) - 1.0) > 1e-6) {
68 variedSources.insert(sourceKey);
69 }
70 }
71 }
72 }
73 for (const ClipSourceID id : m_sourceManager.getAllSourceIDs()) {
74 ClipSource* source = m_sourceManager.getSource(id);
75 if (source == nullptr) {
76 continue;
77 }
78 const auto buffer = source->getBuffer();
79 const bool needed = buffer && buffer->isValid() && ClipPrefilter::isNeeded(buffer->sampleRate, targetRate) &&
80 playedSources.count(id.value) > 0;
81
82 if (!needed) {
83 if (source->hasFilteredVariant()) {
84 source->clearFilteredVariant(); // rate no longer qualifies, or nothing plays it — free the copy
85 }
86 continue;
87 }
88 if (source->hasFilteredVariant() && !source->getFilteredBufferFor(targetRate)) {
89 source->clearFilteredVariant(); // keyed for an old rate/content — free it
90 }
91 if (source->filteredVariantSpec() != 0 &&
92 source->filteredVariantSpec() != ClipPrefilter::kSpecVersion) {
93 source->clearFilteredVariant(); // filter spec changed
94 }
95 // A whole-number rate ratio can be stored decimated: half the memory and no resampling, and the
96 // kernel plays it exactly as it played the full-rate copy (bit for bit) at speed and pitch 1. At any
97 // other speed the two differ (the decimated data is interpolated close to its Nyquist), so a source
98 // some clip plays at another speed needs the full-rate copy. The full-rate copy is sticky: once a
99 // source has it, it keeps it. While an upgrade builds, the decimated copy keeps playing.
100 const uint32_t factor = ClipPrefilter::decimationFactor(buffer->sampleRate, targetRate);
101 const auto current = source->getFilteredBufferFor(targetRate);
102 const bool currentIsFullRate = current && current->sampleRate != targetRate;
103 const uint32_t wantDecimation =
104 (factor > 1 && variedSources.count(id.value) == 0 && !currentIsFullRate) ? factor : 1;
105 if (current && (currentIsFullRate || wantDecimation > 1)) {
106 continue; // ready
107 }
108
109 if (!m_clipPrefilterService) {
110 // Completion callback: atomic-only graph-dirty request (thread-safe from
111 // the worker), so the app's graph pump applies the copy on next drain.
112 m_clipPrefilterService = std::make_unique<ClipPrefilterService>(
113 [this] { requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged); });
114 }
115 ClipPrefilterService::Job job;
116 job.sourceId = id.value;
117 job.contentRevision = source->getContentRevision();
118 job.targetRate = targetRate;
119 job.decimation = wantDecimation;
120 job.source = source->getSharedBuffer();
121 m_clipPrefilterService->enqueue(std::move(job)); // dedupes internally
122 }
123}

133bool TrackManager::prewarmSamplerKeepLength(bool wait) {
134 using Plugins::SamplerPlugin;
135 // Pitches each unit's notes use, across every pattern (a unit's notes can live in any pattern).
136 std::unordered_map<UnitID, std::vector<int>> pitchesByUnit;
137 for (const auto& pattern : m_patternManager.getAllPatterns()) {
138 if (!pattern || !pattern->isMidi()) continue;
139 for (const auto& note : pattern->getMidiNotes()) {
140 auto& pitches = pitchesByUnit[note.unitId];
141 if (std::find(pitches.begin(), pitches.end(), note.pitch) == pitches.end()) {
142 pitches.push_back(note.pitch);
143 }
144 }
145 }
146 bool allReady = true;
147 for (UnitID id : m_unitManager.getAllUnitIDs()) {
148 auto sampler = std::dynamic_pointer_cast<SamplerPlugin>(m_unitManager.getUnitPlugin(id));
149 if (!sampler || sampler->getPitchMode() != SamplerPlugin::PitchMode::KeepLength) continue;
150 const auto it = pitchesByUnit.find(id);
151 if (it == pitchesByUnit.end()) continue;
152 allReady = sampler->prewarmKeepLength(it->second, wait) && allReady;
153 }
154 return allReady;
155}
