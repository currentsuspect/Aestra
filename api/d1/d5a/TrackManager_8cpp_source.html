21void TrackManager::ensureClipPrefilters() {
22 // Graph-build thread only (see the ClipSource threading contract).
23 const double projectRate = m_playlistModel.getProjectSampleRate();
24 const uint32_t targetRate = projectRate > 0.0 ? static_cast<uint32_t>(projectRate) : 0;
25
26 // 1. Apply finished results whose key still matches the live state.
27 if (m_clipPrefilterService) {
28 for (auto& result : m_clipPrefilterService->drainCompleted()) {
29 ClipSource* source = m_sourceManager.getSource(ClipSourceID{result.sourceId});
30 if (source == nullptr || !result.filtered) {
31 continue;
32 }
33 if (source->getContentRevision() != result.contentRevision ||
34 result.targetRate != targetRate ||
35 result.specVersion != ClipPrefilter::kSpecVersion) {
36 continue; // stale (buffer replaced / rate changed since enqueue) — discard
37 }
38 source->setFilteredVariant(std::move(result.filtered), result.targetRate,
39 result.contentRevision, result.specVersion);
40 }
41 }
42
43 if (targetRate == 0) {
44 return;
45 }
46
47 // 2. Clear stale variants; queue missing work for downsampled sources.
48 for (const ClipSourceID id : m_sourceManager.getAllSourceIDs()) {
49 ClipSource* source = m_sourceManager.getSource(id);
50 if (source == nullptr) {
51 continue;
52 }
53 const auto buffer = source->getBuffer();
54 const bool needed =
55 buffer && buffer->isValid() && ClipPrefilter::isNeeded(buffer->sampleRate, targetRate);
56
57 if (!needed) {
58 if (source->hasFilteredVariant()) {
59 source->clearFilteredVariant(); // rate no longer qualifies — free the copy
60 }
61 continue;
62 }
63 if (source->hasFilteredVariant() && !source->getFilteredBufferFor(targetRate)) {
64 source->clearFilteredVariant(); // keyed for an old rate/content — free it
65 }
66 if (source->filteredVariantSpec() != 0 &&
67 source->filteredVariantSpec() != ClipPrefilter::kSpecVersion) {
68 source->clearFilteredVariant(); // filter spec changed
69 }
70 if (source->getFilteredBufferFor(targetRate)) {
71 continue; // ready
72 }
73
74 if (!m_clipPrefilterService) {
75 // Completion callback: atomic-only graph-dirty request (thread-safe from
76 // the worker), so the app's graph pump applies the copy on next drain.
77 m_clipPrefilterService = std::make_unique<ClipPrefilterService>(
78 [this] { requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged); });
79 }
80 ClipPrefilterService::Job job;
81 job.sourceId = id.value;
82 job.contentRevision = source->getContentRevision();
83 job.targetRate = targetRate;
84 job.source = source->getSharedBuffer();
85 m_clipPrefilterService->enqueue(std::move(job)); // dedupes internally
86 }
87}
