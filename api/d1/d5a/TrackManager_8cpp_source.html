20void TrackManager::commitRecordingTake(uint64_t trackId, const RecordingCapture& capture, double fallbackStartBeat,
21 bool forcePlacementStartBeat) {
22 auto* track = getTrack(trackId);
23 if (!track) {
24 Log::warning("[TrackManager] Could not resolve Track for recorded take: " + std::to_string(trackId));
25 return;
26 }
27 const uint32_t channelId = track->channelId;
28 std::vector<float> capturedSamples = capture.stream ? capture.stream->samples() : copyCaptureSamples(capture);
29 if (capture.stream && capture.stream->lostFrames() > 0) {
30 Log::warning("[TrackManager] Recorded take on track " + std::to_string(trackId) + " lost " +
31 std::to_string(capture.stream->lostFrames()) +
32 " frame(s) the disk writer could not keep up with; they are silence in the take.");
33 }
34 if (capturedSamples.empty()) {
35 if (capture.stream) {
36 capture.stream->discardPartialFile();
37 }
38 // Silent no-op here made "record produced nothing" undiagnosable
39 // (deferred-capture beat unreachable, #845). Say something.
40 Log::warning("[TrackManager] Recorded take on track " + std::to_string(trackId) +
41 " captured zero samples — capture start was never reached during the take.");
42 return;
43 }
44
45 if (!getChannelById(channelId)) {
46 Log::warning("[TrackManager] Could not resolve mixer insert for recorded take " +
47 std::to_string(channelId) + " (track " + std::to_string(trackId) + ")");
48 return;
49 }
50
51 std::vector<float> conditionedSamples = std::move(capturedSamples);
52 const float rawPeak = analyzePeak(conditionedSamples);
53 conditionRecordedTakeSamples(conditionedSamples);
54 const float conditionedPeak = analyzePeak(conditionedSamples);
55
56 const double captureStartBeat = capture.startBeat.load(std::memory_order_acquire);
57 double startBeat = forcePlacementStartBeat
58 ? fallbackStartBeat
59 : (captureStartBeat > 0.0 ? captureStartBeat : fallbackStartBeat);
60 // T-6: compensate device latency. The captured signal arrived late by
61 // the input latency and was performed against backing heard late by
62 // the output latency, so the take as placed sounds late by the sum.
63 // Shift placement earlier (samples intact); clamp at zero — a take
64 // starting at the very top can only keep a small residual lateness.
65 // Values are pulled live through the provider so buffer/device/rate
66 // reconfigures from any path are reflected at commit time.
67 const double bpm = std::max(1.0, m_playlistModel.getBPM());
68 double latencyInMs = 0.0, latencyOutMs = 0.0;
69 if (m_recordLatencyProvider) {
70 m_recordLatencyProvider(latencyInMs, latencyOutMs);
71 }
72 const double recordLatencyBeats = (std::max(0.0, latencyInMs) + std::max(0.0, latencyOutMs)) / 1000.0 *
73 bpm / 60.0;
74 startBeat = std::max(0.0, startBeat - recordLatencyBeats);
75 const double durationBeats = framesToBeats(static_cast<double>(conditionedSamples.size()));
76 if (durationBeats <= 0.0) {
77 return;
78 }
79 const double durationSeconds =
80 static_cast<double>(conditionedSamples.size()) / std::max(1.0, m_inputSampleRate);
81 const float playbackGain = computeRecordedTakeGain(conditionedSamples);
82
83 auto buffer = std::make_shared<AudioBufferData>();
84 buffer->sampleRate = static_cast<uint32_t>(std::max(1.0, m_inputSampleRate));
85 buffer->numChannels = 2;
86 buffer->numFrames = conditionedSamples.size();
87 buffer->interleavedData.resize(conditionedSamples.size() * 2);
88 for (size_t i = 0; i < conditionedSamples.size(); ++i) {
89 const float sample = conditionedSamples[i];
90 buffer->interleavedData[i * 2] = sample;
91 buffer->interleavedData[i * 2 + 1] = sample;
92 }
93
94 std::string takePath = buildRecordingTakePath(channelId);
95 if (capture.stream && !capture.stream->partialPath().empty()) {
96 const std::string& partial = capture.stream->partialPath();
97 takePath = partial.substr(0, partial.size() - std::strlen(RecordingTakeStream::kPartialExtension));
98 }
99 if (!writeRecordedTakeWav(takePath, *buffer)) {
100 // The partial file stays: it is now the only copy of this take.
101 Log::error("[TrackManager] Failed to write recorded take: " + takePath);
102 return;
103 }
104 if (capture.stream) {
105 capture.stream->discardPartialFile();
106 }
107
108 const std::string takeName = std::filesystem::path(takePath).stem().string();
109 ClipSourceID sourceId = m_sourceManager.createRecordedSource(takePath, takeName, buffer);
110 if (!sourceId.isValid()) {
111 Log::error("[TrackManager] Failed to register recorded source for: " + takePath);
112 return;
113 }
114
115 AudioSlicePayload payload;
116 payload.audioSourceId = sourceId;
117 payload.durationSeconds = durationSeconds;
118 // Populate the sample-domain fields the serializer persists
119 // (startSamples/lengthSamples). The old {0.0, numFrames} form set
120 // startOffset/duration instead, so the slice saved as start:0 length:0.
121 AudioSlice fullSlice;
122 fullSlice.startSamples = 0.0;
123 fullSlice.lengthSamples = static_cast<double>(buffer->numFrames);
124 payload.slices.push_back(fullSlice);
125
126 PatternID patternId = m_patternManager.createAudioPattern(takeName, durationBeats, payload);
127 if (!patternId.isValid()) {
128 Log::error("[TrackManager] Failed to create audio pattern for recorded take.");
129 return;
130 }
131 m_patternManager.setPatternMixerChannel(patternId, channelId);
132
133 ClipInstance clip;
134 clip.id = ClipInstanceID::generate();
135 clip.name = takeName;
136 clip.startBeat = startBeat;
137 clip.durationBeats = durationBeats;
138 clip.durationSeconds = durationSeconds;
139 clip.patternId = patternId;
140 clip.sourceId = patternId.value;
141 clip.edits.gainLinear = playbackGain;
142
143 if (recordTakeInPlace(m_playlistModel, m_commandHistory, *track, clip)) { // the armed lane, if the span is free
144 requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged); m_modified.store(true, std::memory_order_relaxed);
145 Log::info("[TrackManager] Recorded take committed in the track's own lane: " + takePath);
146 if (m_onTakeCommitted) m_onTakeCommitted(track->activeLaneId);
147 return;
148 }
149 // Recording destination and Playlist placement are separate: a take lane, the PatternSource keeps the armed insert.
150 auto createLane = std::make_shared<CreateLaneCommand>(m_playlistModel, takeName);
151 createLane->execute();
152 const PlaylistLaneID laneId = createLane->getLaneId();
153 if (!laneId.isValid()) {
154 m_patternManager.removePattern(patternId);
155 Log::error("[TrackManager] Failed to create a Playlist lane for recorded take.");
156 return;
157 }
158
159 auto addClip = std::make_shared<AddClipCommand>(m_playlistModel, laneId, clip);
160 addClip->execute();
161 if (!m_playlistModel.getClip(clip.id)) {
162 createLane->undo();
163 m_patternManager.removePattern(patternId);
164 Log::error("[TrackManager] Failed to place recorded take on its Playlist lane.");
165 return;
166 }
167
168 // The take lane is owned by the recording Track (FD-14), joined in the same undoable step as its creation.
169 auto attachLane = std::make_shared<AttachLaneToTrackCommand>(*this, trackId, laneId);
170 attachLane->execute();
171 auto* ownedLane = m_playlistModel.getLane(laneId);
172 if (!ownedLane || ownedLane->trackId != trackId) {
173 attachLane->undo();
174 addClip->undo();
175 createLane->undo();
176 m_patternManager.removePattern(patternId);
177 Log::error("[TrackManager] Failed to attach recorded take lane to track " + std::to_string(trackId));
178 return;
179 }
180
181 auto transaction = std::make_shared<CommandTransaction>("Record Take");
182 transaction->add(createLane);
183 transaction->add(addClip);
184 transaction->add(attachLane);
185 transaction->markExecuted();
186 if (!m_commandHistory.pushExecuted(transaction)) {
187 attachLane->undo();
188 addClip->undo();
189 createLane->undo();
190 m_patternManager.removePattern(patternId);
191 Log::error("[TrackManager] Failed to add recorded take to command history.");
192 return;
193 }
194 requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged); m_modified.store(true, std::memory_order_relaxed);
195 Log::info("[TrackManager] Recorded take committed: " + takePath + " to mixer insert " + std::to_string(channelId) + " at beat " +
196 std::to_string(startBeat) + " with raw peak " + std::to_string(rawPeak) + ", conditioned peak " +
197 std::to_string(conditionedPeak) + ", clip gain " + std::to_string(playbackGain));
198
199 if (m_onTakeCommitted) m_onTakeCommitted(laneId);
200}

215void TrackManager::startTakeDrainThread() {
216 if (m_takeDrainThread.joinable()) {
217 return;
218 }
219 {
220 std::lock_guard<std::mutex> lock(m_takeDrainWakeMutex);
221 m_takeDrainStop = false;
222 }
223 // The capture ring holds m_maxRecordingSeconds; draining every 50 ms
224 // leaves that whole span as headroom for a stalled disk.
225 m_takeDrainThread = std::thread([this]() {
226 std::unique_lock<std::mutex> lock(m_takeDrainWakeMutex);
227 while (!m_takeDrainStop) {
228 m_takeDrainWake.wait_for(lock, std::chrono::milliseconds(50), [this]() { return m_takeDrainStop; });
229 lock.unlock();
230 drainRecordingCaptures();
231 lock.lock();
232 }
233 });
234}

253void TrackManager::ensureClipPrefilters() {
254 // Graph-build thread only (see the ClipSource threading contract).
255 const double projectRate = m_playlistModel.getProjectSampleRate();
256 const uint32_t targetRate = projectRate > 0.0 ? static_cast<uint32_t>(projectRate) : 0;
257
258 // 1. Apply finished results whose key still matches the live state.
259 if (m_clipPrefilterService) {
260 for (auto& result : m_clipPrefilterService->drainCompleted()) {
261 ClipSource* source = m_sourceManager.getSource(ClipSourceID{result.sourceId});
262 if (source == nullptr || !result.filtered) {
263 continue;
264 }
265 if (source->getContentRevision() != result.contentRevision ||
266 result.targetRate != targetRate ||
267 result.specVersion != ClipPrefilter::kSpecVersion) {
268 continue; // stale (buffer replaced / rate changed since enqueue) — discard
269 }
270 source->setFilteredVariant(std::move(result.filtered), result.targetRate,
271 result.contentRevision, result.specVersion);
272 }
273 }
274
275 if (targetRate == 0) {
276 return;
277 }
278
279 // 2. Clear stale variants; queue missing work for downsampled sources THAT A CLIP PLAYS. A filtered
280 // copy is a full float32 copy of the source, so one made for a source nothing plays (a browser preview,
281 // a deleted clip's leftover) is memory for nothing.
282 std::unordered_set<uint64_t> playedSources;
283 std::unordered_set<uint64_t> variedSources; // played by some clip at a speed or pitch other than 1
284 for (size_t l = 0; l < m_playlistModel.getLaneCount(); ++l) {
285 const PlaylistLane* lane = m_playlistModel.getLane(m_playlistModel.getLaneId(l));
286 if (lane == nullptr) {
287 continue;
288 }
289 for (const ClipInstance& clip : lane->clips) {
290 const PatternSource* pattern = m_patternManager.getPattern(clip.patternId);
291 if (pattern != nullptr && pattern->isAudio()) {
292 const uint64_t sourceKey = std::get<AudioSlicePayload>(pattern->payload).audioSourceId.value;
293 playedSources.insert(sourceKey);
294 if (std::fabs(static_cast<double>(clip.edits.effectiveVarispeed()) - 1.0) > 1e-6) {
295 variedSources.insert(sourceKey);
296 }
297 }
298 }
299 }
300 for (const ClipSourceID id : m_sourceManager.getAllSourceIDs()) {
301 ClipSource* source = m_sourceManager.getSource(id);
302 if (source == nullptr) {
303 continue;
304 }
305 const auto buffer = source->getBuffer();
306 const bool needed = buffer && buffer->isValid() && ClipPrefilter::isNeeded(buffer->sampleRate, targetRate) &&
307 playedSources.count(id.value) > 0;
308
309 if (!needed) {
310 if (source->hasFilteredVariant()) {
311 source->clearFilteredVariant(); // rate no longer qualifies, or nothing plays it — free the copy
312 }
313 continue;
314 }
315 if (source->hasFilteredVariant() && !source->getFilteredBufferFor(targetRate)) {
316 source->clearFilteredVariant(); // keyed for an old rate/content — free it
317 }
318 if (source->filteredVariantSpec() != 0 &&
319 source->filteredVariantSpec() != ClipPrefilter::kSpecVersion) {
320 source->clearFilteredVariant(); // filter spec changed
321 }
322 // A whole-number rate ratio can be stored decimated: half the memory and no resampling, and the
323 // kernel plays it exactly as it played the full-rate copy (bit for bit) at speed and pitch 1. At any
324 // other speed the two differ (the decimated data is interpolated close to its Nyquist), so a source
325 // some clip plays at another speed needs the full-rate copy. The full-rate copy is sticky: once a
326 // source has it, it keeps it. While an upgrade builds, the decimated copy keeps playing.
327 const uint32_t factor = ClipPrefilter::decimationFactor(buffer->sampleRate, targetRate);
328 const auto current = source->getFilteredBufferFor(targetRate);
329 const bool currentIsFullRate = current && current->sampleRate != targetRate;
330 const uint32_t wantDecimation =
331 (factor > 1 && variedSources.count(id.value) == 0 && !currentIsFullRate) ? factor : 1;
332 if (current && (currentIsFullRate || wantDecimation > 1)) {
333 continue; // ready
334 }
335
336 if (!m_clipPrefilterService) {
337 // Completion callback: atomic-only graph-dirty request (thread-safe from
338 // the worker), so the app's graph pump applies the copy on next drain.
339 m_clipPrefilterService = std::make_unique<ClipPrefilterService>(
340 [this] { requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged); });
341 }
342 ClipPrefilterService::Job job;
343 job.sourceId = id.value;
344 job.contentRevision = source->getContentRevision();
345 job.targetRate = targetRate;
346 job.decimation = wantDecimation;
347 job.source = source->getSharedBuffer();
348 m_clipPrefilterService->enqueue(std::move(job)); // dedupes internally
349 }
350}

360bool TrackManager::prewarmSamplerKeepLength(bool wait) {
361 using Plugins::SamplerPlugin;
362 // Pitches each unit's notes use, across every pattern (a unit's notes can live in any pattern).
363 std::unordered_map<UnitID, std::vector<int>> pitchesByUnit;
364 for (const auto& pattern : m_patternManager.getAllPatterns()) {
365 if (!pattern || !pattern->isMidi()) continue;
366 for (const auto& note : pattern->getMidiNotes()) {
367 auto& pitches = pitchesByUnit[note.unitId];
368 if (std::find(pitches.begin(), pitches.end(), note.pitch) == pitches.end()) {
369 pitches.push_back(note.pitch);
370 }
371 }
372 }
373 bool allReady = true;
374 for (UnitID id : m_unitManager.getAllUnitIDs()) {
375 auto sampler = std::dynamic_pointer_cast<SamplerPlugin>(m_unitManager.getUnitPlugin(id));
376 if (!sampler || sampler->getPitchMode() != SamplerPlugin::PitchMode::KeepLength) continue;
377 const auto it = pitchesByUnit.find(id);
378 if (it == pitchesByUnit.end()) continue;
379 allReady = sampler->prewarmKeepLength(it->second, wait) && allReady;
380 }
381 return allReady;
382}
