25inline constexpr double kExprStepBeats = 0.25;
26inline constexpr double kMaxSwingDelayBeats = 0.125;
27inline constexpr uint8_t kMaxRatchet = 4;

30inline int gridStepOf(double startBeat) {
31 const double step = std::round(startBeat / kExprStepBeats);
32 return std::abs(startBeat - step * kExprStepBeats) < 0.002 ? static_cast<int>(step) : -1;
33}
34
35inline double swingDelayBeats(double startBeat, float swing) {
36 const int step = gridStepOf(startBeat);
37 if (step < 0 || (step & 1) == 0) return 0.0;
38 return static_cast<double>(std::clamp(swing, 0.0f, 1.0f)) * kMaxSwingDelayBeats;
39}

42inline bool chancePasses(uint64_t patternId, uint32_t instanceId, uint64_t unitId, double startBeat, int pitch,
43 uint64_t pass, float chance) {
44 if (chance >= 1.0f) return true;
45 if (chance <= 0.0f) return false;
46 uint64_t h = 0x9E3779B97F4A7C15ull ^ patternId;
47 auto mix = [&h](uint64_t v) {
48 h ^= v + 0x9E3779B97F4A7C15ull + (h << 6) + (h >> 2);
49 h ^= h >> 33;
50 h *= 0xff51afd7ed558ccdull;
51 h ^= h >> 33;
52 };
53 mix(instanceId);
54 mix(unitId);
55 mix(static_cast<uint64_t>(std::llround(startBeat * 960.0)));
56 mix(static_cast<uint64_t>(pitch));
57 mix(pass);
58 // Full finalizer (splitmix64): the mixing above leaves small input differences small, and a pass
59 // counter is exactly that, so without it every pass of a note gave nearly the same draw.
60 h = (h ^ (h >> 30)) * 0xbf58476d1ce4e5b9ull;
61 h = (h ^ (h >> 27)) * 0x94d049bb133111ebull;
62 h ^= h >> 31;
63 return static_cast<double>(h >> 11) * (1.0 / 9007199254740992.0) < static_cast<double>(chance);
64}

67inline bool needsExpression(const std::vector<MidiNote>& notes, float swing) {
68 if (swing > 0.0f) return true;
69 return std::any_of(notes.begin(), notes.end(), [](const MidiNote& n) { return n.chance < 1.0f || n.ratchet > 1; });
70}

73inline std::vector<MidiNote> expandForPlayback(const std::vector<MidiNote>& notes, float swing, uint64_t patternId,
74 uint32_t instanceId, uint64_t pass) {
75 std::vector<MidiNote> out;
76 out.reserve(notes.size());
77 for (const auto& n : notes) {
78 if (!chancePasses(patternId, instanceId, n.unitId, n.startBeat, n.pitch, pass, n.chance)) continue;
79 const double start = n.startBeat + swingDelayBeats(n.startBeat, swing);
80 const int hits = std::clamp<int>(n.ratchet, 1, kMaxRatchet);
81 if (hits == 1) {
82 MidiNote m = n;
83 m.startBeat = start;
84 out.push_back(m);
85 continue;
86 }
87 const double span = std::max(n.durationBeats, 0.01) / hits;
88 for (int i = 0; i < hits; ++i) {
89 MidiNote m = n;
90 m.startBeat = start + span * i;
91 m.durationBeats = span;
92 m.gate = n.gate / static_cast<float>(hits);
93 m.slide = n.slide && i == hits - 1;
94 m.ratchet = 1;
95 out.push_back(m);
96 }
97 }
98 return out;
99}
