25bool buffersAreFinite(float** buffer, uint32_t numChannels, uint32_t numFrames) noexcept {
26 for (uint32_t ch = 0; ch < numChannels; ++ch) {
27 float* channel = buffer[ch];
28 if (!channel) {
29 continue;
30 }
31 for (uint32_t i = 0; i < numFrames; ++i) {
32 if (!std::isfinite(channel[i])) {
33 return false;

98bool EffectChain::insertPlugin(size_t slotIndex, PluginInstancePtr plugin) {
99 if (reportRealtimeMisuse("EffectChain::insertPlugin")) {
100 return false;
101 }
102 if (slotIndex >= MAX_SLOTS) {
103 return false;
104 }
105
106 if (plugin) {
107 if (m_maxBlockSize > 0 && m_sampleRate > 0.0) {
108 plugin->initialize(m_sampleRate, m_maxBlockSize);
109 }
110 if (!plugin->isActive()) {
111 plugin->activate();
112 }
113 // Prewarm getInfo() here, off the RT path. Several implementations
114 // (e.g. BuiltInPlugins::samplerInfo) return a function-local static
115 // whose FIRST call constructs std::string fields — a heap allocation
116 // plus a __cxa_guard acquire. EffectChainSnapshot::process() calls
117 // getInfo() per slot on the audio thread, so without this the first
118 // block after an insert pays that cost on the RT thread (measured:
119 // 1x31-byte alloc; RTAllocationTrapTest, issue #432).
120 (void)plugin->getInfo();
121 }
122
123 m_slots[slotIndex].plugin = std::move(plugin);
124 m_slots[slotIndex].bypassed.store(false);
125 m_slots[slotIndex].dryWetMix.store(1.0f);
126 m_slots[slotIndex].faultState = std::make_shared<EffectSlotFaultState>();
127
128 publishSnapshot();
129 if (m_onLatencyChanged) {
130 m_onLatencyChanged();
131 }
132 return true;
133}

154bool EffectChain::movePlugin(size_t fromSlot, size_t toSlot) {
155 if (reportRealtimeMisuse("EffectChain::movePlugin")) {
156 return false;
157 }
158 if (fromSlot >= MAX_SLOTS || toSlot >= MAX_SLOTS) {
159 return false;
160 }
161
162 if (fromSlot == toSlot) {
163 return true;
164 }
165
166 // Can only move to empty slot
167 if (!m_slots[toSlot].isEmpty()) {
168 return false;
169 }
170
171 m_slots[toSlot].plugin = std::move(m_slots[fromSlot].plugin);
172 m_slots[toSlot].bypassed.store(m_slots[fromSlot].bypassed.load());
173 m_slots[toSlot].dryWetMix.store(m_slots[fromSlot].dryWetMix.load());
174 m_slots[toSlot].faultState = std::move(m_slots[fromSlot].faultState);
175 if (!m_slots[toSlot].faultState) {
176 m_slots[toSlot].faultState = std::make_shared<EffectSlotFaultState>();
177 }
178
179 m_slots[fromSlot].plugin = nullptr;
180 m_slots[fromSlot].bypassed.store(false);
181 m_slots[fromSlot].dryWetMix.store(1.0f);
182 m_slots[fromSlot].faultState = std::make_shared<EffectSlotFaultState>();
183
184 publishSnapshot();
185 return true;

208
209 m_slots[slot2].bypassed.store(bypass1);
210 m_slots[slot2].dryWetMix.store(mix1);

341void EffectChain::prepare(double sampleRate, uint32_t maxBlockSize) {
342 m_sampleRate = sampleRate;
343 m_maxBlockSize = maxBlockSize;
344
345 // Pre-allocate dry buffer for dry/wet mixing (stereo)
346 if (maxBlockSize == 0)
347 return;
348 const size_t required = static_cast<size_t>(maxBlockSize) * 2;
349 if (m_dryBuffer.size() < required) {
350 m_dryBuffer.resize(required);
351 }
352
353 for (auto& slot : m_slots) {
354 if (!slot.plugin) {
355 continue;
356 }
357 slot.plugin->initialize(m_sampleRate, m_maxBlockSize);
358 if (!slot.plugin->isActive()) {
359 slot.plugin->activate();
360 }
361 }
362
363 // Initialize snapshot for Pass 2
364 publishSnapshot();
365}

367void EffectChain::process(float** buffer, uint32_t numChannels, uint32_t numFrames, const float* const* sidechainInputs,
368 uint32_t numSidechainChannels) {
369 // Skip if entire chain is bypassed
370 if (m_chainBypassed.load(std::memory_order_acquire)) {
371 return;
372 }
373
374 // Process each slot in sequence
375 for (const auto& slot : m_slots) {
376 // Skip empty or bypassed slots
377 if (slot.isEmpty() || slot.bypassed.load(std::memory_order_acquire) || isFaultBypassed(slot.faultState)) {
378 continue;
379 }
380
381 auto& plugin = slot.plugin;
382 if (!plugin) {
383 continue;
384 }
385
386 if (!plugin->isActive()) {
387 // printf("[EffectChain] Plugin exists but inactive!\n");
388 continue;
389 }
390
391 float dryWet = slot.dryWetMix.load(std::memory_order_acquire);
392
393 const PluginInfo& pluginInfo = plugin->getInfo();
394 const bool isBuiltInComp = (pluginInfo.id == "com.Aestrastudios.comp");
395 const uint32_t requestedInputChannels = numChannels + numSidechainChannels;
396 const bool canUseSidechain = sidechainInputs && numSidechainChannels > 0 &&
397 (pluginInfo.numAudioInputs >= requestedInputChannels || isBuiltInComp);
398 std::array<const float*, 4> inputChannels{};
399 const float* const* processInputs = reinterpret_cast<const float* const*>(buffer);
400 uint32_t processInputChannels = numChannels;
401 if (canUseSidechain) {
402 for (uint32_t ch = 0; ch < numChannels && ch < inputChannels.size(); ++ch) {
403 inputChannels[ch] = buffer[ch];
404 }
405 for (uint32_t ch = 0; ch < numSidechainChannels && (numChannels + ch) < inputChannels.size(); ++ch) {
406 inputChannels[numChannels + ch] = sidechainInputs[ch];
407 }
408 processInputs = inputChannels.data();
409 processInputChannels = isBuiltInComp
410 ? requestedInputChannels
411 : std::min<uint32_t>(pluginInfo.numAudioInputs, requestedInputChannels);
412 }
413
414 // If fully wet, process directly
415 if (dryWet >= 0.999f) {
416 const bool processed =
417 processPluginNoexcept(*plugin, processInputs, buffer, processInputChannels, numChannels, numFrames);
418 if (!processed) {
419 clearFloatBuffers(buffer, numChannels, numFrames);
420 } else if (!buffersAreFinite(buffer, numChannels, numFrames)) {
421 markNonFiniteOutputFault(slot.faultState);
422 clearFloatBuffers(buffer, numChannels, numFrames);
423 }
424 }
425 // If not fully wet, need to blend
426 else if (dryWet > 0.001f) {
427 const uint32_t blendChannels = (numChannels < 2) ? numChannels : 2;
428 const size_t requiredDry = static_cast<size_t>(numFrames) * static_cast<size_t>(blendChannels);
429 if (m_dryBuffer.size() < requiredDry) {
430 // Not prepared (or prepared for a smaller block). Stay RT-safe: no allocation.
431 const bool processed =
432 processPluginNoexcept(*plugin, processInputs, buffer, processInputChannels, numChannels, numFrames);
433 if (!processed) {
434 clearFloatBuffers(buffer, numChannels, numFrames);
435 } else if (!buffersAreFinite(buffer, numChannels, numFrames)) {
436 markNonFiniteOutputFault(slot.faultState);
437 clearFloatBuffers(buffer, numChannels, numFrames);
438 }
439 continue;
440 }
441
442 // Save dry signal
443 for (uint32_t ch = 0; ch < blendChannels; ++ch) {
444 std::memcpy(m_dryBuffer.data() + ch * numFrames, buffer[ch], numFrames * sizeof(float));
445 }
446
447 const bool processed =
448 processPluginNoexcept(*plugin, processInputs, buffer, processInputChannels, numChannels, numFrames);
449 if (!processed) {
450 restoreDryChannels(buffer, m_dryBuffer.data(), numChannels, blendChannels, numFrames);
451 continue;
452 }
453 if (!buffersAreFinite(buffer, numChannels, numFrames)) {
454 markNonFiniteOutputFault(slot.faultState);
455 restoreDryChannels(buffer, m_dryBuffer.data(), numChannels, blendChannels, numFrames);
456 continue;
457 }
458
459 // Blend dry/wet
460 float wetGain = dryWet;
461 float dryGain = 1.0f - dryWet;
462
463 for (uint32_t ch = 0; ch < blendChannels; ++ch) {
464 const float* dry = m_dryBuffer.data() + ch * numFrames;
465 float* wet = buffer[ch];
466
467 for (uint32_t i = 0; i < numFrames; ++i) {
468 wet[i] = dry[i] * dryGain + wet[i] * wetGain;
469 }
470 }
471 }
472 // dryWet <= 0.001f means fully dry, skip processing
473 }
474}

480void EffectChainSnapshot::process(float** buffer, uint32_t numChannels, uint32_t numFrames,
481 const float* const* sidechainInputs, uint32_t numSidechainChannels,
482 float* dryBuffer) const {
483 // Skip if entire chain is bypassed
484 if (isChainBypassed) {
485 return;
486 }
487
488 // Process each slot in sequence
489 for (size_t slotIdx = 0; slotIdx < MAX_SLOTS; ++slotIdx) {
490 const auto& slot = m_slots[slotIdx];
491
492 // Skip empty or bypassed slots
493 if (slot.isEmpty() || slot.bypassed || isFaultBypassed(slot.faultState)) {
494 continue;
495 }
496
497 auto& plugin = slot.plugin;
498 if (!plugin) {
499 continue;
500 }
501
502 if (!plugin->isActive()) {
503 continue;
504 }
505
506 float dryWet = slot.dryWetMix;
507
508 const PluginInfo& pluginInfo = plugin->getInfo();
509 const bool isBuiltInComp = (pluginInfo.id == "com.Aestrastudios.comp");
510 const uint32_t requestedInputChannels = numChannels + numSidechainChannels;
511 const bool canUseSidechain = sidechainInputs && numSidechainChannels > 0 &&
512 (pluginInfo.numAudioInputs >= requestedInputChannels || isBuiltInComp);
513 std::array<const float*, 4> inputChannels{};
514 const float* const* processInputs = reinterpret_cast<const float* const*>(buffer);
515 uint32_t processInputChannels = numChannels;
516 if (canUseSidechain) {
517 for (uint32_t ch = 0; ch < numChannels && ch < inputChannels.size(); ++ch) {
518 inputChannels[ch] = buffer[ch];
519 }
520 for (uint32_t ch = 0; ch < numSidechainChannels && (numChannels + ch) < inputChannels.size(); ++ch) {
521 inputChannels[numChannels + ch] = sidechainInputs[ch];
522 }
523 processInputs = inputChannels.data();
524 processInputChannels = isBuiltInComp
525 ? requestedInputChannels
526 : std::min<uint32_t>(pluginInfo.numAudioInputs, requestedInputChannels);
527 }
528
529 // If fully wet, process directly
530 if (dryWet >= 0.999f) {
531 const bool processed =
532 processPluginNoexcept(*plugin, processInputs, buffer, processInputChannels, numChannels, numFrames);
533 if (!processed) {
534 clearFloatBuffers(buffer, numChannels, numFrames);
535 } else if (!buffersAreFinite(buffer, numChannels, numFrames)) {
536 markNonFiniteOutputFault(slot.faultState);
537 clearFloatBuffers(buffer, numChannels, numFrames);
538 }
539 }
540 // If not fully wet, need to blend
541 else if (dryWet > 0.001f) {
542 const uint32_t blendChannels = (numChannels < 2) ? numChannels : 2;
543
544 // Save dry signal into caller-provided buffer
545 for (uint32_t ch = 0; ch < blendChannels; ++ch) {
546 std::memcpy(dryBuffer + ch * numFrames, buffer[ch], numFrames * sizeof(float));
547 }
548
549 const bool processed =
550 processPluginNoexcept(*plugin, processInputs, buffer, processInputChannels, numChannels, numFrames);
551 if (!processed) {
552 restoreDryChannels(buffer, dryBuffer, numChannels, blendChannels, numFrames);
553 continue;
554 }
555 if (!buffersAreFinite(buffer, numChannels, numFrames)) {
556 markNonFiniteOutputFault(slot.faultState);
557 restoreDryChannels(buffer, dryBuffer, numChannels, blendChannels, numFrames);
558 continue;
559 }
560
561 // Blend dry/wet
562 float wetGain = dryWet;
563 float dryGain = 1.0f - dryWet;
564
565 for (uint32_t ch = 0; ch < blendChannels; ++ch) {
566 const float* dry = dryBuffer + ch * numFrames;
567 float* wet = buffer[ch];
568
569 for (uint32_t i = 0; i < numFrames; ++i) {
570 wet[i] = dry[i] * dryGain + wet[i] * wetGain;
571 }
572 }
573 }
574 // dryWet <= 0.001f means fully dry, skip processing
575 }
576}

582std::vector<uint8_t> EffectChain::saveState() const {
583 std::vector<uint8_t> state;
584
585 // Write header
586 state.push_back('N');
587 state.push_back('E');
588 state.push_back('C'); // Aestra Effect Chain
589 state.push_back(1); // Version
590
591 // Write slot count
592 state.push_back(static_cast<uint8_t>(MAX_SLOTS));
593
594 // Write each slot
595 for (size_t i = 0; i < MAX_SLOTS; ++i) {
596 const auto& slot = m_slots[i];
597
598 if (slot.isEmpty()) {
599 state.push_back(0); // Empty flag
600 continue;
601 }
602
603 state.push_back(1); // Has plugin flag
604
605 // Save plugin ID
606 const auto& info = slot.plugin->getInfo();
607 uint32_t idLen = static_cast<uint32_t>(info.id.size());
608 state.insert(state.end(), reinterpret_cast<const uint8_t*>(&idLen),
609 reinterpret_cast<const uint8_t*>(&idLen) + sizeof(idLen));
610 state.insert(state.end(), info.id.begin(), info.id.end());
611
612 // Save bypass state
613 state.push_back(slot.bypassed.load() ? 1 : 0);
614
615 // Save dry/wet
616 float dryWet = slot.dryWetMix.load();
617 state.insert(state.end(), reinterpret_cast<const uint8_t*>(&dryWet),
618 reinterpret_cast<const uint8_t*>(&dryWet) + sizeof(dryWet));
619
620 // Save plugin state
621 auto pluginState = slot.plugin->saveState();
622 uint32_t stateLen = static_cast<uint32_t>(pluginState.size());
623 state.insert(state.end(), reinterpret_cast<const uint8_t*>(&stateLen),
624 reinterpret_cast<const uint8_t*>(&stateLen) + sizeof(stateLen));
625 state.insert(state.end(), pluginState.begin(), pluginState.end());
626 }
627
628 return state;
629}

631bool EffectChain::loadState(const std::vector<uint8_t>& state, PluginManager& manager) {
632 if (reportRealtimeMisuse("EffectChain::loadState")) {
633 return false;
634 }
635 if (state.size() < 5) {
636 return false;
637 }
638
639 // Check header
640 if (state[0] != 'N' || state[1] != 'E' || state[2] != 'C' || state[3] != 1) {
641 return false;
642 }
643
644 uint8_t slotCount = state[4];
645 if (slotCount != MAX_SLOTS) {
646 return false;
647 }
648
649 size_t offset = 5;
650
651 for (size_t i = 0; i < MAX_SLOTS && offset < state.size(); ++i) {
652 uint8_t hasPlugin = state[offset++];
653
654 if (!hasPlugin) {
655 m_slots[i].plugin = nullptr;
656 m_slots[i].faultState = std::make_shared<EffectSlotFaultState>();
657 continue;
658 }
659
660 if (offset + sizeof(uint32_t) > state.size()) {
661 return false;
662 }
663
664 // Read plugin ID
665 uint32_t idLen;
666 std::memcpy(&idLen, &state[offset], sizeof(idLen));
667 offset += sizeof(idLen);
668
669 if (offset + idLen > state.size()) {
670 return false;
671 }
672
673 std::string pluginId(reinterpret_cast<const char*>(&state[offset]), idLen);
674 offset += idLen;
675
676 if (offset + 1 + sizeof(float) > state.size()) {
677 return false;
678 }
679
680 // Read bypass state
681 bool bypassed = state[offset++] != 0;
682
683 // Read dry/wet
684 float dryWet;
685 std::memcpy(&dryWet, &state[offset], sizeof(dryWet));
686 offset += sizeof(dryWet);
687
688 // Read plugin state length
689 uint32_t stateLen;
690 std::memcpy(&stateLen, &state[offset], sizeof(stateLen));
691 offset += sizeof(stateLen);
692
693 if (offset + stateLen > state.size()) {
694 return false;
695 }
696
697 std::vector<uint8_t> pluginState(state.begin() + offset, state.begin() + offset + stateLen);
698 offset += stateLen;
699
700 // Create plugin instance
701 auto instance = manager.createInstanceById(pluginId);
702 if (instance) {
703 instance->initialize(m_sampleRate, m_maxBlockSize);
704 if (!instance->loadState(pluginState)) {
705 Aestra::Log::warning("[EffectChain] Failed to load state for plugin slot " + std::to_string(i) +
706 " — using default state");
707 }
708 instance->activate();
709
710 m_slots[i].plugin = std::move(instance);
711 m_slots[i].bypassed.store(bypassed);
712 m_slots[i].dryWetMix.store(dryWet);
713 m_slots[i].faultState = std::make_shared<EffectSlotFaultState>();
714 }
715 }
716
717 publishSnapshot();
718 return true;
719}

737void EffectChain::reset() {
738 if (reportRealtimeMisuse("EffectChain::reset")) {
739 return;
740 }
741 // 1. Temporarily bypass the chain to silence audio input
742 bool wasBypassed = m_chainBypassed.exchange(true);
743
744 // 2. Reboot each plugin to clear internal buffers (delay lines, etc.)
745 for (auto& slot : m_slots) {
746 if (slot.plugin) {
747 // Check if active before resetting?
748 if (slot.plugin->isActive()) {
749 slot.plugin->deactivate();
750 slot.plugin->activate();
751 }
752 }
753 }
754
755 // 3. Restore original bypass state
756 m_chainBypassed.store(wasBypassed);
757
758 publishSnapshot();
759}
