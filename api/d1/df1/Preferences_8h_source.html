11// =============================================================================
12// User Preferences - Persisted across sessions
13// =============================================================================
14class Preferences {
15public:
16 static Preferences& instance();
17
18 // WHICH OF THESE ACTUALLY GOVERN ANYTHING
19 //
20 // A persisted field that nothing reads is worse than a missing one: it makes
21 // a setting look configurable, and any code written against it validates,
22 // persists, reports success and changes nothing. Each field below says which
23 // it is. Keep this accurate — the audit that produced it is in
24 // docs/technical/settings_view_state_map.md (B4).
25
26 // --- Audio settings: NOT WIRED, and now a DUPLICATE ----------------------
27 // Nothing reads these. audio_settings.conf is the persisted authority — one
28 // parser, AudioSettingsStore — and AudioDeviceManager is the runtime one.
29 //
30 // Device, sample rate and buffer size DO survive a restart as of #649; they
31 // are applied at startup from that file, before any settings UI exists. So
32 // these fields are not an unimplemented feature waiting to be wired: they
33 // are a second, dead copy of settings that already work elsewhere, and the
34 // two have already drifted (this said bufferSize 512 while the live config
35 // said 256).
36 //
37 // Do not wire them. Removing them is tracked in #652; until then, anything
38 // written against these would silently override a working setting.
39 std::string audioDeviceId = "default";
40 int sampleRate = 48000;
41 int bufferSize = 512;
42 bool exclusiveMode = false;
43
44 // --- UI settings ----------------------------------------------------------
45 // showGrid / snapToGrid / gridSize: NOT WIRED. Shadowed by TrackManagerUI's
46 // own m_snapEnabled / m_snapSetting, which are the live authority. Snap
47 // choice does not survive a restart.
48 bool showGrid = true;
49 bool snapToGrid = true;
50 float gridSize = 0.25f; // beats
51 // theme: LIVE. Written by AppearanceSettingsPage::applyChanges, read at
52 // startup by AestraWindowManager. The live authority is NUIThemeManager, so
53 // anything changing the theme must write both.
54 std::string theme = "Aestra-dark";
55
56 // --- Auto-save: LIVE ------------------------------------------------------
57 // Both are read at startup by AestraApp and applied to AutosaveManager.
58 //
59 // load() stores whatever the file says, unvalidated. The clamp to
60 // [10, 3600] happens in AestraApp::resolveAutosaveInterval(), which is the
61 // single place the interval is resolved for both autosave initialisation
62 // and reinitialisation, and which writes the corrected value back. So an
63 // out-of-range value can exist in memory between load() and that call.
64 bool autoSaveEnabled = true;
65 int autoSaveIntervalSeconds = 300;
66
67 // --- Recent files: NOT WIRED ---------------------------------------------
68 // The storage, cap, and add/clear helpers below are implemented and correct;
69 // nothing calls them and no UI surfaces the list. Kept rather than deleted
70 // because a Recent Files menu is a plausible near-term feature and this is
71 // the plumbing it would need — but until something calls addRecentFile(),
72 // this list is always empty.
73 std::vector<std::string> recentFiles;
74 static constexpr size_t MAX_RECENT_FILES = 10;
75
76 // Load/save
77 void load();
78 void save();
79 void addRecentFile(const std::string& path);
80 void clearRecentFiles();
81
82 // Path helpers
83 static std::string getPreferencesPath();
84 static std::string getRecentFilesPath();
85
86private:
87 Preferences() = default;
88 ~Preferences() = default;
89 Preferences(const Preferences&) = delete;
90 Preferences& operator=(const Preferences&) = delete;
91
92 void createDefaults();
93 bool loadFromJson(const std::string& jsonStr);
94 JSON toJson() const;
95 void fromJson(const JSON& json);
96};
