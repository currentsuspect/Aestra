15void DeleteLaneCommand::execute() {
16 if (m_executed) {
17 return;
18 }
19 auto& model = m_trackManager.getPlaylistModel();
20 const PlaylistLane* lane = model.getLane(m_laneId);
21 if (!lane) {
22 return;
23 }
24 m_name = lane->name;
25 m_trackId = lane->trackId;
26 m_clips = lane->clips;
27 m_playlistIndex = model.getLaneIndex(m_laneId);
28 if (m_trackId != 0) {
29 if (const Track* track = m_trackManager.getTrack(m_trackId)) {
30 // A track's last lane cannot be deleted (UI and command policy):
31 // an owned track must keep at least its primary lane.
32 if (track->laneIds.size() <= 1) {
33 return;
34 }
35 const auto it = std::find(track->laneIds.begin(), track->laneIds.end(), m_laneId);
36 if (it != track->laneIds.end()) {
37 m_laneIdsIndex = static_cast<int>(it - track->laneIds.begin());
38 }
39 m_wasActiveLane = track->activeLaneId == m_laneId;
40 }
41 m_trackManager.detachLaneFromTrack(m_trackId, m_laneId);
42 }
43 model.removeLane(m_laneId);
44 m_trackManager.requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged);
45 m_executed = true;
46}

48void DeleteLaneCommand::undo() {
49 if (!m_executed) {
50 return;
51 }
52 auto& model = m_trackManager.getPlaylistModel();
53 const PlaylistLaneID restored = model.createLaneWithId(m_laneId, m_name);
54 for (const auto& clip : m_clips) {
55 model.addClip(restored, clip);
56 }
57 if (m_trackId != 0) {
58 m_trackManager.attachLaneToTrack(m_trackId, restored);
59 if (m_laneIdsIndex >= 0) {
60 m_trackManager.moveLaneWithinTrack(m_trackId, restored, static_cast<size_t>(m_laneIdsIndex));
61 }
62 // detachLaneFromTrack reassigned activeLaneId to lanes.back(); put the
63 // deleted lane back in charge when it was active before the delete.
64 if (m_wasActiveLane) {
65 if (Track* track = m_trackManager.getTrack(m_trackId)) {
66 track->activeLaneId = restored;
67 }
68 }
69 }
70 if (m_playlistIndex >= 0) {
71 model.moveLaneToIndex(restored, static_cast<size_t>(m_playlistIndex));
72 }
73 m_trackManager.requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged);
74 m_executed = false;
75}

77void DeleteLaneCommand::redo() {
78 if (m_executed) {
79 return;
80 }
81 auto& model = m_trackManager.getPlaylistModel();
82 if (m_trackId != 0) {
83 m_trackManager.detachLaneFromTrack(m_trackId, m_laneId);
84 }
85 model.removeLane(m_laneId);
86 m_trackManager.requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged);
87 m_executed = true;
88}
