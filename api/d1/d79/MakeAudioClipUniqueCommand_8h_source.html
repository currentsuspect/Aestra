15public:
16 MakeAudioClipUniqueCommand(TrackManager& manager, ClipInstanceID clipId) : m_manager(manager), m_clipId(clipId) {}
17
18 void execute() override {
19 if (m_executed)
20 return;
21 const auto* clip = m_manager.getPlaylistModel().getClip(m_clipId);
22 if (!clip || !clip->patternId.isValid())
23 return;
24 const auto* sourcePattern = m_manager.getPatternManager().getPattern(clip->patternId);
25 if (!sourcePattern || !sourcePattern->isAudio())
26 return;
27 if (!m_originalPatternId.isValid()) {
28 m_originalPatternId = clip->patternId;
29 }
30 if (m_detachedPattern) {
31 if (!m_manager.getPatternManager().reinsertPattern(m_detachedPattern))
32 return;
33 } else {
34 m_uniquePatternId = m_manager.getPatternManager().clonePattern(m_originalPatternId);
35 if (!m_uniquePatternId.isValid())
36 return;
37 }
38 if (!m_manager.getPlaylistModel().setClipPattern(m_clipId, m_uniquePatternId)) {
39 m_detachedPattern = m_manager.getPatternManager().detachPattern(m_uniquePatternId);
40 return;
41 }
42 m_manager.markModified();
43 m_executed = true;
44 }
45
46 void undo() override {
47 if (!m_executed)
48 return;
49 if (m_manager.getPlaylistModel().setClipPattern(m_clipId, m_originalPatternId)) {
50 m_detachedPattern = m_manager.getPatternManager().detachPattern(m_uniquePatternId);
51 m_manager.markModified();
52 m_executed = false;
53 }
54 }
55
56 void redo() override { execute(); }
57
58 std::string getName() const override { return "Make Audio Clip Unique"; }
59 size_t getSizeInBytes() const override { return sizeof(*this); }
60 bool changesProjectState() const override { return true; }
61
