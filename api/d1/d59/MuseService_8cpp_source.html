24void MuseService::wireHeadlessEngine(const std::shared_ptr<TrackManager>& trackManager,
25 AudioEngine& engine) {
26 engine.setTrackManager(trackManager);
27 engine.setUnitManager(&trackManager->getUnitManager());
28 engine.setPatternPlaybackEngine(&trackManager->getPatternPlaybackEngine());
29 engine.setContinuousParams(trackManager->getContinuousParams());
30 // Meters: the engine writes per-slot peaks/RMS/LUFS only when a snapshot
31 // buffer is installed (the app does this in AestraApp).
32 auto meterBuffer = std::make_shared<MeterSnapshotBuffer>();
33 engine.setMeterSnapshots(meterBuffer);
34 trackManager->setMeterSnapshots(meterBuffer);
35 trackManager->buildAndShareSlotMap();
36 if (auto slotMap = trackManager->getChannelSlotMapShared()) {
37 engine.setChannelSlotMap(slotMap);
38 }
39
40 // Transport commands (play/stop/seek) travel from TrackManager to the
41 // engine through this sink — the same relay AestraContent installs.
42 TrackManager* tm = trackManager.get();
43 AudioEngine* enginePtr = &engine;
44 tm->setCommandSink([enginePtr, tm](const AudioQueueCommand& cmd) -> bool {
45 // Edges deliver reliably (bounded wait, counted in edgeDroppedCount()
46 // on failure); state keeps best-effort drop-newest semantics (#913).
47 bool accepted = false;
48 if (AudioCommandQueue::isEdgeCommand(cmd.type)) {
49 accepted = enginePtr->commandQueue().pushReliable(cmd);
50 } else {
51 accepted = enginePtr->commandQueue().push(cmd);
52 }
53 if (accepted && cmd.type == AudioQueueCommandType::SetTransportState) {
54 const double sampleRate =
55 std::max(1.0, static_cast<double>(enginePtr->getSampleRate()));
56 // onTransportStateApplied resolves the kTransportPreservePosition pause
57 // sentinel and the seconds conversion in one place (#590).
58 tm->onTransportStateApplied(cmd.value1 != 0.0f, cmd.samplePos, sampleRate);
59 }
60 return accepted;
61 });
62}

64std::string MuseService::handleRequest(const std::string& requestJson) {
65 double id = 0.0;
66 std::string verb;
67
68 // Parsing gets its own try so only malformed JSON reports parse_error;
69 // runtime failures later map to execution_error.
70 JSON request;
71 try {
72 request = JSON::parse(requestJson);
73 } catch (const std::exception& e) {
74 return makeError(id, "parse_error", std::string("invalid request: ") + e.what()).toString();
75 } catch (...) {
76 return makeError(id, "parse_error", "invalid request").toString();
77 }
78
79 try {
80 if (!request.isObject()) {
81 return makeError(id, "parse_error", "request must be a JSON object").toString();
82 }
83
84 if (request.has("id")) {
85 if (!request["id"].isNumber()) {
86 // A wrong-typed id would silently echo 0 and mis-correlate
87 // responses; reject instead.
88 return makeError(id, "parse_error", "id must be a number").toString();
89 }
90 id = request["id"].asNumber();
91 }
92 if (!request.has("verb") || !request["verb"].isString()) {
93 return makeError(id, "parse_error", "missing string field: verb").toString();
94 }
95 verb = request["verb"].asString();
96
97 // Host verbs are namespaced (settings.setAudioDevice) and validated by
98 // the registry, which owns their argument schemas. Built-ins are
99 // checked here. A host verb can never shadow a built-in: the registry
100 // refuses the reserved audio. prefix and every built-in is unprefixed.
101 const bool isHostVerb = m_hostVerbs.has(verb);
102
103 // Classify the verb before any dependency checks so protocol status
104 // never depends on how the service happens to be wired.
105 if (!isHostVerb && !isKnownVerb(verb)) {
106 return makeError(id, "parse_error", "unknown command: " + verb).toString();
107 }
108
109 // Most queries take no arguments; accepting any would silently drop
110 // caller intent.
111 if (!isHostVerb && isQueryVerb(verb) && !queryTakesArgs(verb) && request.has("args") &&
112 request["args"].size() > 0) {
113 return makeError(id, "validation_error", verb + " takes no arguments", verb).toString();
114 }
115
116 // The verbs that stay here: the host seam and the manifest. They are
117 // short, they sit between the session and routing families in the old
118 // order, and they are the only ones that need m_hostVerbs.
119 const auto startTime = std::chrono::steady_clock::now();
120 const auto finish = [&](JSON& response) {
121 const auto endTime = std::chrono::steady_clock::now();
122 response.set("executionMs",
123 JSON(std::chrono::duration<double, std::milli>(endTime - startTime).count()));
124 return response.toString();
125 };
126 const auto makeOk = [&]() {
127 JSON response = JSON::object();
128 response.set("id", JSON(id));
129 response.set("status", JSON("ok"));
130 response.set("verb", JSON(verb));
131 return response;
132 };
133
134 // ------------------------------------------------------------------
135 // Verb families.
136 //
137 // The chain that used to be one 2200-line if-statement, now one call
138 // per family. Order is the old order: session state, then the host seam
139 // and the manifest below, then routing diagnostics, library reads,
140 // history, and finally the render actions. Each family returns nullopt
141 // for a verb it does not own, which is what a fall-through past its
142 // internal if-chain used to do.
143 //
144 // A family that answers owns the request outright: it returns the
145 // finished response string, including its own executionMs stamp.
146 // ------------------------------------------------------------------
147 // Named museCtx, not ctx: the mutation fallback further down declares its
148 // own `CommandContext ctx`, lifted verbatim from the original, and two
149 // locals of that name in one scope is a redefinition.
150 const MuseInternal::RequestContext museCtx{
151 id, verb, &request, m_trackManager, m_engine, &m_projectLoadReport};
152 const MuseInternal::ResponseEnvelope envelope(id, verb);
153
154 if (auto handled = MuseInternal::handleSessionVerbs(museCtx, envelope)) return *handled;
155
156 // ------------------------------------------------------------------
157 // Host verbs: capabilities the application registered into the seam.
158 // MuseService does not know what any of them do — it validates against
159 // the declared schema, checks that this process can honour the verb's
160 // thread affinity, and calls the handler.
161 // ------------------------------------------------------------------
162 if (isHostVerb) {
163 JSON noArgs = JSON::object();
164 const JSON& hostArgs = request.has("args") ? request["args"] : noArgs;
165 const HostVerbResult hostResult =
166 m_hostVerbs.invoke(verb, hostArgs, m_hostUiThreadAvailable);
167
168 if (!hostResult.ok) {
169 // The registry's own refusals are argument/affinity problems;
170 // anything else is the host declining to do the thing. Keep the
171 // machine-readable code in the response so callers never have to
172 // pattern-match prose to tell them apart.
173 const bool validation = hostResult.errorCode == "invalid_args" ||
174 hostResult.errorCode == "missing_arg" ||
175 hostResult.errorCode == "unknown_verb";
176 JSON response = makeError(id, validation ? "validation_error" : "execution_error",
177 hostResult.message, verb);
178 response.set("errorCode", JSON(hostResult.errorCode));
179 return finish(response);
180 }
181
182 JSON response = makeOk();
183 response.set("result", hostResult.result);
184 return finish(response);
185 }
186
187 if (verb == "get_capabilities") {
188 // "What can this host do?" — so an agent discovers the surface
189 // instead of hardcoding assumptions about which build it is talking
190 // to. A headless session legitimately answers with an empty list.
191 JSON verbs = JSON::array();
192 for (const auto& spec : m_hostVerbs.capabilities()) {
193 JSON entry = JSON::object();
194 entry.set("verb", JSON(spec.name));
195 entry.set("domain", JSON(std::string(HostVerbRegistry::domainName(spec.domain))));
196 entry.set("description", JSON(spec.description));
197 entry.set("mutates", JSON(spec.mutates));
198 entry.set("requiresHostUi",
199 JSON(spec.affinity == HostThreadAffinity::HostUiThread));
200 JSON args = JSON::array();
201 for (const auto& arg : spec.args) {
202 JSON a = JSON::object();
203 a.set("name", JSON(arg.name));
204 a.set("type", JSON(std::string(HostVerbRegistry::argTypeName(arg.type))));
205 a.set("required", JSON(arg.required));
206 if (!arg.description.empty()) a.set("description", JSON(arg.description));
207 if (!std::isnan(arg.minValue)) a.set("min", JSON(arg.minValue));
208 if (!std::isnan(arg.maxValue)) a.set("max", JSON(arg.maxValue));
209 args.push(a);
210 }
211 entry.set("args", args);
212 verbs.push(entry);
213 }
214 JSON result = JSON::object();
215 result.set("hostVerbs", verbs);
216 result.set("hostUiAvailable", JSON(m_hostUiThreadAvailable));
217 JSON response = makeOk();
218 response.set("result", result);
219 return finish(response);
220 }
221
222 if (verb == "get_schema") {
223 // The tool manifest, over the wire: a socket client can bootstrap
224 // without access to the binary's --schema flag.
225 JSON result = JSON::parse(MuseGrammar::schemaToJsonString());
226 JSON response = makeOk();
227 response.set("result", result);
228 return finish(response);
229 }
230 if (auto handled = MuseInternal::handleRoutingVerbs(museCtx, envelope)) return *handled;
231 if (auto handled = MuseInternal::handleLibraryVerbs(museCtx, envelope)) return *handled;
232 if (auto handled = MuseInternal::handleHistoryVerbs(museCtx, envelope)) return *handled;
233 if (auto handled = MuseInternal::handleRenderVerbs(museCtx, envelope)) return *handled;
234
235 // ------------------------------------------------------------------
236 // Mutation verbs: JSON args -> flag map -> the same schema
237 // validation, registry factories, and CommandHistory the text
238 // parser uses.
239 // ------------------------------------------------------------------
240 if (!m_trackManager) {
241 return makeError(id, "execution_error", "no track manager", verb).toString();
242 }
243
244 std::unordered_map<std::string, std::string> flags;
245 if (request.has("args")) {
246 JSON& args = request["args"];
247 if (!args.isObject()) {
248 return makeError(id, "parse_error", "args must be an object", verb).toString();
249 }
250 std::string convertError;
251 if (!jsonArgsToFlags(args, flags, convertError)) {
252 return makeError(id, "parse_error", convertError, verb).toString();
253 }
254 }
255
256 CommandParser parser;
257 const CommandContext ctx{m_engine, m_trackManager};
258 CommandResult cmdResult =
259 parser.execute(verb, flags, m_trackManager->getCommandHistory(), ctx);
260
261 JSON response = JSON::object();
262 response.set("id", JSON(id));
263 response.set("status", JSON(statusName(cmdResult.status)));
264 response.set("verb", JSON(verb));
265 response.set("message", JSON(cmdResult.message));
266 response.set("undoable", JSON(cmdResult.undoable));
267 if (cmdResult.hasCreatedId) {
268 // Structured id of the object the command created (e.g. the new
269 // pattern from clone_pattern) — agents must not parse the message.
270 JSON result = JSON::object();
271 result.set("createdId", JSON(static_cast<double>(cmdResult.createdId)));
272 response.set("result", result);
273 }
274 response.set("executionMs", JSON(cmdResult.executionMs));
275 return response.toString();
276 } catch (const std::exception& e) {
277 return makeError(id, "execution_error", e.what(), verb).toString();
278 } catch (...) {
279 return makeError(id, "execution_error", "internal error", verb).toString();
280 }
281}
