4// SIGTERM / SIGINT: autosave and exit, never a prompt (SPEC 3 §6.7, owner ruling A).
5//
6// SDL converts these signals into SDL_QUIT, which the platform layer routes to the
7// same close callback as the window's close button, so a signal got the interactive
8// "Unsaved Changes" dialog and the process waited on it forever: a logout or a
9// scripted `kill` never finished. Installing this handler replaces SDL's (SDL only
10// installs its own over SIG_DFL, and on quit only removes a handler that is still
11// its own), so the close button keeps its prompt and a signal takes this path.
12//
13// The handler only records which signal arrived. The main loop polls pending() and
14// does the actual work (autosave, keep the session recoverable, exit) on the main
15// thread, where it is allowed to.
16//
17// Deliberately dependency-free, like CrashFlagPath.h, so it is unit-testable while
18// Source/ has no linkable library target (#666).
19
20#include <atomic>
21#include <csignal> // on POSIX this also declares sigaction
22
23namespace Aestra {
24
25class TerminationSignal {
26public:
29 static bool install() {
30#if defined(_WIN32)
31 return false;
32#else
33 struct sigaction action {};
34 action.sa_handler = &TerminationSignal::handle;
35 sigemptyset(&action.sa_mask);
36 action.sa_flags = SA_RESTART;
37 const bool term = sigaction(SIGTERM, &action, nullptr) == 0;
38 const bool interrupt = sigaction(SIGINT, &action, nullptr) == 0;
39 return term && interrupt;
40#endif
41 }
42
44 static int pending() { return s_pending.load(std::memory_order_relaxed); }
45
47 static void resetForTest() { s_pending.store(0, std::memory_order_relaxed); }
48
49private:
50 // Async-signal-safe: one lock-free store, nothing else.
51 static void handle(int signalNumber) { s_pending.store(signalNumber, std::memory_order_relaxed); }
52
53 static_assert(std::atomic<int>::is_always_lock_free, "the signal handler needs a lock-free flag");
54 // Constant-initialized, so the handler never runs a guarded static initialization.
55 inline static std::atomic<int> s_pending{0};
56};
57
58} // namespace Aestra
