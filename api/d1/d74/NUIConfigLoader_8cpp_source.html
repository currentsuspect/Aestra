31bool NUIConfigLoader::loadConfigFromString(const std::string& content) {
32 config_ = Aestra::JSON::object();
33
34 // saveConfig() serializes the theme as JSON (Aestra::JSON::toString), but the
35 // hand-authored config assets under AestraUI/Config/ are YAML. Detect which
36 // format we were handed: a JSON document must be parsed with the JSON parser,
37 // a YAML document with the line-based parseYAML(). Feeding JSON to parseYAML()
38 // silently restored nothing — loadConfig() returned true but applied no values,
39 // so a saved config never round-tripped. See issue #585.
40 const size_t firstNonSpace = content.find_first_not_of(" \t\r\n");
41 const bool looksLikeJson =
42 firstNonSpace != std::string::npos && content[firstNonSpace] == '{';
43
44 if (looksLikeJson) {
45 Aestra::JSON parsed = Aestra::JSON::parse(content);
46 // parse() yields a null (non-object) JSON on malformed input; accept only a
47 // real object so a truncated or corrupt file fails loudly instead of
48 // wiping the live theme with an empty config.
49 configLoaded_ = parsed.isObject();
50 if (configLoaded_) {
51 config_ = std::move(parsed);
52 }
53 } else {
54 configLoaded_ = parseYAML(content);
55 }
56
57 if (configLoaded_) {
58 applyConfig();
59 }
60
61 return configLoaded_;
62}

129bool NUIConfigLoader::parseYAML(const std::string& content) {
130 // Simple YAML parser - handles basic key: value pairs and nested objects
131 std::istringstream stream(content);
132 std::string line;
133 std::string currentPath;
134 int indentLevel = 0;
135
136 while (std::getline(stream, line)) {
137 // Skip empty lines and comments
138 if (line.empty() || line[0] == '#') continue;
139
140 // Count indentation
141 int currentIndent = 0;
142 for (char c : line) {
143 if (c == ' ') currentIndent++;
144 else break;
145 }
146
147 // Remove leading/trailing whitespace
148 line.erase(0, currentIndent);
149 line = std::regex_replace(line, std::regex("\\s+$"), "");
150
151 if (line.empty()) continue;
152
153 // Check if this is a new object (ends with ':')
154 if (line.back() == ':') {
155 std::string key = line.substr(0, line.length() - 1);
156 key = std::regex_replace(key, std::regex("\\s+$"), "");
157
158 if (currentIndent / 2 > indentLevel) {
159 currentPath += "." + key;
160 indentLevel++;
161 } else if (currentIndent / 2 < indentLevel) {
162 // Go up levels
163 int levelsUp = indentLevel - currentIndent / 2;
164 for (int i = 0; i < levelsUp; i++) {
165 size_t lastDot = currentPath.find_last_of('.');
166 if (lastDot != std::string::npos) {
167 currentPath = currentPath.substr(0, lastDot);
168 }
169 }
170 currentPath += "." + key;
171 indentLevel = currentIndent / 2;
172 } else {
173 size_t lastDot = currentPath.find_last_of('.');
174 if (lastDot != std::string::npos) {
175 currentPath = currentPath.substr(0, lastDot);
176 }
177 currentPath += "." + key;
178 }
179 } else {
180 // This is a value
181 size_t colonPos = line.find(':');
182 if (colonPos != std::string::npos) {
183 std::string key = line.substr(0, colonPos);
184 std::string value = line.substr(colonPos + 1);
185
186 key = std::regex_replace(key, std::regex("^\\s+|\\s+$"), "");
187 value = std::regex_replace(value, std::regex("^\\s+|\\s+$"), "");
188
189 // Set the value in the JSON structure
190 setNestedValue(config_, currentPath + "." + key, value);
191 }
192 }
193 }
194
195 return true;
196}

198void NUIConfigLoader::setNestedValue(Aestra::JSON& json, const std::string& path, const std::string& value) {
199 std::vector<std::string> keys;
200 std::istringstream pathStream(path);
201 std::string key;
202
203 while (std::getline(pathStream, key, '.')) {
204 if (!key.empty()) {
205 keys.push_back(key);
206 }
207 }
208
209 Aestra::JSON* current = &json;
210 for (size_t i = 0; i < keys.size() - 1; i++) {
211 if (current->isObject()) {
212 if (!current->has(keys[i])) {
213 current->set(keys[i], Aestra::JSON::object());
214 }
215 current = &((*current)[keys[i]]);
216 }
217 }
218
219 if (!keys.empty() && current->isObject()) {
220 // Try to parse as number first, then color, then string
221 if (value.find('#') == 0) {
222 // Color value
223 NUIColor color = parseColor(value);
224 current->set(keys.back(), "#" + colorToHex(color));
225 } else if (std::regex_match(value, std::regex("-?\\d*\\.?\\d+"))) {
226 // Number value
227 current->set(keys.back(), std::stod(value));
228 } else {
229 // String value
230 current->set(keys.back(), value);
231 }
232 }
233}

266void NUIConfigLoader::applyLayout(const Aestra::JSON& layout) {
267 auto& themeManager = NUIThemeManager::getInstance();
268 auto& theme = themeManager.getCurrentThemeMutable();
269
270 // Only override a layout value when the JSON supplies a usable one.
271 // parseDimension() returns 0.0f for malformed input (non-numeric, NaN, "px"
272 // typos), so both helpers reject non-finite values and keep the current
273 // default rather than silently clobbering it.
274 //
275 // The zero policy differs by role:
276 // - applyExtent: dimensions that must occupy space (title bar, view toggle,
277 // control sizes). Zero collapses the layout, so require strictly > 0.
278 // - applySpacingDim: margins/padding where zero is a legitimate value
279 // (a flush, no-gap layout), so accept finite >= 0.
280 const auto applyExtent = [&](const char* key, float& target) {
281 if (!layout.has(key)) return;
282 const float value = parseDimension(layout[key]);
283 if (std::isfinite(value) && value > 0.0f) {
284 target = value;
285 }
286 };
287 const auto applySpacingDim = [&](const char* key, float& target) {
288 if (!layout.has(key)) return;
289 const float value = parseDimension(layout[key]);
290 if (std::isfinite(value) && value >= 0.0f) {
291 target = value;
292 }
293 };
294
295 applyExtent("trackHeight", theme.layout.trackHeight);
296 applyExtent("trackControlsWidth", theme.layout.trackControlsWidth);
297 applyExtent("fileBrowserWidth", theme.layout.fileBrowserWidth);
298 applyExtent("transportBarHeight", theme.layout.transportBarHeight);
299 applyExtent("titleBarHeight", theme.layout.titleBarHeight);
300 applyExtent("viewToggleWidth", theme.layout.viewToggleWidth);
301 applyExtent("viewToggleHeight", theme.layout.viewToggleHeight);
302 applyExtent("transportButtonSize", theme.layout.transportButtonSize);
303 applyExtent("controlButtonWidth", theme.layout.controlButtonWidth);
304 applyExtent("controlButtonHeight", theme.layout.controlButtonHeight);
305 applyExtent("gridLineSpacing", theme.layout.gridLineSpacing);
306 applySpacingDim("panelMargin", theme.layout.panelMargin);
307 applySpacingDim("componentPadding", theme.layout.componentPadding);
308}

328NUIColor NUIConfigLoader::parseColor(const std::string& colorStr) {
329 std::string hex = colorStr;
330 if (hex[0] == '#') hex = hex.substr(1);
331
332 if (hex.length() == 6) {
333 uint32_t color = std::stoul(hex, nullptr, 16);
334 return NUIColor::fromHex(color);
335 } else if (hex.length() == 8) {
336 // Handle 8-character hex (AARRGGBB)
337 uint32_t color = std::stoul(hex, nullptr, 16);
338 float a = ((color >> 24) & 0xFF) / 255.0f;
339 float r = ((color >> 16) & 0xFF) / 255.0f;
340 float g = ((color >> 8) & 0xFF) / 255.0f;
341 float b = (color & 0xFF) / 255.0f;
342 return NUIColor(r, g, b, a);
343 }
344
345 return NUIColor::black(); // fallback
346}

348float NUIConfigLoader::parseDimension(const Aestra::JSON& value) {
349 if (value.isNumber()) {
350 return static_cast<float>(value.asNumber());
351 } else if (value.isString()) {
352 std::string str = value.asString();
353 // Remove 'px' suffix if present
354 if (str.size() > 2 && str.substr(str.size() - 2) == "px") {
355 str = str.substr(0, str.size() - 2);
356 }
357 try {
358 return std::stof(str);
359 } catch (const std::invalid_argument&) {
360 return 0.0f;
361 } catch (const std::out_of_range&) {
362 return 0.0f;
363 }
364 }
365 return 0.0f;
366}

368std::string NUIConfigLoader::colorToHex(const NUIColor& color) {
369 uint32_t hex = color.toHex();
370 std::stringstream ss;
371 if (color.a < 0.999f) {
372 // 8-character hex (AARRGGBB)
373 ss << std::hex << std::setw(8) << std::setfill('0') << hex;
374 } else {
375 // 6-character hex (RRGGBB) - exclude alpha channel
376 ss << std::hex << std::setw(6) << std::setfill('0') << (hex & 0xFFFFFF);
377 }
378 return ss.str();
379}
