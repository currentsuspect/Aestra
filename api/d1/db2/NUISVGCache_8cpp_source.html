17NUISVGCache::CacheEntry* NUISVGCache::get(std::string_view svgContent, int w, int h, const NUIColor& tint) {
18 for (auto& pair : cache_) {

34void NUISVGCache::put(const CacheKey& key, std::vector<unsigned char>&& rgba, int w, int h, NUIRenderer* renderer) {
35 // If cache is full, remove oldest entry
36 if (cache_.size() >= maxEntries_) {
37 auto oldestIt = cache_.begin();
38 auto oldestTime = oldestIt->second.lastUsed;
39
40 for (auto it = cache_.begin(); it != cache_.end(); ++it) {
41 if (it->second.lastUsed < oldestTime) {
42 oldestTime = it->second.lastUsed;
43 oldestIt = it;
44 }
45 }
46
47 // Clean up texture if it exists
48 if (oldestIt->second.textureId != 0 && renderer) {
49 renderer->deleteTexture(oldestIt->second.textureId);
50 }
51
52 cache_.erase(oldestIt);
53 }
54
55 // Store new entry with current timestamp
56 CacheEntry entry;
57 entry.rgba = std::move(rgba);
58 entry.width = w;
59 entry.height = h;
60 entry.textureId = 0; // Initialized to 0, renderer will create later
61 entry.lastUsed = std::chrono::steady_clock::now();
62
63 cache_[key] = std::move(entry);
64}

66void NUISVGCache::cleanup(NUIRenderer* renderer, std::chrono::seconds maxAge) {
67 auto now = std::chrono::steady_clock::now();
68
69 for (auto it = cache_.begin(); it != cache_.end(); ) {
70 auto age = std::chrono::duration_cast<std::chrono::seconds>(now - it->second.lastUsed);
71 if (age > maxAge) {
72 // Clean up texture
73 if (it->second.textureId != 0 && renderer) {
74 renderer->deleteTexture(it->second.textureId);
75 }
76 it = cache_.erase(it);
77 } else {
78 ++it;
79 }
80 }
