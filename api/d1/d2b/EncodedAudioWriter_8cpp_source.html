26// Windows loads the encoders only from Aestra's own folder, never by bare name: the default DLL search order includes
27// the current directory, where a planted libmp3lame.dll would run inside Aestra.
28void* loadFromAppFolder(const char* name) {
29 wchar_t exe[MAX_PATH];
30 const DWORD n = GetModuleFileNameW(nullptr, exe, MAX_PATH);
31 if (n == 0 || n >= MAX_PATH)
32 return nullptr;
33 std::wstring path(exe, n);
34 const size_t slash = path.find_last_of(L"\\/");
35 if (slash == std::wstring::npos)
36 return nullptr;
37 path.resize(slash + 1);
38 for (const char* c = name; *c; ++c)
39 path.push_back(static_cast<wchar_t>(*c));
40 return reinterpret_cast<void*>(
41 LoadLibraryExW(path.c_str(), nullptr, LOAD_LIBRARY_SEARCH_DLL_LOAD_DIR | LOAD_LIBRARY_SEARCH_SYSTEM32));
42}
43#endif

45void* openFirst(std::initializer_list<const char*> names, std::string& loaded) {
46 for (const char* name : names) {
47#ifdef _WIN32
48 void* lib = loadFromAppFolder(name);
49#else
50 void* lib = dlopen(name, RTLD_NOW | RTLD_LOCAL);
51#endif
52 if (lib) {
53 loaded = name;
54 return lib;
55 }
56 }
57 return nullptr;
58}

62 return nullptr;

71struct FlacApi {
72 using New = void* (*)();
73 using SetU = int (*)(void*, unsigned);
74 using InitFile = int (*)(void*, const char*, void*, void*);
75 using Process = int (*)(void*, const int32_t*, unsigned);
76 using Finish = int (*)(void*);
77 using Delete = void (*)(void*);
78 New create = nullptr;
79 SetU channels = nullptr, bits = nullptr, rate = nullptr, level = nullptr;
80 InitFile initFile = nullptr;
81 Process process = nullptr;
82 Finish finish = nullptr;
83 Delete destroy = nullptr;
84 bool ok() const { return create && channels && bits && rate && level && initFile && process && finish && destroy; }
85};

178class FlacWriter final : public EncodedAudioWriter {
179public:
180 FlacWriter(const FlacApi& api, void* encoder, uint32_t channels, uint32_t bits)
181 : m_api(api), m_encoder(encoder), m_channels(channels), m_bits(bits), m_rng(0x5eed1234u) {}
182 ~FlacWriter() override {
183 if (m_encoder) {
184 m_api.finish(m_encoder);
185 m_api.destroy(m_encoder);
186 }
187 }
188 bool write(const float* interleaved, uint32_t frames) override {
189 const size_t n = static_cast<size_t>(frames) * m_channels;
190 m_scratch.resize(n);
191 const double full = static_cast<double>((1u << (m_bits - 1)) - 1u);
192 std::uniform_real_distribution<double> lsb(-0.5, 0.5);
193 for (size_t i = 0; i < n; ++i) {
194 double v = std::clamp(static_cast<double>(interleaved[i]), -1.0, 1.0) * full;
195 if (m_bits <= 16)
196 v += lsb(m_rng) + lsb(m_rng); // TPDF dither, as the WAV writer does at 16 bits
197 m_scratch[i] =
198 static_cast<int32_t>(std::clamp(std::lround(v), -static_cast<long>(full) - 1, static_cast<long>(full)));
199 }
200 return m_api.process(m_encoder, m_scratch.data(), frames) != 0;
201 }
202 bool finish() override {
203 if (!m_encoder)
204 return false;
205 const bool ok = m_api.finish(m_encoder) != 0;
206 m_api.destroy(m_encoder);
207 m_encoder = nullptr;
208 return ok;
209 }
210
211private:
212 const FlacApi& m_api;
213 void* m_encoder;
214 uint32_t m_channels, m_bits;
215 std::vector<int32_t> m_scratch;
216 std::mt19937 m_rng;
217};

219class Mp3Writer final : public EncodedAudioWriter {
220public:
221 Mp3Writer(const LameApi& api, void* lame, FILE* file, uint32_t channels)
222 : m_api(api), m_lame(lame), m_file(file), m_channels(channels) {}
223 ~Mp3Writer() override {
224 if (m_lame)
225 m_api.close(m_lame);
226 if (m_file)
227 std::fclose(m_file);
228 }
229 bool write(const float* interleaved, uint32_t frames) override {
230 // LAME's documented worst case: 1.25 * samples + 7200 bytes.
231 m_out.resize(static_cast<size_t>(frames) * 5 / 4 + 7200);
232 const int n =
233 m_api.encode(m_lame, interleaved, static_cast<int>(frames), m_out.data(), static_cast<int>(m_out.size()));
234 return n >= 0 && std::fwrite(m_out.data(), 1, static_cast<size_t>(n), m_file) == static_cast<size_t>(n);
235 }
236 bool finish() override {
237 if (!m_lame)
238 return false;
239 m_out.resize(7200);
240 const int n = m_api.flush(m_lame, m_out.data(), static_cast<int>(m_out.size()));
241 bool ok = n >= 0 && std::fwrite(m_out.data(), 1, static_cast<size_t>(n), m_file) == static_cast<size_t>(n);
242 m_api.close(m_lame);
243 m_lame = nullptr;
244 ok = std::fflush(m_file) == 0 && ok;
245 ok = std::fclose(m_file) == 0 && ok;
246 m_file = nullptr;
247 (void)m_channels;
248 return ok;
249 }
250
251private:
252 const LameApi& m_api;
253 void* m_lame;
254 FILE* m_file;
255 uint32_t m_channels;
256 std::vector<unsigned char> m_out;
257};

259FILE* openForWrite(const std::string& path) {
260#ifdef _WIN32
261 const int n = MultiByteToWideChar(CP_UTF8, 0, path.c_str(), -1, nullptr, 0);
262 if (n <= 0)
263 return nullptr;
264 std::wstring wide(static_cast<size_t>(n), L'\0');
265 MultiByteToWideChar(CP_UTF8, 0, path.c_str(), -1, wide.data(), n);
266 return _wfopen(wide.c_str(), L"wb");
267#else
268 return std::fopen(path.c_str(), "wb");
269#endif
270}

290std::unique_ptr<EncodedAudioWriter> EncodedAudioWriter::open(const std::string& path, const Settings& settings,
291 std::string& error) {
292 const auto& libs = libraries();
293 if (settings.channels == 0 || settings.channels > 2 || settings.sampleRate == 0) {
294 error = "FLAC and MP3 export support mono or stereo at a known sample rate";
295 return nullptr;
296 }
297 if (settings.format == ExportFormat::Flac) {
298 if (!libs.availability.flac) {
299 error = "FLAC export needs libFLAC, which was not found on this computer";
300 return nullptr;
301 }
302 const uint32_t bits = settings.bitsPerSample <= 16 ? 16u : 24u;
303 void* enc = libs.flac.create();
304 if (!enc) {
305 error = "the FLAC encoder could not be created";
306 return nullptr;
307 }
308 libs.flac.channels(enc, settings.channels);
309 libs.flac.bits(enc, bits);
310 libs.flac.rate(enc, settings.sampleRate);
311 libs.flac.level(enc, 5);
312 // FLAC__STREAM_ENCODER_INIT_STATUS_OK is 0.
313 if (libs.flac.initFile(enc, path.c_str(), nullptr, nullptr) != 0) {
314 libs.flac.destroy(enc);
315 error = "the FLAC file could not be created at " + path;
316 return nullptr;
317 }
318 return std::make_unique<FlacWriter>(libs.flac, enc, settings.channels, bits);
319 }
320 if (settings.format == ExportFormat::Mp3) {
321 if (!libs.availability.mp3) {
322 error = "MP3 export needs libmp3lame (LAME), which was not found on this computer";
323 return nullptr;
324 }
325 void* lame = libs.lame.init();
326 if (!lame) {
327 error = "the MP3 encoder could not be created";
328 return nullptr;
329 }
330 libs.lame.inRate(lame, static_cast<int>(settings.sampleRate));
331 libs.lame.outRate(lame, static_cast<int>(settings.sampleRate));
332 libs.lame.channels(lame, static_cast<int>(settings.channels));
333 libs.lame.bitrate(lame, static_cast<int>(settings.mp3Kbps));
334 libs.lame.quality(lame, 2); // high quality
335 if (libs.lame.initParams(lame) < 0) {
336 libs.lame.close(lame);
337 error = "the MP3 encoder rejected these settings (sample rate or bit rate)";
338 return nullptr;
339 }
340 FILE* file = openForWrite(path);
341 if (!file) {
342 libs.lame.close(lame);
343 error = "the MP3 file could not be created at " + path;
344 return nullptr;
345 }
346 return std::make_unique<Mp3Writer>(libs.lame, lame, file, settings.channels);
347 }
348 error = "WAV export does not go through the encoder";
349 return nullptr;
350}
