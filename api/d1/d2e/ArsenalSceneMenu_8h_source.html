64 static bool canDelete(bool usedOnTimeline, size_t sceneCount, std::string* why = nullptr) {
65 if (usedOnTimeline) {
66 if (why) *why = "Delete (it plays in the timeline)";
67 return false;

76 static uint64_t selectAfterDelete(const std::vector<ArsenalSceneStrip::Scene>& scenes, uint64_t deleted) {
77 const auto it = std::find_if(scenes.begin(), scenes.end(), [&](const auto& s) { return s.id == deleted; });
78 if (it == scenes.end()) return 0;
79 if (std::next(it) != scenes.end()) return std::next(it)->id;
80 return it != scenes.begin() ? std::prev(it)->id : 0;
81 }
