30// Zenity 4 can otherwise negotiate its minimum file-chooser size on some
31// Wayland compositors, clipping the file list and action row into a tiny
32// top-left window. These remain hints: the compositor may constrain them on
33// smaller displays.
34constexpr const char* ZENITY_FILE_DIALOG_PRESENTATION = " --modal --width=760 --height=560";

36std::string runDialogCommand(const std::string& command) {
37 std::array<char, 512> buffer{};
38 std::string output;
39 FILE* pipe = popen(command.c_str(), "r");
40 if (!pipe) {
41 return "";
42 }
43
44 while (fgets(buffer.data(), static_cast<int>(buffer.size()), pipe) != nullptr) {
45 output += buffer.data();
46 }
47
48 const int status = pclose(pipe);
49 if (status != 0) {
50 return "";
51 }
52
53 return trimTrailingNewlines(output);
54}

139std::string PlatformUtilsLinux::getAppDataPath(const std::string& appName) const {
140 const char* xdg = std::getenv("XDG_DATA_HOME");
141 std::filesystem::path path;
142 if (xdg && *xdg) {
143 path = std::filesystem::path(xdg);
144 } else {
145 const char* home = std::getenv("HOME");
146 if (home && *home) {
147 path = std::filesystem::path(home) / ".local" / "share";
148 } else {
149 return "/tmp/" + appName;
150 }
151 }
152
153 path /= appName;
154
155 std::error_code ec;
156 if (!std::filesystem::exists(path)) {
157 std::filesystem::create_directories(path, ec);
158 std::filesystem::permissions(path, std::filesystem::perms::owner_all, std::filesystem::perm_options::replace,
159 ec);
160 }
161
162 return path.string();
163}
