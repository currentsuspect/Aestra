285 AestraUI::NUIPoint(area.right(), listY - 2.0f),
286 1.0f,

318 renderer.fillRect(rowRect, theme.getColor("surfaceRaised").withAlpha(0.05f));
319 }
320 renderer.drawLine(
321 AestraUI::NUIPoint(rowRect.x, rowRect.bottom()),
322 AestraUI::NUIPoint(rowRect.right(), rowRect.bottom()),
323 1.0f,
324 theme.getColor("textPrimary").withAlpha(0.08f));
325
326 // Left edge: hover drag handle or active equalizer icon, otherwise track number.
327 const bool showHandle = isHovered && !isCurrent;
328 if (isCurrent) {
329 const float baseX = colNo + 1.0f;
330 const float baseY = y + 12.0f;
331 for (int b = 0; b < 3; ++b) {
332 const float phase = m_animationTime * 8.0f + static_cast<float>(b) * 0.75f;

336 }
337 } else if (showHandle) {
338 const float dotX = colNo + 2.0f;
339 const float dotY = y + 11.0f;
340 const bool handleHovered = (m_queueHandleHoverIndex == static_cast<int>(i));
341 if (handleHovered) {
342 renderer.fillRoundedRect(AestraUI::NUIRect(colNo - 3.0f, y + 8.0f, 16.0f, 20.0f), themeProps.radiusS + 1.0f, theme.getColor("accentPrimary").withAlpha(0.22f));
343 }

1029void AuditionPanel::renderWaveform(AestraUI::NUIRenderer& renderer, const AestraUI::NUIRect& area) {
1030 if (!m_engine) return;
1031
1032 auto source = m_engine->getCurrentSource();
1033 if (!source || !source->isReady()) {
1034 auto& theme = AestraUI::NUIThemeManager::getInstance();
1035 const auto& themeProps = theme.getCurrentTheme();
1036 const float centerX = area.x + area.width * 0.5f;
1037 const float centerY = area.y + area.height * 0.5f;
1038 renderer.drawText("Drop a track to analyze its shape", AestraUI::NUIPoint(centerX - 98.0f, centerY - 12.0f), themeProps.fontSizeXS, theme.getColor("textSecondary").withAlpha(0.9f));
1039 renderer.drawText("Scrub here once a source is loaded", AestraUI::NUIPoint(centerX - 92.0f, centerY + 8.0f), themeProps.fontSizeXS, theme.getColor("textMuted").withAlpha(0.95f));
1040 return;
1041 }
1042
1043 auto buffer = source->getBuffer();
1044 if (!buffer) return;
1045
1046 auto& theme = AestraUI::NUIThemeManager::getInstance();
1047
1048 // Neon Colors
1049 AestraUI::NUIColor playedColorStart = theme.getColor("primary"); // Purple
1050 AestraUI::NUIColor playedColorEnd = theme.getColor("secondary"); // Cyan (or mapped secondary)
1051 AestraUI::NUIColor unplayedColor = theme.getColor("surfaceRaised");
1052 unplayedColor.a = 0.68f;
1053
1054 const float pixelStride = 2.0f;
1055 const float barWidth = 2.0f;
1056 const uint32_t numBars = static_cast<uint32_t>(area.width / pixelStride);
1057 if (numBars == 0) return;
1058
1059 const size_t totalFrames = buffer->numFrames;
1060 const uint32_t channels = buffer->numChannels;
1061 const auto& data = buffer->interleavedData;
1062
1063 float centerY = area.y + area.height / 2.0f;
1064 float halfHeight = area.height / 2.0f;
1065 double progress = m_engine->getPositionNormalized();
1066 float playheadX = area.x + static_cast<float>(progress) * area.width;
1067 const float easedReveal = m_waveformRevealProgress * m_waveformRevealProgress * (3.0f - 2.0f * m_waveformRevealProgress);
1068 const float revealX = area.x + area.width * std::clamp(easedReveal, 0.0f, 1.0f);
1069
1070 for (uint32_t i = 0; i < numBars; ++i) {
1071 float maxAmp = 0.0f;
1072 const size_t startFrame = static_cast<size_t>((static_cast<double>(i) / static_cast<double>(numBars)) * static_cast<double>(totalFrames));
1073 const size_t endFrame = std::max(startFrame + 1,
1074 static_cast<size_t>((static_cast<double>(i + 1) / static_cast<double>(numBars)) * static_cast<double>(totalFrames)));
1075 const size_t spanFrames = std::max<size_t>(1, endFrame - startFrame);
1076 const size_t step = spanFrames > 100 ? (spanFrames / 100) : 1;
1077
1078 for (size_t f = startFrame; f < endFrame; f += step) {
1079 size_t idx = f * channels;
1080 if (idx < data.size()) {
1081 float s = std::abs(data[idx]);
1082 if (s > maxAmp) maxAmp = s;
1083 }
1084 }
1085
1086 if (maxAmp > 0.001f) {
1087 // Apply log scaling or boost for better visuals
1088 float h = maxAmp * halfHeight * 1.5f;
1089 if (h < 2.0f) h = 2.0f;
1090 if (h > halfHeight) h = halfHeight;
1091
1092 float x = area.x + (i * pixelStride);
1093 if (x > revealX) break;
1094
1095 AestraUI::NUIRect barRect(x, centerY - h, barWidth, h * 2.0f);
1096
1097 if (x < playheadX) {
1098 // Vertical Gradient for Played Bars: Purple (Bottom) -> Cyan (Top)
1099 // Since we render from CenterY, we can just use a vibrant vertical gradient
1100 // Top Color
1101 AestraUI::NUIColor colTop = theme.getColor("secondary"); // Cyan
1102 AestraUI::NUIColor colBot = theme.getColor("primary"); // Purple
1103
1104 // Draw in 2 segments for simple gradient simulation
1105 // Top Half
1106 renderer.fillRoundedRect(AestraUI::NUIRect(x, centerY - h, barWidth, h), 1.0f, colTop);
1107 // Bottom Half
1108 renderer.fillRoundedRect(AestraUI::NUIRect(x, centerY, barWidth, h), 1.0f, colBot);
1109 // Or better: manual lerp loop?
1110 // Let's stick to a solid vibrant color that matches the Slider gradient midpoint?
1111 // Or just use the Vertical Logic:
1112 // We'll draw 4 segments vertically
1113 for (int s=0; s<4; ++s) {
1114 float t = s/3.0f;
1115 AestraUI::NUIColor segCol = AestraUI::NUIColor::lerp(colTop, colBot, t);
1116 renderer.fillRoundedRect(
1117 AestraUI::NUIRect(x, (centerY-h) + (s*(h*2.0f)/4.0f), barWidth, (h*2.0f)/4.0f + 0.5f),
1118 0.5f,
1119 segCol
1120 );
1121 }
1122 } else {
1123 // Unplayed: Darker, subtle
1124 renderer.fillRoundedRect(barRect, 1.0f, unplayedColor);
1125 }
1126 }
1127 }
1128
1129 renderer.drawLine(
1130 AestraUI::NUIPoint(area.x, centerY),
1131 AestraUI::NUIPoint(area.right(), centerY),
1132 1.0f,
1133 theme.getColor("border").withAlpha(0.35f)
1134 );
1135}

1145bool AuditionPanel::onMouseEvent(const AestraUI::NUIMouseEvent& event) {
1146 auto bounds = getBounds();
1147 constexpr float queueHeaderH = 24.0f;
1148 constexpr float queueRowPitch = 42.0f;
1149 auto queueRowAt = [&](const AestraUI::NUIPoint& p) -> int {
1150 if (!m_queueArea.contains(p)) return -1;
1151 const float listY = m_queueArea.y + queueHeaderH + 2.0f;
1152 if (p.y < listY) return -1;
1153 const int idx = static_cast<int>((p.y - listY) / queueRowPitch);
1154 if (!m_engine) return -1;
1155 const int size = static_cast<int>(m_engine->getQueue().size());
1156 return (idx >= 0 && idx < size) ? idx : -1;
1157 };
1158 auto isOnQueueHandle = [&](const AestraUI::NUIPoint& p, int row) -> bool {
1159 if (row < 0) return false;
1160 const float y = m_queueArea.y + queueHeaderH + 2.0f + static_cast<float>(row) * queueRowPitch;
1161 return AestraUI::NUIRect(m_queueArea.x + 8.0f, y + 10.0f, 12.0f, 16.0f).contains(p);
1162 };
1163 auto isOnQueueRemove = [&](const AestraUI::NUIPoint& p, int row) -> bool {
1164 if (row < 0) return false;
1165 const float y = m_queueArea.y + queueHeaderH + 2.0f + static_cast<float>(row) * queueRowPitch;
1166 return AestraUI::NUIRect(m_queueArea.right() - 78.0f, y + 9.0f, 18.0f, 18.0f).contains(p);
1167 };
1168 auto queueInsertAt = [&](const AestraUI::NUIPoint& p) -> int {
1169 if (!m_engine) return -1;
1170 const int size = static_cast<int>(m_engine->getQueue().size());
1171 const float listY = m_queueArea.y + queueHeaderH + 2.0f;
1172 const float raw = (p.y - listY) / queueRowPitch;
1173 return std::clamp(static_cast<int>(std::floor(raw + 0.5f)), 0, size);
1174 };
1175 auto scrubNormFromX = [this](float x) -> double {
1176 if (m_waveformArea.width <= 1.0f) return 0.0;
1177 const float relativeX = x - m_waveformArea.x;
1178 const float normPos = relativeX / m_waveformArea.width;
1179 // Avoid hard-locking at exact track end while dragging.
1180 return static_cast<double>(std::clamp(normPos, 0.0f, 0.9995f));
1181 };
1182
1183 // Give the queue context menu first chance to consume input.
1184 if (m_queueContextMenu) {
1185 const bool handled = m_queueContextMenu->onMouseEvent(event);
1186 if (handled) {
1187 return true;
1188 }
1189 if (event.pressed &&
1190 (event.button == AestraUI::NUIMouseButton::Left || event.button == AestraUI::NUIMouseButton::Right)) {
1191 detachContextMenu(m_queueContextMenu);
1192 m_queueContextMenu = nullptr;
1193 setDirty(true);
1194 }
1195 }
1196
1197 // Early exit if mouse is outside our bounds entirely
1198 if (!bounds.contains(event.position) && !m_isScrubbingWaveform) {
1199 m_hoveredQueueIndex = -1;
1200 m_queueRemoveHoverIndex = -1;
1201 m_queueHandleHoverIndex = -1;
1202 m_clearQueueHovered = false;
1203 return NUIComponent::onMouseEvent(event);
1204 }
1205
1206 // 1. Scrubbing (left-click only)
1207 if (event.pressed && event.button == AestraUI::NUIMouseButton::Left) {
1208 if (m_waveformArea.contains(event.position)) {
1209 m_isScrubbingWaveform = true;
1210 if (m_engine) m_engine->seekNormalized(scrubNormFromX(event.position.x));
1211 return true;
1212 }
1213 } else if (event.released) {
1214 m_isScrubbingWaveform = false;
1215 } else if (m_isScrubbingWaveform) {
1216 if (m_engine) m_engine->seekNormalized(scrubNormFromX(event.position.x));
1217 return true;
1218 }
1219
1220 // 2. Queue interactions (clear, remove, reorder, click-to-play)
1221 const bool clearHover = m_clearQueueButtonBounds.contains(event.position);
1222 if (clearHover != m_clearQueueHovered) {
1223 m_clearQueueHovered = clearHover;
1224 setDirty(true);
1225 }
1226
1227 if (event.pressed && event.button == AestraUI::NUIMouseButton::Left && clearHover) {
1228 if (m_engine) m_engine->clearQueue();
1229 m_hoveredQueueIndex = -1;
1230 m_queueRemoveHoverIndex = -1;
1231 m_queueHandleHoverIndex = -1;
1232 m_clearQueueHovered = false;
1233 m_isDraggingQueueItem = false;
1234 m_queueDragFromIndex = -1;
1235 m_queueDragInsertIndex = -1;
1236 setDirty(true);
1237 return true;
1238 }
1239
1240 if (!event.pressed && !event.released && m_isDraggingQueueItem) {
1241 m_queueDragInsertIndex = queueInsertAt(event.position);
1242 setDirty(true);
1243 return true;
1244 }
1245
1246 if (m_queueArea.contains(event.position)) {
1247 const int prevHover = m_hoveredQueueIndex;
1248 const int prevHandleHover = m_queueHandleHoverIndex;
1249 const int prevRemoveHover = m_queueRemoveHoverIndex;
1250 const int index = queueRowAt(event.position);
1251 m_hoveredQueueIndex = index;
1252 m_queueHandleHoverIndex = (index >= 0 && isOnQueueHandle(event.position, index)) ? index : -1;
1253 m_queueRemoveHoverIndex = (index >= 0 && isOnQueueRemove(event.position, index)) ? index : -1;
1254 if (prevHover != m_hoveredQueueIndex || prevHandleHover != m_queueHandleHoverIndex || prevRemoveHover != m_queueRemoveHoverIndex) {
1255 setDirty(true);
1256 }
1257
1258 if (event.pressed && event.button == AestraUI::NUIMouseButton::Left && index >= 0) {
1259 if (m_queueRemoveHoverIndex == index) {
1260 if (m_engine) m_engine->removeFromQueue(static_cast<size_t>(index));
1261 m_hoveredQueueIndex = -1;
1262 m_queueRemoveHoverIndex = -1;
1263 m_queueHandleHoverIndex = -1;
1264 setDirty(true);
1265 return true;
1266 }
1267 if (m_queueHandleHoverIndex == index) {
1268 m_isDraggingQueueItem = true;
1269 m_queueDragFromIndex = index;
1270 m_queueDragInsertIndex = index;
1271 setDirty(true);
1272 return true;
1273 }
1274 if (m_engine) {
1275 m_engine->jumpToTrack(static_cast<size_t>(index));
1276 m_engine->play();
1277 Log::info("[AuditionPanel] Clicked queue item: " + std::to_string(index));
1278 return true;
1279 }
1280 }
1281
1282 if (event.pressed && event.button == AestraUI::NUIMouseButton::Right && index >= 0 && m_engine) {
1283 const auto& queue = m_engine->getQueue();
1284 const int queueSize = static_cast<int>(queue.size());
1285 if (queueSize <= 0) return true;
1286
1287 detachContextMenu(m_queueContextMenu);
1288 m_queueContextMenu = std::make_shared<AestraUI::NUIContextMenu>();
1289 auto menu = m_queueContextMenu;
1290

1292 const bool canMoveDown = index < queueSize - 1;
1293 const bool canMoveTop = index > 0;
1294 const bool canMoveBottom = index < queueSize - 1;

1308 if (!m_engine || index <= 0) return;
1309 m_engine->moveQueueItem(static_cast<size_t>(index), 0);
1310 });
1311 menu->addItem(moveTopItem);
1312

1314 moveUpItem->setEnabled(canMoveUp);
1315 moveUpItem->setOnClick([this, index]() {
1316 if (!m_engine || index <= 0) return;

1340 if (!m_engine) return;
1341 m_engine->removeFromQueue(static_cast<size_t>(index));

1343 menu->addItem(deleteItem);
1344
1345 menu->setOnHide([this]() {
1346 detachContextMenu(m_queueContextMenu);
1347 m_queueContextMenu = nullptr;
1348 setDirty(true);
1349 });
1350 attachAndShowContextMenu(this, menu, event.position);
1351 setDirty(true);
1352 return true;
1353 }
1354
1355 if (event.released && event.button == AestraUI::NUIMouseButton::Left && m_isDraggingQueueItem) {

1358 if (m_engine && from >= 0) {
1359 int destination = insert;

1361 const int size = static_cast<int>(m_engine->getQueue().size());

1363 if (destination != from) {
1364 m_engine->moveQueueItem(static_cast<size_t>(from), static_cast<size_t>(destination));
1365 }
1366 }
1367 m_isDraggingQueueItem = false;
1368 m_queueDragFromIndex = -1;
1369 m_queueDragInsertIndex = -1;
1370 setDirty(true);
1371 return true;
1372 }
1373 } else {
1374 if (m_hoveredQueueIndex != -1 || m_queueRemoveHoverIndex != -1 || m_queueHandleHoverIndex != -1) {
1375 m_hoveredQueueIndex = -1;
1376 m_queueRemoveHoverIndex = -1;
1377 m_queueHandleHoverIndex = -1;
1378 setDirty(true);
1379 }
1380 if (event.released && event.button == AestraUI::NUIMouseButton::Left && m_isDraggingQueueItem) {
1381 m_isDraggingQueueItem = false;
1382 m_queueDragFromIndex = -1;
1383 m_queueDragInsertIndex = -1;
1384 setDirty(true);
1385 return true;
1386 }
1387 }
1388
1389 if (event.pressed && event.button == AestraUI::NUIMouseButton::Right) {
1390 return true;
1391 }
1392
1393 return NUIComponent::onMouseEvent(event);
1394}

1412bool AuditionPanel::isInDropZone(float x, float y) const {
1413 auto bounds = getBounds();
1414 // The queue area starts at bounds.height - queueHeight - padding.
1415 // With queueHeight = 0.30f * bounds.height and padding = 20.0f,
1416 // this is roughly bounds.height * 0.70f - 20.0f.
1417 // So, a simple check for y >= bounds.height * 0.70f is a good approximation for the drop zone.
1418 return (x >= 0.0f && x <= bounds.width &&
1419 y >= bounds.height * 0.70f && y <= bounds.height);
