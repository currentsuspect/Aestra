10void ChannelSlotMap::rebuild(const std::vector<std::unique_ptr<MixerChannel>>& channels) {
11 m_slotToId.fill(INVALID_SLOT);
12 m_channelCount = 0;
13
14 uint32_t slot = 0;
15 for (const auto& channel : channels) {
16 if (channel && slot < MAX_CHANNEL_SLOTS) {
17 uint32_t channelId = channel->getChannelId();
18 m_slotToId[slot] = channelId;
19 ++slot;
20 }
21 }
22 m_channelCount = slot;
23}

25uint32_t ChannelSlotMap::getSlotIndex(uint32_t channelId) const {
26 // Linear scan of at most 127 entries. Deterministic, cache-friendly,
27 // no hash computation. Called from RT path but n is tiny.
28 const uint32_t count = m_channelCount;
29 for (uint32_t i = 0; i < count; ++i) {
30 if (m_slotToId[i] == channelId)
31 return i;
32 }
33 return INVALID_SLOT;
34}
