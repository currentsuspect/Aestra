27NUIColor colorFromUnitValue(uint32_t value) {
28 constexpr float scale = 1.0f / 255.0f;
29 return NUIColor(static_cast<float>((value >> 16) & 0xff) * scale,
30 static_cast<float>((value >> 8) & 0xff) * scale,
31 static_cast<float>(value & 0xff) * scale,
32 1.0f);
33}

68void UnitRow::updateState() {
69 auto* unit = m_manager.getUnit(m_unitId);
70 if (unit) {
71 m_name = unit->name;
72 m_color = unit->color;
73 m_group = unit->group;
74 m_type = unit->type;
75 m_isEnabled = unit->isEnabled;
76 m_isArmed = unit->isArmed;
77 m_isMuted = unit->isMuted;
78 m_isSolo = unit->isSolo;
79 m_audioClip = unit->audioClipPath.empty() ? "" :
80 unit->audioClipPath.substr(unit->audioClipPath.find_last_of("/\\") + 1);
81 m_pluginId = unit->pluginId;
82 m_mixerChannel = unit->targetMixerRoute;
83 m_audioDurationSeconds = unit->audioDurationSeconds;
84 m_audioPreviewWaveform = unit->audioPreviewWaveform;
85
86 switch (m_type) {
87 case Aestra::Audio::UnitType::Sampler:
88 m_groupLabel = "Sampler";
89 break;
90 case Aestra::Audio::UnitType::PitchedSampler:
91 m_groupLabel = "808";
92 break;
93 case Aestra::Audio::UnitType::Instrument:
94 m_groupLabel = "MIDI";
95 break;
96 case Aestra::Audio::UnitType::Audio:
97 m_groupLabel = "Audio";
98 break;
99 default:
100 m_groupLabel = "Sampler";
101 break;
102 }
103
104 if (!m_pluginId.empty()) {
105 m_sourceSummary = "Plugin";
106 } else if (!m_audioClip.empty()) {
107 m_sourceSummary = "Sample";
108 } else {
109 m_sourceSummary = "Empty";
110 }
111 }
112
113 const auto displayType = shouldUseNoteRoll() ? Aestra::Audio::UnitType::Instrument : m_type;
114
115 // Ensure name label exists and stays in sync
116 if (!m_nameLabel) {
117 m_nameLabel = std::make_shared<UnitNameLabel>(m_name, displayType);
118 addChild(m_nameLabel);
119 layoutNameLabel();
120
121 m_nameLabel->m_onOpenEditor = [this]() {
122 if (m_onEditUnit) m_onEditUnit(m_unitId);
123 };
124 m_nameLabel->m_onRename = [this](const std::string& name) {
125 if (m_onRenameUnit) m_onRenameUnit(m_unitId, name);
126 };
127 } else {
128 m_nameLabel->setUnitName(m_name);
129 m_nameLabel->setUnitType(displayType);
130 }
131
132 invalidateVisuals();
133}

135void UnitRow::onRender(NUIRenderer& renderer) {
136 if (m_nameLabel) {
137 m_nameLabel->setUnitType(shouldUseNoteRoll() ? Aestra::Audio::UnitType::Instrument : m_type);
138 }
139
140 // Invalidate cache if flagged (e.g. state change or hover)
141 if (m_needsCacheUpdate) {
142 renderer.invalidateCache(reinterpret_cast<uint64_t>(this));
143 m_needsCacheUpdate = false;
144 }
145
146 auto bounds = getBounds();
147
148 // Render live for now. The widget cache path is currently producing
149 // duplicate/local-space ghosts for Arsenal rows on Linux.
150 drawContent(renderer);
151
152 // Render dynamic children (Input Widget, Context Menu) on top
153 // Push UnitRow position so children (relative coords) draw correctly
154 renderer.pushTransform(bounds.x, bounds.y);
155 renderChildren(renderer);
156 renderer.popTransform();
157}

159void UnitRow::drawContent(NUIRenderer& renderer) {
160 auto bounds = getBounds();
161 auto& theme = NUIThemeManager::getInstance();
162
163 NUIRect cardBounds = bounds;
164 cardBounds.height = 56.0f;
165
166 const auto& themeProps = theme.getCurrentTheme();
167 const float radius = std::max(0.0f, themeProps.radiusM - 1.0f);
168 const NUIColor unitAccent = colorFromUnitValue(m_color);
169 NUIColor cardBg = theme.getColor("backgroundSecondary");
170 NUIColor border = theme.getColor("border").withAlpha(0.08f);
171 if (m_isDropHighlighted) {
172 cardBg = theme.getColor("accentPrimary").withAlpha(0.18f);
173 border = theme.getColor("accentPrimary").withAlpha(0.95f);
174 } else if (m_isSelected) {
175 cardBg = unitAccent.withAlpha(0.10f);
176 border = unitAccent.withAlpha(0.72f);
177 } else if (m_isHovered) {
178 cardBg = theme.getColor("surfaceRaised");
179 border = theme.getColor("accentPrimary").withAlpha(0.22f);
180 }
181
182 renderer.fillRoundedRect(cardBounds, radius, cardBg);
183 renderer.strokeRoundedRect(cardBounds, radius, m_isDropHighlighted ? 2.0f : 1.0f, border);
184 renderer.fillRoundedRect({cardBounds.x + 1.0f, cardBounds.y + 8.0f, 4.0f, cardBounds.height - 16.0f},
185 2.0f, unitAccent.withAlpha(m_isEnabled ? 0.95f : 0.35f));
186
187 m_controlWidth = std::clamp(cardBounds.width * 0.38f, 220.0f, 312.0f);
188
189 NUIRect dragRect(cardBounds.x + 6.0f, cardBounds.y + 6.0f, 32.0f, cardBounds.height - 12.0f);
190 drawDragHandle(renderer, dragRect);
191
192 NUIRect controlRect(cardBounds.x + 42.0f, cardBounds.y,
193 std::max(0.0f, m_controlWidth - 48.0f), cardBounds.height);
194 drawControlBlock(renderer, controlRect);
195
196 float separatorX = cardBounds.x + m_controlWidth;
197 renderer.drawLine(NUIPoint(separatorX, cardBounds.y + 8.0f),
198 NUIPoint(separatorX, cardBounds.y + cardBounds.height - 8.0f),
199 1.0f,
200 theme.getColor("borderSubtle").withAlpha(0.55f));
201
202 NUIRect contextRect(separatorX + 8.0f, cardBounds.y + 6.0f, cardBounds.right() - separatorX - 14.0f, cardBounds.height - 12.0f);
203 renderer.setClipRect(cardBounds);
204 drawContextBlock(renderer, contextRect);
205 renderer.clearClipRect();
206}

357void UnitRow::drawContextBlock(NUIRenderer& renderer, const NUIRect& bounds) {
358 auto& theme = NUIThemeManager::getInstance();
359 struct NoteSpan {
360 int startStep{0};
361 int endStep{0};
362 double startBeat{0.0};
363 double endBeat{0.0};
364 int pitch{48};
365 float velocity{1.0f};
366 };
367
368 const auto& themeProps = theme.getCurrentTheme();
369 const NUIRect contentCard(bounds.x, bounds.y, std::max(0.0f, bounds.width), std::max(0.0f, bounds.height));
370 renderer.fillRoundedRect(contentCard, themeProps.radiusS + 2.0f, theme.getColor("backgroundPrimary"));
371
372 const float lanePadding = 6.0f;
373 const NUIRect timelineStrip(contentCard.x + lanePadding,
374 contentCard.y + lanePadding,
375 std::max(0.0f, contentCard.width - lanePadding * 2.0f),
376 std::max(0.0f, contentCard.height - lanePadding * 2.0f));
377
378 // === Step Grid Layout ===
379 float availWidth = timelineStrip.width;
380 float stepWidth = std::max(availWidth / static_cast<float>(m_stepCount), 20.0f);
381 float totalWidth = stepWidth * m_stepCount;
382
383 // Clamp scroll
384 float maxScroll = std::max(0.0f, totalWidth - availWidth);
385 m_scrollX = std::max(0.0f, std::min(m_scrollX, maxScroll));
386
387 // === Fetch Active Steps ===
388 std::vector<int> activeSteps;
389 std::vector<NoteSpan> noteSpans;
390 std::vector<int> stepPitch(m_stepCount, -1);
391 std::vector<float> stepLength(m_stepCount, 0.0f);
392 bool unitHasContent = !m_pluginId.empty() || !m_audioClip.empty();
393 bool noteRollMode = usesPianoRollForType(m_type);
394 double visibleLengthBeats = static_cast<double>(m_stepCount) * 0.25;
395 if (m_patternId.isValid()) {
396 auto* pattern = m_trackManager->getPatternManager().getPattern(m_patternId);
397 if (pattern && pattern->isMidi()) {
398 visibleLengthBeats = std::max(0.25, pattern->lengthBeats);
399 auto& midi = std::get<Aestra::Audio::MidiPayload>(pattern->payload);
400 for (const auto& note : midi.notes) {
401 if (note.unitId != m_unitId && note.unitId != 0) continue;
402 unitHasContent = true;
403 if (m_type == Aestra::Audio::UnitType::Sampler &&
404 (note.pitch != 60 || std::abs(note.durationBeats - 0.25) > 0.01)) {
405 noteRollMode = true;
406 }
407 const int startStep = std::clamp(static_cast<int>(std::floor(note.startBeat / 0.25 + 0.0001)), 0, m_stepCount - 1);
408 const double endBeat = std::max(note.startBeat + 0.25, note.startBeat + note.durationBeats);
409 const int endExclusive =
410 std::clamp(static_cast<int>(std::ceil(endBeat / 0.25 - 0.0001)), startStep + 1, m_stepCount);
411
412 for (int step = startStep; step < endExclusive; ++step) {
413 activeSteps.push_back(step);
414 }
415 noteSpans.push_back(NoteSpan{startStep, endExclusive, note.startBeat, endBeat, note.pitch,
416 note.velocity});
417 stepPitch[startStep] = note.pitch;
418 stepLength[startStep] = static_cast<float>(note.durationBeats);
419 }
420 }
421 }
422 std::sort(activeSteps.begin(), activeSteps.end());
423 activeSteps.erase(std::unique(activeSteps.begin(), activeSteps.end()), activeSteps.end());
424
425 if (m_type == Aestra::Audio::UnitType::Sampler && !noteRollMode) {
426 const NUIColor inactiveFill = theme.getColor("surfaceTertiary");
427 const NUIColor inactiveStroke = theme.getColor("border").withAlpha(0.10f);
428 const NUIColor activeFill = theme.getColor("primary");
429 const float cellGap = 3.0f;
430 const float cellRadius = 3.0f;
431 const float cellHeight = std::max(10.0f, timelineStrip.height - 6.0f);
432 const float cellY = timelineStrip.y + (timelineStrip.height - cellHeight) * 0.5f;
433
434 for (int step = 0; step < m_stepCount; ++step) {
435 const float cellX = timelineStrip.x + (step * stepWidth) - m_scrollX + (cellGap * 0.5f);
436 const float cellWidth = std::max(8.0f, stepWidth - cellGap);
437 if (cellX + cellWidth < timelineStrip.x || cellX > timelineStrip.right()) {
438 continue;
439 }
440
441 const bool isActive = std::binary_search(activeSteps.begin(), activeSteps.end(), step);
442 NUIRect cellRect(cellX, cellY, cellWidth, cellHeight);
443 renderer.fillRoundedRect(cellRect, cellRadius, isActive ? activeFill : inactiveFill);
444 renderer.strokeRoundedRect(cellRect, cellRadius, 1.0f, inactiveStroke);
445
446 if ((step + 1) % 4 == 0 && step + 1 < m_stepCount) {
447 const float markerX = cellRect.right() + (cellGap * 0.5f);
448 renderer.drawLine(NUIPoint(markerX, timelineStrip.y + 2.0f),
449 NUIPoint(markerX, timelineStrip.bottom() - 2.0f),
450 1.0f,
451 theme.getColor("border").withAlpha(0.12f));
452 }
453 }
454 } else if (m_type == Aestra::Audio::UnitType::PitchedSampler) {
455 const NUIColor inactiveFill = theme.getColor("surfaceTertiary");
456 const NUIColor inactiveStroke = theme.getColor("border").withAlpha(0.10f);
457 const NUIColor activeFill = theme.getColor("primary");
458 const NUIColor pitchText = theme.getColor("textSecondary").withAlpha(0.72f);
459 const float cellGap = 3.0f;
460 const float topHeight = std::max(8.0f, timelineStrip.height * 0.58f);
461 const float pitchLaneHeight = std::max(8.0f, timelineStrip.height * 0.24f);
462 const float lengthLaneHeight = std::max(3.0f, timelineStrip.height * 0.10f);
463 const float topY = timelineStrip.y;
464 const float pitchY = topY + topHeight + 2.0f;
465 const float lengthY = timelineStrip.bottom() - lengthLaneHeight - 1.0f;
466
467 std::vector<NUIPoint> slidePoints;
468 for (int step = 0; step < m_stepCount; ++step) {
469 const float cellX = timelineStrip.x + (step * stepWidth) - m_scrollX + (cellGap * 0.5f);
470 const float cellWidth = std::max(8.0f, stepWidth - cellGap);
471 if (cellX + cellWidth < timelineStrip.x || cellX > timelineStrip.right()) {
472 continue;
473 }
474 const bool isActive = std::binary_search(activeSteps.begin(), activeSteps.end(), step);
475 NUIRect cellRect(cellX, topY, cellWidth, topHeight);
476 renderer.fillRoundedRect(cellRect, 3.0f, isActive ? activeFill : inactiveFill);
477 renderer.strokeRoundedRect(cellRect, 3.0f, 1.0f, inactiveStroke);
478
479 const NUIRect pitchRect(cellX, pitchY, cellWidth, pitchLaneHeight);
480 if (stepPitch[step] >= 0) {
481 const int display = stepPitch[step] - 36;
482 renderer.drawTextCentered((display >= 0 ? "+" : "") + std::to_string(display), pitchRect, 8.0f, pitchText);
483 slidePoints.push_back(NUIPoint(pitchRect.x + pitchRect.width * 0.5f, pitchRect.y + pitchRect.height * 0.5f));
484 } else {
485 renderer.drawTextCentered("·", pitchRect, 8.0f, pitchText.withAlpha(0.35f));
486 }
487
488 const NUIRect lengthBg(cellX, lengthY, cellWidth, lengthLaneHeight);
489 renderer.fillRoundedRect(lengthBg, 1.5f, inactiveStroke.withAlpha(0.35f));
490 if (stepLength[step] > 0.0f) {
491 const float ratio = std::clamp(stepLength[step] / 1.0f, 0.15f, 1.0f);
492 NUIRect lengthFill(cellX, lengthY, std::max(2.0f, cellWidth * ratio), lengthLaneHeight);
493 renderer.fillRoundedRect(lengthFill, 1.5f, activeFill.lightened(0.08f));
494 }
495 }
496
497 for (size_t i = 1; i < slidePoints.size(); ++i) {
498 renderer.drawLine(slidePoints[i - 1], slidePoints[i], 1.5f, activeFill.withAlpha(0.6f));
499 }
500 } else if (noteRollMode) {
501 for (int lane = 0; lane < 5; ++lane) {
502 const float y = timelineStrip.y + ((lane + 1) * timelineStrip.height / 5.0f);
503 renderer.drawLine(NUIPoint(timelineStrip.x, y),
504 NUIPoint(timelineStrip.right(), y),
505 1.0f,
506 theme.getColor("border").withAlpha(0.07f));
507 }
508
509 for (double beat = 4.0; beat < visibleLengthBeats; beat += 4.0) {
510 const float x = timelineStrip.x +
511 static_cast<float>(beat / visibleLengthBeats) * totalWidth - m_scrollX;
512 renderer.drawLine(NUIPoint(x, timelineStrip.y), NUIPoint(x, timelineStrip.bottom()), 1.0f,
513 theme.getColor("border").withAlpha(0.10f));
514 }
515
516 int minPitch = noteSpans.empty() ? 60 : noteSpans[0].pitch;
517 int maxPitch = noteSpans.empty() ? 72 : noteSpans[0].pitch;
518 for (const auto& span : noteSpans) {
519 minPitch = std::min(minPitch, span.pitch);
520 maxPitch = std::max(maxPitch, span.pitch);
521 }
522 minPitch = std::max(0, minPitch - 2);
523 maxPitch = std::min(127, maxPitch + 2);
524 const int pitchRange = std::max(12, maxPitch - minPitch);
525 const NUIColor noteFill = theme.getColor("primary").withAlpha(0.88f);
526 const NUIColor noteStroke = theme.getColor("border").withAlpha(0.10f);
527 const float noteHeight = 4.0f;
528
529 for (const auto& span : noteSpans) {
530 const float startX = timelineStrip.x +
531 static_cast<float>(span.startBeat / visibleLengthBeats) * totalWidth + 1.5f -
532 m_scrollX;
533 const float endX = timelineStrip.x +
534 static_cast<float>(span.endBeat / visibleLengthBeats) * totalWidth - 1.5f -
535 m_scrollX;
536 const float width = std::max(6.0f, endX - startX);
537 if (startX + width < timelineStrip.x || startX > timelineStrip.right()) {
538 continue;
539 }
540
541 float normalizedPitch = static_cast<float>(span.pitch - minPitch) / static_cast<float>(pitchRange);
542 float noteY = timelineStrip.bottom() - 2.0f - (normalizedPitch * (timelineStrip.height - noteHeight - 4.0f)) - noteHeight;
543 NUIRect noteRect(startX, noteY, width, noteHeight);
544 renderer.fillRoundedRect(noteRect, 2.0f, noteFill);
545 renderer.strokeRoundedRect(noteRect, 2.0f, 1.0f, noteStroke);
546 }
547 } else if (m_type == Aestra::Audio::UnitType::Audio) {
548 if (!m_audioPreviewWaveform.empty()) {
549 const float midY = timelineStrip.y + timelineStrip.height * 0.5f;
550 const float ampScale = std::max(4.0f, timelineStrip.height * 0.42f);
551 const float binWidth = timelineStrip.width / static_cast<float>(m_audioPreviewWaveform.size());
552 const NUIColor waveColor = theme.getColor("primary").withAlpha(0.9f);
553
554 for (size_t i = 0; i < m_audioPreviewWaveform.size(); ++i) {
555 const float x = timelineStrip.x + static_cast<float>(i) * binWidth;
556 const float amp = std::clamp(m_audioPreviewWaveform[i], 0.02f, 1.0f) * ampScale;
557 renderer.drawLine(NUIPoint(x, midY - amp), NUIPoint(x, midY + amp), std::max(1.0f, binWidth * 0.6f), waveColor);
558 }
559 }
560 }
561
562 if (!unitHasContent && m_isHovered) {
563 std::string emptyLabel = "Drop a sample or open Step Editor";
564 if (m_type == Aestra::Audio::UnitType::PitchedSampler) {
565 emptyLabel = "Drop a sample - pitch and slide per step";
566 } else if (m_type == Aestra::Audio::UnitType::Instrument) {
567 emptyLabel = "Open Piano Roll to add notes";
568 } else if (m_type == Aestra::Audio::UnitType::Audio) {
569 emptyLabel = "Drop an audio clip";
570 }
571 renderer.drawTextCentered("+", NUIRect(timelineStrip.x, timelineStrip.y + 2.0f, timelineStrip.width, 10.0f), 12.0f, theme.getColor("textSecondary").withAlpha(0.72f));
572 renderer.drawTextCentered(emptyLabel,
573 NUIRect(timelineStrip.x, timelineStrip.y + 10.0f, timelineStrip.width, timelineStrip.height - 10.0f),
574 9.0f,
575 theme.getColor("textSecondary").withAlpha(0.72f));
576 }

579bool UnitRow::shouldUseNoteRoll() const {
580 if (usesPianoRollForType(m_type)) {
581 return true;
582 }
583 if (m_type != Aestra::Audio::UnitType::Sampler || !m_trackManager || !m_patternId.isValid()) {
584 return false;
585 }
586
587 const auto* pattern = m_trackManager->getPatternManager().getPattern(m_patternId);
588 if (!pattern || !pattern->isMidi()) {
589 return false;

591
592 const auto& midi = std::get<Aestra::Audio::MidiPayload>(pattern->payload);
593 return std::any_of(midi.notes.begin(), midi.notes.end(), [this](const Aestra::Audio::MidiNote& note) {
594 const bool belongsToUnit = note.unitId == m_unitId || note.unitId == 0;
595 return belongsToUnit &&
596 (note.pitch != 60 || std::abs(note.durationBeats - 0.25) > 0.01);
597 });

600bool UnitRow::onMouseEvent(const NUIMouseEvent& event) {
601 // Forward to UnitNameLabel first (same local-coordinate transform as old m_nameInput)
602 if (m_nameLabel) {
603 auto bounds = getBounds();
604 NUIMouseEvent localEvent = event;
605 localEvent.position.x -= bounds.x;
606 localEvent.position.y -= bounds.y;
607 if (m_nameLabel->onMouseEvent(localEvent)) {
608 return true;
609 }
610 }
611
612 auto bounds = getBounds();
613 const NUIRect localDragRect(6.0f, 6.0f, 32.0f, 44.0f);
614 m_controlWidth = std::clamp(bounds.width * 0.38f, 220.0f, 312.0f);
615 const NUIRect localControlRect(42.0f, 0.0f, std::max(0.0f, m_controlWidth - 48.0f), 56.0f);
616 const float separatorX = m_controlWidth;
617 const NUIRect localContextRect(separatorX + 8.0f, 6.0f, std::max(0.0f, bounds.width - (separatorX + 14.0f)), 44.0f);
618 const NUIPoint localPoint(event.position.x - bounds.x, event.position.y - bounds.y);
619
620 auto resolveGridStep = [this](const NUIPoint& position, const NUIRect& gridBounds) -> int {
621 float relativeX = position.x - gridBounds.x;
622 float availWidth = gridBounds.width;
623 float stepWidth = std::max(availWidth / static_cast<float>(m_stepCount), 20.0f);
624 float contentX = relativeX + m_scrollX;
625 return static_cast<int>(contentX / stepWidth);
626 };
627
628 auto updateStepEditSpan = [this](int endExclusive) {
629 if (!m_patternId.isValid() || m_stepEditStart < 0 || endExclusive <= m_stepEditStart) {
630 return;
631 }
632
633 m_stepEditEndExclusive = std::max(m_stepEditStart + 1, std::min(m_stepCount, endExclusive));
634 const double targetStart = static_cast<double>(m_stepEditStart) * 0.25;
635 const double targetDuration = static_cast<double>(m_stepEditEndExclusive - m_stepEditStart) * 0.25;
636
637 auto& pm = m_trackManager->getPatternManager();
638 pm.applyPatch(m_patternId, [this, targetStart, targetDuration](Aestra::Audio::PatternSource& p) {
639 if (!p.isMidi()) return;
640 auto& midi = std::get<Aestra::Audio::MidiPayload>(p.payload);
641 auto it = std::find_if(midi.notes.begin(), midi.notes.end(),
642 [this, targetStart](const Aestra::Audio::MidiNote& n) {
643 return n.unitId == m_unitId &&
644 n.pitch == 60 &&
645 std::abs(n.startBeat - targetStart) < 0.01;
646 });
647 if (it != midi.notes.end()) {
648 it->durationBeats = targetDuration;
649 }
650 });
651 };
652
653 auto finishStepEdit = [this]() {
654 if (!m_isStepEditing) {
655 return;
656 }
657 m_isStepEditing = false;
658 if (m_onPatternEdited && m_patternId.isValid()) {
659 m_onPatternEdited(m_patternId);
660 }
661 invalidateVisuals();
662 };
663
664 // === Scroll Handling ===
665 if (std::abs(event.wheelDelta) > 0.0f) {
666 // Only scroll if hovering over context area (right of controls)
667 if (bounds.contains(event.position) && (event.position.x - bounds.x > m_controlWidth)) {
668 float delta = event.wheelDelta;
669 if (delta > 10.0f) delta = 1.0f;
670 if (delta < -10.0f) delta = -1.0f;
671
672 // If notes exist, scroll pitch viewport instead of horizontal
673 bool hasNotes = false;
674 if (m_patternId.isValid()) {
675 auto* pattern = m_trackManager->getPatternManager().getPattern(m_patternId);
676 if (pattern && pattern->isMidi()) {
677 auto& midi = std::get<Aestra::Audio::MidiPayload>(pattern->payload);
678 for (const auto& note : midi.notes) {
679 if (note.unitId == m_unitId || note.unitId == 0) {
680 hasNotes = true;
681 break;
682 }
683 }
684 }
685 }
686
687 if (hasNotes) {
688 // Pitch viewport scroll (up/down = see higher/lower pitches)
689 m_minimapPitchOffset -= delta * 4.0f; // 4 semitones per tick
690 m_minimapPitchOffset = std::clamp(m_minimapPitchOffset, -60.0f, 60.0f);
691 } else {
692 // Horizontal scroll for step grid
693 m_scrollX -= delta * 40.0f;
694 }
695 invalidateVisuals();
696 return true;
697 }
698 }
699
700 // === Hover Detection ===
701 bool wasHovered = m_isHovered;
702 m_isHovered = bounds.contains(event.position);
703
704 // === Step Hover Detection (with Scroll) ===
705 int oldHoveredStep = m_hoveredStep;
706 m_hoveredStep = -1;
707
708 if (m_isHovered && localContextRect.contains(localPoint)) {
709 const int stepIndex = resolveGridStep(localPoint, localContextRect);
710 if (stepIndex >= 0 && stepIndex < m_stepCount) {
711 m_hoveredStep = stepIndex;
712 }
713 }
714
715 if (wasHovered != m_isHovered || oldHoveredStep != m_hoveredStep) {
716 invalidateVisuals();
717 }
718
719 if (m_isStepEditing) {
720 if (event.released || event.type == NUIMouseEventType::Up) {
721 finishStepEdit();
722 return true;
723 }
724
725 const bool isPointerMoveWhileEditing =
726 !event.pressed && !event.released &&
727 (event.type == NUIMouseEventType::Drag ||
728 event.type == NUIMouseEventType::Move ||
729 event.button == NUIMouseButton::None);
730
731 if (isPointerMoveWhileEditing) {
732 int stepIndex = resolveGridStep(localPoint, localContextRect);
733 stepIndex = std::max(m_stepEditStart, std::min(m_stepCount - 1, stepIndex));
734 const int endExclusive = stepIndex + 1;
735 if (endExclusive != m_stepEditEndExclusive) {
736 updateStepEditSpan(endExclusive);
737 invalidateVisuals();
738 }
739 return true;
740 }
741 }
742
743 // === Click Handling ===
744 if (event.pressed) {
745 // Ensure focus
746 if (bounds.contains(event.position)) {
747 setFocused(true);
748 }
749
750 if (event.button == NUIMouseButton::Left) {
751 // Note: NUITextInput handles its own focus loss/commit.
752 // We just handle clicks on other elements.
753
754 if (bounds.contains(event.position)) {
755 if (localDragRect.contains(localPoint)) {
756 m_isDragging = true;
757 m_dragStartPos = event.position;
758 if (m_onDragStart) m_onDragStart(m_unitId);
759 invalidateVisuals();
760 return true;
761 }
762 else if (localControlRect.contains(localPoint)) {
763 handleControlClick(event, localControlRect);
764 return true;
765 }
766 else if (localContextRect.contains(localPoint)) {
767 handleContextClick(event, localContextRect);
768 return true;
769 }
770 }
771 }
772
773 // === Right-click (row body, outside name label which is handled by UnitNameLabel child) ===
774 if (event.button == NUIMouseButton::Right) {
775 if (bounds.contains(event.position)) {
776 showRowContextMenu(event.position);
777 return true;
778 }
779 }
780 }
781
782 // === Release drag ===
783 if (!event.pressed && m_isDragging) {
784 m_isDragging = false;
785 invalidateVisuals();
786 }
787
788 return false;
789}

791void UnitRow::handleControlClick(const NUIMouseEvent& event, const NUIRect& bounds) {
792 const auto rowBounds = getBounds();
793 const float rowLocalX = event.position.x - rowBounds.x;
794 const float rowLocalY = event.position.y - rowBounds.y;
795 const float localX = rowLocalX - bounds.x;
796 const float localY = rowLocalY - bounds.y;
797 const float dotCenterX = 12.0f;
798 const float dotCenterY = 18.0f;
799 const float dx = localX - dotCenterX;
800 const float dy = localY - dotCenterY;
801 if ((dx * dx + dy * dy) <= 36.0f) {
802 m_manager.setUnitEnabled(m_unitId, !m_isEnabled);
803 updateState();
804 invalidateVisuals();
805 return;
806 }
807
808 const NUIRect muteRect(bounds.width - 92.0f, bounds.height * 0.5f - 10.0f, 24.0f, 20.0f);
809 const NUIRect soloRect(bounds.width - 62.0f, bounds.height * 0.5f - 10.0f, 24.0f, 20.0f);
810 const NUIPoint localPoint(localX, localY);
811 if (muteRect.contains(localPoint)) {
812 m_manager.setUnitMute(m_unitId, !m_isMuted);
813 updateState();
814 invalidateVisuals();
815 return;
816 }
817 if (soloRect.contains(localPoint)) {
818 m_manager.setUnitSolo(m_unitId, !m_isSolo);
819 updateState();
820 invalidateVisuals();
821 return;
822 }
823}

825void UnitRow::handleContextClick(const NUIMouseEvent& event, const NUIRect& bounds) {
826 const auto rowBounds = getBounds();
827 const NUIPoint localPoint(event.position.x - rowBounds.x - bounds.x,
828 event.position.y - rowBounds.y - bounds.y);
829
830 if (shouldUseNoteRoll()) {
831 if (m_onOpenPatternEditor && m_patternId.isValid()) {
832 m_onOpenPatternEditor(m_patternId);
833 }
834 return;
835 }
836
837 if (m_type == Aestra::Audio::UnitType::Audio) {
838 if (m_onLoadUnitSample) {
839 m_onLoadUnitSample(m_unitId);
840 }
841 return;
842 }
843
844 // === Double-click to load sample ===
845 auto now = std::chrono::steady_clock::now();
846 long long nowMs = std::chrono::duration_cast<std::chrono::milliseconds>(now.time_since_epoch()).count();
847 bool isDoubleClick = (nowMs - m_lastClipClickTimeMs < 400) || event.doubleClick;
848 m_lastClipClickTimeMs = nowMs;
849
850 if (isDoubleClick) {
851 // Trigger sample file picker
852 if (m_onLoadUnitSample) m_onLoadUnitSample(m_unitId);
853 return;
854 }
855
856 // === Single click: Grid Interaction - Toggle Steps ===
857 float relativeX = localPoint.x;
858 float availWidth = bounds.width;
859 float stepWidth = std::max(availWidth / static_cast<float>(m_stepCount), 20.0f);
860
861 float contentX = relativeX + m_scrollX;
862 int stepIndex = static_cast<int>(contentX / stepWidth);
863
864 if (stepIndex >= 0 && stepIndex < m_stepCount && m_patternId.isValid()) {
865 auto& pm = m_trackManager->getPatternManager();
866
867 bool removedExisting = false;
868 pm.applyPatch(m_patternId, [this, stepIndex, &removedExisting](Aestra::Audio::PatternSource& p) {
869 if (!p.isMidi()) return;
870 auto& midi = std::get<Aestra::Audio::MidiPayload>(p.payload);
871
872 const double targetBeat = stepIndex * 0.25;
873 auto it = std::find_if(midi.notes.begin(), midi.notes.end(),
874 [this, targetBeat](const Aestra::Audio::MidiNote& n) {
875 const double endBeat = std::max(n.startBeat + 0.25, n.startBeat + n.durationBeats);
876 return n.unitId == m_unitId &&
877 n.pitch == 60 &&
878 targetBeat >= n.startBeat - 0.01 &&
879 targetBeat < endBeat - 0.01;
880 });
881
882 if (it != midi.notes.end()) {
883 midi.notes.erase(it);
884 removedExisting = true;
885 return;
886 }
887
888 const int defaultPitch = (m_type == Aestra::Audio::UnitType::PitchedSampler) ? 36 : 60;
889 midi.notes.push_back({defaultPitch, targetBeat, 0.25, 100.0f / 127.0f, m_unitId});
890 });
891
892 if (removedExisting) {
893 if (m_onPatternEdited) {
894 m_onPatternEdited(m_patternId);
895 }
896 } else {
897 m_isStepEditing = true;
898 m_stepEditStart = stepIndex;
899 m_stepEditEndExclusive = stepIndex + 1;
900
901 Aestra::Audio::AudioQueueCommand audCmd;
902 audCmd.type = Aestra::Audio::AudioQueueCommandType::AuditionUnit;
903 audCmd.trackIndex = m_unitId;
904 audCmd.value1 = 0.8f;
905 if (m_trackManager) {
906 m_trackManager->pushAudioCommand(audCmd);
907 }
908 }
909
910 invalidateVisuals();
911 }

1003

1019 updateState();
1020 invalidateVisuals();
1021
1022 result.accepted = true;
1023 result.message = "Sample loaded: " + filename;
1024 Aestra::Log::info("[UnitRow] Sample dropped: " + filename);
1025 }
1026 }
1027
1028 invalidateVisuals();
1029 return result;
