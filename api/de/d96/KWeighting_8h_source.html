47inline Biquad preFilter(double sampleRate) {
48 constexpr double kPi = 3.14159265358979323846;
49 const double f0 = 1681.974450955533;
50 const double G = 3.999843853973347; // dB
51 const double Q = 0.7071752369554196;
52
53 const double K = std::tan(kPi * f0 / sampleRate);
54 const double K2 = K * K;
55 const double Vh = std::pow(10.0, G / 20.0);
56 const double Vb = std::pow(Vh, 0.4996667741545416);
57 const double a0 = 1.0 + K / Q + K2;
58
59 const Biquad c{(Vh + Vb * K / Q + K2) / a0, 2.0 * (K2 - Vh) / a0, (Vh - Vb * K / Q + K2) / a0,
60 2.0 * (K2 - 1.0) / a0, (1.0 - K / Q + K2) / a0};
61 if (std::abs(sampleRate - 48000.0) < 1.0e-9 && !detail::matches(c, kPreFilter48k, 1.0e-9)) {
62 return kPreFilter48k;
63 }
64 return c;
65}

73inline Biquad rlbHighPass(double sampleRate) {
74 constexpr double kPi = 3.14159265358979323846;
75 const double f0 = 38.13547087602444;
76 const double Q = 0.5003270373238773;
77
78 const double K = std::tan(kPi * f0 / sampleRate);
79 const double K2 = K * K;
80 const double norm = 1.0 + K / Q + K2;
81
82 const Biquad c{1.0, -2.0, 1.0, 2.0 * (K2 - 1.0) / norm, (1.0 - K / Q + K2) / norm};
83 if (std::abs(sampleRate - 48000.0) < 1.0e-9 && !detail::matches(c, kRlb48k, 1.0e-9)) {
84 return kRlb48k;
85 }
