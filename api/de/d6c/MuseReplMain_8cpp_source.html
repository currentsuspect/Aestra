3// MuseRepl — the stdin/stdout entry point to Aestra's agentic runtime.
4//
5// One JSON request per input line, one JSON response per output line
6// (JSONL both ways). Everything goes through MuseService: the same schema
7// validation, command factories, and undo history the UI uses. This is how
8// an external agent (or a human with a pipe) drives a headless session:
9//
10// $ MuseRepl <<'EOF'
11// {"id": 1, "verb": "set_bpm", "args": {"value": 142}}
12// {"id": 2, "verb": "add_track", "args": {"name": "Drums"}}
13// {"id": 3, "verb": "list_tracks"}
14// EOF
15//
16// Flags:
17// --schema print the MuseGrammar command schema as JSON and exit
18// (the agent's tool manifest)
19// --sample-rate <hz> engine sample rate (default 48000)
20

37int main(int argc, char** argv) {
38 uint32_t sampleRate = 48000;
39
40 for (int i = 1; i < argc; ++i) {
41 const std::string arg = argv[i];
42 if (arg == "--schema") {
43 std::cout << MuseGrammar::schemaToJsonString();
44 return 0;
45 }
46 if (arg == "--sample-rate" && i + 1 < argc) {
47 const long parsed = std::strtol(argv[++i], nullptr, 10);
48 if (parsed >= 8000 && parsed <= 192000) {
49 sampleRate = static_cast<uint32_t>(parsed);
50 } else {
51 std::cerr << "invalid --sample-rate, using 48000\n";
52 }
53 continue;
54 }
55 if (arg == "--help" || arg == "-h") {
56 std::cout << "MuseRepl: JSONL in on stdin, JSONL out on stdout.\n"
57 " --schema print command schema JSON and exit\n"
58 " --sample-rate <hz> engine sample rate (default 48000)\n";
59 return 0;
60 }
61 std::cerr << "unknown argument: " << arg << "\n";
62 return 2;
63 }
64
65 // Session setup mirrors the app's wiring: TrackManager owns the model and
66 // history, AudioEngine owns transport/tempo, the registry binds both.
67 // Built-in plugins (the sampler load_sample instantiates) register inside
68 // PluginManager::initialize(); without it units stay silent.
69 if (!PluginManager::getInstance().initialize()) {
70 std::cerr << "failed to initialize plugin manager\n";
71 return 1;
72 }
73
74 auto trackManager = std::make_shared<TrackManager>();
75 trackManager->setOutputSampleRate(static_cast<double>(sampleRate));
76 trackManager->setInputSampleRate(static_cast<double>(sampleRate));
77 trackManager->setInputChannelCount(0);
78 // The app wires this in AestraContent; without it, creating a unit does
79 // not create its default MIDI pattern.
80 trackManager->getUnitManager().setPatternManager(&trackManager->getPatternManager());
81
82 AudioEngine engine;
83 engine.setSampleRate(sampleRate);
84 // Must cover AudioExporter's 4096-frame render blocks (render_song):
85 // processBlock does not split blocks larger than the configured maximum.
86 engine.setBufferConfig(4096, 2);
87 MuseService::wireHeadlessEngine(trackManager, engine);
88 if (!engine.initialize()) {
89 std::cerr << "failed to initialize audio engine\n";
90 return 1;
91 }
92
93 CommandRegistry::initialize(trackManager.get());
94 CommandRegistry::setAudioEngine(&engine);
95
96 MuseService service(trackManager.get(), &engine);
97
98 std::string line;
99 while (std::getline(std::cin, line)) {
100 // Skip blank lines so hand-driven sessions can space things out.
101 bool blank = true;
102 for (char c : line) {
103 if (!std::isspace(static_cast<unsigned char>(c))) {
104 blank = false;
105 break;
106 }
107 }
108 if (blank) continue;
109
110 std::cout << service.handleRequest(line) << "\n";
111 std::cout.flush();
112 }
113
114 return 0;
115}
