3// MuseRepl — the stdin/stdout entry point to Aestra's agentic runtime.
4//
5// One JSON request per input line, one JSON response per output line
6// (JSONL both ways). Everything goes through MuseService: the same schema
7// validation, command factories, and undo history the UI uses. This is how
8// an external agent (or a human with a pipe) drives a headless session:
9//
10// $ MuseRepl <<'EOF'
11// {"id": 1, "verb": "set_bpm", "args": {"value": 142}}
12// {"id": 2, "verb": "add_track", "args": {"name": "Drums"}}
13// {"id": 3, "verb": "list_tracks"}
14// EOF
15//
16// Flags:
17// --schema print the MuseGrammar command schema as JSON and exit
18// (the agent's tool manifest)
19// --sample-rate <hz> engine sample rate (default 48000)
20

40int main(int argc, char** argv) {
41 uint32_t sampleRate = 48000;
42
43 int socketPort = -1; // -1 = stdin/stdout mode
44
45 for (int i = 1; i < argc; ++i) {
46 const std::string arg = argv[i];
47 if (arg == "--port" && i + 1 < argc) {
48 const long parsed = std::strtol(argv[++i], nullptr, 10);
49 if (parsed >= 0 && parsed <= 65535) {
50 socketPort = static_cast<int>(parsed);
51 } else {
52 std::cerr << "invalid --port\n";
53 return 2;
54 }
55 continue;
56 }
57 if (arg == "--schema") {
58 std::cout << MuseGrammar::schemaToJsonString();
59 return 0;
60 }
61 if (arg == "--sample-rate" && i + 1 < argc) {
62 const long parsed = std::strtol(argv[++i], nullptr, 10);
63 if (parsed >= 8000 && parsed <= 192000) {
64 sampleRate = static_cast<uint32_t>(parsed);
65 } else {
66 std::cerr << "invalid --sample-rate, using 48000\n";
67 }
68 continue;
69 }
70 if (arg == "--help" || arg == "-h") {
71 std::cout << "MuseRepl: JSONL in on stdin, JSONL out on stdout.\n"
72 " --schema print command schema JSON and exit\n"
73 " --sample-rate <hz> engine sample rate (default 48000)\n"
74 " --port <port> serve JSONL on 127.0.0.1:<port> instead of\n"
75 " stdin/stdout (0 = ephemeral, printed)\n";
76 return 0;
77 }
78 std::cerr << "unknown argument: " << arg << "\n";
79 return 2;
80 }
81
82 // Session setup mirrors the app's wiring: TrackManager owns the model and
83 // history, AudioEngine owns transport/tempo, the registry binds both.
84 // Built-in plugins (the sampler load_sample instantiates) register inside
85 // PluginManager::initialize(); without it units stay silent.
86 if (!PluginManager::getInstance().initialize()) {
87 std::cerr << "failed to initialize plugin manager\n";
88 return 1;
89 }
90
91 auto trackManager = std::make_shared<TrackManager>();
92 trackManager->setOutputSampleRate(static_cast<double>(sampleRate));
93 trackManager->setInputSampleRate(static_cast<double>(sampleRate));
94 trackManager->setInputChannelCount(0);
95 // The app wires this in AestraContent; without it, creating a unit does
96 // not create its default MIDI pattern.
97 trackManager->getUnitManager().setPatternManager(&trackManager->getPatternManager());
98
99 AudioEngine engine;
100 engine.setSampleRate(sampleRate);
101 // Must cover AudioExporter's 4096-frame render blocks (render_song):
102 // processBlock does not split blocks larger than the configured maximum.
103 engine.setBufferConfig(4096, 2);
104 MuseService::wireHeadlessEngine(trackManager, engine);
105 if (!engine.initialize()) {
106 std::cerr << "failed to initialize audio engine\n";
107 return 1;
108 }
109
110 CommandRegistry::initialize();
111
112 MuseService service(trackManager.get(), &engine);
113
114 // Socket mode: the same headless session served over localhost — what
115 // muse-agent connects to when no GUI is running.
116 if (socketPort >= 0) {
117 MuseSocketServer server;
118 std::string error;
119 if (!server.start(static_cast<uint16_t>(socketPort), error)) {
120 std::cerr << "failed to start socket server: " << error << "\n";
121 return 1;
122 }
123 std::cout << "muse-port " << server.port() << std::endl;
124 while (server.isRunning()) {
125 if (server.processPending(service) == 0) {
126 std::this_thread::sleep_for(std::chrono::milliseconds(10));
127 }
128 }
129 return 0;
130 }
131
132 std::string line;
133 while (std::getline(std::cin, line)) {
134 // Skip blank lines so hand-driven sessions can space things out.
135 bool blank = true;
136 for (char c : line) {
137 if (!std::isspace(static_cast<unsigned char>(c))) {
138 blank = false;
139 break;
140 }
141 }
142 if (blank) continue;
143
144 std::cout << service.handleRequest(line) << "\n";
145 std::cout.flush();
146 }
147
148 return 0;
149}
