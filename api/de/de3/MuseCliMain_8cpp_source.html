3// muse — one-shot CLI for Aestra's Muse socket.
4//
5// MuseRepl speaks JSONL on stdin/stdout and muse-agent hands the surface to a
6// model. Neither is what you want when a *person* (or a coding agent) is at a
7// terminal driving a session by hand: those need a request framed, a socket
8// opened, and a line parsed for every single verb. This is that missing piece —
9// one verb per invocation, flags instead of hand-written JSON, the response on
10// stdout, and the outcome in the exit status:
11//
12// muse list_units
13// muse add_unit --type sampler --name Kick
14// muse set_bpm --value 140
15// muse render_pattern --pattern 1 --file /tmp/loop.wav
16//
17// It talks to whatever is listening: the running GUI app (started with
18// AESTRA_MUSE_PORT set) or a headless `MuseRepl --port N`. Edits against the
19// live app land on the main thread through the same command system as user
20// edits, so they share the undo history.
21//
22// Deliberately does NOT link libcurl or any provider code — this is a socket
23// client and a JSON formatter, nothing more.

44void printUsage() {
45 std::cout <<
46 "muse — one verb per invocation against a running Aestra (or MuseRepl).\n"
47 "\n"
48 "Usage:\n"
49 " muse [options] <verb> [--flag value ...]\n"
50 "\n"
51 "Options:\n"
52 " --port <port> Muse socket port (default: $AESTRA_MUSE_PORT)\n"
53 " --host <host> default 127.0.0.1\n"
54 " --args-json <json> args object verbatim; for verbs flat flags cannot\n"
55 " express (batch takes an array of commands)\n"
56 " --result print only the result field, not the envelope\n"
57 " --raw print the response line exactly as received\n"
58 " --compact single-line JSON (default is indented)\n"
59 " --timeout <seconds> give up waiting for a response (default 120,\n"
60 " 0 waits forever). A bounce of a long timeline\n"
61 " is slow but finite; a stalled peer is not.\n"
62 " -h, --help\n"
63 "\n"
64 "Flag values are typed by shape: true/false become booleans, numbers\n"
65 "become numbers, everything else stays a string. A flag with no value\n"
66 "is a boolean true.\n"
67 "\n"
68 "Exit status: 0 the verb succeeded, 1 the service refused it, 2 it was\n"
69 "never asked (bad usage, or nothing listening).\n"
70 "\n"
71 "Start something to talk to:\n"
72 " AESTRA_MUSE_PORT=41952 ./Aestra # drive the live app\n"
73 " ./MuseRepl --port 41952 # headless\n"
74 "\n"
75 "`muse get_schema` lists every verb, its flags, and the semantics that\n"
76 "are not visible in the protocol.\n";
77}

79bool parsePort(const std::string& text, uint16_t& out) {
80 char* end = nullptr;
81 const long parsed = std::strtol(text.c_str(), &end, 10);
82 if (end == nullptr || *end != '\0' || parsed <= 0 || parsed > 65535) {
83 return false;
84 }
85 out = static_cast<uint16_t>(parsed);
86 return true;
87}

91int main(int argc, char** argv) {
92 std::vector<std::string> tokens(argv + 1, argv + argc);
93
94 Aestra::MuseAgent::CliInvocation invocation;
95 std::string error;
96 if (!Aestra::MuseAgent::parseMuseCli(tokens, invocation, error)) {
97 std::cerr << error << "\n\n";
98 printUsage();
99 return EXIT_USAGE;
100 }
101 if (invocation.helpRequested) {
102 printUsage();
103 return EXIT_OK;
104 }
105
106 if (invocation.portText.empty()) {
107 if (const char* env = std::getenv("AESTRA_MUSE_PORT")) {
108 invocation.portText = env;
109 }
110 }
111 if (invocation.portText.empty()) {
112 std::cerr << "no port: pass --port, or set AESTRA_MUSE_PORT.\n"
113 "Start the app with AESTRA_MUSE_PORT=<port> to drive the live\n"
114 "session, or run MuseRepl --port <port> for a headless one.\n";
115 return EXIT_USAGE;
116 }
117
118 uint16_t port = 0;
119 if (!parsePort(invocation.portText, port)) {
120 std::cerr << "invalid port: " << invocation.portText << "\n";
121 return EXIT_USAGE;
122 }
123
124 Aestra::MuseAgent::MuseSocketClient client;
125 if (!client.connect(invocation.host, port, error)) {
126 std::cerr << "cannot reach a Muse socket at " << invocation.host << ":" << port
127 << " — " << error << "\n"
128 << "Is the app running with AESTRA_MUSE_PORT set, or MuseRepl --port?\n";
129 return EXIT_USAGE;
130 }
131
132 client.setReadTimeoutMs(invocation.timeoutSeconds * 1000);
133
134 Aestra::MuseAgent::MuseSocketClient::Outcome outcome{};
135 const std::string line =
136 client.request(Aestra::MuseAgent::buildMuseRequest(invocation).toString(), &outcome);
137
138 if (outcome == Aestra::MuseAgent::MuseSocketClient::Outcome::TimedOut) {
139 std::cerr << "timed out after " << invocation.timeoutSeconds << "s waiting for "
140 << invocation.verb << " — raise --timeout, or 0 to wait indefinitely\n";
141 return EXIT_USAGE;
142 }
143 if (outcome == Aestra::MuseAgent::MuseSocketClient::Outcome::Disconnected || line.empty()) {
144 std::cerr << "no response from " << invocation.host << ":" << port
145 << " (the session may have exited mid-request)\n";
146 return EXIT_USAGE;
147 }
148
149 // Validate before printing, and identically in both modes. --raw controls
150 // the *formatting* of a well-formed envelope; it is not a licence to pass
151 // protocol garbage through with a status derived from a lenient parse.
152 bool consumedAll = false;
153 JSON response = JSON::parseStrict(line, consumedAll);
154 if (!consumedAll || !response.isObject() || !response.has("status")) {
155 std::cerr << "malformed response (not a single JSON envelope): " << line << "\n";
156 return EXIT_USAGE;
157 }
158 const bool ok = response["status"].asString() == "ok";
159
160 if (invocation.raw) {
161 std::cout << line << std::endl;
162 return ok ? EXIT_OK : EXIT_REQUEST_FAILED;
163 }
164
165 const int indent = invocation.compact ? 0 : 2;
166 if (invocation.resultOnly && response.has("result")) {
167 std::cout << response["result"].toString(indent) << std::endl;
168 } else {
169 std::cout << response.toString(indent) << std::endl;
170 }
171
172 return ok ? EXIT_OK : EXIT_REQUEST_FAILED;
173}
