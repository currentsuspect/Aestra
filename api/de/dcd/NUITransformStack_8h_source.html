105inline NUIComposedTransform composeTransformStack(const NUITransform2D* stack, std::size_t count) {
106 NUIComposedTransform out;
107 for (std::size_t i = 0; i < count; ++i) {
108 // Accumulate the translation in the *current* outer scale before
109 // folding this level's scale in. Doing it the other way round is the
110 // exact mistake being fixed here.
111 out.tx += out.scale * stack[i].tx;
112 out.ty += out.scale * stack[i].ty;
113 out.scale *= stack[i].scale;
114 }
115 return out;
116}
