13void InternalPluginRegistry::registerPlugin(Registration registration) {
14 if (!registration.info.isValid() || !registration.createInstance) {
15 return;
16 }
17
18 if (!registration.isAvailable) {
19 registration.isAvailable = [] { return true; };
20 }
21
22 std::lock_guard<std::mutex> lock(m_mutex);
23 m_registrations[registration.info.id] = std::move(registration);
24}

26std::vector<PluginInfo> InternalPluginRegistry::listAvailablePlugins() const {
27 std::vector<PluginInfo> plugins;
28
29 std::lock_guard<std::mutex> lock(m_mutex);
30 plugins.reserve(m_registrations.size());
31 for (const auto& entry : m_registrations) {
32 const auto& registration = entry.second;
33 if (registration.isAvailable && !registration.isAvailable()) {
34 continue;
35 }
36 plugins.push_back(registration.info);
37 }
38
39 return plugins;
40}

42bool InternalPluginRegistry::isRegisteredPlugin(const std::string& pluginId) const {
43 std::lock_guard<std::mutex> lock(m_mutex);

47bool InternalPluginRegistry::isPluginAvailable(const std::string& pluginId) const {
48 std::lock_guard<std::mutex> lock(m_mutex);
49 const auto it = m_registrations.find(pluginId);
50 if (it == m_registrations.end()) {
51 return false;
52 }
53 return !it->second.isAvailable || it->second.isAvailable();
54}

56PluginInstancePtr InternalPluginRegistry::createInstance(const std::string& pluginId) const {
57 std::lock_guard<std::mutex> lock(m_mutex);
58 const auto it = m_registrations.find(pluginId);
59 if (it == m_registrations.end()) {
60 return nullptr;
61 }
62
63 const auto& registration = it->second;
64 if (registration.isAvailable && !registration.isAvailable()) {
65 return nullptr;
66 }
67
68 return registration.createInstance ? registration.createInstance() : nullptr;
69}
