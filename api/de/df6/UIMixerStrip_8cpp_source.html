23 constexpr float HEADER_H = 30.0f;
24 constexpr float KNOB_H = 32.0f;
25 constexpr float FX_H = 30.0f;
26 constexpr float BUTTONS_H = 26.0f;
27 constexpr float FOOTER_H = 22.0f;
28 constexpr float PAD = 8.0f;
29 constexpr float GAP = 8.0f;
30 constexpr float SIDE_INSET = 4.0f;
31 constexpr float SECTION_GAP = 8.0f;
32 constexpr float METER_W = 22.0f;
33 constexpr float MASTER_METER_W = 36.0f;
34 // Labelled Peak / LUFS / Gain blocks at the foot of the master strip.
35 // Each block is a label line stacked over a value line; two of them sit in
36 // the meter column, so the band must fit 2 * BLOCK_H exactly or the second
37 // value falls off the bottom of the strip.
38 constexpr float MASTER_LABEL_H = 12.0f;
39 constexpr float MASTER_VALUE_H = 14.0f;
40 constexpr float MASTER_BLOCK_H = MASTER_LABEL_H + MASTER_VALUE_H; // 26
41 // Two stacked blocks plus bottom slack. With only 4px of slack the second
42 // value landed on the strip's bottom edge and was not drawn at all.
43 constexpr float MASTER_READOUT_H = MASTER_BLOCK_H * 2.0f + 14.0f; // 66
44
45 constexpr float SELECT_TOP_H = 3.0f;
46 constexpr float MIXER_MIN_CHANNEL_HEIGHT = 220.0f;

48 // Resolve a palette index to a renderer colour (matches UIMixerHeader::colorFromARGB).
49 NUIColor trackColorFromIndex(int index)
50 {
51 const uint32_t argb = paletteIndexToARGB(index);
52 return NUIColor(((argb >> 16) & 0xFF) / 255.0f,
53 ((argb >> 8) & 0xFF) / 255.0f,
54 (argb & 0xFF) / 255.0f,
55 ((argb >> 24) & 0xFF) / 255.0f);
56 }

58 std::string compactRouteName(uint32_t targetId, const std::string& targetName)
59 {
60 if (targetId == 0 || targetName == "Master" || targetName == "MASTER") {
61 return "M";
62 }
63 const std::string channelPrefix = "Channel ";
64 if (targetName.rfind(channelPrefix, 0) == 0) {
65 return "C" + targetName.substr(channelPrefix.size());
66 }
67 // Projects saved before strips were renamed still carry "Insert N".
68 const std::string legacyInsertPrefix = "Insert ";
69 if (targetName.rfind(legacyInsertPrefix, 0) == 0) {
70 return "I" + targetName.substr(legacyInsertPrefix.size());
71 }
72 const std::string legacyTrackPrefix = "Track ";
73 if (targetName.rfind(legacyTrackPrefix, 0) == 0) {
74 return "I" + targetName.substr(legacyTrackPrefix.size());
75 }
76 const std::string lanePrefix = "Lane ";
77 if (targetName.rfind(lanePrefix, 0) == 0) {
78 return "L" + targetName.substr(lanePrefix.size());
79 }
80 if (targetName.size() <= 6) {
81 return targetName;
82 }
83 return targetName.substr(0, 6);
84 }

86 std::string buildStripRouteSummary(const Aestra::ChannelViewModel& channel)
87 {
88 if (channel.id == 0) {
89 return channel.routeName;
90 }
91
92 const bool busOnly = !channel.masterSendEnabled && channel.mainOutputId != 0;
93 int audibleSendCount = 0;
94 int sidechainSendCount = 0;
95 std::string audibleTarget;
96 std::string sidechainTarget;
97 std::unordered_set<uint32_t> audibleTargets;
98 std::unordered_set<uint32_t> sidechainTargets;
99
100 for (const auto& send : channel.sends) {
101 if (send.muted) continue;
102 if (send.sidechainOnly) {
103 ++sidechainSendCount;
104 sidechainTargets.insert(send.targetId);
105 if (sidechainTarget.empty()) {
106 sidechainTarget = compactRouteName(send.targetId, send.targetName);
107 }
108 continue;
109 }
110 ++audibleSendCount;
111 audibleTargets.insert(send.targetId);
112 if (audibleTarget.empty()) {
113 audibleTarget = compactRouteName(send.targetId, send.targetName);
114 }
115 }
116
117 std::ostringstream summary;
118 bool first = true;
119 auto appendToken = [&](const std::string& token) {
120 if (token.empty()) return;
121 if (!first) summary << " • ";
122 summary << token;
123 first = false;
124 };
125
126 if (busOnly) {
127 appendToken("Bus " + compactRouteName(channel.mainOutputId, channel.routeName));
128 }
129 if (audibleSendCount > 0) {
130 if (audibleTargets.size() == 1) {
131 appendToken("Snd " + audibleTarget);
132 } else {
133 appendToken("S+" + std::to_string(static_cast<int>(audibleTargets.size())));
134 }
135 }
136 if (sidechainSendCount > 0) {
137 if (sidechainTargets.size() == 1) {
138 appendToken("SC " + sidechainTarget);
139 } else {
140 appendToken("SC+" + std::to_string(static_cast<int>(sidechainTargets.size())));
141 }
142 }
143
144 return summary.str();
145 }

147 std::string buildFxStatus(const Aestra::ChannelViewModel& channel)
148 {
149 if (!channel.channel) {
150 return {};
151 }
152
153 bool hasSidechainSend = false;
154 for (const auto& send : channel.sends) {
155 if (!send.muted && send.sidechainOnly) {
156 hasSidechainSend = true;
157 break;
158 }
159 }
160
161 auto& chain = channel.channel->getEffectChain();
162 for (size_t i = 0; i < Aestra::Audio::EffectChain::MAX_SLOTS; ++i) {
163 auto plugin = chain.getPlugin(i);
164 auto comp = std::dynamic_pointer_cast<Aestra::Audio::Plugins::AestraComp>(plugin);
165 if (!comp) {
166 continue;
167 }
168
169 const float grDb = comp->getCurrentGainReductionDb();
170 if (grDb >= 0.1f) {
171 std::ostringstream out;
172 out.setf(std::ios::fixed);
173 out.precision(grDb >= 10.0f ? 0 : 1);
174 out << "GR " << grDb;
175 return out.str();
176 }
177
178 if (hasSidechainSend && channel.sidechainPeak > 0.001f) {
179 return "SC live";
180 }
181
182 return {};
183 }
184
185 return {};
186 }

188 std::string formatDuckGain(float gain, const std::string& sourceLabel)
189 {
190 const std::string prefix = sourceLabel.empty() ? "DUCK" : sourceLabel;
191 if (!std::isfinite(gain) || gain <= 0.0f) {
192 return prefix;
193 }
194 const float db = 20.0f * std::log10(std::clamp(gain, 1.0e-6f, 1.0f));
195 std::ostringstream out;
196 out.setf(std::ios::fixed);
197 out.precision(std::abs(db) >= 9.5f ? 0 : 1);
198 out << prefix << " " << db << "dB";
199 return out.str();
200 }

425void UIMixerStrip::layoutChildren()
426{
427 auto bounds = getBounds();
428
429 const float contentX = bounds.x + SIDE_INSET;
430 const float contentW = std::max(1.0f, bounds.width - SIDE_INSET * 2.0f);
431 float y = bounds.y + PAD * 0.75f;
432
433 if (m_header) {
434 m_header->setBounds(contentX, y, contentW, HEADER_H);
435 y += HEADER_H + SECTION_GAP;
436 }
437
438 // Input Dropdown Removed from Strip Layout
439
440 // Three-knob row: Trim + Pan + Width (channel controls grouped together)
441 const bool showTrim = (m_trimKnob && m_trimKnob->isVisible());
442 const bool showPan = (m_panKnob && m_panKnob->isVisible());
443 const bool showWidth = (m_widthKnob && m_widthKnob->isVisible());
444 int visibleKnobCount = (showTrim ? 1 : 0) + (showPan ? 1 : 0) + (showWidth ? 1 : 0);
445 if (visibleKnobCount > 0) {
446 const float knobW = (contentW - GAP * (visibleKnobCount - 1)) / visibleKnobCount;
447 float knobX = contentX;
448 if (showTrim) {
449 m_trimKnob->setBounds(knobX, y, knobW, KNOB_H);
450 knobX += knobW + GAP;
451 }
452 if (showPan) {
453 m_panKnob->setBounds(knobX, y, knobW, KNOB_H);
454 knobX += knobW + GAP;
455 }
456 if (showWidth) {
457 m_widthKnob->setBounds(knobX, y, knobW, KNOB_H);
458 }
459 y += KNOB_H + SECTION_GAP;
460 }
461
462 // M/S/R buttons moved below knobs, above FX summary
463 const bool hasButtons = (m_buttons && m_buttons->isVisible());
464 if (hasButtons) {
465 m_buttons->setBounds(contentX, y, contentW, BUTTONS_H);
466 y += BUTTONS_H + 6.0f;
467 }
468
469 // FX Summary (Add Insert) below the channel controls
470 if (m_fxSummary && m_fxSummary->isVisible()) {
471 if (m_channelId == 0) {
472 y += 10.0f;
473 }
474 m_fxSummary->setBounds(contentX, y, contentW, FX_H);
475 y += FX_H + SECTION_GAP;
476 }
477
478 const bool hasFooter = (m_footer && m_footer->isVisible());
479 const float footerH = hasFooter ? FOOTER_H : 0.0f;
480 const float footerY = bounds.y + bounds.height - PAD - footerH;
481 if (hasFooter) {
482 m_footer->setBounds(contentX, footerY, contentW, FOOTER_H);
483 }
484
485 const bool isMaster = (m_channelId == 0);
486 const float meterW = isMaster ? MASTER_METER_W : METER_W;
487
488 // The master reserves a labelled Peak/LUFS/Gain block at the foot of the
489 // strip; its footer is hidden, so that space is otherwise unused.
490 const float readoutH = isMaster ? MASTER_READOUT_H : 0.0f;
491
492 // Meter + fader fill the strip between the channel controls and the footer.
493 // (Previously sized to 62% of the free height and bottom-anchored, which
494 // left a large dead gap above the fader — the tallest, most-used control.)
495 const float meterY = y;
496 const float meterH = std::max(16.0f, (footerY - GAP - readoutH) - y);
497 const float meterX = contentX + 2.0f;
498
499 if (m_meter) {
500 m_meter->setBounds(meterX, meterY, meterW, meterH);
501 }
502
503 if (m_fader) {
504 const float faderX = meterX + meterW + GAP;
505 const float faderW = std::max(16.0f, (contentX + contentW) - faderX);
506 m_fader->setBounds(faderX, meterY, faderW, meterH);
507 }
508
509 // Readouts sit under the column they describe: observed signal (Peak, LUFS)
510 // under the meter, user-controlled gain under the fader. Keeping all three
511 // in one bottom block made the eye travel from the live bars to a detached
512 // table to interpret state, and blurred which side owns which number.
513 if (isMaster) {
514 const float readoutY = footerY - MASTER_READOUT_H;
515 m_masterMeterReadoutRect = NUIRect{meterX, readoutY, meterW + GAP, MASTER_READOUT_H};
516 const float gainX = meterX + meterW + GAP;
517 m_masterGainReadoutRect =
518 NUIRect{gainX, readoutY, std::max(16.0f, (contentX + contentW) - gainX), MASTER_READOUT_H};
519 } else {
520 m_masterMeterReadoutRect = NUIRect{};
521 m_masterGainReadoutRect = NUIRect{};
522 }
523}

525void UIMixerStrip::renderMasterReadout(NUIRenderer& renderer,
526 const Aestra::ChannelViewModel& channel)
527{
528 const NUIRect& meterArea = m_masterMeterReadoutRect;
529 const NUIRect& gainArea = m_masterGainReadoutRect;
530 if (meterArea.width <= 0.0f || meterArea.height <= 0.0f) return;
531
532 auto& theme = NUIThemeManager::getInstance();
533 const NUIColor labelColor = theme.getColor("textSecondary").withAlpha(0.72f);
534 const NUIColor valueColor = theme.getColor("textPrimary").withAlpha(0.95f);
535 const NUIColor clipColor = theme.getColor("meterCrit");
536
537 const bool clipped = channel.clipLatchL || channel.clipLatchR;
538
539 // Peak: prefer the held value, which is what the meter's hold line shows.
540 float peakDb = std::max(channel.smoothedPeakL, channel.smoothedPeakR);
541 if (channel.peakHoldL > Aestra::MixerMath::DB_MIN ||
542 channel.peakHoldR > Aestra::MixerMath::DB_MIN) {
543 peakDb = std::max(channel.peakHoldL, channel.peakHoldR);
544 }
545
546 char buf[32];
547 // Stacked label-over-value: the master columns are too narrow for a
548 // side-by-side label and a signed value.
549 auto block = [&](const NUIRect& area, int index, const char* label,
550 const std::string& value, const NUIColor& color) {
551 const float y = area.y + static_cast<float>(index) * MASTER_BLOCK_H;
552 const NUIRect labelRect{area.x, y, area.width, MASTER_LABEL_H};
553 const NUIRect valueRect{area.x, y + MASTER_LABEL_H, area.width, MASTER_VALUE_H};
554 renderer.drawTextCentered(label, labelRect, 8.5f, labelColor);
555 renderer.drawTextCentered(value, valueRect, 10.0f, color);
556 };
557
558 // --- Observed signal, under the meter ---
559 // PEAK — dBFS, explicitly. Doubles as the clip/over indicator.
560 std::string peakText;
561 if (peakDb <= Aestra::MixerMath::DB_MIN) {
562 peakText = "\xE2\x88\x92\xE2\x88\x9E";
563 } else {
564 std::snprintf(buf, sizeof(buf), "%.1f", peakDb);
565 peakText = buf;
566 }
567 block(meterArea, 0, clipped ? "OVER" : "PEAK dBFS", peakText, clipped ? clipColor : valueColor);
568
569 // LUFS — integrated, and the sign always survives (see UIMixerMeter).
570 // Integrated LUFS is undefined until the gate has material; show the same
571 // silence marker the peak readout uses rather than an ambiguous dash.
572 std::string lufsText = "\xE2\x88\x92\xE2\x88\x9E";
573 if (channel.integratedLufs > -100.0f) {
574 std::snprintf(buf, sizeof(buf), "%.1f", channel.integratedLufs);
575 lufsText = buf;
576 }
577 block(meterArea, 1, "LUFS INT", lufsText, valueColor);
578
579 // --- User-controlled gain, under the fader ---
580 if (gainArea.width > 0.0f) {
581 // Same silence floor as UIMixerFader::updateCachedText. Without it the
582 // fader reads "−∞" while the readout right under it reads "-90.0",
583 // which looks like two different values for one control.
584 std::string gainText = "\xE2\x88\x92\xE2\x88\x9E";
585 if (channel.faderGainDb > UIMixerFader::FADER_FLOOR_THRESHOLD) {
586 std::snprintf(buf, sizeof(buf), "%.1f", channel.faderGainDb);
587 gainText = buf;
588 }
589 block(gainArea, 0, "GAIN dB", gainText, valueColor);
590 }
591}

600void UIMixerStrip::onUpdate(double deltaTime)
601{
602 (void)deltaTime;
603 if (!m_viewModel) {
604 updateChildren(deltaTime);
605 return;
606 }
607
608 auto* channel = m_viewModel->getChannelById(m_channelId);
609 if (!channel) {
610 updateChildren(deltaTime);
611 return;
612 }
613
614 const bool selected = (m_viewModel->getSelectedChannelId() == static_cast<int32_t>(m_channelId));

769void UIMixerStrip::onRender(NUIRenderer& renderer)
770{
771 Aestra::ChannelViewModel* channel = nullptr;
772 if (m_viewModel) {
773 channel = m_viewModel->getChannelById(m_channelId);

775
776 const auto bounds = getBounds();
777 if (bounds.isEmpty()) return;
778
779 const bool selected = m_viewModel && (m_viewModel->getSelectedChannelId() == static_cast<int32_t>(m_channelId));

781 // Unified "Deep Black" background for ALL strips.
782 // Use Squircle (Rounded Rect) for standard Apple-like look
783 float radius = NUIThemeManager::getInstance().getRadius("m");
784 renderer.fillRoundedRect(bounds, radius, m_stripBg);
785
786 // Master gets a slightly different tone to distinguish it (optional, but good for hierarchy)
787 if (m_channelId == 0) {
788 // Subtle accent-tinted vertical gradient for visual hierarchy
789 const NUIColor accent = NUIThemeManager::getInstance().getColor("primary");
790 renderer.fillRoundedRect(bounds, radius,
791 accent.withAlpha(0.04f));
792 renderer.fillRectGradient({bounds.x, bounds.y, bounds.width, 90.0f},
793 accent.withAlpha(0.10f), accent.withAlpha(0.0f), true);
794 renderer.strokeRoundedRect(bounds, radius, 1.0f, m_masterBorder);
795 }
796
797 if (selected) {
798 // Selection = a slight tinted body + a strong top edge. No outline.
799 //
800 // This started at five layers (tint, top bar, inner highlight, outer
801 // glow ring, outline). The outline was the last redundant one: with a
802 // tinted body and a bright edge the strip is already unmistakable, and
803 // the border was what made a selected strip read as a different kind of
804 // component rather than the same strip, selected.
805 renderer.fillRoundedRect(bounds, radius, m_selectedTint);
806 renderer.fillRect(NUIRect{bounds.x, bounds.y, bounds.width, SELECT_TOP_H}, m_selectedTopHighlight);

813 (m_panKnob && m_panKnob->isDragging()) ||

816 invalidateStaticCache();
817 renderChildren(renderer);
818 if (m_channelId == 0 && channel) {
819 renderMasterReadout(renderer, *channel);
820 }
821 if (channel && channel->muted) {
822 renderer.fillRect(getBounds(), m_mutedOverlay);
823 }
824 return;
825 }
826
827 // Static caching Disabled due to HiDPI blurriness issues.
828 // The performance impact of redrawing vector UI is minimal on modern systems.
829 /*
830 if (m_staticCacheInvalidated) {
831 renderer.invalidateCache(m_staticCacheId);
832 m_staticCacheInvalidated = false;
833 }
834
835 const bool usedCache = renderer.renderCachedOrUpdate(m_staticCacheId, bounds, [&]() {
836 renderer.clear(NUIColor(0, 0, 0, 0));
837 renderer.pushTransform(-bounds.x, -bounds.y);
838 renderStaticLayer(renderer);
839 renderer.popTransform();
840 });
841
842 if (usedCache) {
843 if (m_meter) {
844 m_meter->onRender(renderer);

846 if (channel && channel->muted) {
847 renderer.fillRect(getBounds(), m_mutedOverlay);
848 }
849 return;

852
853 renderChildren(renderer);
854 if (m_channelId == 0 && channel) {
855 renderMasterReadout(renderer, *channel);
856 }
857 if (m_channelId == 0 && m_viewModel && m_viewModel->isPreviewDuckingActive()) {
858 auto& theme = NUIThemeManager::getInstance();

876bool UIMixerStrip::onMouseEvent(const NUIMouseEvent& event)
877{
878 if (m_meter && m_viewModel && event.pressed &&
879 (event.button == NUIMouseButton::Left || event.button == NUIMouseButton::Right) &&
880 getBounds().contains(event.position)) {
881 const auto stripBounds = getBounds();
882 const auto meterBounds = m_meter->getBounds();
883 auto* channel = m_viewModel->getChannelById(m_channelId);
884 std::ostringstream hitLog;
885 hitLog << "[MixerClipHit] channel=" << m_channelId
886 << " button=" << static_cast<int>(event.button)
887 << " click=(" << event.position.x << "," << event.position.y << ")"
888 << " strip=(" << stripBounds.x << "," << stripBounds.y << ","
889 << stripBounds.width << "," << stripBounds.height << ")"
890 << " meter=(" << meterBounds.x << "," << meterBounds.y << ","
891 << meterBounds.width << "," << meterBounds.height << ")"
892 << " inMeter=" << (meterBounds.contains(event.position) ? 1 : 0)
893 << " latchL=" << (channel ? (channel->clipLatchL ? 1 : 0) : -1)
894 << " latchR=" << (channel ? (channel->clipLatchR ? 1 : 0) : -1);
895 AESTRA_LOG_INFO(hitLog.str());
896 }
897
898 if (m_meter && m_viewModel && event.pressed &&
899 (event.button == NUIMouseButton::Left || event.button == NUIMouseButton::Right) &&
900 m_meter->getBounds().contains(event.position)) {
901 auto* channel = m_viewModel->getChannelById(m_channelId);
902 if (channel && (channel->clipLatchL || channel->clipLatchR)) {
903 m_viewModel->clearClipLatch(channel->id);
904 if (m_meterSnapshots) {
905 m_meterSnapshots->clearClip(channel->slotIndex);
906 }
907 m_meter->setClipLatch(false, false);
908 invalidateStaticCache();
909 }
910 return true;
911 }
912
913 bool handledSelection = false;
914 if (m_viewModel && event.pressed && event.button == NUIMouseButton::Left) {
915 if (getBounds().contains(event.position)) {
916 m_viewModel->setSelectedChannelId(static_cast<int32_t>(m_channelId));
917 handledSelection = true;
918 }
919 }
920
921 const bool handledByChildren = NUIComponent::onMouseEvent(event);
922 return handledSelection || handledByChildren;
923}
