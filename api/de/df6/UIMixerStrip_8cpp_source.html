38 // Resolve a palette index to a renderer colour (matches UIMixerHeader::colorFromARGB).
39 NUIColor trackColorFromIndex(int index)
40 {
41 const uint32_t argb = paletteIndexToARGB(index);
42 return NUIColor(((argb >> 16) & 0xFF) / 255.0f,
43 ((argb >> 8) & 0xFF) / 255.0f,
44 (argb & 0xFF) / 255.0f,
45 ((argb >> 24) & 0xFF) / 255.0f);
46 }

48 std::string compactRouteName(uint32_t targetId, const std::string& targetName)
49 {
50 if (targetId == 0 || targetName == "Master" || targetName == "MASTER") {
51 return "M";
52 }
53 const std::string trackPrefix = "Track ";
54 if (targetName.rfind(trackPrefix, 0) == 0) {
55 return "T" + targetName.substr(trackPrefix.size());
56 }
57 const std::string lanePrefix = "Lane ";
58 if (targetName.rfind(lanePrefix, 0) == 0) {
59 return "L" + targetName.substr(lanePrefix.size());
60 }
61 if (targetName.size() <= 6) {
62 return targetName;
63 }
64 return targetName.substr(0, 6);
65 }

67 std::string buildStripRouteSummary(const Aestra::ChannelViewModel& channel)
68 {
69 if (channel.id == 0) {
70 return channel.routeName;
71 }
72
73 const bool busOnly = !channel.masterSendEnabled && channel.mainOutputId != 0;
74 int audibleSendCount = 0;
75 int sidechainSendCount = 0;
76 std::string audibleTarget;
77 std::string sidechainTarget;
78 std::unordered_set<uint32_t> audibleTargets;
79 std::unordered_set<uint32_t> sidechainTargets;
80
81 for (const auto& send : channel.sends) {
82 if (send.muted) continue;
83 if (send.sidechainOnly) {
84 ++sidechainSendCount;
85 sidechainTargets.insert(send.targetId);
86 if (sidechainTarget.empty()) {
87 sidechainTarget = compactRouteName(send.targetId, send.targetName);
88 }
89 continue;
90 }
91 ++audibleSendCount;
92 audibleTargets.insert(send.targetId);
93 if (audibleTarget.empty()) {
94 audibleTarget = compactRouteName(send.targetId, send.targetName);
95 }
96 }
97
98 std::ostringstream summary;
99 bool first = true;
100 auto appendToken = [&](const std::string& token) {
101 if (token.empty()) return;
102 if (!first) summary << " • ";
103 summary << token;
104 first = false;
105 };
106
107 if (busOnly) {
108 appendToken("Bus " + compactRouteName(channel.mainOutputId, channel.routeName));
109 }
110 if (audibleSendCount > 0) {
111 if (audibleTargets.size() == 1) {
112 appendToken("Snd " + audibleTarget);
113 } else {
114 appendToken("S+" + std::to_string(static_cast<int>(audibleTargets.size())));
115 }
116 }
117 if (sidechainSendCount > 0) {
118 if (sidechainTargets.size() == 1) {
119 appendToken("SC " + sidechainTarget);
120 } else {
121 appendToken("SC+" + std::to_string(static_cast<int>(sidechainTargets.size())));
122 }
123 }
124
125 return summary.str();
126 }

128 std::string buildFxStatus(const Aestra::ChannelViewModel& channel)
129 {
130 if (!channel.channel) {
131 return {};
132 }
133
134 bool hasSidechainSend = false;
135 for (const auto& send : channel.sends) {
136 if (!send.muted && send.sidechainOnly) {
137 hasSidechainSend = true;
138 break;
139 }
140 }
141
142 auto& chain = channel.channel->getEffectChain();
143 for (size_t i = 0; i < Aestra::Audio::EffectChain::MAX_SLOTS; ++i) {
144 auto plugin = chain.getPlugin(i);
145 auto comp = std::dynamic_pointer_cast<Aestra::Audio::Plugins::AestraComp>(plugin);
146 if (!comp) {
147 continue;
148 }
149
150 const float grDb = comp->getCurrentGainReductionDb();
151 if (grDb >= 0.1f) {
152 std::ostringstream out;
153 out.setf(std::ios::fixed);
154 out.precision(grDb >= 10.0f ? 0 : 1);
155 out << "GR " << grDb;
156 return out.str();
157 }
158
159 if (hasSidechainSend && channel.sidechainPeak > 0.001f) {
160 return "SC live";
161 }
162
163 return {};
164 }
165
166 return {};
167 }

169 std::string formatDuckGain(float gain, const std::string& sourceLabel)
170 {
171 const std::string prefix = sourceLabel.empty() ? "DUCK" : sourceLabel;
172 if (!std::isfinite(gain) || gain <= 0.0f) {
173 return prefix;
174 }
175 const float db = 20.0f * std::log10(std::clamp(gain, 1.0e-6f, 1.0f));
176 std::ostringstream out;
177 out.setf(std::ios::fixed);
178 out.precision(std::abs(db) >= 9.5f ? 0 : 1);
179 out << prefix << " " << db << "dB";
180 return out.str();
181 }

284 m_buttons->onSoloToggled = [this](bool soloed, NUIModifiers modifiers) {
285 if (!m_viewModel) return;
286 auto* channel = m_viewModel->getChannelById(m_channelId);
287 if (!channel || channel->id == 0) return;
288
289 // Solo Safe Logic (Ctrl + Click)
290 const bool isCtrl = (modifiers & NUIModifiers::Ctrl);
291 if (isCtrl) {
292 // Revert the local toggle because pure solo state shouldn't change
293 // But wait, the button row just toggled its internal visual state.
294 // For Solo Safe, we actually want to toggle a *different* visual state
295 // (maybe different color or icon?), but for now let's just update the logic.
296 // NOTE: UIMixerButtonRow tracks 'soloed' boolean. If we are setting 'solo safe',
297 // we should probably revert the 'soloed' state on the button if it wasn't already soloed.
298 // Or, ideally, Solo Safe is an independent state.
299 // Given the button row is simple, let's treat it as:
300 // Ctrl+Click -> Toggle Safe. Restore Solo button state to what it was.
301
302 // Revert button state visually (hacky but works for stateless widget)
303 m_buttons->setSoloed(!soloed);
304
305 // Toggle proper safe state
306 if (auto mc = channel->channel) {
307 bool newSafe = !mc->isSoloSafe();
308 mc->setSoloSafe(newSafe);
309 // Visual feedback? UIMixerStrip doesn't have a distinct 'Safe' icon yet.
310 // We might rely on the user knowing they did it, or add a small indicator later.
311 }
312 return;
313 }
314
315 // Exclusive solo: clear other solos first (matches playlist behavior).
316 if (soloed) {
317 const size_t count = m_viewModel->getChannelCount();
318 for (size_t i = 0; i < count; ++i) {
319 auto* other = m_viewModel->getChannelByIndex(i);
320 if (!other || other->id == channel->id) continue;
321 if (auto otherMC = other->channel) {
322 otherMC->setSolo(false);
323 }
324 other->soloed = false;
325 }
326 }
327
328 channel->soloed = soloed;
329 invalidateStaticCache();
330
331 if (auto mc = channel->channel) {
332 if (soloed && mc->isMuted()) {
333 mc->setMute(false);
334 channel->muted = false;
335 }
336 }
337
338 // Fire undo/redo callback
339 if (onSoloChanged) onSoloChanged(soloed);
340 };

445void UIMixerStrip::layoutChildren()
446{
447 auto bounds = getBounds();
448
449 const float contentX = bounds.x + SIDE_INSET;
450 const float contentW = std::max(1.0f, bounds.width - SIDE_INSET * 2.0f);
451 float y = bounds.y + PAD * 0.75f;
452
453 if (m_header) {
454 m_header->setBounds(contentX, y, contentW, HEADER_H);
455 y += HEADER_H + SECTION_GAP;
456 }
457
458 // Input Dropdown Removed from Strip Layout
459
460 // Three-knob row: Trim + Pan + Width (channel controls grouped together)
461 const bool showTrim = (m_trimKnob && m_trimKnob->isVisible());
462 const bool showPan = (m_panKnob && m_panKnob->isVisible());
463 const bool showWidth = (m_widthKnob && m_widthKnob->isVisible());
464 int visibleKnobCount = (showTrim ? 1 : 0) + (showPan ? 1 : 0) + (showWidth ? 1 : 0);
465 if (visibleKnobCount > 0) {
466 const float knobW = (contentW - GAP * (visibleKnobCount - 1)) / visibleKnobCount;
467 float knobX = contentX;
468 if (showTrim) {
469 m_trimKnob->setBounds(knobX, y, knobW, KNOB_H);
470 knobX += knobW + GAP;
471 }
472 if (showPan) {
473 m_panKnob->setBounds(knobX, y, knobW, KNOB_H);
474 knobX += knobW + GAP;
475 }
476 if (showWidth) {
477 m_widthKnob->setBounds(knobX, y, knobW, KNOB_H);
478 }
479 y += KNOB_H + SECTION_GAP;
480 }
481
482 // M/S/R buttons moved below knobs, above FX summary
483 const bool hasButtons = (m_buttons && m_buttons->isVisible());
484 if (hasButtons) {
485 m_buttons->setBounds(contentX, y, contentW, BUTTONS_H);
486 y += BUTTONS_H + 6.0f;
487 }
488
489 // FX Summary (Add Insert) below the channel controls
490 if (m_fxSummary && m_fxSummary->isVisible()) {
491 if (m_channelId == 0) {
492 y += 10.0f;
493 }
494 m_fxSummary->setBounds(contentX, y, contentW, FX_H);
495 y += FX_H + SECTION_GAP;
496 }
497
498 const bool hasFooter = (m_footer && m_footer->isVisible());
499 const float footerH = hasFooter ? FOOTER_H : 0.0f;
500 const float footerY = bounds.y + bounds.height - PAD - footerH;
501 if (hasFooter) {
502 m_footer->setBounds(contentX, footerY, contentW, FOOTER_H);
503 }
504
505 const float contentY = y;
506 const float contentH = std::max(1.0f, footerY - y);
507
508 const float meterW = (m_channelId == 0) ? MASTER_METER_W : METER_W;
509
510 const float availableH = std::max(1.0f, contentH - PAD);
511 const float meterH = availableH * 0.62f;
512 const float meterY = footerY - GAP - meterH;
513 const float meterX = contentX + 2.0f;
514
515 if (m_meter) {
516 m_meter->setBounds(meterX, meterY, meterW, meterH);
517 }
518
519 if (m_fader) {
520 const float faderX = meterX + meterW + GAP;
521 const float faderW = std::max(16.0f, (contentX + contentW) - faderX);
522 m_fader->setBounds(faderX, meterY, faderW, meterH);
523 }
524}

533void UIMixerStrip::onUpdate(double deltaTime)
534{
535 (void)deltaTime;
536 if (!m_viewModel) {
537 updateChildren(deltaTime);
538 return;
539 }
540
541 auto* channel = m_viewModel->getChannelById(m_channelId);
542 if (!channel) {
543 updateChildren(deltaTime);
544 return;
545 }
546
547 const bool selected = (m_viewModel->getSelectedChannelId() == static_cast<int32_t>(m_channelId));
548 if (m_cachedSelected != selected) {
549 m_cachedSelected = selected;
550 invalidateStaticCache();
551 }
552
553 // Input Dropdown Logic Removed
554
555 if (m_cachedMuted != channel->muted) {
556 m_cachedMuted = channel->muted;
557 invalidateStaticCache();
558 if (m_meter) {
559 m_meter->setDimmed(channel->muted);
560 }
561 }
562 if (m_cachedSoloed != channel->soloed) {
563 m_cachedSoloed = channel->soloed;
564 invalidateStaticCache();
565 }
566 if (m_cachedArmed != channel->armed) {
567 m_cachedArmed = channel->armed;
568 invalidateStaticCache();
569 }
570 if (m_cachedMonitored != channel->monitored) {
571 m_cachedMonitored = channel->monitored;
572 invalidateStaticCache();
573 }
574
575 if (m_header) {
576 if (m_cachedName != channel->name) {
577 m_cachedName = channel->name;
578 invalidateStaticCache();
579 }
580 m_header->setTrackName(channel->name);
581 const std::string stripRoute = buildStripRouteSummary(*channel);
582 if (m_cachedRoute != stripRoute) {
583 m_cachedRoute = stripRoute;
584 invalidateStaticCache();
585 }
586 m_header->setRouteName(stripRoute);
587 if (m_cachedTrackColorArgb != static_cast<uint32_t>(channel->trackColorIndex + 1)) {
588 m_cachedTrackColorArgb = static_cast<uint32_t>(channel->trackColorIndex + 1);
589 invalidateStaticCache();
590
591 // Thread the track colour through the strip's accent controls so each
592 // channel reads as its own instrument. Master (id 0) keeps the neutral
593 // primary accent, and unset tracks fall back to the widget defaults.
594 if (m_channelId != 0 && channel->trackColorIndex >= 0) {
595 const NUIColor accent = trackColorFromIndex(channel->trackColorIndex);
596 if (m_fader) m_fader->setAccentColor(accent);
597 if (m_trimKnob) m_trimKnob->setAccentColor(accent);
598 if (m_panKnob) m_panKnob->setAccentColor(accent);
599 if (m_widthKnob) m_widthKnob->setAccentColor(accent);
600 }
601 }
602 m_header->setTrackColorIndex(channel->trackColorIndex);
603 m_header->setSelected(selected);
604 }
605
606 if (m_buttons && m_buttons->isVisible()) {
607 m_buttons->setMuted(channel->muted);
608 m_buttons->setSoloed(channel->soloed);
609 m_buttons->setArmed(channel->armed);
610 }
611
612 if (m_trimKnob && m_trimKnob->isVisible() && !m_trimKnob->isDragging()) {
613 const bool hovered = m_trimKnob->isHovered();
614 if (m_cachedTrimHovered != hovered) {
615 m_cachedTrimHovered = hovered;
616 invalidateStaticCache();
617 }
618 if (std::abs(m_cachedTrimDb - channel->trimDb) > 1e-3f) {
619 m_cachedTrimDb = channel->trimDb;
620 invalidateStaticCache();
621 }
622 m_trimKnob->setValue(channel->trimDb);
623 }
624
625 if (m_panKnob && m_panKnob->isVisible() && !m_panKnob->isDragging()) {
626 const bool hovered = m_panKnob->isHovered();
627 if (m_cachedPanHovered != hovered) {
628 m_cachedPanHovered = hovered;
629 invalidateStaticCache();
630 }
631 if (std::abs(m_cachedPan - channel->pan) > 1e-4f) {
632 m_cachedPan = channel->pan;
633 invalidateStaticCache();
634 }
635 m_panKnob->setValue(channel->pan);
636 }
637
638 if (m_widthKnob && m_widthKnob->isVisible() && !m_widthKnob->isDragging()) {
639 const bool hovered = m_widthKnob->isHovered();
640 if (m_cachedWidthHovered != hovered) {
641 m_cachedWidthHovered = hovered;
642 invalidateStaticCache();
643 }
644 if (std::abs(m_cachedWidth - channel->width) > 1e-4f) {
645 m_cachedWidth = channel->width;
646 invalidateStaticCache();
647 }
648 m_widthKnob->setValue(channel->width);
649 }
650
651 if (m_fxSummary && m_fxSummary->isVisible()) {
652 if (m_cachedFxCount != channel->fxCount) {
653 m_cachedFxCount = channel->fxCount;
654 invalidateStaticCache();
655 m_fxSummary->setFxCount(channel->fxCount);
656 }
657 const std::string fxStatus = buildFxStatus(*channel);
658 if (m_cachedFxStatus != fxStatus) {
659 m_cachedFxStatus = fxStatus;
660 invalidateStaticCache();
661 m_fxSummary->setStatusText(fxStatus);
662 }
663 }
664
665 if (m_footer && m_footer->isVisible()) {
666 m_footer->setTrackNumber(m_trackNumber);
667 }
668
669 if (m_meter) {
670 m_meter->setLevels(channel->smoothedPeakL, channel->smoothedPeakR);
671 m_meter->setRmsLevels(channel->smoothedRmsL, channel->smoothedRmsR);
672 m_meter->setPeakOverlay(channel->envPeakL, channel->envPeakR);
673 m_meter->setPeakHold(channel->peakHoldL, channel->peakHoldR);
674 m_meter->setClipLatch(channel->clipLatchL, channel->clipLatchR);
675
676 // Pass correlation to meter (will only be drawn if enabled)
677 m_meter->setCorrelation(channel->correlation);
678
679 // Pass LUFS only to master meter (ID 0)
680 if (m_channelId == 0) {
681 m_meter->setIntegratedLufs(channel->integratedLufs);
682 }
683 }
684
685 if (m_fader && !m_fader->isDragging()) {
686 const bool hovered = m_fader->isHovered();
687 if (m_cachedFaderHovered != hovered) {
688 m_cachedFaderHovered = hovered;
689 invalidateStaticCache();
690 }
691
692 if (std::abs(m_cachedFaderDb - channel->faderGainDb) > 1e-3f) {
693 m_cachedFaderDb = channel->faderGainDb;
694 invalidateStaticCache();
695 }
696 m_fader->setValueDb(channel->faderGainDb);
697 }
698
699 updateChildren(deltaTime);

806bool UIMixerStrip::onMouseEvent(const NUIMouseEvent& event)
807{
808 if (m_meter && m_viewModel && event.pressed &&
809 (event.button == NUIMouseButton::Left || event.button == NUIMouseButton::Right) &&
810 getBounds().contains(event.position)) {
811 const auto stripBounds = getBounds();
812 const auto meterBounds = m_meter->getBounds();
813 auto* channel = m_viewModel->getChannelById(m_channelId);
814 std::ostringstream hitLog;
815 hitLog << "[MixerClipHit] channel=" << m_channelId
816 << " button=" << static_cast<int>(event.button)
817 << " click=(" << event.position.x << "," << event.position.y << ")"
818 << " strip=(" << stripBounds.x << "," << stripBounds.y << ","
819 << stripBounds.width << "," << stripBounds.height << ")"
820 << " meter=(" << meterBounds.x << "," << meterBounds.y << ","
821 << meterBounds.width << "," << meterBounds.height << ")"
822 << " inMeter=" << (meterBounds.contains(event.position) ? 1 : 0)
823 << " latchL=" << (channel ? (channel->clipLatchL ? 1 : 0) : -1)
824 << " latchR=" << (channel ? (channel->clipLatchR ? 1 : 0) : -1);
825 AESTRA_LOG_INFO(hitLog.str());
826 }
827
828 if (m_meter && m_viewModel && event.pressed &&
829 (event.button == NUIMouseButton::Left || event.button == NUIMouseButton::Right) &&
830 m_meter->getBounds().contains(event.position)) {
831 auto* channel = m_viewModel->getChannelById(m_channelId);
832 if (channel && (channel->clipLatchL || channel->clipLatchR)) {
833 m_viewModel->clearClipLatch(channel->id);
834 if (m_meterSnapshots) {
835 m_meterSnapshots->clearClip(channel->slotIndex);
836 }
837 m_meter->setClipLatch(false, false);
838 invalidateStaticCache();
839 }
840 return true;
841 }
842
843 bool handledSelection = false;
844 if (m_viewModel && event.pressed && event.button == NUIMouseButton::Left) {
845 if (getBounds().contains(event.position)) {
846 m_viewModel->setSelectedChannelId(static_cast<int32_t>(m_channelId));
847 handledSelection = true;
848 }
849 }
850
851 const bool handledByChildren = NUIComponent::onMouseEvent(event);
852 return handledSelection || handledByChildren;
853}
