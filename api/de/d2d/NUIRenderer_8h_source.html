281 virtual FontMetrics getFontMetrics(float fontSize) const {
282 FontMetrics metrics;
283 // Fallback approximation that matches the renderer's baseline compensation defaults.
284 metrics.ascent = fontSize * 0.8f;
285 metrics.descent = fontSize * 0.2f;
286 metrics.lineHeight = metrics.ascent + metrics.descent;
287 metrics.capHeight = fontSize * 0.7f;
288 return metrics;
289 }
290
301 float calculateTextBaselineY(const NUIRect& rect, float fontSize) const {
302 const FontMetrics metrics = getFontMetrics(fontSize);
303 const float lineHeight = (metrics.lineHeight > 0.0f) ? metrics.lineHeight : fontSize;
304 const float ascent = (metrics.ascent > 0.0f) ? metrics.ascent : (fontSize * 0.8f);
305 const float topY = rect.y + (rect.height - lineHeight) * 0.5f;
306 return topY + ascent;
307 }
308
318 float calculateTextY(const NUIRect& rect, float fontSize) const {
319 const FontMetrics metrics = getFontMetrics(fontSize);
320 const float lineHeight = (metrics.lineHeight > 0.0f) ? metrics.lineHeight : fontSize;
321 return rect.y + (rect.height - lineHeight) * 0.5f;
322 }
323
335 float calculateOpticalTextY(float centreY, float fontSize) const {
336 const FontMetrics metrics = getFontMetrics(fontSize);
337 const float ascent = (metrics.ascent > 0.0f) ? metrics.ascent : (fontSize * 0.8f);
338 const float capHeight = (metrics.capHeight > 0.0f) ? metrics.capHeight : (fontSize * 0.7f);
339 return centreY + capHeight * 0.5f - ascent;
340 }
342 float calculateOpticalTextY(const NUIRect& rect, float fontSize) const {
343 return calculateOpticalTextY(rect.y + rect.height * 0.5f, fontSize);
344 }
