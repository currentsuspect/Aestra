213 virtual FontMetrics getFontMetrics(float fontSize) const {
214 FontMetrics metrics;
215 // Fallback approximation that matches the renderer's baseline compensation defaults.
216 metrics.ascent = fontSize * 0.8f;
217 metrics.descent = fontSize * 0.2f;
218 metrics.lineHeight = metrics.ascent + metrics.descent;
219 return metrics;
220 }
221
230 float calculateTextBaselineY(const NUIRect& rect, float fontSize) const {
231 const FontMetrics metrics = getFontMetrics(fontSize);
232 const float lineHeight = (metrics.lineHeight > 0.0f) ? metrics.lineHeight : fontSize;
233 const float ascent = (metrics.ascent > 0.0f) ? metrics.ascent : (fontSize * 0.8f);
234 const float topY = rect.y + (rect.height - lineHeight) * 0.5f;
235 return topY + ascent;
236 }
237
246 float calculateTextY(const NUIRect& rect, float fontSize) const {
247 const FontMetrics metrics = getFontMetrics(fontSize);
248 const float lineHeight = (metrics.lineHeight > 0.0f) ? metrics.lineHeight : fontSize;
249 return rect.y + (rect.height - lineHeight) * 0.5f;
250 }
