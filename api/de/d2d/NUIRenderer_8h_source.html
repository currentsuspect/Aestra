221 virtual FontMetrics getFontMetrics(float fontSize) const {
222 FontMetrics metrics;
223 // Fallback approximation that matches the renderer's baseline compensation defaults.
224 metrics.ascent = fontSize * 0.8f;
225 metrics.descent = fontSize * 0.2f;
226 metrics.lineHeight = metrics.ascent + metrics.descent;
227 return metrics;
228 }
229
238 float calculateTextBaselineY(const NUIRect& rect, float fontSize) const {
239 const FontMetrics metrics = getFontMetrics(fontSize);
240 const float lineHeight = (metrics.lineHeight > 0.0f) ? metrics.lineHeight : fontSize;
241 const float ascent = (metrics.ascent > 0.0f) ? metrics.ascent : (fontSize * 0.8f);
242 const float topY = rect.y + (rect.height - lineHeight) * 0.5f;
243 return topY + ascent;
244 }
245
254 float calculateTextY(const NUIRect& rect, float fontSize) const {
255 const FontMetrics metrics = getFontMetrics(fontSize);
256 const float lineHeight = (metrics.lineHeight > 0.0f) ? metrics.lineHeight : fontSize;
257 return rect.y + (rect.height - lineHeight) * 0.5f;
258 }
