220 virtual FontMetrics getFontMetrics(float fontSize) const {
221 FontMetrics metrics;
222 // Fallback approximation that matches the renderer's baseline compensation defaults.
223 metrics.ascent = fontSize * 0.8f;
224 metrics.descent = fontSize * 0.2f;
225 metrics.lineHeight = metrics.ascent + metrics.descent;
226 return metrics;
227 }
228
237 float calculateTextBaselineY(const NUIRect& rect, float fontSize) const {
238 const FontMetrics metrics = getFontMetrics(fontSize);
239 const float lineHeight = (metrics.lineHeight > 0.0f) ? metrics.lineHeight : fontSize;
240 const float ascent = (metrics.ascent > 0.0f) ? metrics.ascent : (fontSize * 0.8f);
241 const float topY = rect.y + (rect.height - lineHeight) * 0.5f;
242 return topY + ascent;
243 }
244
253 float calculateTextY(const NUIRect& rect, float fontSize) const {
254 const FontMetrics metrics = getFontMetrics(fontSize);
255 const float lineHeight = (metrics.lineHeight > 0.0f) ? metrics.lineHeight : fontSize;
256 return rect.y + (rect.height - lineHeight) * 0.5f;
257 }
