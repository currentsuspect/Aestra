93 void copyFloats(uint64_t firstSample, uint64_t count, float* out) const {
94 const uint64_t total = storedSamples();
95 const uint64_t avail = firstSample < total ? std::min<uint64_t>(count, total - firstSample) : 0;
96 switch (storage) {
97 case Storage::Pcm16: {
98 const int16_t* src = pcm16.data() + firstSample;
99 for (uint64_t i = 0; i < avail; ++i)
100 out[i] = static_cast<float>(src[i]) * (1.0f / 32768.0f);
101 break;
102 }
103 case Storage::Pcm24: {
104 const uint8_t* src = pcm24.data() + firstSample * 3;
105 for (uint64_t i = 0; i < avail; ++i)
106 out[i] = pcm24ToFloat(src + i * 3);
107 break;
108 }
109 default:
110 if (avail > 0)
111 std::memcpy(out, interleavedData.data() + firstSample, avail * sizeof(float));
112 break;
113 }
114 for (uint64_t i = avail; i < count; ++i)
115 out[i] = 0.0f;
116 }

129 float peakAbs() const {
130 float peak = 0.0f;
131 if (!isCompact()) {
132 for (const float v : interleavedData)
133 if (std::isfinite(v))
134 peak = std::max(peak, std::abs(v));
135 return peak;
136 }
137 float slice[4096];
138 const uint64_t total = storedSamples();
139 for (uint64_t at = 0; at < total; at += 4096) {
140 const uint64_t n = std::min<uint64_t>(4096, total - at);
141 copyFloats(at, n, slice);
142 for (uint64_t i = 0; i < n; ++i)
143 peak = std::max(peak, std::abs(slice[i])); // PCM values are always finite
144 }

164 bool tryCompact() {
165 if (isCompact())
166 return true;
167 const size_t n = interleavedData.size();
168 if (n == 0)
169 return false;
170 const float* f = interleavedData.data();
171
172 // Exactness test per sample: x * 2^bits is an exact integer in range. Scaling by a power of two is
173 // exact in float32, so no rounding is involved; the loop is branch-free per block so it vectorizes,
174 // and a failing file is rejected within its first block.
175 const auto fitsGrid = [&](float scale, int32_t lo, int32_t hi) {
176 constexpr size_t kBlock = 4096;
177 for (size_t at = 0; at < n; at += kBlock) {
178 const size_t end = std::min(n, at + kBlock);
179 bool ok = true;
180 for (size_t i = at; i < end; ++i) {
181 const float t = f[i] * scale;
182 const int32_t k = static_cast<int32_t>(t);
183 ok &= (static_cast<float>(k) == t) & (k >= lo) & (k <= hi);
184 }
185 if (!ok)
186 return false;
187 }
188 return true;
189 };
190
191 if (fitsGrid(32768.0f, -32768, 32767)) {
192 std::vector<int16_t> packed(n);
193 for (size_t i = 0; i < n; ++i)
194 packed[i] = static_cast<int16_t>(static_cast<int32_t>(f[i] * 32768.0f));
195 pcm16 = std::move(packed);
196 std::vector<float>().swap(interleavedData);
197 storage = Storage::Pcm16;
198 return true;
199 }
200
201 if (!fitsGrid(8388608.0f, -8388608, 8388607))
202 return false;
203 std::vector<uint8_t> packed(n * 3);
204 for (size_t i = 0; i < n; ++i) {
205 const int32_t k = static_cast<int32_t>(f[i] * 8388608.0f);
206 packed[i * 3] = static_cast<uint8_t>(k & 0xFF);
207 packed[i * 3 + 1] = static_cast<uint8_t>((k >> 8) & 0xFF);
208 packed[i * 3 + 2] = static_cast<uint8_t>((k >> 16) & 0xFF);
209 }
210 pcm24 = std::move(packed);
211 std::vector<float>().swap(interleavedData);
212 storage = Storage::Pcm24;
213 return true;
214 }
215
216private:
217 static float pcm24ToFloat(const uint8_t* p) {
218 int32_t k = static_cast<int32_t>(p[0]) | (static_cast<int32_t>(p[1]) << 8) | (static_cast<int32_t>(p[2]) << 16);
219 k = (k << 8) >> 8; // sign-extend 24 -> 32
220 return static_cast<float>(k) * (1.0f / 8388608.0f);
221 }
