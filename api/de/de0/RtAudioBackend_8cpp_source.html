44std::vector<AudioDeviceInfo> RtAudioBackend::getDevices() {
45 std::vector<AudioDeviceInfo> devices;
46
47 try {
48 std::vector<unsigned int> deviceIds = m_rtAudio->getDeviceIds();
49 std::cout << "RtAudioBackend::getDevices: Found " << deviceIds.size() << " device IDs" << std::endl;
50
51 for (unsigned int id : deviceIds) {
52 try {
53 RtAudio::DeviceInfo rtInfo = m_rtAudio->getDeviceInfo(id);
54
55 // Skip invalid devices
56 if (rtInfo.outputChannels == 0 && rtInfo.inputChannels == 0) {
57 std::cout << " Device " << id << ": Skipping (no I/O channels)" << std::endl;
58 continue;
59 }
60
61 AudioDeviceInfo info;
62 info.id = id;
63 info.name = rtInfo.name;
64 info.maxInputChannels = rtInfo.inputChannels;
65 info.maxOutputChannels = rtInfo.outputChannels;
66 info.supportedSampleRates = rtInfo.sampleRates;
67 info.preferredSampleRate = rtInfo.preferredSampleRate;
68 info.isDefaultInput = rtInfo.isDefaultInput;
69 info.isDefaultOutput = rtInfo.isDefaultOutput;
70
71 std::cout << " Device " << id << ": " << rtInfo.name << " (out:" << rtInfo.outputChannels
72 << " in:" << rtInfo.inputChannels << ")" << std::endl;
73
74 devices.push_back(info);
75 } catch (const std::exception& e) {
76 std::cerr << " Device " << id << ": Exception getting info: " << e.what() << std::endl;
77 continue;
78 }
79 }
80 } catch (const std::exception& e) {
81 std::cerr << "RtAudioBackend::getDevices: Exception: " << e.what() << std::endl;
82 } catch (...) {
83 std::cerr << "RtAudioBackend::getDevices: Unknown exception" << std::endl;
84 }
85
86 std::cout << "RtAudioBackend::getDevices: Returning " << devices.size() << " valid devices" << std::endl;
87 return devices;
88}

90bool RtAudioBackend::openStream(const AudioStreamConfig& config, AudioCallback callback, void* userData) {
91 if (m_rtAudio->isStreamOpen()) {
92 closeStream();
93 }
94
95 m_userCallback.store(callback, std::memory_order_relaxed);
96 m_userData.store(userData, std::memory_order_relaxed);
97
98 RtAudio::StreamParameters outputParams;
99 outputParams.deviceId = config.deviceId;
100 outputParams.nChannels = config.numOutputChannels;
101 outputParams.firstChannel = 0;
102
103 RtAudio::StreamParameters* inputParams = nullptr;
104 RtAudio::StreamParameters inputParamsData;
105 if (config.numInputChannels > 0 && config.inputDeviceId != 0) {
106 // Guard against monitor/loopback devices — they capture system playback,
107 // causing recording to bleed music from other tracks.
108 if (isMonitorDevice(config.inputDeviceId)) {
109 Log::warning("RtAudioBackend: Refusing monitor/loopback input device — "
110 "this would cause recording bleed. Skipping input capture.");
111 } else {
112 inputParamsData.deviceId = config.inputDeviceId;
113 inputParamsData.nChannels = config.numInputChannels;
114 inputParamsData.firstChannel = 0;
115 inputParams = &inputParamsData;
116 }
117 }
118
119 unsigned int bufferFrames = config.bufferSize;
120 unsigned int sampleRate = config.sampleRate;
121
122 std::cout << "RtAudioBackend::openStream: About to call m_rtAudio->openStream" << std::endl;
123 std::cout << " outputParams.deviceId: " << outputParams.deviceId << std::endl;
124 std::cout << " outputParams.nChannels: " << outputParams.nChannels << std::endl;
125 std::cout << " sampleRate: " << sampleRate << std::endl;
126 std::cout << " bufferFrames: " << bufferFrames << std::endl;
127 std::cout.flush();
128
129 RtAudioErrorType error = m_rtAudio->openStream(&outputParams, inputParams, RTAUDIO_FLOAT32, sampleRate,
130 &bufferFrames, &RtAudioBackend::rtAudioCallback, this);
131
132 std::cout << "RtAudioBackend::openStream: m_rtAudio->openStream returned error code: " << error << std::endl;
133 std::cout << " Final bufferFrames: " << bufferFrames << std::endl;
134 std::cout << " Stream open: " << (m_rtAudio->isStreamOpen() ? "true" : "false") << std::endl;
135 std::cout.flush();
136
137 if (error == RTAUDIO_NO_ERROR) {
138 m_bufferSize.store(bufferFrames, std::memory_order_relaxed); // Store actual buffer size returned by RtAudio
139 }
140
141 return (error == RTAUDIO_NO_ERROR);
142}

168 RtAudioErrorType error = m_rtAudio->startStream();

208int RtAudioBackend::rtAudioCallback(void* outputBuffer, void* inputBuffer, unsigned int numFrames, double streamTime,
209 RtAudioStreamStatus status, void* userData) {
210 (void)status; // Unused parameter
211
212 RtAudioBackend* backend = static_cast<RtAudioBackend*>(userData);
213
214 AudioCallback userCallback = backend->m_userCallback.load(std::memory_order_relaxed);
215 void* callbackUserData = backend->m_userData.load(std::memory_order_relaxed);
216
217 if (userCallback) {
218 int result = userCallback(static_cast<float*>(outputBuffer), static_cast<const float*>(inputBuffer), numFrames,
219 streamTime, callbackUserData);
220 backend->m_stats.callbackCount++;
221 return result;
222 }
223
224 return 0;
225}
