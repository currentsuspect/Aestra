20std::vector<AutomationRegion> visibleRegions(const AutomationCurve& curve) {
21 if (!curve.regions.empty())
22 return curve.regions;
23 if (curve.points.empty())
24 return {};
25 return {{1, curve.points.front().beat, curve.points.back().beat}};

28float screenX(double beat, const AutomationClipLayer::Geometry& g) {
29 return g.gridStartX + static_cast<float>(beat) * g.pixelsPerBeat - g.scrollOffset;
30}

32double beatAt(float x, const AutomationClipLayer::Geometry& g) {
33 return std::max(0.0, static_cast<double>(x - g.gridStartX + g.scrollOffset) / g.pixelsPerBeat);
34}

36AestraUI::NUIRect headerRect(const AutomationRegion& r, const AutomationClipLayer::Geometry& g) {
37 const float x0 = screenX(r.startBeat, g) - 3.0f;
38 const float x1 = std::max(screenX(r.endBeat, g) + 3.0f, x0 + 8.0f);
39 return {x0, g.bounds.y, x1 - x0, AutomationClipLayer::kHeaderHeight};
40}

43AestraUI::NUIRect gripRect(const AutomationRegion& r, const AutomationClipLayer::Geometry& g) {
44 return {screenX(r.startBeat, g) + 6.0f, g.bounds.y, 16.0f, AutomationClipLayer::kHeaderHeight};
45}

48bool pointNear(const PlaylistLane& lane, const AestraUI::NUIPoint& p, const AutomationClipLayer::Geometry& g) {
49 for (const auto& curve : lane.automationCurves)
50 for (const auto& pt : curve.points) {
51 const float x = screenX(pt.beat, g);
52 const float y = g.bounds.y + (1.0f - static_cast<float>(automationHeightOfValue(
53 curve.getAutomationTarget(), pt.value))) * g.bounds.height;
54 if (std::hypot(x - p.x, y - p.y) <= 12.0f)
55 return true;
56 }
57 return false;
58}

74void AutomationClipLayer::render(AestraUI::NUIRenderer& renderer, const PlaylistLane& lane, int editedCurve,
75 const Geometry& g) const {
76 auto& theme = AestraUI::NUIThemeManager::getInstance();
77 const auto accent = theme.getColor("accentCyan");
78 for (size_t c = 0; c < lane.automationCurves.size(); ++c) {
79 const bool edited = static_cast<int>(c) == editedCurve;
80 for (const auto& region : visibleRegions(lane.automationCurves[c])) {
81 const bool selected = static_cast<int>(c) == m_selectedCurve && region.id == m_selectedRegion;
82 const AestraUI::NUIRect head = headerRect(region, g);
83 const AestraUI::NUIRect body(head.x, g.bounds.y, head.width, g.bounds.height);
84 renderer.fillRect(body, accent.withAlpha(edited ? 0.07f : 0.03f));
85 renderer.fillRect(head, accent.withAlpha(selected ? 0.40f : (edited ? 0.20f : 0.10f)));
86 const AestraUI::NUIRect grip = gripRect(region, g);
87 renderer.fillRect(grip, accent.withAlpha(selected ? 0.90f : (edited ? 0.60f : 0.35f)));
88 for (int dot = 0; dot < 3; ++dot) // three dots: a grip
89 renderer.fillRect({grip.x + 4.0f + 4.0f * static_cast<float>(dot), grip.y + 5.0f, 2.0f, 2.0f},
90 theme.getColor("background").withAlpha(0.7f));
91 renderer.strokeRect(body, selected ? 1.5f : 1.0f,
92 selected ? theme.getColor("primary") : accent.withAlpha(edited ? 0.55f : 0.25f));
93 }
94 }

97AutomationClipLayer::Hit AutomationClipLayer::headerAt(const PlaylistLane& lane, const AestraUI::NUIPoint& p,
98 const Geometry& g, bool gripOnly) const {
99 if (pointNear(lane, p, g))
100 return {};
101 // Regions of different curves may share a header strip; the one drawn last (highest curve index) is on top.
102 for (int c = static_cast<int>(lane.automationCurves.size()) - 1; c >= 0; --c) {
103 for (const auto& region : visibleRegions(lane.automationCurves[static_cast<size_t>(c)]))
104 if ((gripOnly ? gripRect(region, g) : headerRect(region, g)).contains(p))
105 return {c, region.id};
106 }
107 return {};
108}

110void AutomationClipLayer::commit(TrackManager& tm, PlaylistLaneID laneId, std::vector<AutomationCurve> before,
111 const char* name, const Hooks& hooks) {
112 auto* lane = tm.getPlaylistModel().getLane(laneId);
113 if (!lane)
114 return;
115 for (auto& curve : lane->automationCurves)
116 R::retighten(curve);
117 if (!automationCurvesEqual(before, lane->automationCurves)) {
118 tm.getCommandHistory().pushAndExecute(std::make_shared<EditAutomationCurvesCommand>(
119 tm, laneId, std::move(before), lane->automationCurves, name));
120 tm.requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged);
121 tm.markModified();
122 }
123 if (hooks.changed)
124 hooks.changed();
125}

127bool AutomationClipLayer::onMouse(const AestraUI::NUIMouseEvent& event, TrackManager& tm, PlaylistLaneID laneId,
128 const Geometry& g, const Hooks& hooks, int& editedCurve) {
129 auto* lane = tm.getPlaylistModel().getLane(laneId);
130 if (!lane)
131 return false;
132 const auto snap = [&](double b) { return hooks.snapBeat ? hooks.snapBeat(b) : b; };
133
134 if (m_dragging) {
135 if (event.released && event.button == AestraUI::NUIMouseButton::Left) {
136 m_dragging = false;
137 if (hooks.captureMouse)
138 hooks.captureMouse(false);
139 // The live edit is already in the lane; the release only records it.
140 auto before = std::move(m_before);
141 m_before.clear();
142 if (m_copy) {
143 // The copy becomes the selection: the newest region of the dragged curve.
144 const auto& curve = lane->automationCurves[static_cast<size_t>(m_dragHit.curve)];
145 if (!curve.regions.empty() && curve.regions.size() > before[static_cast<size_t>(m_dragHit.curve)].regions.size())
146 m_selectedRegion = R::nextId(curve) - 1;
147 }
148 commit(tm, laneId, std::move(before), m_copy ? "Copy Automation Clip" : "Move Automation Clip", hooks);
149 return true;
150 }
151 if (!event.pressed && !event.released) {
152 // Dragging: edit a copy of the curves from the press, and keep the last placement that was allowed.
153 const double delta = snap(m_origStart + beatAt(event.position.x, g) - m_grabBeat) - m_origStart;
154 auto trial = m_before;
155 auto& curve = trial[static_cast<size_t>(m_dragHit.curve)];
156 const bool ok = m_copy ? (std::fabs(delta) < R::kEps || R::copy(curve, m_dragHit.region, delta).has_value())
157 : R::move(curve, m_dragHit.region, delta);
158 if (ok) {
159 lane->automationCurves = std::move(trial);
160 if (hooks.changed)
161 hooks.changed();
162 }
163 return true;
164 }
165 return true;
166 }
167
168 if (!event.pressed)
169 return false;
170 const bool right = event.button == AestraUI::NUIMouseButton::Right;
171 const Hit hit = headerAt(*lane, event.position, g, /*gripOnly=*/!right);
172 if (hit.curve < 0)
173 return false;
174
175 m_selectedCurve = hit.curve;
176 m_selectedRegion = hit.region;
177 editedCurve = hit.curve;
178 if (event.button == AestraUI::NUIMouseButton::Right) {
179 showMenu(tm, laneId, hit, beatAt(event.position.x, g), event.position, hooks);
180 return true;
181 }
182 if (event.button != AestraUI::NUIMouseButton::Left)
183 return false;
184
185 const auto& curve = lane->automationCurves[static_cast<size_t>(hit.curve)];
186 const auto regions = visibleRegions(curve);
187 const auto it = std::find_if(regions.begin(), regions.end(), [&](const auto& r) { return r.id == hit.region; });
188 m_dragging = true;
189 m_copy = (event.modifiers & AestraUI::NUIModifiers::Ctrl) != 0;
190 m_dragHit = hit;
191 m_grabBeat = beatAt(event.position.x, g);
192 m_origStart = it != regions.end() ? it->startBeat : 0.0;
193 m_before = lane->automationCurves;
194 if (hooks.captureMouse)
195 hooks.captureMouse(true);
196 if (hooks.changed)
197 hooks.changed();
198 return true;
199}

201void AutomationClipLayer::showMenu(TrackManager& tm, PlaylistLaneID laneId, Hit hit, double beat,
202 const AestraUI::NUIPoint& at, const Hooks& hooks) {
203 closeMenu();
204 m_menu = std::make_shared<AestraUI::NUIContextMenu>();
205 TrackManager* manager = &tm;
206
207 // Each action edits the lane's curve, then records ONE undo step for it.
208 const auto edit = [manager, laneId, hit, hooks, this](const char* name, auto&& op) {
209 auto* lane = manager->getPlaylistModel().getLane(laneId);
210 if (!lane || hit.curve < 0 || hit.curve >= static_cast<int>(lane->automationCurves.size()))
211 return;
212 auto before = lane->automationCurves;
213 op(lane->automationCurves[static_cast<size_t>(hit.curve)]);
214 commit(*manager, laneId, std::move(before), name, hooks);
215 };
216
217 m_menu->addItem("Duplicate Clip", [edit, hit]() {
218 edit("Duplicate Automation Clip", [&](AutomationCurve& curve) {
219 R::materialize(curve);
220 const auto* region = R::find(curve, hit.region);
221 if (!region)
222 return;
223 const double length = std::max(region->endBeat - region->startBeat, 0.25);
224 // The first free slot to the right, a clip-length at a time.
225 for (int step = 1; step <= 64; ++step)
226 if (R::copy(curve, hit.region, length * step).has_value())
227 return;
228 });
229 });
230 m_menu->addItem("Split Here", [edit, hit, beat]() {
231 edit("Split Automation Clip", [&](AutomationCurve& curve) { R::split(curve, hit.region, beat); });
232 });
233 m_menu->addSeparator();
234 m_menu->addItem("Delete Clip", [edit, hit, this]() {
235 edit("Delete Automation Clip", [&](AutomationCurve& curve) { R::remove(curve, hit.region); });
236 m_selectedCurve = -1;
237 m_selectedRegion = 0;
238 });
239
240 AestraUI::NUIComponent* root = &m_owner;
241 while (root->getParent())
242 root = root->getParent();
243 root->addChild(m_menu);
244 m_menu->setOwner(m_owner.weak_from_this());
245 m_menu->showAt(at);
246 root->repaint();
247}
