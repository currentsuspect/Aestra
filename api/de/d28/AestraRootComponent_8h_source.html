110 void updateMembershipBadge(double deltaTime) {
111 if (!m_rootCustomWindow || !m_rootCustomWindow->getTitleBar()) {
112 return;
113 }
114 if (m_membershipBadgeRefreshSeconds < kMembershipBadgeRefreshIntervalSeconds) {
115 m_membershipBadgeRefreshSeconds +=
116 std::max(std::min(deltaTime, kMembershipBadgeRefreshIntervalSeconds), 0.0);
