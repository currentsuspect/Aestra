45// Wakes the worker the moment process() hands it a block, instead of the worker finding it on its next 1 ms nap
46// (V82-H7 follow-up). post() is one atomic and, only when the worker sleeps, one wake syscall: safe from the render thread.
47class WorkerWake {
48public:
49 WorkerWake() {
50#ifdef _WIN32
51 m_sem = CreateSemaphoreA(nullptr, 0, 1, nullptr);
52#elif defined(__APPLE__)
53 m_sem = dispatch_semaphore_create(0);
54#else
55 m_ok = sem_init(&m_sem, 0, 0) == 0;
56#endif
57 }
58 ~WorkerWake() {
59#ifdef _WIN32
60 if (m_sem)
61 CloseHandle(m_sem);
62#elif defined(__APPLE__)
63 if (m_sem)
64 dispatch_release(m_sem);
65#else
66 if (m_ok)
67 sem_destroy(&m_sem);
68#endif
69 }
70 WorkerWake(const WorkerWake&) = delete;
71 WorkerWake& operator=(const WorkerWake&) = delete;
72
73 void post() noexcept {
74#ifdef _WIN32
75 if (m_sem)
76 ReleaseSemaphore(m_sem, 1, nullptr); // at most one pending wake: a full count just fails
77#elif defined(__APPLE__)
78 if (m_sem)
79 dispatch_semaphore_signal(m_sem);
80#else
81 if (m_ok)
82 sem_post(&m_sem);
83#endif
84 }
86 void wait(std::chrono::microseconds timeout) noexcept {
87#ifdef _WIN32
88 if (m_sem) {
89 WaitForSingleObject(m_sem, static_cast<DWORD>(std::max<long long>(1, timeout.count() / 1000)));
90 return;
91 }
92#elif defined(__APPLE__)
93 if (m_sem) {
94 dispatch_semaphore_wait(m_sem, dispatch_time(DISPATCH_TIME_NOW, timeout.count() * 1000));
95 return;
96 }
97#else
98 if (m_ok) {
99 timespec ts{};
100 clock_gettime(CLOCK_REALTIME, &ts);
101 const long long ns = ts.tv_nsec + static_cast<long long>(timeout.count()) * 1000;
102 ts.tv_sec += static_cast<time_t>(ns / 1000000000);
103 ts.tv_nsec = static_cast<long>(ns % 1000000000);
104 while (sem_timedwait(&m_sem, &ts) != 0 && errno == EINTR) {
105 }
106 return;
107 }
108#endif
109 std::this_thread::sleep_for(timeout);
110 }
111
112private:
113#ifdef _WIN32
114 HANDLE m_sem = nullptr;
115#elif defined(__APPLE__)
116 dispatch_semaphore_t m_sem = nullptr;
117#else
118 sem_t m_sem{};
119 bool m_ok = false;
120#endif
121};

127// (project files, sockets, devices) are inherited across fork by default and
128// would be reachable from untrusted plugin code. Only 0/1/2 survive.
129//
130// Only async-signal-safe work is allowed here — fork in a multithreaded
131// process may hold malloc's lock. close_range is one syscall; the fallback is
132// close() in a bounded loop. opendir/readdir would allocate and can deadlock,
133// so the obvious /proc/self/fd iteration is out.

135// Doubly nested, not &&: invoking the function-like __GLIBC_PREREQ while it
136// is undefined is a hard error on Clang, and short-circuit does not save it —
137// expansion runs before evaluation. Each level keeps undefined macros out of
138// reach: macOS skips at __linux__, musl (no __GLIBC__) takes the loop below.
139#if defined(__linux__)
140#if defined(__GLIBC__)
141#if __GLIBC_PREREQ(2, 34)
142 if (::close_range(3, ~0u, 0) == 0) {
143 return;
144 }
145#endif
146#endif
147#endif

151 lim = rl.rlim_cur;
152 }
153 for (rlim_t fd = 3; fd < lim; ++fd) {
154 ::close(static_cast<int>(fd));

192bool hexDecodeRaw(const std::string& input, std::vector<uint8_t>& output) {
193 if ((input.size() % 2) != 0) {
194 return false;
195 }
196 output.resize(input.size() / 2);
197 for (size_t i = 0; i < output.size(); ++i) {
198 const int hi = hexValue(input[i * 2]);
199 const int lo = hexValue(input[i * 2 + 1]);
200 if (hi < 0 || lo < 0) {
201 return false;
202 }
203 output[i] = static_cast<uint8_t>((hi << 4) | lo);
204 }
205 return true;

241// The frame terminator is '\n' alone, so a '\r' before it is never payload.
242//
243// The helper now sets its stdout to binary mode, but this stays as the transport's
244// own guarantee: without it, a child whose stdout is in Windows text mode silently
245// appends '\r' to every reply. Prefix checks like `response.find("OK") == 0` cannot
246// see that, while PROCESS and SAVESTATE hex-decode the remainder of the line and
247// reject it for having an odd character count — so audio and plugin state break
248// while load and activate look perfectly healthy.
249void stripTrailingCR(std::string& line) {
250 if (!line.empty() && line.back() == '\r') {
251 line.pop_back();
252 }

328 // True once a command or a response was truncated part-way through a line.
329 // The newline framing cannot be recovered from that: the next read returns the
330 // tail of the abandoned response, so every later request/response pair is
331 // mismatched. Callers must fail the instance instead of continuing to talk.
332 //
333 // KNOWN GAP: this catches a reply cut off mid-line, not one that arrives
334 // complete but late. A command that times out with nothing read leaves the
335 // channel usable, and if the child answers afterwards that reply is returned to
336 // whichever command asks next — the same mismatch, undetectable from the byte
337 // stream alone. Closing it needs a sequence tag per command so a stale reply can
338 // be recognised and discarded instead of the channel being thrown away; that is
339 // a protocol change, deliberately not bundled with this fix.
340 bool framingLost() const { return m_framingLost; }

363bool PluginHostProcess::start() {
364#ifdef _WIN32
365 // Windows previously returned false here, which meant plugins were never
366 // sandboxed on Windows at all: the caller fell back to in-process hosting, so
367 // any plugin fault took Aestra down with it. Linux/macOS had fork() isolation
368 // the whole time.
369 if (m_process) {
370 return false; // already started
371 }
372
373 // Same pre-exec check the POSIX path performs. AESTRA_PLUGIN_HOST_PATH is
374 // attacker-influencable, so refuse anything that is not an absolute path to a
375 // regular file before handing it to CreateProcess (SEC-FIX parity).
376 std::error_code ec;
377 const std::filesystem::path exe(m_executablePath);
378 if (!exe.is_absolute() || !std::filesystem::is_regular_file(exe, ec)) {
379 return false;
380 }
381
382 SECURITY_ATTRIBUTES sa{};
383 sa.nLength = sizeof(sa);
384 sa.bInheritHandle = TRUE;
385 sa.lpSecurityDescriptor = nullptr;
386
387 // Every handle below is scoped: on any failure they close themselves, and the
388 // two we keep are released into members only once CreateProcess has succeeded.
389 ScopedHandle childStdinRead;
390 ScopedHandle childStdoutWrite;
391 ScopedHandle stdinWrite;
392 ScopedHandle stdoutRead;
393 if (!CreatePipe(childStdinRead.put(), stdinWrite.put(), &sa, 0)) {
394 return false;
395 }
396 if (!CreatePipe(stdoutRead.put(), childStdoutWrite.put(), &sa, 0)) {
397 return false;
398 }
399
400 // Our ends must NOT be inheritable. If they are, the child holds a duplicate
401 // of the write end of its own stdout and the pipe never reports EOF when the
402 // child dies — readLine() would then block until its timeout on every crash
403 // instead of noticing immediately.
404 SetHandleInformation(stdinWrite.get(), HANDLE_FLAG_INHERIT, 0);
405 SetHandleInformation(stdoutRead.get(), HANDLE_FLAG_INHERIT, 0);
406
407 // STARTF_USESTDHANDLES makes all three std handles significant, so stderr
408 // cannot just be whatever GetStdHandle returns. Two cases break it: a GUI
409 // process has no console and gets back NULL or INVALID_HANDLE_VALUE, and even a
410 // valid console handle is not necessarily inheritable — CreateProcess passes
411 // only inheritable handles. Either way the child would be left with an
412 // unusable stderr, losing exactly the diagnostics that matter most when an
413 // untrusted plugin dies inside the helper. Duplicating it inheritable keeps
414 // them; a NULL here (no console to forward to) is the documented way to say
415 // "no stderr", which is very different from a bogus handle.
416 ScopedHandle childStderr;
417 const HANDLE parentStderr = GetStdHandle(STD_ERROR_HANDLE);
418 if (parentStderr && parentStderr != INVALID_HANDLE_VALUE) {
419 if (!DuplicateHandle(GetCurrentProcess(), parentStderr, GetCurrentProcess(), childStderr.put(), 0,
420 TRUE, // inheritable, or CreateProcess will not pass it on
421 DUPLICATE_SAME_ACCESS)) {
422 childStderr.reset();
423 }
424 }
425
426 // bInheritHandles=TRUE below is all-or-nothing: it hands the child EVERY
427 // inheritable handle Aestra currently holds, not just the three std handles
428 // named here. For a process whose entire purpose is to contain untrusted
429 // third-party plugin code, that is a containment hole — an inheritable file or
430 // socket handle opened anywhere else in the application (a project being
431 // written, an audio device, a network connection) becomes reachable from inside
432 // a plugin, and which handles happen to be inheritable is not something this
433 // call can see. PROC_THREAD_ATTRIBUTE_HANDLE_LIST restricts inheritance to
434 // exactly the handles listed, so the child gets its pipes and nothing else.
435 //
436 // Every handle in the list must itself be inheritable or CreateProcess fails
437 // with ERROR_INVALID_PARAMETER, which is why childStderr is only included when
438 // the duplication above actually produced one.
439 HANDLE inheritList[3] = {childStdinRead.get(), childStdoutWrite.get(), nullptr};
440 DWORD inheritCount = 2;
441 if (childStderr) {
442 inheritList[inheritCount++] = childStderr.get();
443 }
444
445 SIZE_T attributeSize = 0;
446 // Deliberately unchecked: this call is documented to fail with
447 // ERROR_INSUFFICIENT_BUFFER while returning the size, which is the only reason
448 // to make it.
449 InitializeProcThreadAttributeList(nullptr, 1, 0, &attributeSize);
450 if (attributeSize == 0) {
451 return false;
452 }
453 // Element type is a pointer, not char: the list stores pointers internally, and a
454 // char buffer only guarantees char alignment. That is latent UB everywhere and an
455 // actual fault on strict-alignment targets such as Windows on ARM64.
456 std::vector<void*> attributeBuffer((attributeSize + sizeof(void*) - 1) / sizeof(void*));
457 auto* attributeList = reinterpret_cast<LPPROC_THREAD_ATTRIBUTE_LIST>(attributeBuffer.data());
458 if (!InitializeProcThreadAttributeList(attributeList, 1, 0, &attributeSize)) {
459 return false;
460 }
461 if (!UpdateProcThreadAttribute(attributeList, 0, PROC_THREAD_ATTRIBUTE_HANDLE_LIST, inheritList,
462 inheritCount * sizeof(HANDLE), nullptr, nullptr)) {
463 DeleteProcThreadAttributeList(attributeList);
464 return false;
465 }
466
467 STARTUPINFOEXA si{};
468 // Must be sizeof(STARTUPINFOEXA), not sizeof(STARTUPINFOA): the extended struct
469 // is how CreateProcess knows lpAttributeList is present and readable.
470 si.StartupInfo.cb = sizeof(si);
471 si.StartupInfo.dwFlags = STARTF_USESTDHANDLES;
472 si.StartupInfo.hStdInput = childStdinRead.get();
473 si.StartupInfo.hStdOutput = childStdoutWrite.get();
474 si.StartupInfo.hStdError = childStderr.get();
475 si.lpAttributeList = attributeList;
476
477 // CreateProcess mutates lpCommandLine, so it cannot be a string literal or a
478 // c_str(). The path is quoted because plugin host paths routinely sit under
479 // "Program Files".
480 std::string commandLine = "\"" + m_executablePath + "\" --stdio";
481 std::vector<char> mutableCommandLine(commandLine.begin(), commandLine.end());
482 mutableCommandLine.push_back('\0');
483
484 PROCESS_INFORMATION pi{};
485 const BOOL created =
486 CreateProcessA(m_executablePath.c_str(), mutableCommandLine.data(), nullptr, nullptr,
487 TRUE, // narrowed to inheritList by the attribute list above
488 CREATE_NO_WINDOW | EXTENDED_STARTUPINFO_PRESENT, nullptr, nullptr, &si.StartupInfo, &pi);
489 // The list is only consulted during process creation, so it can go immediately —
490 // and must, on both outcomes.
491 DeleteProcThreadAttributeList(attributeList);
492 if (!created) {
493 return false;
494 }
495
496 CloseHandle(pi.hThread);
497 // The child holds its own duplicates of the pipe ends now, so our copies of
498 // those must go — the stdout write end especially, or the pipe never reports EOF
499 // when the child dies. ScopedHandle closes them on the way out of this function;
500 // only the two ends we keep talking on are moved into members.
501 m_process.reset(pi.hProcess);
502 m_stdinWrite.reset(stdinWrite.release());
503 m_stdoutRead.reset(stdoutRead.release());
504 return true;
505#else
506 int inPipe[2] = {-1, -1};
507 int outPipe[2] = {-1, -1};
508 if (pipe(inPipe) != 0 || pipe(outPipe) != 0) {
509 return false;
510 }
511
512 m_pid = fork();
513 if (m_pid < 0) {
514 close(inPipe[0]);
515 close(inPipe[1]);
516 close(outPipe[0]);
517 close(outPipe[1]);
518 return false;
519 }
520
521 if (m_pid == 0) {
522 dup2(inPipe[0], STDIN_FILENO);
523 dup2(outPipe[1], STDOUT_FILENO);
524 close(inPipe[0]);
525 close(inPipe[1]);
526 close(outPipe[0]);
527 close(outPipe[1]);
528 closeUnrelatedDescriptors();
529 // [SEC-FIX] Validate path is absolute and a regular file before exec to prevent
530 // arbitrary code execution via AESTRA_PLUGIN_HOST_PATH environment variable.
531 std::error_code ec;
532 std::filesystem::path p(m_executablePath);
533 if (!p.is_absolute() || !std::filesystem::is_regular_file(p, ec)) {
534 _exit(127);
535 }
536 execl(m_executablePath.c_str(), m_executablePath.c_str(), "--stdio", static_cast<char*>(nullptr));
537 _exit(127);
538 }
539
540 close(inPipe[0]);
541 close(outPipe[1]);
542 m_stdinFd = inPipe[1];
543 m_stdoutFd = outPipe[0];
544 fcntl(m_stdoutFd, F_SETFL, fcntl(m_stdoutFd, F_GETFL, 0) | O_NONBLOCK);
545 return true;
546#endif
547}

549void PluginHostProcess::stop() {
550#ifdef _WIN32
551 // Ask first, then insist. Closing stdin is itself a shutdown signal: the child
552 // sees EOF on its read loop even if it ignored EXIT.
553 if (m_stdinWrite) {
554 sendLine("EXIT");
555 m_stdinWrite.reset(); // closing stdin is itself the second signal
556 }
557 m_stdoutRead.reset();
558 if (m_process) {
559 // Same 100ms grace as POSIX before escalating. A wedged plugin must not be
560 // able to hold shutdown open indefinitely, which is the whole point of
561 // hosting it out of process.
562 if (WaitForSingleObject(m_process.get(), 100) != WAIT_OBJECT_0) {
563 TerminateProcess(m_process.get(), 1);
564 // TerminateProcess only *requests* termination; it returns before the
565 // process object is signalled. That normally completes in microseconds,
566 // but a child blocked in an uninterruptible kernel call — a wedged
567 // driver, a hung network-share read reached from inside a plugin DLL —
568 // can stay unsignalled indefinitely, and untrusted third-party plugin
569 // code is exactly the population that produces those states. stop()
570 // runs from ~PluginHostProcess via shutdown(), so an unbounded wait
571 // here would block project close and application exit: the failure mode
572 // out-of-process hosting exists to prevent. We stop waiting and close our
573 // handle instead; the child stays in whatever state it is stuck in, which
574 // costs nothing we can recover anyway.
575 constexpr DWORD kTerminateWaitMs = 2000;
576 if (WaitForSingleObject(m_process.get(), kTerminateWaitMs) != WAIT_OBJECT_0) {
577 Log::warning("[PluginHost] helper process did not terminate; abandoning it to keep shutdown bounded");
578 }
579 }
580 m_process.reset();
581 }
582#else
583 if (m_stdinFd >= 0) {
584 sendLine("EXIT");
585 close(m_stdinFd);
586 m_stdinFd = -1;
587 }
588 if (m_stdoutFd >= 0) {
589 close(m_stdoutFd);
590 m_stdoutFd = -1;
591 }
592 if (m_pid > 0) {
593 int status = 0;
594 for (int i = 0; i < 10; ++i) {
595 if (waitpid(m_pid, &status, WNOHANG) == m_pid) {
596 m_pid = -1;
597 return;
598 }
599 std::this_thread::sleep_for(std::chrono::milliseconds(10));
600 }
601 kill(m_pid, SIGTERM);
602 waitpid(m_pid, &status, 0);
603 m_pid = -1;
604 }
605#endif
606}

608bool PluginHostProcess::isRunning() {
609#ifdef _WIN32
610 if (!m_process) {
611 return false;
612 }
613 DWORD exitCode = 0;
614 if (!GetExitCodeProcess(m_process.get(), &exitCode)) {
615 return false;
616 }
617 // STILL_ACTIVE is 259, which a process may also legitimately exit with. Confirm
618 // against the process handle rather than trusting the code alone, or a plugin
619 // that exits 259 would look alive forever and never be restarted.
620 if (exitCode == STILL_ACTIVE) {
621 return WaitForSingleObject(m_process.get(), 0) == WAIT_TIMEOUT;
622 }
623 return false;
624#else
625 if (m_pid <= 0) {
626 return false;
627 }
628 int status = 0;
629 const pid_t result = waitpid(m_pid, &status, WNOHANG);
630 if (result == 0) {
631 return true;
632 }
633 if (result == m_pid) {
634 m_pid = -1;
635 }
636 return false;
637#endif
638}

640bool PluginHostProcess::sendLine(const std::string& line) {
641#ifdef _WIN32
642 if (!m_stdinWrite || m_framingLost || !isRunning()) {
643 return false;
644 }
645 std::string framed = line;
646 framed.push_back('\n');
647 const char* data = framed.data();
648 DWORD remaining = static_cast<DWORD>(framed.size());
649 while (remaining > 0) {
650 DWORD written = 0;
651 if (!WriteFile(m_stdinWrite.get(), data, remaining, &written, nullptr) || written == 0) {
652 // A write that stopped part-way leaves half a command in the pipe, and
653 // the child would splice the next command onto its tail. Same
654 // unrecoverable framing loss as a truncated response.
655 if (remaining < framed.size()) {
656 m_framingLost = true;
657 }
658 return false;
659 }
660 data += written;
661 remaining -= written;
662 }
663 return true;
664#else
665 if (m_stdinFd < 0 || m_framingLost || !isRunning()) {
666 return false;
667 }
668 std::string framed = line;
669 framed.push_back('\n');
670 const char* data = framed.data();
671 size_t remaining = framed.size();
672 while (remaining > 0) {
673 const ssize_t written = write(m_stdinFd, data, remaining);
674 if (written <= 0) {
675 // See the Windows branch: a half-written command cannot be recovered
676 // from, because the child splices the next one onto its tail.
677 if (remaining < framed.size()) {
678 m_framingLost = true;
679 }
680 return false;
681 }
682 data += written;
683 remaining -= static_cast<size_t>(written);
684 }
685 return true;
686#endif
687}

689bool PluginHostProcess::readLine(std::string& line, std::chrono::milliseconds timeout) {
690#ifdef _WIN32
691 if (!m_stdoutRead || m_framingLost) {
692 return false;
693 }
694 line.clear();
695 const auto waitStart = std::chrono::steady_clock::now();
696 const auto deadline = waitStart + timeout;
697 for (;;) {
698 // Drained before touching the pipe: a bulk read can cross a response
699 // boundary, so the newline this call needs may already be in hand.
700 const size_t newline = m_readCarry.find('\n');
701 if (newline != std::string::npos) {
702 line.assign(m_readCarry, 0, newline);
703 m_readCarry.erase(0, newline + 1);
704 stripTrailingCR(line);
705 return true;
706 }
707 if (m_readCarry.size() > kMaxLineBytes) {
708 m_framingLost = true;
709 return false; // same runaway-line cap as the POSIX path
710 }
711 if (std::chrono::steady_clock::now() >= deadline) {
712 break;
713 }
714
715 // Anonymous pipes cannot be opened for overlapped I/O, so there is no
716 // WaitForSingleObject-able handle and no way to give ReadFile a timeout.
717 // PeekNamedPipe (which does work on anonymous pipes) supplies the
718 // "would block?" answer that O_NONBLOCK gives the POSIX path. Without the
719 // peek, ReadFile would block past the caller's deadline and a hung child
720 // would hang the caller with it.
721 DWORD available = 0;
722 if (!PeekNamedPipe(m_stdoutRead.get(), nullptr, 0, nullptr, &available, nullptr)) {
723 break; // broken pipe: child is gone
724 }
725 if (available == 0) {
726 if (!isRunning()) {
727 break;
728 }
729 // A block's "OK" usually lands within a couple of milliseconds, and a Windows sleep can last a whole
730 // 15.6 ms tick, longer than the block. So yield for the first 3 ms, then nap (V82-H7 follow-up).
731 if (std::chrono::steady_clock::now() < waitStart + std::chrono::milliseconds(3)) {
732 std::this_thread::yield();
733 } else {
734 std::this_thread::sleep_for(std::chrono::milliseconds(1));
735 }
736 continue;
737 }
738
739 // Take everything the peek reported in one call. Reading a byte at a time
740 // costs two pipe operations per byte, where the POSIX path pays one.
741 char buffer[4096];
742 const DWORD wanted = static_cast<DWORD>(std::min<size_t>(available, sizeof(buffer)));
743 DWORD read = 0;
744 if (!ReadFile(m_stdoutRead.get(), buffer, wanted, &read, nullptr) || read == 0) {
745 break;
746 }
747 m_readCarry.append(buffer, read);
748 }
749
750 // A clean timeout consumed nothing, so the channel stays usable. Bytes left
751 // without a newline mean this response was abandoned part-way through: the
752 // caller gives up and issues its next command, and the tail still sitting here
753 // would come back as that command's reply. Framing is unrecoverable then.
754 if (!m_readCarry.empty()) {
755 m_framingLost = true;
756 }
757 return false;
758#else
759 if (m_stdoutFd < 0 || m_framingLost) {
760 return false;
761 }
762 line.clear();
763 const auto deadline = std::chrono::steady_clock::now() + timeout;
764 while (std::chrono::steady_clock::now() < deadline) {
765 char ch = '\0';
766 const ssize_t n = read(m_stdoutFd, &ch, 1);
767 if (n == 1) {
768 if (ch == '\n') {
769 stripTrailingCR(line);
770 return true;
771 }
772 line.push_back(ch);
773 if (line.size() > kMaxLineBytes) {
774 m_framingLost = true;
775 return false;
776 }
777 } else {
778 if (!isRunning()) {
779 break;
780 }
781 // Sleep until the reply's next byte is there rather than a fixed 1 ms: a block's "OK" back from the
782 // helper used to wait out the rest of a nap every time (V82-H7 follow-up). Capped so a dead child is still noticed.
783 const auto left = std::chrono::duration_cast<std::chrono::milliseconds>(
784 deadline - std::chrono::steady_clock::now());
785 pollfd pfd{m_stdoutFd, POLLIN, 0};
786 ::poll(&pfd, 1, static_cast<int>(std::clamp<long long>(left.count() + 1, 1, 50)));
787 }
788 }
789 // Bytes already taken from the pipe are gone and the rest of this response is
790 // still queued — see the Windows branch for why that cannot be recovered from.
791 if (!line.empty()) {
792 m_framingLost = true;
793 }
794 return false;
795#endif
796}

961bool hexToText(const std::string& hex, std::string& out) {
962 out.clear();
963 if (hex.size() % 2 != 0) return false;
964 auto nibble = [](char c) -> int {
965 if (c >= '0' && c <= '9') return c - '0';
966 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
967 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
968 return -1;
969 };
970 for (size_t i = 0; i < hex.size(); i += 2) {
971 const int hi = nibble(hex[i]), lo = nibble(hex[i + 1]);
972 if (hi < 0 || lo < 0) return false;
973 out.push_back(static_cast<char>(hi * 16 + lo));
974 }
975 return true;
976}

979std::vector<PluginParameter> OutOfProcessPluginInstance::getParameters() const {
980 std::lock_guard<std::mutex> lock(m_paramListMutex);
981 if (m_paramListValid) return m_paramList;
982 if (!m_loaded.load(std::memory_order_acquire) || m_crashed.load(std::memory_order_acquire)) return {};
983
984 std::string response;
985 auto* self = const_cast<OutOfProcessPluginInstance*>(this);
986 if (!self->sendCommand("PARAMS", &response, std::chrono::seconds(10)) || response.rfind("OK ", 0) != 0) return {};
987 std::istringstream in(response.substr(3));
988 size_t count = 0;
989 in >> count;
990 std::vector<PluginParameter> list;
991 list.reserve(std::min<size_t>(count, 100000));
992 std::string token;
993 while (in >> token && list.size() < 100000) {
994 // id:flags:steps:default:name:short:unit
995 std::vector<std::string> field;
996 size_t start = 0;
997 for (size_t at; (at = token.find(':', start)) != std::string::npos; start = at + 1)
998 field.push_back(token.substr(start, at - start));
999 field.push_back(token.substr(start));
1000 if (field.size() != 7) continue;
1001 PluginParameter p;
1002 try {
1003 p.id = static_cast<uint32_t>(std::stoul(field[0]));
1004 const int flags = std::stoi(field[1]);
1005 p.isAutomatable = (flags & 1) != 0;
1006 p.isBypass = (flags & 2) != 0;
1007 p.isReadOnly = (flags & 4) != 0;
1008 p.stepCount = static_cast<uint32_t>(std::stoul(field[2]));
1009 p.defaultValue = std::stof(field[3]);
1010 } catch (const std::exception&) {
1011 continue;
1012 }
1013 if (!hexToText(field[4], p.name) || !hexToText(field[5], p.shortName) || !hexToText(field[6], p.unit)) continue;
1014 list.push_back(std::move(p));
1015 }
1016 m_paramList = std::move(list);
1017 m_paramListValid = true;
1018 return m_paramList;
1019}

1066void OutOfProcessPluginInstance::setParameter(uint32_t id, float value) {
1067 // Non-blocking by design: enqueue and return. No IPC, lock, or allocation
1068 // here. The worker thread forwards the change to the child in order (#238).
1069 // The queue is single-producer (see the header) — call from one thread. A
1070 // full queue drops the newest change and bumps a diagnostic counter
1071 // (parameterDropCount) rather than blocking; the first drop is logged so the
1072 // loss is observable instead of silent.
1073 if (!m_paramQueue.push(ParamChange{id, value})) {
1074 if (m_paramDrops.fetch_add(1, std::memory_order_relaxed) == 0) {
1075 Log::warning("[PluginHost] parameter queue full for " + m_info.name +
1076 "; dropping changes (raise kParamQueueCapacity if this recurs)");
1077 }
1078 }

1081void OutOfProcessPluginInstance::drainParamQueueToChild() {
1082 // Worker-thread only. Each change is a single SETPARAM round-trip on the IPC
1083 // channel this thread owns. Bounded per pass (kMaxParamDrainPerPass) so a big
1084 // burst against a slow child cannot starve PROCESS; the remainder drains on
1085 // the next worker iteration, in order.
1086 ParamChange change;
1087 for (size_t drained = 0;
1088 drained < kMaxParamDrainPerPass && (m_autoQueue.pop(change) || m_paramQueue.pop(change)); ++drained) {
1089 std::ostringstream command;
1090 command << "SETPARAM " << change.id << " " << change.value;
1091 std::string response;
1092 if (!sendCommand(command.str(), &response) || response.find("OK") != 0) {
1093 // A dead/unresponsive child fails the whole instance safely; a plugin
1094 // that rejects one parameter (ERR) is logged but does not crash us.
1095 if (!m_process || !m_process->isRunning()) {
1096 markCrashed();
1097 return;
1098 }
1099 Log::warning("[PluginHost] SETPARAM rejected for " + m_info.name + " (id=" +
1100 std::to_string(change.id) + ")");
1101 }
1102 }

1216bool OutOfProcessPluginInstance::sendCommand(const std::string& command, std::string* response,
1217 std::chrono::milliseconds timeout) {
1218 std::lock_guard<std::mutex> lock(m_ipcMutex);
1219 if (!m_process || !m_process->isRunning()) {
1220 return false;
1221 }
1222 // A truncated command or response desynchronizes every later request/response
1223 // pair on this channel, so it cannot be reused: the next reply read would be
1224 // the tail of the abandoned one. Failing the instance here is what keeps that
1225 // contained — saveState() and loadState() would otherwise hand their callers a
1226 // plausible-looking failure with no sign that the transport itself is broken,
1227 // and drainParamQueueToChild() would keep issuing commands into the desync
1228 // because the child is still alive. Checked at this choke point so every
1229 // caller, present and future, inherits it.
1230 if (!m_process->sendLine(command)) {
1231 if (m_process->framingLost()) {
1232 markCrashed();
1233 }
1234 return false;
1235 }
1236 std::string localResponse;
1237 if (!m_process->readLine(localResponse, timeout)) {
1238 if (m_process->framingLost()) {
1239 markCrashed();
1240 }
1241 return false;
1242 }
1243 if (response) {
1244 *response = std::move(localResponse);
1245 }
1246 return true;
1247}

1275void OutOfProcessPluginInstance::workerLoop() {
1276 while (!m_workerStop.load(std::memory_order_acquire)) {
1277 if (m_crashed.load(std::memory_order_acquire)) {
1278 std::this_thread::sleep_for(std::chrono::milliseconds(5));
1279 continue;
1280 }
1281
1282 // Forward any queued parameter changes to the child first, so a change
1283 // requested before this block takes effect for it. Runs every iteration
1284 // (even with no audio pending) so knob moves apply while transport is
1285 // stopped too (#238).
1286 drainParamQueueToChild();
1287 if (m_crashed.load(std::memory_order_acquire)) {
1288 continue;
1289 }
1290
1291 uint8_t pendingReady = 2;
1292 if (!m_pendingState.compare_exchange_strong(pendingReady, 3, std::memory_order_acq_rel)) {
1293 // process() posts the moment it hands a block over; the timeout keeps knob moves flowing while stopped.
1294 m_wake->wait(std::chrono::milliseconds(2));
1295 continue;
1296 }
1297
1298 const uint32_t frames = m_pendingFrames.load(std::memory_order_acquire);
1299 const size_t midiBytes = m_pendingMidiBytes.load(std::memory_order_acquire);
1300 const size_t sampleCount = static_cast<size_t>(frames) * m_transportChannels;
1301 if (sampleCount <= m_workerInput.size()) {
1302 std::copy_n(m_pendingInput.data(), sampleCount, m_workerInput.data());
1303 }
1304 if (midiBytes <= m_workerMidiData.size()) {
1305 std::copy_n(m_pendingMidiData.data(), midiBytes, m_workerMidiData.data());
1306 }
1307 m_workerTransport = m_pendingTransport;
1308 m_workerStart = m_pendingStart;
1309 m_pendingState.store(0, std::memory_order_release);
1310
1311 if (sampleCount > m_workerInput.size() || midiBytes > m_workerMidiData.size() ||
1312 !processBlockInHelper(m_workerInput, m_transportChannels, frames, m_workerMidiData, midiBytes,
1313 m_workerTransport, m_workerOutput)) {
1314 markCrashed();
1315 continue;
1316 }
1317
1318 uint8_t readyEmpty = 0;
1319 if (m_readyState.compare_exchange_strong(readyEmpty, 1, std::memory_order_acq_rel)) {
1320 const size_t outputSamples = std::min(sampleCount, m_workerOutput.size());
1321 std::copy_n(m_workerOutput.data(), outputSamples, m_readyOutput.data());
1322 m_readyStart = m_workerStart;
1323 m_readyFrames.store(frames, std::memory_order_release);
1324 m_readyState.store(2, std::memory_order_release);
1325 // Published last, so an observer that sees this advance is guaranteed
1326 // to find the result readable rather than half-written.
1327 const uint64_t blocks = m_processedBlocks.fetch_add(1, std::memory_order_acq_rel) + 1;
1328 // A plugin may only know its own latency once it has processed a block (LSP's limiters report theirs
1329 // then), and it can change later: ask after the first block and every kLatencyCheckEveryBlocks after.
1330 if (blocks == 1 || blocks % kLatencyCheckEveryBlocks == 0) {
1331 refreshPluginLatency();
1332 }
1333 }
1334 }
1335}

1349bool OutOfProcessPluginInstance::processBlockInHelper(const std::vector<float>& input, uint32_t channels,
1350 uint32_t frames, const std::vector<uint8_t>& midiData,
1351 size_t midiBytes, const HostTransport& transport,
1352 std::vector<float>& output) {
1353 const size_t sampleCount = static_cast<size_t>(channels) * frames;
1354 if (sampleCount > input.size()) {
1355 return false;
1356 }
1357 if (m_shm && sampleCount <= m_shm->maxSamples() && midiBytes <= m_shm->maxMidiBytes()) {
1358 // The samples go through the shared block; the pipe carries only the sizes and the "OK" (V82-H7 follow-up).
1359 std::copy_n(input.data(), sampleCount, m_shm->input());
1360 if (midiBytes > 0) {
1361 std::copy_n(midiData.data(), midiBytes, m_shm->midi());
1362 }
1363 const std::string command = "PROCESSSHM " + std::to_string(channels) + " " + std::to_string(frames) + " " +
1364 std::to_string(midiBytes) + transport.toWire();
1365 std::string response;
1366 if (!sendCommand(command, &response) || response != "OK") {
1367 return false;
1368 }
1369 if (output.size() < sampleCount) {
1370 output.resize(sampleCount);
1371 }
1372 std::copy_n(m_shm->output(), sampleCount, output.data());
1373 return true;
1374 }
1375 std::ostringstream command;
1376 if (midiBytes > 0) {
1377 command << "PROCESSMIDI " << channels << " " << frames << " "
1378 << hexEncodeBytes(input.data(), sampleCount * sizeof(float)) << " "
1379 << hexEncodeBytes(midiData.data(), midiBytes);
1380 } else {
1381 command << "PROCESS " << channels << " " << frames << " "
1382 << hexEncodeBytes(input.data(), sampleCount * sizeof(float));
1383 }
1384 command << transport.toWire();
1385 std::string response;
1386 if (!sendCommand(command.str(), &response)) {
1387 return false;
1388 }
1389 constexpr const char* kPrefix = "OK ";
1390 if (response.compare(0, 3, kPrefix) != 0) {
1391 return false;
1392 }
1393 return hexDecodeBytes(response.substr(3), output) && output.size() >= sampleCount;
1394}
