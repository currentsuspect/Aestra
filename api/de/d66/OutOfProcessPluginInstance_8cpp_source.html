34std::string hexEncode(const std::string& input) {
35 static constexpr char kHex[] = "0123456789ABCDEF";
36 std::string out;
37 out.reserve(input.size() * 2);
38 for (unsigned char c : input) {
39 out.push_back(kHex[c >> 4]);
40 out.push_back(kHex[c & 0x0F]);
41 }
42 return out;
43}

45std::string hexEncodeBytes(const void* data, size_t size) {
46 static constexpr char kHex[] = "0123456789ABCDEF";
47 const auto* bytes = static_cast<const unsigned char*>(data);
48 std::string out;
49 out.reserve(size * 2);
50 for (size_t i = 0; i < size; ++i) {
51 out.push_back(kHex[bytes[i] >> 4]);
52 out.push_back(kHex[bytes[i] & 0x0F]);
53 }
54 return out;
55}

57int hexValue(char c) {
58 if (c >= '0' && c <= '9')
59 return c - '0';
60 if (c >= 'a' && c <= 'f')
61 return 10 + c - 'a';
62 if (c >= 'A' && c <= 'F')
63 return 10 + c - 'A';
64 return -1;
65}

67bool hexDecodeRaw(const std::string& input, std::vector<uint8_t>& output) {
68 if ((input.size() % 2) != 0) {
69 return false;
70 }
71 output.resize(input.size() / 2);
72 for (size_t i = 0; i < output.size(); ++i) {
73 const int hi = hexValue(input[i * 2]);
74 const int lo = hexValue(input[i * 2 + 1]);
75 if (hi < 0 || lo < 0) {
76 return false;
77 }
78 output[i] = static_cast<uint8_t>((hi << 4) | lo);
79 }
80 return true;
81}

83bool hexDecodeBytes(const std::string& input, std::vector<float>& output) {
84 if ((input.size() % 2) != 0) {
85 return false;
86 }
87 const size_t byteCount = input.size() / 2;
88 if ((byteCount % sizeof(float)) != 0) {
89 return false;
90 }
91 output.resize(byteCount / sizeof(float));
92 auto* bytes = reinterpret_cast<unsigned char*>(output.data());
93 for (size_t i = 0; i < byteCount; ++i) {
94 const int hi = hexValue(input[i * 2]);
95 const int lo = hexValue(input[i * 2 + 1]);
96 if (hi < 0 || lo < 0) {
97 return false;
98 }
99 bytes[i] = static_cast<unsigned char>((hi << 4) | lo);
100 }
101 return true;
102}

116// The frame terminator is '\n' alone, so a '\r' before it is never payload.
117//
118// The helper now sets its stdout to binary mode, but this stays as the transport's
119// own guarantee: without it, a child whose stdout is in Windows text mode silently
120// appends '\r' to every reply. Prefix checks like `response.find("OK") == 0` cannot
121// see that, while PROCESS and SAVESTATE hex-decode the remainder of the line and
122// reject it for having an odd character count — so audio and plugin state break
123// while load and activate look perfectly healthy.
124void stripTrailingCR(std::string& line) {
125 if (!line.empty() && line.back() == '\r') {
126 line.pop_back();
127 }

154// Owns one Win32 HANDLE. start() juggles five of them across four exit paths, and
155// every future early return would otherwise have to remember the right subset of
156// CloseHandle calls by hand.
157class ScopedHandle {
158public:
159 ScopedHandle() = default;
160 explicit ScopedHandle(HANDLE handle) : m_handle(handle) {}
161 ~ScopedHandle() { reset(); }
162
163 ScopedHandle(const ScopedHandle&) = delete;
164 ScopedHandle& operator=(const ScopedHandle&) = delete;
165
166 void reset(HANDLE handle = nullptr) {
167 if (m_handle && m_handle != INVALID_HANDLE_VALUE) {
168 CloseHandle(m_handle);
169 }
170 m_handle = handle;
171 }
172
173 HANDLE get() const { return m_handle; }
174 HANDLE* put() { return &m_handle; }
175
176 // Hand ownership to a long-lived member once the call has fully succeeded.
177 HANDLE release() {
178 HANDLE handle = m_handle;
179 m_handle = nullptr;
180 return handle;
181 }
182
183 explicit operator bool() const { return m_handle && m_handle != INVALID_HANDLE_VALUE; }
184
185private:
186 HANDLE m_handle = nullptr;
187};

203 // True once a command or a response was truncated part-way through a line.
204 // The newline framing cannot be recovered from that: the next read returns the
205 // tail of the abandoned response, so every later request/response pair is
206 // mismatched. Callers must fail the instance instead of continuing to talk.
207 //
208 // KNOWN GAP: this catches a reply cut off mid-line, not one that arrives
209 // complete but late. A command that times out with nothing read leaves the
210 // channel usable, and if the child answers afterwards that reply is returned to
211 // whichever command asks next — the same mismatch, undetectable from the byte
212 // stream alone. Closing it needs a sequence tag per command so a stale reply can
213 // be recognised and discarded instead of the channel being thrown away; that is
214 // a protocol change, deliberately not bundled with this fix.
215 bool framingLost() const { return m_framingLost; }
216
217private:
218 // Matches the POSIX and Windows runaway-line caps; a line this long means the
219 // child is malfunctioning, not that a reply is merely large.
220 static constexpr size_t kMaxLineBytes = 16ull * 1024ull * 1024ull;
221
222 std::string m_executablePath;
223 bool m_framingLost = false;
224#ifdef _WIN32
225 ScopedHandle m_process; // child process
226 ScopedHandle m_stdinWrite; // our end of the child's stdin
227 ScopedHandle m_stdoutRead; // our end of the child's stdout
228 // Bytes read past the current line. A bulk read can cross a response boundary,
229 // so the remainder is kept here rather than thrown away.
230 std::string m_readCarry;
231#else
232 pid_t m_pid = -1;
233 int m_stdinFd = -1;
234 int m_stdoutFd = -1;
235#endif
236};

238bool PluginHostProcess::start() {
239#ifdef _WIN32
240 // Windows previously returned false here, which meant plugins were never
241 // sandboxed on Windows at all: the caller fell back to in-process hosting, so
242 // any plugin fault took Aestra down with it. Linux/macOS had fork() isolation
243 // the whole time.
244 if (m_process) {
245 return false; // already started
246 }
247
248 // Same pre-exec check the POSIX path performs. AESTRA_PLUGIN_HOST_PATH is
249 // attacker-influencable, so refuse anything that is not an absolute path to a
250 // regular file before handing it to CreateProcess (SEC-FIX parity).
251 std::error_code ec;
252 const std::filesystem::path exe(m_executablePath);
253 if (!exe.is_absolute() || !std::filesystem::is_regular_file(exe, ec)) {
254 return false;
255 }
256
257 SECURITY_ATTRIBUTES sa{};
258 sa.nLength = sizeof(sa);
259 sa.bInheritHandle = TRUE;
260 sa.lpSecurityDescriptor = nullptr;
261
262 // Every handle below is scoped: on any failure they close themselves, and the
263 // two we keep are released into members only once CreateProcess has succeeded.
264 ScopedHandle childStdinRead;
265 ScopedHandle childStdoutWrite;
266 ScopedHandle stdinWrite;
267 ScopedHandle stdoutRead;
268 if (!CreatePipe(childStdinRead.put(), stdinWrite.put(), &sa, 0)) {
269 return false;
270 }
271 if (!CreatePipe(stdoutRead.put(), childStdoutWrite.put(), &sa, 0)) {
272 return false;
273 }
274
275 // Our ends must NOT be inheritable. If they are, the child holds a duplicate
276 // of the write end of its own stdout and the pipe never reports EOF when the
277 // child dies — readLine() would then block until its timeout on every crash
278 // instead of noticing immediately.
279 SetHandleInformation(stdinWrite.get(), HANDLE_FLAG_INHERIT, 0);
280 SetHandleInformation(stdoutRead.get(), HANDLE_FLAG_INHERIT, 0);
281
282 // STARTF_USESTDHANDLES makes all three std handles significant, so stderr
283 // cannot just be whatever GetStdHandle returns. Two cases break it: a GUI
284 // process has no console and gets back NULL or INVALID_HANDLE_VALUE, and even a
285 // valid console handle is not necessarily inheritable — CreateProcess passes
286 // only inheritable handles. Either way the child would be left with an
287 // unusable stderr, losing exactly the diagnostics that matter most when an
288 // untrusted plugin dies inside the helper. Duplicating it inheritable keeps
289 // them; a NULL here (no console to forward to) is the documented way to say
290 // "no stderr", which is very different from a bogus handle.
291 ScopedHandle childStderr;
292 const HANDLE parentStderr = GetStdHandle(STD_ERROR_HANDLE);
293 if (parentStderr && parentStderr != INVALID_HANDLE_VALUE) {
294 if (!DuplicateHandle(GetCurrentProcess(), parentStderr, GetCurrentProcess(), childStderr.put(), 0,
295 TRUE, // inheritable, or CreateProcess will not pass it on
296 DUPLICATE_SAME_ACCESS)) {
297 childStderr.reset();
298 }
299 }
300
301 // bInheritHandles=TRUE below is all-or-nothing: it hands the child EVERY
302 // inheritable handle Aestra currently holds, not just the three std handles
303 // named here. For a process whose entire purpose is to contain untrusted
304 // third-party plugin code, that is a containment hole — an inheritable file or
305 // socket handle opened anywhere else in the application (a project being
306 // written, an audio device, a network connection) becomes reachable from inside
307 // a plugin, and which handles happen to be inheritable is not something this
308 // call can see. PROC_THREAD_ATTRIBUTE_HANDLE_LIST restricts inheritance to
309 // exactly the handles listed, so the child gets its pipes and nothing else.
310 //
311 // Every handle in the list must itself be inheritable or CreateProcess fails
312 // with ERROR_INVALID_PARAMETER, which is why childStderr is only included when
313 // the duplication above actually produced one.
314 HANDLE inheritList[3] = {childStdinRead.get(), childStdoutWrite.get(), nullptr};
315 DWORD inheritCount = 2;
316 if (childStderr) {
317 inheritList[inheritCount++] = childStderr.get();
318 }
319
320 SIZE_T attributeSize = 0;
321 // Deliberately unchecked: this call is documented to fail with
322 // ERROR_INSUFFICIENT_BUFFER while returning the size, which is the only reason
323 // to make it.
324 InitializeProcThreadAttributeList(nullptr, 1, 0, &attributeSize);
325 if (attributeSize == 0) {
326 return false;
327 }
328 // Element type is a pointer, not char: the list stores pointers internally, and a
329 // char buffer only guarantees char alignment. That is latent UB everywhere and an
330 // actual fault on strict-alignment targets such as Windows on ARM64.
331 std::vector<void*> attributeBuffer((attributeSize + sizeof(void*) - 1) / sizeof(void*));
332 auto* attributeList = reinterpret_cast<LPPROC_THREAD_ATTRIBUTE_LIST>(attributeBuffer.data());
333 if (!InitializeProcThreadAttributeList(attributeList, 1, 0, &attributeSize)) {
334 return false;
335 }
336 if (!UpdateProcThreadAttribute(attributeList, 0, PROC_THREAD_ATTRIBUTE_HANDLE_LIST, inheritList,
337 inheritCount * sizeof(HANDLE), nullptr, nullptr)) {
338 DeleteProcThreadAttributeList(attributeList);
339 return false;
340 }
341
342 STARTUPINFOEXA si{};
343 // Must be sizeof(STARTUPINFOEXA), not sizeof(STARTUPINFOA): the extended struct
344 // is how CreateProcess knows lpAttributeList is present and readable.
345 si.StartupInfo.cb = sizeof(si);
346 si.StartupInfo.dwFlags = STARTF_USESTDHANDLES;
347 si.StartupInfo.hStdInput = childStdinRead.get();
348 si.StartupInfo.hStdOutput = childStdoutWrite.get();
349 si.StartupInfo.hStdError = childStderr.get();
350 si.lpAttributeList = attributeList;
351
352 // CreateProcess mutates lpCommandLine, so it cannot be a string literal or a
353 // c_str(). The path is quoted because plugin host paths routinely sit under
354 // "Program Files".
355 std::string commandLine = "\"" + m_executablePath + "\" --stdio";
356 std::vector<char> mutableCommandLine(commandLine.begin(), commandLine.end());
357 mutableCommandLine.push_back('\0');
358
359 PROCESS_INFORMATION pi{};
360 const BOOL created =
361 CreateProcessA(m_executablePath.c_str(), mutableCommandLine.data(), nullptr, nullptr,
362 TRUE, // narrowed to inheritList by the attribute list above
363 CREATE_NO_WINDOW | EXTENDED_STARTUPINFO_PRESENT, nullptr, nullptr, &si.StartupInfo, &pi);
364 // The list is only consulted during process creation, so it can go immediately —
365 // and must, on both outcomes.
366 DeleteProcThreadAttributeList(attributeList);
367 if (!created) {
368 return false;
369 }
370
371 CloseHandle(pi.hThread);
372 // The child holds its own duplicates of the pipe ends now, so our copies of
373 // those must go — the stdout write end especially, or the pipe never reports EOF
374 // when the child dies. ScopedHandle closes them on the way out of this function;
375 // only the two ends we keep talking on are moved into members.
376 m_process.reset(pi.hProcess);
377 m_stdinWrite.reset(stdinWrite.release());
378 m_stdoutRead.reset(stdoutRead.release());
379 return true;
380#else
381 int inPipe[2] = {-1, -1};
382 int outPipe[2] = {-1, -1};
383 if (pipe(inPipe) != 0 || pipe(outPipe) != 0) {
384 return false;
385 }
386
387 m_pid = fork();
388 if (m_pid < 0) {
389 close(inPipe[0]);
390 close(inPipe[1]);
391 close(outPipe[0]);
392 close(outPipe[1]);
393 return false;
394 }
395
396 if (m_pid == 0) {
397 dup2(inPipe[0], STDIN_FILENO);
398 dup2(outPipe[1], STDOUT_FILENO);
399 close(inPipe[0]);
400 close(inPipe[1]);
401 close(outPipe[0]);
402 close(outPipe[1]);
403 // [SEC-FIX] Validate path is absolute and a regular file before exec to prevent
404 // arbitrary code execution via AESTRA_PLUGIN_HOST_PATH environment variable.
405 std::error_code ec;
406 std::filesystem::path p(m_executablePath);
407 if (!p.is_absolute() || !std::filesystem::is_regular_file(p, ec)) {
408 _exit(127);
409 }
410 execl(m_executablePath.c_str(), m_executablePath.c_str(), "--stdio", static_cast<char*>(nullptr));
411 _exit(127);
412 }
413
414 close(inPipe[0]);
415 close(outPipe[1]);
416 m_stdinFd = inPipe[1];
417 m_stdoutFd = outPipe[0];
418 fcntl(m_stdoutFd, F_SETFL, fcntl(m_stdoutFd, F_GETFL, 0) | O_NONBLOCK);
419 return true;
420#endif
421}

423void PluginHostProcess::stop() {
424#ifdef _WIN32
425 // Ask first, then insist. Closing stdin is itself a shutdown signal: the child
426 // sees EOF on its read loop even if it ignored EXIT.
427 if (m_stdinWrite) {
428 sendLine("EXIT");
429 m_stdinWrite.reset(); // closing stdin is itself the second signal
430 }
431 m_stdoutRead.reset();
432 if (m_process) {
433 // Same 100ms grace as POSIX before escalating. A wedged plugin must not be
434 // able to hold shutdown open indefinitely, which is the whole point of
435 // hosting it out of process.
436 if (WaitForSingleObject(m_process.get(), 100) != WAIT_OBJECT_0) {
437 TerminateProcess(m_process.get(), 1);
438 // TerminateProcess only *requests* termination; it returns before the
439 // process object is signalled. That normally completes in microseconds,
440 // but a child blocked in an uninterruptible kernel call — a wedged
441 // driver, a hung network-share read reached from inside a plugin DLL —
442 // can stay unsignalled indefinitely, and untrusted third-party plugin
443 // code is exactly the population that produces those states. stop()
444 // runs from ~PluginHostProcess via shutdown(), so an unbounded wait
445 // here would block project close and application exit: the failure mode
446 // out-of-process hosting exists to prevent. We stop waiting and close our
447 // handle instead; the child stays in whatever state it is stuck in, which
448 // costs nothing we can recover anyway.
449 constexpr DWORD kTerminateWaitMs = 2000;
450 if (WaitForSingleObject(m_process.get(), kTerminateWaitMs) != WAIT_OBJECT_0) {
451 Log::warning("[PluginHost] helper process did not terminate; abandoning it to keep shutdown bounded");
452 }
453 }
454 m_process.reset();
455 }
456#else
457 if (m_stdinFd >= 0) {
458 sendLine("EXIT");
459 close(m_stdinFd);
460 m_stdinFd = -1;
461 }
462 if (m_stdoutFd >= 0) {
463 close(m_stdoutFd);
464 m_stdoutFd = -1;
465 }
466 if (m_pid > 0) {
467 int status = 0;
468 for (int i = 0; i < 10; ++i) {
469 if (waitpid(m_pid, &status, WNOHANG) == m_pid) {
470 m_pid = -1;
471 return;
472 }
473 std::this_thread::sleep_for(std::chrono::milliseconds(10));
474 }
475 kill(m_pid, SIGTERM);
476 waitpid(m_pid, &status, 0);
477 m_pid = -1;
478 }
479#endif
480}

482bool PluginHostProcess::isRunning() {
483#ifdef _WIN32
484 if (!m_process) {
485 return false;
486 }
487 DWORD exitCode = 0;
488 if (!GetExitCodeProcess(m_process.get(), &exitCode)) {
489 return false;
490 }
491 // STILL_ACTIVE is 259, which a process may also legitimately exit with. Confirm
492 // against the process handle rather than trusting the code alone, or a plugin
493 // that exits 259 would look alive forever and never be restarted.
494 if (exitCode == STILL_ACTIVE) {
495 return WaitForSingleObject(m_process.get(), 0) == WAIT_TIMEOUT;
496 }
497 return false;
498#else
499 if (m_pid <= 0) {
500 return false;
501 }
502 int status = 0;
503 const pid_t result = waitpid(m_pid, &status, WNOHANG);
504 if (result == 0) {
505 return true;
506 }
507 if (result == m_pid) {
508 m_pid = -1;
509 }
510 return false;
511#endif
512}

514bool PluginHostProcess::sendLine(const std::string& line) {
515#ifdef _WIN32
516 if (!m_stdinWrite || m_framingLost || !isRunning()) {
517 return false;
518 }
519 std::string framed = line;
520 framed.push_back('\n');
521 const char* data = framed.data();
522 DWORD remaining = static_cast<DWORD>(framed.size());
523 while (remaining > 0) {
524 DWORD written = 0;
525 if (!WriteFile(m_stdinWrite.get(), data, remaining, &written, nullptr) || written == 0) {
526 // A write that stopped part-way leaves half a command in the pipe, and
527 // the child would splice the next command onto its tail. Same
528 // unrecoverable framing loss as a truncated response.
529 if (remaining < framed.size()) {
530 m_framingLost = true;
531 }
532 return false;
533 }
534 data += written;
535 remaining -= written;
536 }
537 return true;
538#else
539 if (m_stdinFd < 0 || m_framingLost || !isRunning()) {
540 return false;
541 }
542 std::string framed = line;
543 framed.push_back('\n');
544 const char* data = framed.data();
545 size_t remaining = framed.size();
546 while (remaining > 0) {
547 const ssize_t written = write(m_stdinFd, data, remaining);
548 if (written <= 0) {
549 // See the Windows branch: a half-written command cannot be recovered
550 // from, because the child splices the next one onto its tail.
551 if (remaining < framed.size()) {
552 m_framingLost = true;
553 }
554 return false;
555 }
556 data += written;
557 remaining -= static_cast<size_t>(written);
558 }
559 return true;
560#endif
561}

563bool PluginHostProcess::readLine(std::string& line, std::chrono::milliseconds timeout) {
564#ifdef _WIN32
565 if (!m_stdoutRead || m_framingLost) {
566 return false;
567 }
568 line.clear();
569 const auto deadline = std::chrono::steady_clock::now() + timeout;
570 for (;;) {
571 // Drained before touching the pipe: a bulk read can cross a response
572 // boundary, so the newline this call needs may already be in hand.
573 const size_t newline = m_readCarry.find('\n');
574 if (newline != std::string::npos) {
575 line.assign(m_readCarry, 0, newline);
576 m_readCarry.erase(0, newline + 1);
577 stripTrailingCR(line);
578 return true;
579 }
580 if (m_readCarry.size() > kMaxLineBytes) {
581 m_framingLost = true;
582 return false; // same runaway-line cap as the POSIX path
583 }
584 if (std::chrono::steady_clock::now() >= deadline) {
585 break;
586 }
587
588 // Anonymous pipes cannot be opened for overlapped I/O, so there is no
589 // WaitForSingleObject-able handle and no way to give ReadFile a timeout.
590 // PeekNamedPipe (which does work on anonymous pipes) supplies the
591 // "would block?" answer that O_NONBLOCK gives the POSIX path. Without the
592 // peek, ReadFile would block past the caller's deadline and a hung child
593 // would hang the caller with it.
594 DWORD available = 0;
595 if (!PeekNamedPipe(m_stdoutRead.get(), nullptr, 0, nullptr, &available, nullptr)) {
596 break; // broken pipe: child is gone
597 }
598 if (available == 0) {
599 if (!isRunning()) {
600 break;
601 }
602 std::this_thread::sleep_for(std::chrono::milliseconds(1));
603 continue;
604 }
605
606 // Take everything the peek reported in one call. Reading a byte at a time
607 // costs two pipe operations per byte, where the POSIX path pays one.
608 char buffer[4096];
609 const DWORD wanted = static_cast<DWORD>(std::min<size_t>(available, sizeof(buffer)));
610 DWORD read = 0;
611 if (!ReadFile(m_stdoutRead.get(), buffer, wanted, &read, nullptr) || read == 0) {
612 break;
613 }
614 m_readCarry.append(buffer, read);
615 }
616
617 // A clean timeout consumed nothing, so the channel stays usable. Bytes left
618 // without a newline mean this response was abandoned part-way through: the
619 // caller gives up and issues its next command, and the tail still sitting here
620 // would come back as that command's reply. Framing is unrecoverable then.
621 if (!m_readCarry.empty()) {
622 m_framingLost = true;
623 }
624 return false;
625#else
626 if (m_stdoutFd < 0 || m_framingLost) {
627 return false;
628 }
629 line.clear();
630 const auto deadline = std::chrono::steady_clock::now() + timeout;
631 while (std::chrono::steady_clock::now() < deadline) {
632 char ch = '\0';
633 const ssize_t n = read(m_stdoutFd, &ch, 1);
634 if (n == 1) {
635 if (ch == '\n') {
636 stripTrailingCR(line);
637 return true;
638 }
639 line.push_back(ch);
640 if (line.size() > kMaxLineBytes) {
641 m_framingLost = true;
642 return false;
643 }
644 } else {
645 if (!isRunning()) {
646 break;
647 }
648 std::this_thread::sleep_for(std::chrono::milliseconds(1));
649 }
650 }
651 // Bytes already taken from the pipe are gone and the rest of this response is
652 // still queued — see the Windows branch for why that cannot be recovered from.
653 if (!line.empty()) {
654 m_framingLost = true;
655 }
656 return false;
657#endif
658}

748void OutOfProcessPluginInstance::process(const float* const* inputs, float** outputs, uint32_t numInputChannels,
749 uint32_t numOutputChannels, uint32_t numFrames, const MidiBuffer* midiInput,
750 MidiBuffer* midiOutput) {
751 if (midiOutput) {
752 midiOutput->clear();
753 }
754
755 if (m_crashed.load(std::memory_order_acquire) || m_maxBlockSize == 0 || numFrames > m_maxBlockSize) {
756 passThrough(inputs, outputs, numInputChannels, numOutputChannels, numFrames);
757 return;
758 }
759
760 uint8_t readyEmpty = 2;
761 bool usedReadyOutput = false;
762 if (m_readyState.compare_exchange_strong(readyEmpty, 3, std::memory_order_acq_rel)) {
763 const uint32_t frames = std::min<uint32_t>(m_readyFrames.load(std::memory_order_acquire), numFrames);
764 for (uint32_t ch = 0; ch < numOutputChannels; ++ch) {
765 if (!outputs || !outputs[ch]) {
766 continue;
767 }
768 if (ch < m_transportChannels) {
769 for (uint32_t frame = 0; frame < frames; ++frame) {
770 outputs[ch][frame] = m_readyOutput[static_cast<size_t>(frame) * m_transportChannels + ch];
771 }
772 if (frames < numFrames) {
773 std::memset(outputs[ch] + frames, 0, static_cast<size_t>(numFrames - frames) * sizeof(float));
774 }
775 } else {
776 std::memset(outputs[ch], 0, static_cast<size_t>(numFrames) * sizeof(float));
777 }
778 }
779 m_readyState.store(0, std::memory_order_release);
780 usedReadyOutput = true;
781 }
782
783 uint8_t pendingEmpty = 0;
784 if (m_pendingState.compare_exchange_strong(pendingEmpty, 1, std::memory_order_acq_rel)) {
785 for (uint32_t frame = 0; frame < numFrames; ++frame) {
786 for (uint32_t ch = 0; ch < m_transportChannels; ++ch) {
787 const size_t index = static_cast<size_t>(frame) * m_transportChannels + ch;
788 m_pendingInput[index] = (inputs && ch < numInputChannels && inputs[ch]) ? inputs[ch][frame] : 0.0f;
789 }
790 }
791 size_t midiBytes = 0;
792 if (midiInput) {
793 const size_t maxEvents = m_pendingMidiData.size() / 8;
794 const size_t eventCount = std::min(midiInput->getEventCount(), maxEvents);
795 for (size_t i = 0; i < eventCount; ++i) {
796 const auto& event = midiInput->getEvent(i);
797 if (event.size != 3 || midiBytes + 8 > m_pendingMidiData.size()) {
798 continue;
799 }
800 const uint32_t sampleOffset = std::min<uint32_t>(event.sampleOffset, numFrames > 0 ? numFrames - 1 : 0);
801 std::memcpy(m_pendingMidiData.data() + midiBytes, &sampleOffset, sizeof(sampleOffset));
802 m_pendingMidiData[midiBytes + 4] = event.size;
803 std::memcpy(m_pendingMidiData.data() + midiBytes + 5, event.data, 3);
804 midiBytes += 8;
805 }
806 }
807 m_pendingMidiBytes.store(midiBytes, std::memory_order_release);
808 m_pendingFrames.store(numFrames, std::memory_order_release);
809 m_pendingState.store(2, std::memory_order_release);
810 }
811
812 if (!usedReadyOutput) {
813 passThrough(inputs, outputs, numInputChannels, numOutputChannels, numFrames);
814 }
815}

822void OutOfProcessPluginInstance::setParameter(uint32_t id, float value) {
823 // Non-blocking by design: enqueue and return. No IPC, lock, or allocation
824 // here. The worker thread forwards the change to the child in order (#238).
825 // The queue is single-producer (see the header) — call from one thread. A
826 // full queue drops the newest change and bumps a diagnostic counter
827 // (parameterDropCount) rather than blocking; the first drop is logged so the
828 // loss is observable instead of silent.
829 if (!m_paramQueue.push(ParamChange{id, value})) {
830 if (m_paramDrops.fetch_add(1, std::memory_order_relaxed) == 0) {
831 Log::warning("[PluginHost] parameter queue full for " + m_info.name +
832 "; dropping changes (raise kParamQueueCapacity if this recurs)");
833 }
834 }
835}
836
837void OutOfProcessPluginInstance::drainParamQueueToChild() {
838 // Worker-thread only. Each change is a single SETPARAM round-trip on the IPC
839 // channel this thread owns. Bounded per pass (kMaxParamDrainPerPass) so a big
840 // burst against a slow child cannot starve PROCESS; the remainder drains on
841 // the next worker iteration, in order.
842 ParamChange change;
843 for (size_t drained = 0; drained < kMaxParamDrainPerPass && m_paramQueue.pop(change); ++drained) {
844 std::ostringstream command;
845 command << "SETPARAM " << change.id << " " << change.value;
846 std::string response;
847 if (!sendCommand(command.str(), &response) || response.find("OK") != 0) {
848 // A dead/unresponsive child fails the whole instance safely; a plugin
849 // that rejects one parameter (ERR) is logged but does not crash us.
850 if (!m_process || !m_process->isRunning()) {
851 markCrashed();
852 return;
853 }
854 Log::warning("[PluginHost] SETPARAM rejected for " + m_info.name + " (id=" +
855 std::to_string(change.id) + ")");
856 }
857 }

922void OutOfProcessPluginInstance::resetWatchdog() {
923 m_watchdogMaxExecutionTimeNs.store(0, std::memory_order_release);
924 m_watchdogAvgExecutionTimeNs.store(0, std::memory_order_release);
925 m_watchdogViolationCount.store(0, std::memory_order_release);
926 m_watchdogBypassed.store(false, std::memory_order_release);
927 // A live child is not sufficient: a wedged-but-running child whose channel lost
928 // framing satisfies isRunning(), and un-crashing there would take process() off
929 // the passthrough path and onto a transport that cannot answer correctly. The
930 // next sendCommand() would re-crash it, so this is a narrow window rather than a
931 // lasting wrong state — but the invariant is "once framing is lost, stop
932 // talking", and this function is a caller like any other.
933 if (m_process && m_process->isRunning() && !m_process->framingLost()) {
934 m_crashed.store(false, std::memory_order_release);
935 }

938bool OutOfProcessPluginInstance::sendCommand(const std::string& command, std::string* response,
939 std::chrono::milliseconds timeout) {
940 std::lock_guard<std::mutex> lock(m_ipcMutex);
941 if (!m_process || !m_process->isRunning()) {
942 return false;
943 }
944 // A truncated command or response desynchronizes every later request/response
945 // pair on this channel, so it cannot be reused: the next reply read would be
946 // the tail of the abandoned one. Failing the instance here is what keeps that
947 // contained — saveState() and loadState() would otherwise hand their callers a
948 // plausible-looking failure with no sign that the transport itself is broken,
949 // and drainParamQueueToChild() would keep issuing commands into the desync
950 // because the child is still alive. Checked at this choke point so every
951 // caller, present and future, inherits it.
952 if (!m_process->sendLine(command)) {
953 if (m_process->framingLost()) {
954 markCrashed();
955 }
956 return false;
957 }
958 std::string localResponse;
959 if (!m_process->readLine(localResponse, timeout)) {
960 if (m_process->framingLost()) {
961 markCrashed();
962 }
963 return false;
964 }
965 if (response) {
966 *response = std::move(localResponse);
967 }
968 return true;
969}

988void OutOfProcessPluginInstance::workerLoop() {
989 while (!m_workerStop.load(std::memory_order_acquire)) {
990 if (m_crashed.load(std::memory_order_acquire)) {
991 std::this_thread::sleep_for(std::chrono::milliseconds(5));
992 continue;
993 }
994
995 // Forward any queued parameter changes to the child first, so a change
996 // requested before this block takes effect for it. Runs every iteration
997 // (even with no audio pending) so knob moves apply while transport is
998 // stopped too (#238).
999 drainParamQueueToChild();
1000 if (m_crashed.load(std::memory_order_acquire)) {
1001 continue;
1002 }
1003
1004 uint8_t pendingReady = 2;
1005 if (!m_pendingState.compare_exchange_strong(pendingReady, 3, std::memory_order_acq_rel)) {
1006 std::this_thread::sleep_for(std::chrono::milliseconds(1));
1007 continue;
1008 }
1009
1010 const uint32_t frames = m_pendingFrames.load(std::memory_order_acquire);
1011 const size_t midiBytes = m_pendingMidiBytes.load(std::memory_order_acquire);
1012 const size_t sampleCount = static_cast<size_t>(frames) * m_transportChannels;
1013 if (sampleCount <= m_workerInput.size()) {
1014 std::copy_n(m_pendingInput.data(), sampleCount, m_workerInput.data());
1015 }
1016 if (midiBytes <= m_workerMidiData.size()) {
1017 std::copy_n(m_pendingMidiData.data(), midiBytes, m_workerMidiData.data());
1018 }
1019 m_pendingState.store(0, std::memory_order_release);
1020
1021 if (sampleCount > m_workerInput.size() || midiBytes > m_workerMidiData.size() ||
1022 !processBlockInHelper(m_workerInput, m_transportChannels, frames, m_workerMidiData, midiBytes,
1023 m_workerOutput)) {
1024 markCrashed();
1025 continue;
1026 }
1027
1028 uint8_t readyEmpty = 0;
1029 if (m_readyState.compare_exchange_strong(readyEmpty, 1, std::memory_order_acq_rel)) {
1030 const size_t outputSamples = std::min(sampleCount, m_workerOutput.size());
1031 std::copy_n(m_workerOutput.data(), outputSamples, m_readyOutput.data());
1032 m_readyFrames.store(frames, std::memory_order_release);
1033 m_readyState.store(2, std::memory_order_release);
1034 // Published last, so an observer that sees this advance is guaranteed
1035 // to find the result readable rather than half-written.
1036 m_processedBlocks.fetch_add(1, std::memory_order_acq_rel);
1037 }
1038 }

1041bool OutOfProcessPluginInstance::processBlockInHelper(const std::vector<float>& input, uint32_t channels,
1042 uint32_t frames, const std::vector<uint8_t>& midiData,
1043 size_t midiBytes, std::vector<float>& output) {
1044 const size_t sampleCount = static_cast<size_t>(channels) * frames;
1045 if (sampleCount > input.size()) {
1046 return false;
1047 }
1048 std::ostringstream command;
1049 if (midiBytes > 0) {
1050 command << "PROCESSMIDI " << channels << " " << frames << " "
1051 << hexEncodeBytes(input.data(), sampleCount * sizeof(float)) << " "
1052 << hexEncodeBytes(midiData.data(), midiBytes);
1053 } else {
1054 command << "PROCESS " << channels << " " << frames << " "
1055 << hexEncodeBytes(input.data(), sampleCount * sizeof(float));
1056 }
1057 std::string response;
1058 if (!sendCommand(command.str(), &response)) {
1059 return false;
1060 }
1061 constexpr const char* kPrefix = "OK ";
1062 if (response.compare(0, 3, kPrefix) != 0) {
1063 return false;
1064 }
1065 return hexDecodeBytes(response.substr(3), output) && output.size() >= sampleCount;
1066}
