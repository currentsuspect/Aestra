25std::string UIMixerSendList::formatLevel(float db) {
26 if (!std::isfinite(db) || db <= -90.0f)
27 return "\xE2\x88\x92\xE2\x88\x9E"; // minus infinity
28 char buf[16];
29 if (std::fabs(db) < 0.05f)
30 return "0";
31 if (std::fabs(db) >= 9.95f)
32 std::snprintf(buf, sizeof(buf), "%d", static_cast<int>(std::lround(std::fabs(db))));
33 else
34 std::snprintf(buf, sizeof(buf), "%.1f", std::fabs(db));
35 return std::string(db < 0.0f ? "\xE2\x88\x92" : "+") + buf;
36}

50void UIMixerSendList::onRender(NUIRenderer& renderer) {
51 const auto b = getBounds();
52 if (b.isEmpty() || m_rows.empty())
53 return;
54 auto& theme = NUIThemeManager::getInstance();
55 const NUIColor text = theme.getColor("textPrimary");
56 const NUIColor muted = theme.getColor("textSecondary");
57 const NUIColor amber = theme.getColor("warning");
58
59 const int capacity = std::max(1, visibleRowCapacity());
60 const bool overflow = static_cast<int>(m_rows.size()) > capacity;
61 const int shown = overflow ? std::max(1, capacity - 1) : static_cast<int>(m_rows.size());
62 float y = b.y + 1.0f;
63 for (int i = 0; i < shown; ++i) {
64 const Row& row = m_rows[static_cast<size_t>(i)];
65 const float alpha = row.muted ? 0.4f : 1.0f;
66 const float cy = y + kRowH * 0.5f;
67 if (row.sidechain) {
68 // An amber diamond: outline only, rotated square.
69 const float s = 4.0f;
70 renderer.drawLine({b.x + 8.0f, cy - s}, {b.x + 8.0f + s, cy}, 1.4f, amber.withAlpha(alpha));
71 renderer.drawLine({b.x + 8.0f + s, cy}, {b.x + 8.0f, cy + s}, 1.4f, amber.withAlpha(alpha));
72 renderer.drawLine({b.x + 8.0f, cy + s}, {b.x + 8.0f - s, cy}, 1.4f, amber.withAlpha(alpha));
73 renderer.drawLine({b.x + 8.0f - s, cy}, {b.x + 8.0f, cy - s}, 1.4f, amber.withAlpha(alpha));
74 } else {
75 renderer.fillRect(NUIRect{b.x + 5.0f, cy - 3.0f, 6.0f, 6.0f}, row.color.withAlpha(alpha));
76 }
77 const std::string level = row.sidechain ? std::string("SC") : formatLevel(row.levelDb);
78 const float levelW = renderer.measureText(level, kFont).width;
79 const float textX = b.x + 16.0f;
80 const float avail = std::max(1.0f, b.right() - 6.0f - levelW - 4.0f - textX);
81 std::string name = row.name;
82 bool cut = false;
83 while (name.size() > 1 && renderer.measureText(name, kFont).width > avail) {
84 name.pop_back();
85 cut = true;
86 }
87 if (cut)
88 name += "\xE2\x80\xA6";
89 const float ty = y + (kRowH - kFont) * 0.5f - 1.0f;
90 renderer.drawText(name, {textX, ty}, kFont, text.withAlpha(0.85f * alpha));
91 renderer.drawText(level, {b.right() - 6.0f - levelW, ty}, kFont, muted.withAlpha(0.85f * alpha));
92 y += kRowH;
93 }
94 if (overflow) {
95 renderer.drawText("+" + std::to_string(static_cast<int>(m_rows.size()) - shown) + " more",
96 {b.x + 16.0f, y + (kRowH - kFont) * 0.5f - 1.0f}, kFont, muted.withAlpha(0.7f));
97 }
98}

100bool UIMixerSendList::onMouseEvent(const NUIMouseEvent& event) {
101 if (!isVisible() || !isEnabled() || m_rows.empty())
102 return false;
103 const bool inside = getBounds().contains(event.position);
104 if (event.pressed && event.button == NUIMouseButton::Left && inside) {
105 m_pressed = true;
106 return true;
107 }
108 if (event.released && event.button == NUIMouseButton::Left) {
109 const bool fire = m_pressed && inside;
110 m_pressed = false;
111 if (fire && onClicked)
112 onClicked();
113 return fire;
114 }
115 return false;
116}
