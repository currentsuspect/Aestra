36// Mangled RTTI name, trimmed of the Itanium "N8AestraUI...E" noise for reading.
37std::string typeName(const NUIComponent& component) {
38 std::string name = typeid(component).name();
39 for (const char* prefix : {"N8AestraUI", "N6Aestra5Audio", "N6Aestra"}) {
40 if (name.rfind(prefix, 0) == 0) name = name.substr(std::char_traits<char>::length(prefix));
41 }
42 size_t i = 0;
43 while (i < name.size() && name[i] >= '0' && name[i] <= '9') ++i; // length prefix
44 name = name.substr(i);
45 if (!name.empty() && name.back() == 'E') name.pop_back();
46 return name;
47}

49template <typename T>
50std::string topN(const std::map<std::string, T>& values, size_t n, double divisor, const char* format) {
51 std::vector<std::pair<double, std::string>> ranked;
52 ranked.reserve(values.size());
53 for (const auto& [key, value] : values) ranked.push_back({static_cast<double>(value) / divisor, key});
54 std::sort(ranked.rbegin(), ranked.rend());
55 std::string out;
56 std::array<char, kLineBuffer> buf{};
57 for (size_t i = 0; i < ranked.size() && i < n; ++i) {
58 std::snprintf(buf.data(), buf.size(), format, ranked[i].second.c_str(), ranked[i].first);
59 out += buf.data();
60 }
61 return out;
62}

92void frameWorkEnd(bool presented) {
93 if (level() <= 0) return;
94 auto& f = frameWork();
95 const auto now = std::chrono::steady_clock::now();
96 if (presented) f.ms.push_back(std::chrono::duration<double, std::milli>(now - f.workStart).count());
97 if (now - f.windowStart < std::chrono::seconds(1)) return;
98 f.windowStart = now;
99 if (f.ms.empty()) return;
100
101 std::sort(f.ms.begin(), f.ms.end());
102 double sum = 0.0;
103 for (const double ms : f.ms) sum += ms;
104 const auto over = [&](double budget) {
105 return static_cast<long>(std::count_if(f.ms.begin(), f.ms.end(), [budget](double ms) { return ms > budget; }));
106 };
107 const std::size_t p95 = (f.ms.size() * 95 + 99) / 100 - 1; // nearest rank: ceil(0.95 n) - 1
108 std::array<char, 256> line{};
109 std::snprintf(line.data(), line.size(),
110 "[FrameStats] frames=%zu work avg=%.2fms p95=%.2fms max=%.2fms over16.7=%ld over33=%ld",
111 f.ms.size(), sum / static_cast<double>(f.ms.size()), f.ms[p95], f.ms.back(), over(16.7), over(33.3));
112 Aestra::Log::info(line.data());
113 if (enabled()) {
114 std::istringstream report(takeReport(f.ms.size()));
115 for (std::string detail; std::getline(report, detail);) Aestra::Log::info(detail);
116 }
117 f.ms.clear();
118}

135std::string takeReport(std::size_t presentedFrames) {
136 auto& c = counters();
137 const double frames = static_cast<double>(std::max<size_t>(1, presentedFrames));
138 std::string report = "[FrameWhy]" + topN(c.presentReasons, kTopN, 1.0, " %s=%.0f");
139 report += "\n[DirtyWho] per second:" + topN(c.dirtyOrigins, kTopN, 1.0, " %s=%.0f");
140 report += "\n[RenderSelf] ms/frame:" + topN(c.renderSelfMs, kTopN, frames, " %s=%.2f");
141 std::array<char, kLineBuffer> phase{};
142 std::snprintf(phase.data(), phase.size(), "\n[Phase] ms/frame: tree=%.2f submit=%.2f swap=%.2f", c.phaseMs[0] / frames,
143 c.phaseMs[1] / frames, c.phaseMs[2] / frames);
144 report += phase.data();
145 c = Counters{};
146 return report;
147}
