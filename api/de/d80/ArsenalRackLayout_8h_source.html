50 static ArsenalRackLayout compute(const std::vector<Audio::UnitGroup>& groups, const std::vector<float>& extras = {}) {
51 ArsenalRackLayout out;
52 static constexpr Audio::UnitGroup order[] = {Audio::UnitGroup::Drums, Audio::UnitGroup::Synth,
53 Audio::UnitGroup::Audio, Audio::UnitGroup::Unknown};
54 float y = 0.0f;
55 for (const auto g : order) {
56 int count = 0;
57 for (const auto& rg : groups) {
58 // Anything that is not one of the three named groups lands under "Other".
59 const bool named = rg == Audio::UnitGroup::Drums || rg == Audio::UnitGroup::Synth || rg == Audio::UnitGroup::Audio;
60 if (rg == g || (g == Audio::UnitGroup::Unknown && !named)) ++count;
61 }
62 if (count == 0) continue;
63 ArsenalRackItem header;
64 header.kind = ArsenalRackItem::Kind::Header;
65 header.group = g;
66 header.count = count;
67 header.y = y;
68 header.h = kHeaderH;
69 out.items.push_back(header);
70 y += kHeaderH;
71 for (size_t i = 0; i < groups.size(); ++i) {
72 const auto rg = groups[i];
73 const bool named = rg == Audio::UnitGroup::Drums || rg == Audio::UnitGroup::Synth || rg == Audio::UnitGroup::Audio;
74 if (!(rg == g || (g == Audio::UnitGroup::Unknown && !named))) continue;
75 ArsenalRackItem row;
76 row.kind = ArsenalRackItem::Kind::Row;
77 row.group = g;
78 row.rowIndex = i;
79 row.y = y;
80 row.h = kRowH;
81 row.extraH = i < extras.size() ? extras[i] : 0.0f;
82 out.items.push_back(row);
83 out.displayOrder.push_back(i);
84 y += kRowH + row.extraH + kRowGap;
85 }
86 }
87 out.contentHeight = y;
88 return out;
89 }

102 size_t dropSlotForY(float y) const {
103 size_t slot = 0;
104 for (const auto& item : items) {
105 if (item.kind != ArsenalRackItem::Kind::Row) continue;
106 if (item.y + item.h * 0.5f < y) ++slot;
107 }
108 return slot;
109 }

114 static size_t finalIndexAfterMove(const std::vector<Audio::UnitID>& order, Audio::UnitID unit, size_t insertBefore) {
115 const auto it = std::find(order.begin(), order.end(), unit);
116 const size_t old = it == order.end() ? insertBefore : static_cast<size_t>(it - order.begin());
117 return insertBefore > old ? insertBefore - 1 : insertBefore;
118 }

121 size_t insertIndexForSlot(size_t slot, size_t rowCount) const {
122 return slot < displayOrder.size() ? displayOrder[slot] : rowCount;
123 }
