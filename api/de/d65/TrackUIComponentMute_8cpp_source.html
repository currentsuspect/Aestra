2// The Mute tool (owner, 2026-10-20: "I could not mute individual patterns ... without muting the whole track"; it takes
3// the place Erase had, since right-click and Delete already remove a clip). A left click on a clip mutes it, or unmutes
4// it if it was muted; holding the button and moving sets every clip the pointer crosses on this lane to the state the
5// first one went to, so a sweep never flips some back. Each change is its own undo step. Alt+click does the same once
6// with any tool active. Kept out of TrackUIComponent.cpp, which is at its size pin.

17bool TrackUIComponent::handleMuteTool(const AestraUI::NUIMouseEvent& event) {
18 auto* manager = dynamic_cast<TrackManagerUI*>(getParent());
19 const bool automation = m_playlistMode == PlaylistMode::Automation;
20 const bool toolActive = manager && manager->getCurrentTool() == PlaylistTool::Mute && !automation;
21 const bool leftPress = event.pressed && event.button == AestraUI::NUIMouseButton::Left;
22 const bool altClick = manager && !automation && leftPress && (event.modifiers & AestraUI::NUIModifiers::Alt);
23 if (!toolActive && !altClick) {
24 m_muteStroke = false;
25 return false;
26 }
27 if (event.released && event.button == AestraUI::NUIMouseButton::Left) {
28 const bool wasMuting = m_muteStroke;
29 m_muteStroke = false;
30 m_muteStrokeDone.clear();
31 return wasMuting;
32 }
33 if (!leftPress && !m_muteStroke) return false;
34 if (leftPress) {
35 // The header keeps its own buttons: a press there is not a mute.
36 const float controlsWidth = AestraUI::NUIThemeManager::getInstance().getLayoutDimensions().trackControlsWidth;
37 if (!getBounds().contains(event.position) || event.position.x < getBounds().x + controlsWidth) return false;
38 m_muteStroke = toolActive; // Alt+click is one click, not a stroke
39 m_muteStrokeDone.clear();
40 }
41
42 ClipInstanceID hit;
43 for (const auto& [clipId, clipBounds] : m_allClipBounds) {
44 if (clipBounds.contains(event.position)) {
45 hit = clipId;
46 break;
47 }
48 }
49 if (!hit.isValid()) return true; // a press on empty grid with the tool active is still the tool's
50 if (std::find(m_muteStrokeDone.begin(), m_muteStrokeDone.end(), hit) != m_muteStrokeDone.end()) return true;
51
52 if (m_muteStrokeDone.empty()) m_muteStrokeTarget = !manager->isClipMuted(hit); // the first clip decides the sweep
53 manager->toggleClipMute(hit, m_muteStrokeTarget);
54 m_muteStrokeDone.push_back(hit);
55 return true;
56}
