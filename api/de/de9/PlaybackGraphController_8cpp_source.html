51void PlaybackGraphController::rebuildGraph(double sampleRate) {

53 return;
54 }
55
56 auto graph = AudioGraphBuilder::buildFromTrackManager(*m_trackManager);
57 m_engine->setGraph(graph);

59 if (auto slotMap = m_trackManager->getChannelSlotMapShared()) {

65
66 m_graphGeneration.fetch_add(1, std::memory_order_release);
