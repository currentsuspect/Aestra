68void SettingsForm::layout(const AestraUI::NUIRect& area) {
69 m_area = area;
70 const float maxScroll = std::max(0.0f, contentHeight() - area.height);
71 m_scroll = std::clamp(m_scroll, 0.0f, maxScroll);
72
73 const float x = area.x + kSidePadding;
74 const float width = std::max(0.0f, area.width - kSidePadding * 2.0f);
75 float y = area.y + 6.0f - m_scroll;
76 for (auto& item : m_items) {
77 item.bounds = AestraUI::NUIRect(x, y, width, item.height);
78 if (item.control) {
79 if (item.kind == Kind::Block) {
80 item.controlBounds = item.bounds;
81 } else {
82 const float cw = std::min(item.controlWidth, std::max(0.0f, width - kLabelWidth));
83 item.controlBounds =
84 AestraUI::NUIRect(x + kLabelWidth, y + (item.height - kControlHeight) * 0.5f, cw, kControlHeight);
85 }
86 item.control->setBounds(item.controlBounds);
87 // A control scrolled out of the area must not take clicks; partly visible ones are clipped when drawn.
88 const bool inside = item.controlBounds.bottom() > area.y && item.controlBounds.y < area.bottom();
89 item.control->setVisible(inside);
90 }
91 y += item.height;
92 }
93}

95void SettingsForm::paint(AestraUI::NUIRenderer& renderer) const {
96 auto& theme = AestraUI::NUIThemeManager::getInstance();
97 const auto& props = theme.getCurrentTheme();
98 const AestraUI::NUIColor caption = theme.getColor("textMuted");
99 const AestraUI::NUIColor label = theme.getColor("textPrimary").withAlpha(0.88f);
100 const AestraUI::NUIColor hint = theme.getColor("textMuted");
101 const AestraUI::NUIColor line = theme.getColor("border").withAlpha(0.55f);
102 const float captionFont = props.fontSizeXS;
103 const float labelFont = 13.0f;
104 const float hintFont = 11.5f;
105
106 renderer.setClipRect(m_area);
107 for (const auto& item : m_items) {
108 const auto& b = item.bounds;
109 if (b.bottom() < m_area.y || b.y > m_area.bottom()) {
110 continue;
111 }
112 if (item.kind == Kind::Section) {
113 AestraUI::NUIRect text(b.x, b.bottom() - kCaptionHeight, b.width, kCaptionHeight - 6.0f);
114 renderer.drawText(item.text, {b.x, std::round(renderer.calculateTextY(text, captionFont))}, captionFont,
115 caption);
116 continue;
117 }
118 // The board's hairline sits on top of every row.
119 renderer.drawLine({b.x, b.y + 0.5f}, {b.right(), b.y + 0.5f}, 1.0f, line);
120 if (item.kind == Kind::Block) {
121 continue;
122 }
123 renderer.drawText(item.text, {b.x, std::round(renderer.calculateTextY(b, labelFont))}, labelFont, label);
124 if (!item.hint.empty()) {
125 const float hx = (item.control ? item.controlBounds.right() : b.x + kLabelWidth) + kHintGap;
126 renderer.drawText(item.hint, {hx, std::round(renderer.calculateTextY(b, hintFont))}, hintFont, hint);
127 }
128 }
129 renderer.clearClipRect();
130}

132bool SettingsForm::onMouseWheel(const AestraUI::NUIMouseEvent& event) {
133 if (event.wheelDelta == 0.0f || !m_area.contains(event.position)) {
134 return false;
135 }
136 const float maxScroll = std::max(0.0f, contentHeight() - m_area.height);
137 if (maxScroll <= 0.0f) {
138 return false;
139 }
140 const float before = m_scroll;
141 m_scroll = std::clamp(m_scroll - event.wheelDelta * kRowHeight, 0.0f, maxScroll);
142 if (m_scroll == before) {
143 return false;
144 }
145 layout(m_area);
146 return true;
147}
