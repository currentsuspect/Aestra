2// MasterSafetyLimiter — final output safety limiter for the master bus.
3// Uses a light soft-knee region ahead of a hard safety ceiling so accidental
4// spikes get caught before they turn into speaker- or ear-hostile blasts.
5
6#pragma once
7
8#include <algorithm>
9#include <cmath>
10
11namespace Aestra {
12namespace Audio {
13
14class MasterSafetyLimiter {
15public:
16 MasterSafetyLimiter() = default;
17
18 // Process a stereo frame in-place. Returns true if limiting occurred.
19 bool process(double& L, double& R) {
20 const double inL = L;
21 const double inR = R;
22 L = apply(L);
23 R = apply(R);
24 const bool limited = (L != inL) || (R != inR);
25 return limited;
26 }
27
28 bool process(float& L, float& R) {
29 const float inL = L;
30 const float inR = R;
31 L = apply(L);
32 R = apply(R);
33 return (L != inL) || (R != inR);
34 }
35
36 void reset() {}
37
38 // Constants -- do not change without updating tests.
39 static constexpr float kKneeStart = 0.98f;
40 static constexpr float kCeiling = 0.9997f;
41 static constexpr float kKneeRange = kCeiling - kKneeStart;
42
43 // Compilation-compatible aliases. Values intentionally reflect the
44 // reshaped limiter, not the old 0.85/0.95 thresholds.
45 static constexpr double SOFT_KNEE_START = kKneeStart;
46 static constexpr double OUTPUT_CEILING = kCeiling;
47
48 static float softKnee(float x) {
49 const float t = (x - kKneeStart) / kKneeRange;
50 const float shaped = t * t * (3.0f - 2.0f * t);
51 return kKneeStart + shaped * kKneeRange;
52 }
53
54 static float apply(float s) {
55 if (!std::isfinite(s)) {
56 return 0.0f;
57 }
58
59 const float a = std::abs(s);
60 if (a <= kKneeStart) {
61 return s;
62 }
63 if (a >= kCeiling) {
64 return std::copysign(kCeiling, s);
65 }
66 return std::copysign(std::min(softKnee(a), a), s);
67 }
68
69 static double apply(double s) {
70 if (!std::isfinite(s)) {
71 return 0.0;
72 }
73
74 const double a = std::abs(s);
75 if (a <= static_cast<double>(kKneeStart)) {
76 return s;
77 }
78 if (a >= static_cast<double>(kCeiling)) {
79 return std::copysign(static_cast<double>(kCeiling), s);
80 }
81
82 const double t = (a - static_cast<double>(kKneeStart)) / static_cast<double>(kKneeRange);
83 const double shaped = t * t * (3.0 - 2.0 * t);
84 const double soft = static_cast<double>(kKneeStart) + shaped * static_cast<double>(kKneeRange);
85 return std::copysign(std::min(soft, a), s);
86 }
87};
