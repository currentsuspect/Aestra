38AutomationTarget TrackManagerUI::automationLaneTarget(PlaylistLaneID laneId) const {
39 const auto* view = findAutomationLaneView(laneId);
40 if (view && view->chosen) return view->target;
41 // Nothing picked yet: the curve the lane already has (a pan-only lane edits its pan), else Volume.
42 const auto* lane = m_trackManager ? m_trackManager->getPlaylistModel().getLane(laneId) : nullptr;
43 if (lane && !lane->automationCurves.empty()) return lane->automationCurves.front().getAutomationTarget();
44 return AutomationTarget::Volume;
45}

47void TrackManagerUI::showAutomationLane(PlaylistLaneID laneId, AutomationTarget target, bool deferred) {
48 auto& view = automationLaneView(laneId);
49 const bool changed = !view.shown || view.target != target || !view.chosen;
50 view.shown = true;
51 view.target = target;
52 view.chosen = true;
53 if (!changed) return;
54 if (deferred) {
55 m_needsTrackRefresh = true; // rebuilt on the next paint (TrackManagerUIRender)
56 invalidateCache();
57 setDirty(true);
58 return;
59 }
60 refreshTracks();
61}

63void TrackManagerUI::hideAutomationLane(PlaylistLaneID laneId, bool deferred) {
64 auto& view = automationLaneView(laneId);
65 if (!view.shown) return;
66 view.shown = false; // the curve stays on the lane; the track row marks it (setHasHiddenAutomation)
67 if (deferred) {
68 m_needsTrackRefresh = true;
69 invalidateCache();
70 setDirty(true);
71 return;
72 }
73 refreshTracks();
74}

76void TrackManagerUI::toggleAutomationForSelectedTracks() {
77 std::vector<PlaylistLaneID> lanes;
78 for (const auto& row : m_trackUIComponents) {
79 if (row && !row->isAutomationRow() && m_trackSelection.contains(row->getLaneId())) {
80 lanes.push_back(row->getLaneId());
81 }
82 }
83 if (lanes.empty()) return;
84 // One press opens them all, or closes them all when every selected track already shows its lane.
85 const bool allShown = std::all_of(lanes.begin(), lanes.end(),
86 [this](const PlaylistLaneID& id) { return isAutomationLaneShown(id); });
87 for (const auto& id : lanes) automationLaneView(id).shown = !allShown;
88 refreshTracks();
89}

102std::shared_ptr<TrackUIComponent> TrackManagerUI::makeAutomationRow(const std::shared_ptr<TrackUIComponent>& trackRow,
103 AutomationTarget target) {
104 const PlaylistLaneID laneId = trackRow->getLaneId();
105 auto row = std::make_shared<TrackUIComponent>(laneId, nullptr, m_trackManager.get());
106 row->setOnCacheInvalidationNeeded([this]() { this->invalidateCache(); });
107 row->setOnTrackSelected([this](TrackUIComponent* comp, TrackSelectionIntent intent) { selectTrack(comp, intent); });
108 row->setPixelsPerBeat(m_pixelsPerBeat);
109 row->setBeatsPerBar(m_beatsPerBar);
110 row->setTimelineScrollOffset(m_timelineScrollOffset);
111 row->setSnapSetting(m_snapSetting);
112 row->setSnapEnabled(m_snapEnabled);
113 row->setPlatformBridge(m_window);
114 row->setOnEditedAutomationTargetChanged([this, laneId](AutomationTarget changed) {
115 auto& view = automationLaneView(laneId);
116 view.target = changed;
117 view.chosen = true;
118 });
119 // Both arrive from inside the row's own handlers, so the rebuild waits for the next paint.
120 row->setAutomationRow(
121 target, [this, laneId](AutomationTarget picked) { showAutomationLane(laneId, picked, true); },
122 [this, laneId]() { hideAutomationLane(laneId, true); });
123 row->updateUI();
124 return row;
125}
