36AESTRA_AVX2_TARGET
37inline void sincDotProductAVX2(const float* c0,
38 const float* c1,
39 float alpha,
40 const float* samples, // Interleaved L/R stereo
41 float& sumL, float& sumR) {
42 __m256 vSumL = _mm256_setzero_ps();
43 __m256 vSumR = _mm256_setzero_ps();
44 __m256 vAlpha = _mm256_set1_ps(alpha);
45
46 for (int i = 0; i < 64; i += 8) {
47 // Load base and adjacent phase coefficients
48 __m256 vC0 = _mm256_loadu_ps(&c0[i]);
49 __m256 vC1 = _mm256_loadu_ps(&c1[i]);
50
51 // Phase interpolation: coeff = c0 + alpha * (c1 - c0)
52 __m256 vCoeff = _mm256_fmadd_ps(_mm256_sub_ps(vC1, vC0), vAlpha, vC0);
53
54 // Optimize: Load 16 samples (8 stereo pairs) using 2 vector loads
55 __m256 ra = _mm256_loadu_ps(&samples[i * 2]);
56 __m256 rb = _mm256_loadu_ps(&samples[i * 2 + 8]);
57
58 // De-interleave L and R channels
59 __m256 vMixL = _mm256_shuffle_ps(ra, rb, _MM_SHUFFLE(2, 0, 2, 0));
60 __m256 vMixR = _mm256_shuffle_ps(ra, rb, _MM_SHUFFLE(3, 1, 3, 1));
61
62 __m256 vL = _mm256_castpd_ps(_mm256_permute4x64_pd(_mm256_castps_pd(vMixL), _MM_SHUFFLE(3, 1, 2, 0)));
63 __m256 vR = _mm256_castpd_ps(_mm256_permute4x64_pd(_mm256_castps_pd(vMixR), _MM_SHUFFLE(3, 1, 2, 0)));
64
65 vSumL = _mm256_fmadd_ps(vL, vCoeff, vSumL);
66 vSumR = _mm256_fmadd_ps(vR, vCoeff, vSumR);
67 }
68
69 alignas(32) float hL[8], hR[8];
70 _mm256_store_ps(hL, vSumL);
71 _mm256_store_ps(hR, vSumR);
72
73 sumL = hL[0] + hL[1] + hL[2] + hL[3] + hL[4] + hL[5] + hL[6] + hL[7];
74 sumR = hR[0] + hR[1] + hR[2] + hR[3] + hR[4] + hR[5] + hR[6] + hR[7];
75}

81AESTRA_AVX2_TARGET
82inline void sincDotProductAVX2_Reversed(const float* c0,
83 const float* c1,
84 float alpha,
85 const float* samples, float& sumL, float& sumR) {
86 __m256 vSumL = _mm256_setzero_ps();
87 __m256 vSumR = _mm256_setzero_ps();
88 __m256 vAlpha = _mm256_set1_ps(alpha);
89
90 // Reverse mask: 7, 6, 5, 4, 3, 2, 1, 0
91 const __m256i vRevMask = _mm256_set_epi32(0, 1, 2, 3, 4, 5, 6, 7);
92
93 for (int i = 0; i < 64; i += 8) {
94 // Load coeffs from the end backwards
95 const float* c0Ptr = &c0[64 - 8 - i];
96 const float* c1Ptr = &c1[64 - 8 - i];
97 __m256 vC0 = _mm256_loadu_ps(c0Ptr);
98 __m256 vC1 = _mm256_loadu_ps(c1Ptr);
99
100 // Reverse the coefficients in the register
101 vC0 = _mm256_permutevar8x32_ps(vC0, vRevMask);
102 vC1 = _mm256_permutevar8x32_ps(vC1, vRevMask);
103
104 // Phase interpolation
105 __m256 vCoeff = _mm256_fmadd_ps(_mm256_sub_ps(vC1, vC0), vAlpha, vC0);
106
107 // Load pairs (same as forward)
108 __m256 ra = _mm256_loadu_ps(&samples[i * 2]);
109 __m256 rb = _mm256_loadu_ps(&samples[i * 2 + 8]);
110
111 __m256 vMixL = _mm256_shuffle_ps(ra, rb, _MM_SHUFFLE(2, 0, 2, 0));
112 __m256 vMixR = _mm256_shuffle_ps(ra, rb, _MM_SHUFFLE(3, 1, 3, 1));
113
114 __m256 vL = _mm256_castpd_ps(_mm256_permute4x64_pd(_mm256_castps_pd(vMixL), _MM_SHUFFLE(3, 1, 2, 0)));
115 __m256 vR = _mm256_castpd_ps(_mm256_permute4x64_pd(_mm256_castps_pd(vMixR), _MM_SHUFFLE(3, 1, 2, 0)));
116
117 vSumL = _mm256_fmadd_ps(vL, vCoeff, vSumL);
118 vSumR = _mm256_fmadd_ps(vR, vCoeff, vSumR);
119 }
120
121 alignas(32) float hL[8], hR[8];
122 _mm256_store_ps(hL, vSumL);
123 _mm256_store_ps(hR, vSumR);
124
125 sumL = hL[0] + hL[1] + hL[2] + hL[3] + hL[4] + hL[5] + hL[6] + hL[7];
126 sumR = hR[0] + hR[1] + hR[2] + hR[3] + hR[4] + hR[5] + hR[6] + hR[7];
127}

134inline void sincDotProductAVX2(const float* c0,
135 const float* c1,
136 float alpha,
137 const float* samples,
138 float& sumL,
139 float& sumR) {
140 sumL = 0.0f;
141 sumR = 0.0f;
142 for (int i = 0; i < 64; ++i) {
143 const float coeff = c0[i] + alpha * (c1[i] - c0[i]);
144 sumL += samples[i * 2] * coeff;
145 sumR += samples[i * 2 + 1] * coeff;
146 }
147}
