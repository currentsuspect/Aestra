20void PianoRollKeyLane::onRender(NUIRenderer& renderer) {
21 if (!isVisible()) return;
22 auto b = getBounds();
23 auto& themeManager = NUIThemeManager::getInstance();
24
25 renderer.setClipRect(b);
26
27 const auto laneBg = themeManager.getColor("backgroundSecondary").darkened(0.02f);
28 const auto naturalKey = themeManager.getColor("surfaceRaised").withAlpha(0.72f);
29 const auto naturalHover = themeManager.getColor("surfaceRaised").lightened(0.08f).withAlpha(0.92f);
30 const auto accidentalKey = themeManager.getColor("backgroundPrimary").darkened(0.08f);
31 const auto accidentalHover = themeManager.getColor("accentPrimary").darkened(0.35f).withAlpha(0.92f);
32 const auto separator = themeManager.getColor("border").withAlpha(0.34f);
33 const auto rootAccent = themeManager.getColor("accentPrimary").withAlpha(0.86f);
34 const auto naturalText = themeManager.getColor("textPrimary").withAlpha(0.72f);
35
36 int startPitch = 127 - static_cast<int>((scrollY_) / keyHeight_);
37 int endPitch = 127 - static_cast<int>((scrollY_ + b.height) / keyHeight_);
38
39 // Expand range for safety margin (2 extra keys on each end)
40 startPitch = std::clamp(startPitch + 2, 0, 127);
41 endPitch = std::clamp(endPitch - 2, 0, 127);
42
43 renderer.fillRect(b, laneBg);
44
45 for (int p = startPitch; p >= endPitch; --p) {
46 float worldY = (127 - p) * keyHeight_;
47 float y = b.y + worldY - scrollY_;
48 const bool accidental = isBlackKey(p);
49 const bool hovered = p == hoveredKey_ || p == previewPitch_;
50 const float rightPad = 3.0f;
51 const float accidentalW = std::round(b.width * 0.68f);
52 NUIRect keyRect(b.x + 3.0f,
53 y + 1.0f,
54 accidental ? accidentalW : b.width - 3.0f - rightPad,
55 std::max(4.0f, keyHeight_ - 2.0f));
56
57 renderer.fillRoundedRect(keyRect,
58 accidental ? 2.0f : 3.0f,
59 hovered ? (accidental ? accidentalHover : naturalHover)
60 : (accidental ? accidentalKey : naturalKey));
61 renderer.strokeRoundedRect(keyRect, accidental ? 2.0f : 3.0f, 1.0f, separator);
62
63 if (!accidental) {
64 renderer.drawLine(NUIPoint(keyRect.x, keyRect.bottom()),
65 NUIPoint(keyRect.right(), keyRect.bottom()),
66 1.0f,
67 separator.withAlpha(0.52f));
68 }
69
70 if (keyHeight_ >= 15.0f && p % 12 == 0) {
71 const std::string label = getNoteLabel(p);
72 const float fontSize = 10.0f;
73 const auto textSize = renderer.measureText(label, fontSize);
74 const float textX = keyRect.right() - textSize.width - 8.0f;
75 const float textY = keyRect.y + (keyRect.height - textSize.height) * 0.5f;
76 renderer.drawText(label,
77 NUIPoint(textX, textY),
78 fontSize,
79 naturalText.lightened(0.12f));
80 }
81
82 if (p % 12 == 0) {
83 renderer.fillRoundedRect(NUIRect(b.x + 3.0f, y + 4.0f, 2.0f, std::max(2.0f, keyHeight_ - 8.0f)),
84 1.0f,
85 rootAccent);
86 }
87 }
88
89 renderer.drawLine(NUIPoint(b.right() - 0.5f, b.y),
90 NUIPoint(b.right() - 0.5f, b.bottom()),
91 1.0f,
92 separator.withAlpha(0.9f));
93
94 renderer.clearClipRect();
95}

97bool PianoRollKeyLane::onMouseEvent(const NUIMouseEvent& event) {
98 if (!getBounds().contains(event.position)) {
99 if (hoveredKey_ != -1) {
100 hoveredKey_ = -1;
101 if (onHoveredKeyChanged_) onHoveredKeyChanged_(-1);
102 repaint();
103 }
104 // A press that started on a key can be released outside the lane —
105 // still end the preview so the next press re-fires cleanly.
106 if (event.released && event.button == NUIMouseButton::Left) {
107 previewPitch_ = -1;
108 }
109 return false;
110 }
111
112 auto b = getBounds();
113 float localY = event.position.y - b.y + scrollY_;
114 int pitch = 127 - static_cast<int>(localY / keyHeight_);
115 pitch = std::clamp(pitch, 0, 127);
116
117 if (hoveredKey_ != pitch) {
118 hoveredKey_ = pitch;
119 if (onHoveredKeyChanged_) onHoveredKeyChanged_(pitch);
120 repaint();
121 }
122
123 if (m_isPlayingCallback && m_isPlayingCallback()) {
124 return NUIComponent::onMouseEvent(event);
125 }
126
127 if (event.pressed && event.button == NUIMouseButton::Left) {
128 previewPitch_ = pitch;
129 if (onPreviewNote_) {
130 onPreviewNote_(pitch, 100);
131 }
132 }
133 else if (event.released && event.button == NUIMouseButton::Left && previewPitch_ != -1) {
134 // The engine audition voice is one-shot (auto note-off, velocity
135 // clamped to >=1) — sending a velocity-0 "note-off" would retrigger a
136 // near-silent voice on top of it, so just clear the preview state.
137 previewPitch_ = -1;
138 }
139
140 return NUIComponent::onMouseEvent(event);
141}
