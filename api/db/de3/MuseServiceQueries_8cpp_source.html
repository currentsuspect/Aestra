30JSON trackToJson(const MixerChannel& channel, size_t index) {
31 JSON t = JSON::object();
32 t.set("index", JSON(static_cast<double>(index)));
33 t.set("id", JSON(static_cast<double>(channel.getChannelId())));
34 t.set("name", JSON(channel.getName()));
35 t.set("volume", JSON(static_cast<double>(channel.getVolume())));
36 t.set("pan", JSON(static_cast<double>(channel.getPan())));
37 t.set("muted", JSON(channel.isMuted()));
38 t.set("soloed", JSON(channel.isSoloed()));
39 return t;
40}

46std::optional<std::string> handleSessionVerbs(const RequestContext& ctx, const ResponseEnvelope& env) {
47 // Bound to the names the moved bodies already use, so no verb body had to be
48 // edited to make the move.
49 const double id = ctx.id;
50 const std::string& verb = ctx.verb;
51 JSON& request = *ctx.request;
52 TrackManager* const m_trackManager = ctx.trackManager;
53 AudioEngine* const m_engine = ctx.engine;
54 // A reference, not a pointer: get_project_load_report tests this for a value
55 // and then dereferences it, so it has to carry std::optional's own semantics.
56 // A pointer would be truthy whenever the address is non-null — including when
57 // the optional it points at is empty — which is not the same question.
58 const std::optional<JSON>& m_projectLoadReport = *ctx.projectLoadReport;
59
60 const auto makeOk = [&env]() { return env.ok(); };
61 const auto finish = [&env](JSON& response) { return env.finish(response); };
62
63 if (verb == "get_transport") {
64 JSON result = JSON::object();
65 if (m_engine) {
66 result.set("bpm", JSON(static_cast<double>(m_engine->getBPM())));
67 result.set("playing", JSON(m_engine->isTransportPlaying()));
68 result.set("positionSeconds", JSON(m_engine->getPositionSeconds()));
69 } else if (m_trackManager) {
70 result.set("bpm", JSON(m_trackManager->getTimelineClock().getCurrentTempo()));
71 result.set("playing", JSON(m_trackManager->isPlaying()));
72 result.set("positionSeconds", JSON(m_trackManager->getPosition()));
73 } else {
74 return makeError(id, "execution_error", "no engine or track manager", verb).toString();
75 }
76 JSON response = makeOk();
77 response.set("result", result);
78 return finish(response);
79 }
80
81 if (verb == "list_tracks") {
82 if (!m_trackManager) {
83 return makeError(id, "execution_error", "no track manager", verb).toString();
84 }
85 JSON tracks = JSON::array();
86 const size_t count = m_trackManager->getChannelCount();
87 for (size_t i = 0; i < count; ++i) {
88 if (const MixerChannel* ch = m_trackManager->getChannel(i)) {
89 tracks.push(trackToJson(*ch, i));
90 }
91 }
92 JSON result = JSON::object();
93 result.set("tracks", tracks);
94 JSON response = makeOk();
95 response.set("result", result);
96 return finish(response);
97 }
98
99 if (verb == "list_clips") {
100 if (!m_trackManager) {
101 return makeError(id, "execution_error", "no track manager", verb).toString();
102 }
103 auto& playlist = m_trackManager->getPlaylistModel();
104 JSON lanes = JSON::array();
105 const size_t laneCount = playlist.getLaneCount();
106 for (size_t i = 0; i < laneCount; ++i) {
107 PlaylistLaneID laneId = playlist.getLaneId(i);
108 PlaylistLane* lane = playlist.getLane(laneId);
109 if (!lane) continue;
110 JSON laneJson = JSON::object();
111 laneJson.set("index", JSON(static_cast<double>(i)));
112 laneJson.set("id", JSON(laneId.toString())); // full UUID — lossless
113 laneJson.set("name", JSON(lane->name));
114 JSON clips = JSON::array();
115 for (const auto& clip : lane->clips) {
116 JSON c = JSON::object();
117 c.set("id", JSON(clip.id.toString())); // full UUID — lossless
118 c.set("name", JSON(clip.name));
119 c.set("startBeat", JSON(clip.startBeat));
120 c.set("durationBeats", JSON(clip.durationBeats));
121 // 0 = not a pattern clip
122 c.set("pattern", JSON(static_cast<double>(clip.patternId.value)));
123 clips.push(c);
124 }
125 laneJson.set("clips", clips);
126 lanes.push(laneJson);
127 }
128 JSON result = JSON::object();
129 result.set("lanes", lanes);
130 JSON response = makeOk();
131 response.set("result", result);
132 return finish(response);
133 }
134
135 if (verb == "get_session_state") {
136 if (!m_trackManager) {
137 return makeError(id, "execution_error", "no track manager", verb).toString();
138 }
139 JSON result = JSON::object();
140
141 JSON transport = JSON::object();
142 if (m_engine) {
143 transport.set("bpm", JSON(static_cast<double>(m_engine->getBPM())));
144 transport.set("playing", JSON(m_engine->isTransportPlaying()));
145 } else {
146 transport.set("bpm", JSON(m_trackManager->getTimelineClock().getCurrentTempo()));
147 transport.set("playing", JSON(m_trackManager->isPlaying()));
148 }
149 result.set("transport", transport);
150
151 JSON tracks = JSON::array();
152 const size_t count = m_trackManager->getChannelCount();
153 for (size_t i = 0; i < count; ++i) {
154 if (const MixerChannel* ch = m_trackManager->getChannel(i)) {
155 tracks.push(trackToJson(*ch, i));
156 }
157 }
158 result.set("tracks", tracks);
159 result.set("laneCount",
160 JSON(static_cast<double>(m_trackManager->getPlaylistModel().getLaneCount())));
161 result.set("unitCount",
162 JSON(static_cast<double>(m_trackManager->getUnitManager().getUnitCount())));
163 result.set("canUndo", JSON(m_trackManager->getCommandHistory().canUndo()));
164
165 JSON response = makeOk();
166 response.set("result", result);
167 return finish(response);
168 }
169
170 if (verb == "list_units") {
171 if (!m_trackManager) {
172 return makeError(id, "execution_error", "no track manager", verb).toString();
173 }
174 auto& unitManager = m_trackManager->getUnitManager();
175 JSON units = JSON::array();
176 for (UnitID unitId : unitManager.getAllUnitIDs()) {
177 const UnitInfo* unit = unitManager.getUnit(unitId);
178 if (!unit) continue;
179 JSON u = JSON::object();
180 u.set("id", JSON(static_cast<double>(unit->id)));
181 u.set("name", JSON(unit->name));
182 u.set("type", JSON(unitTypeName(unit->type)));
183 u.set("enabled", JSON(unit->isEnabled));
184 u.set("gain", JSON(static_cast<double>(unit->gain)));
185 u.set("muted", JSON(unit->isMuted));
186 u.set("soloed", JSON(unit->isSolo));
187 u.set("defaultPatternId",
188 JSON(static_cast<double>(unit->defaultPatternId.value)));
189 u.set("mixerChannelId", JSON(static_cast<double>(unitManager.getUnitMixerChannel(unitId))));
190 // Compatibility metadata retained for older clients. It no longer controls audio routing.
191 u.set("timelineLane",
192 JSON(static_cast<double>(unitManager.getUnitTimelineLane(unitId))));
193 u.set("samplePath", JSON(unit->audioClipPath));
194 u.set("sampleDurationSeconds", JSON(unit->audioDurationSeconds));
195 units.push(u);
196 }
197 JSON result = JSON::object();
198 result.set("units", units);
199 JSON response = makeOk();
200 response.set("result", result);
201 return finish(response);
202 }
203 // No verb matched. In the single-file version this fell out of the if-chain
204 // into the next family's checks and eventually the mutation path, which is
205 // exactly what nullopt asks handleRequest to do.
206 return std::nullopt;
207}
