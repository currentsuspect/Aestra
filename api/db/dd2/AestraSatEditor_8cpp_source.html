32void drawArc(NUIRenderer& renderer, NUIPoint center, float radius, float startAngle, float endAngle, float thickness,
33 NUIColor color) {
34 if (endAngle - startAngle <= 0.001f)
35 return;
36 std::array<NUIPoint, 49> pts{};
37 const float div = static_cast<float>(pts.size() - 1);
38 for (size_t i = 0; i < pts.size(); ++i) {
39 const float t = static_cast<float>(i) / div;
40 const float a = startAngle + (endAngle - startAngle) * t;
41 pts[i] = {center.x + std::cos(a) * radius, center.y + std::sin(a) * radius};
42 }
43 renderer.drawPolyline(pts.data(), static_cast<int>(pts.size()), thickness, color);
44}

104void AestraSatEditor::drawKnob(NUIRenderer& renderer, const NUIRect& rect, uint32_t paramId, const char* label,
105 bool large) {
106 auto& theme = NUIThemeManager::getInstance();
107 const float value = m_instance ? m_instance->getParameter(paramId) : 0.0f;
108 const NUIPoint c = rect.center();
109 const float r = rect.width * 0.5f - 4.0f;
110 const float angle = kKnobStart + value * kKnobSweep;
111
112 renderer.fillCircle(c, r + 4.0f, insetSurface());
113 renderer.strokeCircle(c, r + 4.0f, 1.0f, editorInk(0.060f));
114
115 drawArc(renderer, c, r - 3.0f, kKnobStart, kKnobStart + kKnobSweep, large ? 4.0f : 3.0f,
116 editorNeutral(0.199f, 1.0f));
117 drawArc(renderer, c, r - 3.0f, kKnobStart, angle, large ? 4.0f : 3.0f, accent().withAlpha(0.92f));
118
119 const float needleLen = r - (large ? 14.0f : 10.0f);
120 const NUIPoint tip(c.x + std::cos(angle) * needleLen, c.y + std::sin(angle) * needleLen);
121 renderer.drawLine(c, tip, 2.0f, accent().withAlpha(0.85f));
122 renderer.fillCircle(tip, large ? 3.5f : 2.5f, accent());
123
124 const float wellR = r * (large ? 0.34f : 0.30f);
125 renderer.fillCircle(c, wellR, editorNeutral(0.045f, 0.96f));
126 renderer.strokeCircle(c, wellR, 1.2f, accent().withAlpha(0.36f));
127
128 renderer.drawTextCentered(label, NUIRect(rect.x, rect.bottom() + 4.0f, rect.width, 14.0f), 10.5f,
129 theme.getColor("textPrimary").withAlpha(0.95f));
130 const std::string valStr = m_instance ? m_instance->getParameterDisplay(paramId) : "";
131 renderer.drawTextCentered(valStr, NUIRect(rect.x - 10.0f, rect.bottom() + 19.0f, rect.width + 20.0f, 13.0f),
132 large ? 10.0f : 9.0f, accent().withAlpha(0.96f));
133}

193bool AestraSatEditor::onMouseEvent(const NUIMouseEvent& event) {
194 if (!isVisible())
195 return false;
196 if (AestraPanelWindow::onMouseEvent(event))
197 return true;
198 if (!m_instance)
199 return false;
200
201 const float mx = event.position.x;
202
203 // Bypass pill
204 if (m_bypassRect.contains(event.position) && event.pressed && event.button == NUIMouseButton::Left) {
205 const float cur = m_instance->getParameter(AestraSat::kBypass);
206 m_instance->setParameter(AestraSat::kBypass, cur > 0.5f ? 0.0f : 1.0f);
207 setDirty();
208 return true;
209 }
210 m_bypassHovered = m_bypassRect.contains(event.position);
211
212 // Mode segments
213 if (event.pressed && event.button == NUIMouseButton::Left) {
214 for (size_t i = 0; i < m_modeRects.size(); ++i) {
215 if (m_modeRects[i].contains(event.position)) {
216 m_instance->setParameter(AestraSat::kMode, AestraSat::normFromMode(static_cast<AestraSat::Mode>(i)));
217 setDirty();
218 return true;
219 }
220 }
221 }
222
223 // Knob vertical drag (rotary → cursor capture)
224 if (m_draggingParam >= 0) {
225 if (event.released) {
226 endKnobCapture();
227 m_draggingParam = -1;
228 return true;
229 }
230 if (event.button == NUIMouseButton::None) {
231 // Service-owned frame delta (up = increase); accumulates into the
232 // current value, so total travel matches the old absolute mapping.
233 const float step = knobDragStep(event, kDragRangePx);
234 const float cur = m_instance->getParameter(static_cast<uint32_t>(m_draggingParam));
235 m_instance->setParameter(static_cast<uint32_t>(m_draggingParam),
236 std::clamp(cur + step, 0.0f, 1.0f));
237 setDirty();
238 return true;
239 }
240 }
241 if (event.pressed && event.button == NUIMouseButton::Left) {
242 const struct {
243 NUIRect rect;
244 uint32_t param;
245 } knobs[] = {
246 {m_driveRect, AestraSat::kDrive},
247 {m_toneRect, AestraSat::kTone},
248 {m_outputRect, AestraSat::kOutput},
249 };
250 for (const auto& k : knobs) {
251 if (k.rect.contains(event.position)) {
252 m_draggingParam = static_cast<int>(k.param);
253 m_dragStartValue = m_instance->getParameter(k.param);
254 beginKnobCapture({k.rect.x + k.rect.width * 0.5f, k.rect.y + k.rect.height * 0.5f},
255 event.position);
256 return true;
257 }
258 }
259 }
260
261 // Mix slider drag
262 const NUIRect mixTrack(m_mixRect.x + 38.0f, m_mixRect.y + 6.0f, m_mixRect.width - 78.0f, m_mixRect.height - 12.0f);
263 if (m_draggingMix) {
264 if (event.released) {
265 m_draggingMix = false;
266 return true;
267 }
268 if (event.button == NUIMouseButton::None) {
269 const float t = std::clamp((mx - mixTrack.x) / mixTrack.width, 0.0f, 1.0f);
270 m_instance->setParameter(AestraSat::kMix, t);
271 setDirty();
272 return true;
273 }
274 }
275 if (event.pressed && event.button == NUIMouseButton::Left && mixTrack.contains(event.position)) {
276 m_draggingMix = true;
277 const float t = std::clamp((mx - mixTrack.x) / mixTrack.width, 0.0f, 1.0f);
278 m_instance->setParameter(AestraSat::kMix, t);
279 setDirty();
280 return true;
281 }
282
283 return consumeInsideBounds(event);
284}
