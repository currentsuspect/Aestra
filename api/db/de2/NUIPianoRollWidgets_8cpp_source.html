24static bool isBlackKey(int midiPitch) {
25 int m = midiPitch % 12;
26 return (m == 1 || m == 3 || m == 6 || m == 8 || m == 10);
27}

29static std::string getNoteLabel(int midiPitch) {
30 static const char* noteNames[] = { "C", "C#", "D", "D#", "E", "F", "F#", "G", "G#", "A", "A#", "B" };
31 int octave = (midiPitch / 12) - 2; // C3=60 standard
32 return std::string(noteNames[midiPitch % 12]) + std::to_string(octave);
33}

35static float safeClampScroll(float value, float upper) {
36 if (!std::isfinite(value) || !std::isfinite(upper) || upper <= 0.0f) {
37 return 0.0f;
38 }
39 if (value <= 0.0f) {
40 return 0.0f;
41 }
42 if (value >= upper) {
43 return upper;
44 }
45 return value;
46}

48static float safeClampRange(float value, float lower, float upper) {
49 if (!std::isfinite(value) || !std::isfinite(lower) || !std::isfinite(upper)) {
50 return std::max(0.0f, lower);
51 }
52 if (upper < lower) {
53 upper = lower;
54 }
55 if (value <= lower) {
56 return lower;
57 }
58 if (value >= upper) {
59 return upper;
60 }
61 return value;
62}

89void PianoRollKeyLane::onRender(NUIRenderer& renderer) {
90 if (!isVisible()) return;
91 auto b = getBounds();
92 auto& themeManager = NUIThemeManager::getInstance();
93
94 renderer.setClipRect(b);
95
96 const auto laneBg = themeManager.getColor("backgroundSecondary").darkened(0.02f);
97 const auto naturalKey = themeManager.getColor("surfaceRaised").withAlpha(0.72f);
98 const auto naturalHover = themeManager.getColor("surfaceRaised").lightened(0.08f).withAlpha(0.92f);
99 const auto accidentalKey = themeManager.getColor("backgroundPrimary").darkened(0.08f);
100 const auto accidentalHover = themeManager.getColor("accentPrimary").darkened(0.35f).withAlpha(0.92f);
101 const auto separator = themeManager.getColor("border").withAlpha(0.34f);
102 const auto rootAccent = themeManager.getColor("accentPrimary").withAlpha(0.86f);
103 const auto naturalText = themeManager.getColor("textPrimary").withAlpha(0.72f);
104
105 int startPitch = 127 - static_cast<int>((scrollY_) / keyHeight_);
106 int endPitch = 127 - static_cast<int>((scrollY_ + b.height) / keyHeight_);
107
108 // Expand range for safety margin (2 extra keys on each end)
109 startPitch = std::clamp(startPitch + 2, 0, 127);
110 endPitch = std::clamp(endPitch - 2, 0, 127);
111
112 renderer.fillRect(b, laneBg);
113
114 for (int p = startPitch; p >= endPitch; --p) {
115 float worldY = (127 - p) * keyHeight_;
116 float y = b.y + worldY - scrollY_;
117 const bool accidental = isBlackKey(p);
118 const bool hovered = p == hoveredKey_ || p == previewPitch_;
119 const float rightPad = 3.0f;
120 const float accidentalW = std::round(b.width * 0.68f);
121 NUIRect keyRect(b.x + 3.0f,
122 y + 1.0f,
123 accidental ? accidentalW : b.width - 3.0f - rightPad,
124 std::max(4.0f, keyHeight_ - 2.0f));
125
126 renderer.fillRoundedRect(keyRect,
127 accidental ? 2.0f : 3.0f,
128 hovered ? (accidental ? accidentalHover : naturalHover)
129 : (accidental ? accidentalKey : naturalKey));
130 renderer.strokeRoundedRect(keyRect, accidental ? 2.0f : 3.0f, 1.0f, separator);
131
132 if (!accidental) {
133 renderer.drawLine(NUIPoint(keyRect.x, keyRect.bottom()),
134 NUIPoint(keyRect.right(), keyRect.bottom()),
135 1.0f,
136 separator.withAlpha(0.52f));
137 }
138
139 if (keyHeight_ >= 15.0f && p % 12 == 0) {
140 const std::string label = getNoteLabel(p);
141 const float fontSize = 10.0f;
142 const auto textSize = renderer.measureText(label, fontSize);
143 const float textX = keyRect.right() - textSize.width - 8.0f;
144 const float textY = keyRect.y + (keyRect.height - textSize.height) * 0.5f;
145 renderer.drawText(label,
146 NUIPoint(textX, textY),
147 fontSize,
148 naturalText.lightened(0.12f));
149 }
150
151 if (p % 12 == 0) {
152 renderer.fillRoundedRect(NUIRect(b.x + 3.0f, y + 4.0f, 2.0f, std::max(2.0f, keyHeight_ - 8.0f)),
153 1.0f,
154 rootAccent);
155 }
156 }
157
158 renderer.drawLine(NUIPoint(b.right() - 0.5f, b.y),
159 NUIPoint(b.right() - 0.5f, b.bottom()),
160 1.0f,
161 separator.withAlpha(0.9f));
162
163 renderer.clearClipRect();
164}

166bool PianoRollKeyLane::onMouseEvent(const NUIMouseEvent& event) {
167 if (!getBounds().contains(event.position)) {
168 if (hoveredKey_ != -1) {
169 hoveredKey_ = -1;
170 if (onHoveredKeyChanged_) onHoveredKeyChanged_(-1);
171 repaint();
172 }
173 // A press that started on a key can be released outside the lane —
174 // still end the preview so the next press re-fires cleanly.
175 if (event.released && event.button == NUIMouseButton::Left) {
176 previewPitch_ = -1;
177 }
178 return false;
179 }
180
181 auto b = getBounds();
182 float localY = event.position.y - b.y + scrollY_;
183 int pitch = 127 - static_cast<int>(localY / keyHeight_);
184 pitch = std::clamp(pitch, 0, 127);
185
186 if (hoveredKey_ != pitch) {
187 hoveredKey_ = pitch;
188 if (onHoveredKeyChanged_) onHoveredKeyChanged_(pitch);
189 repaint();
190 }
191
192 if (m_isPlayingCallback && m_isPlayingCallback()) {
193 return NUIComponent::onMouseEvent(event);
194 }
195
196 if (event.pressed && event.button == NUIMouseButton::Left) {
197 previewPitch_ = pitch;
198 if (onPreviewNote_) {
199 onPreviewNote_(pitch, 100);
200 }
201 }
202 else if (event.released && event.button == NUIMouseButton::Left && previewPitch_ != -1) {
203 // The engine audition voice is one-shot (auto note-off, velocity
204 // clamped to >=1) — sending a velocity-0 "note-off" would retrigger a
205 // near-silent voice on top of it, so just clear the preview state.
206 previewPitch_ = -1;
207 }
208
209 return NUIComponent::onMouseEvent(event);
210}

244void PianoRollMinimap::setView(double start, double duration, bool preserveEdge) {
245 if (isDragging_) return;
246 if (preserveEdge) {
247 const double previousStart = startBeat_;
248 const double previousEnd = startBeat_ + viewDuration_;
249 viewDuration_ = std::clamp(duration, 0.25, totalDuration_);
250 if (std::abs(previousStart) <= 1e-9) {
251 startBeat_ = 0.0;
252 } else if (std::abs(previousEnd - totalDuration_) <= 1e-6) {
253 startBeat_ = std::max(0.0, totalDuration_ - viewDuration_);
254 } else {
255 startBeat_ = std::clamp(start, 0.0, std::max(0.0, totalDuration_ - viewDuration_));
256 }
257 } else {
258 viewDuration_ = std::clamp(duration, 0.25, totalDuration_);
259 startBeat_ = std::clamp(start, 0.0, std::max(0.0, totalDuration_ - viewDuration_));
260 }
261 repaint();
262}

290void PianoRollMinimap::onRender(NUIRenderer& renderer) {
291 if (!isVisible()) return;
292 auto b = getBounds();
293 auto& theme = NUIThemeManager::getInstance();
294
295 const auto panelBg = theme.getColor("backgroundPrimary").darkened(0.03f);
296 const auto border = theme.getColor("border").withAlpha(0.48f);
297 renderer.fillRoundedRect(NUIRect(b.x + 4.0f, b.y + 3.0f, b.width - 8.0f, b.height - 6.0f), 5.0f, panelBg);
298 renderer.strokeRoundedRect(NUIRect(b.x + 4.0f, b.y + 3.0f, b.width - 8.0f, b.height - 6.0f), 5.0f, 1.0f, border);
299
300 // View Rect
301 float x1 = b.x + beatToX(startBeat_);
302 float w = beatToX(viewDuration_);
303 NUIRect viewRect(x1, b.y + 2, w, b.height - 4);
304
305 auto thumbCol = theme.getColor("accentPrimary").withAlpha(0.13f);
306 auto borderCol = theme.getColor("accentPrimary").withAlpha(0.72f);
307
308 renderer.fillRoundedRect(viewRect, 5.0f, thumbCol);
309 renderer.strokeRoundedRect(viewRect, 5.0f, 1.0f, borderCol);
310
311 renderer.setClipRect(NUIRect(b.x + 2.0f, b.y + 2.0f, b.width - 4.0f, b.height - 4.0f));
312 int minPitch = 127;
313 int maxPitch = 0;
314 for (const auto& note : notes_) {
315 if (note.isDeleted || note.durationBeats <= 0.0) continue;
316 minPitch = std::min(minPitch, std::clamp(note.pitch, 0, 127));
317 maxPitch = std::max(maxPitch, std::clamp(note.pitch, 0, 127));
318 }
319 if (minPitch > maxPitch) {
320 minPitch = 48;
321 maxPitch = 72;
322 } else {
323 minPitch = std::max(0, minPitch - 2);
324 maxPitch = std::min(127, maxPitch + 2);
325 }
326
327 const float noteAreaY = b.y + 4.0f;
328 const float noteAreaH = std::max(8.0f, b.height - 8.0f);
329 const int pitchSpan = std::max(1, maxPitch - minPitch + 1);
330 const float laneHeight = noteAreaH / static_cast<float>(pitchSpan);
331 const float visibleNoteH = std::max(3.0f, std::min(5.0f, laneHeight * 0.95f));
332 const auto noteFill = theme.getColor("accentPrimary").lightened(0.08f).withAlpha(0.94f);
333 const auto noteEdge = theme.getColor("accentSecondary").withAlpha(0.55f);
334
335 for (const auto& note : notes_) {
336 if (note.isDeleted || note.durationBeats <= 0.0) continue;
337 const int clampedPitch = std::clamp(note.pitch, minPitch, maxPitch);
338 const int relativePitch = maxPitch - clampedPitch;
339 const float x = std::round(b.x + beatToX(note.startBeat));
340 const float wNote = std::max(4.0f, std::round(beatToX(std::max(0.05, note.durationBeats))));
341 const float laneY = noteAreaY + static_cast<float>(relativePitch) * laneHeight;
342 const float y = std::round(laneY + std::max(0.0f, (laneHeight - visibleNoteH) * 0.5f));
343 const float maxW = std::max(0.0f, (b.x + b.width - 4.0f) - x);
344 if (maxW <= 0.0f) continue;
345 NUIRect noteRect(x, y, std::min(wNote, maxW), visibleNoteH);
346 renderer.fillRoundedRect(noteRect, std::min(2.0f, visibleNoteH * 0.45f), noteFill);
347 renderer.strokeRoundedRect(noteRect, std::min(2.0f, visibleNoteH * 0.45f), 1.0f, noteEdge);
348 }
349 renderer.clearClipRect();
350
351 const auto playheadColor = NUIThemeManager::getInstance().getColor("accentPrimary");
352 const float playheadX = b.x + beatToX(playheadBeat_);
353 if (playheadX >= b.x && playheadX <= b.x + b.width) {
354 renderer.drawLine(NUIPoint(playheadX, b.y + 1.0f),
355 NUIPoint(playheadX, b.y + b.height - 1.0f),
356 2.0f,
357 playheadColor.withAlpha(0.9f));
358 }
359
360 // Handles (Visual only, logic in mouse)
361 const float handleW = 4.0f;
362 const float handleH = std::max(8.0f, b.height - 8.0f);
363 const auto handleFill = borderCol.withAlpha(0.5f);
364 NUIRect leftHandle(x1 + 2.0f, b.y + 4.0f, handleW, handleH);
365 NUIRect rightHandle(x1 + w - 6.0f, b.y + 4.0f, handleW, handleH);
366 renderer.fillRoundedRect(leftHandle, 2.0f, handleFill);
367 renderer.fillRoundedRect(rightHandle, 2.0f, handleFill);
368}

370bool PianoRollMinimap::onMouseEvent(const NUIMouseEvent& event) {
371 if (!getBounds().contains(event.position) && !isDragging_) return false;
372
373 auto b = getBounds();
374 float localX = event.position.x - b.x;
375 float localY = event.position.y - b.y;
376
377 float x1 = beatToX(startBeat_);
378 float w = beatToX(viewDuration_);
379 float x2 = x1 + w;
380 const float handleW = 8.0f;
381 const float handleH = std::max(8.0f, b.height - 8.0f);
382 const NUIRect leftHandleRect(x1 + 2.0f, 4.0f, handleW, handleH);
383 const NUIRect rightHandleRect(x1 + w - handleW - 2.0f, 4.0f, handleW, handleH);
384 const NUIRect viewportRect(x1, 2.0f, w, b.height - 4.0f);
385 const NUIPoint localPos(localX, localY);
386
387 if (event.pressed && event.button == NUIMouseButton::Left) {
388 // Hit Test
389 if (leftHandleRect.contains(localPos)) {
390 isDragging_ = true;
391 isResizingL_ = true;
392 dragStartPos_ = event.position;
393 dragStartStart_ = startBeat_;
394 dragStartDuration_ = viewDuration_;
395 } else if (rightHandleRect.contains(localPos)) {
396 isDragging_ = true;
397 isResizingR_ = true;
398 dragStartPos_ = event.position;
399 dragStartStart_ = startBeat_;
400 dragStartDuration_ = viewDuration_;
401 } else if (viewportRect.contains(localPos)) {
402 isDragging_ = true;
403 dragStartPos_ = event.position;
404 dragStartStart_ = startBeat_;
405 dragStartDuration_ = viewDuration_;
406 } else {
407 double beat = xToBeat(localX);
408 startBeat_ = std::clamp(beat - viewDuration_ * 0.5, 0.0, std::max(0.0, totalDuration_ - viewDuration_));
409 isDragging_ = false;
410 isResizingL_ = false;
411 isResizingR_ = false;
412 if (onViewChanged) onViewChanged(startBeat_, viewDuration_);
413 repaint();
414 return true;
415 }
416 return true;
417 }
418 else if (event.released && event.button == NUIMouseButton::Left) {
419 isDragging_ = false;
420 isResizingL_ = false;
421 isResizingR_ = false;
422 return true;
423 }
424 else if (!event.pressed && isDragging_) {
425 float dx = event.position.x - dragStartPos_.x;
426 double db = dx / b.width * totalDuration_;
427
428 if (isResizingL_) {
429 double newStart = dragStartStart_ + db;
430 double newDur = dragStartDuration_ - db;
431
432 if (newDur < 0.25) { newStart -= (0.25 - newDur); newDur = 0.25; }
433
434 viewDuration_ = std::clamp(newDur, 0.25, totalDuration_);
435 startBeat_ = std::clamp(newStart, 0.0, std::max(0.0, totalDuration_ - viewDuration_));
436 }
437 else if (isResizingR_) {
438 double newDur = dragStartDuration_ + db;
439 viewDuration_ = std::clamp(newDur, 0.25, totalDuration_);
440 startBeat_ = std::clamp(startBeat_, 0.0, std::max(0.0, totalDuration_ - viewDuration_));
441 }
442 else {
443 startBeat_ = std::clamp(dragStartStart_ + db, 0.0, std::max(0.0, totalDuration_ - viewDuration_));
444 }
445
446 if (onViewChanged) onViewChanged(startBeat_, viewDuration_);
447 repaint();
448 return true;
449 }
450
451 return NUIComponent::onMouseEvent(event);
452}

477 const double localX = static_cast<double>(event.position.x - bounds.x + scrollX_);

485 isScrubbing_ = true;
486 scrubToPointer();

490 if (event.released && event.button == NUIMouseButton::Left) {
491 scrubToPointer();
492 isScrubbing_ = false;
493 if (onPlayheadScrubbed) onPlayheadScrubbed(playheadBeat_, false);
494 return true;
495 }
496 scrubToPointer();
497 return true;
498 }

502// Ruler Render: "Mature" Playlist Style
503void PianoRollRuler::onRender(NUIRenderer& renderer) {
504 if (!isVisible()) return;
505 const auto bounds = getBounds();
506 auto& theme = NUIThemeManager::getInstance();
507
508 const auto rulerBackground = NUIColor(0.012f, 0.012f, 0.012f, 1.0f);
509 const auto border = NUIColor::white().withAlpha(0.075f);
510 const auto highlight = NUIColor::white().withAlpha(0.014f);
511 const auto text = theme.getColor("textPrimary").withAlpha(0.86f);
512 const auto tick = NUIColor::fromHex(0x2e2e2e).withAlpha(0.82f);
513
514 renderer.fillRoundedRect(bounds, 3.0f, rulerBackground);
515 renderer.fillRect(NUIRect(bounds.x, bounds.y, bounds.width, 1.0f), highlight);
516 renderer.strokeRoundedRect(bounds, 3.0f, 1.0f, border);
517 renderer.drawLine(NUIPoint(bounds.x, bounds.bottom() - 1.0f),
518 NUIPoint(bounds.right(), bounds.bottom() - 1.0f),
519 1.0f,
520 border);
521
522 renderer.setClipRect(bounds);
523
524 const int beatsPerBar = std::max(1, beatsPerBar_);
525 const float pixelsPerBar = pixelsPerBeat_ * beatsPerBar;
526 int barStride = 1;
527 while ((pixelsPerBar * static_cast<float>(barStride)) < 28.0f && barStride < 128) {
528 barStride *= 2;
529 }
530
531 int startBar = static_cast<int>(std::floor(scrollX_ / pixelsPerBar));
532 startBar = static_cast<int>(std::floor(static_cast<double>(startBar) / barStride)) * barStride;
533 const int visibleBars = static_cast<int>(std::ceil((scrollX_ + bounds.width) / pixelsPerBar)) - startBar;
534 const int endBar = startBar + visibleBars + barStride;
535
536 for (int bar = startBar; bar <= endBar; bar += barStride) {
537 const float x = snapVerticalLineX(bounds.x + bar * pixelsPerBar - scrollX_);
538 const int barNumber = bar + 1;
539 const bool isMajorBar = barStride > 1 || barNumber == 1 || ((barNumber - 1) % 4 == 0);
540 const float fontSize = isMajorBar ? 11.0f : 10.0f;
541 const float textY = std::round(renderer.calculateTextY(bounds, fontSize));
542 renderer.drawText(std::to_string(barNumber),
543 NUIPoint(x + 4.0f, textY),
544 fontSize,
545 isMajorBar ? text : text.withAlpha(0.76f));
546
547 const float tickHeight = isMajorBar ? bounds.height * 0.58f : bounds.height * 0.24f;
548 renderer.drawLine(NUIPoint(x, bounds.bottom() - tickHeight),
549 NUIPoint(x, bounds.bottom()),
550 isMajorBar ? 1.15f : 1.0f,
551 isMajorBar ? tick : tick.withAlpha(0.58f));
552
553 if (pixelsPerBeat_ >= 10.0f && barStride == 1) {
554 for (int beat = 1; beat < beatsPerBar; ++beat) {
555 const float beatX = x + beat * pixelsPerBeat_;
556 renderer.drawLine(NUIPoint(beatX, bounds.bottom() - bounds.height * 0.22f),
557 NUIPoint(beatX, bounds.bottom()),
558 1.0f,
559 NUIColor::fromHex(0x262626).withAlpha(0.36f));
560 }
561 }
562 }
563
564 const float playheadX =
565 snapVerticalLineX(beatToScreenX(playheadBeat_, pixelsPerBeat_, scrollX_, bounds.x));
566 if (playheadX >= bounds.x && playheadX <= bounds.right()) {
567 const auto accent = theme.getColor("accentPrimary");
568 const float markerRadius = isScrubbing_ ? 6.5f : 6.0f;
569 const NUIPoint markerCenter(playheadX, bounds.bottom() - markerRadius);
570 renderer.fillCircle(markerCenter,
571 markerRadius + 2.0f,
572 theme.getColor("backgroundPrimary").withAlpha(0.94f));

1097int PianoRollNoteLayer::snapPitchToScale(int pitch) {
1098 if (!snapToScale_ || scaleType_ == ScaleType::Chromatic) return pitch;
1099 if (MusicTheory::isNoteInScale(pitch, rootKey_, scaleType_)) return pitch;
1100 int bestPitch = pitch;
1101 int bestDist = 128;
1102 for (int candidate = pitch - 6; candidate <= pitch + 6; ++candidate) {
1103 if (candidate < 0 || candidate > 127) continue;
1104 if (MusicTheory::isNoteInScale(candidate, rootKey_, scaleType_)) {
1105 int dist = std::abs(candidate - pitch);
1106 if (dist < bestDist) {
1107 bestDist = dist;
1108 bestPitch = candidate;
1109 }
1110 }
1111 }
1112 return bestPitch;
1113}

1115void PianoRollNoteLayer::auditionPitch(int pitch) {
1116 // Never talk over the transport — playback owns the audio focus. Clear the
1117 // sounding pitch while suppressed so the very next idle placement re-fires
1118 // (otherwise the same-pitch guard below would swallow it after playback stops).
1119 if (isPlayingCallback_ && isPlayingCallback_()) {
1120 auditionPitch_ = -1;
1121 return;
1122 }
1123 pitch = std::clamp(pitch, 0, 127);
1124 // The audition path is a one-shot voice that auto-releases (~125 ms), so a
1125 // fresh press must always fire. auditionStop() resets the guard on release;
1126 // the guard's only job is to avoid machine-gunning one pitch while a drag
1127 // jitters within the same row.
1128 if (pitch == auditionPitch_) return;
1129 if (onPreviewNote_) {
1130 onPreviewNote_(pitch, static_cast<int>(std::lround(lastNoteVelocity_ * 127.0f)));
1131 }
1132 auditionPitch_ = pitch;
1133}

1137 // placement or pitch-drag can audition again, even on the same pitch.
1138 auditionPitch_ = -1;

1141std::vector<int> PianoRollNoteLayer::buildTriad(int rootPitch) const {
1142 rootPitch = std::clamp(rootPitch, 0, 127);
1143 // No scale context → a plain major triad is the sensible default.
1144 if (scaleType_ == ScaleType::Chromatic) {
1145 std::vector<int> chord = {rootPitch};
1146 if (rootPitch + 4 <= 127) chord.push_back(rootPitch + 4);
1147 if (rootPitch + 7 <= 127) chord.push_back(rootPitch + 7);
1148 return chord;
1149 }
1150
1151 // Diatonic triad = root + the 2nd and 4th scale tones above it (the third and
1152 // fifth), so the chord quality follows the degree the root sits on.
1153 std::vector<int> chord = {rootPitch};
1154 const int wantDegrees[] = {2, 4};
1155 int found = 0;
1156 int scaleStepsUp = 0;
1157 for (int p = rootPitch + 1; p <= 127 && found < 2; ++p) {
1158 if (!MusicTheory::isNoteInScale(p, rootKey_, scaleType_)) continue;
1159 ++scaleStepsUp;
1160 if (scaleStepsUp == wantDegrees[found]) {
1161 chord.push_back(p);
1162 ++found;
1163 }
1164 }
1165 return chord;
1166}

1251void PianoRollNoteLayer::glueSelectedNotes() {
1252 // Collect selected note indices grouped by pitch.
1253 std::vector<int> selected;
1254 for (size_t i = 0; i < notes_.size(); ++i) {
1255 if (notes_[i].selected && !notes_[i].isDeleted) selected.push_back(static_cast<int>(i));
1256 }
1257 if (selected.size() < 2) return;
1258
1259 auto oldNotes = notes_;
1260 std::vector<MidiNote> merged; // rebuilt selected notes
1261 std::vector<bool> consumed(notes_.size(), false);
1262
1263 // For each pitch, sweep left→right and coalesce runs that overlap or touch.
1264 std::sort(selected.begin(), selected.end(), [&](int a, int b) {
1265 if (notes_[a].pitch != notes_[b].pitch) return notes_[a].pitch < notes_[b].pitch;
1266 return notes_[a].startBeat < notes_[b].startBeat;
1267 });
1268
1269 bool changed = false;
1270 size_t k = 0;
1271 while (k < selected.size()) {
1272 MidiNote run = notes_[selected[k]];
1273 double runEnd = run.startBeat + run.durationBeats;
1274 size_t j = k + 1;
1275 while (j < selected.size() && notes_[selected[j]].pitch == run.pitch &&
1276 notes_[selected[j]].startBeat <= runEnd + 0.0001) {
1277 runEnd = std::max(runEnd, notes_[selected[j]].startBeat + notes_[selected[j]].durationBeats);
1278 changed = true; // at least two notes folded together
1279 ++j;
1280 }
1281 run.durationBeats = runEnd - run.startBeat;
1282 run.selected = true;
1283 merged.push_back(run);
1284 for (size_t m = k; m < j; ++m) consumed[selected[m]] = true;
1285 k = j;
1286 }
1287
1288 if (!changed) return;
1289
1290 // Keep everything that wasn't glued, then append the merged notes.
1291 std::vector<MidiNote> rebuilt;
1292 rebuilt.reserve(notes_.size());
1293 for (size_t i = 0; i < notes_.size(); ++i) {
1294 if (!consumed[i]) rebuilt.push_back(notes_[i]);
1295 }
1296 for (auto& m : merged) rebuilt.push_back(m);
1297 notes_ = std::move(rebuilt);
1298
1299 pushUndo("Glue", oldNotes, notes_);
1300 commitNotes();
1301 repaint();
1302}

1304void PianoRollNoteLayer::strumSelectedNotes(double spreadBeats) {
1305 std::vector<int> selected;
1306 for (size_t i = 0; i < notes_.size(); ++i) {
1307 if (notes_[i].selected && !notes_[i].isDeleted) selected.push_back(static_cast<int>(i));
1308 }
1309 if (selected.size() < 2) return;
1310
1311 // Anchor at the earliest selected start so the strum begins where the chord
1312 // sits, then cascade low pitch → high pitch.
1313 double baseStart = std::numeric_limits<double>::max();
1314 for (int i : selected) baseStart = std::min(baseStart, notes_[i].startBeat);
1315 std::sort(selected.begin(), selected.end(),
1316 [&](int a, int b) { return notes_[a].pitch < notes_[b].pitch; });
1317
1318 auto oldNotes = notes_;
1319 for (size_t k = 0; k < selected.size(); ++k) {
1320 notes_[selected[k]].startBeat = std::max(0.0, baseStart + static_cast<double>(k) * spreadBeats);
1321 }
1322 pushUndo("Strum", oldNotes, notes_);
1323 commitNotes();
1324 repaint();
1325}

1407void PianoRollNoteLayer::applyNotePropertyDelta(int field, float dy, bool coarseStep) {
1408 if (propNoteIndex_ < 0 || propNoteIndex_ >= static_cast<int>(notes_.size())) return;
1409 auto& n = notes_[propNoteIndex_];
1410 const MidiNote& s = propDragStartNote_;
1411
1412 double timeStep = MusicTheory::getSnapDuration(snap_);
1413 if (timeStep <= 0.0) timeStep = 0.25; // SnapGrid::None still steps musically
1414 if (!coarseStep) timeStep = 0.01; // Alt: fine time adjustment
1415
1416 switch (field) {
1417 case 0: { // Pitch — one semitone per few pixels, auditioned as it moves
1418 const int np = std::clamp(s.pitch + static_cast<int>(std::lround(dy / 6.0f)), 0, 127);
1419 if (np != n.pitch) {
1420 n.pitch = np;
1421 auditionPitch(np);
1422 }
1423 break;
1424 }
1425 case 1: // Velocity
1426 n.velocity = std::clamp(s.velocity + dy / 160.0f, 0.0f, 1.0f);
1427 break;
1428 case 2: // Pan
1429 n.pan = std::clamp(s.pan + dy / 90.0f, -1.0f, 1.0f);
1430 break;
1431 case 3: { // Start time
1432 const double steps = std::trunc(dy / 8.0f);
1433 n.startBeat = std::max(0.0, s.startBeat + steps * timeStep);
1434 break;
1435 }
1436 case 4: { // Length
1437 const double steps = std::trunc(dy / 8.0f);
1438 n.durationBeats = std::max(0.125, s.durationBeats + steps * timeStep);
1439 break;
1440 }
1441 default:
1442 break;
1443 }
1444 repaint();
1445}

1447bool PianoRollNoteLayer::handleNotePropertiesMouse(const NUIMouseEvent& event) {
1448 if (propNoteIndex_ < 0) return false;
1449 if (propNoteIndex_ >= static_cast<int>(notes_.size())) {
1450 propNoteIndex_ = -1;
1451 return false;
1452 }
1453
1454 const NUIRect& r = propPanelRect_;
1455 const float rowsTop = r.y + kPropPad + kPropTitleH;
1456 const auto fieldAt = [&](const NUIPoint& p) -> int {
1457 if (p.x < r.x + 4.0f || p.x > r.right() - 4.0f) return -1;
1458 const int row = static_cast<int>(std::floor((p.y - rowsTop) / kPropRowH));
1459 return (row >= 0 && row < kPropRowCount) ? row : -1;
1460 };
1461
1462 if (event.pressed && event.button == NUIMouseButton::Left) {
1463 if (!r.contains(event.position)) {
1464 closeNoteProperties(true); // click-away accepts, like Accept
1465 return true;
1466 }
1467 const float btnY = r.bottom() - kPropPad - 22.0f;
1468 const NUIRect resetRect(r.x + kPropPad, btnY, 88.0f, 22.0f);
1469 const NUIRect acceptRect(r.right() - kPropPad - 88.0f, btnY, 88.0f, 22.0f);
1470 if (resetRect.contains(event.position)) {
1471 notes_[propNoteIndex_] = propOriginalNote_;
1472 repaint();
1473 return true;
1474 }
1475 if (acceptRect.contains(event.position)) {
1476 closeNoteProperties(true);
1477 return true;
1478 }
1479 const int field = fieldAt(event.position);
1480 if (field != -1) {
1481 propDragField_ = field;
1482 propDragStartPos_ = event.position;
1483 propDragStartNote_ = notes_[propNoteIndex_];
1484 }
1485 return true;
1486 }
1487
1488 if (event.released) {
1489 // No commitNotes() while the popup is open: it sorts notes_ by start
1490 // beat, which would silently retarget propNoteIndex_ after a Start
1491 // edit. closeNoteProperties() commits once, on Accept/cancel.
1492 propDragField_ = -1;
1493 return true;
1494 }
1495
1496 if (propDragField_ != -1) {
1497 // Row drag: up increases. Alt refines the time fields.
1498 const float dy = propDragStartPos_.y - event.position.y;
1499 applyNotePropertyDelta(propDragField_, dy, !(event.modifiers & NUIModifiers::Alt));
1500 return true;
1501 }
1502
1503 if (event.wheelDelta != 0.0f) {
1504 const int field = fieldAt(event.position);
1505 if (field != -1) {
1506 // One step per notch, expressed as the drag distance of one step.
1507 static constexpr float kStepPixels[kPropRowCount] = {6.0f, 4.0f, 4.0f, 8.0f, 8.0f};
1508 propDragStartNote_ = notes_[propNoteIndex_];
1509 applyNotePropertyDelta(field,
1510 (event.wheelDelta > 0.0f ? 1.0f : -1.0f) * kStepPixels[field],
1511 !(event.modifiers & NUIModifiers::Alt));
1512 }
1513 return true;
1514 }
1515
1516 return true; // modal: swallow hover/motion so nothing edits underneath
1517}

1846bool PianoRollNoteLayer::onMouseEvent(const NUIMouseEvent& event) {

1848 // Reset hover when leaving bounds
1849 if (hoveredNoteIndex_ != -1) {
1850 hoveredNoteIndex_ = -1;
1851 hoverOnRightEdge_ = false;
1852 hoverOnLeftEdge_ = false;
1853 if (platformBridge_) platformBridge_->setCursorStyle(NUICursorStyle::Arrow);
1854 }
1855 if (hoveredPitch_ != -1) {
1856 hoveredPitch_ = -1;
1857 if (onHoveredPitchChanged_) onHoveredPitchChanged_(-1);
1858 }
1859 if (hoverBeat_ >= 0.0) {
1860 hoverBeat_ = -1.0;
1861 repaint();
1862 }
1863 return false;
1864 }
1865
1866 auto b = getBounds();
1867 float localX = event.position.x - b.x + scrollX_;
1868 float localY = event.position.y - b.y + scrollY_;
1869
1870 // Holding Alt mid-drag bypasses the grid for fine placement. Clone drags
1871 // (CopyDragging) are excluded — Alt is what starts them, so it can't
1872 // double as the fine toggle there. Recomputed every event so it can never
1873 // go stale between gestures.
1874 fineDrag_ = (event.modifiers & NUIModifiers::Alt) &&
1875 (state_ == State::Painting || state_ == State::Moving ||
1876 state_ == State::Resizing || state_ == State::ResizingLeft);
1877
1878 // The Note Properties popup is modal within the layer while open.
1879 if (propNoteIndex_ >= 0) {
1880 return handleNotePropertiesMouse(event);
1881 }
1882
1883 const int cursorPitch = std::clamp(127 - static_cast<int>(localY / keyHeight_), 0, 127);
1884 if (hoveredPitch_ != cursorPitch) {
1885 hoveredPitch_ = cursorPitch;
1886 if (onHoveredPitchChanged_) onHoveredPitchChanged_(cursorPitch);
1887 // Vertical moves within the same snapped beat leave hoverBeat_
1888 // unchanged, so without this the pencil's phantom note would linger
1889 // on the previous pitch row.
1890 repaint();
1891 }
1892
1893 // --- HOVER / SMART CURSOR (no button activity) ---
1894 if (state_ == State::None && !event.pressed && !event.released) {
1895 int hitIdx = findNoteAt(localX, localY);
1896 // Track the snapped beat the cursor sits on so the pencil can render a
1897 // phantom of the note a click would place. Only meaningful over empty
1898 // space with the pencil active; suppressed elsewhere so it never lingers.
1899 const double snappedHoverBeat =
1900 (tool_ == GlobalTool::Pencil && hitIdx == -1)
1901 ? snapToGrid(std::max(0.0, static_cast<double>(localX) / pixelsPerBeat_))
1902 : -1.0;
1903 if (hoverBeat_ != snappedHoverBeat) {
1904 hoverBeat_ = snappedHoverBeat;
1905 repaint();
1906 }
1907 if (hitIdx == -1) {
1908 if (hoveredNoteIndex_ != -1) {
1909 hoveredNoteIndex_ = -1;
1910 hoverOnRightEdge_ = false;
1911 hoverOnLeftEdge_ = false;
1912 repaint();
1913 }
1914 if (platformBridge_) {
1915 platformBridge_->setCursorStyle(
1916 tool_ == GlobalTool::Pencil ? NUICursorStyle::Crosshair : NUICursorStyle::Arrow);
1917 }
1918 } else {
1919 const auto& n = notes_[hitIdx];
1920 float nx = b.x + static_cast<float>(n.startBeat * pixelsPerBeat_) - scrollX_;
1921 float nw = static_cast<float>(n.durationBeats * pixelsPerBeat_);
1922 float edgeZone = std::min(10.0f, nw * 0.30f);
1923 bool onLeftEdge = (event.position.x <= nx + edgeZone);
1924 bool onRightEdge = (event.position.x >= nx + nw - edgeZone);
1925
1926 if (hoveredNoteIndex_ != hitIdx || hoverOnRightEdge_ != onRightEdge || hoverOnLeftEdge_ != onLeftEdge) {
1927 hoveredNoteIndex_ = hitIdx;
1928 hoverOnRightEdge_ = onRightEdge;
1929 hoverOnLeftEdge_ = onLeftEdge;
1930 repaint();
1931 }
1932 if (platformBridge_) {
1933 if (onLeftEdge || onRightEdge)
1934 platformBridge_->setCursorStyle(NUICursorStyle::ResizeEW);
1935 else
1936 platformBridge_->setCursorStyle(NUICursorStyle::Grab);
1937 }
1938 }

1941 // --- ALT + WHEEL OVER A NOTE = VELOCITY ---
1942 // Alt shapes the velocity of the note under the cursor (and the whole
1943 // selection if that note is part of it). Kept behind a modifier so plain
1944 // wheel always scrolls — mixing "scroll the grid" and "edit the note" on
1945 // the same gesture made scrolling feel unpredictable.
1946 if (event.wheelDelta != 0.0f && (event.modifiers & NUIModifiers::Alt) && hoveredNoteIndex_ >= 0 &&
1947 hoveredNoteIndex_ < static_cast<int>(notes_.size()) && !notes_[hoveredNoteIndex_].isDeleted) {
1948 auto oldNotes = notes_;
1949 const float delta = event.wheelDelta * 0.04f; // ~5 MIDI steps per notch
1950 const bool editSelection = notes_[hoveredNoteIndex_].selected;
1951 for (auto& n : notes_) {
1952 if (n.isDeleted) continue;
1953 if (editSelection ? n.selected : (&n == &notes_[hoveredNoteIndex_])) {
1954 n.velocity = std::clamp(n.velocity + delta, 0.0f, 1.0f);
1955 }
1956 }
1957 lastNoteVelocity_ = notes_[hoveredNoteIndex_].velocity; // adopt for new notes
1958 velocityBubbleIndex_ = hoveredNoteIndex_;
1959 // Coalesce a continuous scrub into one undo step instead of one per notch.
1960 if (!undoStack_.empty() && undoStack_.back().description == "Velocity") {
1961 undoStack_.back().notesAfter = notes_;
1962 } else {
1963 pushUndo("Velocity", oldNotes, notes_);
1964 }
1965 commitNotes();
1966 repaint();
1967 return true;
1968 }

1970 // --- RIGHT CLICK / ERASER (FAST ERASE) ---
1971 if (event.button == NUIMouseButton::Right) {
1972 if (event.pressed) {
1973 state_ = State::Erasing;
1974
1975 // Delete immediately on press
1976 int idx = findNoteAt(localX, localY);
1977 if (idx != -1) {
1978 auto oldNotes = notes_;
1979 notes_.erase(notes_.begin() + idx);
1980 pushUndo("Delete Note", oldNotes, notes_);
1981 commitNotes();
1982 repaint();
1983 }

1986 }

1992 notes_.erase(notes_.begin() + idx);

2007 std::chrono::steady_clock::now().time_since_epoch())

2056 // pencil keeps drawing, and Ctrl temporarily borrows a selection box.

2106 if (!(event.modifiers & NUIModifiers::Shift)) {

2110 const int rootPitch = snapPitchToScale(std::clamp(127 - static_cast<int>(localY / keyHeight_), 0, 127));

2115 occupied = true;

2134 for (int p : buildTriad(rootPitch)) {

2136 }

2749void PianoRollNoteLayer::pushUndo(const std::string& desc, const std::vector<MidiNote>& oldN, const std::vector<MidiNote>& newN) {
2750 PianoRollCommand cmd;
2751 cmd.description = desc;
2752 cmd.notesBefore = oldN;
2753 cmd.notesAfter = newN;
2754 undoStack_.push_back(cmd);
2755 redoStack_.clear();
2756
2757 // Enforce Limits (Count & Memory)
2758 // 1. Hard count limit
2759 if (undoStack_.size() > 50) {
2760 undoStack_.erase(undoStack_.begin());
2761 }
2762
2763 // 2. Memory Cap (100MB) - "Cockroach Chrysalis"
2764 // Calculate total size and evict from front (LRU)
2765 size_t totalBytes = 0;
2766 const size_t kMaxBytes = 100 * 1024 * 1024; // 100MB
2767
2768 // Reverse iterate to count from newest (keep these)
2769 // Actually simpler to just calc total and pop front.
2770 for (const auto& c : undoStack_) {
2771 totalBytes += c.description.capacity();
2772 totalBytes += c.notesBefore.capacity() * sizeof(MidiNote);
2773 totalBytes += c.notesAfter.capacity() * sizeof(MidiNote);
2774 }
2775
2776 while (totalBytes > kMaxBytes && !undoStack_.empty()) {
2777 const auto& c = undoStack_.front();
2778 size_t cmdSize = c.description.capacity() +
2779 c.notesBefore.capacity() * sizeof(MidiNote) +
2780 c.notesAfter.capacity() * sizeof(MidiNote);
2781
2782 if (totalBytes >= cmdSize) totalBytes -= cmdSize;
2783 else totalBytes = 0;
2784
2785 undoStack_.erase(undoStack_.begin());
2786 }
2787}

3676 if (!getBounds().contains(event.position) && !m_isResizingPanel) return false;
3677
3678 // The shortcut sheet captures the pointer: any click or scroll dismisses it
3679 // instead of reaching the editors underneath.
3680 if (m_showShortcutHelp) {
3681 if (event.pressed || event.wheelDelta != 0.0f) {
3682 m_showShortcutHelp = false;

3685 return true;
3686 }
3687
3688 if (m_toolbar) {
3689 if (auto menu = m_toolbar->getActiveContextMenu(); menu && menu->isVisible()) {
3690 if (menu->onMouseEvent(event)) return true;

3693 const auto menuBounds = menu->getGlobalBounds();

3695 if (!menuBounds.contains(event.position) &&
3696 !toolbarBounds.contains(event.position)) {
3697 m_toolbar->dismissActiveContextMenu();

3700 }
3701 }
3702 }
3703
3704 auto b = getBounds();
3705 float splitterY = b.y + b.height - m_controlPanelHeight;

3724 float newH = m_dragStartPanelHeight - dy;
