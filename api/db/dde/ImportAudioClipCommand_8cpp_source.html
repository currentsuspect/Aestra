10void ImportAudioClipCommand::rollbackFailedFirstExecute() {
11 auto& patterns = m_manager.getPatternManager();
12 if (m_patternId.isValid()) {
13 m_detachedPattern = patterns.detachPattern(m_patternId);
14 m_detachedPattern.reset();
15 m_patternId = PatternID{};
16 }
17 // Only a source this command introduced may be withdrawn, and only here:
18 // nothing has referenced it yet. A source the project already had stays.
19 if (m_createdSource && m_sourceId.isValid()) {
20 m_manager.getSourceManager().removeSource(m_sourceId);
21 m_sourceId = ClipSourceID{};
22 m_createdSource = false;
23 }
24}

26void ImportAudioClipCommand::execute() {
27 if (m_executed) {
28 return;
29 }
30
31 auto& sources = m_manager.getSourceManager();
32 auto& patterns = m_manager.getPatternManager();
33 auto& playlist = m_manager.getPlaylistModel();
34
35 const bool firstRun = !m_sourceId.isValid();
36
37 // Register the decoded audio here rather than in the factory, so the whole
38 // import is one mutation the history owns.
39 if (firstRun) {
40 if (!m_buffer || !m_buffer->isValid()) {
41 Log::error("[ImportAudioClipCommand] No decoded audio to import: " + m_filePath);
42 return;
43 }
44 const ClipSourceID existing = sources.findSourceByPath(m_filePath);
45 if (existing.isValid()) {
46 // Already in the project (path is the dedupe key): adopt it, and
47 // never withdraw it on failure.
48 m_sourceId = existing;
49 m_createdSource = false;
50 } else {
51 m_sourceId = sources.createRecordedSource(m_filePath, m_displayName, m_buffer);
52 if (!m_sourceId.isValid()) {
53 Log::error("[ImportAudioClipCommand] Could not register the decoded source for: " + m_filePath);
54 return;
55 }
56 m_createdSource = true;
57 }
58 }
59
60 const ClipSource* source = sources.getSource(m_sourceId);
61 const uint64_t sourceFrames = source ? source->getNumFrames() : 0;
62 if (sourceFrames == 0) {
63 Log::error("[ImportAudioClipCommand] The registered source carries no audio: " + m_filePath);
64 if (firstRun) {
65 rollbackFailedFirstExecute();
66 }
67 return;
68 }
69
70 if (m_detachedPattern) {
71 // Redo: the pattern already describes this import, so put it back
72 // rather than minting a second one for the same audio.
73 if (!patterns.reinsertPattern(m_detachedPattern)) {
74 return;
75 }
76 } else {
77 AudioSlicePayload payload;
78 payload.audioSourceId = m_sourceId;
79 payload.durationSeconds = m_durationSeconds;
80 AudioSlice fullSlice;
81 fullSlice.startSamples = 0.0;
82 fullSlice.lengthSamples = static_cast<double>(sourceFrames);
83 payload.slices.push_back(fullSlice);
84
85 m_patternId = patterns.createAudioPattern(m_displayName, m_durationBeats, payload);
86 if (!m_patternId.isValid()) {
87 Log::error("[ImportAudioClipCommand] Could not create the audio pattern for: " + m_displayName);
88 if (firstRun) {
89 rollbackFailedFirstExecute();
90 }
91 return;
92 }
93 }
94
95 ClipInstance clip;
96 clip.id = m_clipId.isValid() ? m_clipId : ClipInstanceID::generate();
97 clip.name = m_displayName;
98 clip.startBeat = m_startBeat;
99 clip.durationBeats = m_durationBeats;
100 clip.durationSeconds = m_durationSeconds;
101 clip.patternId = m_patternId;
102 clip.sourceId = m_patternId.value;
103 clip.edits = ClipEdits::forNewAudioClip();
104
105 const ClipInstanceID placed = playlist.addClip(m_laneId, clip);
106 if (!placed.isValid() || !playlist.getClip(placed)) {
107 Log::error("[ImportAudioClipCommand] Could not place the imported clip on its lane.");
108 if (firstRun) {
109 rollbackFailedFirstExecute();
110 } else {
111 // A failed redo keeps the source: undo is what put it in reach.
112 m_detachedPattern = patterns.detachPattern(m_patternId);
113 }
114 return;
115 }
116
117 m_clipId = placed;
118 m_manager.markModified();
119 m_manager.requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged);
120 m_executed = true;
121}

123void ImportAudioClipCommand::undo() {
124 if (!m_executed) {
125 return;
126 }
127 auto& playlist = m_manager.getPlaylistModel();
128 playlist.removeClip(m_clipId);
129 if (playlist.getClip(m_clipId)) {
130 return;
131 }
132 // Detach rather than remove, and keep the source: redo reuses both instead
133 // of decoding the file a second time.
134 m_detachedPattern = m_manager.getPatternManager().detachPattern(m_patternId);
135 m_manager.markModified();
136 m_manager.requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged);
137 m_executed = false;
138}
