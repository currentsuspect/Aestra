44void UIMixerFader::cacheThemeColors()
45{
46 auto& theme = NUIThemeManager::getInstance();
47 // Track: Deep Glass Slot
48 m_trackBg = theme.getColor("meterBackground").withAlpha(0.72f);
49 // Rail fill is deliberately NEUTRAL. Colouring it with the channel accent
50 // made the fader read as a second level meter — the fader shows where the

58 m_border = theme.getColor("borderStrong");
59 m_tick = theme.getColor("textSecondary").withAlpha(0.34f);
60 m_tickUnity = theme.getColor("textPrimary").withAlpha(0.72f);
61 m_tooltipBg = theme.getColor("elevatedPanel").withAlpha(0.98f);

76void UIMixerFader::updateCachedText()
77{
78 if (std::abs(m_cachedDbValue - m_valueDb) < 0.01f) {

123float UIMixerFader::dbToY(float db, float trackTop, float trackHeight) const
124{
125 const float norm = (db - m_minDb) / std::max(1e-3f, (m_maxDb - m_minDb));
126 return trackTop + (1.0f - std::clamp(norm, 0.0f, 1.0f)) * trackHeight;

129void UIMixerFader::renderScale(NUIRenderer& renderer, float trackX, float trackWidth,
130 float trackTop, float trackHeight, bool showLabels)
131{
132 // Ticks flank the rail on both sides so the cap never hides the scale.
133 constexpr float TICK_LEN = 4.0f;
134 constexpr float TICK_GAP = 3.0f;
135 const float leftEnd = trackX - TICK_GAP;
136 const float rightStart = trackX + trackWidth + TICK_GAP;
137
138 // The scale is structural, not a second readout. Unity is useful orientation
139 // at a glance; the complete dB ladder is revealed for the master strip or
140 // while the musician is directly manipulating this fader.
141 char labelBuf[8];
142 for (float db : SCALE_TICKS) {
143 if (db > m_maxDb || db < m_minDb) continue;
144
145 const bool unity = (std::abs(db) < 0.01f);
146 const float y = dbToY(db, trackTop, trackHeight);
147 const float len = unity ? TICK_LEN + 2.0f : TICK_LEN;
148 const float thickness = unity ? 2.0f : 1.0f;
149 const NUIColor color =
150 unity ? m_tickUnity : (showLabels ? m_tick : m_tick.withAlpha(m_tick.a * 0.58f));
151
152 renderer.fillRect(NUIRect{leftEnd - len, y - thickness * 0.5f, len, thickness}, color);
153 renderer.fillRect(NUIRect{rightStart, y - thickness * 0.5f, len, thickness}, color);
154
155 if (!showLabels && !unity) {
156 continue;
157 }
158
159 const char* label = "0";
160 if (!unity) {
161 std::snprintf(labelBuf, sizeof(labelBuf), "%+d", static_cast<int>(db));
162 label = labelBuf;
163 }
164 const float labelSize = renderer.measureText(label, 7.5f).width;
165 // Right-align labels against the tick so they remain a contextual
166 // measurement aid instead of forming a permanent text column.
167 renderer.drawText(label,
168 {leftEnd - TICK_LEN - 5.0f - labelSize, y - 3.5f},
169 unity ? 8.0f : 7.5f,
170 unity ? m_tickUnity : m_tick.withAlpha(showLabels ? m_tick.a : m_tick.a * 0.72f));
171 }

174void UIMixerFader::onRender(NUIRenderer& renderer)
175{
176 // Safe point to free a committed/cancelled editor: we are no longer inside
177 // any NUITextInput callback.
178 m_retiredInput.reset();
179
180 auto bounds = getBounds();
181
182 // Track area
183 const float trackTop = bounds.y + TOP_PAD;
184 const float trackBottom = bounds.y + bounds.height - BOTTOM_PAD;
185 const float trackHeight = std::max(1.0f, trackBottom - trackTop);
186
187 const float trackWidth = TRACK_WIDTH;
188 const float trackX = bounds.x + (bounds.width - trackWidth) * 0.5f;
189 NUIRect trackRect{trackX, trackTop, trackWidth, trackHeight};
190
191 // 1. Scale first, so the rail and cap sit on top of it.
192 const bool showScaleLabels = m_showScaleLabels || isHovered() || m_dragging;
193 renderScale(renderer, trackX, trackWidth, trackTop, trackHeight, showScaleLabels);
194
195 // 2. Track Background (deep recessed slot)
196 renderer.fillRoundedRect(trackRect, 3.0f, m_trackBg);
197
198 // 3. Filled Portion — neutral. This is control state, not signal.
199 const float norm = (m_valueDb - m_minDb) / std::max(1e-3f, (m_maxDb - m_minDb));
200 const float filledH = std::clamp(norm, 0.0f, 1.0f) * trackHeight;
201
202 if (filledH > 0.0f) {
203 NUIRect fillRect{trackX, trackBottom - filledH, trackWidth, filledH};
204 renderer.fillRoundedRect(fillRect, 3.0f, m_railFill);
205 }
206
207 // 4. Fader Handle (distinct grab block)
208 const float handleY = std::clamp(trackBottom - filledH - HANDLE_HEIGHT * 0.5f,
209 trackTop - HANDLE_HEIGHT * 0.5f,
210 trackBottom - HANDLE_HEIGHT * 0.5f);
211
212 const float handleW = handleWidth();
213 const float handleX = bounds.x + (bounds.width - handleW) * 0.5f;
214 const float handleH = HANDLE_HEIGHT;
215
216 NUIRect handleRect{handleX, handleY, handleW, handleH};
217 float handleRad = 4.0f;
218
219 // Handle Body (dark surface) — flat, no drop shadow; the border + grip
220 // give it enough definition.
221 renderer.fillRoundedRect(handleRect, handleRad, m_handle.withAlpha(0.98f));
222
223 // Handle Border — the one place the per-channel accent appears on the fader,
224 // and only while the control is actually engaged.
225 bool handleActive = isHovered() || m_dragging;
226 NUIColor handleBorder = handleActive ? m_accent : m_border;
227 renderer.strokeRoundedRect(handleRect, handleRad, 1.0f, handleBorder);
228
229 // Grip: three short lines read as "grab me" better than a single slit.
230 const float gripW = std::min(18.0f, handleW - 10.0f);
231 const float gripX = handleX + (handleW - gripW) * 0.5f;
232 const float gripCy = handleY + handleH * 0.5f;
233 const NUIColor gripColor = handleActive ? m_accent : m_textSecondary.withAlpha(0.85f);
234 for (int i = -1; i <= 1; ++i) {
235 renderer.fillRoundedRect(
236 NUIRect{gripX, gripCy + static_cast<float>(i) * 4.0f - 1.0f, gripW, 2.0f},
237 1.0f,
238 i == 0 ? gripColor : gripColor.withAlpha(gripColor.a * 0.45f));
239 }
240
241 // 5. dB readout at top. It remains available for exact entry, but recedes
242 // on ordinary channel strips so the control shape and the adjacent meter
243 // retain visual priority. The master strip keeps a stronger readout.
244 if (m_editing && m_textInput) {
245 renderChildren(renderer);
246 } else {
247 const bool valueActive = isHovered() || m_dragging || m_showScaleLabels;
248 const float fontSize = m_showScaleLabels ? 11.0f : 10.0f;
249 const NUIRect textRect = readoutRect();
250 if (isHovered()) {
251 renderer.fillRoundedRect(textRect, 3.0f, m_trackBg.withAlpha(0.55f));
252 }
253 renderer.drawTextCentered(m_cachedText,
254 textRect,
255 fontSize,
256 valueActive ? m_text : m_textSecondary.withAlpha(0.72f));
257 }
258
259 // 6. Drag Value Tooltip (only while dragging)
260 if (m_dragging) {
261 const float tipW = 38.0f;
262 const float tipH = 18.0f;
263 float tipX = handleX + (handleW - tipW) * 0.5f;
264 float tipY = handleY - tipH - 4.0f;
265
266 // Flip to bottom if near top edge
267 if (tipY < bounds.y) {
268 tipY = handleY + handleH + 4.0f;
269 }
270
271 renderer.fillRoundedRect({tipX, tipY, tipW, tipH}, 3.0f, m_tooltipBg);
272 renderer.strokeRoundedRect({tipX, tipY, tipW, tipH}, 3.0f, 1.0f, m_accent.withAlpha(0.5f));
273 renderer.drawTextCentered(m_cachedText, {tipX, tipY, tipW, tipH}, 10.5f, m_text);
274 }
275}

282void UIMixerFader::beginEdit()
283{
284 if (m_editing && m_textInput) {
285 return;
286 }
287
288 m_editing = true;
289
290 // Seed with the plain number so the user can just type over it. "−∞" is not
291 // round-trippable, so an inaudible fader starts from an empty field.
292 std::string seed;
293 if (m_valueDb > FADER_FLOOR_THRESHOLD) {
294 char buf[16];
295 std::snprintf(buf, sizeof(buf), "%.1f", m_valueDb);
296 seed = buf;
297 }
298
299 auto& theme = NUIThemeManager::getInstance();
300 m_textInput = std::make_shared<NUITextInput>(seed);
301
302 // NUITextInput always draws its text and caret at the theme's "m" size, and
303 // sizes the caret to that font's full line height. The 14 px readout band is
304 // shorter than that, so the caret overflowed the field. Give the editor the
305 // height the font actually needs; it grows down over the top of the track,
306 // which is fine for a transient editor.
307 const float fontM = theme.getFontSize("m");
308 const float editH = std::max(readoutRect().height, std::ceil(fontM * 1.9f) + 4.0f);
309 const NUIRect readout = readoutRect();
310 m_textInput->setBounds(NUIRect{readout.x, getBounds().y, readout.width, editH});
311 // Default 8 px padding on each side leaves almost nothing in a ~66 px strip.
312 m_textInput->setPadding(4.0f);
313 m_textInput->setInputType(NUITextInput::InputType::Number);
314 m_textInput->setJustification(NUITextInput::Justification::Center);
315 m_textInput->setTextColor(theme.getColor("textPrimary"));
316 m_textInput->setBackgroundColor(theme.getColor("inputBgDefault"));
317 m_textInput->setBorderColor(theme.getColor("inputBorderFocus"));
318 m_textInput->setBorderWidth(1.0f);
319 m_textInput->setBorderRadius(3.0f);
320 m_textInput->setFocusedBorderColor(m_accent);
321
322 m_textInput->setOnReturnKey([this]() { commitEdit(); });
323 m_textInput->setOnEscapeKey([this]() { cancelEdit(); });
324 m_textInput->setOnFocusLost([this]() { commitEdit(); });
325
326 addChild(m_textInput);
327 // Printable characters are routed to the globally focused component, so the
328 // field must take focus or it will never see any input.
329 m_textInput->setFocused(true);
330 m_textInput->setCaretPosition(static_cast<int>(seed.length()));
331 m_textInput->selectAll();
332 repaint();
333}

335void UIMixerFader::commitEdit()
336{
337 if (!m_editing) {
338 return;
339 }
340
341 const std::string entered = m_textInput ? m_textInput->getText() : std::string{};
342 m_editing = false;
343 if (m_textInput) {
344 removeChild(m_textInput);
345 // Park, don't free: this can run from the editor's own onFocusLost.
346 m_retiredInput = std::move(m_textInput);
347 m_textInput.reset();
348 }
349
350 // An empty field means "silence"; anything unparseable leaves the value alone.
351 if (entered.find_first_not_of(" \t") == std::string::npos) {
352 setValueDb(m_minDb);
353 } else {
354 // Parsed in the C locale, not the process locale. A DAW loads arbitrary
355 // third-party plugin binaries, and a plugin that calls
356 // setlocale(LC_ALL, "") on a comma-decimal system would otherwise make
357 // std::stof read "-6.5" as -6.
358 std::istringstream stream(entered);
359 stream.imbue(std::locale::classic());
360 float parsed = 0.0f;
361 stream >> parsed;
362
363 // Reject "nan"/"inf": clampDb happens to swallow those today only
364 // because of its comparison order, and that is not a guarantee worth
365 // depending on. Garbage input leaves the current value alone.
366 if (!stream.fail() && std::isfinite(parsed)) {
367 setValueDb(parsed);
368 }
369 }
370
371 repaint();
372}

409UIMixerFader::~UIMixerFader() {
410 // Torn down mid-drag: cancel the capture so the bridge never routes to a
411 // dangling owner and the cursor is never stranded hidden.
412 if (m_platformBridge && m_platformBridge->isCursorCaptureOwner(this)) {
413 m_platformBridge->cancelCursorCapture();
414 }
415
416 // Torn down mid-edit: the editor's callbacks capture `this`, and it can
417 // outlive the fader — removeChild() defers while an event is dispatching,
418 // parking a strong reference elsewhere. Nothing fires focus loss during
419 // teardown today, so this is not a live crash; dropping the callbacks makes
420 // that independent of a future change to component destruction order.
421 if (m_textInput) {
422 m_textInput->setOnReturnKey(nullptr);
423 m_textInput->setOnEscapeKey(nullptr);
424 m_textInput->setOnFocusLost(nullptr);
425 }

428bool UIMixerFader::onMouseEvent(const NUIMouseEvent& event)
429{
430 if (!isVisible() || !isEnabled()) return false;
431
432 auto bounds = getBounds();
433
434 // While the value is being typed, the field owns the pointer.
435 if (m_editing && m_textInput) {
436 if (m_textInput->onMouseEvent(event)) {
437 return true;
438 }
439 if (event.pressed && !bounds.contains(event.position)) {
440 commitEdit();
441 }
442 return false;
443 }
444
445 if (!event.cursorCaptured) {
446 setHovered(bounds.contains(event.position));
447 }
448 if (!bounds.contains(event.position) && !m_dragging) return false;
449
450 if (event.pressed && event.button == NUIMouseButton::Left) {
451 // The platform never populates event.doubleClick, so presses are paired
452 // up by the shared tracker — without this the reset below is dead code.
453 const bool isDoubleClick = m_clickTracker.registerPress() || event.doubleClick;
454
455 // The readout is an entry field, not part of the track: clicking it used
456 // to slam the fader to whatever value that Y mapped to.
457 if (readoutRect().contains(event.position)) {
458 beginEdit();
459 return true;
460 }
461
462 if (isDoubleClick) {
463 setValueDb(m_defaultDb);
464 return true;
465 }
466 }
467
468 if (event.pressed && event.button == NUIMouseButton::Left) {
469 m_dragging = true;
470 m_dragLatched = false;
471 m_dragStartPos = event.position;
472 const float trackTop = bounds.y + TOP_PAD;
473 const float trackBottom = bounds.y + bounds.height - BOTTOM_PAD;
474 const float trackHeight = std::max(1.0f, trackBottom - trackTop);
475 const float currentNorm = (m_valueDb - m_minDb) / std::max(1e-3f, (m_maxDb - m_minDb));
476 const float filledH = std::clamp(currentNorm, 0.0f, 1.0f) * trackHeight;
477 const float handleY = std::clamp(trackBottom - filledH - HANDLE_HEIGHT * 0.5f,
478 trackTop - HANDLE_HEIGHT * 0.5f,
479 trackBottom - HANDLE_HEIGHT * 0.5f);
480 const float handleW = handleWidth();
481 const float handleH = HANDLE_HEIGHT;
482 const float handleX = bounds.x + (bounds.width - handleW) * 0.5f;
483 const NUIRect handleRect{handleX, handleY, handleW, handleH};
484
485 if (!handleRect.contains(event.position)) {
486 const float norm = std::clamp(1.0f - (event.position.y - trackTop) / trackHeight, 0.0f, 1.0f);
487 const float clickedDb = m_minDb + norm * (m_maxDb - m_minDb);
488 setValueDb(clickedDb);
489 }
490
491 m_dragStartDb = m_valueDb;
492
493 // Cursor capture for infinite drag: service hides + confines and will
494 // restore at the fader handle on release.
495 if (m_platformBridge) {
496 m_lastDragY = event.position.y;
497 m_platformBridge->beginCursorCapture(
498 this, NUICursorRestorePolicy::ThumbPosition,
499 static_cast<int>(event.position.x), static_cast<int>(event.position.y));
500 }
501
502 return true;
503 }
504
505 if (event.released && event.button == NUIMouseButton::Left && m_dragging) {
506 m_dragging = false;
507 m_dragLatched = false;
508
509 if (m_platformBridge) {
510 // Warp cursor to the fader handle position (matches current value),
511 // not to the click origin — user expects cursor at the slider thumb.
512 auto bounds = getBounds();
513 const float trackTop = bounds.y + TOP_PAD;
514 const float trackBottom = bounds.y + bounds.height - BOTTOM_PAD;
515 const float trackHeight = std::max(1.0f, trackBottom - trackTop);
516 const float norm = (m_valueDb - m_minDb) / std::max(1e-3f, (m_maxDb - m_minDb));
517 const float filledH = std::clamp(norm, 0.0f, 1.0f) * trackHeight;
518 const float handleY = std::clamp(trackBottom - filledH - HANDLE_HEIGHT * 0.5f,
519 trackTop - HANDLE_HEIGHT * 0.5f,
520 trackBottom - HANDLE_HEIGHT * 0.5f);
521 const float handleW = handleWidth();
522 const float handleX = bounds.x + (bounds.width - handleW) * 0.5f;
523
524 // End capture: service warps to the handle, unhides, releases
525 // confinement — in that order.
526 m_platformBridge->endCursorCapture(
527 static_cast<int>(handleX + handleW * 0.5f),
528 static_cast<int>(handleY + HANDLE_HEIGHT * 0.5f));
529 }
530
531 return true;
532 }
533
534 // Dragging (mouse move events set button = None)
535 if (m_dragging && event.button == NUIMouseButton::None) {
536 // Cursor-warp mode: use frame-to-frame delta
537 if (m_platformBridge) {
538 // Service-owned delta (recentered; no absolute-coord read).
539 float dy = event.delta.y;
540
541 // Invert Y for fader: up = positive
542 const float deltaPx = -dy;
543 const float trackTop = bounds.y + TOP_PAD;
544 const float trackBottom = bounds.y + bounds.height - BOTTOM_PAD;
545 const float trackHeight = std::max(1.0f, trackBottom - trackTop);
546 const float dbPerPixel = (m_maxDb - m_minDb) / trackHeight;
547
548 // Reduced sensitivity for cursor-warp mode
549 float sensitivity = (event.modifiers & NUIModifiers::Shift) ? 0.1f : 0.5f;
550 float nextDb = m_valueDb + deltaPx * dbPerPixel * sensitivity;
551
552 if ((event.modifiers & NUIModifiers::Ctrl) || (event.modifiers & NUIModifiers::Alt)) {
553 nextDb = std::round(nextDb / SNAP_DB) * SNAP_DB;
554 }
555
556 setValueDb(nextDb);
557 return true;
558 }
559
560 // Non-warp mode: use drag start position with latching
561 const float deltaPx = m_dragStartPos.y - event.position.y;
562 const float trackTop = bounds.y + TOP_PAD;
563 const float trackBottom = bounds.y + bounds.height - BOTTOM_PAD;
564 const float trackHeight = std::max(1.0f, trackBottom - trackTop);
565 const float dbPerPixel = (m_maxDb - m_minDb) / trackHeight;
566 const float absDelta = std::abs(deltaPx);
567
568 float sensitivity = 1.0f;
569 if (event.modifiers & NUIModifiers::Shift) {
570 sensitivity *= 0.2f;
571 }
572 if (!m_dragLatched && absDelta < DRAG_SLOP) {
573 return true;
574 }
575 if (!m_dragLatched) {
576 m_dragLatched = true;
577 m_dragStartDb = m_valueDb;
578 m_dragStartPos = event.position;
579 return true;
580 }
581
582 float nextDb = m_dragStartDb + deltaPx * dbPerPixel * sensitivity;
583
584 if ((event.modifiers & NUIModifiers::Ctrl) || (event.modifiers & NUIModifiers::Alt)) {
585 nextDb = std::round(nextDb / SNAP_DB) * SNAP_DB;
586 }
587
588 setValueDb(nextDb);
589 return true;
590 }
591
592 return false;
593}
