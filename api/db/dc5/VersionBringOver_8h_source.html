48inline void mix(uint64_t& h, uint64_t v) { h ^= v + 0x9e3779b97f4a7c15ULL + (h << 6) + (h >> 2); }
49inline uint64_t q(double v) { return static_cast<uint64_t>(std::llround(v * 1000000.0)); }
50inline uint64_t strHash(const std::string& s) { return std::hash<std::string>{}(s); }
51
53inline uint64_t patternSignature(const PatternSource& p, const SourceManager& sources) {
54 uint64_t h = 1469598103934665603ULL;
55 mix(h, q(p.lengthBeats));
56 if (p.isMidi()) {
57 for (const auto& n : p.getMidiNotes()) {
58 mix(h, static_cast<uint64_t>(n.pitch)); mix(h, q(n.startBeat)); mix(h, q(n.durationBeats)); mix(h, q(n.velocity));
59 }
60 } else if (p.isAudio()) {
61 const auto& a = std::get<AudioSlicePayload>(p.payload);
62 if (const auto* source = sources.getSource(a.audioSourceId)) mix(h, strHash(source->getFilePath()));
63 mix(h, q(a.startOffsetSeconds)); mix(h, q(a.durationSeconds));
64 for (const auto& s : a.slices) { mix(h, q(s.startSamples)); mix(h, q(s.lengthSamples)); }
65 }
66 return h;
67}
68
70inline bool instrumentsExist(const PatternSource& p, const TrackManager& current) {
71 if (!p.isMidi()) return true;
72 for (const auto& n : p.getMidiNotes())
73 if (n.unitId != 0 && current.getUnitManager().getUnit(n.unitId) == nullptr) return false;
74 return true;
75}

80inline Plan plan(TrackManager& other, TrackManager& current, const std::string& otherName) {
81 Plan out;
82 auto& op = other.getPatternManager();
83 auto& cp = current.getPatternManager();
84 std::unordered_map<std::string, uint64_t> currentPatterns;
85 cp.forEachPattern([&](const PatternSource& p) { currentPatterns[p.name] = detail::patternSignature(p, current.getSourceManager()); });
86
87 std::unordered_map<std::string, uint64_t> currentTracks;
88 for (Track* t : current.getTracks()) {
89 uint64_t h = 7;
90 for (const auto& laneId : t->laneIds) {
91 const auto* lane = current.getPlaylistModel().getLane(laneId);
92 if (!lane) continue;
93 for (const auto& c : lane->clips) {
94 detail::mix(h, detail::q(c.startBeat)); detail::mix(h, detail::q(c.durationBeats));
95 if (const auto* p = cp.getPattern(c.patternId)) detail::mix(h, detail::patternSignature(*p, current.getSourceManager()));
96 }
97 }
98 currentTracks[t->name] = h;
99 }
100
101 for (Track* t : other.getTracks()) {
102 Item item;
103 item.kind = Kind::Track;
104 item.id = t->trackId;
105 item.name = t->name;
106 uint64_t h = 7;
107 size_t clips = 0;
108 bool blocked = false;
109 for (const auto& laneId : t->laneIds) {
110 const auto* lane = other.getPlaylistModel().getLane(laneId);
111 if (!lane) continue;
112 for (const auto& c : lane->clips) {
113 ++clips;
114 detail::mix(h, detail::q(c.startBeat)); detail::mix(h, detail::q(c.durationBeats));
115 if (const auto* p = op.getPattern(c.patternId)) {
116 detail::mix(h, detail::patternSignature(*p, other.getSourceManager()));
117 blocked = blocked || !detail::instrumentsExist(*p, current);
118 }
119 }
120 }
121 const auto it = currentTracks.find(t->name);
122 if (blocked) { item.status = Status::Blocked; item.detail = "needs an instrument this version does not have"; }
123 else if (it == currentTracks.end()) { item.status = Status::New; item.detail = "new in " + otherName; }
124 else if (it->second == h) { item.status = Status::Same; item.detail = "same as current"; }
125 else { item.status = Status::Changed; item.detail = std::to_string(clips) + (clips == 1 ? " clip" : " clips") + " \xc2\xb7 differs from current"; }
126 out.tracks.push_back(std::move(item));
127 }
128
129 op.forEachPattern([&](const PatternSource& p) {
130 Item item;
131 item.kind = Kind::Pattern;
132 item.id = p.id.value;
133 item.name = p.name.empty() ? "Pattern " + std::to_string(p.id.value) : p.name;
134 const auto it = currentPatterns.find(p.name);
135 const uint64_t sig = detail::patternSignature(p, other.getSourceManager());
136 if (!detail::instrumentsExist(p, current)) { item.status = Status::Blocked; item.detail = "needs an instrument this version does not have"; }
137 else if (it == currentPatterns.end()) { item.status = Status::New; item.detail = "new in " + otherName; }
138 else if (it->second == sig) { item.status = Status::Same; item.detail = "same as current"; }
139 else { item.status = Status::Changed; item.detail = "differs from current"; }
140 out.patterns.push_back(std::move(item));
141 });
142 std::sort(out.patterns.begin(), out.patterns.end(), [](const Item& a, const Item& b) { return a.name < b.name; });
143 return out;
144}

150class Command : public ICommand {
151public:
152 Command(TrackManager& current, TrackManager& other, const std::vector<uint64_t>& trackIds, const std::vector<uint64_t>& patternIds)
153 : m_current(current) {
154 auto& op = other.getPatternManager();
155 std::unordered_map<uint64_t, size_t> patternIndex; // other's pattern id -> index into m_patterns
156 const auto takePattern = [&](PatternID id) -> int {
157 const auto found = patternIndex.find(id.value);
158 if (found != patternIndex.end()) return static_cast<int>(found->second);
159 const PatternSource* src = op.getPattern(id);
160 if (!src) return -1;
161 PatternCopy copy;
162 copy.pattern = *src;
163 if (src->isAudio()) {
164 const auto& a = std::get<AudioSlicePayload>(src->payload);
165 if (const auto* source = other.getSourceManager().getSource(a.audioSourceId)) {
166 copy.audioPath = source->getFilePath();
167 copy.audioName = source->getName();
168 copy.buffer = source->getBuffer();
169 }
170 }
171 m_patterns.push_back(std::move(copy));
172 patternIndex[id.value] = m_patterns.size() - 1;
173 return static_cast<int>(m_patterns.size() - 1);
174 };
175 for (const uint64_t trackId : trackIds) {
176 const Track* t = other.getTrack(trackId);
177 if (!t) continue;
178 TrackCopy tc;
179 tc.name = t->name;
180 for (const auto& laneId : t->laneIds) {
181 const auto* lane = other.getPlaylistModel().getLane(laneId);
182 if (!lane) continue;
183 LaneCopy lc;
184 lc.name = lane->name; lc.volume = lane->volume; lc.pan = lane->pan; lc.muted = lane->muted; lc.color = lane->colorRGBA;
185 for (const auto& clip : lane->clips) {
186 ClipCopy cc;
187 cc.clip = clip;
188 cc.clip.id = ClipInstanceID{};
189 cc.pattern = clip.patternId.value ? takePattern(clip.patternId) : -1;
190 lc.clips.push_back(std::move(cc));
191 }
192 tc.lanes.push_back(std::move(lc));
193 }
194 m_tracks.push_back(std::move(tc));
195 }
196 for (const uint64_t id : patternIds) takePattern(PatternID{id});
197 }

199 void execute() override {
200 auto& patterns = m_current.getPatternManager();
201 auto& sources = m_current.getSourceManager();
202 auto& playlist = m_current.getPlaylistModel();
203 m_patternIds.clear();
204 for (auto& copy : m_patterns) {
205 PatternSource pattern = copy.pattern;
206 if (pattern.isAudio()) {
207 auto& payload = std::get<AudioSlicePayload>(pattern.payload);
208 const ClipSourceID sourceId = copy.audioPath.empty() ? ClipSourceID{} : sources.getOrCreateSource(copy.audioPath, copy.audioName);
209 if (auto* source = sources.getSource(sourceId); source && !source->isReady() && copy.buffer) source->setBuffer(copy.buffer);
210 payload.audioSourceId = sourceId;
211 pattern.setMixerChannelId(0); // Master: mixer channels are not brought over
212 }
213 m_patternIds.push_back(patterns.adoptPattern(pattern));
214 }
215 m_made.clear();
216 for (const auto& tc : m_tracks) {
217 Made made;
218 for (size_t i = 0; i < tc.lanes.size(); ++i) {
219 const auto& lc = tc.lanes[i];
220 const PlaylistLaneID laneId = playlist.createLane(lc.name);
221 if (auto* lane = playlist.getLane(laneId)) { lane->volume = lc.volume; lane->pan = lc.pan; lane->muted = lc.muted; lane->colorRGBA = lc.color; }
222 if (i == 0) made.trackId = m_current.createTrack(laneId, tc.name);
223 else if (auto* track = m_current.getTrack(made.trackId)) { track->laneIds.push_back(laneId); if (auto* lane = playlist.getLane(laneId)) lane->trackId = made.trackId; }
224 made.lanes.push_back(laneId);
225 for (const auto& cc : lc.clips) {
226 ClipInstance clip = cc.clip;
227 if (cc.pattern >= 0) {
228 clip.patternId = m_patternIds[static_cast<size_t>(cc.pattern)];
229 if (const auto* p = patterns.getPattern(clip.patternId); p && p->isAudio())
230 clip.sourceId = std::get<AudioSlicePayload>(p->payload).audioSourceId.value;
231 }
232 made.clips.push_back(playlist.addClip(laneId, clip));
233 }
234 }
235 m_made.push_back(std::move(made));
236 }
237 m_current.markModified();
238 }

240 void undo() override {
241 auto& playlist = m_current.getPlaylistModel();
242 for (auto it = m_made.rbegin(); it != m_made.rend(); ++it) {
243 for (const auto& clipId : it->clips) playlist.removeClip(clipId);
244 if (it->trackId) m_current.removeTrack(it->trackId);
245 for (const auto& laneId : it->lanes) playlist.removeLane(laneId);
246 }
247 m_made.clear();
248 for (const auto& id : m_patternIds) m_current.getPatternManager().removePattern(id);
249 m_patternIds.clear();
250 m_current.markModified();
251 }
252
253 void redo() override { execute(); }
