57inline double timelineGridOffsetToBeat(float offsetFromGridStart, float scrollOffset, float pixelsPerBeat) {
58 if (!(pixelsPerBeat > 0.0f) || !std::isfinite(pixelsPerBeat)) {
59 return 0.0;
60 }
61 return (static_cast<double>(offsetFromGridStart) + static_cast<double>(scrollOffset)) /
62 static_cast<double>(pixelsPerBeat);
63}

66inline float timelineBeatToGridOffset(double beat, float scrollOffset, float pixelsPerBeat) {
67 if (!(pixelsPerBeat > 0.0f) || !std::isfinite(pixelsPerBeat)) {
68 return 0.0f;
69 }
70 return static_cast<float>(beat * static_cast<double>(pixelsPerBeat)) - scrollOffset;
71}

104inline float safeClampFloat(float value, float a, float b) {
105 // Bounds first — "in range" is meaningless without a range.
106 if (!std::isfinite(a) && !std::isfinite(b))
107 return 0.0f;
108 if (!std::isfinite(a))
109 a = b;
110 if (!std::isfinite(b))
111 b = a;
112
113 const float lo = std::min(a, b);
114 const float hi = std::max(a, b);
115
116 // NaN compares false against everything, so it would otherwise fall through
117 // both bounds checks and be returned unchanged.
118 if (std::isnan(value))
119 return lo;
120
121 // These two also absorb -inf and +inf, which order correctly against finite
122 // bounds and so need no special case of their own.
123 if (value <= lo)
124 return lo;
125 if (value >= hi)
126 return hi;
127 return value;
128}
