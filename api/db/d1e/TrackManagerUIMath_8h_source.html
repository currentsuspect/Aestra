87inline double timelineGridOffsetToBeat(float offsetFromGridStart, float scrollOffset, float pixelsPerBeat) {
88 if (!(pixelsPerBeat > 0.0f) || !std::isfinite(pixelsPerBeat)) {
89 return 0.0;
90 }
91 return (static_cast<double>(offsetFromGridStart) + static_cast<double>(scrollOffset)) /
92 static_cast<double>(pixelsPerBeat);
93}

96inline float timelineBeatToGridOffset(double beat, float scrollOffset, float pixelsPerBeat) {
97 if (!(pixelsPerBeat > 0.0f) || !std::isfinite(pixelsPerBeat)) {
98 return 0.0f;
99 }
100 return static_cast<float>(beat * static_cast<double>(pixelsPerBeat)) - scrollOffset;
101}

134inline float safeClampFloat(float value, float a, float b) {
135 // Bounds first — "in range" is meaningless without a range.
136 if (!std::isfinite(a) && !std::isfinite(b))
137 return 0.0f;
138 if (!std::isfinite(a))
139 a = b;
140 if (!std::isfinite(b))
141 b = a;
142
143 const float lo = std::min(a, b);
144 const float hi = std::max(a, b);
145
146 // NaN compares false against everything, so it would otherwise fall through
147 // both bounds checks and be returned unchanged.
148 if (std::isnan(value))
149 return lo;
150
151 // These two also absorb -inf and +inf, which order correctly against finite
152 // bounds and so need no special case of their own.
153 if (value <= lo)
154 return lo;
155 if (value >= hi)
156 return hi;
157 return value;
158}
