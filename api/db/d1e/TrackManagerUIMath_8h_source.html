21// Above the track rows sits ONE time band, top to bottom: the overview

83inline double timelineGridOffsetToBeat(float offsetFromGridStart, float scrollOffset, float pixelsPerBeat) {
84 if (!(pixelsPerBeat > 0.0f) || !std::isfinite(pixelsPerBeat)) {
85 return 0.0;
86 }
87 return (static_cast<double>(offsetFromGridStart) + static_cast<double>(scrollOffset)) /
88 static_cast<double>(pixelsPerBeat);
89}

92inline float timelineBeatToGridOffset(double beat, float scrollOffset, float pixelsPerBeat) {
93 if (!(pixelsPerBeat > 0.0f) || !std::isfinite(pixelsPerBeat)) {
94 return 0.0f;
95 }
96 return static_cast<float>(beat * static_cast<double>(pixelsPerBeat)) - scrollOffset;
97}

130inline float safeClampFloat(float value, float a, float b) {
131 // Bounds first — "in range" is meaningless without a range.
132 if (!std::isfinite(a) && !std::isfinite(b))
133 return 0.0f;
134 if (!std::isfinite(a))
135 a = b;
136 if (!std::isfinite(b))
137 b = a;
138
139 const float lo = std::min(a, b);
140 const float hi = std::max(a, b);
141
142 // NaN compares false against everything, so it would otherwise fall through
143 // both bounds checks and be returned unchanged.
144 if (std::isnan(value))
145 return lo;
146
147 // These two also absorb -inf and +inf, which order correctly against finite
148 // bounds and so need no special case of their own.
149 if (value <= lo)
150 return lo;
151 if (value >= hi)
152 return hi;
153 return value;
154}
