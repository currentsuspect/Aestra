49struct StartupTimer {
50 const char* name;
51 std::chrono::steady_clock::time_point start;
52 explicit StartupTimer(const char* n)
53 : name(n), start(std::chrono::steady_clock::now()) {}
54 ~StartupTimer() {
55 auto elapsed = std::chrono::duration_cast<std::chrono::milliseconds>(
56 std::chrono::steady_clock::now() - start)
57 .count();
58 Log::info(std::string("[Startup] ") + name + ": " + std::to_string(elapsed) + " ms");
59 }
60};

101std::optional<std::string> AestraApp::getAppDataPath() {
102 IPlatformUtils* utils = Platform::getUtils();
103 if (!utils) {
104 // Deliberately no fallback. This used to return the process working
105 // directory, which is a perfectly valid-looking path that happens to
106 // be the wrong one — callers could not tell, and neither could the
107 // log. That is what made both #675 defects silent. There is no
108 // portable-mode feature in the tree that wants a working-directory
109 // app-data location, so failing is the honest answer (#676).
110 Log::warning("[AppData] Platform utilities unavailable; no app-data path can be resolved");
111 return std::nullopt;
112 }
113 std::string appDataDir = utils->getAppDataPath("Aestra");
114 std::error_code ec;
115 if (!std::filesystem::create_directories(appDataDir, ec) && ec) {
116 // create_directories() returns false without setting ec when the
117 // directory already exists, so a set ec is a genuine failure. A
118 // directory we cannot create is one we cannot write to either.
119 Log::warning("[AppData] Cannot create app-data directory " + appDataDir + ": " + ec.message());
120 return std::nullopt;
121 }
122 return appDataDir;
123}

166void AestraApp::writeCrashFlag() {
167 // initialize() primes this immediately after Platform::initialize(). Prime
168 // here only as a fallback for callers that reach the write without going
169 // through startup — re-resolving unconditionally would let detection use
170 // one path while the write and the clear use another (#675).
171 if (!CrashFlagPath::isPrimed()) {
172 const auto resolved = getCrashFlagPath();
173 if (!resolved) {
174 Log::warning("[CrashDetection] Cannot resolve crash flag path; no crash flag written this session");
175 return;
176 }
177 CrashFlagPath::prime(*resolved);
178 }
179 const std::string flagPath = CrashFlagPath::get();
180 std::ofstream out(flagPath, std::ios::trunc);
181 if (out) {
182 out << std::chrono::system_clock::now().time_since_epoch().count() << "\n";
183 // The flag is a crash sentinel: without fsync a hard crash or power loss
184 // right after startup can lose it to the OS cache — exactly the case
185 // crash detection exists for (issue #284).
186 if (!Aestra::syncOfstream(out, flagPath)) {
187 // Best-effort: the unsynced flag stays on disk, but don't log success.
188 Log::warning("[CrashDetection] Failed to sync crash flag: " + flagPath);
189 out.close();
190 } else {
191 out.close();
192 Log::info("[CrashDetection] Wrote crash flag: " + flagPath);
193 }
194 } else {
195 Log::warning("[CrashDetection] Failed to write crash flag: " + flagPath);
196 }
197}

199void AestraApp::clearCrashFlag() {
200 // Use the path retained at startup. Resolving it here would go through
201 // getAppDataPath(), which falls back to the working directory once
202 // Platform::shutdown() has released the platform utilities — the flag would
203 // then never be found and never be cleared (#675).
204 //
205 // Not primed means writeCrashFlag() never ran, so there is nothing this
206 // process wrote to clear. Say so loudly rather than resolving a path that
207 // is wrong by construction at this point in shutdown.
208 if (!CrashFlagPath::isPrimed()) {
209 Log::warning("[CrashDetection] Crash flag path was never resolved; nothing to clear");
210 return;
211 }
212 const std::string& flagPath = CrashFlagPath::get();
213 std::error_code ec;
214 if (std::filesystem::exists(flagPath, ec)) {
215 std::filesystem::remove(flagPath, ec);
216 if (!ec) {
217 Log::info("[CrashDetection] Cleared crash flag");
218 } else {
219 Log::warning("[CrashDetection] Failed to clear crash flag: " + ec.message());
220 }
221 }

224bool AestraApp::isCrashedSession() {
225 const auto flagPath = activeCrashFlagPath();
226 if (!flagPath) {
227 // Unresolvable is not the same as clean, but there is nothing to read
228 // and no substitute worth inventing. Say so rather than reporting a
229 // confident "no crash" derived from the wrong directory (#676).
230 Log::warning("[CrashDetection] Cannot resolve crash flag path; treating this session as not crashed");
231 return false;
232 }
233 std::error_code ec;
234 return std::filesystem::exists(*flagPath, ec);
235}

284bool AestraApp::initialize(const std::string& projectPath) {
285 StartupTimer totalTimer("Total startup");
286
287 if (!transitionToInitializing()) return false;
288
289 Log::info("Aestra v1.0.0 - Initializing...");
290
291 {
292 StartupTimer t("Platform init");
293 if (!Platform::initialize()) {
294 Log::error("Failed to initialize platform");
295 return false;
296 }
297 }
298
299 // Resolve the crash-flag path ONCE, now that platform utilities exist, and
300 // reuse it for every crash-flag operation in this process (#675).
301 //
302 // Every one of those operations previously called getAppDataPath()
303 // independently, and that function silently falls back to the process
304 // working directory whenever platform utilities are absent. Detection used
305 // to run BEFORE Platform::initialize() and clearing runs AFTER
306 // Platform::shutdown(), so both sat in exactly that window: the flag was
307 // written to app-data, then looked for in the working directory. Clean
308 // exits never cleared it, and — worse — a real crash was never detected,
309 // because the read missed the file too. Proven by launching with a flag in
310 // the working directory only: recovery fired.
311 //
312 // Detection therefore moved below platform init. Nothing between the old
313 // and new positions consumed the result.
314 if (const auto crashFlagPath = getCrashFlagPath()) {
315 CrashFlagPath::prime(*crashFlagPath);
316 } else {
317 Log::error("[CrashDetection] Cannot resolve crash flag path at startup; "
318 "crash detection and recovery are unavailable this session");
319 }
320
321 const bool crashedSession = isCrashedSession();
322 m_previousRecoverySessionToken = crashedSession ? readCrashFlagToken() : "";
323
324 // An app-data autosave is recovery state, never an implicit canonical project.
325 m_documentState.startUntitled(autosavePathOrEmpty());
326
327 // ALWAYS write crash flag at session start. It is cleared only on clean
328 // shutdown. If the app crashes at any point, the flag persists and the
329 // next launch will detect it.
330 writeCrashFlag();
331 m_recoverySessionToken = readCrashFlagToken();
332
333 // Load preferences and UI state early
334 Preferences::instance().load();
335 bool autoSaveEnabled = Preferences::instance().autoSaveEnabled;
336
337 UIState uiState;
338 uiState.load();
339
340 {
341 StartupTimer t("Platform + Window");
342 if (!initializePlatformAndWindow(uiState)) return false;
343 }
344 {
345 StartupTimer t("Audio init");
346 initializeAudio();
347 }
348 {
349 StartupTimer t("Content init");
350 initializeContent();
351 }
352 {
353 StartupTimer t("Autosave init");
354 initializeAutosave(autoSaveEnabled);
355 }
356 {
357 StartupTimer t("Recovery dialog");
358 buildRecoveryDialog();
359 }
360 {
361 StartupTimer t("Menu bar");
362 buildMenuBar();
363 }
364 {
365 StartupTimer t("Plugins init");
366 initializePlugins();
367 }
368 {
369 StartupTimer t("Project load/recovery");
370 loadOrRecoverProject(projectPath, crashedSession);
371 }
372 {
373 StartupTimer t("UI state restore");
374 restoreUIState(uiState);
375 }
376
377 startMuseSocketIfConfigured();
378
379 return transitionToRunning();
380}

464std::chrono::seconds AestraApp::resolveAutosaveInterval() {
465 // ONE place decides this. Three numbers used to describe the autosave
466 // interval and disagreed: Preferences said 300, AutosaveManager's own
467 // default said 30, and initializeAutosave hard-coded 60 — which won,
468 // because the other two were never consulted.
469 //
470 // reinitAutosaveManager() hard-coded 60 a second time, so a user's stored
471 // interval was silently discarded whenever the autosave manager was rebuilt
472 // (project load, save-as). Both sites now resolve through here.
473 //
474 // Clamped because this is user-editable JSON on disk, and the corrected
475 // value is written back so the file and the running config agree.
476 const int persisted = Preferences::instance().autoSaveIntervalSeconds;
477 const int seconds = std::clamp(persisted, 10, 3600);
478 if (seconds != persisted) {
479 Log::warning("[AutoSave] autoSaveIntervalSeconds " + std::to_string(persisted) +
480 " out of range; using " + std::to_string(seconds));
481 Preferences::instance().autoSaveIntervalSeconds = seconds;
482 }
483 return std::chrono::seconds(seconds);
484}

486void AestraApp::initializeAutosave(bool enabled) {
487 Aestra::Audio::AutosaveManager::Config config;
488 config.enabled = enabled;
489 // Three numbers used to describe this interval and disagreed: Preferences
490 // said 300, AutosaveManager's own default said 30, and this line hard-coded
491 // 60 — which won, because the other two were never consulted. Preferences is
492 // the persisted authority; clamped because it is user-editable JSON on disk.
493 config.autosaveInterval = resolveAutosaveInterval();
494 config.captureSnapshotOnCallingThread = true;
495 config.autosavePathOverride = m_documentState.autosavePath();
496 const std::string canonicalProjectPath = m_documentState.canonicalPath();
497 config.onAutosaveCommitted = [this, canonicalProjectPath](const std::string& autosavePath) {
498 writeRecoveryMarkerForAutosave(autosavePath, canonicalProjectPath);
499 };
500 config.serializer = [this](std::string& outData) -> bool {
501 if (!m_content || !m_content->getTrackManager()) return false;
502 const double tempo = (m_audioController && m_audioController->getEngine())
503 ? static_cast<double>(m_audioController->getEngine()->getBPM())
504 : (m_content->getTransportBar() ? static_cast<double>(m_content->getTransportBar()->getTempo()) : 120.0);
505 const double playhead = m_content->getTrackManager()->getPosition();
506 auto ser = ProjectSerializer::serialize(m_content->getTrackManager(), tempo, playhead, 0);
507 if (!ser.ok) return false;
508 outData = std::move(ser.contents);
509 return true;
510 };
511 m_autoSaveManager.initialize(m_documentState.canonicalPath(), std::move(config));
512}

760void AestraApp::loadOrRecoverProject(const std::string& projectPath, bool crashedSession) {
761 const bool hasRequestedProject = !projectPath.empty() && std::filesystem::exists(projectPath);
762 const std::string autosavePath = m_documentState.autosavePath();
763 std::string timestamp;
764 const std::string recoveryMarkerPath = getRecoveryMarkerPath(autosavePath);
765 std::vector<std::string> recoveryCandidates;
766 const bool primaryAutosaveDetected =
767 crashedSession && !m_previousRecoverySessionToken.empty() &&
768 RecoveryDialog::detectAutosave(autosavePath, recoveryMarkerPath, m_previousRecoverySessionToken, timestamp);
769 if (primaryAutosaveDetected) {
770 recoveryCandidates.push_back(autosavePath);
771 }
772 if (crashedSession && !m_previousRecoverySessionToken.empty()) {
773 for (auto& backup : Aestra::Audio::AutosaveManager::listBackupsForAutosavePath(autosavePath)) {
774 recoveryCandidates.push_back(std::move(backup));
775 }
776 }
777
778 if (!recoveryCandidates.empty()) {
779 const std::string recoveryDisplayPath = recoveryCandidates.front();
780 Log::info("[Recovery] Crash detected, showing recovery dialog");
781 m_pendingAutosavePath = recoveryDisplayPath;
782 m_recoveryHandled = false;
783
784 if (auto recoveryDialog = m_windowManager->getRecoveryDialog()) {
785 auto alive = m_aliveToken;
786 const std::string originalProjectPath =
787 readRecoveryOriginalProjectPath(recoveryMarkerPath, m_previousRecoverySessionToken);
788 recoveryDialog->show(recoveryDisplayPath,
789 [this, alive, autosavePath, projectPath, hasRequestedProject, originalProjectPath,
790 recoveryCandidates](Aestra::RecoveryResponse response) {
791 if (!*alive)
792 return;
793 m_recoveryHandled = true;
794 m_pendingAutosavePath.clear();
795 if (response == Aestra::RecoveryResponse::Recover) {
796 beginProjectLoadAttempt();
797 auto selected = ProjectSerializer::loadFirstValid(
798 recoveryCandidates, m_content ? m_content->getTrackManager() : nullptr, originalProjectPath);
799 auto result = std::move(selected.result);
800 recordProjectLoadAttempt(result, ProjectLoadSource::Recovery);
801 if (result.ok) {
802 result = applyLoadedProject(selected.loadedPath, ProjectLoadSource::Recovery,
803 originalProjectPath, std::move(result));
804 if (m_content && m_content->getTrackManager()) {
805 m_content->getTrackManager()->markModified();
806 }
807 m_autoSaveManager.markDirty();
808 updateWindowTitle();
809 Log::info("[Recovery] Autosave recovered successfully from: " + selected.loadedPath);
810 } else {
811 Log::error("[Recovery] Failed to load autosave");
812 if (m_content)
813 m_content->resetToDefaultProject();
814 clearProjectLoadReport();
815 m_documentState.startUntitled(autosavePathOrEmpty());
816 reinitAutosaveManager();
817 syncRecordingProjectPath(m_content, m_documentState.canonicalPath());
818 }
819 } else if (response == Aestra::RecoveryResponse::Discard) {
820 std::error_code ec;
821 std::filesystem::remove(autosavePath, ec);
822 std::filesystem::remove(getRecoveryMarkerPath(autosavePath), ec);
823 if (ec) {
824 Log::warning("[Recovery] Failed to remove autosave: " + ec.message());
825 } else {
826 Log::info("[Recovery] Autosave discarded");
827 }
828 if (hasRequestedProject) {
829 auto result = loadProjectFromPath(projectPath);
830 if (!result.ok) {
831 Log::error("Failed to load requested project after discarding recovery: " + projectPath);
832 }
833 } else {
834 if (m_content)
835 m_content->resetToDefaultProject();
836 clearProjectLoadReport();
837 m_documentState.startUntitled(autosavePathOrEmpty());
838 reinitAutosaveManager();
839 syncRecordingProjectPath(m_content, m_documentState.canonicalPath());
840 updateWindowTitle();
841 }
842 }
843 });
844 Log::info("[Recovery] Showing recovery dialog for autosave");
845 } else {
846 Log::warning("[Recovery] RecoveryDialog not available — discarding autosave");
847 std::error_code ec;
848 std::filesystem::remove(autosavePath, ec);
849 std::filesystem::remove(recoveryMarkerPath, ec);
850 m_recoveryHandled = true;
851 if (hasRequestedProject) {
852 loadProjectFromPath(projectPath);
853 } else {
854 clearProjectLoadReport();
855 }
856 }
857 return;
858 }
859
860 m_recoveryHandled = true;
861 if (hasRequestedProject) {
862 auto result = loadProjectFromPath(projectPath);
863 if (!result.ok) {
864 Log::error("Failed to load requested project: " + projectPath + " (" + result.errorMessage + ")");
865 }
866 }

985void AestraApp::finalizeAudioSetup() {
986 if (m_audioStreamReady) return;
987 StartupTimer t("Deferred audio stream start");
988 // Pass nullptr so openDefaultStream falls back to 'this' (the controller)
989 // as the audio-callback userData, which the callback expects.
990 if (m_audioController->openDefaultStream(nullptr)) {
991 // Establish the effective DSP configuration BEFORE the stream starts
992 // producing callbacks. This is not merely tidier ordering: setThreadCount
993 // replaces the engine's thread pool outright
994 // (m_threadPool = std::make_unique<RealTimeThreadPool>(count)), so running
995 // it against a live stream races the audio thread that is using that pool.
996 applyPersistedEngineSettings();
997
998 m_audioController->startStream();
999 if (m_content) {
1000 m_content->setAudioStatus(true);
1001 }
1002 connectAudioToUI(); // Sync configs now that stream is open
1003 m_audioStreamReady = true;
1004
1005 // Re-wire the performance HUD in case the engine wasn't constructed
1006 // yet when the HUD was created in initializeContent() (e.g. audio init
1007 // failed at startup and recovered here). Idempotent when already set.
1008 if (auto* hud = m_windowManager->getUnifiedHUD()) {
1009 hud->setAudioEngine(m_audioController->getEngine());
1010 }
1011 }
1012}
1013
1014// Apply the persisted DSP configuration once the engine exists (#649).
1015//
1016// These settings used to be applied by AudioSettingsPage's constructor, reached
1017// lazily through ensureSettingsAndDialogs() — so whether the engine ran with the
1018// user's safety-limiter, dither and multi-threading choices depended on whether
1019// they had opened the Settings dialog this session. The safety limiter in
1020// particular defaults to ON (AudioEngine.h), so a user who turned it off got it
1021// back every launch and lost it again mid-session on opening Settings.
1022//
1023// Startup owns this now. The Settings dialog edits and displays; it does not
1024// initialise.
1025void AestraApp::applyPersistedEngineSettings() {
1026 auto* engine = m_audioController ? m_audioController->getEngine() : nullptr;
1027 if (!engine) {
1028 return;
1029 }
1030
1031 const Aestra::AudioSettings saved = Aestra::loadAudioSettings();
1032
1033 if (Aestra::isSet(saved.masterLimiter)) {
1034 engine->setSafetyLimiterEnabled(saved.masterLimiter != 0);
1035 }
1036 if (Aestra::isSet(saved.dcRemoval)) {
1037 engine->setDCRemovalEnabled(saved.dcRemoval != 0);
1038 }
1039 if (Aestra::isSet(saved.dithering)) {
1040 if (Aestra::isPlausibleDitheringMode(saved.dithering)) {
1041 engine->setDitheringMode(static_cast<Aestra::Audio::DitheringMode>(saved.dithering));
1042 } else {
1043 Log::warning("[Audio] dithering " + std::to_string(saved.dithering) +
1044 " is not a valid mode; leaving the engine default");
1045 }
1046 }
1047 if (Aestra::isSet(saved.previewDuckingDb)) {
1048 engine->setPreviewDuckingAttenuationDb(static_cast<float>(saved.previewDuckingDb));
1049 }
1050 if (Aestra::isSet(saved.resampling) && !Aestra::isPlausibleResamplingIndex(saved.resampling)) {
1051 Log::warning("[Audio] resampling " + std::to_string(saved.resampling) +
1052 " is not a valid quality index; leaving the engine default");
1053 } else if (Aestra::isSet(saved.resampling)) {
1054 // Index-to-quality mapping mirrors AudioSettingsPage::applyChanges. It is
1055 // duplicated rather than shared only because the page cannot be linked
1056 // headlessly; unifying it belongs with the page rewire.
1057 using Aestra::Audio::Interpolators::InterpolationQuality;
1058 InterpolationQuality engineQ = InterpolationQuality::Sinc16;
1059 Aestra::Audio::ClipResamplingQuality globalQ = Aestra::Audio::ClipResamplingQuality::High;
1060 switch (saved.resampling) {
1061 case 0: engineQ = InterpolationQuality::Cubic; globalQ = Aestra::Audio::ClipResamplingQuality::Fast; break;
1062 case 1: engineQ = InterpolationQuality::Sinc32; globalQ = Aestra::Audio::ClipResamplingQuality::Draft; break;
1063 case 2: engineQ = InterpolationQuality::Cubic; globalQ = Aestra::Audio::ClipResamplingQuality::Standard; break;
1064 case 3: engineQ = InterpolationQuality::Sinc64; globalQ = Aestra::Audio::ClipResamplingQuality::High; break;
1065 default: break; // unreachable: guarded by isPlausibleResamplingIndex above
1066 }
1067 engine->setInterpolationQuality(engineQ);
1068 Aestra::Audio::PlaylistMixer::setResamplingQuality(globalQ);
1069 }
1070 if (Aestra::isSet(saved.multiThreading)) {
1071 const bool enabled = saved.multiThreading != 0;
1072 engine->setMultiThreadingEnabled(enabled);
1073 if (enabled && Aestra::isSet(saved.threads)) {
1074 const unsigned hw = std::thread::hardware_concurrency();
1075 const int maxThreads = hw > 0 ? static_cast<int>(hw) : 8;
1076 const int clamped = std::max(1, std::min(saved.threads, maxThreads));
1077 if (clamped != saved.threads) {
1078 Log::warning("[Audio] threads " + std::to_string(saved.threads) + " out of range; using " +
1079 std::to_string(clamped));
1080 }
1081 engine->setThreadCount(clamped);
1082 }
1083 }
1084
1085 Log::info("[Audio] Applied persisted engine settings at startup");
1086}

1118void AestraApp::run() {
1119 m_windowManager->render(); // Initial render — window visible NOW
1120
1121 // Complete heavy audio init after window is already showing,
1122 // so the UI feels instant even if the stream takes a second.
1123 finalizeAudioSetup();
1124
1125 while (m_running && m_windowManager->processEvents()) {
1126 UnifiedProfiler::getInstance().beginFrame();
1127 m_windowManager->beginFrame(); // Start timing
1128
1129 {
1130 AESTRA_ZONE("UI_Update");
1131 // Sync Transport State
1132 if (m_audioController->getEngine() && m_content && m_content->getTrackManager()) {
1133 auto engine = m_audioController->getEngine();
1134 auto tm = m_content->getTrackManager();
1135
1136 if (tm->isUserScrubbing()) {
1137 double scrubPos = tm->getPosition();
1138 uint32_t sr = engine->getSampleRate();
1139 if (sr > 0) engine->setGlobalSamplePos(static_cast<uint64_t>(scrubPos * sr));
1140 } else if (engine->isTransportPlaying() && tm->isPlaying()) {
1141 double realTime = engine->getPositionSeconds();
1142 tm->syncPositionFromEngine(realTime);
1143 }
1144
1145 if (m_content->getTransportBar()) {
1146 auto* transportBar = m_content->getTransportBar();
1147 transportBar->setPosition(tm->getPosition());
1148 transportBar->syncTransportState(tm->isPlaying(), tm->isPaused(), tm->isRecordArmed());
1149 }
1150 updateWindowTitle();
1151
1152 // Playback needs the 60 FPS target for smooth playhead/meters;
1153 // when stopped this decays back to the idle target via the
1154 // governor's timeout.
1155 if (auto* fps = m_windowManager->getAdaptiveFPS()) {
1156 fps->setAudioVisualizationActive(tm->isPlaying());
1157 }
1158 }
1159
1160 // Rebuild graph check - uses PlaybackGraphController for canonical drain
1161 if (m_audioController->getEngine() && m_content) {
1162 auto* controller = m_content->getPlaybackGraphController();
1163 if (controller) {
1164 controller->drainIfDirty(m_audioController->getSampleRate());
1165 }
1166 }
1167
1168 // Muse socket: execute agent requests on the main thread, exactly
1169 // like user edits (shared undo history, shared command system).
1170 if (m_museSocketServer && m_museService) {
1171 if (m_museSocketServer->processPending(*m_museService) > 0) {
1172 // Defeat idle frame elision so agent edits show up.
1173 if (auto* root = m_windowManager->getRootComponent()) {
1174 root->setDirty(true);
1175 }
1176 }
1177 }
1178 }
1179
1180 // ---- Idle frame elision (labs/perf/idle-frame-elision-spec.md) ----
1181 // Events and updates above always run at full cadence (input latency
1182 // is untouched); only the render+swap work is elided when nothing can
1183 // have changed on screen. A ~3 fps heartbeat keeps the window alive.
1184 const bool presentThisFrame = shouldRenderThisFrame();
1185
1186 if (presentThisFrame) {
1187 AESTRA_ZONE("Render_Prep");
1188 // Consume the pending-invalidation bit at render START: everything
1189 // dirty up to this point is included in this frame; anything that
1190 // dirties DURING render survives and triggers the next frame.
1191 // (Clearing after present would eat mid-render invalidations; the
1192 // skip path never clears.)
1193 if (auto* root = m_windowManager->getRootComponent()) {
1194 root->setDirty(false);
1195 }
1196 m_windowManager->render();
1197 }
1198
1199 // Capture the mutable project graph only on its owning application
1200 // thread. AutosaveManager commits the resulting immutable string on its
1201 // worker, so active edits never race ProjectSerializer traversal.
1202 m_autoSaveManager.captureSnapshotIfDue();
1203
1204 // Non-realtime audio work (graph rebuilds, plugin state commits) should
1205 // happen before frame pacing so it doesn't eat into the sleep budget.
1206 if (m_audioController && m_audioController->getEngine()) {
1207 m_audioController->getEngine()->performNonRealtimeMaintenance();
1208 }
1209
1210 double sleepTime = m_windowManager->endFrame();
1211 if (presentThisFrame) {
1212 m_windowManager->swapBuffers();
1213 m_lastPresentedFrame = std::chrono::steady_clock::now();
1214 }
1215 if (sleepTime > 0.0) {
1216 if (auto fps = m_windowManager->getAdaptiveFPS()) {
1217 fps->sleep(sleepTime);
1218 } else {
1219 std::this_thread::sleep_for(std::chrono::duration<double>(sleepTime));
1220 }
1221 }
1222
1223 UnifiedProfiler::getInstance().endFrame();
1224 }
1225}
1226
1227bool AestraApp::shouldRenderThisFrame() {
1228 // Conservative v1 gate: render unless EVERY skip condition holds
1229 // (dirty == false && realtime_visuals == false && input_recent == false).
1230
1231 // Pending invalidation — any component's setDirty(true) propagates here.
1232 auto* root = m_windowManager->getRootComponent();
1233 if (!root || root->isDirty())
1234 return true;
1235
1236 // Input recency — the adaptive FPS governor already tracks this (fed by
1237 // mouse/key callbacks); align with its idle timeout.
1238 auto* fps = m_windowManager->getAdaptiveFPS();
1239 if (!fps || fps->getIdleTime() < 2.0)
1240 return true;
1241
1242 // Realtime visuals — transport (playhead/meters), record-arm input
1243 // monitoring, file preview, and Audition playback.
1244 if (m_audioController && m_audioController->getEngine() && m_audioController->getEngine()->isTransportPlaying()) {
1245 return true;
1246 }
1247 if (m_content) {
1248 if (auto tm = m_content->getTrackManager()) {
1249 if (tm->isPlaying() || tm->isRecordArmed())
1250 return true;
1251 }
1252 if (m_content->hasRealtimePlaybackVisuals())
1253 return true;
1254 }
1255
1256 // Overlays that animate or need fresh samples (HUD, dialogs, menus).
1257 if (m_windowManager->requiresContinuousRender())
1258 return true;
1259
1260 // Deeply idle: heartbeat only (~3.3 fps) so caret blink, tooltips and any
1261 // unsignaled change still surface within ~300 ms.
1262 const auto now = std::chrono::steady_clock::now();
1263 const double sinceLastPresent = std::chrono::duration<double>(now - m_lastPresentedFrame).count();
1264 return sinceLastPresent >= 0.3;
1265}

1584ProjectSerializer::LoadResult AestraApp::applyLoadedProject(const std::string& path, ProjectLoadSource source,
1585 const std::string& canonicalPath,
1586 ProjectSerializer::LoadResult result) {
1587
1588 // Snapshot callers may pass the current canonical path by reference.
1589 // Resolve it before mutating the document state.
1590 const std::string resolvedCanonicalPath =
1591 canonicalPath.empty() ? m_documentState.canonicalPath() : canonicalPath;
1592
1593 switch (source) {
1594 case ProjectLoadSource::Canonical:
1595 m_documentState.openCanonical(path);
1596 break;
1597 case ProjectLoadSource::Recovery:
1598 m_documentState.recoverFrom(path, resolvedCanonicalPath);
1599 break;
1600 case ProjectLoadSource::Snapshot:
1601 m_documentState.restoreSnapshot(path, resolvedCanonicalPath);
1602 break;
1603 }
1604 if (result.integrity == ProjectSerializer::LoadIntegrity::Mismatch) {
1605 m_documentState.protectCanonicalFromOverwrite();
1606 }
1607
1608 if (m_audioController && m_audioController->getEngine()) {
1609 auto* engine = m_audioController->getEngine();
1610 engine->setTransportPlaying(false);
1611 engine->setBPM(static_cast<float>(result.tempo));
1612 if (m_content) {
1613 m_content->setPluginTempo(static_cast<float>(result.tempo));
1614 }
1615 const double sampleRate = std::max(1.0, static_cast<double>(engine->getSampleRate()));
1616 engine->setGlobalSamplePos(static_cast<uint64_t>(std::max(0.0, result.playhead) * sampleRate));
1617 }
1618
1619 if (m_content) {
1620 if (auto trackManager = m_content->getTrackManager()) {
1621 trackManager->getPlaylistModel().setBPM(result.tempo);
1622 trackManager->getTimelineClock().setTempo(result.tempo);
1623 trackManager->getPatternPlaybackEngine().flush();
1624 trackManager->setPosition(result.playhead);
1625 trackManager->setPlayStartPosition(result.playhead);
1626 trackManager->getCommandHistory().clear();
1627 // Only an in-memory transformation requires a save. Merely
1628 // advancing an older no-op schema stamp must not create a false
1629 // dirty prompt (#662).
1630 trackManager->setModified(result.requiresSaveAfterLoad());
1631 }
1632
1633 if (auto transportBar = m_content->getTransportBar()) {
1634 transportBar->setTempo(static_cast<float>(result.tempo));
1635 transportBar->setPosition(result.playhead);
1636 }
1637
1638 m_content->refreshProjectViews();
1639 }
1640
1641 // Project load - use PlaybackGraphController if available, otherwise fallback
1642 if (m_audioController && m_audioController->getEngine() && m_content) {
1643 auto* controller = m_content->getPlaybackGraphController();
1644 if (controller) {
1645 controller->requestRebuild(Aestra::Audio::PlaybackGraphController::GraphDirtyReason::ProjectLoaded);
1646 controller->drainIfDirty(m_audioController->getSampleRate());
1647 } else if (m_content->getTrackManager()) {
1648 auto graph = AudioGraphBuilder::buildFromTrackManager(*m_content->getTrackManager());
1649 m_audioController->getEngine()->setGraph(graph);
1650 m_content->getTrackManager()->rebuildAndPushSnapshot();
1651 }
1652 }
1653
1654 syncRecordingProjectPath(m_content, m_documentState.canonicalPath());
1655 reinitAutosaveManager();
1656 if (result.ui) {
1657 applyUIState(*result.ui);
1658 }
1659 // Loading may change the project (and its takes manifest) out from under
1660 // an open Takes panel — re-pull the providers.
1661 if (m_content) {
1662 if (auto takesPanel = m_content->getTakesPanel(); takesPanel && takesPanel->isVisible()) {
1663 takesPanel->refreshTakes();
1664 }
1665 }
1666 updateWindowTitle();
1667 Log::info("Project loaded into app state from " + path);
1668 return result;
1669}

1812ProjectSerializer::LoadResult AestraApp::switchToTake(const std::string& takeId) {
1813 ProjectSerializer::LoadResult result;
1814 if (takeId.empty()) {
1815 result.errorMessage = "Take id is empty";
1816 return result;
1817 }
1818 if (m_documentState.canonicalPath().empty()) {
1819 result.errorMessage = "Project path is empty";
1820 return result;
1821 }
1822
1823 if (!saveProject()) {
1824 result.errorMessage = "Could not save the current Take before switching";
1825 return result;
1826 }
1827
1828 const auto manifest = TakeManager::loadManifest(m_documentState.canonicalPath());
1829 if (!manifest.ok) {
1830 result.errorMessage = manifest.errorMessage;
1831 return result;
1832 }
1833
1834 const auto* take = manifest.findTake(takeId);
1835 if (!take) {
1836 result.errorMessage = "Take not found: " + takeId;
1837 return result;
1838 }
1839
1840 const std::string snapshotPath = TakeManager::resolveSnapshotPath(m_documentState.canonicalPath(), *take);
1841 if (snapshotPath.empty() || !std::filesystem::exists(snapshotPath)) {
1842 result.errorMessage = "Take snapshot is missing: " + snapshotPath;
1843 return result;
1844 }
1845
1846 result = loadProjectFromPath(snapshotPath, ProjectLoadSource::Snapshot, m_documentState.canonicalPath());
1847 if (!result.ok) {
1848 return result;
1849 }
1850
1851 auto activeResult = TakeManager::setActiveTake(m_documentState.canonicalPath(), takeId);
1852 if (!activeResult.ok) {
1853 auto rollback = loadProjectFromPath(m_documentState.canonicalPath());
1854 if (!rollback.ok) {
1855 Log::error("[Takes] CRITICAL: Failed to rollback after failed Take switch: " + rollback.errorMessage);
1856 }
1857 result.ok = false;
1858 result.errorMessage = activeResult.errorMessage;
1859 return result;
1860 }
1861
1862 if (!saveProject()) {
1863 auto rollback = loadProjectFromPath(m_documentState.canonicalPath());
1864 if (!rollback.ok) {
1865 Log::error("[Takes] CRITICAL: Failed to rollback after failed mirror save: " + rollback.errorMessage);
1866 }
1867 result.ok = false;
1868 result.errorMessage = "Switched Take but could not mirror it to the canonical project file";
1869 return result;
1870 }
1871
1872 Log::info("[Takes] Switched to Take: " + activeResult.take.name);
1873 return result;
1874}

1876bool AestraApp::branchFromTake(const std::string& takeId) {
1877 if (m_documentState.canonicalPath().empty() || takeId.empty()) {
1878 return false;
1879 }
1880
1881 // Preserve the working state before anything else touches the manifest.
1882 if (!saveProject()) {
1883 Log::warning("[Takes] Could not save the current Take before branching");
1884 return false;
1885 }
1886
1887 const auto manifest = TakeManager::loadManifest(m_documentState.canonicalPath());
1888 const auto* source = manifest.ok ? manifest.findTake(takeId) : nullptr;
1889 const std::string branchName = source ? (source->name + " Branch") : "";
1890
1891 auto dup = TakeManager::duplicateTake(m_documentState.canonicalPath(), takeId, branchName);
1892 if (!dup.ok) {
1893 Log::warning("[Takes] Could not branch from Take: " + dup.errorMessage);
1894 return false;
1895 }
1896
1897 auto result = switchToTake(dup.take.id);
1898 if (!result.ok) {
1899 // Don't leave a half-made branch behind: the duplicate is not active
1900 // (the failed switch rolled back), so it can be deleted safely.
1901 auto rollback = TakeManager::deleteTake(m_documentState.canonicalPath(), dup.take.id);
1902 Log::error("[Takes] Branch created but could not switch to it: " + result.errorMessage +
1903 (rollback.ok ? " (branch rolled back)" : " (rollback also failed: " + rollback.errorMessage + ")"));
1904 return false;
1905 }
1906 return true;
1907}
