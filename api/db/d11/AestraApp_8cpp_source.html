42struct StartupTimer {
43 const char* name;
44 std::chrono::steady_clock::time_point start;
45 explicit StartupTimer(const char* n)
46 : name(n), start(std::chrono::steady_clock::now()) {}
47 ~StartupTimer() {
48 auto elapsed = std::chrono::duration_cast<std::chrono::milliseconds>(
49 std::chrono::steady_clock::now() - start)
50 .count();
51 Log::info(std::string("[Startup] ") + name + ": " + std::to_string(elapsed) + " ms");
52 }
53};

125 std::error_code ec;
126 std::ofstream out(flagPath, std::ios::trunc);
127 if (out) {

179bool AestraApp::initialize(const std::string& projectPath) {
180 StartupTimer totalTimer("Total startup");
181
182 if (!transitionToInitializing()) return false;
183
184 Log::info("Aestra v1.0.0 - Initializing...");
185
186 // Check for crashed session BEFORE initializing platform.
187 bool crashedSession = isCrashedSession();
188 m_previousRecoverySessionToken = crashedSession ? readCrashFlagToken() : "";
189
190 {
191 StartupTimer t("Platform init");
192 if (!Platform::initialize()) {
193 Log::error("Failed to initialize platform");
194 return false;
195 }
196 }
197
198 // Initialize m_projectPath AFTER Platform is initialized
199 m_projectPath = getAutosavePath();
200
201 // ALWAYS write crash flag at session start. It is cleared only on clean
202 // shutdown. If the app crashes at any point, the flag persists and the
203 // next launch will detect it.
204 writeCrashFlag();
205 m_recoverySessionToken = readCrashFlagToken();
206
207 // Load preferences and UI state early
208 Preferences::instance().load();
209 bool autoSaveEnabled = Preferences::instance().autoSaveEnabled;
210
211 UIState uiState;
212 uiState.load();
213
214 {
215 StartupTimer t("Platform + Window");
216 if (!initializePlatformAndWindow(uiState)) return false;
217 }
218 {
219 StartupTimer t("Audio init");
220 initializeAudio();
221 }
222 {
223 StartupTimer t("Content init");
224 initializeContent();
225 }
226 {
227 StartupTimer t("Autosave init");
228 initializeAutosave(autoSaveEnabled);
229 }
230 {
231 StartupTimer t("Recovery dialog");
232 buildRecoveryDialog();
233 }
234 {
235 StartupTimer t("Menu bar");
236 buildMenuBar();
237 }
238 {
239 StartupTimer t("Plugins init");
240 initializePlugins();
241 }
242 {
243 StartupTimer t("Project load/recovery");
244 loadOrRecoverProject(projectPath, crashedSession);
245 }
246 {
247 StartupTimer t("UI state restore");
248 restoreUIState(uiState);
249 }
250
251 return transitionToRunning();
252}

322void AestraApp::initializeAutosave(bool enabled) {
323 Aestra::Audio::AutosaveManager::Config config;
324 config.enabled = enabled;
325 config.autosaveInterval = std::chrono::seconds(60);
326 config.autosavePathOverride = getAutosavePath();
327 config.onAutosaveCommitted = [this](const std::string& autosavePath) {
328 writeRecoveryMarkerForAutosave(autosavePath);
329 };
330 config.serializer = [this](std::string& outData) -> bool {
331 if (!m_content || !m_content->getTrackManager()) return false;
332 const double tempo = (m_audioController && m_audioController->getEngine())
333 ? static_cast<double>(m_audioController->getEngine()->getBPM())
334 : (m_content->getTransportBar() ? static_cast<double>(m_content->getTransportBar()->getTempo()) : 120.0);
335 const double playhead = m_content->getTrackManager()->getPosition();
336 auto ser = ProjectSerializer::serialize(m_content->getTrackManager(), tempo, playhead, 0);
337 if (!ser.ok) return false;
338 outData = std::move(ser.contents);
339 return true;
340 };
341 m_autoSaveManager.initialize(m_projectPath, std::move(config));
342}

654void AestraApp::loadOrRecoverProject(const std::string& projectPath, bool crashedSession) {
655 if (!projectPath.empty() && std::filesystem::exists(projectPath)) {
656 const std::string previousProjectPath = m_projectPath;
657 m_projectPath = projectPath;
658 auto result = loadProject();
659 if (result.ok) {
660 reinitAutosaveManager();
661 syncRecordingProjectPath(m_content, m_projectPath);
662 if (result.ui) applyUIState(*result.ui);
663 } else {
664 m_projectPath = previousProjectPath;
665 }
666 return;
667 }
668
669 std::string autosavePath = getAutosavePath();
670 std::string timestamp;
671
672 const std::string recoveryMarkerPath = getRecoveryMarkerPath(autosavePath);
673
674 if (crashedSession && !m_previousRecoverySessionToken.empty() &&

677 m_pendingAutosavePath = autosavePath;
678 m_recoveryHandled = false;
679
680 if (auto recoveryDialog = m_windowManager->getRecoveryDialog()) {
681 auto alive = m_aliveToken;
682 recoveryDialog->show(autosavePath, [this, alive, autosavePath](Aestra::RecoveryResponse response) {
683 if (!*alive) return;
684 m_recoveryHandled = true;
685 if (response == Aestra::RecoveryResponse::Recover) {
686 const std::string previousProjectPath = m_projectPath;
687 m_projectPath = autosavePath;
688 auto result = loadProject();
689 if (result.ok) {
690 reinitAutosaveManager();
691 syncRecordingProjectPath(m_content, m_projectPath);
692 if (result.ui) applyUIState(*result.ui);
693 Log::info("[Recovery] Autosave recovered successfully");
694 } else {
695 m_projectPath = previousProjectPath;
696 Log::error("[Recovery] Failed to load autosave");
697 if (m_content) m_content->resetToDefaultProject();
698 }
699 } else if (response == Aestra::RecoveryResponse::Discard) {
700 std::error_code ec;
701 std::filesystem::remove(autosavePath, ec);
702 std::filesystem::remove(getRecoveryMarkerPath(autosavePath), ec);
703 if (ec) {
704 Log::warning("[Recovery] Failed to remove autosave: " + ec.message());
705 } else {
706 Log::info("[Recovery] Autosave discarded");
707 }
708 if (m_content) m_content->resetToDefaultProject();
709 m_projectPath = getAutosavePath();
710 reinitAutosaveManager();
711 syncRecordingProjectPath(m_content, m_projectPath);
712 }
713 });
714 Log::info("[Recovery] Showing recovery dialog for autosave");
715 } else {
716 Log::warning("[Recovery] RecoveryDialog not available — discarding autosave");
717 std::error_code ec;
718 std::filesystem::remove(autosavePath, ec);
719 std::filesystem::remove(recoveryMarkerPath, ec);
720 m_recoveryHandled = true;
721 }
722 } else {
723 m_recoveryHandled = true;

900void AestraApp::run() {
901 m_windowManager->render(); // Initial render — window visible NOW
902
903 // Complete heavy audio init after window is already showing,
904 // so the UI feels instant even if the stream takes a second.
905 finalizeAudioSetup();
906
907 while (m_running && m_windowManager->processEvents()) {
908 UnifiedProfiler::getInstance().beginFrame();
909 m_windowManager->beginFrame(); // Start timing
910
911 {
912 AESTRA_ZONE("UI_Update");
913 // Sync Transport State
914 if (m_audioController->getEngine() && m_content && m_content->getTrackManager()) {
915 auto engine = m_audioController->getEngine();
916 auto tm = m_content->getTrackManager();
917
918 if (tm->isUserScrubbing()) {
919 double scrubPos = tm->getPosition();
920 uint32_t sr = engine->getSampleRate();
921 if (sr > 0) engine->setGlobalSamplePos(static_cast<uint64_t>(scrubPos * sr));
922 } else if (engine->isTransportPlaying() && tm->isPlaying()) {
923 double realTime = engine->getPositionSeconds();
924 tm->syncPositionFromEngine(realTime);
925 }
926
927 if (m_content->getTransportBar()) {
928 auto* transportBar = m_content->getTransportBar();
929 transportBar->setPosition(tm->getPosition());
930 transportBar->syncTransportState(tm->isPlaying(), tm->isPaused(), tm->isRecordArmed());
931 }
932 updateWindowTitle();
933
934 // Playback needs the 60 FPS target for smooth playhead/meters;
935 // when stopped this decays back to the idle target via the
936 // governor's timeout.
937 if (auto* fps = m_windowManager->getAdaptiveFPS()) {
938 fps->setAudioVisualizationActive(tm->isPlaying());
939 }
940 }
941
942 // Rebuild graph check - uses PlaybackGraphController for canonical drain
943 if (m_audioController->getEngine() && m_content) {
944 auto* controller = m_content->getPlaybackGraphController();
945 if (controller) {
946 controller->drainIfDirty(m_audioController->getSampleRate());
947 }
948 }
949 }
950
951 // ---- Idle frame elision (labs/perf/idle-frame-elision-spec.md) ----
952 // Events and updates above always run at full cadence (input latency
953 // is untouched); only the render+swap work is elided when nothing can
954 // have changed on screen. A ~3 fps heartbeat keeps the window alive.
955 const bool presentThisFrame = shouldRenderThisFrame();
956
957 if (presentThisFrame) {
958 AESTRA_ZONE("Render_Prep");
959 // Consume the pending-invalidation bit at render START: everything
960 // dirty up to this point is included in this frame; anything that
961 // dirties DURING render survives and triggers the next frame.
962 // (Clearing after present would eat mid-render invalidations; the
963 // skip path never clears.)
964 if (auto* root = m_windowManager->getRootComponent()) {
965 root->setDirty(false);
966 }
967 m_windowManager->render();
968 }
969
970 // Autosave is handled entirely by AutosaveManager's background thread,
971 // triggered via markDirty() from TrackManager::setOnModified().
972
973 // Non-realtime audio work (graph rebuilds, plugin state commits) should
974 // happen before frame pacing so it doesn't eat into the sleep budget.
975 if (m_audioController && m_audioController->getEngine()) {
976 m_audioController->getEngine()->performNonRealtimeMaintenance();
977 }
978
979 double sleepTime = m_windowManager->endFrame();
980 if (presentThisFrame) {
981 m_windowManager->swapBuffers();
982 m_lastPresentedFrame = std::chrono::steady_clock::now();
983 }
984 if (sleepTime > 0.0) {
985 if (auto fps = m_windowManager->getAdaptiveFPS()) {
986 fps->sleep(sleepTime);
987 } else {
988 std::this_thread::sleep_for(std::chrono::duration<double>(sleepTime));
989 }
990 }
991
992 UnifiedProfiler::getInstance().endFrame();
993 }

996bool AestraApp::shouldRenderThisFrame() {
997 // Conservative v1 gate: render unless EVERY skip condition holds
998 // (dirty == false && realtime_visuals == false && input_recent == false).
999
1000 // Pending invalidation — any component's setDirty(true) propagates here.
1001 auto* root = m_windowManager->getRootComponent();
1002 if (!root || root->isDirty())
1003 return true;
1004
1005 // Input recency — the adaptive FPS governor already tracks this (fed by
1006 // mouse/key callbacks); align with its idle timeout.
1007 auto* fps = m_windowManager->getAdaptiveFPS();
1008 if (!fps || fps->getIdleTime() < 2.0)
1009 return true;
1010
1011 // Realtime visuals — transport (playhead/meters), record-arm input
1012 // monitoring, file preview, and Audition playback.
1013 if (m_audioController && m_audioController->getEngine() && m_audioController->getEngine()->isTransportPlaying()) {
1014 return true;
1015 }
1016 if (m_content) {
1017 if (auto tm = m_content->getTrackManager()) {
1018 if (tm->isPlaying() || tm->isRecordArmed())
1019 return true;
1020 }
1021 if (m_content->hasRealtimePlaybackVisuals())
1022 return true;
1023 }
1024
1025 // Overlays that animate or need fresh samples (HUD, dialogs, menus).
1026 if (m_windowManager->requiresContinuousRender())
1027 return true;
1028
1029 // Deeply idle: heartbeat only (~3.3 fps) so caret blink, tooltips and any
1030 // unsignaled change still surface within ~300 ms.
1031 const auto now = std::chrono::steady_clock::now();
1032 const double sinceLastPresent = std::chrono::duration<double>(now - m_lastPresentedFrame).count();
1033 return sinceLastPresent >= 0.3;
1034}

1045 // the autosave is still available for recovery on next launch.

1047 Log::info("[SHUTDOWN] Emergency autosave before shutdown...");

1081 }

1084 if (m_content && m_content->getFileBrowser()) {

1101
1102 // Issue #120: Capture track view zoom/scroll state
1103 if (m_content && m_content->getTrackManagerUI()) {

1108 Log::info("[UIState] Captured track view state: zoom=" + std::to_string(uiState.horizontalZoom) +
1109 ", hScroll=" + std::to_string(uiState.scrollPositionX) +
1110 ", vScroll=" + std::to_string(uiState.scrollPositionY));
1111 }
1112
1113 uiState.save();
1114
1115 if (m_audioController) {
1116 m_audioController->stopStream();
1117 m_audioController->closeStream();
1118 }
1119
1120 // AestraContent owns UI-facing preview state and detaches it from AudioEngine
1121 // in its destructor. Destroy content while the engine still exists.
1122 if (m_windowManager) {
1123 m_windowManager->shutdown();
1124 }
1125 m_content.reset();
1126
1127 Aestra::Audio::PluginManager::getInstance().shutdown();
1128
1129 if (m_audioController) {
1130 m_audioController->shutdown();
1131 }
1132
1133 Platform::shutdown();

1136
1137 // Clear crash flag LAST — if anything above crashes, the flag persists.
1138 // Also clear ServiceLocator last so shutdown paths can still resolve services.
1139 Aestra::ServiceLocator::clear();
1140 clearCrashFlag();
1141 Log::shutdown();
1142}
