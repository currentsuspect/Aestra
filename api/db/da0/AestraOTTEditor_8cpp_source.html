98void AestraOTTEditor::drawBypassPill(NUIRenderer& renderer) {
99 auto& theme = NUIThemeManager::getInstance();
100 const bool bypassed = m_instance && m_instance->getParameter(AestraOTT::kBypass) > 0.5f;
101 constexpr float kRadius = 7.0f;
102 constexpr float kFont = 10.0f;
103 if (bypassed) {
104 renderer.fillRoundedRect(m_bypassRect, kRadius,
105 NUIColor(0.92f, 0.28f, 0.22f).withAlpha(m_bypassHovered ? 0.94f : 0.78f));
106 renderer.strokeRoundedRect(m_bypassRect, kRadius, 1.0f, NUIColor(0.92f, 0.28f, 0.22f).withAlpha(0.50f));
107 renderer.drawTextCentered("BYPASSED", m_bypassRect, kFont, theme.getColor("textPrimary"));
108 } else {
109 renderer.fillRoundedRect(m_bypassRect, kRadius,
110 theme.getColor("success").withAlpha(m_bypassHovered ? 0.30f : 0.18f));
111 renderer.strokeRoundedRect(m_bypassRect, kRadius, 1.0f, theme.getColor("success").withAlpha(0.40f));
112 renderer.drawTextCentered("ACTIVE", m_bypassRect, kFont, theme.getColor("success"));
113 }
114}

116bool AestraOTTEditor::onMouseEvent(const NUIMouseEvent& event) {
117 if (!isVisible())
118 return false;
119 if (AestraPanelWindow::onMouseEvent(event))
120 return true;
121 if (!m_instance)
122 return false;
123
124
125 // Bypass pill
126 if (m_bypassRect.contains(event.position) && event.pressed && event.button == NUIMouseButton::Left) {
127 const float cur = m_instance->getParameter(AestraOTT::kBypass);
128 m_instance->setParameter(AestraOTT::kBypass, cur > 0.5f ? 0.0f : 1.0f);
129 setDirty();
130 return true;
131 }
132 m_bypassHovered = m_bypassRect.contains(event.position);
133
134 // Knob vertical drag (rotary -> cursor capture): AestraPanelWindow::handleKnobDrag.
135 const KnobTarget knobs[] = {
136 {m_depthRect, AestraOTT::kDepth}, {m_timeRect, AestraOTT::kTime}, {m_inRect, AestraOTT::kInGain},
137 {m_outRect, AestraOTT::kOutGain}, {m_lowRect, AestraOTT::kLowGain}, {m_midRect, AestraOTT::kMidGain},
138 {m_highRect, AestraOTT::kHighGain}, {m_xloRect, AestraOTT::kXoverLow}, {m_xhiRect, AestraOTT::kXoverHigh},
139 };
140 const auto get = [this](uint32_t id) { return m_instance->getParameter(id); };
141 const auto set = [this](uint32_t id, float v) { m_instance->setParameter(id, v); };
142 if (handleKnobDrag(event, knobs, std::size(knobs), kDragRangePx, get, set))
143 return true;
144
145 return consumeInsideBounds(event);
146}
