32void drawArc(NUIRenderer& renderer, NUIPoint center, float radius, float startAngle, float endAngle, float thickness,
33 NUIColor color) {
34 if (endAngle < startAngle)
35 std::swap(startAngle, endAngle);
36 if (endAngle - startAngle <= 0.001f)
37 return;
38 std::array<NUIPoint, 49> pts{};
39 const float div = static_cast<float>(pts.size() - 1);
40 for (size_t i = 0; i < pts.size(); ++i) {
41 const float t = static_cast<float>(i) / div;
42 const float a = startAngle + (endAngle - startAngle) * t;
43 pts[i] = {center.x + std::cos(a) * radius, center.y + std::sin(a) * radius};
44 }
45 renderer.drawPolyline(pts.data(), static_cast<int>(pts.size()), thickness, color);
46}

107void AestraOTTEditor::drawKnob(NUIRenderer& renderer, const NUIRect& rect, uint32_t paramId, const char* label,
108 bool large, bool bipolar) {
109 auto& theme = NUIThemeManager::getInstance();
110 const float value = m_instance ? m_instance->getParameter(paramId) : 0.0f;
111 const NUIPoint c = rect.center();
112 const float r = rect.width * 0.5f - 4.0f;
113 const float angle = kKnobStart + value * kKnobSweep;
114
115 renderer.fillCircle(c, r + 4.0f, insetSurface());
116 renderer.strokeCircle(c, r + 4.0f, 1.0f, editorInk(0.060f));
117
118 drawArc(renderer, c, r - 3.0f, kKnobStart, kKnobStart + kKnobSweep, large ? 4.0f : 3.0f,
119 editorNeutral(0.199f, 1.0f));
120 if (bipolar) {
121 // Bipolar knobs fill from the top-center detent toward the value.
122 const float centerAngle = kKnobStart + 0.5f * kKnobSweep;
123 drawArc(renderer, c, r - 3.0f, std::min(centerAngle, angle), std::max(centerAngle, angle), 3.0f,
124 accent().withAlpha(0.92f));
125 } else {
126 drawArc(renderer, c, r - 3.0f, kKnobStart, angle, large ? 4.0f : 3.0f, accent().withAlpha(0.92f));
127 }
128
129 const float needleLen = r - (large ? 14.0f : 9.0f);
130 const NUIPoint tip(c.x + std::cos(angle) * needleLen, c.y + std::sin(angle) * needleLen);
131 renderer.drawLine(c, tip, 2.0f, accent().withAlpha(0.85f));
132 renderer.fillCircle(tip, large ? 3.5f : 2.5f, accent());
133
134 const float wellR = r * (large ? 0.34f : 0.28f);
135 renderer.fillCircle(c, wellR, editorNeutral(0.045f, 0.96f));
136 renderer.strokeCircle(c, wellR, 1.2f, accent().withAlpha(0.36f));
137
138 renderer.drawTextCentered(label, NUIRect(rect.x, rect.bottom() + 4.0f, rect.width, 14.0f), 10.5f,
139 theme.getColor("textPrimary").withAlpha(0.95f));
140 const std::string valStr = m_instance ? m_instance->getParameterDisplay(paramId) : "";
141 renderer.drawTextCentered(valStr, NUIRect(rect.x - 14.0f, rect.bottom() + 19.0f, rect.width + 28.0f, 13.0f),
142 large ? 10.0f : 9.0f, accent().withAlpha(0.96f));
143}

145void AestraOTTEditor::drawBypassPill(NUIRenderer& renderer) {
146 auto& theme = NUIThemeManager::getInstance();
147 const bool bypassed = m_instance && m_instance->getParameter(AestraOTT::kBypass) > 0.5f;
148 constexpr float kRadius = 7.0f;
149 constexpr float kFont = 10.0f;
150 if (bypassed) {
151 renderer.fillRoundedRect(m_bypassRect, kRadius,
152 NUIColor(0.92f, 0.28f, 0.22f).withAlpha(m_bypassHovered ? 0.94f : 0.78f));
153 renderer.strokeRoundedRect(m_bypassRect, kRadius, 1.0f, NUIColor(0.92f, 0.28f, 0.22f).withAlpha(0.50f));
154 renderer.drawTextCentered("BYPASSED", m_bypassRect, kFont, theme.getColor("textPrimary"));
155 } else {
156 renderer.fillRoundedRect(m_bypassRect, kRadius,
157 theme.getColor("success").withAlpha(m_bypassHovered ? 0.30f : 0.18f));
158 renderer.strokeRoundedRect(m_bypassRect, kRadius, 1.0f, theme.getColor("success").withAlpha(0.40f));
159 renderer.drawTextCentered("ACTIVE", m_bypassRect, kFont, theme.getColor("success"));
160 }

163bool AestraOTTEditor::onMouseEvent(const NUIMouseEvent& event) {
164 if (!isVisible())
165 return false;
166 if (AestraPanelWindow::onMouseEvent(event))
167 return true;
168 if (!m_instance)
169 return false;
170
171 const float my = event.position.y;
172
173 // Bypass pill
174 if (m_bypassRect.contains(event.position) && event.pressed && event.button == NUIMouseButton::Left) {
175 const float cur = m_instance->getParameter(AestraOTT::kBypass);
176 m_instance->setParameter(AestraOTT::kBypass, cur > 0.5f ? 0.0f : 1.0f);
177 setDirty();
178 return true;
179 }
180 m_bypassHovered = m_bypassRect.contains(event.position);
181
182 // Knob vertical drag
183 if (m_draggingParam >= 0) {
184 if (event.released) {
185 m_draggingParam = -1;
186 return true;
187 }
188 if (event.button == NUIMouseButton::None) {
189 const float delta = (m_dragStartY - my) / kDragRangePx;
190 m_instance->setParameter(static_cast<uint32_t>(m_draggingParam),
191 std::clamp(m_dragStartValue + delta, 0.0f, 1.0f));
192 setDirty();
193 return true;
194 }
195 }
196 if (event.pressed && event.button == NUIMouseButton::Left) {
197 const struct {
198 NUIRect rect;
199 uint32_t param;
200 } knobs[] = {
201 {m_depthRect, AestraOTT::kDepth}, {m_timeRect, AestraOTT::kTime}, {m_inRect, AestraOTT::kInGain},
202 {m_outRect, AestraOTT::kOutGain}, {m_lowRect, AestraOTT::kLowGain}, {m_midRect, AestraOTT::kMidGain},
203 {m_highRect, AestraOTT::kHighGain}, {m_xloRect, AestraOTT::kXoverLow}, {m_xhiRect, AestraOTT::kXoverHigh},
204 };
205 for (const auto& k : knobs) {
206 if (k.rect.contains(event.position)) {
207 m_draggingParam = static_cast<int>(k.param);
208 m_dragStartY = my;
209 m_dragStartValue = m_instance->getParameter(k.param);
210 return true;
211 }
212 }
213 }
214
215 return consumeInsideBounds(event);
216}
