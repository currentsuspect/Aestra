34class AudioArena {
35public:
36 explicit AudioArena(size_t capacityBytes)
37 : m_capacity(capacityBytes) {
38 // Allocate with max_align_t alignment (typically 16 bytes).
39 // For stricter alignments, we add padding to guarantee alignment
40 // up to 64 bytes regardless of the base pointer alignment.
41 m_buffer = new (std::nothrow) uint8_t[capacityBytes + 63];
42 m_rawBuffer = m_buffer;
43 // Align the buffer start to 64 bytes (covers all common alignments)
44 uintptr_t addr = reinterpret_cast<uintptr_t>(m_rawBuffer);
45 uintptr_t aligned = (addr + 63) & ~static_cast<uintptr_t>(63);
46 m_buffer = reinterpret_cast<uint8_t*>(aligned);
47 m_bufferSize = capacityBytes + 63 - (aligned - addr);
48 m_offset.store(0, std::memory_order_relaxed);
49 m_peak.store(0, std::memory_order_relaxed);
50 m_allocCount.store(0, std::memory_order_relaxed);
51 }
52
53 ~AudioArena() {
54 delete[] m_rawBuffer;
55 }
56
57 // Non-copyable, non-movable
58 AudioArena(const AudioArena&) = delete;
59 AudioArena& operator=(const AudioArena&) = delete;
60 AudioArena(AudioArena&&) = delete;
61 AudioArena& operator=(AudioArena&&) = delete;
62
72 void* allocate(size_t size, size_t alignment = alignof(std::max_align_t)) noexcept {
73 if (!m_buffer || size == 0)
74 return nullptr;
75
76 // Alignment must be power of 2
77 if ((alignment & (alignment - 1)) != 0)
78 return nullptr;
79
80 size_t currentOffset;
81 size_t newOffset;
82 size_t alignedOffset;
83
84 do {
85 currentOffset = m_offset.load(std::memory_order_relaxed);
86
87 // Align the current offset
88 alignedOffset = (currentOffset + alignment - 1) & ~(alignment - 1);
89 newOffset = alignedOffset + size;
90
91 // Check if we have room
92 if (newOffset > m_capacity)
93 return nullptr;
94
95 // Try to claim this space via atomic CAS
96 } while (!m_offset.compare_exchange_weak(currentOffset, newOffset,
97 std::memory_order_relaxed,
98 std::memory_order_relaxed));
99
100 // Update peak usage (best-effort, no CAS needed — monotonic is fine)
101 size_t peak = m_peak.load(std::memory_order_relaxed);
102 while (newOffset > peak) {
103 if (m_peak.compare_exchange_weak(peak, newOffset,
104 std::memory_order_relaxed,
105 std::memory_order_relaxed))
106 break;
107 }
108
109 m_allocCount.fetch_add(1, std::memory_order_relaxed);
110
111 return static_cast<void*>(m_buffer + alignedOffset);
112 }
113
120 void reset() noexcept {
121 m_offset.store(0, std::memory_order_relaxed);
122 m_peak.store(0, std::memory_order_relaxed);
123 m_allocCount.store(0, std::memory_order_relaxed);
124 std::memset(m_buffer, 0, m_capacity);
125 }
126
128 size_t capacity() const noexcept { return m_capacity; }
129
131 size_t used() const noexcept { return m_offset.load(std::memory_order_relaxed); }
132
134 size_t remaining() const noexcept { return m_capacity - m_offset.load(std::memory_order_relaxed); }
135
137 size_t peakUsage() const noexcept { return m_peak.load(std::memory_order_relaxed); }
138
140 size_t allocationCount() const noexcept { return m_allocCount.load(std::memory_order_relaxed); }
141
143 bool isEmpty() const noexcept { return m_offset.load(std::memory_order_relaxed) == 0; }
144
145private:
146 uint8_t* m_rawBuffer{nullptr};
147 uint8_t* m_buffer{nullptr};
148 size_t m_capacity{0};
149 size_t m_bufferSize{0};
150 std::atomic<size_t> m_offset{0};
151 std::atomic<size_t> m_peak{0};
152 std::atomic<size_t> m_allocCount{0};
153};
