2// What a hosted plugin is told about the song: tempo, metre, playing or stopped, and where the playhead is.
3//
4// VST3 (ProcessContext) and CLAP (clap_event_transport) plugins used to get nothing, so a tempo-synced delay or LFO
5// had no tempo, an arpeggiator never knew the song was playing, and Vital reset its own BPM. The engine publishes
6// one snapshot per render block (AudioEngine::renderGraph); every plugin host reads it in process(). In the sandbox
7// the app sends the snapshot with each block and the helper publishes it in its own process, so the same hosts
8// read it there too.
9//
10// One writer (the render thread), any number of readers, no locks: a sequence counter brackets each write, and a
11// reader that sees it move retries a few times, then reports nothing rather than a torn snapshot.

25struct HostTransport {
26 bool valid = false;
27 bool playing = false;
28 bool looping = false;
29 double tempo = 120.0;
30 int32_t timeSigNumerator = 4;
31 int32_t timeSigDenominator = 4;
32 int64_t samplePosition = 0;
33 double sampleRate = 48000.0;
34 double ppqPosition = 0.0;
35 double barStartPpq = 0.0;
36 double loopStartPpq = 0.0;
37 double loopEndPpq = 0.0;

40 static HostTransport at(int64_t samplePosition, double sampleRate, double tempo, int32_t beatsPerBar, bool playing,
41 bool looping, double loopStartBeat, double loopEndBeat) noexcept {
42 HostTransport t;
43 t.valid = sampleRate > 0.0 && tempo > 0.0;
44 t.playing = playing;
45 t.looping = looping;
46 t.tempo = tempo;
47 t.timeSigNumerator = beatsPerBar > 0 ? beatsPerBar : 4;
48 t.timeSigDenominator = 4;
49 t.samplePosition = samplePosition;
50 t.sampleRate = sampleRate;
51 if (t.valid) {
52 t.ppqPosition = static_cast<double>(samplePosition) / sampleRate * tempo / 60.0;
53 t.barStartPpq = std::floor(t.ppqPosition / t.timeSigNumerator) * t.timeSigNumerator;
54 }
55 t.loopStartPpq = loopStartBeat;
56 t.loopEndPpq = loopEndBeat;
57 return t;
58 }

64 std::string toWire() const {
65 if (!valid)
66 return {};
67 char buf[256];
68 const int n = std::snprintf(buf, sizeof buf, " T %u %d %d %lld", 1u | (playing ? 2u : 0u) | (looping ? 4u : 0u),
69 timeSigNumerator, timeSigDenominator, static_cast<long long>(samplePosition));
70 std::string out(buf, n > 0 ? static_cast<size_t>(n) : 0);
71 for (double d : {tempo, sampleRate, ppqPosition, barStartPpq, loopStartPpq, loopEndPpq}) {
72 uint64_t bits = 0;
73 std::memcpy(&bits, &d, sizeof bits);
74 std::snprintf(buf, sizeof buf, " %016llx", static_cast<unsigned long long>(bits));
75 out += buf;
76 }
77 return out;
78 }

82 static HostTransport fromWire(const std::string* tokens, size_t count) noexcept {
83 HostTransport t;
84 if (count != 10)
85 return t;
86 const auto integer = [](const std::string& s, long long& out) {
87 char* end = nullptr;
88 out = std::strtoll(s.c_str(), &end, 10);
89 return !s.empty() && end && *end == '\0';
90 };
91 const auto real = [](const std::string& s, double& out) {
92 char* end = nullptr;
93 const unsigned long long bits = std::strtoull(s.c_str(), &end, 16);
94 if (s.size() != 16 || !end || *end != '\0')
95 return false;
96 const uint64_t b = bits;
97 std::memcpy(&out, &b, sizeof out);
98 return std::isfinite(out);
99 };
100 long long flags = 0, num = 0, den = 0, sample = 0;
101 if (!integer(tokens[0], flags) || !integer(tokens[1], num) || !integer(tokens[2], den) ||
102 !integer(tokens[3], sample))
103 return t;
104 double d[6] = {};
105 for (size_t i = 0; i < 6; ++i)
106 if (!real(tokens[4 + i], d[i]))
107 return t;
108 if ((flags & 1) == 0 || num < 1 || num > 64 || den < 1 || den > 64 || d[0] <= 0.0 || d[0] > 1000.0 ||
109 d[1] <= 0.0)
110 return t;
111 t.playing = (flags & 2) != 0;
112 t.looping = (flags & 4) != 0;
113 t.timeSigNumerator = static_cast<int32_t>(num);
114 t.timeSigDenominator = static_cast<int32_t>(den);
115 t.samplePosition = sample;
116 t.tempo = d[0];
117 t.sampleRate = d[1];
118 t.ppqPosition = d[2];
119 t.barStartPpq = d[3];
120 t.loopStartPpq = d[4];
121 t.loopEndPpq = d[5];
122 t.valid = true;
123 return t;
124 }
125};

129struct ClapTransportValues {
130 static constexpr double kFactor = 2147483648.0; // CLAP_BEATTIME_FACTOR and CLAP_SECTIME_FACTOR, 1 << 31
131 uint32_t flags = 0;
132 int64_t songPosBeats = 0;
133 int64_t songPosSeconds = 0;
134 double tempo = 0.0;
135 int64_t loopStartBeats = 0;
136 int64_t loopEndBeats = 0;
137 int64_t loopStartSeconds = 0;
138 int64_t loopEndSeconds = 0;
139 int64_t barStart = 0;
140 int32_t barNumber = 0;
141 uint16_t tsigNum = 4;
142 uint16_t tsigDenom = 4;
143
144 static ClapTransportValues from(const HostTransport& t) noexcept {
145 // CLAP_TRANSPORT_HAS_TEMPO | HAS_BEATS_TIMELINE | HAS_SECONDS_TIMELINE | HAS_TIME_SIGNATURE, then IS_PLAYING
146 // (1 << 4) and IS_LOOP_ACTIVE (1 << 6).
147 ClapTransportValues v;
148 if (!t.valid)
149 return v;
150 v.flags = 1u | 2u | 4u | 8u;
151 if (t.playing)
152 v.flags |= 1u << 4;
153 const double secondsPerBeat = 60.0 / t.tempo;
154 const auto beats = [](double ppq) { return static_cast<int64_t>(std::llround(ppq * kFactor)); };
155 const auto seconds = [&](double ppq) { return static_cast<int64_t>(std::llround(ppq * secondsPerBeat * kFactor)); };
156 v.songPosBeats = beats(t.ppqPosition);
157 v.songPosSeconds = static_cast<int64_t>(std::llround(static_cast<double>(t.samplePosition) / t.sampleRate * kFactor));
158 v.tempo = t.tempo;
159 if (t.looping && t.loopEndPpq > t.loopStartPpq) {
160 v.flags |= 1u << 6;
161 v.loopStartBeats = beats(t.loopStartPpq);
162 v.loopEndBeats = beats(t.loopEndPpq);
163 v.loopStartSeconds = seconds(t.loopStartPpq);
164 v.loopEndSeconds = seconds(t.loopEndPpq);
165 }
166 v.barStart = beats(t.barStartPpq);
167 v.barNumber = static_cast<int32_t>(t.barStartPpq / t.timeSigNumerator);
168 v.tsigNum = static_cast<uint16_t>(t.timeSigNumerator);
169 v.tsigDenom = static_cast<uint16_t>(t.timeSigDenominator);
170 return v;
171 }
172};

174class HostTransportClock {
175public:
177 static void publish(const HostTransport& t) noexcept {
178 Slot& s = slot();
179 const uint32_t seq = s.seq.load(std::memory_order_relaxed);
180 s.seq.store(seq + 1, std::memory_order_relaxed); // odd: a write is under way
181 std::atomic_thread_fence(std::memory_order_release);
182 s.flags.store((t.valid ? 1u : 0u) | (t.playing ? 2u : 0u) | (t.looping ? 4u : 0u), std::memory_order_relaxed);
183 s.tempo.store(t.tempo, std::memory_order_relaxed);
184 s.timeSig.store((static_cast<uint32_t>(t.timeSigNumerator) << 16) |
185 static_cast<uint32_t>(t.timeSigDenominator & 0xFFFF),
186 std::memory_order_relaxed);
187 s.samplePosition.store(t.samplePosition, std::memory_order_relaxed);
188 s.sampleRate.store(t.sampleRate, std::memory_order_relaxed);
189 s.ppqPosition.store(t.ppqPosition, std::memory_order_relaxed);
190 s.barStartPpq.store(t.barStartPpq, std::memory_order_relaxed);
191 s.loopStartPpq.store(t.loopStartPpq, std::memory_order_relaxed);
192 s.loopEndPpq.store(t.loopEndPpq, std::memory_order_relaxed);
193 s.seq.store(seq + 2, std::memory_order_release);
194 }
195
197 static HostTransport current() noexcept {
198 const Slot& s = slot();
199 for (int attempt = 0; attempt < 4; ++attempt) {
200 const uint32_t before = s.seq.load(std::memory_order_acquire);
201 if (before & 1u)
202 continue;
203 HostTransport t;
204 const uint32_t flags = s.flags.load(std::memory_order_relaxed);
205 t.valid = (flags & 1u) != 0;
206 t.playing = (flags & 2u) != 0;
207 t.looping = (flags & 4u) != 0;
208 t.tempo = s.tempo.load(std::memory_order_relaxed);
209 const uint32_t sig = s.timeSig.load(std::memory_order_relaxed);
210 t.timeSigNumerator = static_cast<int32_t>(sig >> 16);
211 t.timeSigDenominator = static_cast<int32_t>(sig & 0xFFFF);
212 t.samplePosition = s.samplePosition.load(std::memory_order_relaxed);
213 t.sampleRate = s.sampleRate.load(std::memory_order_relaxed);
214 t.ppqPosition = s.ppqPosition.load(std::memory_order_relaxed);
215 t.barStartPpq = s.barStartPpq.load(std::memory_order_relaxed);
216 t.loopStartPpq = s.loopStartPpq.load(std::memory_order_relaxed);
217 t.loopEndPpq = s.loopEndPpq.load(std::memory_order_relaxed);
218 std::atomic_thread_fence(std::memory_order_acquire);
219 if (s.seq.load(std::memory_order_relaxed) == before)
220 return t;
221 }
222 return {};
223 }

226 static void reset() noexcept { publish(HostTransport{}); }
