31inline AutomationHeaderReadout automationHeaderReadout(const AutomationCurve* curve, double defaultValue,
32 double playheadBeat) {
33 if (!curve || curve->getPoints().empty()) return {defaultValue, false};
34 const auto& points = curve->getPoints();
35 size_t onlyIndex = 0;
36 int selected = 0;
37 for (size_t i = 0; i < points.size(); ++i) {
38 if (!points[i].selected) continue;
39 onlyIndex = i;
40 ++selected;
41 }
42 if (selected == 1) {
43 AutomationHeaderReadout readout{points[onlyIndex].value, true, {}};
44 if (points.size() > 1) {
45 const bool hasPrevious = onlyIndex > 0;
46 const AutomationPoint& neighbour = points[hasPrevious ? onlyIndex - 1 : onlyIndex + 1];
47 readout.delta = automationDeltaLabel(curve->getAutomationTarget(), readout.value, neighbour.value) +
48 (hasPrevious ? " vs prev" : " vs next");
49 }
50 return readout;
51 }
52 return {curve->getValueAtBeat(playheadBeat), false, {}};
53}

59struct SegmentSpan {

61 float t1{1.0f};
62 bool empty() const { return t1 <= t0; }
63};
64inline SegmentSpan visibleSegmentSpan(float x1, float x2, float left, float right) {
65 SegmentSpan span;
66 if (x2 > x1) {
67 span.t0 = std::clamp((left - x1) / (x2 - x1), 0.0f, 1.0f);
68 span.t1 = std::clamp((right - x1) / (x2 - x1), 0.0f, 1.0f);
69 } else if (x1 < left || x1 > right) {
70 span.t1 = span.t0; // a zero-length segment outside the lane
71 }
72 return span;
73}

135template <typename ScreenOf>
136AutomationPointHit hitAutomationPoint(const std::vector<AutomationCurve>& curves, int editedCurve, float x, float y,
137 float radius, ScreenOf screenOf) {
138 auto search = [&](int c) -> AutomationPointHit {
139 if (c < 0 || c >= static_cast<int>(curves.size()) || !curves[static_cast<size_t>(c)].isVisible()) return {};
140 const auto& points = curves[static_cast<size_t>(c)].getPoints();
141 for (int i = 0; i < static_cast<int>(points.size()); ++i) {
142 const auto [px, py] = screenOf(c, points[static_cast<size_t>(i)]);
143 if (std::hypot(px - x, py - y) < radius) return {c, i};
144 }
145 return {};
146 };
147 if (const auto hit = search(editedCurve); hit.found()) return hit;
148 for (int c = 0; c < static_cast<int>(curves.size()); ++c) {
149 if (c == editedCurve) continue;
150 if (const auto hit = search(c); hit.found()) return hit;
151 }
152 return {};
153}

160inline void selectAutomationPoint(std::vector<AutomationCurve>& curves, int curve, int point, bool extend) {
161 if (curve < 0 || curve >= static_cast<int>(curves.size())) return;
162 auto& points = curves[static_cast<size_t>(curve)].getPoints();
163 if (point < 0 || point >= static_cast<int>(points.size())) return;
164 const bool wasSelected = points[static_cast<size_t>(point)].selected;
165 if (!extend) {
166 for (auto& c : curves) {
167 for (auto& p : c.getPoints()) p.selected = false;
168 }
169 }
170 points[static_cast<size_t>(point)].selected = extend ? !wasSelected : true;
171}

193inline std::string automationValueLabel(AutomationTarget target, double value) {
194 char text[32];
195 if (target == AutomationTarget::Pan) {
196 const int percent = static_cast<int>(std::lround(std::clamp(value, -1.0, 1.0) * 100.0));
197 if (percent == 0) return "C";
198 std::snprintf(text, sizeof(text), "%c %d", percent < 0 ? 'L' : 'R', std::abs(percent));
199 return text;
200 }
201 if (target != AutomationTarget::Volume) {
202 std::snprintf(text, sizeof(text), "%.0f%%", std::clamp(value, 0.0, 1.0) * 100.0);
203 return text;
204 }
205 if (!(value > 1.0e-5)) return "-\xE2\x88\x9E dB"; // -∞ in UTF-8
206 const double db = 20.0 * std::log10(value);
207 std::snprintf(text, sizeof(text), "%.1f dB", std::abs(db) < 0.05 ? 0.0 : db);
208 return text;
209}

214inline std::string automationDeltaLabel(AutomationTarget target, double value, double neighbour) {
215 char text[32];
216 if (target == AutomationTarget::Pan) {
217 const int points = static_cast<int>(std::lround((std::clamp(value, -1.0, 1.0) - std::clamp(neighbour, -1.0, 1.0)) * 100.0));
218 std::snprintf(text, sizeof(text), "%+d", points);
219 return points == 0 ? "0" : text;
220 }
221 if (target != AutomationTarget::Volume) {
222 const double points = (std::clamp(value, 0.0, 1.0) - std::clamp(neighbour, 0.0, 1.0)) * 100.0;
223 std::snprintf(text, sizeof(text), "%+.0f%%", points);
224 return std::fabs(points) < 0.5 ? "0%" : text;
225 }
226 const bool silent = !(value > 1.0e-5);
227 const bool neighbourSilent = !(neighbour > 1.0e-5);
228 if (silent && neighbourSilent) return "0.0 dB";
229 if (neighbourSilent) return "+\xE2\x88\x9E dB";
230 if (silent) return "-\xE2\x88\x9E dB";
231 const double db = 20.0 * std::log10(value / neighbour);
232 if (std::fabs(db) < 0.05) return "0.0 dB";
233 std::snprintf(text, sizeof(text), "%+.1f dB", db);
234 return text;
235}
