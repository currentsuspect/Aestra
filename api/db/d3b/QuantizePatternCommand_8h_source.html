24class QuantizePatternCommand : public ICommand {
25public:
26 QuantizePatternCommand(PatternManager& patternManager, PatternID patternId, double gridBeats,
27 double strength, uint64_t unitId)
28 : m_patternManager(patternManager), m_patternId(patternId), m_gridBeats(gridBeats),
29 m_strength(strength), m_unitId(unitId) {}
30
31 void execute() override {
32 if (m_executed) return;
33 m_originalStarts.clear();
34
35 m_patternManager.applyPatch(m_patternId, [this](PatternSource& pattern) {
36 if (!pattern.isMidi()) return;
37 auto& notes = std::get<MidiPayload>(pattern.payload).notes;
38 for (size_t i = 0; i < notes.size(); ++i) {
39 MidiNote& note = notes[i];
40 if (m_unitId != 0 && note.unitId != m_unitId) continue;
41 const double snapped = std::round(note.startBeat / m_gridBeats) * m_gridBeats;
42 const double moved = note.startBeat + (snapped - note.startBeat) * m_strength;
43 if (moved != note.startBeat) {
44 m_originalStarts.emplace_back(i, note.startBeat);
45 note.startBeat = moved;
46 }
47 }
48 });
49
50 m_executed = true;
51 }
52
53 void undo() override {
54 if (!m_executed) return;
55
56 m_patternManager.applyPatch(m_patternId, [this](PatternSource& pattern) {
57 if (!pattern.isMidi()) return;
58 auto& notes = std::get<MidiPayload>(pattern.payload).notes;
59 for (const auto& [index, start] : m_originalStarts) {
60 if (index < notes.size()) {
61 notes[index].startBeat = start;
62 }
63 }
64 });
65
66 m_executed = false;
67 }
68
69 void redo() override {
70 if (m_executed) return;
71 execute();
72 }
73
74 std::string getName() const override { return "Quantize Pattern"; }
75 size_t getSizeInBytes() const override {
76 return sizeof(*this) + m_originalStarts.size() * sizeof(std::pair<size_t, double>);
77 }
78 bool changesProjectState() const override { return true; }
79
80private:
81 PatternManager& m_patternManager;
82 PatternID m_patternId;
83 double m_gridBeats;
84 double m_strength;
85 uint64_t m_unitId; // 0 = all units
86
87 // Recorded by execute() for undo: (note index, original startBeat)
88 std::vector<std::pair<size_t, double>> m_originalStarts;
89 bool m_executed = false;
90};
