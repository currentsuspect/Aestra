16void ArsenalPanel::showSceneMenu(uint64_t patternId, const AestraUI::NUIPoint& at) {
17 if (!m_trackManager) return;
18 ArsenalSceneMenu::Request request;
19 request.tracks = m_trackManager.get();
20 request.owner = this;
21 request.id = patternId;
22 request.activeId = m_activePatternID.value;
23 request.at = at;
24 request.chip = m_scenes.chipRectOf(patternId);
25 request.scenes = m_scenes.scenes();
26 request.changed = [this](uint64_t select) { sceneMenuChanged(select); };
27 m_sceneMenu.show(request);

30void ArsenalPanel::reconcilePatterns() {
31 if (!m_trackManager) return;
32 auto& patterns = m_trackManager->getPatternManager();
33 size_t signature = 0;
34 std::vector<ArsenalPatternPick::Candidate> midi;
35 patterns.forEachPattern([&](const PatternSource& p) {
36 signature += std::hash<std::string>{}(p.name) * 31u + static_cast<size_t>(p.id.value); // order-free: the store is unordered
37 if (p.isMidi()) midi.push_back({p.id.value, std::get<MidiPayload>(p.payload).notes.size()});
38 });
39 const uint64_t toShow = ArsenalPatternPick::patternToShow(midi);
40 if (!patterns.getPattern(m_activePatternID) && toShow != 0) {
41 // Nothing on show (a project just opened, undoing a duplicate, a delete elsewhere): show a pattern with notes
42 // rather than an empty one the arrangement never uses (#999).
43 m_activePatternID = PatternID{toShow};
44 m_trackManager->preparePatternForArsenal(m_activePatternID);
45 if (m_onActivePatternChanged) m_onActivePatternChanged(m_activePatternID);
46 }
47 if (signature != m_patternSignature) { // renamed, added or removed: Ctrl+Z reaches here without the browser knowing
48 m_patternSignature = signature;
49 if (m_patternBrowser) m_patternBrowser->refreshPatterns();
50 repaint();
51 }
52}

54void ArsenalPanel::sceneMenuChanged(uint64_t select) {
55 if (m_patternBrowser) m_patternBrowser->refreshPatterns();
56 // After Duplicate the copy becomes active (as in the browser); after deleting the active pattern its neighbour does.
57 if (select != 0) setActivePattern(PatternID{select});
58 refreshUnits();
59 repaint();
60}
