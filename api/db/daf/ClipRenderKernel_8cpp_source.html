14ClipRenderState makeClipRenderState(const ClipRuntimeInfo& clipInfo, double projectSampleRate) {
15 ClipRenderState clip;
16 // ClipRenderState.bufferOwner holds a shared_ptr<AudioBufferData>
17 // that extends the buffer's lifetime independently of SourceManager.
18 // audioData is extracted from the same buffer as a raw pointer for hot-path use.
19
20 if (clipInfo.isAudio()) {
21 clip.audioData = clipInfo.audioData->interleavedData.data();
22 clip.bufferOwner = clipInfo.sharedAudioData;
23 clip.totalFrames = clipInfo.audioData->numFrames;
24 clip.sourceSampleRate = static_cast<double>(clipInfo.sourceSampleRate);
25 clip.channels = clipInfo.sourceChannels;
26 }
27
28 clip.startSample = clipInfo.startTime;
29 clip.endSample = clipInfo.getEndTime();
30
31 // Convert sourceStart (Project Rate) to sampleOffset (Source Rate)
32 // Use double precision to prevent audio popping due to sub-sample drift
33 if (projectSampleRate > 0.0 && clip.sourceSampleRate > 0.0) {
34 clip.sampleOffset =
35 static_cast<double>(clipInfo.sourceStart) * (clip.sourceSampleRate / projectSampleRate);
36 } else {
37 clip.sampleOffset = static_cast<double>(clipInfo.sourceStart);
38 }
39
40 clip.gain = clipInfo.gainLinear;
41 clip.pan = clipInfo.pan;
42 clip.playbackRate = clipInfo.playbackRate;
43 clip.pitchSemitones = clipInfo.pitchSemitones;
44 clip.fadeInSamples = clipInfo.fadeInSamples;
45 clip.fadeOutSamples = clipInfo.fadeOutSamples;
46 return clip;
47}

49ClipRenderResult renderClipInto(const ClipRenderState& clip, uint64_t blockStart, uint64_t blockEnd,
50 double* destination, uint32_t engineSampleRate,
51 Interpolators::InterpolationQuality quality) {
52 // Verbatim relocation of the per-clip body of renderClips(). The only
53 // changes are mechanical: the loop-carried telemetry flag became this
54 // result, and each clip-level `continue` became a return of it. Transport
55 // and pattern-mode guards stay at the call site; this never sees them.
56 ClipRenderResult result{};
57
58 // Preconditions the call site used to guarantee.
59 //
60 // AudioEngine::renderGraph() returns early when the engine sample rate
61 // is zero, so this body never saw one. As a free function that the
62 // consolidation renderer also calls, it has to say so itself: with a
63 // zero output rate and a clip whose own rate is unset, ratio becomes
64 // 0/0 = NaN, the `phase >= totalFrames` guard is false for NaN, and
65 // the frame-count clamp then casts NaN to uint32_t — undefined.
66 //
67 // A zero-length source is the same class: both existing bounds checks
68 // are gated on totalFrames > 0, so it would fall through to the
69 // direct-rate branch and read past the buffer.
70 if (engineSampleRate == 0 || clip.totalFrames == 0) {
71 return result;
72 }
73
74 if (!clip.audioData || blockEnd <= clip.startSample || blockStart >= clip.endSample) {
75 return result;
76 }
77
78 const uint64_t start = std::max(blockStart, clip.startSample);
79 const uint64_t end = std::min(blockEnd, clip.endSample);
80 const uint32_t localOffset = static_cast<uint32_t>(start - blockStart);
81 uint32_t framesToRender = static_cast<uint32_t>(end - start);
82
83 // Sample rate ratio
84 const double outputRate = static_cast<double>(engineSampleRate);
85 const double srcRate = clip.sourceSampleRate > 0.0 ? clip.sourceSampleRate : outputRate;
86 const double playbackRate =
87 std::isfinite(clip.playbackRate) ? MixMath::clampD(static_cast<double>(clip.playbackRate), 0.25, 4.0) : 1.0;
88 // Pitch folds into the same varispeed envelope as Speed: 2^(st/12)
89 // multiplies the rate, and the combined factor stays bounded by the
90 // existing 0.25..4.0 clamp so phase advancement stays bounded (#746).
91 const double pitchSemitones = std::isfinite(clip.pitchSemitones)
92 ? MixMath::clampD(static_cast<double>(clip.pitchSemitones), -24.0, 24.0)
93 : 0.0;
94 const double varispeed =
95 MixMath::clampD(playbackRate * std::pow(2.0, pitchSemitones / 12.0), 0.25, 4.0);
96 const double ratio = (srcRate / outputRate) * varispeed;
97
98 // Source position
99 const double outputFrameOffset = static_cast<double>(start - clip.startSample);
100 double phase = clip.sampleOffset + outputFrameOffset * ratio;
101
102 // Bounds
103 const int64_t totalFrames = static_cast<int64_t>(clip.totalFrames);
104 const uint64_t totalFrameCount = static_cast<uint64_t>(totalFrames);
105 if (totalFrames > 0 && phase >= static_cast<double>(totalFrames)) {
106 return result;
107 }
108 if (totalFrames > 0) {
109 const double remaining = static_cast<double>(totalFrames) - phase;
110 const uint32_t maxFrames = static_cast<uint32_t>(remaining / ratio);
111 framesToRender = std::min(framesToRender, maxFrames);
112 }
113 if (framesToRender == 0)
114 return result;
115
116 const uint32_t channels = clip.channels;
117 const uint32_t stride = channels;
118
119 double* dstBase = destination;
120 const float* data = clip.audioData;
121 double* dst = dstBase + static_cast<size_t>(localOffset) * 2;
122
123 const uint64_t clipLength = clip.endSample - clip.startSample;
124 const uint64_t fadeInLen =
125 std::min(clipLength, std::max<uint64_t>(kClipEdgeFadeSamples, clip.fadeInSamples));
126 const uint64_t fadeOutLen =
127 std::min(clipLength, std::max<uint64_t>(kClipEdgeFadeSamples, clip.fadeOutSamples));
128 const auto clipFadeAt = [&clip, fadeInLen, fadeOutLen](uint64_t projectSample) {
129 double fade = 1.0;
130 if (fadeInLen > 0 && projectSample < clip.startSample + fadeInLen) {
131 fade = std::min(fade, static_cast<double>(projectSample - clip.startSample) /
132 static_cast<double>(fadeInLen));
133 }
134 if (fadeOutLen > 0 && projectSample + fadeOutLen > clip.endSample) {
135 fade = std::min(fade, static_cast<double>(clip.endSample - projectSample) /
136 static_cast<double>(fadeOutLen));
137 }
138 return fade;
139 };
140 // Clip pan is an instance-level stereo balance before the insert.
141 // Centre must remain unity so opening the editor cannot change old
142 // projects or stack a second -3 dB centre law with the insert pan.
143 const double clipPan = MixMath::clampD(static_cast<double>(clip.pan), -1.0, 1.0);
144 const double clipGain = static_cast<double>(clip.gain);
145 const double clipGainL = clipGain * (clipPan > 0.0 ? 1.0 - clipPan : 1.0);
146 const double clipGainR = clipGain * (clipPan < 0.0 ? 1.0 + clipPan : 1.0);
147
148 // Fast path: matching sample rates - direct copy to double
149 if (std::abs(ratio - 1.0) < 1e-9) {
150 const uint64_t srcStart = static_cast<uint64_t>(phase);
151 const float* src = data + srcStart * stride;
152 for (uint32_t i = 0; i < framesToRender; ++i) {
153 // Micro-fade at clip edges to avoid clicks/crackles.
154 const uint64_t projectSample = start + i;
155 const double fade = clipFadeAt(projectSample);
156
157 // Sanitize clip source reads: a corrupted audio file can
158 // contain NaN/Inf that would otherwise poison the mix.
159 double sL, sR;
160 if (channels == 1) {
161 sL = MixMath::sanitizeMix(static_cast<double>(src[i]));
162 sR = sL;
163 } else {
164 sL = MixMath::sanitizeMix(static_cast<double>(src[i * 2]));
165 sR = MixMath::sanitizeMix(static_cast<double>(src[i * 2 + 1]));
166 }
167
168 dst[i * 2] += sL * clipGainL * fade;
169 dst[i * 2 + 1] += sR * clipGainR * fade;
170 }
171 } else {
172 result.usedSampleRateConversion = true;
173 // Resampling - use selected quality, pre-compute end condition
174 const double phaseEnd = static_cast<double>(totalFrames);
175
176 if (channels == 1) {
177 // Mono Resampling — use the same quality interpolators as stereo.
178 // The mono path reads a single float per frame, so we duplicate to L/R.
179 switch (quality) {
180 case Interpolators::InterpolationQuality::Cubic:
181 for (uint32_t i = 0; i < framesToRender && phase < phaseEnd; ++i) {
182 const uint64_t projectSample = start + i;
183 const double fade = clipFadeAt(projectSample);
184 float sample = 0.0f;
185 uint64_t idx = static_cast<uint64_t>(phase);
186 double frac = phase - static_cast<double>(idx);
187 // Catmull-Rom 4-point on mono data
188 float s0 = (idx > 0) ? data[idx - 1] : data[idx];
189 float s1 = data[idx];
190 float s2 = (idx + 1 < totalFrameCount) ? data[idx + 1] : data[idx];
191 float s3 = (idx + 2 < totalFrameCount) ? data[idx + 2] : s2;
192 double f = frac;
193 sample = static_cast<float>(0.5 * ((2.0 * s1) + (-s0 + s2) * f +
194 (2.0 * s0 - 5.0 * s1 + 4.0 * s2 - s3) * f * f +
195 (-s0 + 3.0 * s1 - 3.0 * s2 + s3) * f * f * f));
196 const double s = MixMath::sanitizeMix(static_cast<double>(sample));
197 dst[i * 2] += s * clipGainL * fade;
198 dst[i * 2 + 1] += s * clipGainR * fade;
199 phase += ratio;
200 }
201 return result;
202 case Interpolators::InterpolationQuality::Sinc8:
203 case Interpolators::InterpolationQuality::Sinc16:
204 case Interpolators::InterpolationQuality::Sinc32:
205 case Interpolators::InterpolationQuality::Sinc64:
206 // Sinc on mono: compute weighted sum, duplicate to L/R
207 for (uint32_t i = 0; i < framesToRender && phase < phaseEnd; ++i) {
208 const uint64_t projectSample = start + i;
209 const double fade = clipFadeAt(projectSample);
210 double val =
211 Interpolators::sincInterpolateMono(data, totalFrames, phase, quality);
212 const double v = MixMath::sanitizeMix(static_cast<double>(val));
213 dst[i * 2] += v * clipGainL * fade;
214 dst[i * 2 + 1] += v * clipGainR * fade;
215 phase += ratio;
216 }
217 return result;
218 default:
219 // Linear fallback
220 for (uint32_t i = 0; i < framesToRender && phase < phaseEnd; ++i) {
221 const uint64_t projectSample = start + i;
222 const double fade = clipFadeAt(projectSample);
223 uint64_t idx = static_cast<uint64_t>(phase);
224 double frac = phase - static_cast<double>(idx);
225 float s0 = data[idx];
226 float s1 = (idx + 1 < static_cast<uint64_t>(totalFrames)) ? data[idx + 1] : s0;
227 double val = s0 + frac * (s1 - s0);
228 const double v = MixMath::sanitizeMix(static_cast<double>(val));
229 dst[i * 2] += v * clipGainL * fade;
230 dst[i * 2 + 1] += v * clipGainR * fade;
231 phase += ratio;
232 }
233 return result;
234 }
235 }
236
237 // Select interpolator at block level, not per-sample
238 // Block snapshot, matching mono above and every other cached
239 // atomic renderGraph() takes once per block (slot map, params,
240 // meter snapshots, pattern mode, and this same quality value at
241 // its line 2168 read). Loading it again here let stereo observe a
242 // different quality from mono inside one block, and potentially a
243 // different one per clip — incoherent with the snapshot the
244 // context already carries, and unreachable once this body becomes
245 // a free kernel that has no engine to load from.
246 switch (quality) {
247 case Interpolators::InterpolationQuality::Cubic:
248 for (uint32_t i = 0; i < framesToRender && phase < phaseEnd; ++i) {
249 float outL, outR;
250 Interpolators::CubicInterpolator::interpolate(data, totalFrames, phase, outL, outR);
251 const uint64_t projectSample = start + i;
252 const double fade = clipFadeAt(projectSample);
253 dst[i * 2] += MixMath::sanitizeMix(static_cast<double>(outL)) * clipGainL * fade;
254 dst[i * 2 + 1] += MixMath::sanitizeMix(static_cast<double>(outR)) * clipGainR * fade;
255 phase += ratio;
256 }
257 break;
258 case Interpolators::InterpolationQuality::Sinc8:
259 for (uint32_t i = 0; i < framesToRender && phase < phaseEnd; ++i) {
260 float outL, outR;
261 Interpolators::Sinc8Interpolator::interpolate(data, totalFrames, phase, outL, outR);
262 const uint64_t projectSample = start + i;
263 const double fade = clipFadeAt(projectSample);
264 dst[i * 2] += MixMath::sanitizeMix(static_cast<double>(outL)) * clipGainL * fade;
265 dst[i * 2 + 1] += MixMath::sanitizeMix(static_cast<double>(outR)) * clipGainR * fade;
266 phase += ratio;
267 }
268 break;
269 case Interpolators::InterpolationQuality::Sinc16:
270 for (uint32_t i = 0; i < framesToRender && phase < phaseEnd; ++i) {
271 float outL, outR;
272 Interpolators::Sinc16Interpolator::interpolate(data, totalFrames, phase, outL, outR);
273 const uint64_t projectSample = start + i;
274 const double fade = clipFadeAt(projectSample);
275 dst[i * 2] += MixMath::sanitizeMix(static_cast<double>(outL)) * clipGainL * fade;
276 dst[i * 2 + 1] += MixMath::sanitizeMix(static_cast<double>(outR)) * clipGainR * fade;
277 phase += ratio;
278 }
279 break;
280 case Interpolators::InterpolationQuality::Sinc32:
281 for (uint32_t i = 0; i < framesToRender && phase < phaseEnd; ++i) {
282 float outL, outR;
283 Interpolators::Sinc32Interpolator::interpolate(data, totalFrames, phase, outL, outR);
284 const uint64_t projectSample = start + i;
285 const double fade = clipFadeAt(projectSample);
286 dst[i * 2] += MixMath::sanitizeMix(static_cast<double>(outL)) * clipGainL * fade;
287 dst[i * 2 + 1] += MixMath::sanitizeMix(static_cast<double>(outR)) * clipGainR * fade;
288 phase += ratio;
289 }
290 break;
291 case Interpolators::InterpolationQuality::Sinc64:
292 for (uint32_t i = 0; i < framesToRender && phase < phaseEnd; ++i) {
293 float outL, outR;
294 Interpolators::Sinc64Interpolator::interpolate(data, totalFrames, phase, outL, outR);
295 const uint64_t projectSample = start + i;
296 const double fade = clipFadeAt(projectSample);
297 dst[i * 2] += MixMath::sanitizeMix(static_cast<double>(outL)) * clipGainL * fade;
298 dst[i * 2 + 1] += MixMath::sanitizeMix(static_cast<double>(outR)) * clipGainR * fade;
299 phase += ratio;
300 }
301 break;
302 }
303 }
304 return result;
305}
