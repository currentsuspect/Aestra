59 static std::vector<float> thumbnail(const std::vector<Audio::MidiNote>& notes, double lengthBeats) {
60 std::vector<float> bars(kThumbBars, 0.0f);
61 if (lengthBeats <= 0.0) return bars;
62 for (const auto& n : notes) {
63 const size_t b = std::min(kThumbBars - 1, static_cast<size_t>(std::max(0.0, n.startBeat) / lengthBeats * kThumbBars));
64 bars[b] += 1.0f;
65 }
66 const float most = *std::max_element(bars.begin(), bars.end());
67 if (most > 0.0f) for (auto& v : bars) if (v > 0.0f) v = std::max(0.2f, v / most);
68 return bars;
69 }

72 static std::vector<Placed> place(const std::vector<float>& widths, int activeIndex, float availW, float offset,
73 bool followActive, float& outOffset) {
74 std::vector<Placed> out;
75 float x = 0.0f;
76 for (size_t i = 0; i < widths.size(); ++i) {
77 out.push_back({i, x, widths[i]});
78 x += widths[i] + kGap;
79 }
80 const float total = std::max(0.0f, x - kGap);
81 const float viewW = std::max(0.0f, availW - kAddW - kGap);
82 const float maxOff = std::max(0.0f, total - viewW);
83 if (followActive && activeIndex >= 0 && activeIndex < static_cast<int>(out.size())) {
84 const auto& a = out[static_cast<size_t>(activeIndex)];
85 if (a.x < offset) offset = a.x;
86 if (a.x + a.w > offset + viewW) offset = a.x + a.w - viewW;
87 }
88 outOffset = std::clamp(offset, 0.0f, maxOff);
89 for (auto& p : out) p.x -= outOffset;
90 return out;
91 }

104 template <class TrackManagerT>
105 static std::vector<uint64_t> quietDefaults(const TrackManagerT& tm) {
106 std::vector<uint64_t> placed, quiet;
107 tm.getPlaylistModel().withLanes([&](const auto& lanes) {
108 for (const auto& lane : lanes)
109 for (const auto& c : lane.clips) placed.push_back(c.patternId.value);
110 return 0;
111 });
112 const auto& units = tm.getUnitManager();
113 for (const auto id : units.getAllUnitIDs()) {
114 const auto* u = units.getUnit(id);
115 if (!u || !u->defaultPatternId.isValid()) continue;
116 const auto* p = tm.getPatternManager().getPattern(u->defaultPatternId);
117 if (!p || !p->isMidi() || !std::get<Audio::MidiPayload>(p->payload).notes.empty()) continue;
118 if (std::find(placed.begin(), placed.end(), p->id.value) == placed.end()) quiet.push_back(p->id.value);
119 }
120 return quiet;
121 }

125 void refresh(const Audio::PatternManager& pm, uint64_t activeId, const AestraUI::NUIRect& rect, const std::vector<uint64_t>& hidden = {}) {
126 m_rect = rect;
127 m_scenes.clear();
128 for (const auto& p : pm.getAllPatterns()) {
129 if (p && p->isMidi() && p->id.value != activeId && std::find(hidden.begin(), hidden.end(), p->id.value) != hidden.end()) continue;
130 if (p && p->isMidi()) m_scenes.push_back({p->id.value, p->name.empty() ? "Pattern" : p->name, thumbnail(std::get<Audio::MidiPayload>(p->payload).notes, p->lengthBeats)});
131 }
132 std::sort(m_scenes.begin(), m_scenes.end(), [](const Scene& a, const Scene& b) { return a.id < b.id; });
133 std::vector<float> widths;
134 int active = -1;
135 for (size_t i = 0; i < m_scenes.size(); ++i) {
136 widths.push_back(chipWidth(m_scenes[i].name));
137 if (m_scenes[i].id == activeId) active = static_cast<int>(i);
138 }
139 m_followActive = m_followActive || activeId != m_lastActive;
140 m_lastActive = activeId;
141 m_placed = place(widths, active, rect.width - kRightW - kGap, m_offset, m_followActive, m_offset);
142 const auto* ap = pm.getPattern(Audio::PatternID{activeId});
143 m_swing = ap ? ap->swing : 0.0f;
144 m_followActive = false;
145 m_activeId = activeId;
146 }

152 const float limit = m_rect.right() - kRightW - kGap - kAddW;

169 Hit hitTest(const AestraUI::NUIPoint& pt) const {
170 if (!m_rect.contains(pt)) return {};
171 if (placeRect().contains(pt)) return {Hit::Kind::Place, 0};
172 if (swingRect().contains(pt)) return {Hit::Kind::Swing, 0};
173 if (addRect().contains(pt)) return {Hit::Kind::Add, 0};
174 if (pt.x >= addRect().x - kGap) return {};
175 for (const auto& p : m_placed) {
176 if (chipRect(p).contains(pt)) return {Hit::Kind::Scene, m_scenes[p.index].id};
177 }
178 return {};
179 }

182 static std::string fitName(AestraUI::NUIRenderer& renderer, std::string name, float font, float room) {
183 if (renderer.measureText(name, font).width <= room) return name;
184 while (name.size() > 1 && renderer.measureText(name + "\xE2\x80\xA6", font).width > room) name.pop_back();
185 return name + "\xE2\x80\xA6";

188 void draw(AestraUI::NUIRenderer& renderer) const {
189 if (m_rect.width <= 0.0f) return;
190 auto& theme = AestraUI::NUIThemeManager::getInstance();
191 const auto& tp = theme.getCurrentTheme();
192 const AestraUI::NUIRect view{m_rect.x, m_rect.y, std::max(0.0f, addRect().x - kGap - m_rect.x), m_rect.height};
193 renderer.setClipRect(view);
194 for (const auto& p : m_placed) {
195 const auto r = chipRect(p);
196 const auto& scene = m_scenes[p.index];
197 const bool on = scene.id == m_activeId;
198 renderer.fillRoundedRect(r, tp.radiusXS, on ? theme.getColor("surfaceRaised") : theme.getColor("backgroundSecondary"));
199 renderer.strokeRoundedRect(r, tp.radiusXS, 1.0f, on ? theme.getColor("textPrimary").withAlpha(0.9f) : theme.getColor("borderSubtle").withAlpha(0.7f));
200 renderer.setClipRect(AestraUI::intersectRects(view, r));
201 renderer.drawText(fitName(renderer, scene.name, tp.fontSizeXS + 1.0f, r.width - 24.0f), {r.x + 12.0f, r.y + 8.0f}, tp.fontSizeXS + 1.0f, theme.getColor("textPrimary").withAlpha(on ? 1.0f : 0.9f));
202 const float tw = r.width - 24.0f, bw = tw / static_cast<float>(kThumbBars), maxH = 8.0f;
203 for (size_t i = 0; i < scene.thumb.size(); ++i) {
204 const float h = scene.thumb[i] > 0.0f ? std::max(1.0f, maxH * scene.thumb[i]) : 1.0f;
205 renderer.fillRect({r.x + 12.0f + static_cast<float>(i) * bw, r.bottom() - 9.0f - h, std::max(1.0f, bw - 1.0f), h},
206 theme.getColor(on ? "textSecondary" : "textDisabled").withAlpha(scene.thumb[i] > 0.0f ? 0.9f : 0.35f));
207 }
208 renderer.setClipRect(view);
209 }
210 renderer.clearClipRect();
211 const auto add = addRect();
212 renderer.fillRoundedRect(add, tp.radiusXS, theme.getColor("backgroundPrimary"));
213 renderer.strokeRoundedRect(add, tp.radiusXS, 1.0f, theme.getColor("borderSubtle").withAlpha(0.7f));
214 renderer.drawTextCentered("+", add, tp.fontSizeM, theme.getColor("textSecondary").withAlpha(0.9f));
215 const auto sw = swingRect();
216 const float ty = sw.y + (sw.height - tp.fontSizeXS) * 0.5f - 2.0f;
217 renderer.drawText("Swing", {sw.x, ty}, tp.fontSizeXS + 1.0f, theme.getColor("textSecondary").withAlpha(0.9f));
218 const std::string pct = std::to_string(swingPercent(m_swing)) + "%";
219 renderer.drawText(pct, {sw.x + renderer.measureText("Swing ", tp.fontSizeXS + 1.0f).width, ty}, tp.fontSizeXS + 1.0f, theme.getColor("textPrimary"));
220 renderer.fillRect({sw.x, sw.bottom() - 10.0f, sw.width, 1.0f}, theme.getColor("borderSubtle").withAlpha(0.7f));
221 renderer.fillRect({sw.x, sw.bottom() - 10.0f, sw.width * m_swing, 1.0f}, theme.getColor("textSecondary"));
222 const auto pl = placeRect(); // the one primary action of the header (the mock's purple button)
223 renderer.fillRoundedRect(pl, tp.radiusXS, theme.getColor("accentPrimary"));
224 renderer.drawTextCentered("Place in timeline", pl, tp.fontSizeXS + 1.0f, AestraUI::NUIColor(1.0f, 1.0f, 1.0f, 1.0f));
225 }

229 bool onMouse(const AestraUI::NUIMouseEvent& e, Audio::PatternManager& pm, uint64_t activeId, Action& out) {
230 const bool left = e.button == AestraUI::NUIMouseButton::Left;
231 if (m_swingDrag) {
232 setSwing(pm, activeId, e.position.x);
233 out = {Action::Kind::SwingEdited, activeId};
234 if (!e.pressed && left) m_swingDrag = false;
235 return true;
236 }
237 if (!m_rect.contains(e.position)) return false;
238 if (std::abs(e.wheelDelta) > 0.001f) { scrollBy(-e.wheelDelta * 40.0f); return true; }
239 if (e.pressed && e.button == AestraUI::NUIMouseButton::Right) { // a chip's menu (V82-AR1, FD-52)
240 if (const Hit h = hitTest(e.position); h.kind == Hit::Kind::Scene) out = {Action::Kind::Menu, h.id, e.position};
241 return true;
242 }
243 if (!e.pressed || !left) return true;
244 const Hit hit = hitTest(e.position);
245 if (hit.kind == Hit::Kind::Scene) out = {Action::Kind::Select, hit.id};
246 if (hit.kind == Hit::Kind::Add) out = {Action::Kind::PatternAdded, pm.createMidiPattern(nextName(m_scenes), 8.0, {}).value};
247 if (hit.kind == Hit::Kind::Place) out = {Action::Kind::PlaceInTimeline, activeId};
248 if (hit.kind == Hit::Kind::Swing) { m_swingDrag = true; setSwing(pm, activeId, e.position.x); out = {Action::Kind::SwingEdited, activeId}; }
249 return true;
250 }

256 void setSwing(Audio::PatternManager& pm, uint64_t id, float x) {
257 const auto r = swingRect();
258 m_swing = std::clamp((x - r.x) / r.width, 0.0f, 1.0f);
259 m_swing = std::round(m_swing * 100.0f) / 100.0f;
260 pm.applyPatch(Audio::PatternID{id}, [this](Audio::PatternSource& p) { p.swing = m_swing; });
261 }
262 bool m_swingDrag{false};
263 float m_swing{0.0f};
264 uint64_t m_activeId{0};
265 uint64_t m_lastActive{0};
266 float m_offset{0.0f};
267 bool m_followActive{true};
