14struct BodySink {

16 size_t cap;
17 bool overflowed = false;

20size_t writeToString(char* data, size_t size, size_t nmemb, void* userData) {
21 auto* sink = static_cast<BodySink*>(userData);
22 const size_t length = size * nmemb;
23 if (!appendWithinCap(*sink->body, data, length, sink->cap)) {
24 // Returning short makes libcurl abort the transfer (CURLE_WRITE_ERROR).
25 sink->overflowed = true;
26 return 0;
27 }
28 return length;
29}

31size_t onHeader(char* data, size_t size, size_t nmemb, void* userData) {
32 auto* seconds = static_cast<int*>(userData);
33 const size_t length = size * nmemb;
34 static const char kName[] = "retry-after:";
35 constexpr size_t kNameLen = sizeof(kName) - 1;
36 if (length > kNameLen) {
37 bool match = true;
38 for (size_t i = 0; i < kNameLen && match; ++i) {
39 match = std::tolower(static_cast<unsigned char>(data[i])) == kName[i];
40 }
41 if (match) {
42 size_t i = kNameLen;
43 while (i < length && (data[i] == ' ' || data[i] == '\t')) ++i;
44 long value = 0;
45 size_t digits = 0;
46 while (i < length && data[i] >= '0' && data[i] <= '9' && digits < 9) {
47 value = value * 10 + (data[i] - '0');
48 ++i;
49 ++digits;
50 }
51 // Delta-seconds only; the HTTP-date form is ignored (the schedule's backoff applies).
52 while (i < length && (data[i] == ' ' || data[i] == '\r' || data[i] == '\n')) ++i;
53 if (digits > 0 && i == length) *seconds = static_cast<int>(value);
54 }
55 }
56 return length;
57}

66HttpResponse httpPostJson(const std::string& url,
67 const std::vector<std::pair<std::string, std::string>>& headers,
68 const std::string& jsonBody, long timeoutSeconds,
69 size_t maxResponseBytes) {
70 ensureCurlInit();
71 HttpResponse response;
72
73 CURL* curl = curl_easy_init();
74 if (!curl) {
75 response.error = "curl_easy_init failed";
76 return response;
77 }
78
79 // A failed append would silently drop a header, and the request would go
80 // out without its auth, so treat it as a failure of the whole call.
81 curl_slist* headerList = nullptr;
82 bool headersOk = true;
83 const auto addHeader = [&](const std::string& line) {
84 curl_slist* grown = curl_slist_append(headerList, line.c_str());
85 if (grown == nullptr) {
86 headersOk = false;
87 return;
88 }
89 headerList = grown;
90 };
91 addHeader("content-type: application/json");
92 for (const auto& [key, value] : headers) {
93 addHeader(key + ": " + value);
94 }
95 if (!headersOk) {
96 response.error = "out of memory building request headers";
97 curl_slist_free_all(headerList);
98 curl_easy_cleanup(curl);
99 return response;
100 }
101
102 curl_easy_setopt(curl, CURLOPT_URL, url.c_str());
103 curl_easy_setopt(curl, CURLOPT_HTTPHEADER, headerList);
104 curl_easy_setopt(curl, CURLOPT_POSTFIELDS, jsonBody.c_str());
105 curl_easy_setopt(curl, CURLOPT_POSTFIELDSIZE, static_cast<long>(jsonBody.size()));
106 curl_easy_setopt(curl, CURLOPT_HEADERFUNCTION, onHeader);
107 curl_easy_setopt(curl, CURLOPT_HEADERDATA, &response.retryAfterSeconds);
108 curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, writeToString);
109 BodySink sink{&response.body, maxResponseBytes};
110 curl_easy_setopt(curl, CURLOPT_WRITEDATA, &sink);
111 // Second line of defence: refuse up front when the server declares a size over the cap.
112 curl_easy_setopt(curl, CURLOPT_MAXFILESIZE_LARGE, static_cast<curl_off_t>(maxResponseBytes));
113 curl_easy_setopt(curl, CURLOPT_TIMEOUT, timeoutSeconds);
114 curl_easy_setopt(curl, CURLOPT_FOLLOWLOCATION, 0L);
115
116 const CURLcode result = curl_easy_perform(curl);
117 if (sink.overflowed || result == CURLE_FILESIZE_EXCEEDED) {
118 response.bodyTooLarge = true;
119 response.error = "response exceeded " + std::to_string(maxResponseBytes) + " bytes";
120 } else if (result != CURLE_OK) {
121 response.error = curl_easy_strerror(result);
122 } else {
123 curl_easy_getinfo(curl, CURLINFO_RESPONSE_CODE, &response.status);
124 }
125
126 curl_slist_free_all(headerList);
127 curl_easy_cleanup(curl);
128 return response;
129}
