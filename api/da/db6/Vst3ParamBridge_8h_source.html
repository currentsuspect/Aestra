4// V82-H4/H5: how parameter values travel between the host, a VST3 controller (the half with the editor) and its
5// processor (the half that makes sound). VST3 keeps them apart on purpose: setting a value on the controller does not
6// reach the processor, so a host that only calls IEditController::setParamNormalized() hears nothing change. The values
7// have to be handed to process() as IParameterChanges, and anything the plugin's own window changes arrives through
8// IComponentHandler::performEdit and has to take the same road.
9//
10// * ParamBridge values waiting to go to the processor (any thread pushes; the audio thread drains, never blocks)
11// * HostParameterChanges the pending values as the IParameterChanges handed to process(), fixed capacity, no allocation
12// * OutputParameterChanges what the processor reports back (meters, its own automation), read by the host after process()
13// * HostComponentHandler the controller's side: edits made in the plugin window become pending values

29constexpr size_t kMaxParamChangesPerBlock = 128;
30
32struct ParamChange {
33 uint32_t id = 0;
34 double value = 0.0;
35};
36
37class ParamBridge {
38public:
41 void push(uint32_t id, double value) {
42 std::lock_guard<std::mutex> lock(m_mutex);
43 for (auto& pending : m_pending) {
44 if (pending.id == id) {
45 pending.value = value;
46 return;
47 }
48 }
49 m_pending.push_back({id, value});
50 }
51
54 size_t drain(ParamChange* out, size_t max) {
55 std::unique_lock<std::mutex> lock(m_mutex, std::try_to_lock);
56 if (!lock.owns_lock() || m_pending.empty()) return 0;
57 const size_t n = std::min(max, m_pending.size());
58 for (size_t i = 0; i < n; ++i) out[i] = m_pending[i];
59 m_pending.erase(m_pending.begin(), m_pending.begin() + static_cast<std::ptrdiff_t>(n));
60 return n;
61 }
62
63 size_t pendingCount() const {
64 std::lock_guard<std::mutex> lock(m_mutex);
65 return m_pending.size();
66 }
67
68private:
69 mutable std::mutex m_mutex;
70 std::vector<ParamChange> m_pending;
71};

109 void fill(const ParamChange* values, size_t n) {
110 m_count = static_cast<Steinberg::int32>(std::min(n, m_queues.size()));

142 Steinberg::Vst::IParamValueQueue* PLUGIN_API getParameterData(Steinberg::int32 index) override {
143 return (index >= 0 && index < m_count) ? &m_queues[static_cast<size_t>(index)] : nullptr;
144 }
145 Steinberg::Vst::IParamValueQueue* PLUGIN_API addParameterData(const Steinberg::Vst::ParamID& id,
146 Steinberg::int32& index) override {
147 for (Steinberg::int32 i = 0; i < m_count; ++i) {
148 if (m_queues[static_cast<size_t>(i)].id == id) {
149 index = i;
150 return &m_queues[static_cast<size_t>(i)];
151 }
152 }
153 if (m_count >= static_cast<Steinberg::int32>(m_queues.size())) return nullptr;
154 index = m_count++;
155 m_queues[static_cast<size_t>(index)].id = id;
156 return &m_queues[static_cast<size_t>(index)];
157 }
158 Steinberg::tresult PLUGIN_API queryInterface(const Steinberg::TUID, void** obj) override {
159 if (obj) *obj = nullptr;
160 return Steinberg::kNoInterface;
161 }
162 Steinberg::uint32 PLUGIN_API addRef() override { return 1; }
163 Steinberg::uint32 PLUGIN_API release() override { return 1; }
164
165 void clear() { m_count = 0; }
167 size_t collect(ParamChange* out, size_t max) const {
168 const size_t n = std::min(max, static_cast<size_t>(m_count));
169 for (size_t i = 0; i < n; ++i) out[i] = {m_queues[i].id, m_queues[i].value};
170 return n;
171 }
172
173private:
174 std::array<detail::SinglePointQueue, kMaxParamChangesPerBlock> m_queues;
175 Steinberg::int32 m_count = 0;
176};
