109void drawArc(NUIRenderer& renderer, NUIPoint center, float radius, float startAngle, float endAngle, float thickness,
110 NUIColor color) {
111 if (endAngle <= startAngle + 0.001f) {
112 return;
113 }
114 std::array<NUIPoint, 49> points{};
115 for (size_t i = 0; i < points.size(); ++i) {
116 const float t = static_cast<float>(i) / static_cast<float>(points.size() - 1);
117 const float angle = startAngle + (endAngle - startAngle) * t;
118 points[i] = {center.x + std::cos(angle) * radius, center.y + std::sin(angle) * radius};
119 }
120 renderer.drawPolyline(points.data(), static_cast<int>(points.size()), thickness, color);
121}

147void RumblePluginEditor::buildControls() {
148 m_controls.clear();
149 if (!m_instance) {
150 return;
151 }
152
153 struct Meta {
154 uint32_t parameterId;
155 const char* title;
156 const char* subtitle;
157 Zone zone;
158 };
159 const Meta meta[] = {
160 {9, "PUNCH", "Chest hit", Zone::Impact},
161 {4, "SWEEP", "Pitch strike", Zone::Impact},
162 {0, "DECAY", "Tail length", Zone::Impact},
163 {1, "DRIVE", "Weight + grit", Zone::Body},
164 {23, "HARMONICS", "Speaker translation", Zone::Body},
165 {24, "CLEAN SUB", "Fundamental hold", Zone::Body},
166 {2, "TONE", "Top-end focus", Zone::Body},
167 {3, "OUTPUT", "Final level", Zone::Body},
168 };
169
170 const auto parameters = m_instance->getParameters();
171 // The macro layout, preset application, and preset matching all assume the
172 // full 25-parameter contract. Silently dropping a missing ID would shift
173 // the index-based zones while presets still write every ID — reject the
174 // incompatible instance outright instead of rendering a lying panel.
175 for (const auto& item : meta) {
176 if (item.parameterId >= parameters.size()) {
177 Aestra::Log::error("[RumbleEditor] Instance exposes " + std::to_string(parameters.size()) +
178 " parameters; id " + std::to_string(item.parameterId) +
179 " missing — refusing to build controls");
180 m_controls.clear();
181 return;
182 }
183 }
184 m_controls.reserve(std::size(meta));
185 for (const auto& item : meta) {
186 MacroControl control;
187 control.parameterId = item.parameterId;
188 control.title = item.title;
189 control.subtitle = item.subtitle;
190 control.defaultValue = parameters[item.parameterId].defaultValue;
191 control.normalizedValue = m_instance->getParameter(item.parameterId);
192 control.zone = item.zone;
193 m_controls.push_back(std::move(control));
194 }
195 layoutControls();
196}

198void RumblePluginEditor::layoutControls() {
199 const auto bounds = getBounds();
200 const float contentTop = bounds.y + AestraPanelWindow::TITLE_BAR_H + kPadding;
201 const float contentWidth = std::max(0.0f, bounds.width - kPadding * 2.0f);
202 m_headerBounds = NUIRect(bounds.x + kPadding, contentTop, contentWidth, kHeaderHeight);
203
204 const float presetWidth = std::clamp(contentWidth * 0.43f, 310.0f, 370.0f);
205 m_previousPresetBounds = NUIRect(m_headerBounds.right() - presetWidth, m_headerBounds.y + 14.0f, 34.0f, 48.0f);
206 m_nextPresetBounds = NUIRect(m_headerBounds.right() - 34.0f, m_headerBounds.y + 14.0f, 34.0f, 48.0f);

210 const float mainY = m_headerBounds.bottom() + 12.0f;
211 const float mainHeight = std::max(0.0f, bounds.bottom() - kPadding - mainY);
212 const float gap = 12.0f;
213 const float impactWidth = std::round((contentWidth - gap) * 0.53f);
214 m_impactPanelBounds = NUIRect(bounds.x + kPadding, mainY, impactWidth, mainHeight);
215 m_bodyPanelBounds = NUIRect(m_impactPanelBounds.right() + gap, mainY, contentWidth - impactWidth - gap, mainHeight);
216 m_impactShapeBounds = NUIRect(m_impactPanelBounds.x + 16.0f, m_impactPanelBounds.y + 42.0f,

220 const float impactKnobWidth = (m_impactPanelBounds.width - 44.0f) / 3.0f;
221 for (size_t i = 0; i < std::min<size_t>(3, m_controls.size()); ++i) {
222 auto& control = m_controls[i];
223 control.bounds = NUIRect(m_impactPanelBounds.x + 12.0f + static_cast<float>(i) * (impactKnobWidth + 10.0f),

225 const float knobSize = std::min(88.0f, control.bounds.width - 20.0f);
226 control.knobBounds =
227 NUIRect(control.bounds.center().x - knobSize * 0.5f, control.bounds.y + 16.0f, knobSize, knobSize);
228 }
229
230 const float bodyTop = m_bodyPanelBounds.y + 45.0f;
231 const float bodyGap = 8.0f;
232 const float bodyCellWidth = (m_bodyPanelBounds.width - 32.0f - bodyGap * 2.0f) / 3.0f;
233 const float bodyCellHeight = (m_bodyPanelBounds.height - 59.0f - bodyGap) * 0.5f;
234 for (size_t offset = 0; offset < 5 && offset + 3 < m_controls.size(); ++offset) {
235 auto& control = m_controls[offset + 3];
236 const size_t row = offset < 3 ? 0 : 1;
237 const size_t column = offset < 3 ? offset : offset - 3;
238 const float rowX = row == 0 ? m_bodyPanelBounds.x + 12.0f
239 : m_bodyPanelBounds.center().x - (bodyCellWidth * 2.0f + bodyGap) * 0.5f;
240 control.bounds =
241 NUIRect(rowX + static_cast<float>(column) * (bodyCellWidth + bodyGap),
242 bodyTop + static_cast<float>(row) * (bodyCellHeight + bodyGap), bodyCellWidth, bodyCellHeight);
243 const float knobSize = std::min(68.0f, std::min(control.bounds.width - 18.0f, control.bounds.height - 54.0f));
244 control.knobBounds =
245 NUIRect(control.bounds.center().x - knobSize * 0.5f, control.bounds.y + 12.0f, knobSize, knobSize);
246 }
247
248 m_presetMenuBounds = NUIRect(m_impactPanelBounds.x, m_impactPanelBounds.y,
249 m_bodyPanelBounds.right() - m_impactPanelBounds.x, m_impactPanelBounds.height);
250 const float menuPad = 18.0f;
251 const float columnGap = 10.0f;
252 const float columnWidth = (m_presetMenuBounds.width - menuPad * 2.0f - columnGap * 3.0f) / 4.0f;
253 const float firstItemY = m_presetMenuBounds.y + 56.0f;
254 for (size_t i = 0; i < m_presetItemBounds.size(); ++i) {
255 const size_t column = i / 4;
256 const size_t row = i % 4;
257 m_presetItemBounds[i] =
258 NUIRect(m_presetMenuBounds.x + menuPad + static_cast<float>(column) * (columnWidth + columnGap),
259 firstItemY + static_cast<float>(row) * 48.0f, columnWidth, 40.0f);
260 }
261}

357void RumblePluginEditor::drawImpactShape(NUIRenderer& renderer) {
358 auto& theme = NUIThemeManager::getInstance();
359 renderer.fillRoundedRect(m_impactShapeBounds, 11.0f, insetSurface());
360 renderer.strokeRoundedRect(m_impactShapeBounds, 11.0f, 1.0f, impactAccent().withAlpha(0.16f));
361 for (int i = 1; i < 4; ++i) {
362 const float x = m_impactShapeBounds.x + m_impactShapeBounds.width * static_cast<float>(i) / 4.0f;
363 renderer.drawLine({x, m_impactShapeBounds.y + 12.0f}, {x, m_impactShapeBounds.bottom() - 12.0f}, 1.0f,
364 editorInk(0.035f));
365 }
366 const float punch = m_controls.size() > 0 ? m_controls[0].normalizedValue : 0.0f;
367 const float sweep = m_controls.size() > 1 ? m_controls[1].normalizedValue : 0.0f;
368 const float decay = m_controls.size() > 2 ? m_controls[2].normalizedValue : 0.0f;
369 std::array<NUIPoint, 97> points{};
370 float phase = 0.0f;
371 for (size_t i = 0; i < points.size(); ++i) {
372 const float t = static_cast<float>(i) / static_cast<float>(points.size() - 1);
373 const float frequency = 2.0f + 8.0f * sweep * std::exp(-t * 7.0f);
374 phase += frequency * 0.13f;
375 const float envelope = std::exp(-t * (2.0f + 7.0f * (1.0f - decay)));
376 const float attack = 0.34f + punch * 0.62f * std::exp(-t * 13.0f);
377 points[i] = {m_impactShapeBounds.x + 10.0f + t * (m_impactShapeBounds.width - 20.0f),
378 m_impactShapeBounds.center().y -
379 std::sin(phase) * envelope * attack * (m_impactShapeBounds.height * 0.40f)};
380 }
381 renderer.drawPolyline(points.data(), static_cast<int>(points.size()), 2.2f, impactAccent().withAlpha(0.92f));
382 renderer.drawText("TRANSIENT SHAPE", {m_impactShapeBounds.x + 11.0f, m_impactShapeBounds.y + 9.0f}, 7.5f,
383 theme.getColor("textSecondary").withAlpha(0.60f));
384}

386void RumblePluginEditor::drawKnob(NUIRenderer& renderer, const MacroControl& control, int controlIndex, bool primary) {
387 auto& theme = NUIThemeManager::getInstance();
388 const bool hovered = m_hoveredControl == controlIndex;
389 const bool dragging = m_draggingControl == controlIndex;
390 const NUIColor accent = controlAccent(control.parameterId);
391 renderer.fillRoundedRect(control.bounds, 11.0f,
392 hovered || dragging ? editorNeutral(NUIColor(0.068f, 0.064f, 0.080f, 0.98f))
393 : editorNeutral(NUIColor(0.036f, 0.034f, 0.043f, 0.82f)));
394 renderer.strokeRoundedRect(control.bounds, 11.0f, 1.0f,
395 hovered || dragging ? accent.withAlpha(0.50f) : editorInk(0.045f));
396
397 const NUIPoint center = control.knobBounds.center();
398 const float radius = control.knobBounds.width * 0.5f;
399 const float valueAngle = kKnobStart + control.normalizedValue * kKnobSweep;
400 renderer.fillCircle(center, radius, insetSurface());
401 renderer.strokeCircle(center, radius, 1.0f, editorInk(0.07f));
402 drawArc(renderer, center, radius - 6.0f, kKnobStart, kKnobStart + kKnobSweep, primary ? 4.0f : 3.2f,
403 editorInk(0.09f));
404 drawArc(renderer, center, radius - 6.0f, kKnobStart, valueAngle, primary ? 4.0f : 3.2f, accent.withAlpha(0.94f));
405 const float needleLength = radius - (primary ? 16.0f : 13.0f);
406 const NUIPoint tip{center.x + std::cos(valueAngle) * needleLength, center.y + std::sin(valueAngle) * needleLength};
407 renderer.drawLine(center, tip, 2.0f, accent.withAlpha(0.86f));
408 renderer.fillCircle(center, primary ? 7.0f : 5.5f, editorNeutral(NUIColor(0.075f, 0.071f, 0.088f, 1.0f)));
409 renderer.fillCircle(center, primary ? 2.8f : 2.2f, accent.withAlpha(0.90f));
410
411 const float labelY = control.knobBounds.bottom() + (primary ? 8.0f : 5.0f);
412 renderer.drawTextCentered(control.title, NUIRect(control.bounds.x, labelY, control.bounds.width, 14.0f),
413 primary ? 10.5f : 9.0f, theme.getColor("textPrimary").withAlpha(0.94f));
414 renderer.drawTextCentered(m_instance ? m_instance->getParameterDisplay(control.parameterId) : "",
415 NUIRect(control.bounds.x, labelY + 17.0f, control.bounds.width, 13.0f),
416 primary ? 10.0f : 8.5f, accent.withAlpha(0.96f));
417 if (primary && control.bounds.bottom() - labelY > 47.0f) {
418 renderer.drawTextCentered(control.subtitle,
419 NUIRect(control.bounds.x, labelY + 35.0f, control.bounds.width, 12.0f), 8.0f,
420 theme.getColor("textSecondary").withAlpha(0.62f));
421 }

424void RumblePluginEditor::drawPresetBrowser(NUIRenderer& renderer) {
425 auto& theme = NUIThemeManager::getInstance();
426 renderer.fillRoundedRect(m_presetMenuBounds, 16.0f, editorNeutral(NUIColor(0.018f, 0.017f, 0.023f, 1.0f)));
427 renderer.strokeRoundedRect(m_presetMenuBounds, 16.0f, 1.2f, bodyAccent().withAlpha(0.58f));
428 renderer.drawText("FACTORY SOUNDS", {m_presetMenuBounds.x + 18.0f, m_presetMenuBounds.y + 17.0f}, 11.0f,
429 theme.getColor("textPrimary"));
430 renderer.drawText("16 starting points / 4 characters",
431 {m_presetMenuBounds.x + 132.0f, m_presetMenuBounds.y + 19.0f}, 8.5f,
432 theme.getColor("textSecondary").withAlpha(0.70f));
433
434 for (size_t column = 0; column < 4; ++column) {
435 const size_t firstIndex = column * 4;
436 const auto& firstPreset = Aestra::Plugins::kRumbleFactoryPresets[firstIndex];
437 renderer.drawText(std::string(firstPreset.category),
438 {m_presetItemBounds[column * 4].x + 3.0f, m_presetMenuBounds.y + 41.0f}, 8.0f,
439 presetAccent(firstIndex).withAlpha(0.88f));
440 }
441 for (size_t i = 0; i < Aestra::Plugins::kRumbleFactoryPresets.size(); ++i) {
442 const bool active = i == m_activePreset;
443 const bool hovered = static_cast<int>(i) == m_hoveredPreset;
444 const NUIColor accent = presetAccent(i);

514int RumblePluginEditor::matchingFactoryPreset() const {
515 if (!m_instance) {
516 return -1;
517 }
518 for (size_t presetIndex = 0; presetIndex < Aestra::Plugins::kRumbleFactoryPresets.size(); ++presetIndex) {
519 const auto& values = Aestra::Plugins::kRumbleFactoryPresets[presetIndex].values;
520 bool matches = true;
521 for (uint32_t parameterId = 0; parameterId < values.size(); ++parameterId) {
522 if (std::abs(m_instance->getParameter(parameterId) - values[parameterId]) > 1.0e-5f) {
523 matches = false;
524 break;
525 }
526 }
527 if (matches) {
528 return static_cast<int>(presetIndex);
529 }
530 }
531 return -1;

534bool RumblePluginEditor::onMouseEvent(const NUIMouseEvent& event) {
535 if (!isVisible()) {
536 return false;
537 }
538 if (AestraPanelWindow::onMouseEvent(event)) {
539 return true;
540 }
541 if (!m_instance) {
542 return false;
543 }
544
545 // Handle an active knob drag before the bounds check: a release outside
546 // the panel must still end the drag, or the knob stays latched to the
547 // pointer when it re-enters.
548 if (m_draggingControl >= 0) {
549 if (event.released) {
550 m_draggingControl = -1;
551 setDirty(true);
552 return true;
553 }
554 if (event.button == NUIMouseButton::None || event.type == NUIMouseEventType::Drag) {
555 const float delta = (m_dragStartY - event.position.y) / kDragRangePixels;
556 setControlValue(m_draggingControl, m_dragStartValue + delta);
557 return true;
558 }
559 }
560
561 const bool contains = getBounds().contains(event.position);
562 if (!contains && !isDraggingWindow()) {
563 return false;
564 }
565
566 if (event.pressed && event.button == NUIMouseButton::Left) {
567 if (m_previousPresetBounds.contains(event.position)) {
568 const size_t count = Aestra::Plugins::kRumbleFactoryPresets.size();
569 applyFactoryPreset((m_activePreset + count - 1) % count);
570 return true;
571 }
572 if (m_nextPresetBounds.contains(event.position)) {
573 applyFactoryPreset((m_activePreset + 1) % Aestra::Plugins::kRumbleFactoryPresets.size());
574 return true;
575 }
576 if (m_presetButtonBounds.contains(event.position)) {
577 m_presetMenuOpen = !m_presetMenuOpen;
578 m_hoveredPreset = -1;
579 setDirty(true);
580 return true;
581 }
582 if (m_presetMenuOpen) {
583 for (size_t i = 0; i < m_presetItemBounds.size(); ++i) {
584 if (m_presetItemBounds[i].contains(event.position)) {
585 applyFactoryPreset(i);
586 m_presetMenuOpen = false;
587 return true;
588 }
589 }
590 if (!m_presetMenuBounds.contains(event.position)) {
591 m_presetMenuOpen = false;
592 setDirty(true);
593 }
594 return true;
595 }
596
597 const int controlIndex = hitTestControl(event.position);
598 if (controlIndex >= 0) {
599 // The platform never populates event.doubleClick — pair up quick
600 // same-spot presses manually so reset-to-default actually fires.
601 const long long nowMs = std::chrono::duration_cast<std::chrono::milliseconds>(
602 std::chrono::steady_clock::now().time_since_epoch())
603 .count();
604 const bool isDoubleClick =
605 event.doubleClick ||
606 ((nowMs - m_lastClickTimeMs) < 400 &&
607 std::abs(event.position.x - m_lastClickPos.x) < 5.0f &&
608 std::abs(event.position.y - m_lastClickPos.y) < 5.0f);
609 m_lastClickTimeMs = isDoubleClick ? 0 : nowMs;
610 m_lastClickPos = event.position;
611 if (isDoubleClick) {
612 setControlValue(controlIndex, m_controls[static_cast<size_t>(controlIndex)].defaultValue);
613 return true;
614 }
615 m_draggingControl = controlIndex;
616 m_dragStartY = event.position.y;
617 m_dragStartValue = m_controls[static_cast<size_t>(controlIndex)].normalizedValue;
618 return true;
619 }
620 }
621
622 if (!event.pressed && !event.released) {
623 m_presetButtonHovered = m_presetButtonBounds.contains(event.position);
624 m_previousPresetHovered = m_previousPresetBounds.contains(event.position);
625 m_nextPresetHovered = m_nextPresetBounds.contains(event.position);
626 m_hoveredControl = m_presetMenuOpen ? -1 : hitTestControl(event.position);
627 m_hoveredPreset = -1;
628 if (m_presetMenuOpen) {
629 for (size_t i = 0; i < m_presetItemBounds.size(); ++i) {
630 if (m_presetItemBounds[i].contains(event.position)) {
631 m_hoveredPreset = static_cast<int>(i);
632 break;
633 }
634 }
635 }
636 setDirty(true);
637 }
638
639 if (event.type == NUIMouseEventType::Scroll && m_hoveredControl >= 0 && !m_presetMenuOpen) {
640 const float direction = event.wheelDelta > 0.0f ? 1.0f : -1.0f;
641 setControlValue(m_hoveredControl,
642 m_controls[static_cast<size_t>(m_hoveredControl)].normalizedValue + direction * 0.02f);
643 return true;
644 }
645 return contains;
646}
