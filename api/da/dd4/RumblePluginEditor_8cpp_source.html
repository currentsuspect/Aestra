24void RumblePluginEditor::buildControls() {
25 m_controls.clear();
26 if (!m_instance) {
27 return;
28 }
29
30 struct Meta {
31 const char* title;
32 const char* subtitle;
33 };
34
35 const Meta meta[] = {
36 {"Decay", "Tail length"},
37 {"Drive", "Body + grit"},
38 {"Tone", "Low-end focus"},
39 {"Gain", "Final level"},
40 };
41
42 auto params = m_instance->getParameters();
43 const size_t count = std::min<size_t>(params.size(), 4);
44 m_controls.reserve(count);
45 for (size_t i = 0; i < count; ++i) {
46 MacroControl control;
47 control.parameterId = params[i].id;
48 control.title = meta[i].title;
49 control.subtitle = meta[i].subtitle;
50 control.normalizedValue = m_instance->getParameter(params[i].id);
51 m_controls.push_back(std::move(control));
52 }
53
54 layoutControls();
55}

57void RumblePluginEditor::layoutControls() {
58 const auto bounds = getBounds();
59 const float contentX = bounds.x + kPadding;
60 const float contentY = bounds.y + AestraPanelWindow::TITLE_BAR_H + kPadding;
61 const float contentW = bounds.width - kPadding * 2.0f;
62 const float cardsY = contentY + kHeroHeight + 12.0f;
63 const float gap = 10.0f;
64 const float cardW = (contentW - gap * 3.0f) / 4.0f;
65 const float cardH = bounds.height - (cardsY - bounds.y) - kPadding;
66
67 for (size_t i = 0; i < m_controls.size(); ++i) {
68 auto& control = m_controls[i];
69 const float x = contentX + static_cast<float>(i) * (cardW + gap);
70 control.cardBounds = NUIRect(x, cardsY, cardW, cardH);
71 control.trackBounds = NUIRect(x + 16.0f, cardsY + 82.0f, cardW - 32.0f, 8.0f);
72
73 const float thumbX = control.trackBounds.x + control.trackBounds.width * control.normalizedValue - 7.0f;
74 control.thumbBounds = NUIRect(thumbX, control.trackBounds.y - 5.0f, 14.0f, 18.0f);
75 }
76}

95void RumblePluginEditor::drawControl(NUIRenderer& renderer, const MacroControl& control, bool hovered) {
96 auto& theme = NUIThemeManager::getInstance();
97 const NUIColor accent = theme.getColor("accentPrimary");
98 const NUIColor cardColor = hovered || control.isDragging ? NUIColor(0.147f, 0.147f, 0.147f, 0.98f)
99 : NUIColor(0.114f, 0.114f, 0.114f, 0.96f);
100
101 renderer.fillRoundedRect(control.cardBounds, kCardRadius, cardColor);
102 renderer.strokeRoundedRect(control.cardBounds, kCardRadius, 1.0f,
103 hovered || control.isDragging ? accent.withAlpha(0.60f)
104 : NUIColor(1.0f, 1.0f, 1.0f, 0.07f));
105
106 renderer.drawText(control.title, {control.cardBounds.x + 14.0f, control.cardBounds.y + 18.0f}, 12.0f,
107 theme.getColor("textPrimary"));
108 renderer.drawText(control.subtitle, {control.cardBounds.x + 14.0f, control.cardBounds.y + 38.0f}, 10.0f,
109 theme.getColor("textSecondary"));
110 renderer.drawText(m_instance ? m_instance->getParameterDisplay(control.parameterId) : "0",
111 {control.cardBounds.x + 14.0f, control.cardBounds.y + 58.0f}, 10.0f,
112 accent.withAlpha(0.95f));
113
114 renderer.fillRoundedRect(control.trackBounds, 4.0f, NUIColor(0.021f, 0.021f, 0.021f, 0.78f));
115 const float filledWidth = std::max(0.0f, std::min(control.trackBounds.width, control.trackBounds.width * control.normalizedValue));
116 if (filledWidth > 0.0f) {
117 renderer.fillRoundedRect({control.trackBounds.x, control.trackBounds.y, filledWidth, control.trackBounds.height},
118 4.0f, accent.withAlpha(0.90f));
119 }
120
121 renderer.fillRoundedRect(control.thumbBounds, 7.0f,
122 hovered || control.isDragging ? NUIColor(1.0f, 1.0f, 1.0f, 1.0f)
123 : NUIColor(0.86f, 0.86f, 0.90f, 0.95f));
124
125 const float meterBottom = control.cardBounds.bottom() - 18.0f;
126 const float meterTop = meterBottom - 92.0f;
127 const float levelHeight = (meterBottom - meterTop) * control.normalizedValue;
128 renderer.fillRoundedRect({control.cardBounds.x + 18.0f, meterBottom - levelHeight, control.cardBounds.width - 36.0f, levelHeight},
129 10.0f, accent.withAlpha(0.20f));
130}

141int RumblePluginEditor::hitTestControl(float x, float y) const {
142 for (size_t i = 0; i < m_controls.size(); ++i) {
143 if (m_controls[i].cardBounds.contains({x, y})) {

150void RumblePluginEditor::updateControlValue(int controlIndex, float normalizedValue) {
151 if (!m_instance || controlIndex < 0 || controlIndex >= static_cast<int>(m_controls.size())) {
152 return;
153 }
154
155 auto& control = m_controls[static_cast<size_t>(controlIndex)];
156 control.normalizedValue = std::clamp(normalizedValue, 0.0f, 1.0f);
157 m_instance->setParameter(control.parameterId, control.normalizedValue);
158 const float thumbX = control.trackBounds.x + control.trackBounds.width * control.normalizedValue - 7.0f;
159 control.thumbBounds.x = thumbX;
160 setDirty(true);
161}

163bool RumblePluginEditor::onMouseEvent(const NUIMouseEvent& event) {
164 if (!isVisible()) {
165 return false;
166 }
167
168 // Let base class handle title bar / close / drag first
169 if (AestraPanelWindow::onMouseEvent(event)) {
170 return true;
171 }
172
173 const auto bounds = getBounds();
174 const bool contains = bounds.contains(event.position);
175
176 if (!contains && !isDraggingWindow()) {
177 return false;
178 }
179
180 if (event.pressed && event.button == NUIMouseButton::Left) {
181 m_hoveredControl = hitTestControl(event.position.x, event.position.y);
182 if (m_hoveredControl >= 0) {
183 m_controls[static_cast<size_t>(m_hoveredControl)].isDragging = true;
184 const auto& track = m_controls[static_cast<size_t>(m_hoveredControl)].trackBounds;
185 updateControlValue(m_hoveredControl, (event.position.x - track.x) / std::max(1.0f, track.width));
186 return true;
187 }
188 }
189
190 if (!event.pressed && !event.released && m_hoveredControl >= 0 && event.button == NUIMouseButton::Left) {
191 auto& control = m_controls[static_cast<size_t>(m_hoveredControl)];
192 if (control.isDragging) {
193 updateControlValue(m_hoveredControl,
194 (event.position.x - control.trackBounds.x) / std::max(1.0f, control.trackBounds.width));
195 return true;
196 }
197 }
198
199 if (!event.pressed && event.button == NUIMouseButton::Left) {
200 for (auto& control : m_controls) {
201 control.isDragging = false;
202 }
203 }
204
205 if (!event.pressed && !event.released) {
206 const int hovered = contains ? hitTestControl(event.position.x, event.position.y) : -1;
207 if (hovered != m_hoveredControl) {
208 m_hoveredControl = hovered;
209 setDirty(true);
210 }
211 }
212
213 return contains;
214}
