43bool PluginManager::initialize() {
44 std::lock_guard<std::mutex> lock(m_mutex);
45
46 if (m_initialized) {
47 return true;
48 }
49
50#ifdef _WIN32
51 // Initialize COM for VST3 plugin hosting
52 // Use COINIT_APARTMENTTHREADED for UI compatibility
53 HRESULT hr = CoInitializeEx(nullptr, COINIT_APARTMENTTHREADED);
54 if (FAILED(hr) && hr != S_FALSE && hr != RPC_E_CHANGED_MODE) {
55 // S_FALSE means already initialized, RPC_E_CHANGED_MODE means different mode
56 // Both are acceptable
57 return false;
58 }
59 m_comInitialized = (hr == S_OK || hr == S_FALSE);
60#endif
61
62 BuiltInPlugins::registerCoreBuiltIns();
63
64 // Add default search paths
65 m_scanner.addDefaultSearchPaths();
66
67 // Built-ins stay in-process. Third-party native plugin formats are routed
68 // through the helper process so a plugin crash does not take down the DAW.
69 m_factory = std::make_unique<HybridPluginFactory>();
70
71 // Load plugin cache when platform utilities are available.
72 // In headless/internal-plugin test flows the platform layer may not be initialized,
73 // and that should not be treated as an error.
74 const bool platformInitialized = Platform::isInitialized();
75 if (platformInitialized) {
76 if (auto* utils = Platform::getUtils()) {
77 std::string appData = utils->getAppDataPath("AESTRA");
78 if (!appData.empty()) {
79 std::filesystem::path cachePath = std::filesystem::path(appData) / "plugin_cache.bin";
80
81 // Ensure directory exists
82 std::error_code ec;
83 std::filesystem::create_directories(cachePath.parent_path(), ec);
84
85 if (m_scanner.loadScanCache(cachePath)) {
86 Aestra::Log::info("Plugin cache loaded: " + std::to_string(m_scanner.getScannedPlugins().size()) +
87 " plugins");
88 } else {
89 Aestra::Log::warning("Plugin cache not found or invalid. Triggering initial scan...");
90 m_scanner.scanAsync();
91 }
92 } else {
93 Aestra::Log::warning("AppData path is empty in PluginManager::initialize; skipping plugin cache");
94 m_scanner.scanAsync();
95 }
96 } else {
97 Aestra::Log::warning("Platform utilities unavailable in PluginManager::initialize; skipping plugin cache");
98 m_scanner.scanAsync();
99 }
100 }
101
102 m_initialized = true;
103 return true;
104}

106void PluginManager::shutdown() {
107 std::lock_guard<std::mutex> lock(m_mutex);
108
109 if (!m_initialized) {
110 return;
111 }
112
113 // Cancel any ongoing scan and join the scanner thread.
114 // This ensures the thread is stopped before static singleton destruction,
115 // which prevents the shutdown hang that forced the quick_exit workaround.
116 m_scanner.cancelAndJoin();
117
118 // Save plugin cache only when platform utilities are available.
119 const bool platformInitialized = Platform::isInitialized();
120 if (platformInitialized) {
121 if (auto* utils = Platform::getUtils()) {
122 std::string appData = utils->getAppDataPath("AESTRA");
123 if (!appData.empty()) {
124 std::filesystem::path cachePath = std::filesystem::path(appData) / "plugin_cache.bin";
125
126 // Ensure directory exists (just in case)
127 std::error_code ec;
128 std::filesystem::create_directories(cachePath.parent_path(), ec);
129
130 if (m_scanner.saveScanCache(cachePath)) {
131 Aestra::Log::info("Plugin cache saved to: " + cachePath.string());
132 } else {
133 Aestra::Log::error("Failed to save plugin cache to: " + cachePath.string());
134 }
135 } else {
136 Aestra::Log::warning("AppData path is empty in PluginManager::shutdown; skipping plugin cache save");
137 }
138 } else {
139 Aestra::Log::warning("Platform utilities unavailable in PluginManager::shutdown; skipping plugin cache save");
140 }
141 }
142
143 // Clear all active instances
144 m_activeInstances.clear();
145
146#ifdef _WIN32
147 // Uninitialize COM
148 if (m_comInitialized) {
149 CoUninitialize();
150 m_comInitialized = false;
151 }
152#endif
153
154 m_initialized = false;
155}

161PluginInstancePtr PluginManager::createInstance(const PluginInfo& info) {
162 if (!m_initialized || !m_factory) {
163 return nullptr;
164 }
165
166 // Adapt Async Factory to Synchronous API (Blocking)
167 // This allows gradual migration while using the new abstraction
168 std::promise<PluginInstancePtr> promise;
169 auto future = promise.get_future();
170
171 m_factory->createPluginAsync(info, [&](PluginInstancePtr instance) { promise.set_value(instance); });
172
173 // Block until complete (In-Process factory is synchronous anyway, so this is safe)
174 // For true async factories, this would hang the UI, which is why we must migrate consumers.
175 PluginInstancePtr instance = future.get();
176
177 if (instance) {
178 // Track the instance
179 std::lock_guard<std::mutex> lock(m_mutex);
180 cleanupExpiredInstances();
181 m_activeInstances.push_back(instance);
182 }
183
184 return instance;
185}

187PluginInstancePtr PluginManager::createInstanceById(const std::string& pluginId) {
188 const PluginInfo* info = findPlugin(pluginId);
189 if (!info) {
190 // Internal plugins are registered with the InternalPluginRegistry, which
191 // is the authority for their availability. A project may reference one
192 // registered after the last scan (tests, dynamically added built-ins),
193 // so fall back to the registry directly — still gated by availability.
194 auto& registry = InternalPluginRegistry::instance();
195 if (registry.isRegisteredPlugin(pluginId) && registry.isPluginAvailable(pluginId)) {
196 return registry.createInstance(pluginId);
197 }
198 return nullptr;
199 }
200 return createInstance(*info);
201}

203void PluginManager::createInstanceAsync(const PluginInfo& info, std::function<void(PluginInstancePtr)> callback) {
204 if (!m_initialized || !m_factory) {
205 if (callback)
206 callback(nullptr);
207 return;
208 }
209
210 m_factory->createPluginAsync(info, [this, callback](PluginInstancePtr instance) {
211 if (instance) {
212 std::lock_guard<std::mutex> lock(m_mutex);
213 cleanupExpiredInstances();
214 m_activeInstances.push_back(instance);
215 }
216 if (callback)
217 callback(instance);
218 });
219}

243std::vector<PluginInstancePtr> PluginManager::getActiveInstances() const {
244 std::lock_guard<std::mutex> lock(m_mutex);
245
246 std::vector<PluginInstancePtr> result;
247 for (const auto& weak : m_activeInstances) {
248 if (auto ptr = weak.lock()) {
249 result.push_back(ptr);
250 }
251 }
252 return result;
253}

255size_t PluginManager::getActiveInstanceCount() const {
256 std::lock_guard<std::mutex> lock(m_mutex);
257
258 size_t count = 0;
259 for (const auto& weak : m_activeInstances) {
260 if (!weak.expired()) {
261 ++count;
262 }
263 }
264 return count;
