20inline FirstFreeMixerChannelResult selectOrCreateFirstFreeMixerChannel(TrackManager& manager, UnitID unitId,
21 const std::string& destinationName,
22 uint32_t color) {
23 std::unordered_set<uint32_t> usedChannelIds;
24 for (const UnitID otherUnitId : manager.getUnitManager().getAllUnitIDs()) {
25 if (otherUnitId == unitId)
26 continue;
27 const uint32_t channelId = manager.getUnitManager().getUnitMixerChannel(otherUnitId);
28 if (channelId != MASTER_MIXER_CHANNEL_ID) {
29 usedChannelIds.insert(channelId);
30 }
31 }
32 for (const auto& pattern : manager.getPatternManager().getAllPatterns()) {
33 if (!pattern || !pattern->isAudio())
34 continue;
35 const uint32_t channelId = pattern->getMixerChannelId();
36 if (channelId != MASTER_MIXER_CHANNEL_ID) {
37 usedChannelIds.insert(channelId);
38 }
39 }
40
41 for (size_t i = 0; i < manager.getChannelCount(); ++i) {
42 const auto* channel = manager.getChannel(i);
43 if (channel && usedChannelIds.find(channel->getChannelId()) == usedChannelIds.end()) {
44 return {channel->getChannelId(), false};
45 }
46 }
47
48 auto* channel = manager.addChannel(destinationName);
49 if (!channel)
50 return {};
51 channel->setColor(color);
52 return {channel->getChannelId(), true};
53}

63inline bool routeUnitToFirstFreeMixerChannel(TrackManager& manager, UnitID unitId, const std::string& destinationName,
64 uint32_t color, bool preserveModifiedState = false) {
65 if (!manager.getUnitManager().getUnit(unitId))
66 return false;
67
68 const bool wasModified = manager.isModified();
69 const auto result = detail::selectOrCreateFirstFreeMixerChannel(manager, unitId, destinationName, color);
70 if (result.channelId == MASTER_MIXER_CHANNEL_ID) {
71 if (preserveModifiedState)
72 manager.setModified(wasModified);
73 return false;
74 }
75
76 manager.getUnitManager().setUnitMixerChannel(unitId, result.channelId);
77 if (preserveModifiedState) {
78 manager.setModified(wasModified);
79 } else {
80 manager.markModified();
81 }
82 return true;
83}

86class AssignUnitToFirstFreeMixerChannelCommand final : public ICommand {
87public:
88 AssignUnitToFirstFreeMixerChannelCommand(TrackManager& manager, UnitID unitId, std::string destinationName,
89 uint32_t color)
90 : m_manager(manager), m_unitId(unitId), m_destinationName(std::move(destinationName)), m_color(color) {}
91
92 void execute() override {
93 if (m_executed || !m_manager.getUnitManager().getUnit(m_unitId))
94 return;
95
96 if (!m_capturedPreviousRoute) {
97 m_previousChannelId = m_manager.getUnitManager().getUnitMixerChannel(m_unitId);
98 m_capturedPreviousRoute = true;
99 }
100
101 if (m_createdDestination) {
102 if (!m_manager.reinsertChannel(m_detachedChannel, m_channelIndex))
103 return;
104 } else if (m_destinationChannelId == MASTER_MIXER_CHANNEL_ID) {
105 const size_t channelCountBefore = m_manager.getChannelCount();
106 const auto result =
107 detail::selectOrCreateFirstFreeMixerChannel(m_manager, m_unitId, m_destinationName, m_color);
108 m_destinationChannelId = result.channelId;
109 m_createdDestination = result.created;
110 if (m_destinationChannelId == MASTER_MIXER_CHANNEL_ID)
111 return;
112 if (m_createdDestination) {
113 if (m_manager.getChannelCount() <= channelCountBefore)
114 return;
115 m_channelIndex = m_manager.getChannelCount() - 1;
116 }
117 }
118
119 m_manager.getUnitManager().setUnitMixerChannel(m_unitId, m_destinationChannelId);
120 m_manager.markModified();
121 m_executed = true;
122 }
123
124 void undo() override {
125 if (!m_executed)
126 return;
127
128 m_manager.getUnitManager().setUnitMixerChannel(m_unitId, m_previousChannelId);
129 if (m_createdDestination) {
130 m_detachedChannel = m_manager.detachChannelById(m_destinationChannelId, m_channelIndex);
131 if (!m_detachedChannel) {
132 m_manager.getUnitManager().setUnitMixerChannel(m_unitId, m_destinationChannelId);
133 return;
134 }
135 }
136 m_manager.markModified();
137 m_executed = false;
138 }
139
140 void redo() override { execute(); }
141
142 std::string getName() const override { return "Assign Unit to First Free Mixer Channel"; }
143 size_t getSizeInBytes() const override { return sizeof(*this); }
144 bool changesProjectState() const override { return true; }
145 bool isUndoable() const override { return m_executed; }
146
147private:
148 TrackManager& m_manager;
149 UnitID m_unitId;
150 std::string m_destinationName;
151 uint32_t m_color;
152 uint32_t m_previousChannelId{MASTER_MIXER_CHANNEL_ID};
153 uint32_t m_destinationChannelId{MASTER_MIXER_CHANNEL_ID};
154 std::unique_ptr<MixerChannel> m_detachedChannel;
155 size_t m_channelIndex{0};
156 bool m_capturedPreviousRoute{false};
157 bool m_createdDestination{false};
158 bool m_executed{false};
159};
