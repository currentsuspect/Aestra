38 // edited to make the move.

45 // and then dereferences it, so it has to carry std::optional's own semantics.

212 const auto addUnresolved = [&](const char* issueCode, const char* routeType,
213 const std::string& sourceNodeId, uint32_t targetId,
214 uint64_t sendId = 0) {
215 JSON evidence = JSON::object();
216 evidence.set("sourceNodeId", JSON(sourceNodeId));
217 evidence.set("targetMixerChannelId", JSON(static_cast<double>(targetId)));
218 evidence.set("stableSourceIdentityAvailable", JSON(true));
219 evidence.set("stableSendIdAvailable", JSON(sendId != 0));
220 if (sendId != 0) evidence.set("sendId", JSON(std::to_string(sendId)));
221
222 JSON issue = JSON::object();
223 issue.set("issueCode", JSON(issueCode));
224 issue.set("routeType", JSON(routeType));
225 issue.set("evidence", evidence);
226 unresolvedRoutes.push(issue);
227 };

229 JSON master = JSON::object();
230 master.set("nodeId", JSON("master"));
231 master.set("destinationType", JSON("master"));
232 master.set("mixerChannelId", JSON(0.0));
233 master.set("stableIdentityAvailable", JSON(true));
234 if (auto* masterChannel = m_trackManager->getMasterChannel()) {
235 master.set("insertChainAvailable", JSON(true));
236 const auto& masterChain = masterChannel->getEffectChain();
237 JSON pluginSlots = JSON::array();
238 for (size_t slotIndex = 0; slotIndex < masterChain.slotCount(); ++slotIndex) {
239 JSON position = JSON::object();
240 position.set("mixerChannelId", JSON(0.0));
241 position.set("slotIndex", JSON(static_cast<double>(slotIndex)));
242
243 JSON slot = JSON::object();
244 slot.set("slotIndex", JSON(static_cast<double>(slotIndex)));
245 slot.set("stableIdentityAvailable", JSON(false));
246 slot.set("positionalIdentityAvailable", JSON(true));
247 slot.set("position", position);
248
249 if (auto plugin = masterChain.getPlugin(slotIndex)) {
250 slot.set("state", JSON("active"));
251 slot.set("pluginId", JSON(plugin->getInfo().id));
252 slot.set("pluginName", JSON(plugin->getInfo().name));
253 slot.set("bypassed", JSON(masterChain.isSlotBypassed(slotIndex)));
254 } else {
255 const std::string missingPluginId = masterChain.getMissingPluginId(slotIndex);
256 if (!missingPluginId.empty()) {
257 slot.set("state", JSON("missing_plugin_placeholder"));
258 slot.set("pluginId", JSON(missingPluginId));
259 slot.set("placeholderPreserved", JSON(true));
260 } else {
261 slot.set("state", JSON("empty"));
262 }
263 }
264 pluginSlots.push(slot);
265 }
266
267 JSON insertChain = JSON::object();
268 insertChain.set("slotCount", JSON(static_cast<double>(masterChain.slotCount())));
269 insertChain.set("identityKind", JSON("positional"));
270 insertChain.set("stableSlotIdentityAvailable", JSON(false));
271 insertChain.set("slots", pluginSlots);
272 master.set("insertChain", insertChain);
273 } else {
274 master.set("insertChainAvailable", JSON(false));
275 }
276 destinations.push(master);

294 slot.set("positionalIdentityAvailable", JSON(true));

297 if (auto plugin = chain.getPlugin(slotIndex)) {

300 slot.set("pluginName", JSON(plugin->getInfo().name));
301 slot.set("bypassed", JSON(chain.isSlotBypassed(slotIndex)));
302 } else {
303 const std::string missingPluginId = chain.getMissingPluginId(slotIndex);
304 if (!missingPluginId.empty()) {
305 slot.set("state", JSON("missing_plugin_placeholder"));
306 slot.set("pluginId", JSON(missingPluginId));
307 slot.set("placeholderPreserved", JSON(true));
308 } else {
309 slot.set("state", JSON("empty"));
310 }

312 pluginSlots.push(slot);

316 insertChain.set("slotCount", JSON(static_cast<double>(chain.slotCount())));
317 insertChain.set("identityKind", JSON("positional"));
318 insertChain.set("stableSlotIdentityAvailable", JSON(false));

321 JSON destination = JSON::object();
322 destination.set("nodeId", JSON(sourceNodeId));
323 destination.set("destinationType", JSON("mixer_channel"));

645 uint32_t dstNodeIdx{0};
646 bool sidechain{false};
647 size_t trackIndex{0};
648 size_t sendIndex{0}; // positional index into the RT send-edge-delay snapshot

650 bool main{false};
651 };
652 std::vector<CurrentEdge> currentEdges;
653 for (size_t trackIndex = 0; trackIndex < channels.size(); ++trackIndex) {
654 const auto* channel = channels[trackIndex];
655 if (!channel)
656 continue;
657 const auto src = topologyNodeByChannelId.find(channel->getChannelId());
658 if (src == topologyNodeByChannelId.end())
659 continue;
660
661 const auto main = topologyNodeByChannelId.find(channel->getMainOutputId());
662 if (main != topologyNodeByChannelId.end() && main->second != src->second) {
663 currentEdges.push_back({static_cast<uint32_t>(src->second), static_cast<uint32_t>(main->second),
664 false, trackIndex, 0, 0, true});
665 }
666 const auto sends = channel->getSends();
667 for (size_t sendIndex = 0; sendIndex < sends.size(); ++sendIndex) {
668 const auto& send = sends[sendIndex];
669 if (send.mute)
670 continue;
671 const auto destination = topologyNodeByChannelId.find(send.targetChannelId);
672 if (destination == topologyNodeByChannelId.end() || destination->second == src->second) {
673 continue;
674 }
675 currentEdges.push_back({static_cast<uint32_t>(src->second),
676 static_cast<uint32_t>(destination->second), send.sidechainOnly, trackIndex,
677 sendIndex, send.sendId, false});
678 }

681 // Match solved edges to current routing by identity, not position.
682 // Pairing solved edge `i` with currentEdges[i] was only correct
683 // while the model was unchanged: inserting or removing one route
684 // shifted everything after it, so a single routing edit reported
685 // pdc_edge_mapping_mismatch on every subsequent edge instead of
686 // the one that actually moved. pdc_edge_count_mismatch already
687 // reports that the shapes differ.
688 //
689 // (src, dst, sidechain) is not unique — a channel may hold two
690 // sends to the same target — so equal keys are consumed in
691 // discovery order, which keeps the pairing deterministic and
692 // still localizes a mismatch to the edge that changed.
693 // Mixed-radix rather than bit-packed: both indices are bounded by
694 // nodes.size(), so this is exact and collision-free without
695 // assuming either fits in a fixed bit width.
696 const uint64_t nodeCount = static_cast<uint64_t>(topology.nodes.size());
697 const auto edgeKey = [nodeCount](uint32_t src, uint32_t dst, bool sidechain) -> uint64_t {
698 return ((static_cast<uint64_t>(src) * nodeCount) + static_cast<uint64_t>(dst)) * 2ull +
699 (sidechain ? 1ull : 0ull);
700 };
701 std::unordered_map<uint64_t, std::vector<size_t>> currentEdgesByKey;
702 for (size_t j = 0; j < currentEdges.size(); ++j) {
703 currentEdgesByKey[edgeKey(currentEdges[j].srcNodeIdx, currentEdges[j].dstNodeIdx,
704 currentEdges[j].sidechain)]
705 .push_back(j);
706 }
707 std::unordered_map<uint64_t, size_t> currentEdgeCursor;
708 constexpr size_t kNoCurrentEdge = static_cast<size_t>(-1);
709
710 for (size_t i = 0; i < topology.edges.size(); ++i) {
711 const auto& solution = topology.edges[i];
712 const bool sourceValid = solution.srcNodeIdx < topology.nodes.size();
713 const bool destinationValid = solution.dstNodeIdx < topology.nodes.size();
714 const std::string sourceNodeId = sourceValid ? nodeId(topology.nodes[solution.srcNodeIdx].channelId)
715 : "unknown:" + std::to_string(solution.srcNodeIdx);
716 const std::string destinationNodeId = destinationValid
717 ? nodeId(topology.nodes[solution.dstNodeIdx].channelId)
718 : "unknown:" + std::to_string(solution.dstNodeIdx);
719
720 size_t currentEdgeIndex = kNoCurrentEdge;
721 if (sourceValid && destinationValid) {
722 const uint64_t key = edgeKey(solution.srcNodeIdx, solution.dstNodeIdx, solution.sidechain);
723 const auto candidates = currentEdgesByKey.find(key);
724 if (candidates != currentEdgesByKey.end()) {
725 auto& cursor = currentEdgeCursor[key];
726 if (cursor < candidates->second.size()) {
727 currentEdgeIndex = candidates->second[cursor++];
728 }
729 }
730 }
731 const bool mappingMatches = currentEdgeIndex != kNoCurrentEdge;
732 bool main = false;
733 size_t sendIndex = 0;
734 uint64_t sendId = 0;
735 bool appliedAvailable = false;
736 uint32_t appliedCompensation = 0;
737 bool mismatch = !mappingMatches;
738
739 if (mappingMatches) {
740 const auto& current = currentEdges[currentEdgeIndex];
741 main = current.main;
742 sendIndex = current.sendIndex;
743 sendId = current.sendId;
744 const auto snapshot = m_engine->getTrackEdgeDelaySnapshot(current.trackIndex);
745 if (snapshot.valid && main) {
746 appliedCompensation = snapshot.mainOutEdgeDelay.compensationSamples;
747 appliedAvailable = true;
748 } else if (snapshot.valid && sendIndex < snapshot.sendEdgeDelays.size()) {
749 appliedCompensation = snapshot.sendEdgeDelays[sendIndex].compensationSamples;
750 appliedAvailable = true;
751 }
752 mismatch = !appliedAvailable || appliedCompensation != solution.compensationSamples;

757 evidence.set("edgeIndex", JSON(static_cast<double>(i)));

759 evidence.set("destinationNodeId", JSON(destinationNodeId));
760 evidence.set("stableEndpointIdentityAvailable", JSON(sourceValid && destinationValid));
761 evidence.set("stableSendIdAvailable", JSON(mappingMatches && !main && sendId != 0));
762 if (mappingMatches && !main && sendId != 0) {
763 evidence.set("sendId", JSON(std::to_string(sendId)));

765 evidence.set("solvedCompensationSamples", JSON(static_cast<double>(solution.compensationSamples)));
766 evidence.set("appliedCompensationAvailable", JSON(appliedAvailable));

769 mappingMatches ? "the RT-side edge delay does not match the solved topology"
770 : "the solved edge cannot be mapped to the current routing model",
