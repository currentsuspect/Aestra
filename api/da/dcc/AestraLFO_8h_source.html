1// © 2026 Aestra Studios — All Rights Reserved. Licensed for personal & educational use only.
2// AestraLFO V1 — tempo-syncable LFO that shapes the audio flowing through it
3// (LFOTool-style insert effect): rhythmic volume gating/ducking, auto-pan, or
4// filter-cutoff wobble.
5//
6// V1 deliberately modulates its own audio path instead of other plugins'
7// parameters: engine-side parameter modulation routing does not exist yet
8// (issue #467); when it lands, this LFO core can be reused as a source.
9//
10// The LFO free-runs from activate() — it locks to tempo *rate* via setBPM()
11// (same push pattern as AestraDelay) but not to transport *phase*; bar-aligned
12// restart needs a host transport hook that IPluginInstance does not have yet.
13//
14// Signal path (per sample):
15// lfo phase -> waveform -> smoothing (slew) -> target:
16// Volume: gain = 1 - depth * (1 - u) (u = unipolar LFO, 1 = open)
17// Pan: equal-power L/R, unity at center
18// Cutoff: ZDF SVF low-pass swept down up to 6 octaves from 20 kHz
19

103 void process(const float* const* inputs, float** outputs, uint32_t numInputChannels, uint32_t numOutputChannels,
104 uint32_t numFrames, const MidiBuffer* midiInput = nullptr, MidiBuffer* midiOutput = nullptr) AESTRA_RT_NONBLOCKING override {
105 (void)midiInput;
106 (void)midiOutput;
107
108 if (!m_active.load(std::memory_order_relaxed) || isBypassed()) {
109 copyOrClear(inputs, outputs, numInputChannels, numOutputChannels, numFrames);
110 return;
111 }
112
113 const uint32_t channels = std::min<uint32_t>(2, numOutputChannels);
114 const bool stereo = channels >= 2;
115 const float sr = static_cast<float>(m_sampleRate);
116 const auto target = targetFromNorm(paramValue(kTarget));
117 const auto wave = waveFromNorm(paramValue(kWave));
118 const bool sync = paramValue(kSyncMode) > 0.5f;
119 const float bpm = m_bpm.load(std::memory_order_relaxed);
120 const float maxCutoff = 0.45f * sr;
121
122 // Control-block architecture: parameters, the phase increment (pow /
123 // divide) and the slew coefficient (exp) are computed once per block.
124 // The waveform, S&H and slew stay per sample (that is the LFO). Volume
125 // is a cheap linear function of the LFO so it is applied exactly per
126 // sample; the Pan (cos/sin) and Cutoff (exp2/tan) transforms are
127 // evaluated once per block from the block-boundary LFO value and their
128 // coefficients are linearly interpolated across the block, so no
129 // transcendental runs in the per-sample path.
130 for (uint32_t blockStart = 0; blockStart < numFrames; blockStart += kCtrlBlock) {
131 const uint32_t blockEnd = std::min(blockStart + kCtrlBlock, numFrames);
132 const uint32_t blockLen = blockEnd - blockStart;
133
134 m_depthSmoothed += (getParameter(kDepth) - m_depthSmoothed) * m_smoothCoeff;
135 m_phaseOffSmoothed += (getParameter(kPhase) - m_phaseOffSmoothed) * m_smoothCoeff;
136 const float depth = m_depthSmoothed;
137
138 float periodSec;
139 if (sync) {
140 const int div = noteDivisionIndexFromParam(getParameter(kNoteDivision));
141 periodSec = noteDivisionMultiplier(div) * 60.0f / bpm;
142 } else {
143 periodSec = 1.0f / rateHzFromNorm(getParameter(kRateHz));
144 }
145 const float phaseInc = 1.0f / (std::max(1.0e-3f, periodSec) * sr);
146
147 const float smoothMs = smoothMsFromNorm(getParameter(kSmooth));
148 const bool hardSlew = smoothMs < 0.01f;
149 const float slewCoeff = hardSlew ? 1.0f : 1.0f - std::exp(-1.0f / (smoothMs * 0.001f * sr));
150
151 // Pan/Cutoff coefficient targets from the block-boundary LFO value;
152 // ramp the applied coefficients toward them across the block.
153 float dPanGL = 0.0f, dPanGR = 0.0f, dca1 = 0.0f, dca2 = 0.0f, dca3 = 0.0f;
154 if (target == kTargetPan) {
155 const float pan = depth * m_lfoSlewed; // -1 .. +1
156 const float a = (pan + 1.0f) * 0.25f * 3.14159265358979323846f;
157 const float tgtGL = std::cos(a) * 1.41421356f;
158 const float tgtGR = std::sin(a) * 1.41421356f;
159 if (m_coeffsDirty) {
160 m_panGL = tgtGL;
161 m_panGR = tgtGR;
162 } else {
163 dPanGL = (tgtGL - m_panGL) / static_cast<float>(blockLen);
164 dPanGR = (tgtGR - m_panGR) / static_cast<float>(blockLen);
165 }
166 } else if (target == kTargetCutoff) {
167 const float u = 0.5f + 0.5f * m_lfoSlewed;
168 const float fc =
169 std::clamp(kCutoffTopHz * std::exp2(-kCutoffSweepOctaves * depth * (1.0f - u)), 20.0f, maxCutoff);
170 const float g = std::tan(3.14159265358979323846f * fc / sr);
171 const float k = 1.41421356f; // Butterworth
172 const float ta1 = 1.0f / (1.0f + g * (g + k));
173 const float ta2 = g * ta1;
174 const float ta3 = g * ta2;
175 if (m_coeffsDirty) {
176 m_ca1 = ta1;
177 m_ca2 = ta2;
178 m_ca3 = ta3;
179 } else {
180 dca1 = (ta1 - m_ca1) / static_cast<float>(blockLen);
181 dca2 = (ta2 - m_ca2) / static_cast<float>(blockLen);
182 dca3 = (ta3 - m_ca3) / static_cast<float>(blockLen);
183 }
184 }
185 m_coeffsDirty = false;
186
187 for (uint32_t i = blockStart; i < blockEnd; ++i) {
188 // ── Advance the LFO phase ──
189 m_phase += phaseInc;
190 if (m_phase >= 1.0f)
191 m_phase -= std::floor(m_phase);
192
193 float p = m_phase + m_phaseOffSmoothed;
194 p -= std::floor(p);
195
196 // S&H draws a fresh value each cycle (wrap of the offset phase).
197 if (p < m_prevPhase) {
198 m_lcgState = m_lcgState * 1664525u + 1013904223u;
199 m_shValue = static_cast<float>(m_lcgState >> 8) * (1.0f / 8388608.0f) - 1.0f;
200 }
201 m_prevPhase = p;
202
203 // ── Waveform, bipolar [-1, 1], "open" (+1) at phase 0 ──
204 float b;
205 switch (wave) {
206 case kWaveSine:
207 b = std::cos(2.0f * 3.14159265358979323846f * p);
208 break;
209 case kWaveTriangle:
210 b = 4.0f * std::abs(p - 0.5f) - 1.0f;
211 break;
212 case kWaveSawDown:
213 b = 1.0f - 2.0f * p;
214 break;
215 case kWaveSawUp:
216 b = 2.0f * p - 1.0f;
217 break;
218 case kWaveSquare:
219 b = (p < 0.5f) ? 1.0f : -1.0f;
220 break;
221 default:
222 b = m_shValue;
223 break;
224 }
225
226 // ── Output slew (one-pole toward the raw wave) ──
227 if (hardSlew) {
228 m_lfoSlewed = b;
229 } else {
230 m_lfoSlewed += (b - m_lfoSlewed) * slewCoeff;
231 }
232
233 const float inL = sanitizeSample(readInput(inputs, numInputChannels, 0, i));
234 const float inR = stereo ? sanitizeSample(readInput(inputs, numInputChannels, 1, i)) : inL;
235
236 float outL = inL;
237 float outR = inR;
238 switch (target) {
239 case kTargetVolume: {
240 const float u = 0.5f + 0.5f * m_lfoSlewed; // unipolar, 1 = open
241 const float gain = 1.0f - depth * (1.0f - u);
242 outL = inL * gain;
243 outR = inR * gain;
244 break;
245 }
246 case kTargetPan: {
247 m_panGL += dPanGL;
248 m_panGR += dPanGR;
249 outL = inL * m_panGL;
250 outR = inR * m_panGR;
251 break;
252 }
253 default: { // kTargetCutoff
254 m_ca1 += dca1;
255 m_ca2 += dca2;
256 m_ca3 += dca3;
257 for (uint32_t ch = 0; ch < 2; ++ch) {
258 const float x = (ch == 0) ? inL : inR;
259 float& ic1 = m_ic1[ch];
260 float& ic2 = m_ic2[ch];
261 const float v3 = x - ic2;
262 const float v1 = m_ca1 * ic1 + m_ca2 * v3;
263 const float v2 = ic2 + m_ca2 * ic1 + m_ca3 * v3;
264 ic1 = 2.0f * v1 - ic1;
265 ic2 = 2.0f * v2 - ic2;
266 if (ch == 0)
267 outL = v2;
268 else
269 outR = v2;
270 }
271 break;
272 }
273 }
274
275 if (outputs[0])
276 outputs[0][i] = flushDenormal(outL);
277 if (numOutputChannels > 1 && outputs[1])
278 outputs[1][i] = flushDenormal(stereo ? outR : outL);
279 for (uint32_t ch = 2; ch < numOutputChannels; ++ch) {
280 if (outputs[ch])
281 outputs[ch][i] = 0.0f;
282 }
283 }
284 }
285
286 m_lfoLevel.store(u01(m_lfoSlewed), std::memory_order_relaxed);
287 }

309 std::string getParameterDisplay(uint32_t id) const override {
310 if (id >= kParamCount)
311 return "";
312 const float v = getParameter(id);
313 char buf[32];
314 switch (id) {
315 case kTarget:
316 switch (targetFromNorm(v)) {
317 case kTargetVolume:
318 return "Volume";
319 case kTargetPan:
320 return "Pan";
321 default:
322 return "Cutoff";
323 }
324 case kWave:
325 switch (waveFromNorm(v)) {
326 case kWaveSine:
327 return "Sine";
328 case kWaveTriangle:
329 return "Triangle";
330 case kWaveSawDown:
331 return "Saw Down";
332 case kWaveSawUp:
333 return "Saw Up";
334 case kWaveSquare:
335 return "Square";
336 default:
337 return "S&H";
338 }
339 case kSyncMode:
340 return v > 0.5f ? "Sync" : "Free";
341 case kRateHz:
342 std::snprintf(buf, sizeof(buf), "%.2f Hz", rateHzFromNorm(v));
343 return buf;
344 case kNoteDivision:
345 return noteDivisionLabel(noteDivisionIndexFromParam(v));
346 case kDepth:
347 return std::to_string(static_cast<int>(std::round(v * 100.0f))) + "%";
348 case kPhase:
349 std::snprintf(buf, sizeof(buf), "%.0f deg", v * 360.0f);
350 return buf;
351 case kSmooth:
352 std::snprintf(buf, sizeof(buf), "%.0f ms", smoothMsFromNorm(v));
353 return buf;
354 case kBypass:
355 return v > 0.5f ? "ON" : "OFF";
356 default:
357 return "";
358 }

367 // Sample-rate-relative so the reported tail is constant in time, not in

481 void resetRuntimeState() {
482 m_phase = 0.0f;
483 m_prevPhase = 0.0f;
484 m_lfoSlewed = 1.0f; // "open" so volume target starts transparent
485 m_lcgState = 0x12345678u;
486 m_shValue = 0.0f;
487 m_ic1[0] = m_ic1[1] = 0.0f;
488 m_ic2[0] = m_ic2[1] = 0.0f;
489 m_coeffsDirty = true;
490 m_panGL = 1.0f;
491 m_panGR = 1.0f;
492 m_ca1 = m_ca2 = m_ca3 = 0.0f;
493 const float sr = static_cast<float>(m_sampleRate);
494 // Smoothing runs once per control block, so the coefficient is scaled
495 // to the block interval to keep the 2 ms time constant.
496 m_smoothCoeff = 1.0f - std::exp(-static_cast<float>(kCtrlBlock) / (0.002f * sr));
497 m_lfoLevel.store(1.0f, std::memory_order_relaxed);
498 }

505 // ── Utility ──
506 static void copyOrClear(const float* const* inputs, float** outputs, uint32_t numInputChannels,
507 uint32_t numOutputChannels, uint32_t numFrames) {
508 for (uint32_t ch = 0; ch < numOutputChannels; ++ch) {
509 if (outputs[ch] && ch < numInputChannels && inputs[ch]) {
510 // Skip the copy when the host renders in place (out == in):
511 // memcpy on aliased buffers is undefined and the data is already
512 // where it needs to be.
513 if (outputs[ch] != inputs[ch])
514 std::memcpy(outputs[ch], inputs[ch], numFrames * sizeof(float));
515 } else if (outputs[ch]) {
516 std::memset(outputs[ch], 0, numFrames * sizeof(float));
517 }
518 }
519 }

521 static float readInput(const float* const* inputs, uint32_t channels, uint32_t channel, uint32_t frame) {
522 if (channel >= channels || !inputs[channel]) {
523 return (channel == 1 && channels > 0 && inputs[0]) ? inputs[0][frame] : 0.0f;
524 }
525 return inputs[channel][frame];
526 }
