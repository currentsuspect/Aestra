1// © 2026 Aestra Studios — All Rights Reserved. Licensed for personal & educational use only.
2// AestraLFO V1 — tempo-syncable LFO that shapes the audio flowing through it
3// (LFOTool-style insert effect): rhythmic volume gating/ducking, auto-pan, or
4// filter-cutoff wobble.
5//
6// V1 deliberately modulates its own audio path instead of other plugins'
7// parameters: engine-side parameter modulation routing does not exist yet
8// (issue #467); when it lands, this LFO core can be reused as a source.
9//
10// The LFO free-runs from activate() — it locks to tempo *rate* via setBPM()
11// (same push pattern as AestraDelay) but not to transport *phase*; bar-aligned
12// restart needs a host transport hook that IPluginInstance does not have yet.
13//
14// Signal path (per sample):
15// lfo phase -> waveform -> smoothing (slew) -> target:
16// Volume: gain = 1 - depth * (1 - u) (u = unipolar LFO, 1 = open)
17// Pan: equal-power L/R, unity at center
18// Cutoff: ZDF SVF low-pass swept down up to 6 octaves from 20 kHz
19

109 void process(const float* const* inputs, float** outputs, uint32_t numInputChannels, uint32_t numOutputChannels,
110 uint32_t numFrames, const MidiBuffer* midiInput = nullptr, MidiBuffer* midiOutput = nullptr) override {
111 (void)midiInput;
112 (void)midiOutput;
113
114 if (!m_active.load(std::memory_order_relaxed) || m_params[kBypass].load(std::memory_order_relaxed) > 0.5f) {
115 copyOrClear(inputs, outputs, numInputChannels, numOutputChannels, numFrames);
116 return;
117 }
118
119 const uint32_t channels = std::min<uint32_t>(2, numOutputChannels);
120 const bool stereo = channels >= 2;
121 const float sr = static_cast<float>(m_sampleRate);
122 const auto target = targetFromNorm(m_params[kTarget].load(std::memory_order_relaxed));
123 const auto wave = waveFromNorm(m_params[kWave].load(std::memory_order_relaxed));
124 const bool sync = m_params[kSyncMode].load(std::memory_order_relaxed) > 0.5f;
125 const float bpm = m_bpm.load(std::memory_order_relaxed);
126 const float maxCutoff = 0.45f * sr;
127
128 // Control-block architecture: parameters, the phase increment (pow /
129 // divide) and the slew coefficient (exp) are computed once per block.
130 // The waveform, S&H and slew stay per sample (that is the LFO). Volume
131 // is a cheap linear function of the LFO so it is applied exactly per
132 // sample; the Pan (cos/sin) and Cutoff (exp2/tan) transforms are
133 // evaluated once per block from the block-boundary LFO value and their
134 // coefficients are linearly interpolated across the block, so no
135 // transcendental runs in the per-sample path.
136 for (uint32_t blockStart = 0; blockStart < numFrames; blockStart += kCtrlBlock) {
137 const uint32_t blockEnd = std::min(blockStart + kCtrlBlock, numFrames);
138 const uint32_t blockLen = blockEnd - blockStart;
139
140 m_depthSmoothed += (getParameter(kDepth) - m_depthSmoothed) * m_smoothCoeff;
141 m_phaseOffSmoothed += (getParameter(kPhase) - m_phaseOffSmoothed) * m_smoothCoeff;
142 const float depth = m_depthSmoothed;
143
144 float periodSec;
145 if (sync) {
146 const int div = noteDivisionIndexFromParam(getParameter(kNoteDivision));
147 periodSec = noteDivisionMultiplier(div) * 60.0f / bpm;
148 } else {
149 periodSec = 1.0f / rateHzFromNorm(getParameter(kRateHz));
150 }
151 const float phaseInc = 1.0f / (std::max(1.0e-3f, periodSec) * sr);
152
153 const float smoothMs = smoothMsFromNorm(getParameter(kSmooth));
154 const bool hardSlew = smoothMs < 0.01f;
155 const float slewCoeff = hardSlew ? 1.0f : 1.0f - std::exp(-1.0f / (smoothMs * 0.001f * sr));
156
157 // Pan/Cutoff coefficient targets from the block-boundary LFO value;
158 // ramp the applied coefficients toward them across the block.
159 float dPanGL = 0.0f, dPanGR = 0.0f, dca1 = 0.0f, dca2 = 0.0f, dca3 = 0.0f;
160 if (target == kTargetPan) {
161 const float pan = depth * m_lfoSlewed; // -1 .. +1
162 const float a = (pan + 1.0f) * 0.25f * 3.14159265358979323846f;
163 const float tgtGL = std::cos(a) * 1.41421356f;
164 const float tgtGR = std::sin(a) * 1.41421356f;
165 if (m_coeffsDirty) {
166 m_panGL = tgtGL;
167 m_panGR = tgtGR;
168 } else {
169 dPanGL = (tgtGL - m_panGL) / static_cast<float>(blockLen);
170 dPanGR = (tgtGR - m_panGR) / static_cast<float>(blockLen);
171 }
172 } else if (target == kTargetCutoff) {
173 const float u = 0.5f + 0.5f * m_lfoSlewed;
174 const float fc =
175 std::clamp(kCutoffTopHz * std::exp2(-kCutoffSweepOctaves * depth * (1.0f - u)), 20.0f, maxCutoff);
176 const float g = std::tan(3.14159265358979323846f * fc / sr);
177 const float k = 1.41421356f; // Butterworth
178 const float ta1 = 1.0f / (1.0f + g * (g + k));
179 const float ta2 = g * ta1;
180 const float ta3 = g * ta2;
181 if (m_coeffsDirty) {
182 m_ca1 = ta1;
183 m_ca2 = ta2;
184 m_ca3 = ta3;
185 } else {
186 dca1 = (ta1 - m_ca1) / static_cast<float>(blockLen);
187 dca2 = (ta2 - m_ca2) / static_cast<float>(blockLen);
188 dca3 = (ta3 - m_ca3) / static_cast<float>(blockLen);
189 }
190 }
191 m_coeffsDirty = false;
192
193 for (uint32_t i = blockStart; i < blockEnd; ++i) {
194 // ── Advance the LFO phase ──
195 m_phase += phaseInc;
196 if (m_phase >= 1.0f)
197 m_phase -= std::floor(m_phase);
198
199 float p = m_phase + m_phaseOffSmoothed;
200 p -= std::floor(p);
201
202 // S&H draws a fresh value each cycle (wrap of the offset phase).
203 if (p < m_prevPhase) {
204 m_lcgState = m_lcgState * 1664525u + 1013904223u;
205 m_shValue = static_cast<float>(m_lcgState >> 8) * (1.0f / 8388608.0f) - 1.0f;
206 }
207 m_prevPhase = p;
208
209 // ── Waveform, bipolar [-1, 1], "open" (+1) at phase 0 ──
210 float b;
211 switch (wave) {
212 case kWaveSine:
213 b = std::cos(2.0f * 3.14159265358979323846f * p);
214 break;
215 case kWaveTriangle:
216 b = 4.0f * std::abs(p - 0.5f) - 1.0f;
217 break;
218 case kWaveSawDown:
219 b = 1.0f - 2.0f * p;
220 break;
221 case kWaveSawUp:
222 b = 2.0f * p - 1.0f;
223 break;
224 case kWaveSquare:
225 b = (p < 0.5f) ? 1.0f : -1.0f;
226 break;
227 default:
228 b = m_shValue;
229 break;
230 }
231
232 // ── Output slew (one-pole toward the raw wave) ──
233 if (hardSlew) {
234 m_lfoSlewed = b;
235 } else {
236 m_lfoSlewed += (b - m_lfoSlewed) * slewCoeff;
237 }
238
239 const float inL = sanitizeSample(readInput(inputs, numInputChannels, 0, i));
240 const float inR = stereo ? sanitizeSample(readInput(inputs, numInputChannels, 1, i)) : inL;
241
242 float outL = inL;
243 float outR = inR;
244 switch (target) {
245 case kTargetVolume: {
246 const float u = 0.5f + 0.5f * m_lfoSlewed; // unipolar, 1 = open
247 const float gain = 1.0f - depth * (1.0f - u);
248 outL = inL * gain;
249 outR = inR * gain;
250 break;
251 }
252 case kTargetPan: {
253 m_panGL += dPanGL;
254 m_panGR += dPanGR;
255 outL = inL * m_panGL;
256 outR = inR * m_panGR;
257 break;
258 }
259 default: { // kTargetCutoff
260 m_ca1 += dca1;
261 m_ca2 += dca2;
262 m_ca3 += dca3;
263 for (uint32_t ch = 0; ch < 2; ++ch) {
264 const float x = (ch == 0) ? inL : inR;
265 float& ic1 = m_ic1[ch];
266 float& ic2 = m_ic2[ch];
267 const float v3 = x - ic2;
268 const float v1 = m_ca1 * ic1 + m_ca2 * v3;
269 const float v2 = ic2 + m_ca2 * ic1 + m_ca3 * v3;
270 ic1 = 2.0f * v1 - ic1;
271 ic2 = 2.0f * v2 - ic2;
272 if (ch == 0)
273 outL = v2;
274 else
275 outR = v2;
276 }
277 break;
278 }
279 }
280
281 if (outputs[0])
282 outputs[0][i] = flushDenormal(outL);
283 if (numOutputChannels > 1 && outputs[1])
284 outputs[1][i] = flushDenormal(stereo ? outR : outL);
285 for (uint32_t ch = 2; ch < numOutputChannels; ++ch) {
286 if (outputs[ch])
287 outputs[ch][i] = 0.0f;
288 }
289 }
290 }
291
292 m_lfoLevel.store(u01(m_lfoSlewed), std::memory_order_relaxed);
293 }

369 return buf;
370 case kBypass:

377 std::vector<uint8_t> saveState() const override {
378 struct Blob {
379 uint32_t magic = kStateMagic;
380 uint32_t version = 1;
381 float params[kParamCount] = {};
382 } blob;
383 for (uint32_t i = 0; i < kParamCount; ++i) {
384 blob.params[i] = getParameter(i);
385 }
386 const auto* data = reinterpret_cast<const uint8_t*>(&blob);
387 return {data, data + sizeof(blob)};
388 }
389
390 bool loadState(const std::vector<uint8_t>& state) override {
391 if (state.size() < sizeof(uint32_t) * 2)
392 return false;
393 uint32_t magic = 0;
394 std::memcpy(&magic, state.data(), sizeof(magic));
395 if (magic != kStateMagic)
396 return false;
397 struct Blob {
398 uint32_t magic;
399 uint32_t version;
400 float params[kParamCount];
401 };
402 if (state.size() < sizeof(Blob))
403 return false;
404 Blob blob{};
405 std::memcpy(&blob, state.data(), sizeof(blob));
406 if (blob.version < 1 || blob.version > 1)
407 return false;
408 // Validate the entire decoded set before mutating anything. setParameter
409 // silently drops non-finite values, so applying in place would leave a
410 // half-updated state while still reporting success — fail atomically.
411 for (uint32_t i = 0; i < kParamCount; ++i) {
412 if (!std::isfinite(blob.params[i]) || blob.params[i] < 0.0f || blob.params[i] > 1.0f)
413 return false;
414 }
415 for (uint32_t i = 0; i < kParamCount; ++i) {
416 setParameter(i, blob.params[i]);
417 }
418 return true;
419 }

549 void resetRuntimeState() {
550 m_phase = 0.0f;
551 m_prevPhase = 0.0f;
552 m_lfoSlewed = 1.0f; // "open" so volume target starts transparent
553 m_lcgState = 0x12345678u;
554 m_shValue = 0.0f;
555 m_ic1[0] = m_ic1[1] = 0.0f;
556 m_ic2[0] = m_ic2[1] = 0.0f;
557 m_coeffsDirty = true;
558 m_panGL = 1.0f;
559 m_panGR = 1.0f;
560 m_ca1 = m_ca2 = m_ca3 = 0.0f;
561 const float sr = static_cast<float>(m_sampleRate);
562 // Smoothing runs once per control block, so the coefficient is scaled
563 // to the block interval to keep the 2 ms time constant.
564 m_smoothCoeff = 1.0f - std::exp(-static_cast<float>(kCtrlBlock) / (0.002f * sr));
565 m_lfoLevel.store(1.0f, std::memory_order_relaxed);
566 }

573 // ── Utility ──
574 static void copyOrClear(const float* const* inputs, float** outputs, uint32_t numInputChannels,
575 uint32_t numOutputChannels, uint32_t numFrames) {
576 for (uint32_t ch = 0; ch < numOutputChannels; ++ch) {
577 if (outputs[ch] && ch < numInputChannels && inputs[ch]) {
578 // Skip the copy when the host renders in place (out == in):
579 // memcpy on aliased buffers is undefined and the data is already
580 // where it needs to be.
581 if (outputs[ch] != inputs[ch])
582 std::memcpy(outputs[ch], inputs[ch], numFrames * sizeof(float));
583 } else if (outputs[ch]) {
584 std::memset(outputs[ch], 0, numFrames * sizeof(float));
585 }
586 }
587 }

589 static float readInput(const float* const* inputs, uint32_t channels, uint32_t channel, uint32_t frame) {
590 if (channel >= channels || !inputs[channel]) {
