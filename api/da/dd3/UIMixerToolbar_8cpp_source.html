60void UIMixerToolbar::onRender(NUIRenderer& renderer) {
61 const auto b = getBounds();
62 if (b.isEmpty()) return;
63 auto& theme = NUIThemeManager::getInstance();
64 const NUIColor text = theme.getColor("textPrimary");
65 const NUIColor muted = theme.getColor("textSecondary");
66 const NUIColor border = theme.getColor("border").withAlpha(0.45f);
67 const NUIColor accent = theme.getColor("accentPrimary");
68
69 // The title and summary take only the room the controls leave: the summary shortens, then goes, then the title.
70 const float room = densityRect(MixerDensity::Wide).x - 8.0f;
71 const float titleW = renderer.measureText("Mixer", 11.0f).width;
72 if (b.x + 8.0f + titleW <= room)
73 renderer.drawText("Mixer", {b.x + 8.0f, b.y + (b.height - 11.0f) * 0.5f - 1.0f}, 11.0f, text.withAlpha(0.95f));
74 const float sx = b.x + 8.0f + titleW + 10.0f;
75 std::string summary = m_summary;
76 if (sx + renderer.measureText(summary, kFont).width > room) summary = m_summary.substr(0, m_summary.find(" \xC2\xB7"));
77 if (sx + renderer.measureText(summary, kFont).width <= room)
78 renderer.drawText(summary, {sx, b.y + (b.height - kFont) * 0.5f - 1.0f}, kFont, muted.withAlpha(0.8f));
79
80 static constexpr const char* names[3] = {"Wide", "Normal", "Narrow"};
81 const auto seg = NUIRect{densityRect(MixerDensity::Wide).x, densityRect(MixerDensity::Wide).y, segW() * 3.0f, kH};
82 renderer.strokeRoundedRect(seg, 3.0f, 1.0f, border);
83 for (int i = 0; i < 3; ++i) {
84 const auto r = densityRect(static_cast<MixerDensity>(i));
85 const bool on = static_cast<int>(m_density) == i;
86 if (on) renderer.fillRoundedRect(r, 3.0f, accent.withAlpha(0.28f));
87 else if (m_hover == i) renderer.fillRoundedRect(r, 3.0f, border.withAlpha(0.35f));
88 renderer.drawTextCentered(names[i], r, kFont, on ? text : muted);
89 }
90
91 const auto g = groupRect();
92 renderer.strokeRoundedRect(g, 3.0f, 1.0f, border);
93 // The switch: a small track with a knob that sits right when on.
94 const NUIRect track{g.x + 8.0f, g.y + (g.height - 10.0f) * 0.5f, 22.0f, 10.0f};
95 renderer.fillRoundedRect(track, 5.0f, m_group ? accent.withAlpha(0.85f) : border);
96 renderer.fillRoundedRect(NUIRect{m_group ? track.right() - 9.0f : track.x + 1.0f, track.y + 1.0f, 8.0f, 8.0f}, 4.0f, text);
97 renderer.drawText(compact() ? "Group" : "Group by output", {track.right() + 6.0f, g.y + (g.height - 9.0f) * 0.5f - 1.0f}, 9.0f,
98 m_hover == 3 ? text : muted);
99
100 const auto r = resetRect();
101 if (m_hover == 4) renderer.fillRoundedRect(r, 3.0f, border.withAlpha(0.35f));
102 renderer.strokeRoundedRect(r, 3.0f, 1.0f, border);
103 renderer.drawTextCentered(compact() ? "Reset" : "Reset peaks", r, kFont, m_hover == 4 ? text : muted);
104}

111bool UIMixerToolbar::onMouseEvent(const NUIMouseEvent& event) {
112 if (!isVisible() || !isEnabled()) return false;
113 int hit = -1;
114 for (int i = 0; i < 3; ++i)
115 if (densityRect(static_cast<MixerDensity>(i)).contains(event.position)) hit = i;
116 if (groupRect().contains(event.position)) hit = 3;
117 if (resetRect().contains(event.position)) hit = 4;
118
119 if (event.button == NUIMouseButton::None) {
120 if (hit != m_hover) { m_hover = hit; repaint(); }
121 return false;
122 }
123 if (event.pressed && event.button == NUIMouseButton::Left && hit >= 0) {
124 if (hit < 3) {
125 const auto d = static_cast<MixerDensity>(hit);
126 if (d != m_density) {
127 m_density = d;
128 repaint();
129 if (onDensityChanged) onDensityChanged(d);
130 }
131 } else if (hit == 3) {
132 m_group = !m_group;
133 repaint();
134 if (onGroupToggled) onGroupToggled(m_group);
135 } else if (onResetPeaks) {
136 onResetPeaks();
137 }
138 return true;
139 }
140 return false;
141}
