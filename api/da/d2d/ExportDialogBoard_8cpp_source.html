26std::string formatLabel(Aestra::Audio::ExportFormat f) {
27 switch (f) {
28 case Aestra::Audio::ExportFormat::Flac:
29 return "FLAC";
30 case Aestra::Audio::ExportFormat::Mp3:
31 return "MP3";
32 default:
33 return "WAV";
34 }
35}

37std::vector<std::string> depthsFor(Aestra::Audio::ExportFormat f) {
38 switch (f) {
39 case Aestra::Audio::ExportFormat::Flac:
40 return {"16-bit", "24-bit"};
41 case Aestra::Audio::ExportFormat::Mp3:
42 return {"320 kbps"};
43 default:
44 return {"16-bit", "24-bit", "32-bit float"};

49void ExportDialog::buildBoardControls() {
50 // The rows of the old layout go: scope and stems are segmented controls painted here, the depth menu follows the
51 // format, and the old labels are not needed.
52 for (const auto& label : {m_outputLabel, m_scopeLabel, m_rateLabel, m_depthLabel, m_tailLabel})
53 label->setVisible(false);
54 m_scopeDropdown->setVisible(false);
55 m_stemsCheckbox->setVisible(false);
56
57 m_formats = {Aestra::Audio::ExportFormat::Wav};
58 for (const auto& option : m_bitDepthOptions) {
59 if (option.rfind("FLAC", 0) == 0 && m_formats.back() != Aestra::Audio::ExportFormat::Flac)
60 m_formats.push_back(Aestra::Audio::ExportFormat::Flac);
61 if (option.rfind("MP3", 0) == 0)
62 m_formats.push_back(Aestra::Audio::ExportFormat::Mp3);
63 }
64 m_formatDropdown = std::make_shared<AestraUI::NUIDropdown>();
65 for (size_t i = 0; i < m_formats.size(); ++i)
66 m_formatDropdown->addItem(formatLabel(m_formats[i]), static_cast<int>(i));
67 m_formatDropdown->setOnSelectionChanged([this](int i) {
68 if (i >= 0 && static_cast<size_t>(i) < m_formats.size())
69 setFormat(m_formats[static_cast<size_t>(i)], -1);
70 });
71 addChild(m_formatDropdown);
72 m_depthDropdown->setOnSelectionChanged([this](int i) { setFormat(m_format, i); });
73 m_browseButton->setText("in folder");
74 m_outputInput->setOnTextChange([this](const std::string&) {
75 refreshOutputPath();
76 setDirty(true);
77 });
78}

80// m_selectedBitDepth stays the index into m_bitDepthOptions ("WAV 24-bit", "FLAC 16-bit", "MP3 320 kbps"), so the
81// saved export settings keep their meaning; the two chips are a view of it.
82void ExportDialog::setFormat(Aestra::Audio::ExportFormat format, int depthChoice) {
83 const bool formatChanged = format != m_format;
84 m_format = format;
85 const auto depths = depthsFor(format);
86 if (depthChoice < 0 || static_cast<size_t>(depthChoice) >= depths.size()) {
87 depthChoice = format == Aestra::Audio::ExportFormat::Mp3 ? 0 : 1; // 24-bit by default
88 }
89 const std::string wanted = formatLabel(format) + " " + depths[static_cast<size_t>(depthChoice)];
90 for (size_t i = 0; i < m_bitDepthOptions.size(); ++i) {
91 if (m_bitDepthOptions[i] == wanted)
92 m_selectedBitDepth = static_cast<int>(i);
93 }
94 if (formatChanged)
95 syncFormatChips();
96 refreshOutputPath();
97 setDirty(true);
98}

100void ExportDialog::syncFormatChips() {
101 const std::string& option = m_bitDepthOptions[static_cast<size_t>(
102 std::clamp(m_selectedBitDepth, 0, static_cast<int>(m_bitDepthOptions.size()) - 1))];
103 m_format = formatOf(m_selectedBitDepth);
104 for (size_t i = 0; i < m_formats.size(); ++i)
105 if (m_formats[i] == m_format)
106 m_formatDropdown->setSelectedIndex(static_cast<int>(i));
107 const auto depths = depthsFor(m_format);
108 m_depthDropdown->clearItems();
109 int selected = 0;
110 for (size_t i = 0; i < depths.size(); ++i) {
111 m_depthDropdown->addItem(depths[i], static_cast<int>(i));
112 if (option.find(depths[i]) != std::string::npos)
113 selected = static_cast<int>(i);
114 }
115 m_depthDropdown->setSelectedIndex(selected);
116}

119// The file name without any extension a format owns; "Untitled" when nothing is left (a new project's default
120// export name is only the extension).
121std::string bareName(std::string name) {
122 for (const char* ext : {".wav", ".flac", ".mp3"}) {
123 const size_t n = std::strlen(ext);
124 if (name.size() >= n && name.compare(name.size() - n, n, ext) == 0)
125 name.resize(name.size() - n);
126 }
127 return name.empty() ? std::string("Untitled") : name;
128}

130// Where an export goes when the project has no folder yet: Renders in the Aestra folder, else home.
131std::string defaultExportFolder() {
132 const char* home = std::getenv("HOME");
133#ifdef _WIN32
134 if (!home)
135 home = std::getenv("USERPROFILE");
136#endif
137 if (!home)
138 return std::filesystem::current_path().string();
139 std::error_code ec;
140 const auto renders = std::filesystem::path(home) / "Documents" / "Aestra" / "Renders";
141 return std::filesystem::is_directory(renders, ec) ? renders.string() : std::string(home);
142}

165std::string ExportDialog::summaryText() const {
166 double seconds = -1.0;
167 if (m_engine && m_trackManager) {
168 const double bpm = m_engine->getBPM() > 0.0 ? m_engine->getBPM() : 120.0;
169 double beats = -1.0;
170 if (m_selectedScope == 0)
171 beats = m_trackManager->getPlaylistModel().getTotalDurationBeats();
172 else if (m_selectedScope == 1)
173 beats = m_engine->getLoopEndBeat() - m_engine->getLoopStartBeat();
174 if (beats > 0.0)
175 seconds = beats * 60.0 / bpm + m_tailSeconds;
176 }
177 char length[32] = "\xe2\x80\x94"; // em dash: the selection's length is not known here
178 if (seconds > 0.0) {
179 const int s = static_cast<int>(std::lround(seconds));
180 std::snprintf(length, sizeof(length), "%d:%02d", s / 60, s % 60);
181 }
182 std::string size = "\xe2\x80\x94";
183 if (seconds > 0.0 && !m_stems) {
184 const double rate = kSampleRates[std::clamp(m_selectedSampleRate, 0, 3)];
185 const std::string& option = m_bitDepthOptions[static_cast<size_t>(
186 std::clamp(m_selectedBitDepth, 0, static_cast<int>(m_bitDepthOptions.size()) - 1))];
187 double bytes = 0.0;
188 if (m_format == Aestra::Audio::ExportFormat::Mp3) {
189 bytes = seconds * 320000.0 / 8.0;
190 } else {
191 const double sampleBytes = option.find("16") != std::string::npos ? 2.0
192 : option.find("32") != std::string::npos ? 4.0
193 : 3.0;
194 bytes = seconds * rate * 2.0 * sampleBytes;
195 if (m_format == Aestra::Audio::ExportFormat::Flac)
196 bytes *= 0.6; // typical for music
197 }
198 char buf[32];
199 std::snprintf(buf, sizeof(buf), "%s%.0f MB", m_format == Aestra::Audio::ExportFormat::Flac ? "about " : "",
200 bytes / 1.0e6);
201 size = buf;
202 }
203 const std::string files = m_stems ? "one WAV per track, plus the master" : "one stereo file";
204 return std::string(length) + " \xc2\xb7 " + size + " \xc2\xb7 " + files;
205}

207void ExportDialog::layoutSettingsBoard() {
208 const float x = m_dialogRect.x + kPad;
209 const float w = m_dialogRect.width - kPad * 2.0f;
210 float y = m_dialogRect.y + 32.0f + 14.0f;
211
212 m_whatCaption = {x, y, w, kCaptionH};
213 y += kCaptionH + 2.0f;
214 const float segW3 = w / 3.0f;
215 for (size_t i = 0; i < m_scopeSegments.size(); ++i)
216 m_scopeSegments[i] = {x + segW3 * static_cast<float>(i), y, segW3, kSegH};
217 y += kSegH + 6.0f;
218 for (size_t i = 0; i < m_stemsSegments.size(); ++i)
219 m_stemsSegments[i] = {x + w * 0.5f * static_cast<float>(i), y, w * 0.5f, kSegH};
220 y += kSegH + 16.0f;
221
222 m_whereCaption = {x, y, w, kCaptionH};
223 y += kCaptionH + 2.0f;
224 const float folderW = 120.0f;
225 m_outputInput->setBounds({x, y, w - folderW - 6.0f, kRowH});
226 m_browseButton->setBounds({x + w - folderW, y, folderW, kRowH});
227 y += kRowH + 4.0f;
228 m_pathLine = {x, y, w, 16.0f};
229 y += 16.0f + 12.0f;
230
231 m_formatCaption = {x, y, w, kCaptionH};
232 y += kCaptionH + 2.0f;
233 float cx = x;
234 const auto chip = [&](const std::shared_ptr<AestraUI::NUIComponent>& c, float cw) {
235 c->setBounds({cx, y, cw, kChipH});
236 cx += cw + 6.0f;
237 };
238 chip(m_formatDropdown, 78.0f);
239 chip(m_depthDropdown, 112.0f);
240 chip(m_rateDropdown, 96.0f);
241 cx += 34.0f; // room for the "Tail" word painted before the field
242 chip(m_tailField, 56.0f);
243 y += kChipH + 12.0f;
244 m_summaryLine = {x, y, w, 22.0f};
245}

247void ExportDialog::paintSettingsBoard(AestraUI::NUIRenderer& renderer) {
248 auto& theme = AestraUI::NUIThemeManager::getInstance();
249 const float capFont = theme.getCurrentTheme().fontSizeXS;
250 const auto muted = theme.getColor("textMuted");
251 const auto border = theme.getColor("border");
252 const auto caption = [&](const AestraUI::NUIRect& r, const char* text) {
253 renderer.drawText(text, {r.x, renderer.calculateTextY(r, capFont)}, capFont, muted);
254 };
255 caption(m_whatCaption, "WHAT");
256 caption(m_whereCaption, "WHERE");
257 caption(m_formatCaption, "FORMAT");
258
259 // A segmented control: one outline, the chosen segment filled.
260 const auto segmented = [&](const auto& segments, const char* const* labels, int chosen) {
261 const AestraUI::NUIRect all(segments.front().x, segments.front().y,
262 segments.back().right() - segments.front().x, segments.front().height);
263 renderer.strokeRoundedRect(all, 4.0f, 1.0f, border);
264 for (size_t i = 0; i < segments.size(); ++i) {
265 const auto& s = segments[i];
266 const bool on = static_cast<int>(i) == chosen;
267 if (on) {
268 renderer.fillRoundedRect({s.x + 3.0f, s.y + 3.0f, s.width - 6.0f, s.height - 6.0f}, 3.0f,
269 theme.getColor("primary").withAlpha(0.24f));
270 }
271 renderer.drawTextCentered(labels[i], s, 13.0f, theme.getColor(on ? "textPrimary" : "textSecondary"));
272 }
273 };
274 static const char* const kScope[] = {"Full song", "Loop", "Selection"};
275 static const char* const kStems[] = {"One file", "Stems + master"};
276 segmented(m_scopeSegments, kScope, m_selectedScope);
277 segmented(m_stemsSegments, kStems, m_stems ? 1 : 0);
278
279 const std::string ext = std::string(".") + Aestra::Audio::exportFormatExtension(m_format);
280
281 std::string where = m_outputPath;
282 if (const char* home = std::getenv("HOME"); home && where.rfind(home, 0) == 0)
283 where = "~" + where.substr(std::strlen(home));
284 if (m_stems) {
285 const std::filesystem::path base(m_outputPath);
286 where = std::filesystem::path(where).parent_path().string() + "/" + base.stem().string() + " - 01 Track" + ext +
287 ", ...";
288 }
289 renderer.drawText(where, {m_pathLine.x, renderer.calculateTextY(m_pathLine, 11.0f)}, 11.0f, muted);
290
291 const auto& tail = m_tailField->getBounds();
292 renderer.drawText("Tail", {tail.x - 32.0f, renderer.calculateTextY(tail, 12.5f)}, 12.5f,
293 theme.getColor("textSecondary"));
294
295 renderer.drawLine({m_summaryLine.x, m_summaryLine.y - 4.0f}, {m_summaryLine.right(), m_summaryLine.y - 4.0f}, 1.0f,
296 border.withAlpha(0.6f));
297 renderer.drawText(summaryText(), {m_summaryLine.x, renderer.calculateTextY(m_summaryLine, 11.5f)}, 11.5f, muted);
298}

300bool ExportDialog::handleBoardClick(const AestraUI::NUIMouseEvent& event) {
301 if (!event.pressed || event.button != AestraUI::NUIMouseButton::Left)
302 return false;
303 for (size_t i = 0; i < m_scopeSegments.size(); ++i) {
304 if (m_scopeSegments[i].contains(event.position)) {
305 m_selectedScope = static_cast<int>(i);
306 setDirty(true);
307 return true;
308 }
309 }
310 for (size_t i = 0; i < m_stemsSegments.size(); ++i) {
311 if (m_stemsSegments[i].contains(event.position)) {
312 m_stems = i == 1;
313 // Stems are written as WAV (exportStems): show that rather than a format they will not get.
314 if (m_stems && m_format != Aestra::Audio::ExportFormat::Wav)
315 setFormat(Aestra::Audio::ExportFormat::Wav, -1);
316 m_formatDropdown->setEnabled(!m_stems);
317 refreshOutputPath();
318 setDirty(true);
319 return true;
320 }
321 }
322 return false;
323}
