3// PluginEditorRegistry — which editor a plugin gets, as data.
4//
5// P4's registry. The dispatch was twelve arms of `else if (pluginId == "...")`
6// in PluginUIController.cpp, each doing the identical four steps:
7//
8// auto ed = std::make_shared<XEditor>(instance);
9// wireEditorClose(ed);
10// ed->setPlatformBridge(m_platformBridge);
11// editorComp = ed;
12//
13// A typo in a plugin id compiled clean and silently degraded the plugin to the
14// generic editor. A new special editor meant finding the right place in an
15// if/else chain in a 700-line controller. And the four steps being copy-pasted
16// twelve times is the shape that lets one arm forget the close-wiring and leak
17// an editor, which nothing would catch.
18//
19// The registry inverts that: one table, and the four steps exist once.
20//
21// WHAT THIS IS NOT. It is not a layout kit, and it does not try to be. The eleven
22// editors are 12,297 lines, AestraEQEditor alone is 5,646, and AestraVerbEditor
23// carries a visualizer. Making a new effect's editor "a layout plus, if needed, a
24// visualizer" means rewriting those as declarative layouts, and that cannot be
25// verified headlessly -- it needs eyes on rendering. That work is deliberately
26// left for v0.8.0 rather than attempted blind inside a hardening pass.
27//
28// The exit criterion P4 actually needed is mostly met already: a new effect needs
29// no editor at all today, because the fallback hands it GenericPluginEditor. What
30// was missing is that the fallback sat inside an if/else chain. This moves it into
31// a table where the entries are enumerable, testable, and checkable against the
32// registry of plugins that actually exist.

125inline constexpr bool pluginEditorHasDedicatedEntry(const std::string& pluginId) {
126 for (const PluginEditorRegistration& r : pluginEditorRegistry) {
127 if (r.pluginId[0] != '\0' && std::strcmp(r.pluginId, pluginId.c_str()) == 0)
128 return true;
129 }
130 return false;
131}

133inline constexpr const char* pluginEditorNameFor(const std::string& pluginId) {
134 for (const PluginEditorRegistration& r : pluginEditorRegistry) {
135 if (r.pluginId[0] != '\0' && std::strcmp(r.pluginId, pluginId.c_str()) == 0)
136 return r.editorName;
137 }
138 return pluginEditorRegistry.front().editorName;
139}
