22class AddEffectCommand : public ICommand {
23public:
24 AddEffectCommand(TrackManager& trackManager, MixerChannel& channel, size_t slotIndex,
25 PluginInstancePtr plugin, std::string effectName)
26 : m_trackManager(trackManager), m_channel(channel), m_slotIndex(slotIndex),
27 m_plugin(std::move(plugin)), m_effectName(std::move(effectName)) {}
28
29 void execute() override {
30 if (m_executed) return;
31 // A redo must restore the identity the first execute minted, not invent a
32 // second one for the same plugin (#667) — see AddPluginCommand. 0 on the
33 // first pass is exactly the "mint one" signal.
34 m_channel.getEffectChain().insertPlugin(m_slotIndex, m_plugin, m_instanceId);
35 m_instanceId = m_channel.getEffectChain().getSlotInstanceId(m_slotIndex);
36 m_trackManager.requestAudioGraphRebuild(GraphDirtyReason::EffectChainChanged);
37 m_executed = true;
38 }
39
40 void undo() override {
41 if (!m_executed) return;
42 m_channel.getEffectChain().removePlugin(m_slotIndex);
43 m_trackManager.requestAudioGraphRebuild(GraphDirtyReason::EffectChainChanged);
44 m_executed = false;
45 }
46
47 void redo() override {
48 if (m_executed) return;
49 execute();
50 }
51
52 std::string getName() const override { return "Add " + m_effectName; }
53 size_t getSizeInBytes() const override { return sizeof(*this); }
54 bool changesProjectState() const override { return true; }
55
56 size_t getSlotIndex() const { return m_slotIndex; }
58 bool wasExecuted() const { return m_executed; }
59
60private:
61 TrackManager& m_trackManager;
62 MixerChannel& m_channel;
63 size_t m_slotIndex;
64 PluginInstancePtr m_plugin;
65 std::string m_effectName;
66 uint64_t m_instanceId{0};
67 bool m_executed = false;
68};

73class RemoveEffectCommand : public ICommand {
74public:
75 RemoveEffectCommand(TrackManager& trackManager, MixerChannel& channel, size_t slotIndex)
76 : m_trackManager(trackManager), m_channel(channel), m_slotIndex(slotIndex) {}
77
78 void execute() override {
79 if (m_executed) return;
80 // Capture before removing so undo re-seats the SAME plugin rather than a
81 // new one wearing its slot (#667) — see RemovePluginCommand.
82 m_removedInstanceId = m_channel.getEffectChain().getSlotInstanceId(m_slotIndex);
83 m_removedPlugin = m_channel.getEffectChain().removePlugin(m_slotIndex);
84 m_trackManager.requestAudioGraphRebuild(GraphDirtyReason::EffectChainChanged);
85 m_executed = true;
86 }
87
88 void undo() override {
89 if (!m_executed) return;
90 if (m_removedPlugin) {
91 m_channel.getEffectChain().insertPlugin(m_slotIndex, m_removedPlugin, m_removedInstanceId);
92 }
93 m_trackManager.requestAudioGraphRebuild(GraphDirtyReason::EffectChainChanged);
94 m_executed = false;
95 }
96
97 void redo() override {
98 if (m_executed) return;
99 execute();
100 }
101
102 std::string getName() const override { return "Remove Effect"; }
103 size_t getSizeInBytes() const override { return sizeof(*this); }
104 bool changesProjectState() const override { return true; }
105
106private:
107 TrackManager& m_trackManager;
108 MixerChannel& m_channel;
109 size_t m_slotIndex;
110 PluginInstancePtr m_removedPlugin;
111 uint64_t m_removedInstanceId{0};
112 bool m_executed = false;
113};

207class MovePluginCommand : public ICommand {
208public:
209 MovePluginCommand(TrackManager& trackManager, MixerChannel& channel, size_t fromSlot, size_t toSlot)
210 : m_trackManager(trackManager), m_channel(channel), m_fromSlot(fromSlot), m_toSlot(toSlot) {}
211
212 void execute() override {
213 auto& chain = m_channel.getEffectChain();
214 if (m_fromSlot >= EffectChain::MAX_SLOTS || m_toSlot >= EffectChain::MAX_SLOTS) {
215 throw std::runtime_error("MovePlugin: slot out of range");
216 }
217 if (m_fromSlot == m_toSlot) {
218 throw std::runtime_error("MovePlugin: source and destination are the same slot");
219 }
220 if (chain.getSlot(m_fromSlot) == nullptr || chain.getSlot(m_fromSlot)->isEmpty()) {
221 throw std::runtime_error("MovePlugin: source slot is empty");
222 }
223 const auto* target = chain.getSlot(m_toSlot);
224 const bool targetEmpty = !target || target->isEmpty();
225 if (targetEmpty) {
226 if (!chain.movePlugin(m_fromSlot, m_toSlot)) {
227 throw std::runtime_error("MovePlugin: move refused by the chain");
228 }
229 m_usedSwap = false;
230 } else {
231 if (!chain.swapPlugins(m_fromSlot, m_toSlot)) {
232 throw std::runtime_error("MovePlugin: swap refused by the chain");
233 }
234 m_usedSwap = true;
235 }
236 m_executed = true;
237 m_trackManager.requestAudioGraphRebuild(GraphDirtyReason::EffectChainChanged);
238 }
239 void undo() override {
240 if (!m_executed) {
241 return;
242 }
243 auto& chain = m_channel.getEffectChain();
244 // A refused revert must not leave the history believing the reorder
245 // was undone while the chain still holds the moved layout.
246 const bool reverted = m_usedSwap ? chain.swapPlugins(m_toSlot, m_fromSlot)
247 : chain.movePlugin(m_toSlot, m_fromSlot);
248 if (!reverted) {
249 throw std::runtime_error("MovePlugin: undo refused by the chain");
250 }
251 m_executed = false;
252 m_trackManager.requestAudioGraphRebuild(GraphDirtyReason::EffectChainChanged);
253 }
254 void redo() override { execute(); }
255
256 std::string getName() const override { return "Move Plugin"; }
257 bool changesProjectState() const override { return true; }
258 std::string type() const override { return "move_plugin"; }
259
260private:
261 TrackManager& m_trackManager;
262 MixerChannel& m_channel;
263 size_t m_fromSlot;
264 size_t m_toSlot;
265 bool m_usedSwap = false;
266 bool m_executed = false;
267};
