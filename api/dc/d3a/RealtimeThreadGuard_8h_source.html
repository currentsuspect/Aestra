183// Installed once at startup from the main thread, never from the callback, so
184// it stays outside both the annotation and the waiver below.
185inline RealtimeMisuseHandler setRealtimeMisuseHandler(RealtimeMisuseHandler handler) noexcept {
186 return g_realtimeMisuseHandler.exchange(handler, std::memory_order_acq_rel);
187}

189// ── Waiver region ──
190// Everything from here to the matching pop touches g_realtimeAudioThreadDepth
191// or dispatches through the handler pointer, the two things Clang's effect
192// analysis cannot clear on its own. The justification is the initial-exec pin
193// and the handler contract, both argued at the top of this file. Nothing else
194// belongs in here: an unrelated function parked inside this region would have
195// its diagnostics silenced by an argument that does not cover it.

201inline bool isRealtimeAudioThread() noexcept AESTRA_RT_NONBLOCKING {
202 return g_realtimeAudioThreadDepth > 0;
203}
204
205inline bool reportRealtimeMisuse(const char* apiName) noexcept AESTRA_RT_NONBLOCKING {
206 if (!isRealtimeAudioThread()) {
207 return false;
208 }
209
210 if (auto handler = g_realtimeMisuseHandler.load(std::memory_order_acquire)) {
211 handler(apiName);
212 } else {
213#ifndef NDEBUG
214 assert(false && "Non-real-time API called from the audio thread");
215#endif
216 }
217
218 return true;
219}
