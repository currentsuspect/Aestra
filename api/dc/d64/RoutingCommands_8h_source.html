25class SetMainOutputCommand : public ICommand {
26public:
27 SetMainOutputCommand(TrackManager& manager, uint32_t channelId, uint32_t targetId)
28 : m_manager(manager), m_channelId(channelId), m_targetId(targetId) {}
29
30 void execute() override {
31 MixerChannel* channel = m_manager.getChannelById(m_channelId);
32 if (!channel) {
33 throw std::runtime_error("SetMainOutput: channel no longer exists");
34 }
35 if (!m_manager.canRouteTo(m_channelId, m_targetId)) {
36 throw std::runtime_error("SetMainOutput: routing loop or illegal destination");
37 }
38 m_previousId = channel->getMainOutputId();
39 channel->setMainOutputId(m_targetId);
40 m_manager.requestAudioGraphRebuild(GraphDirtyReason::RoutingChanged);
41 }
42 void undo() override {
43 if (m_previousId == 0u) {
44 return;
45 }
46 if (auto* channel = m_manager.getChannelById(m_channelId)) {
47 channel->setMainOutputId(m_previousId);
48 m_manager.requestAudioGraphRebuild(GraphDirtyReason::RoutingChanged);
49 }
50 }
51 void redo() override { execute(); }
52
53 std::string getName() const override { return "Reroute Track"; }
54 bool changesProjectState() const override { return true; }
55 std::string type() const override { return "set_main_output"; }
56
57private:
58 TrackManager& m_manager;
59 uint32_t m_channelId;
60 uint32_t m_targetId;
61 uint32_t m_previousId{0u};
62};

67class AddSendCommand : public ICommand {
68public:
69 AddSendCommand(TrackManager& manager, uint32_t channelId, const AudioRoute& route)
70 : m_manager(manager), m_channelId(channelId), m_route(route) {}
71
72 void execute() override {
73 MixerChannel* channel = m_manager.getChannelById(m_channelId);
74 if (!channel) {
75 throw std::runtime_error("AddSend: channel no longer exists");
76 }
77 if (m_route.targetChannelId == kMasterId) {
78 throw std::runtime_error("AddSend: sends to master are illegal (Contract D4)");
79 }
80 if (!m_manager.canRouteTo(m_channelId, m_route.targetChannelId)) {
81 throw std::runtime_error("AddSend: routing loop or illegal destination");
82 }
83 channel->addSend(m_route);
84 // Capture the minted stable id so undo removes the right send even
85 // after other sends were added/removed around it, and redo restores
86 // the SAME id (round trip, not a new identity wearing the old slot).
87 const auto sends = channel->getSends();
88 m_sendId = sends.empty() ? 0u : sends.back().sendId;
89 m_route.sendId = m_sendId;
90 m_manager.requestAudioGraphRebuild(GraphDirtyReason::RoutingChanged);
91 }
92 void undo() override {
93 if (m_sendId == 0u) {
94 return;
95 }
96 if (auto* channel = m_manager.getChannelById(m_channelId)) {
97 channel->removeSend(m_sendId);
98 m_manager.requestAudioGraphRebuild(GraphDirtyReason::RoutingChanged);
99 }
100 }
101 void redo() override { execute(); }
102
103 std::string getName() const override { return "Add Send"; }
104 bool changesProjectState() const override { return true; }
105 std::string type() const override { return "add_send"; }
106
107private:
108 TrackManager& m_manager;
109 uint32_t m_channelId;
110 AudioRoute m_route;
111 uint64_t m_sendId{0};
112};

120class RemoveSendCommand : public ICommand {
121public:
122 RemoveSendCommand(TrackManager& manager, uint32_t channelId, uint64_t sendId)
123 : m_manager(manager), m_channelId(channelId), m_sendId(sendId) {}
124
125 void execute() override {
126 MixerChannel* channel = m_manager.getChannelById(m_channelId);
127 if (!channel) {
128 throw std::runtime_error("RemoveSend: channel no longer exists");
129 }
130 m_sendIndex = channel->findSendIndex(m_sendId);
131 if (m_sendIndex < 0) {
132 throw std::runtime_error("RemoveSend: send no longer exists");
133 }
134 m_removedRoute = channel->getSends()[static_cast<size_t>(m_sendIndex)];
135 channel->removeSend(m_sendId);
136 m_manager.requestAudioGraphRebuild(GraphDirtyReason::RoutingChanged);
137 }
138 void undo() override {
139 if (m_sendIndex < 0) {
140 return;
141 }
142 if (auto* channel = m_manager.getChannelById(m_channelId)) {
143 channel->insertSend(m_sendIndex, m_removedRoute);
144 m_manager.requestAudioGraphRebuild(GraphDirtyReason::RoutingChanged);
145 }
146 }
147 void redo() override { execute(); }
148
149 std::string getName() const override { return "Remove Send"; }
150 bool changesProjectState() const override { return true; }
151 std::string type() const override { return "remove_send"; }
152
153private:
154 TrackManager& m_manager;
155 uint32_t m_channelId;
156 uint64_t m_sendId;
157 int m_sendIndex{-1};
158 AudioRoute m_removedRoute;
159};

169class EditSendCommand : public ICommand {
170public:
171 EditSendCommand(TrackManager& manager, uint32_t channelId, uint64_t sendId, const AudioRoute& newRoute)
172 : m_manager(manager), m_channelId(channelId), m_sendId(sendId), m_newRoute(newRoute) {}
173
174 void execute() override {
175 MixerChannel* channel = m_manager.getChannelById(m_channelId);
176 if (!channel) {
177 throw std::runtime_error("EditSend: channel no longer exists");
178 }
179 const int sendIndex = channel->findSendIndex(m_sendId);
180 if (sendIndex < 0) {
181 throw std::runtime_error("EditSend: send no longer exists");
182 }
183 m_previousRoute = channel->getSends()[static_cast<size_t>(sendIndex)];
184 if (m_previousRoute.targetChannelId != m_newRoute.targetChannelId) {
185 if (m_newRoute.targetChannelId == kMasterId) {
186 throw std::runtime_error("EditSend: sends to master are illegal (Contract D4)");
187 }
188 if (!m_manager.canRouteTo(m_channelId, m_newRoute.targetChannelId)) {
189 throw std::runtime_error("EditSend: routing loop or illegal destination");
190 }
191 }
192 channel->setSend(m_sendId, m_newRoute);
193 if (sendTopologyChanged()) {
194 // mute/sidechainOnly/postFader/destination all change the compiled
195 // topology (edges, sidechain buffers, PDC classification) — not
196 // just level/pan. Keep model and graph in step.
197 m_manager.requestAudioGraphRebuild(GraphDirtyReason::RoutingChanged);
198 }
199 }
200 void undo() override {
201 if (auto* channel = m_manager.getChannelById(m_channelId)) {
202 channel->setSend(m_sendId, m_previousRoute);
203 if (sendTopologyChanged()) {
204 m_manager.requestAudioGraphRebuild(GraphDirtyReason::RoutingChanged);
205 }
206 }
207 }
208 void redo() override { execute(); }
209
210 std::string getName() const override { return "Edit Send"; }
211 bool changesProjectState() const override { return true; }
212 std::string type() const override { return "edit_send"; }
213
214private:
215 bool sendTopologyChanged() const {
216 return m_previousRoute.targetChannelId != m_newRoute.targetChannelId ||
217 m_previousRoute.mute != m_newRoute.mute ||
218 m_previousRoute.sidechainOnly != m_newRoute.sidechainOnly ||
219 m_previousRoute.postFader != m_newRoute.postFader;
220 }
221
222 TrackManager& m_manager;
223 uint32_t m_channelId;
224 uint64_t m_sendId;
225 AudioRoute m_newRoute;
226 AudioRoute m_previousRoute;
227};
