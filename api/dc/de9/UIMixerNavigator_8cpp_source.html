40NUIRect UIMixerNavigator::tickRect(size_t index) const {

67void UIMixerNavigator::onRender(NUIRenderer& renderer) {
68 const auto b = getBounds();
69 if (b.isEmpty() || m_ticks.empty()) return;
70 auto& theme = NUIThemeManager::getInstance();
71 const NUIColor text = theme.getColor("textPrimary");
72 const NUIColor muted = theme.getColor("textSecondary");
73 const NUIColor accent = theme.getColor("accentPrimary");
74 const NUIColor safe = theme.getColor("meterSafe");
75 const NUIColor crit = theme.getColor("meterCrit");
76 const NUIColor solo = theme.getColor("soloed");
77
78 for (size_t i = 0; i < m_groups.size(); ++i) {
79 const auto r = groupLabelRect(i);
80 renderer.fillRect(NUIRect{r.x, r.bottom() - 1.0f, std::max(2.0f, r.width - 2.0f), 1.0f}, muted.withAlpha(0.35f));
81 std::string label = m_groups[i].label;
82 while (label.size() > 1 && renderer.measureText("\xE2\x86\x92 " + label, 9.0f).width > r.width - 4.0f) label.pop_back();
83 if (r.width > 20.0f) renderer.drawText("\xE2\x86\x92 " + label, {r.x + 1.0f, r.y}, 9.0f, muted.withAlpha(0.85f));
84 }
85
86 for (size_t i = 0; i < m_ticks.size(); ++i) {
87 const auto& t = m_ticks[i];
88 const auto r = tickRect(i);
89 renderer.fillRect(r, theme.getColor("border").withAlpha(0.22f));
90 const float lvl = std::clamp(t.level, 0.0f, 1.0f);
91 const float bodyH = r.height - 4.0f;
92 renderer.fillRect(NUIRect{r.x, r.bottom() - std::max(0.0f, bodyH * lvl), r.width, std::max(0.0f, bodyH * lvl)},
93 safe.withAlpha(0.55f));
94 renderer.fillRect(NUIRect{r.x, r.y, r.width, 3.0f}, t.color);
95 if (t.selected) renderer.strokeRect(r, 1.0f, text.withAlpha(0.7f));
96 // Flags stay visible however far the channel is from the bracket.
97 if (t.clipped) renderer.fillRect(NUIRect{r.x, r.bottom() - 3.0f, r.width, 3.0f}, crit);
98 else if (t.soloed) renderer.fillRect(NUIRect{r.x, r.bottom() - 3.0f, r.width, 3.0f}, solo);
99 }
100
101 const auto br = bracketRect();
102 renderer.fillRoundedRect(br, 3.0f, accent.withAlpha(0.10f));
103 renderer.strokeRoundedRect(br, 3.0f, 1.25f, accent.withAlpha(m_draggingBracket ? 0.95f : 0.65f));
104}

106bool UIMixerNavigator::onMouseEvent(const NUIMouseEvent& event) {
107 if (!isVisible() || !isEnabled() || m_ticks.empty()) return false;
108 const auto b = getBounds();
109
110 if (m_draggingBracket) {
111 if (event.released && event.button == NUIMouseButton::Left) {
112 m_draggingBracket = false;
113 repaint();
114 return true;
115 }
116 if (event.button == NUIMouseButton::None && onScrollRequested)
117 onScrollRequested(m_dragStartScroll + toContentDelta(event.position.x - m_dragStartX));
118 return true;
119 }
120 if (!b.contains(event.position)) return false;
121 if (!(event.pressed && event.button == NUIMouseButton::Left)) return false;
122
123 if (bracketRect().contains(event.position)) {
124 m_draggingBracket = true;
125 m_dragStartX = event.position.x;
126 m_dragStartScroll = m_scrollX;
127 repaint();
128 return true;
129 }
130 for (size_t i = 0; i < m_groups.size(); ++i) {
131 if (groupLabelRect(i).contains(event.position)) {
132 if (onScrollRequested) onScrollRequested(m_groups[i].x);
133 return true;
134 }
135 }
136 for (size_t i = 0; i < m_ticks.size(); ++i) {
137 auto r = tickRect(i);
138 r.x -= 0.5f;
139 r.width += 1.0f; // the gap between two ticks belongs to the nearer one
140 if (r.contains(event.position)) {
141 if (onScrollRequested) onScrollRequested(scrollToCentre(m_ticks[i].x, m_ticks[i].w));
142 if (onTickClicked) onTickClicked(m_ticks[i].channelId);
143 return true;
144 }
145 }
146 return true; // empty space inside the navigator is still the navigator's
147}
