21bool isSafeSamplerStatePath(const std::string& path) {
22 if (path.empty() || path.find("://") != std::string::npos) {
23 return false;
24 }
25
26 // Block POSIX absolute paths, Windows drive paths, root-relative Windows
27 // paths, and UNC/network paths before filesystem normalization can vary by
28 // platform.
29 if (path[0] == '/' || path[0] == '\\') {
30 return false;
31 }
32 if (path.size() >= 2 && std::isalpha(static_cast<unsigned char>(path[0])) && path[1] == ':') {
33 return false;
34 }
35 if (path.rfind("\\\\", 0) == 0 || path.rfind("//", 0) == 0) {
36 return false;
37 }
38
39 std::string normalized = path;
40 std::replace(normalized.begin(), normalized.end(), '\\', '/');
41 std::filesystem::path fsPath(normalized);
42 if (fsPath.is_absolute()) {
43 return false;
44 }
45 for (const auto& part : fsPath) {
46 if (part == "..") {
47 return false;
48 }
49 }
50 return true;
51}

160bool SamplerPlugin::loadSampleData(const std::string& path, std::vector<float> data, uint32_t rate, uint32_t channels) {
161 if (data.empty() || rate == 0 || channels == 0) {
162 return false;
163 }
164
165 // Decode output is interleaved. Keep stereo data in-place and only upmix
166 // mono defensively for callers that bypass decodeAudioFile's forceStereo().
167 const size_t numFrames = data.size() / std::max<uint32_t>(channels, 1);
168 if (channels == 1) {
169 std::vector<float> interleaved(numFrames * 2);
170 for (size_t i = 0; i < numFrames; ++i) {
171 interleaved[i * 2] = data[i];
172 interleaved[i * 2 + 1] = data[i];
173 }
174 data = std::move(interleaved);
175 } else if (channels != 2) {
176 return false;
177 }
178
179 // Prepare new data container
180 auto newData = std::make_shared<SampleData>();
181 newData->data = std::move(data);
182 newData->rate = rate;
183 newData->channels = 2;
184 newData->path = path;
185
186 // Atomic Swap (Thread-Safe, Lock-Free-ish)
187 // std::atomic_exchange uses standard atomics for shared_ptr
188 auto oldData = m_data.exchange(newData, std::memory_order_acq_rel);
189
190 // Safely dispose of old data via Garbage Collector (avoids delete on Audio Thread)
191 GarbageCollector::instance().release(oldData, "SamplerPlugin::SampleData");
192
193 return true;
194}

196bool SamplerPlugin::normalizeSample(float targetPeak) {
197 auto currentData = m_data.load(std::memory_order_acquire);
198 if (!currentData || currentData->data.empty()) {
199 return false;
200 }
201
202 float peak = 0.0f;
203 for (float s : currentData->data) {
204 peak = std::max(peak, std::abs(s));
205 }
206 if (peak <= 0.0001f) {
207 return false;
208 }
209
210 const float gain = std::clamp(targetPeak, 0.1f, 1.0f) / peak;
211 auto edited = std::make_shared<SampleData>(*currentData);
212 for (auto& s : edited->data) {
213 s *= gain;
214 }
215
216 auto oldData = m_data.exchange(edited, std::memory_order_acq_rel);
217 GarbageCollector::instance().release(oldData, "SamplerPlugin::SampleData");
218 return true;
219}

221bool SamplerPlugin::reverseSample() {
222 auto currentData = m_data.load(std::memory_order_acquire);
223 if (!currentData || currentData->data.empty() || currentData->channels == 0) {
224 return false;
225 }
226
227 const uint32_t channels = currentData->channels;
228 const size_t frames = currentData->data.size() / channels;
229 if (frames < 2) {
230 return false;
231 }
232
233 auto edited = std::make_shared<SampleData>(*currentData);
234 auto& d = edited->data;
235 for (size_t i = 0; i < frames / 2; ++i) {
236 const size_t j = frames - 1 - i;
237 for (uint32_t ch = 0; ch < channels; ++ch) {
238 std::swap(d[i * channels + ch], d[j * channels + ch]);
239 }
240 }
241
242 auto oldData = m_data.exchange(edited, std::memory_order_acq_rel);
243 GarbageCollector::instance().release(oldData, "SamplerPlugin::SampleData");
244 return true;
245}

247void SamplerPlugin::process(const float* const* inputs, float** outputs, uint32_t numInputChannels,
248 uint32_t numOutputChannels, uint32_t numFrames, const MidiBuffer* midiInput,
249 MidiBuffer* midiOutput) {
250 // Clear Outputs
251 for (uint32_t c = 0; c < numOutputChannels; ++c) {
252 std::memset(outputs[c], 0, numFrames * sizeof(float));
253 }
254
255 // RT-safe hard reset (requested by engine on transport restart)
256 if (m_resetVoicesRequested.exchange(false, std::memory_order_acq_rel)) {
257 for (auto& v : m_voices) {
258 v.active = false;
259 v.stage = EnvStage::Off;
260 v.position = 0.0;
261 v.playbackRate = 1.0;
262 v.targetPlaybackRate = 1.0;
263 v.glideActive = false;
264 v.stageTime = 0.0;
265 v.currentGain = 0.0f;
266 v.releaseGain = 0.0f;
267 }
268 }
269
270 // Thread-safe access to sample data
271 auto currentData = m_data.load(std::memory_order_acquire);
272 if (!currentData || currentData->data.empty())
273 return;
274
275 // Source Rate correction
276 double baseRate = (double)currentData->rate / m_sampleRate;
277
278 // Preload per-block constants from atomics (UI thread writes, audio thread reads)
279 // These change rarely — loading once eliminates per-voice per-sample atomic traffic
280 const float attackMs = std::max(m_params[kParamAttack].load(std::memory_order_relaxed), 0.001f);
281 const float decayMs = std::max(m_params[kParamDecay].load(std::memory_order_relaxed), 0.001f);
282 const float sustainLevel = m_params[kParamSustain].load(std::memory_order_relaxed);
283 const float releaseMs = std::max(m_params[kParamRelease].load(std::memory_order_relaxed), 0.001f);
284 const float cachedGlideTimeMs = m_glideTimeMs.load(std::memory_order_relaxed);
285 const double invSampleRate = 1.0 / m_sampleRate;
286
287 // Events
288 uint32_t eventIdx = 0;
289 size_t eventCount = midiInput ? midiInput->getEventCount() : 0;
290
291 const auto& sampleVec = currentData->data;
292 // Data is now interleaved stereo (2 channels packed as L,R,L,R...)
293 const double totalFrames = static_cast<double>(sampleVec.size() / 2);
294 if (totalFrames <= 1.0) {
295 return;
296 }
297 const float startNorm = std::clamp(m_loopStartNorm.load(std::memory_order_relaxed), 0.0f, 0.999f);
298 const float endNorm = std::clamp(m_loopEndNorm.load(std::memory_order_relaxed), startNorm + 0.001f, 1.0f);
299 const double startFrame = startNorm * (totalFrames - 1.0);
300 const double endFrame = std::max(startFrame + 1.0, endNorm * totalFrames);
301 const double loopLength = std::max(1.0, endFrame - startFrame);
302 const bool loopEnabled = m_loopEnabled.load(std::memory_order_relaxed);
303 int activeVoiceCount = 0;
304 for (const auto& v : m_voices) {
305 if (v.active) {
306 ++activeVoiceCount;
307 }
308 }
309 bool activeVoiceCountDirty = false;
310
311 for (uint32_t i = 0; i < numFrames; ++i) {
312 // Handle MIDI
313 while (midiInput && eventIdx < eventCount) {
314 const auto& e = midiInput->getEvent(eventIdx);
315 if (e.sampleOffset == i) {
316 handleMidiEvent(e, baseRate, currentData);
317 activeVoiceCountDirty = true;
318 eventIdx++;
319 } else if (e.sampleOffset < i) {
320 eventIdx++; // Skip past
321 } else {
322 break; // Future
323 }
324 }
325
326 // Render Voices
327 float L = 0.0f;
328 float R = 0.0f;
329 if (activeVoiceCountDirty) {
330 activeVoiceCount = 0;
331 for (const auto& v : m_voices) {
332 if (v.active) {
333 ++activeVoiceCount;
334 }
335 }
336 activeVoiceCountDirty = false;
337 }
338 const float voiceComp = 1.0f / std::sqrt(static_cast<float>(std::max(1, activeVoiceCount)));
339
340 for (auto& v : m_voices) {
341 if (!v.active)
342 continue;
343
344 if (v.glideActive) {
345 if (cachedGlideTimeMs <= 0.0f) {
346 v.playbackRate = v.targetPlaybackRate;
347 v.glideActive = false;
348 } else {
349 const double glideSamples = std::max(1.0, (static_cast<double>(cachedGlideTimeMs) * m_sampleRate) / 1000.0);
350 const double glideCoeff = 1.0 - std::exp(-1.0 / glideSamples);
351 v.playbackRate += (v.targetPlaybackRate - v.playbackRate) * glideCoeff;
352 if (std::abs(v.targetPlaybackRate - v.playbackRate) < 1.0e-5) {
353 v.playbackRate = v.targetPlaybackRate;
354 v.glideActive = false;
355 }
356 }
357 }
358
359 // One-shot end-of-sample fade (#452). No note-off may ever arrive
360 // in one-shot mode, so fade toward the endpoint with a bounded,
361 // stateless multiplier on top of the normal ADSR — never by
362 // mutating the ADSR stage (the old code entered Release at
363 // trigger time for samples shorter than the release, capturing
364 // releaseGain from a still-zero attack: permanent silence).
365 //
366 // Units: v.position/endFrame are SOURCE frames; the configured
367 // release is OUTPUT time. Convert via |playbackRate|, and clamp
368 // the effective fade to the playable one-shot duration so short
369 // samples fade across what they have instead of starting at zero.
370 float oneShotTail = 1.0f;
371 if (!loopEnabled) {
372 const double absRate = std::abs(v.playbackRate);
373 if (absRate > 1.0e-9) {
374 const double remainingOutputFrames = (endFrame - v.position) / absRate;
375 const double configuredReleaseFrames = static_cast<double>(releaseMs) * m_sampleRate;
376 const double playableOutputFrames = (endFrame - startFrame) / absRate;
377 const double effectiveTailFrames =
378 std::max(1.0, std::min(configuredReleaseFrames, playableOutputFrames));
379 oneShotTail = static_cast<float>(std::clamp(remainingOutputFrames / effectiveTailFrames, 0.0, 1.0));
380 }
381 }
382
383 // Envelope (uses preloaded ADSR — no atomic loads)
384 float env = getEnvelopeLevel(v, invSampleRate, attackMs, decayMs, sustainLevel, releaseMs) * oneShotTail;
385 if (v.stage == EnvStage::Off) {
386 v.active = false;
387 activeVoiceCountDirty = true;
388 continue;
389 }
390
391 // Sample Lookup
392 if (v.position < startFrame) {
393 v.position = startFrame;
394 }
395 if (v.position >= endFrame) {
396 if (loopEnabled) {
397 const double offset = v.position - startFrame;
398 const double wraps = std::floor(offset / loopLength);
399 v.position = startFrame + offset - wraps * loopLength;
400 } else {
401 v.active = false;
402 activeVoiceCountDirty = true;
403 continue;
404 }
405 }
406 if (v.position >= totalFrames) {
407 v.active = false;
408 activeVoiceCountDirty = true;
409 continue;
410 }
411
412 // Sinc64Turbo Interpolation (data is now interleaved)
413 float outL = 0.0f, outR = 0.0f;
414 if (v.position >= 0.0 && v.position < static_cast<double>(totalFrames)) {
415 Interpolators::Sinc64Turbo::interpolate(
416 sampleVec.data(), // interleaved stereo [L,R,L,R...]
417 totalFrames, // total frames (not samples)
418 v.position, // fractional position
419 outL, outR // output
420 );
421 }
422
423 float gain = (v.velocity * v.velocity) * env * kSamplerOutputHeadroom * voiceComp;
424 L += outL * gain * v.gainL;
425 R += outR * gain * v.gainR;
426
427 // Advance
428 v.position += v.playbackRate;
429 }
430
431 outputs[0][i] = L;
432 outputs[1][i] = R;
433 }
434}

436void SamplerPlugin::handleMidiEvent(const MidiBuffer::Event& event, double baseRate,
437 const std::shared_ptr<SampleData>& currentData) {
438 uint8_t status = event.data[0] & 0xF0;
439 uint8_t note = event.data[1];
440 uint8_t velocity = event.data[2];
441
442 if (status == 0xA0) { // Per-note pan, sent just before the matching note-on
443 m_pendingNotePan[note & 0x7F] = velocity; // 1..127, 64 = centre
444 return;
445 }
446
447 if (status == 0x90 && velocity > 0) { // Note On
448 // Consume the pending pan for this note (0 = none → centre) and
449 // re-arm the slot so later notes without pan stay centred. Unity at
450 // centre keeps unpanned projects bit-identical to before.
451 float noteGainL = 1.0f;
452 float noteGainR = 1.0f;
453 {
454 const uint8_t panByte = m_pendingNotePan[note & 0x7F];
455 m_pendingNotePan[note & 0x7F] = 0;
456 if (panByte != 0 && panByte != 64) {
457 const float pan = std::clamp((static_cast<float>(panByte) - 64.0f) / 63.0f, -1.0f, 1.0f);
458 noteGainL = std::min(1.0f, 1.0f - pan);
459 noteGainR = std::min(1.0f, 1.0f + pan);
460 }
461 }
462 const float pitchParam = m_params[kParamPitch].load(std::memory_order_relaxed);
463 const float globalSemitones = (pitchParam - 0.5f) * 24.0f + (m_fineTuneCents.load(std::memory_order_relaxed) / 100.0f);
464 const int rootMidiNote = m_rootMidiNote.load(std::memory_order_relaxed);
465 const float noteSemitones = static_cast<float>(note) - static_cast<float>(rootMidiNote);
466 const float ratio = std::pow(2.0f, (globalSemitones + noteSemitones) / 12.0f);
467 const double targetRate = baseRate * ratio;
468
469 if (m_monoMode.load(std::memory_order_relaxed)) {
470 double noteStartFrame = 0.0;
471 if (currentData && currentData->channels > 0) {
472 const double totalFrames = static_cast<double>(currentData->data.size() / currentData->channels);
473 noteStartFrame = std::clamp(static_cast<double>(m_loopStartNorm.load(std::memory_order_relaxed)), 0.0, 0.999) *
474 std::max(1.0, totalFrames - 1.0);
475 }
476
477 auto& voice = m_voices[0];
478 const bool legato = voice.active && voice.stage != EnvStage::Release && voice.stage != EnvStage::Off;
479
480 voice.active = true;
481 voice.note = note;
482 voice.velocity = velocity / 127.0f;
483 voice.gainL = noteGainL;
484 voice.gainR = noteGainR;
485
486 if (legato) {
487 voice.targetPlaybackRate = targetRate;
488 const float glideTimeMs = m_glideTimeMs.load(std::memory_order_relaxed);
489 if (glideTimeMs <= 0.0f) {
490 voice.playbackRate = targetRate;
491 voice.targetPlaybackRate = targetRate;
492 voice.glideActive = false;
493 } else {
494 voice.glideActive = std::abs(voice.targetPlaybackRate - voice.playbackRate) > 1.0e-5;
495 }
496 } else {
497 voice.position = noteStartFrame;
498 voice.playbackRate = targetRate;
499 voice.targetPlaybackRate = targetRate;
500 voice.glideActive = false;
501 voice.stage = EnvStage::Attack;
502 voice.stageTime = 0.0;
503 voice.currentGain = 0.0f;
504 }
505 return;
506 }
507
508 const int maxVoices = std::clamp(m_maxVoices.load(std::memory_order_relaxed), 1, kMaxVoices);
509 double noteStartFrame = 0.0;
510 if (currentData && currentData->channels > 0) {
511 const double totalFrames = static_cast<double>(currentData->data.size() / currentData->channels);
512 noteStartFrame = std::clamp(static_cast<double>(m_loopStartNorm.load(std::memory_order_relaxed)), 0.0, 0.999) *
513 std::max(1.0, totalFrames - 1.0);
514 }
515 // Find free voice
516 Voice* freeVoice = nullptr;
517 Voice* oldestRelease = nullptr;
518 Voice* oldestActive = nullptr;
519 double maxReleaseTime = -1.0;
520 double maxActiveTime = -1.0;
521
522 for (int i = 0; i < maxVoices; ++i) {
523 auto& v = m_voices[static_cast<size_t>(i)];
524 if (!v.active) {
525 freeVoice = &v;
526 break;
527 }
528 if (v.stage == EnvStage::Release) {
529 if (v.stageTime > maxReleaseTime) {
530 maxReleaseTime = v.stageTime;
531 oldestRelease = &v;
532 }
533 } else {
534 if (v.position > maxActiveTime) {
535 maxActiveTime = v.position;
536 oldestActive = &v;
537 }
538 }
539 }
540
541 if (!freeVoice) {
542 freeVoice = oldestRelease ? oldestRelease : oldestActive;
543 if (!freeVoice)
544 freeVoice = &m_voices[0]; // Fallback
545 }
546
547 freeVoice->active = true;
548 freeVoice->note = note;
549 freeVoice->velocity = velocity / 127.0f;
550 freeVoice->gainL = noteGainL;
551 freeVoice->gainR = noteGainR;
552 freeVoice->position = noteStartFrame;
553 freeVoice->playbackRate = targetRate;
554 freeVoice->targetPlaybackRate = targetRate;
555 freeVoice->glideActive = false;
556 freeVoice->stage = EnvStage::Attack;
557 freeVoice->stageTime = 0.0;
558 freeVoice->currentGain = 0.0f;
559 } else if (status == 0x80 || (status == 0x90 && velocity == 0)) { // Note Off
560 // ADSR-driven one-shot: note-off enters release so envelope fully shapes each hit.
561 for (auto& v : m_voices) {
562 if (v.active && v.note == note && v.stage != EnvStage::Release) {
563 v.stage = EnvStage::Release;
564 v.stageTime = 0.0;
565 v.releaseGain = v.currentGain;
566 }
567 }
568 }
569}

571float SamplerPlugin::getEnvelopeLevel(Voice& v, double dt, float attack, float decay, float sustain, float release) {
572 v.stageTime += dt;
573
574 switch (v.stage) {
575 case EnvStage::Attack:
576 if (v.stageTime >= attack) {
577 v.stage = EnvStage::Decay;
578 v.stageTime = 0.0;
579 v.currentGain = 1.0f;
580 } else {
581 v.currentGain = static_cast<float>(v.stageTime / attack);
582 }
583 break;
584 case EnvStage::Decay:
585 if (v.stageTime >= decay) {
586 v.stage = EnvStage::Sustain;
587 v.stageTime = 0.0;
588 v.currentGain = sustain;
589 } else {
590 v.currentGain = 1.0f - static_cast<float>(v.stageTime / decay) * (1.0f - sustain);
591 }
592 break;
593 case EnvStage::Sustain:
594 v.currentGain = sustain;
595 break;
596 case EnvStage::Release:
597 if (v.stageTime >= release) {
598 v.stage = EnvStage::Off;
599 v.currentGain = 0.0f;
600 } else {
601 v.currentGain = v.releaseGain * static_cast<float>(1.0 - v.stageTime / release);
602 }
603 break;
604 case EnvStage::Off:
605 v.currentGain = 0.0f;
606 break;
607 }
608 return v.currentGain;
609}

615std::vector<PluginParameter> SamplerPlugin::getParameters() const {
616 std::vector<PluginParameter> params;
617 params.push_back({kParamAttack, "Attack", "Att", "s", 0.01f, 0.0f, 2.0f});
618 params.push_back({kParamDecay, "Decay", "Dec", "s", 0.1f, 0.0f, 2.0f});

647std::vector<uint8_t> SamplerPlugin::saveState() const {
648 Aestra::JSON json = Aestra::JSON::object();
649
650 // Params
651 Aestra::JSON pArr = Aestra::JSON::array();
652 for (const auto& p : m_params) {
653 pArr.push(Aestra::JSON(static_cast<double>(p.load())));
654 }
655 json.set("params", pArr);
656 json.set("fineTuneCents", Aestra::JSON(static_cast<double>(m_fineTuneCents.load(std::memory_order_relaxed))));
657 json.set("loopStartNorm", Aestra::JSON(static_cast<double>(m_loopStartNorm.load(std::memory_order_relaxed))));
658 json.set("loopEndNorm", Aestra::JSON(static_cast<double>(m_loopEndNorm.load(std::memory_order_relaxed))));
659 json.set("loopEnabled", Aestra::JSON(m_loopEnabled.load(std::memory_order_relaxed)));
660 json.set("maxVoices", Aestra::JSON(static_cast<double>(m_maxVoices.load(std::memory_order_relaxed))));
661 json.set("rootMidiNote", Aestra::JSON(static_cast<double>(m_rootMidiNote.load(std::memory_order_relaxed))));
662 json.set("monoMode", Aestra::JSON(m_monoMode.load(std::memory_order_relaxed)));
663 json.set("glideTimeMs", Aestra::JSON(static_cast<double>(m_glideTimeMs.load(std::memory_order_relaxed))));
664
665 // Sample Path
666 {
667 auto current = m_data.load(std::memory_order_acquire);
668 if (current && !current->path.empty()) {
669 json.set("samplePath", Aestra::JSON(current->path));
670 }
671 }
672
673 std::string s = json.toString();
674 return std::vector<uint8_t>(s.begin(), s.end());

677bool SamplerPlugin::loadState(const std::vector<uint8_t>& state) {
678 if (state.empty())
679 return false;
680 std::string s(state.begin(), state.end());
681
682 Aestra::JSON json = Aestra::JSON::parse(s);
683 if (!json.isObject())
684 return false;
685
686 if (json["params"].isArray()) {
687 const auto& pArr = json["params"];
688 for (size_t i = 0; i < pArr.size() && i < kParamCount; ++i) {
689 m_params[i].store(static_cast<float>(pArr[i].asNumber()));
690 }
691 }
692
693 if (json.has("fineTuneCents")) {
694 m_fineTuneCents.store(std::clamp(static_cast<float>(json["fineTuneCents"].asNumber()), -100.0f, 100.0f),
695 std::memory_order_relaxed);
696 }
697 if (json.has("loopStartNorm") || json.has("loopEndNorm")) {
698 const float startNorm = json.has("loopStartNorm") ? static_cast<float>(json["loopStartNorm"].asNumber()) : 0.0f;
699 const float endNorm = json.has("loopEndNorm") ? static_cast<float>(json["loopEndNorm"].asNumber()) : 1.0f;
700 setSampleWindow(startNorm, endNorm);
701 }
702 if (json.has("loopEnabled")) {
703 m_loopEnabled.store(json["loopEnabled"].asBool(), std::memory_order_relaxed);
704 }
705 if (json.has("maxVoices")) {
706 setMaxVoices(static_cast<int>(json["maxVoices"].asNumber()));
707 }
708 if (json.has("rootMidiNote")) {
709 setRootMidiNote(static_cast<int>(json["rootMidiNote"].asNumber()));
710 }
711 if (json.has("monoMode")) {
712 setMonoMode(json["monoMode"].asBool());
713 }
714 if (json.has("glideTimeMs")) {
715 setGlideTimeMs(static_cast<float>(json["glideTimeMs"].asNumber()));
716 }
717
718 if (json.has("samplePath")) {
719 std::string path = json["samplePath"].asString();
720 if (!path.empty()) {
721 if (!isSafeSamplerStatePath(path)) {
722 return false;
723 }
724 loadSample(path);
725 }
726 }
727 return true;
728}
