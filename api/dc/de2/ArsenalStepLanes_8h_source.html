27 enum class Lane { Velocity, Pitch, Length, Chance, Ratchet };

29 static constexpr float kLaneH = 30.0f;
30 static constexpr float kLaneGap = 4.0f;
31 static constexpr float kPad = 8.0f;
32 static constexpr float kStepBeats = 0.25f;
33 static constexpr int kPitchRange = 12;
34 static constexpr float kGateMin = 0.1f, kGateMax = 2.0f;
35 static constexpr float kVelMin = 0.05f;
36
37 static constexpr int kRatchetMax = 4;
38
39 static std::vector<Lane> lanesFor(Audio::UnitType type) {
40 if (type == Audio::UnitType::Audio) return {};
41 if (type == Audio::UnitType::PitchedSampler)
42 return {Lane::Velocity, Lane::Pitch, Lane::Length, Lane::Chance, Lane::Ratchet};
43 return {Lane::Velocity, Lane::Chance, Lane::Ratchet};
44 }
45 static const char* label(Lane l) {
46 switch (l) {
47 case Lane::Velocity: return "VEL";
48 case Lane::Pitch: return "PITCH";
49 case Lane::Length: return "LENGTH";
50 case Lane::Chance: return "CHANCE";
51 case Lane::Ratchet: return "RATCHET";
52 }
53 return "";
54 }
55 static float heightFor(size_t laneCount) {
56 return laneCount == 0 ? 0.0f : kPad * 2.0f + static_cast<float>(laneCount) * kLaneH + static_cast<float>(laneCount - 1) * kLaneGap;
57 }

60 static float toFraction(Lane l, float v) {
61 switch (l) {
62 case Lane::Velocity: return std::clamp(v, 0.0f, 1.0f);
63 case Lane::Pitch: return (std::clamp(v, -static_cast<float>(kPitchRange), static_cast<float>(kPitchRange)) + kPitchRange) / (2.0f * kPitchRange);
64 case Lane::Length: return (std::clamp(v, kGateMin, kGateMax) - kGateMin) / (kGateMax - kGateMin);
65 case Lane::Chance: return std::clamp(v, 0.0f, 1.0f);
66 case Lane::Ratchet: return (std::clamp(v, 1.0f, static_cast<float>(kRatchetMax)) - 1.0f) / (kRatchetMax - 1);
67 }
68 return 0.0f;
69 }
71 static float fromFraction(Lane l, float f) {
72 f = std::clamp(f, 0.0f, 1.0f);
73 switch (l) {
74 case Lane::Velocity: return std::max(kVelMin, f);
75 case Lane::Pitch: return std::round(f * 2.0f * kPitchRange - kPitchRange);
76 case Lane::Length: return kGateMin + f * (kGateMax - kGateMin);
77 case Lane::Chance: return f;
78 case Lane::Ratchet: return std::round(1.0f + f * (kRatchetMax - 1));
79 }
80 return 0.0f;
81 }

85 static float readValue(const Audio::MidiNote& n, Lane l, int rootMidi) {
86 switch (l) {
87 case Lane::Velocity: return n.velocity;
88 case Lane::Pitch: return static_cast<float>(n.pitchOffset == 0 && n.pitch > 0 && n.pitch != rootMidi ? n.pitch - rootMidi : n.pitchOffset);
89 case Lane::Length: return n.gate;
90 case Lane::Chance: return n.chance;
91 case Lane::Ratchet: return static_cast<float>(n.ratchet);
92 }
93 return 0.0f;
94 }
95 static void writeValue(Audio::MidiNote& n, Lane l, float v, int rootMidi) {
96 switch (l) {
97 case Lane::Velocity: n.velocity = std::clamp(v, kVelMin, 1.0f); break;
98 case Lane::Pitch: {
99 const int semis = std::clamp(static_cast<int>(std::lround(v)), -kPitchRange, kPitchRange);
100 n.pitchOffset = static_cast<int8_t>(semis);
101 n.pitch = std::clamp(rootMidi + semis, 0, 127);
102 break;
103 }
104 case Lane::Length: n.gate = std::clamp(v, kGateMin, kGateMax); break;
105 case Lane::Chance: n.chance = std::clamp(v, 0.0f, 1.0f); break;
106 case Lane::Ratchet: n.ratchet = static_cast<uint8_t>(std::clamp(static_cast<int>(std::lround(v)), 1, kRatchetMax)); break;
107 }
108 }

111 static int stepAtX(float x, float stripX, float stripW, float stepW, float groupGap, float scrollX, int steps) {
112 const float rel = x - stripX + scrollX;
113 if (stepW <= 0.0f || x < stripX || x >= stripX + stripW || rel < 0.0f) return -1;
114 for (int s = 0; s < steps; ++s) {
115 const float left = static_cast<float>(s) * stepW + static_cast<float>(s / 4) * groupGap;
116 if (rel >= left && rel < left + stepW) return s;
117 }
118 return -1;
119 }
120 static float stepCentreX(int step, float stripX, float stepW, float groupGap, float scrollX) {
121 return stripX + static_cast<float>(step) * stepW + static_cast<float>(step / 4) * groupGap - scrollX + stepW * 0.5f;
122 }

139 void draw(AestraUI::NUIRenderer& renderer, const Context& c) const {
140 const auto lanes = lanesFor(c.type);
141 if (lanes.empty() || !c.tm) return;
142 auto& theme = AestraUI::NUIThemeManager::getInstance();
143 const auto& tp = theme.getCurrentTheme();
144 const auto notes = notesOf(c);
145 renderer.fillRoundedRect(c.area, tp.radiusS, theme.getColor("backgroundPrimary").withAlpha(0.9f));
146 for (size_t i = 0; i < lanes.size(); ++i) {
147 const auto r = laneRect(c, i);
148 renderer.drawText(label(lanes[i]), {c.area.x + 12.0f, r.y + r.height * 0.5f - 5.0f}, tp.fontSizeMicro, theme.getColor("textSecondary").withAlpha(0.8f));
149 ArsenalClip::within(renderer, {c.strip.x, r.y, c.strip.width, r.height}, c.clip);
150 renderer.fillRect({c.strip.x, r.y + r.height * 0.5f, c.strip.width, 1.0f}, theme.getColor("border").withAlpha(lanes[i] == Lane::Pitch ? 0.5f : 0.0f));
151 for (int s = 0; s < c.steps; ++s) {
152 const Audio::MidiNote* n = noteAt(notes, c.unit, s);
153 const float cx = stepCentreX(s, c.strip.x, c.stepW, c.groupGap, c.scrollX);
154 const float bw = std::max(2.0f, c.stepW - 3.0f);
155 const AestraUI::NUIRect cell{cx - bw * 0.5f, r.y, bw, r.height};
156 renderer.fillRect(cell, theme.getColor("surfaceTertiary").withAlpha(0.35f));
157 if (!n) continue;
158 const float f = toFraction(lanes[i], readValue(*n, lanes[i], c.root));
159 const bool centred = lanes[i] == Lane::Pitch;
160 const float top = r.y + r.height * (1.0f - f);
161 const float base = centred ? r.y + r.height * 0.5f : r.bottom();
162 const AestraUI::NUIRect bar{cell.x, std::min(top, base), bw, std::max(2.0f, std::abs(base - top))};
163 renderer.fillRect(bar, c.accent);
164 // The value, when it is not the default and the column is wide enough to read it (the mock's +3, 50%, x3).
165 const float v = readValue(*n, lanes[i], c.root);
166 const bool shown = lanes[i] == Lane::Pitch ? v != 0.0f : lanes[i] == Lane::Length ? std::abs(v - 1.0f) > 0.01f : lanes[i] == Lane::Chance ? v < 0.995f : v > 1.0f;
167 if (shown && bw >= 18.0f) {
168 const std::string txt = lanes[i] == Lane::Pitch ? (v > 0 ? "+" : "") + std::to_string(static_cast<int>(v)) : lanes[i] == Lane::Ratchet ? "x" + std::to_string(static_cast<int>(v)) : std::to_string(static_cast<int>(std::lround(v * 100.0f))) + "%";
169 renderer.drawTextCentered(txt, {cell.x, std::max(r.y, bar.y - 12.0f), bw, 11.0f}, tp.fontSizeMicro, theme.getColor("textSecondary"));
170 }
171 }
172 ArsenalClip::restore(renderer, c.clip);
173 }

177 bool press(const Context& c, const AestraUI::NUIPoint& p) {
178 const auto lanes = lanesFor(c.type);
179 for (size_t i = 0; i < lanes.size(); ++i) {
180 if (!laneRect(c, i).contains(p) || p.x < c.strip.x || p.x >= c.strip.x + c.strip.width) continue;
181 m_lane = static_cast<int>(i);
182 m_before = notesOf(c);
183 m_changed = false;
184 paint(c, p);
185 return true;
186 }
187 return false;
188 }
189 bool dragging() const { return m_lane >= 0; }
191 bool takeEdited() { const bool e = m_edited; m_edited = false; return e; }
192 void drag(const Context& c, const AestraUI::NUIPoint& p) { if (m_lane >= 0) paint(c, p); }
194 void release(const Context& c) {
195 if (m_lane < 0) return;
196 if (m_changed && c.tm && c.pattern.isValid()) {
197 auto after = notesOf(c);
198 c.tm->getCommandHistory().pushExecuted(std::make_shared<Audio::EditPatternNotesCommand>(
199 c.tm->getPatternManager(), c.pattern, std::move(m_before), std::move(after), "Edit Step Lane"));
200 }
201 m_lane = -1;
202 m_before.clear();
203 }

205 static AestraUI::NUIRect laneRect(const Context& c, size_t index) {
206 return {c.area.x, c.area.y + kPad + static_cast<float>(index) * (kLaneH + kLaneGap), c.area.width, kLaneH};
207 }
208 static std::vector<Audio::MidiNote> notesOf(const Context& c) {
209 if (!c.tm || !c.pattern.isValid()) return {};
210 const auto* p = c.tm->getPatternManager().getPattern(c.pattern);
211 if (!p || !p->isMidi()) return {};
212 return std::get<Audio::MidiPayload>(p->payload).notes;
213 }
214 static const Audio::MidiNote* noteAt(const std::vector<Audio::MidiNote>& notes, Audio::UnitID unit, int step) {
215 for (const auto& n : notes) {
216 if (n.unitId == unit && static_cast<int>(std::floor(n.startBeat / kStepBeats + 0.0001)) == step) return &n;
217 }
218 return nullptr;
219 }
220
221private:
222 void paint(const Context& c, const AestraUI::NUIPoint& p) {
223 const auto lanes = lanesFor(c.type);
224 if (m_lane < 0 || static_cast<size_t>(m_lane) >= lanes.size() || !c.tm || !c.pattern.isValid()) return;
225 const int step = stepAtX(p.x, c.strip.x, c.strip.width, c.stepW, c.groupGap, c.scrollX, c.steps);
226 if (step < 0) return;
227 const auto r = laneRect(c, static_cast<size_t>(m_lane));
228 const Lane lane = lanes[static_cast<size_t>(m_lane)];
229 const float v = fromFraction(lane, 1.0f - (p.y - r.y) / r.height);
230 c.tm->getPatternManager().applyPatch(c.pattern, [&](Audio::PatternSource& ps) {
231 if (!ps.isMidi()) return;
232 for (auto& n : std::get<Audio::MidiPayload>(ps.payload).notes) {
233 if (n.unitId == c.unit && static_cast<int>(std::floor(n.startBeat / kStepBeats + 0.0001)) == step) {
234 writeValue(n, lane, v, c.root);
235 m_changed = true;
236 m_edited = true;
237 return;
238 }
239 }
240 });
241 }
242
243 int m_lane{-1};
244 bool m_changed{false};
245 bool m_edited{false};
246 std::vector<Audio::MidiNote> m_before;
247};
