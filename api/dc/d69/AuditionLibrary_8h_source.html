39inline uint32_t seedOf(const std::string& s) {
40 uint32_t h = 2166136261u;
41 for (unsigned char c : s) { h ^= c; h *= 16777619u; }
42 return h;
43}

46inline std::vector<Item> versionItems(const std::string& projectName, const std::string& projectPath,
47 const TakeManager::Manifest& manifest, uint64_t nowMs) {
48 std::vector<Item> out;
49 if (!manifest.ok) return out;
50 // Only versions that were bounced are listed, and they are numbered among themselves, oldest = v1.
51 struct Bounced { const TakeManager::TakeEntry* version; std::string wav; VersionBounceInfo info; };
52 std::vector<Bounced> bounced;
53 for (const auto& v : manifest.takes) {
54 if (v.kind == TakeManager::TakeEntry::Kind::Autosave) continue;
55 const std::string wav = TakeManager::resolveBouncePath(projectPath, v);
56 VersionBounceInfo info;
57 if (wav.empty() || !std::filesystem::exists(wav) || !VersionBounce::readInfo(wav, info)) continue;
58 bounced.push_back({&v, wav, info});
59 }
60 std::stable_sort(bounced.begin(), bounced.end(), [](const Bounced& a, const Bounced& b) { return a.version->createdAtMs < b.version->createdAtMs; });
61 for (size_t i = bounced.size(); i-- > 0;) {
62 const auto& v = *bounced[i].version;
63 Item item;
64 item.kind = Item::Kind::Version;
65 item.id = "v:" + v.id;
66 item.title = projectName;
67 item.badge = "v" + std::to_string(i + 1);
68 item.subtitle = (v.name.empty() ? v.id : v.name) + " \xc2\xb7 " + VersionTime::ago(v.updatedAtMs ? v.updatedAtMs : v.createdAtMs, nowMs);
69 item.path = bounced[i].wav;
70 item.seconds = bounced[i].info.seconds;
71 item.lufs = bounced[i].info.lufs;
72 item.hasLufs = bounced[i].info.lufs > -69.0;
73 item.seed = seedOf(projectName); // a project's versions share a cover, as in the concept
74 out.push_back(std::move(item));
75 }
76 return out;
77}

95inline Item itemOf(const Reference& r) {
96 Item item;
97 item.kind = Item::Kind::Reference;
98 item.id = "r:" + r.path;
99 item.title = r.title;
100 item.badge = "REF";
101 item.path = r.path;
102 item.seconds = r.seconds;
103 item.lufs = r.lufs;
104 item.hasLufs = r.hasLufs;
105 item.seed = seedOf(r.title);
106 char buf[64];
107 const int mins = static_cast<int>(r.seconds) / 60, secs = static_cast<int>(r.seconds) % 60;
108 if (r.hasLufs) std::snprintf(buf, sizeof buf, "%.1f LUFS \xc2\xb7 %d:%02d", r.lufs, mins, secs);
109 else std::snprintf(buf, sizeof buf, "%d:%02d", mins, secs);
110 item.subtitle = buf;
111 return item;
112}

115inline bool save(const std::string& file, const std::vector<Reference>& refs) {
116 std::error_code ec;
117 std::filesystem::create_directories(std::filesystem::path(file).parent_path(), ec);
118 std::ofstream out(file, std::ios::trunc);
119 if (!out) return false;
120 const auto clean = [](std::string s) { std::replace(s.begin(), s.end(), '\t', ' '); std::replace(s.begin(), s.end(), '\n', ' '); return s; };
121 for (const auto& r : refs)
122 out << clean(r.path) << '\t' << clean(r.title) << '\t' << r.seconds << '\t' << (r.hasLufs ? std::to_string(r.lufs) : "-") << '\t' << r.addedMs << '\n';
123 return static_cast<bool>(out);
124}

126inline std::vector<Reference> load(const std::string& file) {
127 std::vector<Reference> refs;
128 std::ifstream in(file);
129 for (std::string line; std::getline(in, line);) {
130 std::vector<std::string> f;
131 std::stringstream ss(line);
132 for (std::string cell; std::getline(ss, cell, '\t');) f.push_back(cell);
133 if (f.size() < 5 || f[0].empty()) continue;
134 Reference r;
135 r.path = f[0];
136 r.title = f[1];
137 try {
138 r.seconds = std::stod(f[2]);
139 if (f[3] != "-") { r.lufs = std::stod(f[3]); r.hasLufs = true; }
140 r.addedMs = static_cast<uint64_t>(std::stoull(f[4]));
141 } catch (...) { continue; }
142 refs.push_back(std::move(r));
143 }
144 return refs;
145}
