58 ClipSourceID getOrCreateSourceWithId(ClipSourceID requestedId, const std::string& filePath,
59 const std::string& displayName = {}) {
60 auto it = m_pathToId.find(filePath);
61 if (it != m_pathToId.end()) {
62 return it->second;
63 }
64
65 // Projects store ids as JSON numbers, exact only up to 2^53-1. Restoring an
66 // id near that edge would push nextId past it, and the next new source would
67 // save an id no load can read back. Above 2^52 the loader's idMap assigns a
68 // fresh id instead, which leaves ~2^52 ids of headroom before the edge.
69 constexpr uint64_t MAX_RESTORABLE_SOURCE_ID = uint64_t{1} << 52;
70 ClipSourceID id{};
71 if (requestedId.isValid() && requestedId.value <= MAX_RESTORABLE_SOURCE_ID
72 && m_sources.find(requestedId.value) == m_sources.end()) {
73 id = requestedId;
74 if (requestedId.value >= nextId) {
75 nextId = requestedId.value + 1;
76 }
77 } else {
78 id = ClipSourceID{nextId++};
79 }
80
81 auto source = std::make_unique<ClipSource>(id, displayName.empty() ? makeDisplayName(filePath) : displayName);
82 source->setFilePath(filePath);
83 source->setOwner(this); // so a later readiness flip can bump the revision
84
85 m_sources[id.value] = std::move(source);
86 m_pathToId[filePath] = id;
87 ++m_revision;
88
89 return id;

106 void attachBuffer(ClipSource* source, std::shared_ptr<AudioBufferData> buffer) {
107 if (!source) {
108 return;
109 }
110 // Reject invalid payloads before they can clobber ready content: a
111 // failed attach must never discard valid audio and its cache.
112 if (!buffer || !buffer->isValid()) {
113 return;
114 }
115 // First manager to attach owns the source; a later attach through a
116 // different manager still proceeds (single-manager process today),
117 // and this manager's revision is what the sweep watches.
118 source->setOwner(this);
119 const bool wasReady = source->isValid();
120 source->setBuffer(std::move(buffer)); // bumps itself on a readiness flip
121 if (wasReady && source->isValid()) {
122 ++m_revision; // ready -> ready replacement still discards the cache
123 }
124 }

133 ClipSourceID createRecordedSource(const std::string& filePath, const std::string& displayName,
134 std::shared_ptr<AudioBufferData> buffer) {
135 if (!buffer || !buffer->isValid()) {
136 return ClipSourceID{};

141 if (!source) {
142 return ClipSourceID{};

145 // Canonical attach: bumps the revision whether this flips an existing
146 // path-deduped source to ready or replaces an earlier take's buffer —
147 // both are transitions waveform-cache builders watch for.
148 attachBuffer(source, std::move(buffer));

184 bool relinkSource(ClipSourceID id, const std::string& newFilePath) {
185 if (newFilePath.empty()) {
186 return false;
187 }
188 auto it = m_sources.find(id.value);
189 if (it == m_sources.end()) {
190 return false;
191 }
192 auto taken = m_pathToId.find(newFilePath);
193 if (taken != m_pathToId.end() && taken->second != id) {
194 return false;
195 }
196 const std::string oldPath = it->second->getFilePath();
197 it->second->setFilePath(newFilePath);
198 if (oldPath != newFilePath) {
199 m_pathToId.erase(oldPath);
200 m_pathToId[newFilePath] = id;
201 ++m_revision;
202 }
203 return true;
204 }
