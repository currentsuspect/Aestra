33void AestraApp::refreshVersionChip() {
34 if (!m_windowManager) return;
35 const std::string path = m_documentState.canonicalPath();
36 const auto manifest = path.empty() ? TakeManager::Manifest{} : TakeManager::loadManifest(path);
37 const auto chip = VersionPicker::chip(manifest);
38 const auto* open = manifest.ok ? manifest.activeTake() : nullptr;
39 m_recordedSteps = open ? static_cast<int>(open->steps.size()) : 0;
40 m_windowManager->setVersionChip(chip.branch, AestraUI::NUIColor::fromHex(VersionsTimeline::laneHex(chip.lane)), chip.version);
41}

43void AestraApp::pushProjectStatus() {
44 if (!m_windowManager) return;
45 auto* tm = m_content ? m_content->getTrackManager().get() : nullptr;
46 const bool modified = tm && tm->isModified();
47 std::string note = m_documentState.statusNote();
48 if (note.empty() && modified) { // "7 unsaved": what has been done since the open version was saved
49 const int n = std::max(0, static_cast<int>(tm->getCommandHistory().getUndoStack().size()) - m_recordedSteps);
50 if (n > 0) note = std::to_string(n) + " unsaved";
51 }
52 m_windowManager->setProjectStatus(m_documentState.displayName(), modified, note);
53}

55void AestraApp::showVersionPicker(float x) {
56 const auto manifest = TakeManager::loadManifest(m_documentState.canonicalPath());
57 const auto entries = VersionPicker::entries(manifest);
58 if (!m_windowManager || entries.empty()) return;
59 auto menu = std::make_shared<AestraUI::NUIContextMenu>();
60 auto header = std::make_shared<AestraUI::NUIContextMenuItem>("BRANCHES");
61 header->setEnabled(false);
62 menu->addItem(header);
63 for (const auto& e : entries) {
64 auto item = std::make_shared<AestraUI::NUIContextMenuItem>(e.branch);
65 item->setShortcut(e.versionName);
66 item->setSwatch(AestraUI::NUIColor::fromHex(VersionsTimeline::laneHex(e.lane)));
67 item->setChecked(e.current);
68 if (!e.current) {
69 item->setOnClick([this, id = e.versionId]() {
70 if (!switchToTake(id).ok) return;
71 refreshVersionChip();
72 if (auto panel = m_content->getTakesPanel()) panel->refreshTakes();
73 });
74 }
75 menu->addItem(item);
76 }
77 menu->addSeparator();
78 menu->addItem("Open Versions...", [this]() { m_content->setViewOpen(Audio::ViewType::Takes, true); });
79 menu->addItem("Save as an alt...", [this]() { if (createTakeFromCurrentProject(true)) refreshVersionChip(); });
80 m_windowManager->showDropdownMenu(menu, x);
81}

84void AestraApp::wireAudition() {
85 if (!m_content) return;
86 AuditionHooks hooks;
87 hooks.project = [this]() { return std::make_pair(m_documentState.displayName(), m_documentState.canonicalPath()); };
88 const std::string autosave = autosavePathOrEmpty();
89 if (!autosave.empty()) hooks.libraryFile = (std::filesystem::path(autosave).parent_path() / "audition_references.txt").string();
90 hooks.importReference = [this]() {
91 const std::string filter("Audio\0*.wav;*.flac;*.mp3;*.ogg;*.aiff;*.aif;*.m4a\0All Files\0*.*\0", sizeof("Audio\0*.wav;*.flac;*.mp3;*.ogg;*.aiff;*.aif;*.m4a\0All Files\0*.*\0") - 1);
92 pickFileAsync([filter](Aestra::IPlatformUtils& utils) { return utils.openFileDialog("Import a reference", filter); },
93 [this](const std::string& picked) { if (!picked.empty() && m_content) m_content->addAuditionReference(picked); });
94 };
95 hooks.bounceNow = [this]() {
96 const auto manifest = TakeManager::loadManifest(m_documentState.canonicalPath());
97 const auto* open = manifest.ok ? manifest.activeTake() : nullptr;
98 return open && bounceSavedVersion(m_documentState.canonicalPath(), open->id, [this](std::vector<float>& a, uint32_t& c, uint32_t& r) { return m_content->renderMixToBuffer(a, c, r); });
99 };
100 m_content->setAuditionHooks(std::move(hooks));
101}
