4// Which piano-roll keys show pressed (SPEC 3 §5.1), kept apart from the note layer so the
5// rules are one small, testable unit: a single-pitch press releases the other keys, a chord
6// press adds to them, and every change is reported once through `changed(pitch, down)`.
7
8#include <algorithm>
9#include <functional>
10#include <vector>
11
12namespace AestraUI {
13
14inline void pressPianoRollKey(std::vector<int>& pressed, int pitch, bool addToPress,
15 const std::function<void(int, bool)>& changed) {
16 if (!addToPress) {
17 // One pitch at a time (a drag onto a new row): the previous key comes back up.
18 for (const int held : pressed) {
19 if (held != pitch && changed) changed(held, false);
20 }
21 pressed.erase(std::remove_if(pressed.begin(), pressed.end(), [pitch](int held) { return held != pitch; }),
22 pressed.end());
23 }
24 if (std::find(pressed.begin(), pressed.end(), pitch) == pressed.end()) {
25 pressed.push_back(pitch);
26 if (changed) changed(pitch, true);
27 }
28}
29
30inline void releasePianoRollKeys(std::vector<int>& pressed, const std::function<void(int, bool)>& changed) {
31 for (const int held : pressed) {
32 if (changed) changed(held, false);
33 }
34 pressed.clear();
35}
