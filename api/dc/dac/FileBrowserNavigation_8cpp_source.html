29 auto& themeManager = NUIThemeManager::getInstance();
30 navHits_.clear();

32 const NUIColor paneBg = themeManager.getColor("backgroundSecondary");
33 const NUIColor sectionColor = themeManager.getColor("textSecondary").withAlpha(0.58f); // Stronger section headers
34 const NUIColor rowText = themeManager.getColor("textPrimary").withAlpha(0.78f);
35 const NUIColor muted = themeManager.getColor("textSecondary").withAlpha(0.48f);
36 const NUIColor selectedBg = themeManager.getColor("accentPrimary").withAlpha(0.05f);

47 // second row on item counts and duplicate structural rules.

50 renderer.drawLine({navHeader.x, navHeader.bottom()}, {navHeader.right(), navHeader.bottom()},

79 // windows keep the tail rows reachable without moving the header. Start at the
80 // full header height so the first nav row lines up with the first file row (the

88 // Start nav content at the same Y as the right column labels

92 auto collectionCount = [&](const std::string& tag) {
93 int count = 0;

95 if (std::find(tags.begin(), tags.end(), tag) != tags.end()) {
96 ++count;

102 auto drawSection = [&](const std::string& label) {
103 if (compact) {
104 // No rule for label-less compact sections. The explicit drawDivider()
105 // between groups already separates them; a section rule here doubled the
106 // divider at group boundaries and drew a lone line that isolated the
107 // first icon (the star) from the header.
108 return;
109 }
110 std::string upper = label;
111 std::transform(upper.begin(), upper.end(), upper.begin(), [](unsigned char c) { return static_cast<char>(std::toupper(c)); });
112 NUIRect labelRect(layout.navPane.x + themeProps.spacingM, y, layout.navPane.width - themeProps.spacingM * 2.0f, 22.0f);
113 renderer.drawText(upper, {labelRect.x, std::round(renderer.calculateTextY(labelRect, themeProps.fontSizeXS))}, themeProps.fontSizeXS, sectionColor);
114 if (label == "Collections") {
115 // Small up-chevron replacing "^" text glyph
116 const float upCx = layout.navPane.right() - 14.0f;
117 const float upCy = labelRect.y + labelRect.height * 0.5f;
118 const float upS = 2.5f;
119 renderer.drawLine({upCx - upS, upCy + upS * 0.5f}, {upCx, upCy - upS * 0.5f}, 1.3f, sectionColor.withAlpha(0.72f));
120 renderer.drawLine({upCx, upCy - upS * 0.5f}, {upCx + upS, upCy + upS * 0.5f}, 1.3f, sectionColor.withAlpha(0.72f));
121 }
122 y += 22.0f;
123 };

126 const NUIColor iconColor = selected ? themeManager.getColor("textPrimary").withAlpha(0.86f) : muted.withAlpha(0.70f);

130 auto drawSvgIcon = [&](const char* svg) {
131 // Keyed by the SVG literal: one action (SystemPlace) draws several glyphs.
132 static std::unordered_map<const char*, std::shared_ptr<NUIIcon>> iconCache;
133 const char* key = svg;
134 auto it = iconCache.find(key);
135 if (it == iconCache.end()) {
136 auto icon = std::make_shared<NUIIcon>(svg);
137 it = iconCache.emplace(key, icon).first;
138 }
139 auto& icon = it->second;
140 icon->setBounds({std::round(cx - 8.0f), std::round(cy - 8.0f), 16.0f, 16.0f});
141 icon->setColor(iconColor);
142 icon->onRender(renderer);
143 };

255 auto drawRow = [&](BrowserNavAction action, const std::string& label, int count = -1, std::string path = {}) {
256 NUIRect row(layout.navPane.x + 5.0f, y, std::max(0.0f, layout.navPane.width - 10.0f), BROWSER_NAV_ROW_H);
257 const bool isPlaceRow = action == BrowserNavAction::CustomPlace || action == BrowserNavAction::SystemPlace;
258 const bool placeActive = activeNavAction_ == BrowserNavAction::CustomPlace ||
259 activeNavAction_ == BrowserNavAction::SystemPlace;
260 bool selected = activeNavAction_ == action;
261 if (isPlaceRow) {
262 selected = placeActive && !isShowingListing() && !path.empty() &&
263 mapKeyForPath(activeNavPath_) == mapKeyForPath(path);
264 } else if (action == BrowserNavAction::Collection) {
265 selected = listingKind_ == ListingKind::Collection && listingTag_ == path;
266 }
267 if (selected) {
268 renderer.fillRoundedRect(row, themeProps.radiusS, selectedBg);
269 renderer.fillRoundedRect({row.x, row.y + 4.0f, 2.0f, row.height - 8.0f}, 1.0f,
270 themeManager.getColor("accentPrimary").withAlpha(0.92f));
271 }
272 // Nav hover wash also lives in renderHoverOverlays(), outside the cache.
273 drawIcon(row, action, selected, label);
274 if (!compact) {
275 renderer.drawText(label, {row.x + 33.0f, std::round(renderer.calculateTextY(row, themeProps.fontSizeS))},
276 themeProps.fontSizeS,
277 selected ? themeManager.getColor("textPrimary").withAlpha(0.90f) : rowText);
278 }
279 if (!compact && count > 0) {
280 const std::string countText = std::to_string(count);
281 const auto countSize = renderer.measureText(countText, themeManager.getFontSize("s"));
282 renderer.drawText(countText,
283 {row.right() - countSize.width - 8.0f, std::round(renderer.calculateTextY(row, 11.0f))},
284 11.0f,
285 muted.withAlpha(selected ? 0.82f : 0.54f));
286 }
287 navHits_.push_back({action, row, label, std::move(path)});
288 y += BROWSER_NAV_ROW_H;
289 ++hitIndex;
290 };

363 if (placesTop > 0 && placesBottom > placesTop) {
364 NUIRect placesOverlay = {layout.navPane.x, placesTop, layout.navPane.width, placesBottom - placesTop};
365 renderer.fillRect(placesOverlay, themeManager.getColor("dragTarget"));
366 renderer.strokeRoundedRect(placesOverlay, themeManager.getRadius("s"), 1.0f,
367 themeManager.getColor("focusRing"));
368 }
369 }
370
371 // Measure content for next frame's scroll clamp (y is post-offset screen
372 // space; add the offset back to recover the intrinsic content height).
373 navContentHeight_ = (y + navScrollOffset_) - navContentTop;
374
375 // Restore the full-pane clip and draw a thin scroll thumb when content
376 // overflows the available viewport.
377 renderer.setClipRect(layout.navPane);
378 const float navOverflow = navContentHeight_ - navViewportHeight_;
379 if (navOverflow > 0.5f && navViewportHeight_ > 0.0f) {
380 // Same painter and same gutter width as the file list beside it, so the
381 // Collections rail is not a third scrollbar dialect (spec 2 §2).
382 const float sbW = AestraUI::kOverlayScrollbarThickness;
383 const NUIRect gutter(layout.navPane.right() - sbW - 2.0f, navContentTop, sbW, navViewportHeight_);
384 const float thumbH = std::max(AestraUI::kOverlayScrollbarMinThumb,
385 navViewportHeight_ * (navViewportHeight_ / navContentHeight_));
386 const float thumbY = navContentTop + (navScrollOffset_ / navOverflow) * (navViewportHeight_ - thumbH);
387 AestraUI::drawOverlayScrollbar(renderer, gutter, NUIRect(gutter.x, thumbY, sbW, thumbH),
388 AestraUI::ScrollbarPaintState{});
389 }
390
391 renderer.clearClipRect();
392}

441bool FileBrowser::handleNavigationMouseEvent(const NUIMouseEvent& event, const BrowserLayout& layout) {
442 if (event.cursorCaptured) return false;
443
444 // Ignore the fixed folder-name header band: rows scrolled up under it are
445 // visually clipped, so they must not be clickable there either. navViewport is
446 // the rect the renderer clips to, so the two cannot disagree.
447 const bool insideNav = layout.navViewport.contains(event.position);
448 int newHovered = -1;
449 if (insideNav) {
450 for (int i = 0; i < static_cast<int>(navHits_.size()); ++i) {
451 if (navHits_[i].bounds.contains(event.position)) {
452 newHovered = i;
453 break;
454 }
455 }
456 }
457
458 if (newHovered != hoveredNavIndex_) {
459 hoveredNavIndex_ = newHovered;
460 setDirty(true); // hover overlay only — no cache rebuild
461 }
462
463 if (usesCompactNavigation() && newHovered >= 0 && newHovered < static_cast<int>(navHits_.size())) {
464 NUIPoint tooltipPosition = event.position;
465 tooltipPosition.x = layout.navPane.right() + 8.0f;
466 NUIComponent::showRemoteTooltip(navHits_[newHovered].label, tooltipPosition, this);
467 } else if (layout.navPane.contains(event.position)) {
468 // Only dismiss the navigation tooltip while the pointer is still in
469 // this region. The file list shares this component as its tooltip
470 // owner and will manage its own tooltip outside the navigation pane.
471 NUIComponent::hideRemoteTooltip(this);
472 }
473
474 if (event.pressed && event.button == NUIMouseButton::Right && newHovered >= 0 &&
475 newHovered < static_cast<int>(navHits_.size())) {
476 const BrowserNavHit hit = navHits_[newHovered];
477 switch (hit.action) {
478 case BrowserNavAction::CustomPlace:
479 case BrowserNavAction::SystemPlace:
480 showPlaceContextMenu(hit, event.position);
481 return true;
482 case BrowserNavAction::Favorites:
483 showFavoritesMenu(); // quick-jump list + Clear
484 return true;
485 case BrowserNavAction::AddFolder:
486 showAddFolderMenu();
487 return true;
488 case BrowserNavAction::Collection:
489 showCollectionContextMenu(hit.path, event.position);
490 return true;
491 default:
492 return false;
493 }
494 }
495
496 if (!event.pressed || event.button != NUIMouseButton::Left || newHovered < 0 ||
497 newHovered >= static_cast<int>(navHits_.size())) {
498 return false;
499 }
500
501 const BrowserNavAction previousAction = activeNavAction_;
502 const BrowserNavAction action = navHits_[newHovered].action;
503 activeNavAction_ = action;
504 activeNavPath_ = navHits_[newHovered].path;
505 updateContentViews();
506 auto setFilter = [this](QuickFilter filter) {
507 if (activeQuickFilter_ != filter) {
508 activeQuickFilter_ = filter;
509 applyFilter();
510 } else {
511 invalidateCache();
512 }
513 };
514
515 switch (action) {
516 // Favorites and Collections list their items from wherever they live,
517 // not just the ones inside the folder currently on screen.
518 case BrowserNavAction::Favorites:
519 activeQuickFilter_ = QuickFilter::All;
520 showFavorites();
521 break;
522 case BrowserNavAction::Collection:
523 activeQuickFilter_ = QuickFilter::All;
524 showCollection(navHits_[newHovered].path);
525 break;
526 case BrowserNavAction::AddCollection:
527 activeNavAction_ = previousAction; // an action, not a destination
528 updateContentViews();
529 beginCollectionRename(createUntitledCollection());
530 break;
531 // Sounds narrows whatever is on screen to audio. Samples is a place:
532 // the User Library folder for one-shots and loops, like Drums.
533 case BrowserNavAction::Sounds:
534 activeTagFilter_.clear();
535 setFilter(QuickFilter::Audio);
536 break;
537 case BrowserNavAction::Samples: {
538 auto path = std::filesystem::path(rootPath_) / "User Library" / "Samples";
539 std::error_code ec;
540 std::filesystem::create_directories(path, ec);
541 activeTagFilter_.clear();
542 activeQuickFilter_ = QuickFilter::Audio;
543 navigateTo(path.string());
544 applyFilter();
545 break;
546 }
547 case BrowserNavAction::Drums: {
548 auto path = std::filesystem::path(rootPath_) / "User Library" / "Drums";
549 std::error_code ec;
550 std::filesystem::create_directories(path, ec);
551 activeTagFilter_.clear();
552 activeQuickFilter_ = QuickFilter::Audio;
553 activeNavAction_ = BrowserNavAction::Drums;
554 navigateTo(path.string());
555 applyFilter();
556 break;
557 }
558 case BrowserNavAction::Instruments: {
559 auto path = std::filesystem::path(rootPath_) / "User Library" / "Instruments";
560 std::error_code ec;
561 std::filesystem::create_directories(path, ec);
562 activeTagFilter_.clear();
563 activeQuickFilter_ = QuickFilter::All;
564 activeNavAction_ = BrowserNavAction::Instruments;
565 navigateTo(path.string());
566 applyFilter();
567 break;
568 }
569 case BrowserNavAction::AudioEffects: {
570 auto path = std::filesystem::path(rootPath_) / "User Library" / "Effects";
571 std::error_code ec;
572 std::filesystem::create_directories(path, ec);
573 activeTagFilter_.clear();
574 activeQuickFilter_ = QuickFilter::All;
575 activeNavAction_ = BrowserNavAction::AudioEffects;
576 navigateTo(path.string());
577 applyFilter();
578 break;
579 }
580 case BrowserNavAction::Patterns:
581 activeTagFilter_.clear();
582 setFilter(QuickFilter::All);
583 break;
584 case BrowserNavAction::Clips: {
585 auto path = std::filesystem::path(rootPath_) / "User Library" / "Clips";
586 std::error_code ec;
587 std::filesystem::create_directories(path, ec);
588 activeTagFilter_.clear();
589 activeQuickFilter_ = QuickFilter::All;
590 activeNavAction_ = BrowserNavAction::Clips;
591 navigateTo(path.string());
592 applyFilter();
593 break;
594 }
595 case BrowserNavAction::CurrentProject:
596 // The audio the open project uses — it used to open the library root.
597 activeQuickFilter_ = QuickFilter::All;
598 showCurrentProject();
599 break;
600 case BrowserNavAction::UserLibrary:
601 case BrowserNavAction::Packs: {
602 std::filesystem::path base = rootPath_.empty() ? std::filesystem::path(currentPath_) : std::filesystem::path(rootPath_);
603 std::filesystem::path target = base / (action == BrowserNavAction::Packs ? "Packs" : "User Library");
604 std::error_code ec;
605 std::filesystem::create_directories(target, ec);
606 navigateTo(target.string());
607 activeTagFilter_.clear();
608 setFilter(QuickFilter::All);
609 break;
610 }
611 case BrowserNavAction::CustomPlace:
612 case BrowserNavAction::SystemPlace:
613 if (!navHits_[newHovered].path.empty()) {
614 navigateTo(navHits_[newHovered].path);
615 }
616 activeTagFilter_.clear();
617 setFilter(QuickFilter::All);
618 break;
619 case BrowserNavAction::AddFolder:
620 showAddFolderMenu();
621 break;
622 default:
623 activeTagFilter_.clear();
624 setFilter(QuickFilter::All);
625 break;
626 }
627
628 // Returning from an embedded view (Plugins/Patterns) to a file-backed view:
629 // while embedded, search-text changes route to the embedded view rather than
630 // applyFilter(), so the file list still reflects the pre-embedded query. The
631 // per-case setFilter() above only re-filters when the quick filter changed, so
632 // reapply here to honor the current search text on the way back to files.
633 const bool leavingEmbeddedView =
634 (previousAction == BrowserNavAction::Plugins || previousAction == BrowserNavAction::Patterns);
635 const bool enteringFileView =
636 (action != BrowserNavAction::Plugins && action != BrowserNavAction::Patterns);
637 if (leavingEmbeddedView && enteringFileView) {
638 applyFilter();
639 }
640
641 if (onNavActionSelected_) {
642 onNavActionSelected_(action);
643 }
644
645 invalidateCache();
646 return true;
647}
