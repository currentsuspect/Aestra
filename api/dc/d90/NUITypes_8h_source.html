123 return NUIColor(

125 ((argb >> 8) & 0xFF) / 255.0f,

127 alphaByte == 0 ? 1.0f : alphaByte / 255.0f
128 );
129 }

183 // HSL color space support
184 struct HSL {
185 float h = 0.0f; // Hue (0-360)
186 float s = 0.0f; // Saturation (0-1)
187 float l = 0.0f; // Lightness (0-1)
188 float a = 1.0f; // Alpha
189
190 HSL() = default;
191 HSL(float h, float s, float l, float a = 1.0f) : h(h), s(s), l(l), a(a) {}
192 };
193
194 // Convert RGB to HSL
195 HSL toHSL() const {
196 float max = std::max({r, g, b});
197 float min = std::min({r, g, b});
198 float delta = max - min;
199
200 float h = 0.0f;
201 if (delta != 0.0f) {
202 if (max == r) {
203 h = 60.0f * fmodf((g - b) / delta, 6.0f);
204 } else if (max == g) {
205 h = 60.0f * ((b - r) / delta + 2.0f);
206 } else {
207 h = 60.0f * ((r - g) / delta + 4.0f);
208 }
209 }
210 if (h < 0.0f) h += 360.0f;
211
212 float l = (max + min) / 2.0f;
213 float s = (delta == 0.0f) ? 0.0f : delta / (1.0f - std::abs(2.0f * l - 1.0f));
214
215 return HSL(h, s, l, a);
216 }
217
218 // Convert HSL to RGB
219 static NUIColor fromHSL(const HSL& hsl) {
220 float c = (1.0f - std::abs(2.0f * hsl.l - 1.0f)) * hsl.s;
221 float x = c * (1.0f - std::abs(fmodf(hsl.h / 60.0f, 2.0f) - 1.0f));
222 float m = hsl.l - c / 2.0f;
223
224 float r, g, b;
225 if (hsl.h < 60.0f) {
226 r = c; g = x; b = 0.0f;
227 } else if (hsl.h < 120.0f) {
228 r = x; g = c; b = 0.0f;
229 } else if (hsl.h < 180.0f) {
230 r = 0.0f; g = c; b = x;
231 } else if (hsl.h < 240.0f) {
232 r = 0.0f; g = x; b = c;
233 } else if (hsl.h < 300.0f) {
234 r = x; g = 0.0f; b = c;
235 } else {
236 r = c; g = 0.0f; b = x;
237 }
238
239 return NUIColor(r + m, g + m, b + m, hsl.a);
240 }

305 // Hex color support
306 uint32_t toHex() const {
307 uint32_t r8 = static_cast<uint32_t>(r * 255.0f) & 0xFF;
308 uint32_t g8 = static_cast<uint32_t>(g * 255.0f) & 0xFF;
309 uint32_t b8 = static_cast<uint32_t>(b * 255.0f) & 0xFF;
310 uint32_t a8 = static_cast<uint32_t>(a * 255.0f) & 0xFF;
311 return (a8 << 24) | (r8 << 16) | (g8 << 8) | b8;
312 }

314 // Smooth color transitions
315 static NUIColor lerpHSL(const NUIColor& a, const NUIColor& b, float t) {
316 HSL hslA = a.toHSL();
317 HSL hslB = b.toHSL();
318
319 // Handle hue wrapping
320 float h = hslA.h;
321 float hDiff = hslB.h - hslA.h;
322 if (hDiff > 180.0f) hDiff -= 360.0f;
323 else if (hDiff < -180.0f) hDiff += 360.0f;
324 h += hDiff * t;
325 if (h < 0.0f) h += 360.0f;
326 if (h >= 360.0f) h -= 360.0f;
327
328 return fromHSL(HSL(
329 h,
330 hslA.s + (hslB.s - hslA.s) * t,
331 hslA.l + (hslB.l - hslA.l) * t,
332 hslA.a + (hslB.a - hslA.a) * t
333 ));
334 }

576inline std::vector<NUIRect> NUIStackHorizontal(const NUIRect& parent, const std::vector<NUISize>& children, float spacing, size_t startIndex = 0) {
577 std::vector<NUIRect> rects;
578 float currentX = parent.x + startIndex * (children[startIndex].width + spacing);
579 for (size_t i = startIndex; i < children.size(); ++i) {
580 float y = parent.y + (parent.height - children[i].height) / 2.0f; // Center vertically
581 rects.push_back(NUIRect(currentX, y, children[i].width, children[i].height));
582 currentX += children[i].width + spacing;
583 }
584 return rects;
585}

713inline NUIRect NUIUnionRects(const std::vector<NUIRect>& rects) {
714 if (rects.empty()) return NUIRect(0, 0, 0, 0);
715
716 float minX = rects[0].x;
717 float minY = rects[0].y;
718 float maxX = rects[0].x + rects[0].width;
719 float maxY = rects[0].y + rects[0].height;
720
721 for (const auto& rect : rects) {
722 minX = std::min(minX, rect.x);
723 minY = std::min(minY, rect.y);
724 maxX = std::max(maxX, rect.x + rect.width);
725 maxY = std::max(maxY, rect.y + rect.height);
726 }
727
728 return NUIRect(minX, minY, maxX - minX, maxY - minY);
729}

758template<typename T>
759T lerp(T a, T b, T t) {

763template<typename T>
764T smoothstep(T edge0, T edge1, T x) {
765 T t = clamp((x - edge0) / (edge1 - edge0), static_cast<T>(0), static_cast<T>(1));
766 return t * t * (static_cast<T>(3) - static_cast<T>(2) * t);
767}

769template<typename T>
770T easeInOut(T t) {
771 return t < static_cast<T>(0.5) ?
772 static_cast<T>(2) * t * t :
773 static_cast<T>(1) - static_cast<T>(2) * (-t + static_cast<T>(2)) * (-t + static_cast<T>(2)) / static_cast<T>(2);
774}

883 float h = 0.0f; // Hue (0-360)
884 float s = 0.0f; // Saturation (0-1)
885 float v = 0.0f; // Value (0-1)
886 float a = 1.0f; // Alpha (0-1)

892// Convert RGB to HSV
893inline NUIColorHSV rgbToHsv(const NUIColor& rgb) {
894 float r = rgb.r, g = rgb.g, b = rgb.b;
895 float max = std::max({r, g, b});
896 float min = std::min({r, g, b});
897 float delta = max - min;
898
899 float h = 0.0f;
900 if (delta != 0.0f) {
901 if (max == r) {
902 h = 60.0f * fmodf((g - b) / delta, 6.0f);
903 } else if (max == g) {
904 h = 60.0f * ((b - r) / delta + 2.0f);
905 } else {
906 h = 60.0f * ((r - g) / delta + 4.0f);
907 }
908 }
909 if (h < 0.0f) h += 360.0f;
910
911 float s = max == 0.0f ? 0.0f : delta / max;
912 float v = max;
913
914 return NUIColorHSV(h, s, v, rgb.a);
915}
916
917// Convert HSV to RGB
918inline NUIColor hsvToRgb(const NUIColorHSV& hsv) {
919 float c = hsv.v * hsv.s;
920 float x = c * (1.0f - std::abs(fmodf(hsv.h / 60.0f, 2.0f) - 1.0f));
921 float m = hsv.v - c;
922
923 float r, g, b;
924 if (hsv.h < 60.0f) {
925 r = c; g = x; b = 0.0f;
926 } else if (hsv.h < 120.0f) {
927 r = x; g = c; b = 0.0f;
928 } else if (hsv.h < 180.0f) {
929 r = 0.0f; g = c; b = x;
930 } else if (hsv.h < 240.0f) {
931 r = 0.0f; g = x; b = c;
932 } else if (hsv.h < 300.0f) {
933 r = x; g = 0.0f; b = c;
934 } else {
935 r = c; g = 0.0f; b = x;
936 }
937
938 return NUIColor(r + m, g + m, b + m, hsv.a);
939}
