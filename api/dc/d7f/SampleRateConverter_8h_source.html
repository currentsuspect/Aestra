74// Precomputed Filter Table (Polyphase Sinc Coefficients)
75// =============================================================================
76
87struct PolyphaseFilterBank {
88 // Coefficient storage: [phase][tap]
89 // Aligned for SIMD (32-byte for AVX, 16-byte for SSE)
90 alignas(32) float coeffs[SRCConstants::POLYPHASE_PHASES][SRCConstants::MAX_FILTER_TAPS];
91
92 uint32_t numTaps{0}; // Active taps for current quality
93 uint32_t halfTaps{0}; // numTaps / 2 (for centering)
94
95 // Clear all coefficients
96 void clear() noexcept {
97 for (auto& phase : coeffs) {
98 for (float& c : phase)
99 c = 0.0f;
100 }
101 numTaps = 0;
102 halfTaps = 0;
103 }
104};

106// =============================================================================
107// Sample History Ring Buffer
108// =============================================================================
109
116struct SampleHistory {
117 // Planar + mirrored storage for SIMD-friendly contiguous windows.
118 // Layout: data[channel][index], where index spans kMirrorFactor * HISTORY_SIZE.
119 // Mirroring eliminates wrap checks for tap windows (RT-safe, fixed size).
120 // kMirrorFactor = 2 is sufficient: writePos + rel <= 127 + 127 = 254 < 256.
121 static constexpr uint32_t kMirrorFactor = 2;
122
123 alignas(32) float data[SRCConstants::MAX_CHANNELS][SRCConstants::HISTORY_SIZE * kMirrorFactor];
124
125 uint32_t writePos{0}; // Current write position in ring
126 uint32_t channels{0}; // Number of active channels
127 uint32_t size{0}; // Total frames in history
128
129 // Initialize the history buffer
130 void init(uint32_t numChannels) noexcept {
131 channels = numChannels;
132 size = SRCConstants::HISTORY_SIZE;
133 writePos = 0;
134 // Clear buffer
135 for (auto& chBuf : data) {
136 for (float& s : chBuf) {
137 s = 0.0f;
138 }
139 }
140 }
141
142 // Push a frame (all channels) into the ring buffer
143 void push(const float* frame) noexcept {
144 const uint32_t base0 = writePos;
145 const uint32_t base1 = base0 + size;
146
147 for (uint32_t ch = 0; ch < channels; ++ch) {
148 const float s = frame[ch];
149 data[ch][base0] = s;
150 data[ch][base1] = s;
151 }
152 // size is always a power of 2 (128), so use bitwise AND instead of modulo
153 writePos = (writePos + 1) & (size - 1);
154 }
155
156 // Get a contiguous window pointer for the given channel starting at relPos
157 // (0 = oldest, size-1 = newest). relPos wraps like get().
158 const float* getWindow(uint32_t channel, int32_t relPos) const noexcept {
159 if (size == 0)
160 return nullptr;
161 const int32_t sizeMask = static_cast<int32_t>(size) - 1;
162 int32_t rel = relPos & sizeMask;
163 if (rel < 0)
164 rel += static_cast<int32_t>(size);
165
166 // Chronological ring is laid out contiguously at [writePos .. writePos+size-1].
167 // Mirroring extends past wrap so tap windows are contiguous.
168 const uint32_t idx = writePos + static_cast<uint32_t>(rel);
169 return &data[channel][idx];
170 }
171
172 // Scalar accessor (kept for reference/testing and edge-case fallbacks)
173 float get(int32_t relPos, uint32_t channel) const noexcept {
174 const float* p = getWindow(channel, relPos);
175 return p ? *p : 0.0f;
176 }
177};

442inline uint32_t estimateOutputFrames(uint32_t inputFrames, uint32_t srcRate, uint32_t dstRate,
443 uint32_t latency = 0) noexcept {
444 const double ratio = static_cast<double>(dstRate) / static_cast<double>(srcRate);
445 return static_cast<uint32_t>(std::ceil(inputFrames * ratio)) + latency + 1;
