14class AssignUnitToFirstFreeInsertCommand final : public ICommand {
15public:
16 AssignUnitToFirstFreeInsertCommand(TrackManager& manager, UnitID unitId, std::string destinationName,
17 uint32_t color)
18 : m_manager(manager), m_unitId(unitId), m_destinationName(std::move(destinationName)), m_color(color) {}
19
20 void execute() override {
21 if (m_executed || !m_manager.getUnitManager().getUnit(m_unitId))
22 return;
23
24 if (!m_capturedPreviousRoute) {
25 m_previousChannelId = m_manager.getUnitManager().getUnitMixerChannel(m_unitId);
26 m_capturedPreviousRoute = true;
27 }
28
29 if (m_createdDestination) {
30 auto& playlist = m_manager.getPlaylistModel();
31 const auto restoredLaneId = playlist.createLaneWithId(m_laneId, m_destinationName);
32 if (restoredLaneId != m_laneId) {
33 playlist.removeLane(restoredLaneId);
34 return;
35 }
36 if (!m_manager.reinsertChannel(m_detachedChannel, m_channelIndex)) {
37 playlist.removeLane(m_laneId);
38 return;
39 }
40 if (auto* lane = playlist.getLane(m_laneId)) {
41 lane->colorRGBA = m_color;
42 }
43 } else if (m_destinationChannelId == MASTER_MIXER_CHANNEL_ID) {
44 selectOrCreateDestination();
45 if (m_destinationChannelId == MASTER_MIXER_CHANNEL_ID)
46 return;
47 }
48
49 m_manager.getUnitManager().setUnitMixerChannel(m_unitId, m_destinationChannelId);
50 m_manager.markModified();
51 m_executed = true;
52 }
53
54 void undo() override {
55 if (!m_executed)
56 return;
57
58 m_manager.getUnitManager().setUnitMixerChannel(m_unitId, m_previousChannelId);
59 if (m_createdDestination) {
60 m_detachedChannel = m_manager.detachChannelById(m_destinationChannelId, m_channelIndex);
61 if (!m_detachedChannel) {
62 m_manager.getUnitManager().setUnitMixerChannel(m_unitId, m_destinationChannelId);
63 return;
64 }
65 m_manager.getPlaylistModel().removeLane(m_laneId);
66 }
67 m_manager.markModified();
68 m_executed = false;
69 }
70
71 void redo() override { execute(); }
72
73 std::string getName() const override { return "Assign Unit to First Free Insert"; }
74 size_t getSizeInBytes() const override { return sizeof(*this); }
75 bool changesProjectState() const override { return true; }
76 bool isUndoable() const override { return m_executed; }

78private:
79 void selectOrCreateDestination() {
80 std::unordered_set<uint32_t> usedChannelIds;
81 for (const UnitID unitId : m_manager.getUnitManager().getAllUnitIDs()) {
82 if (unitId == m_unitId)
83 continue;
84 const uint32_t channelId = m_manager.getUnitManager().getUnitMixerChannel(unitId);
85 if (channelId != MASTER_MIXER_CHANNEL_ID) {
86 usedChannelIds.insert(channelId);
87 }
88 }
89 for (const auto& pattern : m_manager.getPatternManager().getAllPatterns()) {
90 if (!pattern || !pattern->isAudio())
91 continue;
92 const uint32_t channelId = pattern->getMixerChannelId();
93 if (channelId != MASTER_MIXER_CHANNEL_ID) {
94 usedChannelIds.insert(channelId);
95 }
96 }
97
98 for (size_t i = 0; i < m_manager.getChannelCount(); ++i) {
99 const auto* channel = m_manager.getChannel(i);
100 if (channel && usedChannelIds.find(channel->getChannelId()) == usedChannelIds.end()) {
101 m_destinationChannelId = channel->getChannelId();
102 return;
103 }
104 }
105
106 auto& playlist = m_manager.getPlaylistModel();
107 m_laneId = playlist.createLane(m_destinationName);
108 if (!m_laneId.isValid())
109 return;
110
111 auto* channel = m_manager.addChannel(m_destinationName);
112 if (!channel) {
113 playlist.removeLane(m_laneId);
114 m_laneId = {};
115 return;
116 }
117
118 m_destinationChannelId = channel->getChannelId();
119 channel->setColor(m_color);
120 if (auto* lane = playlist.getLane(m_laneId)) {
121 lane->colorRGBA = m_color;
122 }
123 m_createdDestination = true;
124 }
125
126 TrackManager& m_manager;
127 UnitID m_unitId;
128 std::string m_destinationName;
129 uint32_t m_color;
130 uint32_t m_previousChannelId{MASTER_MIXER_CHANNEL_ID};
131 uint32_t m_destinationChannelId{MASTER_MIXER_CHANNEL_ID};
132 PlaylistLaneID m_laneId;
133 std::unique_ptr<MixerChannel> m_detachedChannel;
134 size_t m_channelIndex{0};
135 bool m_capturedPreviousRoute{false};
136 bool m_createdDestination{false};
137 bool m_executed{false};
138};
