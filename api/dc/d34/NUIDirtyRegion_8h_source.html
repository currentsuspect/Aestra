54 // Dirty state tracking for widgets
55 enum class DirtyFlags : uint32_t {
56 None = 0,
57 Position = 1 << 0, // Widget moved
58 Size = 1 << 1, // Widget resized
59 Content = 1 << 2, // Widget content changed (text, color, etc.)
60 Children = 1 << 3, // Children changed
61 Visibility = 1 << 4, // Visibility changed
62 State = 1 << 5, // Widget state changed (hover, pressed, etc.)
63 All = 0xFFFFFFFF
64 };
65
66 inline DirtyFlags operator|(DirtyFlags a, DirtyFlags b) {
67 return static_cast<DirtyFlags>(static_cast<uint32_t>(a) | static_cast<uint32_t>(b));
68 }
69
70 inline DirtyFlags operator&(DirtyFlags a, DirtyFlags b) {
71 return static_cast<DirtyFlags>(static_cast<uint32_t>(a) & static_cast<uint32_t>(b));
72 }
73
74 inline DirtyFlags& operator|=(DirtyFlags& a, DirtyFlags b) {
75 a = a | b;
76 return a;
77 }
