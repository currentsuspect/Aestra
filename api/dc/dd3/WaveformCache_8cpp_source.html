24void WaveformCache::buildFromBuffer(const AudioBufferData& buffer, uint32_t baseSamplesPerPeak, uint32_t numLevels) {
25 if (!buffer.isValid()) {
26 Log::warning("WaveformCache: Cannot build from invalid buffer");
27 return;
28 }
29
30 if (buffer.isCompact()) {
31 // Compact PCM: build level 0 a peak at a time (no whole-file float copy), then the same mips.
32 std::vector<WaveformMipLevel> localLevels(numLevels);
33 buildLevelFromBuffer(buffer, baseSamplesPerPeak, localLevels[0]);
34 for (uint32_t i = 1; i < numLevels; ++i) {
35 buildNextLevel(localLevels[i - 1], localLevels[i]);
36 }
37 {
38 std::unique_lock<std::shared_mutex> lock(m_mutex);
39 std::swap(m_levels, localLevels);
40 m_numChannels = buffer.numChannels;
41 m_sourceFrames = static_cast<SampleIndex>(buffer.numFrames);
42 m_ready.store(true, std::memory_order_release);
43 }
44 return;
45 }
46
47 buildFromRaw(buffer.interleavedData.data(), buffer.numFrames, buffer.numChannels, baseSamplesPerPeak, numLevels);
48}

50void WaveformCache::buildLevelFromBuffer(const AudioBufferData& buffer, uint32_t samplesPerPeak,
51 WaveformMipLevel& outLevel) {
52 const SampleIndex numFrames = static_cast<SampleIndex>(buffer.numFrames);
53 const uint32_t numChannels = buffer.numChannels;
54 outLevel.samplesPerPeak = samplesPerPeak;
55 outLevel.numChannels = numChannels;
56 outLevel.numPeaks = (numFrames + samplesPerPeak - 1) / samplesPerPeak;
57 outLevel.peaks.resize(static_cast<size_t>(outLevel.numPeaks * numChannels));
58
59 std::vector<float> scratch(static_cast<size_t>(samplesPerPeak) * numChannels);
60 for (SampleIndex peakIdx = 0; peakIdx < outLevel.numPeaks; ++peakIdx) {
61 const SampleIndex startFrame = peakIdx * samplesPerPeak;
62 const SampleIndex endFrame = std::min(startFrame + samplesPerPeak, numFrames);
63 const uint32_t count = static_cast<uint32_t>(endFrame - startFrame);
64 buffer.copyFloats(static_cast<uint64_t>(startFrame) * numChannels, static_cast<uint64_t>(count) * numChannels,
65 scratch.data());
66 for (uint32_t ch = 0; ch < numChannels; ++ch) {
67 float minVal, maxVal;
68 double sumSq = 0.0;
69 WaveformSIMD::minMaxRMSChannel(scratch.data(), count, numChannels, ch, 0, count, minVal, maxVal, sumSq);
70 const float rms = (count > 0) ? static_cast<float>(std::sqrt(sumSq / static_cast<double>(count))) : 0.0f;
71 const size_t idx = static_cast<size_t>(peakIdx * numChannels + ch);
72 outLevel.peaks[idx] = WaveformPeak(minVal, maxVal, rms, count);
73 outLevel.peaks[idx].sanitize();
74 }
75 }
76}

78void WaveformCache::buildFromRaw(const float* data, SampleIndex numFrames, uint32_t numChannels,
79 uint32_t baseSamplesPerPeak, uint32_t numLevels) {
80 if (!data || numFrames <= 0 || numChannels == 0) {
81 Log::warning("WaveformCache: Invalid parameters for build");
82 return;
83 }
84
85 // Shadow Build Strategy:
86 // Build levels locally without holding the lock to prevent UI blocking.
87 std::vector<WaveformMipLevel> localLevels;
88 localLevels.resize(numLevels);
89
90 // Build first level from raw data
91 buildLevel(data, numFrames, numChannels, baseSamplesPerPeak, localLevels[0]);
92
93 // Build subsequent levels from previous level
94 for (uint32_t i = 1; i < numLevels; ++i) {
95 buildNextLevel(localLevels[i - 1], localLevels[i]);
96 }
97
98 // Critical Section: atomic swap
99 {
100 std::unique_lock<std::shared_mutex> lock(m_mutex);
101 std::swap(m_levels, localLevels);
102 m_numChannels = numChannels;
103 m_sourceFrames = numFrames;
104 m_ready.store(true, std::memory_order_release);
105 }
106 // localLevels now holds the OLD data and is destroyed here (outside lock)
107
108}

137void WaveformCache::buildNextLevel(const WaveformMipLevel& source, WaveformMipLevel& dest) {
138 dest.samplesPerPeak = source.samplesPerPeak * MIP_LEVEL_MULTIPLIER;
139 dest.numChannels = source.numChannels;
140 dest.numPeaks = (source.numPeaks + MIP_LEVEL_MULTIPLIER - 1) / MIP_LEVEL_MULTIPLIER;
141
142 dest.peaks.resize(static_cast<size_t>(dest.numPeaks * dest.numChannels));
143
144 for (SampleIndex peakIdx = 0; peakIdx < dest.numPeaks; ++peakIdx) {
145 SampleIndex startSourcePeak = peakIdx * MIP_LEVEL_MULTIPLIER;
146 SampleIndex endSourcePeak = std::min(startSourcePeak + MIP_LEVEL_MULTIPLIER, source.numPeaks);
147
148 for (uint32_t ch = 0; ch < dest.numChannels; ++ch) {
149 WaveformPeak merged = source.getPeak(ch, startSourcePeak);
150
151 for (SampleIndex i = startSourcePeak + 1; i < endSourcePeak; ++i) {
152 merged.merge(source.getPeak(ch, i));
153 }
154
155 size_t idx = static_cast<size_t>(peakIdx * dest.numChannels + ch);
156 dest.peaks[idx] = merged;
157 }
158 }

169size_t WaveformCache::selectLevel(double samplesPerPixel) const {
170 if (m_levels.empty())
171 return 0;
172
173 // Prefer coarsest level where samplesPerPeak <= samplesPerPixel,
174 // capping at the finest level if zoomed in beyond LOD0.
175 size_t best = 0;
176 for (size_t i = 0; i < m_levels.size(); ++i) {
177 if (m_levels[i].samplesPerPeak <= samplesPerPixel) {
178 best = i;
179 } else {
180 break;
181 }
182 }
183 return best;
184}

192void WaveformCache::getPeaksForRangePrecise(uint32_t channel, double startSample, double endSample,
193 uint32_t numPixels, std::vector<WaveformPeak>& outPeaks) const {
194 std::shared_lock<std::shared_mutex> lock(m_mutex);
195
196 outPeaks.clear();
197 outPeaks.resize(numPixels);
198
199 if (!m_ready.load(std::memory_order_acquire) || m_levels.empty() || numPixels == 0 ||
200 !std::isfinite(startSample) || !std::isfinite(endSample)) {
201 return;
202 }
203
204 const double sourceFrames = static_cast<double>(m_sourceFrames);
205 startSample = std::clamp(startSample, 0.0, sourceFrames);
206 endSample = std::clamp(endSample, 0.0, sourceFrames);
207 if (channel >= m_numChannels || startSample >= endSample) {
208 return;
209 }
210
211 const double samplesPerPixel = (endSample - startSample) / static_cast<double>(numPixels);
212 const WaveformMipLevel& level = m_levels[selectLevel(samplesPerPixel)];
213 mergePixelRange(level, channel, startSample, samplesPerPixel, numPixels, outPeaks);
214}

245void WaveformCache::mergePixelRange(const WaveformMipLevel& level, uint32_t channel, double startSample,
246 double samplesPerPixel, uint32_t numPixels,
247 std::vector<WaveformPeak>& outPeaks) const {
248 // Raw-pointer walk over the level's interleaved peak storage. Same math as
249 // getPeakRange(), minus the per-entry bounds check and 16-byte by-value
250 // return that dominated profiles on wide clips.
251 const WaveformPeak* peaks = level.peaks.data();
252 const size_t stride = level.numChannels;
253 const size_t ch = channel;
254
255 for (uint32_t pixel = 0; pixel < numPixels; ++pixel) {
256 const double pixelStart = startSample + static_cast<double>(pixel) * samplesPerPixel;
257 const double pixelEnd = startSample + static_cast<double>(pixel + 1) * samplesPerPixel;
258 SampleIndex startPeak = static_cast<SampleIndex>(std::floor(pixelStart / level.samplesPerPeak));
259 SampleIndex endPeak = static_cast<SampleIndex>(std::ceil(pixelEnd / level.samplesPerPeak));
260 endPeak = std::max(endPeak, startPeak + 1);
261
262 SampleIndex s = std::max<SampleIndex>(startPeak, 0);
263 SampleIndex e = std::min<SampleIndex>(endPeak, level.numPeaks);
264 if (s >= e) {
265 outPeaks[pixel] = WaveformPeak();
266 continue;
267 }
268
269 WaveformPeak acc = peaks[static_cast<size_t>(s) * stride + ch];
270 for (SampleIndex i = s + 1; i < e; ++i) {
271 acc.merge(peaks[static_cast<size_t>(i) * stride + ch]);
272 }
273 outPeaks[pixel] = acc;
274 }
275}

277WaveformPeak WaveformCache::getQuickPeak(uint32_t channel, SampleIndex startSample, SampleIndex numSamples) const {
278 std::shared_lock<std::shared_mutex> lock(m_mutex);
279
280 if (!m_ready.load(std::memory_order_acquire) || m_levels.empty()) {
281 return WaveformPeak();
282 }
283
284 if (channel >= m_numChannels || numSamples <= 0) {
285 return WaveformPeak();
286 }
287
288 // Use coarsest level that still covers the range
289 const WaveformMipLevel& level = m_levels.back();
290
291 SampleIndex startPeak = startSample / level.samplesPerPeak;
292 SampleIndex endPeak = (startSample + numSamples + level.samplesPerPeak - 1) / level.samplesPerPeak;
293
294 return level.getPeakRange(channel, startPeak, endPeak);
295}

329void WaveformCacheBuilder::buildAsync(const ClipSource& source, CompletionCallback callback) {
330 if (!source.isReady()) {
331 Log::warning("WaveformCacheBuilder: Source not ready");
332 if (callback)
333 callback(nullptr);
334 return;
335 }
336
337 m_impl->pendingCount.fetch_add(1);
338
339 // Capture buffer by shared_ptr for thread safety
340 auto buffer = source.getBuffer();
341 auto impl = m_impl; // shared_ptr copy keeps Impl alive
342
343 std::thread([buffer, callback, impl]() {
344 struct PendingDecrement {
345 std::atomic<size_t>* cnt;
346 explicit PendingDecrement(std::atomic<size_t>* c) : cnt(c) {}
347 ~PendingDecrement() { cnt->fetch_sub(1); }
348 } guard(&impl->pendingCount);
349
350 if (impl->cancelFlag.load()) {
351 if (callback)
352 callback(nullptr);
353 return;
354 }
355
356 auto cache = std::make_shared<WaveformCache>();
357 cache->buildFromBuffer(*buffer);
358
359 if (callback) {
360 callback(cache);
361 }
362 }).detach();
363}

376void WaveformCacheBuilder::cancelAll() {
377 m_impl->cancelFlag.store(true);
378
379 // Wait up to 5 seconds for pending builds to finish
380 int waits = 0;
381 constexpr int kMaxWaits = 5000; // 5 seconds
382 while (m_impl->pendingCount.load(std::memory_order_acquire) > 0 && waits < kMaxWaits) {
383 std::this_thread::sleep_for(std::chrono::milliseconds(1));
384 ++waits;
385 }
386
387 m_impl->cancelFlag.store(false);
388}
