33void WaveformCache::buildFromRaw(const float* data, SampleIndex numFrames, uint32_t numChannels,
34 uint32_t baseSamplesPerPeak, uint32_t numLevels) {
35 if (!data || numFrames <= 0 || numChannels == 0) {
36 Log::warning("WaveformCache: Invalid parameters for build");
37 return;
38 }
39
40 // Shadow Build Strategy:
41 // Build levels locally without holding the lock to prevent UI blocking.
42 std::vector<WaveformMipLevel> localLevels;
43 localLevels.resize(numLevels);
44
45 // Build first level from raw data
46 buildLevel(data, numFrames, numChannels, baseSamplesPerPeak, localLevels[0]);
47
48 // Build subsequent levels from previous level
49 for (uint32_t i = 1; i < numLevels; ++i) {
50 buildNextLevel(localLevels[i - 1], localLevels[i]);
51 }
52
53 // Critical Section: atomic swap
54 {
55 std::unique_lock<std::shared_mutex> lock(m_mutex);
56 std::swap(m_levels, localLevels);
57 m_numChannels = numChannels;
58 m_sourceFrames = numFrames;
59 m_ready.store(true, std::memory_order_release);
60 }
61 // localLevels now holds the OLD data and is destroyed here (outside lock)
62
63}

65void WaveformCache::buildLevel(const float* data, SampleIndex numFrames, uint32_t numChannels, uint32_t samplesPerPeak,
66 WaveformMipLevel& outLevel) {
67 outLevel.samplesPerPeak = samplesPerPeak;
68 outLevel.numChannels = numChannels;
69 outLevel.numPeaks = (numFrames + samplesPerPeak - 1) / samplesPerPeak;
70
71 outLevel.peaks.resize(static_cast<size_t>(outLevel.numPeaks * numChannels));
72
73 for (SampleIndex peakIdx = 0; peakIdx < outLevel.numPeaks; ++peakIdx) {
74 SampleIndex startFrame = peakIdx * samplesPerPeak;
75 SampleIndex endFrame = std::min(startFrame + samplesPerPeak, numFrames);
76 uint32_t count = static_cast<uint32_t>(endFrame - startFrame);
77
78 for (uint32_t ch = 0; ch < numChannels; ++ch) {
79 float minVal, maxVal;
80 double sumSq = 0.0;
81
82 WaveformSIMD::minMaxRMSChannel(data, numFrames, numChannels, ch, startFrame, endFrame, minVal, maxVal, sumSq);
83
84 float rms = (count > 0) ? static_cast<float>(std::sqrt(sumSq / static_cast<double>(count))) : 0.0f;
85 size_t idx = static_cast<size_t>(peakIdx * numChannels + ch);
86 outLevel.peaks[idx] = WaveformPeak(minVal, maxVal, rms, count);
87 outLevel.peaks[idx].sanitize();
88 }
89 }
90}

92void WaveformCache::buildNextLevel(const WaveformMipLevel& source, WaveformMipLevel& dest) {
93 dest.samplesPerPeak = source.samplesPerPeak * MIP_LEVEL_MULTIPLIER;
94 dest.numChannels = source.numChannels;
95 dest.numPeaks = (source.numPeaks + MIP_LEVEL_MULTIPLIER - 1) / MIP_LEVEL_MULTIPLIER;
96
97 dest.peaks.resize(static_cast<size_t>(dest.numPeaks * dest.numChannels));
98
99 for (SampleIndex peakIdx = 0; peakIdx < dest.numPeaks; ++peakIdx) {
100 SampleIndex startSourcePeak = peakIdx * MIP_LEVEL_MULTIPLIER;
101 SampleIndex endSourcePeak = std::min(startSourcePeak + MIP_LEVEL_MULTIPLIER, source.numPeaks);
102
103 for (uint32_t ch = 0; ch < dest.numChannels; ++ch) {
104 WaveformPeak merged = source.getPeak(ch, startSourcePeak);
105
106 for (SampleIndex i = startSourcePeak + 1; i < endSourcePeak; ++i) {
107 merged.merge(source.getPeak(ch, i));
108 }
109
110 size_t idx = static_cast<size_t>(peakIdx * dest.numChannels + ch);
111 dest.peaks[idx] = merged;
112 }
113 }

124size_t WaveformCache::selectLevel(double samplesPerPixel) const {
125 if (m_levels.empty())
126 return 0;
127
128 // Prefer coarsest level where samplesPerPeak <= samplesPerPixel,
129 // capping at the finest level if zoomed in beyond LOD0.
130 size_t best = 0;
131 for (size_t i = 0; i < m_levels.size(); ++i) {
132 if (m_levels[i].samplesPerPeak <= samplesPerPixel) {
133 best = i;
134 } else {
135 break;
136 }
137 }
138 return best;
139}

147void WaveformCache::getPeaksForRangePrecise(uint32_t channel, double startSample, double endSample,
148 uint32_t numPixels, std::vector<WaveformPeak>& outPeaks) const {
149 std::shared_lock<std::shared_mutex> lock(m_mutex);
150
151 outPeaks.clear();
152 outPeaks.resize(numPixels);
153
154 if (!m_ready.load(std::memory_order_acquire) || m_levels.empty() || numPixels == 0 ||
155 !std::isfinite(startSample) || !std::isfinite(endSample)) {
156 return;
157 }
158
159 const double sourceFrames = static_cast<double>(m_sourceFrames);
160 startSample = std::clamp(startSample, 0.0, sourceFrames);
161 endSample = std::clamp(endSample, 0.0, sourceFrames);
162 if (channel >= m_numChannels || startSample >= endSample) {
163 return;
164 }
165
166 // Calculate samples per pixel
167 const double samplesPerPixel = (endSample - startSample) / static_cast<double>(numPixels);
168
169 // Select appropriate mip level: coarsest where samplesPerPeak <= samplesPerPixel
170 size_t levelIdx = 0;
171 for (size_t i = 0; i < m_levels.size(); ++i) {
172 if (m_levels[i].samplesPerPeak <= samplesPerPixel) {
173 levelIdx = i;
174 } else {
175 break;
176 }
177 }
178
179 const WaveformMipLevel& level = m_levels[levelIdx];
180
181 // Generate peaks for each pixel by merging LOD entries overlapping the pixel's sample range.
182 // This avoids interpolation of min/max (which is physically meaningless) and ensures
183 // deterministic, source-anchored mapping. The fractional range is retained until the
184 // cache level is queried, so a clip split cannot reset the pixel grid at its seam.
185 for (uint32_t pixel = 0; pixel < numPixels; ++pixel) {
186 const double pixelStart = startSample + static_cast<double>(pixel) * samplesPerPixel;
187 const double pixelEnd = startSample + static_cast<double>(pixel + 1) * samplesPerPixel;
188 const double startPeakF = pixelStart / level.samplesPerPeak;
189 const double endPeakF = pixelEnd / level.samplesPerPeak;
190
191 SampleIndex startPeak = static_cast<SampleIndex>(std::floor(startPeakF));
192 SampleIndex endPeak = static_cast<SampleIndex>(std::ceil(endPeakF));
193 endPeak = std::max(endPeak, startPeak + 1);
194 outPeaks[pixel] = level.getPeakRange(channel, startPeak, endPeak);
195 }
196}

198WaveformPeak WaveformCache::getQuickPeak(uint32_t channel, SampleIndex startSample, SampleIndex numSamples) const {
199 std::shared_lock<std::shared_mutex> lock(m_mutex);
200
201 if (!m_ready.load(std::memory_order_acquire) || m_levels.empty()) {
202 return WaveformPeak();
203 }
204
205 if (channel >= m_numChannels || numSamples <= 0) {
206 return WaveformPeak();
207 }
208
209 // Use coarsest level that still covers the range
210 const WaveformMipLevel& level = m_levels.back();
211
212 SampleIndex startPeak = startSample / level.samplesPerPeak;
213 SampleIndex endPeak = (startSample + numSamples + level.samplesPerPeak - 1) / level.samplesPerPeak;
214
215 return level.getPeakRange(channel, startPeak, endPeak);
216}

226size_t WaveformCache::getMemoryUsage() const {
227 std::shared_lock<std::shared_mutex> lock(m_mutex);
228 size_t total = 0;
229 for (const auto& level : m_levels) {
230 total += level.peaks.size() * sizeof(WaveformPeak);
231 }
232 return total;

250void WaveformCacheBuilder::buildAsync(const ClipSource& source, CompletionCallback callback) {
251 if (!source.isReady()) {
252 Log::warning("WaveformCacheBuilder: Source not ready");
253 if (callback)
254 callback(nullptr);
255 return;
256 }
257
258 m_impl->pendingCount.fetch_add(1);
259
260 // Capture buffer by shared_ptr for thread safety
261 auto buffer = source.getBuffer();
262 auto impl = m_impl; // shared_ptr copy keeps Impl alive
263
264 std::thread([buffer, callback, impl]() {
265 struct PendingDecrement {
266 std::atomic<size_t>* cnt;
267 explicit PendingDecrement(std::atomic<size_t>* c) : cnt(c) {}
268 ~PendingDecrement() { cnt->fetch_sub(1); }
269 } guard(&impl->pendingCount);
270
271 if (impl->cancelFlag.load()) {
272 if (callback)
273 callback(nullptr);
274 return;
275 }
276
277 auto cache = std::make_shared<WaveformCache>();
278 cache->buildFromBuffer(*buffer);
279
280 if (callback) {
281 callback(cache);
282 }
283 }).detach();
284}

297void WaveformCacheBuilder::cancelAll() {
298 m_impl->cancelFlag.store(true);
299
300 // Wait up to 5 seconds for pending builds to finish
301 int waits = 0;
302 constexpr int kMaxWaits = 5000; // 5 seconds
303 while (m_impl->pendingCount.load(std::memory_order_acquire) > 0 && waits < kMaxWaits) {
304 std::this_thread::sleep_for(std::chrono::milliseconds(1));
305 ++waits;
306 }
307
308 m_impl->cancelFlag.store(false);
309}
