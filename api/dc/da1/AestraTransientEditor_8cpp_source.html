54void AestraTransientEditor::layoutControls() {
55 const auto b = getBounds();
56 const float contentTop = b.y + AestraPanelWindow::TITLE_BAR_H;
57
58 // Two large bipolar knobs (Attack left, Sustain right) flanking the
59 // envelope sketch in the middle.
60 const float knobRow = contentTop + 16.0f;
61 constexpr float kKnobSize = 130.0f;
62 m_attackRect = NUIRect(b.x + 28.0f, knobRow, kKnobSize, kKnobSize);
63 m_sustainRect = NUIRect(b.right() - 28.0f - kKnobSize, knobRow, kKnobSize, kKnobSize);
64
65 // Envelope sketch sits between the knobs, at the same height so its
66 // top/bottom edges align with the bipolar knob wells.
67 const float sketchLeft = m_attackRect.right() + 18.0f;
68 const float sketchRight = m_sustainRect.x - 18.0f;
69 m_sketchRect = NUIRect(sketchLeft, knobRow + 6.0f, sketchRight - sketchLeft, kKnobSize - 12.0f);
70
71 // Output knob and Mix slider share a bottom band; the band starts below
72 // the bipolar knobs' label/value/range captions so nothing overlaps.
73 // Captions run to bottom + 4 + 14 + 2 + 16 + 2 + 12 = bottom + 50.
74 const float captionBottom = m_attackRect.bottom() + 50.0f;
75 const float bottomRow = captionBottom + 12.0f;
76 constexpr float kOutSize = 64.0f;
77 m_outputRect = NUIRect(b.x + 30.0f, bottomRow, kOutSize, kOutSize);
78 m_mixRect =
79 NUIRect(m_outputRect.right() + 30.0f, bottomRow + 18.0f, b.right() - m_outputRect.right() - 60.0f, 28.0f);
80
81 // Bypass pill sits in the bottom band, just below the Mix slider —
82 // the cleanest real estate in the editor (nothing else lives there).
83 // Computed last so m_mixRect is initialised.
84 constexpr float kBypassW = 88.0f;
85 constexpr float kBypassH = 22.0f;
86 const float bypassY = m_mixRect.bottom() + 6.0f;

112void AestraTransientEditor::drawBipolarKnob(NUIRenderer& renderer, const NUIRect& rect, uint32_t paramId,
113 const char* label, const char* rangeText) {
114 auto& theme = NUIThemeManager::getInstance();
115 const float value = m_instance->getParameter(paramId);
116 // value is in [0,1] where 0.5 is neutral; convert to bipolar [-1, 1].
117 const float bipolar = AestraTransient::bipolarFromNorm(value);
118
119 const NUIPoint c = rect.center();
120 const float r = rect.width * 0.5f - 4.0f;
121
122 renderer.fillCircle(c, r + 4.0f, insetSurface());
123 renderer.strokeCircle(c, r + 4.0f, 1.0f, editorInk(0.060f));
124
125 // Backplate arc: full bidirectional sweep, dimmer than the active fill.
126 drawArc(renderer, c, r - 3.0f, kKnobStart, kKnobStart + kKnobSweep, 4.0f, editorNeutral(0.199f, 1.0f));
127
128 // Active fill: from the centre detent out toward the current side.
129 // The fill colour itself flips with sign (blue right, amber left) so
130 // "boost" and "cut" read at a glance without staring at the value.
131 const float halfSweep = kKnobSweep * 0.5f;
132 if (bipolar >= 0.0f) {
133 drawArc(renderer, c, r - 3.0f, kKnobCenter, kKnobCenter + halfSweep * std::min(bipolar, 1.0f), 4.0f,
134 accent().withAlpha(0.95f));
135 } else {
136 const NUIColor cutAccent(0.94f, 0.62f, 0.30f, 0.95f);
137 drawArc(renderer, c, r - 3.0f, kKnobCenter - halfSweep * std::min(-bipolar, 1.0f), kKnobCenter, 4.0f,
138 cutAccent);
139 }
140
141 // Centre detent marker: a short white tick at 12 o'clock.
142 const float tickInner = r - 9.0f;
143 const float tickOuter = r - 3.0f;
144 const NUIPoint tickIn{c.x + std::cos(kKnobCenter) * tickInner, c.y + std::sin(kKnobCenter) * tickInner};
145 const NUIPoint tickOut{c.x + std::cos(kKnobCenter) * tickOuter, c.y + std::sin(kKnobCenter) * tickOuter};
146 renderer.drawLine(tickIn, tickOut, 2.0f, theme.getColor("textPrimary").withAlpha(0.85f));
147
148 // Needle always points from 12 o'clock; the fill arc conveys the value.
149 const float needleLen = r - 16.0f;
150 const float needleAngle =
151 kKnobCenter + (bipolar >= 0.0f ? 1.0f : -1.0f) * halfSweep * std::min(std::abs(bipolar), 1.0f);
152 const NUIPoint tip(c.x + std::cos(needleAngle) * needleLen, c.y + std::sin(needleAngle) * needleLen);
153 renderer.drawLine(c, tip, 2.0f, theme.getColor("textPrimary").withAlpha(0.85f));
154 renderer.fillCircle(tip, 3.0f, theme.getColor("textPrimary").withAlpha(0.95f));
155
156 // Inner well, mirrored from Sat.
157 const float wellR = r * 0.32f;
158 renderer.fillCircle(c, wellR, editorNeutral(0.045f, 0.96f));
159 renderer.strokeCircle(c, wellR, 1.0f, accent().withAlpha(0.30f));
160
161 // Label band beneath the knob: name, then signed value (e.g. "+12%", "0%", "−100%").
162 const NUIRect labelRect(rect.x, rect.bottom() + 4.0f, rect.width, 14.0f);
163 renderer.drawTextCentered(label, labelRect, 11.0f, theme.getColor("textPrimary").withAlpha(0.95f));
164 const std::string valStr = m_instance->getParameterDisplay(paramId);
165 const NUIRect valRect(rect.x, rect.bottom() + 20.0f, rect.width, 16.0f);
166 renderer.drawTextCentered(valStr, valRect, 12.5f, accent().withAlpha(0.96f));
167 const NUIRect rangeRect(rect.x, rect.bottom() + 38.0f, rect.width, 12.0f);
168 renderer.drawTextCentered(rangeText, rangeRect, 9.5f, theme.getColor("textSecondary").withAlpha(0.55f));
169}

171void AestraTransientEditor::drawUnipolarKnob(NUIRenderer& renderer, const NUIRect& rect, uint32_t paramId,
172 const char* label, const char* rangeText) {
173 auto& theme = NUIThemeManager::getInstance();
174 const float value = m_instance->getParameter(paramId);
175 const NUIPoint c = rect.center();
176 const float r = rect.width * 0.5f - 3.0f;
177
178 renderer.fillCircle(c, r + 3.0f, insetSurface());
179 renderer.strokeCircle(c, r + 3.0f, 1.0f, editorInk(0.055f));
180
181 // Output trims a bipolar range but is presented as a single sweep with
182 // 12 o'clock = 0 dB, the conventional "trim" affordance. Sweep midpoint
183 // sits at kPi (180° + 3π/4 in screen-coords y-down) so a normalized
184 // value of 0.5 (i.e. unity gain) renders at 12 o'clock.
185 const float sweepStart = kPi * 0.75f; // 135° (7 o'clock)
186 const float sweepEnd = kPi * 2.25f; // 405° / 45° (1 o'clock)
187 drawArc(renderer, c, r - 2.0f, sweepStart, sweepEnd, 3.0f, editorNeutral(0.199f, 1.0f));
188 const float fillEnd = sweepStart + (sweepEnd - sweepStart) * value;
189 drawArc(renderer, c, r - 2.0f, sweepStart, fillEnd, 3.0f, accent().withAlpha(0.92f));
190
191 const float needleLen = r - 8.0f;
192 const float angle = sweepStart + (sweepEnd - sweepStart) * value;
193 const NUIPoint tip(c.x + std::cos(angle) * needleLen, c.y + std::sin(angle) * needleLen);
194 renderer.drawLine(c, tip, 1.8f, accent().withAlpha(0.85f));
195 renderer.fillCircle(tip, 2.6f, accent());
196
197 const NUIRect labelRect(rect.x, rect.bottom() + 4.0f, rect.width, 13.0f);
198 renderer.drawTextCentered(label, labelRect, 10.5f, theme.getColor("textPrimary").withAlpha(0.95f));
199 const std::string valStr = m_instance->getParameterDisplay(paramId);
200 const NUIRect valRect(rect.x, rect.bottom() + 19.0f, rect.width, 14.0f);
201 renderer.drawTextCentered(valStr, valRect, 11.5f, accent().withAlpha(0.92f));
202 const NUIRect rangeRect(rect.x, rect.bottom() + 35.0f, rect.width, 11.0f);
203 renderer.drawTextCentered(rangeText, rangeRect, 9.0f, theme.getColor("textSecondary").withAlpha(0.50f));
204}

206void AestraTransientEditor::drawEnvelopeSketch(NUIRenderer& renderer, const NUIRect& rect) {
207 auto& theme = NUIThemeManager::getInstance();
208 renderer.fillRoundedRect(rect, 10.0f, sketchSurface());
209 renderer.strokeRoundedRect(rect, 10.0f, 1.0f, editorInk(0.060f));
210
211 // Two envelope halves derived from the current settings: attack portion
212 // rises and is amplified by Attack amount; decay portion falls and is
213 // amplified by Sustain amount. The sketch updates live as the user
214 // turns the knobs, so the surface "shows" what the plugin will do without
215 // any audio-thread tap.
216 const float attackAmt = AestraTransient::bipolarFromNorm(m_instance->getParameter(AestraTransient::kAttack));
217 const float sustainAmt = AestraTransient::bipolarFromNorm(m_instance->getParameter(AestraTransient::kSustain));
218
219 const float padX = 18.0f;
220 const float padY = 16.0f;
221 const NUIRect plotRect{rect.x + padX, rect.y + padY, rect.width - padX * 2, rect.height - padY * 2};
222 // Centre baseline (zero gain line)
223 const float cy = plotRect.y + plotRect.height * 0.5f;
224 renderer.drawLine({plotRect.x, cy}, {plotRect.right(), cy}, 1.0f, editorInk(0.05f));
225
226 constexpr int kSamples = 96;
227 std::array<NUIPoint, kSamples> polyline{};
228 const float attackEnd = 0.18f; // first 18% of the plot is the attack portion
229 // Y-extent of the unboosted peak as a fraction of the plot height, so
230 // the curve uses the real rect rather than a 0.85-pixel line.
231 const float peak = plotRect.height * 0.35f;
232 const float ampA = attackAmt; // −1…+1
233 const float ampS = sustainAmt;
234
235 for (int i = 0; i < kSamples; ++i) {
236 const float t = static_cast<float>(i) / static_cast<float>(kSamples - 1);
237 // Reference envelope: a peak in the first slice, exponential decay after.
238 float env;
239 if (t < attackEnd) {
240 const float u = t / attackEnd;
241 // Fast attack: rising from 0 to 1 across the attack slice
242 env = 1.0f - (1.0f - u) * (1.0f - u); // ease-out
243 } else {
244 const float u = (t - attackEnd) / (1.0f - attackEnd);
245 env = std::exp(-u * 3.0f);
246 }
247 // Apply bipolar shaping: positive in the attack region, negative in the decay region
248 float shaped;
249 if (t < attackEnd) {
250 shaped = env * (1.0f + ampA * 0.7f);
251 } else {
252 shaped = env * (1.0f + ampS * 0.7f);
253 }
254 const float x = plotRect.x + plotRect.width * t;
255 const float y = cy - shaped * peak;
256 polyline[i] = {x, y};
257 }
258 renderer.drawPolyline(polyline.data(), kSamples, 2.0f, accent().withAlpha(0.92f));
259
260 // Centre marker
261 const float cx = plotRect.x + plotRect.width * attackEnd;
262 renderer.drawLine({cx, plotRect.y}, {cx, plotRect.bottom()}, 1.0f, editorInk(0.10f));
263 renderer.drawTextCentered("attack",
264 NUIRect(plotRect.x, plotRect.bottom() - 14.0f, plotRect.width * attackEnd, 12.0f), 9.0f,
265 theme.getColor("textSecondary").withAlpha(0.55f));
266 renderer.drawTextCentered("sustain", NUIRect(cx, plotRect.bottom() - 14.0f, plotRect.right() - cx, 12.0f), 9.0f,
267 theme.getColor("textSecondary").withAlpha(0.55f));
268}

305 if (m_attackRect.contains(p)) {
306 return 0;

308 if (m_sustainRect.contains(p)) {
309 return 1;
310 }

312 return 2;

317bool AestraTransientEditor::onMouseEvent(const NUIMouseEvent& event) {
318 if (AestraPanelWindow::onMouseEvent(event)) {
319 return true;
320 }
321 if (!m_instance) {
322 return false;
323 }
324
325 // Bypass pill: click toggles
326 if (bypassContains(event.position) && event.pressed && event.button == NUIMouseButton::Left) {
327 const float cur = m_instance->getParameter(kBypass);
328 m_instance->setParameter(kBypass, cur > 0.5f ? 0.0f : 1.0f);
329 setDirty();
330 return true;
331 }
332 m_bypassHovered = bypassContains(event.position);
333
334 if (m_dragging >= 0 && m_dragging < 3) {
335 if (event.released) {
336 endKnobCapture();
337 m_dragging = -1;
338 return true;
339 }
340 if (event.button == NUIMouseButton::None) {
341 // Service-owned frame delta for the knob drag
342 // (AestraPanelWindow::knobDragStep, same path Sat uses).
343 const uint32_t paramId = static_cast<uint32_t>(m_dragging);
344 const float step = knobDragStep(event, kDragRangePx);
345 const float cur = m_instance->getParameter(paramId);
346 m_instance->setParameter(paramId, std::clamp(cur + step, 0.0f, 1.0f));
347 setDirty();
348 return true;
349 }
350 }
351
352 if (event.pressed && event.button == NUIMouseButton::Left) {
353 const int hit = knobAtPoint(event.position);
354 if (hit >= 0) {
355 m_dragging = hit;
356 const NUIRect& rect = hit == 0 ? m_attackRect : (hit == 1 ? m_sustainRect : m_outputRect);
357 beginKnobCapture({rect.x + rect.width * 0.5f, rect.y + rect.height * 0.5f}, event.position);
358 return true;
359 }
360 if (mixContains(event.position)) {
361 m_dragging = 3; // mix
362 // Jump the value to the press position so a click anywhere on
363 // the track selects, not just a drag from the current thumb.
364 const float t = std::clamp((event.position.x - m_mixRect.x) / m_mixRect.width, 0.0f, 1.0f);
365 m_instance->setParameter(kMix, t);
366 setDirty();
367 return true;
368 }
369 }
370
371 // Mix horizontal drag: uses the live cursor x, not the platform's
372 // capture-delta (the horizontal slider does NOT use cursor capture —
373 // that pattern is for knob vertical-drag; capturing here would feel
374 // weird because the cursor moves laterally). Release resets the
375 // drag state so the next press starts fresh.
376 if (m_dragging == 3) {
377 if (event.released) {
378 m_dragging = -1;
379 return true;
380 }
381 if (!event.pressed) {
382 const float t = std::clamp((event.position.x - m_mixRect.x) / m_mixRect.width, 0.0f, 1.0f);
383 m_instance->setParameter(kMix, t);
384 setDirty();
385 return true;
386 }
387 }
388
389 return consumeInsideBounds(event);
390}
