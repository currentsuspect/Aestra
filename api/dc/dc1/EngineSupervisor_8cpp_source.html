13void EngineSupervisor::reportPluginTimeout(uint64_t pluginId, uint64_t timestamp) noexcept {
14 (void)timestamp;
15 m_timeoutCount.fetch_add(1, std::memory_order_relaxed);
16
17 auto profile = m_profile.load(std::memory_order_relaxed);
18 uint32_t threshold;
19 switch (profile) {
20 case RecoveryProfile::LivePerformance: threshold = kLiveTimeoutThreshold; break;
21 case RecoveryProfile::OfflineRender: threshold = kOfflineTimeoutThreshold; break;
22 default: threshold = kStudioTimeoutThreshold; break;
23 }
24
25 // Lock-free: update slot (only safe path on audio thread)
26 getSlot(pluginId).incrementTimeout(threshold);
27
28 // Note: m_pluginHealth (mutex-protected, heap-allocating) is NOT updated from the
29 // audio thread. The slot array is the authoritative lock-free state. Non-RT threads
30 // that need detailed health info can read the slot flags and call getPluginHealth()
31 // which accesses m_pluginHealth under mutex.
32}

58PluginHealthState EngineSupervisor::getPluginHealth(uint64_t pluginId) const {
59 // NOT lock-free: mutex-protected. Only call from non-RT threads.
60 std::lock_guard lock(m_healthMutex);
61 auto it = m_pluginHealth.find(pluginId);
62 if (it == m_pluginHealth.end()) return {};
63 return it->second;

66void EngineSupervisor::decayHealthScores(double decayFactor) {
67 // NOT lock-free: mutex-protected. Only call from non-RT threads.
68
69 // Sanitize decayFactor: NaN/Inf → 0.0, clamp to [0.0, 1.0]
70 if (!std::isfinite(decayFactor)) decayFactor = 0.0;
71 if (decayFactor < 0.0) decayFactor = 0.0;
72 if (decayFactor > 1.0) decayFactor = 1.0;
73
74 std::lock_guard lock(m_healthMutex);
75 for (auto& [id, state] : m_pluginHealth) {
76 // Defensive: reset non-finite scores to zero
77 if (!std::isfinite(state.rollingCpuScore)) {
78 state.rollingCpuScore = 0.0;
79 }
80 state.rollingCpuScore *= decayFactor;
81 if (state.rollingCpuScore < 0.01) {
82 state.consecutiveTimeouts = 0;
83 state.isBypassed = false;
84 // Also clear the lock-free slot
85 getSlot(id).reset();
86 }
87 }
88}
