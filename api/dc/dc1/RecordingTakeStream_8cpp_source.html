16// Mono 32-bit float WAV with the plain 44-byte header: the data chunk starts at
17// a fixed offset, so the two size fields can be rewritten in place each drain.
18constexpr size_t kHeaderBytes = 44;
19constexpr std::streamoff kRiffSizeOffset = 4;
20constexpr std::streamoff kDataSizeOffset = 40;
21constexpr uint16_t kFormatIeeeFloat = 3;
22constexpr uint16_t kChannels = 1;
23constexpr uint16_t kBitsPerSample = 32;
24constexpr uint16_t kBlockAlign = kChannels * (kBitsPerSample / 8);
25// A WAV's size fields are 32-bit; past this the take stays in memory only.
26constexpr uint64_t kMaxDataBytes = 0xFFFFFFFFull - kHeaderBytes;
27constexpr auto kSyncInterval = std::chrono::seconds(1);
28
29void putU16(std::array<char, kHeaderBytes>& h, size_t at, uint16_t v) {
30 std::memcpy(h.data() + at, &v, 2);
31}
32void putU32(std::array<char, kHeaderBytes>& h, size_t at, uint32_t v) {
33 std::memcpy(h.data() + at, &v, 4);
34}

36std::array<char, kHeaderBytes> makeHeader(uint32_t sampleRate, uint32_t dataBytes) {
37 std::array<char, kHeaderBytes> h{};
38 std::memcpy(h.data(), "RIFF", 4);
39 putU32(h, 4, static_cast<uint32_t>(36u + dataBytes));
40 std::memcpy(h.data() + 8, "WAVEfmt ", 8);
41 putU32(h, 16, 16);
42 putU16(h, 20, kFormatIeeeFloat);
43 putU16(h, 22, kChannels);
44 putU32(h, 24, sampleRate);
45 putU32(h, 28, sampleRate * kBlockAlign);
46 putU16(h, 32, kBlockAlign);
47 putU16(h, 34, kBitsPerSample);
48 std::memcpy(h.data() + 36, "data", 4);
49 putU32(h, 40, dataBytes);
50 return h;
51}

69void RecordingTakeStream::openFile() {
70 if (m_partialPath.empty()) {
71 return;
72 }
73 std::error_code ec;
74 const std::filesystem::path path(m_partialPath);
75 if (path.has_parent_path()) {
76 std::filesystem::create_directories(path.parent_path(), ec);
77 }
78 m_file.open(m_partialPath, std::ios::binary | std::ios::trunc);
79 if (!m_file.is_open()) {
80 Log::warning("[Recording] Cannot open in-progress take file; this take is held in memory only: " +
81 m_partialPath);
82 return;
83 }
84 const auto header = makeHeader(m_sampleRate, 0);
85 m_file.write(header.data(), static_cast<std::streamsize>(header.size()));
86 m_fileHealthy = static_cast<bool>(m_file);
87 m_lastSync = std::chrono::steady_clock::now();
88}

90void RecordingTakeStream::drain(const std::atomic<float>* ring, size_t capacity,
91 const std::atomic<uint64_t>& published) {
92 if (!ring || capacity == 0) {
93 return;
94 }
95 const uint64_t end = published.load(std::memory_order_acquire);
96 if (end <= m_drainedFrames) {
97 return;
98 }
99
100 const size_t before = m_samples.size();
101 uint64_t from = m_drainedFrames;
102 // Frames older than one ring length have already been overwritten.
103 if (end - from > capacity) {
104 const uint64_t lost = (end - capacity) - from;
105 m_samples.resize(m_samples.size() + lost, 0.0f);
106 m_lostFrames += lost;
107 from += lost;
108 }
109 for (uint64_t f = from; f < end; ++f) {
110 m_samples.push_back(ring[f % capacity].load(std::memory_order_relaxed));
111 }
112
113 // The audio thread kept writing while we copied. Anything it has wrapped
114 // past since may have been torn mid-copy: silence it rather than keep a
115 // mix of two moments.
116 const uint64_t after = published.load(std::memory_order_acquire);
117 if (after > capacity && after - capacity > from) {
118 const uint64_t tornEnd = std::min(end, after - capacity);
119 const size_t first = before + static_cast<size_t>(from - m_drainedFrames);
120 const size_t count = static_cast<size_t>(tornEnd - from);
121 std::fill_n(m_samples.begin() + static_cast<std::ptrdiff_t>(first), count, 0.0f);
122 m_lostFrames += count;
123 }
124 m_drainedFrames = end;
125 appendToFile(m_samples.data() + before, m_samples.size() - before);
126}

128void RecordingTakeStream::appendToFile(const float* data, size_t count) {
129 if (!partialFileHealthy() || count == 0) {
130 return;
131 }
132 if (static_cast<uint64_t>(m_samples.size()) * sizeof(float) > kMaxDataBytes) {
133 Log::warning("[Recording] Take passed the WAV size limit; the rest stays in memory only: " + m_partialPath);
134 m_fileHealthy = false;
135 return;
136 }
137 m_file.seekp(0, std::ios::end);
138 m_file.write(reinterpret_cast<const char*>(data), static_cast<std::streamsize>(count * sizeof(float)));
139 rewriteHeader();
140 m_file.flush();
141 if (!m_file) {
142 Log::warning("[Recording] Writing the in-progress take failed; it is held in memory only: " + m_partialPath);
143 m_fileHealthy = false;
144 return;
145 }
146 const auto now = std::chrono::steady_clock::now();
147 if (now - m_lastSync >= kSyncInterval) {
148 if (!Aestra::syncOfstream(m_file, m_partialPath)) {
149 Log::warning("[Recording] Syncing the in-progress take failed: " + m_partialPath);
150 }
151 m_lastSync = now;
152 }
153}

155void RecordingTakeStream::rewriteHeader() {
156 const auto dataBytes = static_cast<uint32_t>(m_samples.size() * sizeof(float));
157 const uint32_t riffBytes = 36u + dataBytes;
158 m_file.seekp(kRiffSizeOffset);
159 m_file.write(reinterpret_cast<const char*>(&riffBytes), sizeof(riffBytes));
160 m_file.seekp(kDataSizeOffset);
161 m_file.write(reinterpret_cast<const char*>(&dataBytes), sizeof(dataBytes));
162 m_file.seekp(0, std::ios::end);
163}

176int64_t RecordingTakeStream::readPartialFrameCount(const std::string& path) {
177 std::ifstream in(path, std::ios::binary);
178 std::array<char, kHeaderBytes> h{};
179 if (!in.read(h.data(), static_cast<std::streamsize>(h.size()))) {
180 return -1;
181 }
182 if (std::memcmp(h.data(), "RIFF", 4) != 0 || std::memcmp(h.data() + 8, "WAVEfmt ", 8) != 0 ||
183 std::memcmp(h.data() + 36, "data", 4) != 0) {
184 return -1;
185 }
186 uint32_t dataBytes = 0;
187 std::memcpy(&dataBytes, h.data() + 40, 4);
188 return static_cast<int64_t>(dataBytes / kBlockAlign);
189}
