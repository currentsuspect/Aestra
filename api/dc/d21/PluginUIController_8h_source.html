198 // Wire an editor's close callback. Captures the editor by weak_ptr (so the
199 // callback does not form a retain cycle that leaks the editor), and removes
200 // it from the popup layer and the active-editor list on close. removeChild()
201 // is deferred while an event dispatch is in flight, so the removal is safe
202 // even though close fires from within the editor's own onMouseEvent.
203 template <typename EditorT>
204 void wireEditorClose(const std::shared_ptr<EditorT>& editor) {
205 std::weak_ptr<EditorT> weak = editor;
206 editor->setOnClose([this, weak]() {
207 auto ed = weak.lock();
208 if (!ed) return;
209 std::shared_ptr<NUIComponent> comp = ed;
210 if (m_popupLayer) m_popupLayer->removeChild(comp);
211 m_activeEditors.erase(std::remove(m_activeEditors.begin(), m_activeEditors.end(), comp),
212 m_activeEditors.end());
213 });
214 }
215
216 // Backend references
217 Aestra::Audio::PluginScanner* m_scanner = nullptr;
218 Aestra::Audio::PluginManager* m_manager = nullptr;
219
220 // Platform bridge for cursor capture
221 NUIPlatformBridge* m_platformBridge = nullptr;
222
223 // Bound widgets
224 PluginBrowserPanel* m_browser = nullptr;

226 // Rack bindings (rack -> chain)
