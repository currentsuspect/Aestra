258 // Wire an editor's close callback. Captures the editor by weak_ptr (so the
259 // callback does not form a retain cycle that leaks the editor), and removes
260 // it from the popup layer and the active-editor list on close. removeChild()
261 // is deferred while an event dispatch is in flight, so the removal is safe
262 // even though close fires from within the editor's own onMouseEvent.
263 template <typename EditorT>
264 void wireEditorClose(const std::shared_ptr<EditorT>& editor) {
265 std::weak_ptr<EditorT> weak = editor;
266 editor->setOnClose([this, weak]() {
267 auto ed = weak.lock();
268 if (!ed) return;
269 std::shared_ptr<NUIComponent> comp = ed;
270 if (m_popupLayer) m_popupLayer->removeChild(comp);
271 m_activeEditors.erase(std::remove(m_activeEditors.begin(), m_activeEditors.end(), comp),
272 m_activeEditors.end());
273 });
274 }
