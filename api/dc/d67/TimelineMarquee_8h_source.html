3// TimelineMarquee — widget-independent state machine for the timeline's
4// marquee (multi-select) drag (#847).
5//
6// Header-only and widget-free by design, like TimelineInteractionPolicy.h:
7// the drag contract is reachable headless (AESTRA_CI=ON disables linking
8// AestraUI) so the two failure modes that presented as an app hang are pinned
9// by tests:
10// 1. An active marquee OWNS every mouse event until the button that began
11// it is released — the pointer can never escape a drag, and a foreign
12// button can never finalize someone else's gesture or move its band.
13// 2. Per-move work is pure bookkeeping (endpoint update only). This type
14// carries no invalidation, cache, or render surface at all, so a move
15// cannot trigger a per-move invalidation storm.
16//
17// The widget maps NUIMouseButton onto MarqueeButton tokens, clamps cursor
18// positions into the grid area, and routes an active marquee BEFORE sibling
19// controls (the minimap consumes in-bounds releases even when idle, which
20// would strand the gesture) — clamping and routing policy stay with the
21// widget, sequencing rules live here.

36enum class MarqueeEventKind {
37 Move, // pointer move/drag: endpoint follows the position
38 Press, // button down: consumed, no state change
39 Release, // button up: endpoint follows, finalizes when initiating button
40 Other, // scroll, enter/leave, synthetic: consumed, no state change
41};

44enum class MarqueeOrigin {
45 Tool, // the Multi-Select tool: either button, anywhere in the track area
46 Grid, // a left press on empty grid in the Select tool: the default gesture
47};

65constexpr MarqueeReleasePlan marqueeReleasePlan(MarqueeOrigin origin, float width, float height) {
66 if (origin == MarqueeOrigin::Grid) {
67 const bool click = width < kGridMarqueeClickSlopPx && height < kGridMarqueeClickSlopPx;
68 return {!click, false};
69 }
70 return {true, true};
71}

73struct TimelineMarqueeDrag {
76 bool begin(bool pressed, MarqueeButton btn, float x, float y, bool inTrackArea) {
77 if (!pressed || !inTrackArea || m_active || btn == 0) {
78 return false;
79 }
80 m_active = true;
81 m_button = btn;
82 m_startX = x;
83 m_startY = y;
84 m_endX = x;
85 m_endY = y;
86 return true;
87 }
88
90 bool ownsEvent() const { return m_active; }
91
96 bool onEvent(MarqueeEventKind kind, MarqueeButton btn, float x, float y) {
97 if (!m_active) {
98 return false;
99 }
100 if (kind == MarqueeEventKind::Release) {
101 if (btn != m_button) {
102 return false; // foreign release: consumed, endpoint untouched
103 }
104 m_endX = x;
105 m_endY = y;
106 return true;
107 }
108 if (kind == MarqueeEventKind::Move) {
109 m_endX = x;
110 m_endY = y;
111 }
112 return false; // Press and Other: consumed, no state change
113 }
114
116 void finalize() {
117 m_active = false;
118 m_button = 0;
119 }
120
121 // ── Read-only queries (consumers + tests) ──
122 bool active() const { return m_active; }
123 MarqueeButton button() const { return m_button; }
124 float startX() const { return m_startX; }
125 float startY() const { return m_startY; }
126 float endX() const { return m_endX; }
127 float endY() const { return m_endY; }
128
131 float rectMinX() const { return std::min(m_startX, m_endX); }
132 float rectMinY() const { return std::min(m_startY, m_endY); }
133 float rectMaxX() const { return std::max(m_startX, m_endX); }
134 float rectMaxY() const { return std::max(m_startY, m_endY); }
135 float rectWidth() const { return rectMaxX() - rectMinX(); }
136 float rectHeight() const { return rectMaxY() - rectMinY(); }
137
138private:
139 bool m_active = false;
140 MarqueeButton m_button = 0; // the button that began the drag
141 float m_startX = 0.0f;
142 float m_startY = 0.0f;
143 float m_endX = 0.0f;
144 float m_endY = 0.0f;
145};
