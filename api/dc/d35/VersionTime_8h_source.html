50inline std::string ago(uint64_t epochMs, uint64_t nowMs) {
51 if (epochMs == 0) return {};
52 const std::tm then = local(epochMs), now = local(nowMs);
53 const int days = (now.tm_year - then.tm_year) * 365 + (now.tm_yday - then.tm_yday);
54 char buf[48];
55 if (days <= 0) { std::snprintf(buf, sizeof buf, "today %02d:%02d", then.tm_hour, then.tm_min); return buf; }
56 if (days == 1) return "yesterday";
57 if (days < 7) return std::to_string(days) + " days ago";
58 static const char* months[] = {"Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"};
59 std::snprintf(buf, sizeof buf, "%s %d", months[std::clamp(then.tm_mon, 0, 11)], then.tm_mday);
