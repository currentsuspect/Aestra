26bool framesToSamples(uint64_t frames, uint32_t channels, size_t& outSamples) {
27 if (channels == 0) {
28 return false;
29 }
30 if (frames > std::numeric_limits<size_t>::max() / channels) {
31 return false;
32 }
33 outSamples = static_cast<size_t>(frames) * channels;
34 return true;
35}

39ClipRenderService::SourceRegion ClipRenderService::resolveClipRegion(const ClipInstance& clip,
40 double projectSampleRate) const {
41 SourceRegion region;
42 if (!clip.patternId.isValid()) {
43 return region;
44 }
45 const PatternSource* pattern = m_patterns.getPattern(clip.patternId);
46 if (!pattern || !pattern->isAudio()) {
47 return region;
48 }
49
50 const auto& payload = std::get<AudioSlicePayload>(pattern->payload);
51 const ClipSource* source = m_sources.getSource(payload.audioSourceId);
52 if (!source) {
53 return region;
54 }
55 auto buffer = source->getSharedBuffer();
56 if (!buffer || !buffer->isValid()) {
57 return region;
58 }
59
60 // Patterns written by import and by recording both carry one full-extent
61 // slice; a missing or empty slice means "the whole source".
62 uint64_t startFrame = 0;
63 uint64_t frameCount = buffer->numFrames;
64 if (!payload.slices.empty()) {
65 const AudioSlice& slice = payload.slices.front();
66 if (std::isfinite(slice.startSamples) && slice.startSamples > 0.0) {
67 startFrame = static_cast<uint64_t>(slice.startSamples);
68 }
69 if (std::isfinite(slice.lengthSamples) && slice.lengthSamples > 0.0) {
70 frameCount = static_cast<uint64_t>(slice.lengthSamples);
71 }
72 }
73 // Same offsets the runtime snapshot applies (PlaylistModel::buildSnapshot):
74 // the canonical per-clip offset, then the instance slip. sourceStart is in
75 // project-rate samples, so convert it into this source's frames.
76 const double sourceRate = static_cast<double>(buffer->sampleRate);
77 double offsetFrames = 0.0;
78 if (clip.durationSeconds > 0.0 && std::isfinite(clip.sourceOffsetSeconds) && clip.sourceOffsetSeconds > 0.0) {
79 offsetFrames += clip.sourceOffsetSeconds * sourceRate;
80 }
81 if (std::isfinite(clip.edits.sourceStart) && clip.edits.sourceStart > 0.0) {
82 const double rateScale = (projectSampleRate > 0.0) ? sourceRate / projectSampleRate : 1.0;
83 offsetFrames += clip.edits.sourceStart * rateScale;
84 }
85 if (offsetFrames > 0.0) {
86 const double maxOffset = static_cast<double>(buffer->numFrames);
87 startFrame += static_cast<uint64_t>(std::min(offsetFrames, maxOffset));
88 }
89
90 if (startFrame >= buffer->numFrames) {
91 return region;
92 }
93 frameCount = std::min(frameCount, buffer->numFrames - startFrame);
94
95 // A trimmed clip plays only its own duration, so that is what gets
96 // rendered; otherwise reversing a trim would drag in audio the user
97 // cannot hear.
98 if (clip.durationSeconds > 0.0 && std::isfinite(clip.durationSeconds)) {
99 const double audibleFrames = clip.durationSeconds * sourceRate;
100 if (audibleFrames >= 1.0) {
101 frameCount = std::min(frameCount, static_cast<uint64_t>(audibleFrames));
102 }
103 }
104
105 region.buffer = std::move(buffer);
106 region.startFrame = startFrame;
107 region.frameCount = frameCount;
108 region.mixerChannelId = pattern->getMixerChannelId();
109 region.lengthBeats = pattern->lengthBeats;
110 region.name = pattern->name.empty() ? clip.name : pattern->name;
111 return region;
112}

114std::shared_ptr<AudioBufferData> ClipRenderService::extractRegion(const AudioBufferData& source, uint64_t startFrame,
115 uint64_t frameCount) {
116 if (!source.isValid() || frameCount == 0 || startFrame >= source.numFrames) {
117 return nullptr;
118 }
119
120 // Clamp an overhanging range instead of reading past the buffer. A clip
121 // whose length outlives its source is normal after tempo edits.
122 const uint64_t available = source.numFrames - startFrame;
123 const uint64_t framesToCopy = std::min(frameCount, available);
124
125 size_t sampleCount = 0;
126 size_t sampleOffset = 0;
127 if (!framesToSamples(framesToCopy, source.numChannels, sampleCount) ||
128 !framesToSamples(startFrame, source.numChannels, sampleOffset)) {
129 return nullptr;
130 }
131 if (sampleOffset + sampleCount > source.interleavedData.size()) {
132 return nullptr;
133 }
134
135 auto out = std::make_shared<AudioBufferData>();
136 out->sampleRate = source.sampleRate;
137 out->numChannels = source.numChannels;
138 out->numFrames = framesToCopy;
139 out->interleavedData.assign(source.interleavedData.begin() + static_cast<std::ptrdiff_t>(sampleOffset),
140 source.interleavedData.begin() +
141 static_cast<std::ptrdiff_t>(sampleOffset + sampleCount));
142 return out;
143}

145void ClipRenderService::reverseInPlace(AudioBufferData& buffer) {
146 if (!buffer.isValid() || buffer.numFrames < 2) {
147 return;
148 }
149 const uint32_t channels = buffer.numChannels;
150 size_t usableSamples = 0;
151 if (!framesToSamples(buffer.numFrames, channels, usableSamples) ||
152 usableSamples > buffer.interleavedData.size()) {
153 return;
154 }
155
156 // Swap whole frames so a stereo image survives the reverse; reversing the
157 // flat sample array would also swap L and R.
158 float* data = buffer.interleavedData.data();
159 for (uint64_t front = 0, back = buffer.numFrames - 1; front < back; ++front, --back) {
160 float* a = data + static_cast<size_t>(front) * channels;
161 float* b = data + static_cast<size_t>(back) * channels;
162 for (uint32_t ch = 0; ch < channels; ++ch) {
163 std::swap(a[ch], b[ch]);
164 }
165 }
166}

169 if (!std::isfinite(gainLinear) || gainLinear == 1.0f) {
170 return;
171 }
172 for (float& sample : buffer.interleavedData) {
173 sample *= gainLinear;

177void ClipRenderService::applyFades(AudioBufferData& buffer, uint64_t fadeInFrames, uint64_t fadeOutFrames) {
178 if (!buffer.isValid() || buffer.numFrames == 0) {
179 return;
180 }
181 const uint32_t channels = buffer.numChannels;
182 size_t usableSamples = 0;
183 if (!framesToSamples(buffer.numFrames, channels, usableSamples) ||
184 usableSamples > buffer.interleavedData.size()) {
185 return;
186 }
187
188 // Clamp BEFORE any arithmetic on the pair: a fade longer than the clip is
189 // legal input (a corrupt project, or a tempo change shrinking the clip),
190 // and summing two unclamped uint64 lengths can wrap past the guard and
191 // send the ramp loop off the end of the buffer.
192 const uint64_t fadeIn = std::min(fadeInFrames, buffer.numFrames);
193 const uint64_t fadeOut = std::min(fadeOutFrames, buffer.numFrames);
194 if (fadeIn == 0 && fadeOut == 0) {
195 return;
196 }
197
198 // Mirrors AudioEngine's clipFadeAt so a committed clip sounds like the one
199 // that was playing: the fade-in starts at zero, the fade-out's seam sample
200 // stays at unity (strict >), and overlapping ramps take the MINIMUM rather
201 // than multiplying into a notch.
202 //
203 // The engine's CLIP_EDGE_FADE_SAMPLES micro-fade is deliberately not baked
204 // in: it is a playback-time anti-click that still applies to the committed
205 // clip, so baking it too would attenuate those 128 samples twice.
206 float* data = buffer.interleavedData.data();
207 for (uint64_t frame = 0; frame < buffer.numFrames; ++frame) {
208 double gain = 1.0;
209 if (fadeIn > 0 && frame < fadeIn) {
210 gain = std::min(gain, static_cast<double>(frame) / static_cast<double>(fadeIn));
211 }
212 if (fadeOut > 0 && frame + fadeOut > buffer.numFrames) {
213 gain = std::min(gain, static_cast<double>(buffer.numFrames - frame) / static_cast<double>(fadeOut));
214 }
215 if (gain >= 1.0) {
216 continue;
217 }
218 float* f = data + static_cast<size_t>(frame) * channels;
219 for (uint32_t ch = 0; ch < channels; ++ch) {
220 f[ch] = static_cast<float>(static_cast<double>(f[ch]) * gain);
221 }
222 }

225float ClipRenderService::peakMagnitude(const AudioBufferData& buffer) {
226 float peak = 0.0f;
227 for (const float sample : buffer.interleavedData) {
228 if (std::isfinite(sample)) {
229 peak = std::max(peak, std::fabs(sample));
230 }
231 }
232 return peak;
233}

235std::string ClipRenderService::uniqueRenderPath(const std::string& directory, const std::string& baseName) {
236 namespace fs = std::filesystem;
237
238 // Strip anything that would be awkward in a filename; the pattern name is
239 // user-supplied and may contain separators.
240 std::string safe;
241 safe.reserve(baseName.size());
242 for (const char c : baseName) {
243 const bool ok = (c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') || c == '-' ||
244 c == '_' || c == ' ';
245 safe.push_back(ok ? c : '_');
246 }
247 if (safe.empty()) {
248 safe = "render";
249 }
250
251 const auto now = std::chrono::system_clock::now();
252 const auto time = std::chrono::system_clock::to_time_t(now);
253 std::tm tm{};
254#if defined(_WIN32)
255 localtime_s(&tm, &time);
256#else
257 localtime_r(&time, &tm);
258#endif
259
260 static std::atomic<uint64_t> s_counter{0};
261 const uint64_t uniq = s_counter.fetch_add(1, std::memory_order_relaxed);
262
263 std::ostringstream oss;
264 oss << safe << "_" << std::put_time(&tm, "%Y%m%d_%H%M%S") << "_" << uniq << ".wav";
265 return (fs::path(directory) / oss.str()).string();
266}

268bool ClipRenderService::writeFloatWav(const std::string& path, const AudioBufferData& buffer) {
269 if (buffer.numChannels == 0 || buffer.numChannels > std::numeric_limits<uint16_t>::max() ||
270 buffer.sampleRate == 0 || buffer.numFrames > std::numeric_limits<uint32_t>::max() ||
271 buffer.interleavedData.size() > (std::numeric_limits<uint32_t>::max() / sizeof(float))) {
272 return false;
273 }
274
275 const uint16_t audioFormat = 3; // IEEE float
276 const uint16_t numChannels = static_cast<uint16_t>(buffer.numChannels);
277 const uint32_t sampleRate = buffer.sampleRate;
278 const uint16_t bitsPerSample = 32;
279 const uint16_t blockAlign = static_cast<uint16_t>(numChannels * (bitsPerSample / 8));
280 if (blockAlign == 0 || sampleRate > std::numeric_limits<uint32_t>::max() / blockAlign) {
281 return false;
282 }
283 const uint32_t byteRate = sampleRate * blockAlign;
284 const uint32_t dataSize = static_cast<uint32_t>(buffer.interleavedData.size() * sizeof(float));
285 const uint32_t sampleCount = static_cast<uint32_t>(buffer.numFrames);
286 const uint32_t factChunkSize = 4;
287 if (dataSize > std::numeric_limits<uint32_t>::max() - 48u) {
288 return false;
289 }
290 const uint32_t riffChunkSize = 48u + dataSize;
291 const uint32_t fmtChunkSize = 16;
292
293 std::ofstream file(path, std::ios::binary | std::ios::trunc);
294 if (!file.is_open()) {
295 return false;
296 }
297
298 file.write("RIFF", 4);
299 file.write(reinterpret_cast<const char*>(&riffChunkSize), sizeof(riffChunkSize));
300 file.write("WAVE", 4);
301 file.write("fmt ", 4);
302 file.write(reinterpret_cast<const char*>(&fmtChunkSize), sizeof(fmtChunkSize));
303 file.write(reinterpret_cast<const char*>(&audioFormat), sizeof(audioFormat));
304 file.write(reinterpret_cast<const char*>(&numChannels), sizeof(numChannels));
305 file.write(reinterpret_cast<const char*>(&sampleRate), sizeof(sampleRate));
306 file.write(reinterpret_cast<const char*>(&byteRate), sizeof(byteRate));
307 file.write(reinterpret_cast<const char*>(&blockAlign), sizeof(blockAlign));
308 file.write(reinterpret_cast<const char*>(&bitsPerSample), sizeof(bitsPerSample));
309 file.write("fact", 4);
310 file.write(reinterpret_cast<const char*>(&factChunkSize), sizeof(factChunkSize));
311 file.write(reinterpret_cast<const char*>(&sampleCount), sizeof(sampleCount));
312 file.write("data", 4);
313 file.write(reinterpret_cast<const char*>(&dataSize), sizeof(dataSize));
314 file.write(reinterpret_cast<const char*>(buffer.interleavedData.data()), dataSize);
315
316 return file.good();
317}

319ClipRenderService::CommitResult ClipRenderService::commit(const AudioBufferData& buffer,
320 const std::string& renderDirectory,
321 const std::string& baseName, double lengthBeats,
322 uint32_t mixerChannelId) {
323 CommitResult result;
324 if (!buffer.isValid() || buffer.numFrames == 0) {
325 return result;
326 }
327
328 namespace fs = std::filesystem;
329 std::error_code ec;
330 fs::create_directories(renderDirectory, ec);
331 if (ec) {
332 Log::error("[ClipRenderService] Could not create render directory: " + renderDirectory);
333 return result;
334 }
335
336 const std::string path = uniqueRenderPath(renderDirectory, baseName);
337 if (!writeFloatWav(path, buffer)) {
338 Log::error("[ClipRenderService] Failed to write rendered audio: " + path);
339 // A partial file would look like a valid source on the next load.
340 fs::remove(path, ec);
341 return result;
342 }
343
344 // Hand the in-memory copy straight to the source so playback does not have
345 // to wait for a decode of the file we just wrote.
346 auto owned = std::make_shared<AudioBufferData>(buffer);
347 const ClipSourceID sourceId = m_sources.createRecordedSource(path, baseName, std::move(owned));
348 if (!sourceId.isValid()) {
349 Log::error("[ClipRenderService] Failed to register rendered source: " + path);
350 fs::remove(path, ec);
351 return result;
352 }
353
354 AudioSlicePayload payload;
355 payload.audioSourceId = sourceId;
356 payload.durationSeconds = buffer.durationSeconds();
357 AudioSlice fullSlice;
358 fullSlice.startSamples = 0.0;
359 fullSlice.lengthSamples = static_cast<double>(buffer.numFrames);
360 payload.slices.push_back(fullSlice);
361
362 const double beats = (std::isfinite(lengthBeats) && lengthBeats > 0.0) ? lengthBeats : 4.0;
363 const PatternID patternId = m_patterns.createAudioPattern(baseName, beats, payload);
364 if (!patternId.isValid()) {
365 Log::error("[ClipRenderService] Failed to create pattern for rendered audio: " + path);
366 return result;
367 }
368 m_patterns.setPatternMixerChannel(patternId, mixerChannelId);
369
370 result.patternId = patternId;
371 result.sourceId = sourceId;
372 result.filePath = path;
373 return result;
374}
