25void PluginBrowserPanel::loadFavorites() {
26 m_favoritesSet.clear();
27 auto path = getFavoritesPath();
28 std::ifstream f(path);
29 if (!f.is_open()) return;
30 std::string content((std::istreambuf_iterator<char>(f)), std::istreambuf_iterator<char>());
31 f.close();
32 if (content.empty()) return;
33 try {
34 Aestra::JSON j = Aestra::JSON::parse(content);
35 if (j.has("favorites") && j["favorites"].isArray()) {
36 auto arr = j["favorites"].asArray();
37 for (size_t i = 0; i < arr.size(); ++i) {
38 if (arr[i].isString()) {
39 m_favoritesSet.insert(arr[i].asString());
40 }
41 }
42 }
43 } catch (...) {}
44}

46void PluginBrowserPanel::setFavoritesProvider(FavoritesLoader load, FavoritesSaver save) {
47 m_favoritesSaver = std::move(save);
48 if (const auto stored = load ? load() : std::nullopt) {
49 m_favoritesSet = std::unordered_set<std::string>(stored->begin(), stored->end());
50 for (auto* list : {&m_allPlugins, &m_filteredPlugins}) {
51 for (auto& p : *list) p.isFavorite = m_favoritesSet.count(p.id) > 0;
52 }
53 applyFilters();
54 } else if (m_favoritesSaver && !m_favoritesSet.empty()) {
55 saveFavorites(); // one-way import of favorites.json into the store
56 }
57}

59void PluginBrowserPanel::saveFavorites() {
60 if (m_favoritesSaver) {
61 std::vector<std::string> ids(m_favoritesSet.begin(), m_favoritesSet.end());
62 std::sort(ids.begin(), ids.end()); // a stable order on disk
63 m_favoritesSaver(ids);
64 return;
65 }
66 Aestra::JSON j = Aestra::JSON::object();
67 j.set("version", Aestra::JSON(1.0));
68 Aestra::JSON arr = Aestra::JSON::array();
69 for (const auto& id : m_favoritesSet) {
70 arr.push(Aestra::JSON(id));
71 }
72 j.set("favorites", arr);
73 std::ofstream f(getFavoritesPath());
74 if (f.is_open()) f << j.toString(2);
75}

77void PluginBrowserPanel::toggleFavorite(const std::string& pluginId) {
78 if (m_favoritesSet.count(pluginId)) {
79 m_favoritesSet.erase(pluginId);
80 } else {
81 m_favoritesSet.insert(pluginId);
82 }
83 bool isFav = m_favoritesSet.count(pluginId) > 0;
84 auto patch = [&](std::vector<PluginListItem>& list) {
85 for (auto& p : list)
86 if (p.id == pluginId) { p.isFavorite = isFav; break; }
87 };
88 patch(m_allPlugins);
89 patch(m_filteredPlugins);
90 saveFavorites();
91 applyFilters();
92}

101int PluginBrowserPanel::recentRank(const std::string& id) const {
102 const auto it = std::find(m_recents.begin(), m_recents.end(), id);
103 return it == m_recents.end() ? -1 : static_cast<int>(it - m_recents.begin());

106void PluginBrowserPanel::recordRecent(const std::string& pluginId) {
107 if (pluginId.empty()) return;
108 m_recents.erase(std::remove(m_recents.begin(), m_recents.end(), pluginId), m_recents.end());
109 m_recents.insert(m_recents.begin(), pluginId);
110 if (m_recents.size() > kMaxRecents) m_recents.resize(kMaxRecents);
111 if (m_recentsSaver) m_recentsSaver(m_recents);
112}
