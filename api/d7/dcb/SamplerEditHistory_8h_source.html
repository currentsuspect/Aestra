29 bool commit(Audio::TrackManager& tm, std::function<void()> refreshEditor) {
30 if (m_unit == 0) return false;
31 tm.getUnitManager().captureUnitPluginState(m_unit);
32 tm.markModified();
33 auto sampler = sampler_(tm, m_unit);
34 if (!sampler || !m_has) return false;
35 const auto now = Audio::SamplerEditState::capture(*sampler);
36 if (now == m_baseline) return false;
37 const bool recorded = tm.getCommandHistory().pushExecuted(
38 std::make_shared<Audio::SetSamplerStateCommand>(tm.getUnitManager(), m_unit, m_baseline, now, std::move(refreshEditor)));
39 m_baseline = now;
40 return recorded;
41 }
42
43private:

46 }
47 Audio::UnitID m_unit{0};
48 bool m_has{false};
