28class ScheduleSnapshotPublisher {
29public:
31 void publish(const ScheduleSnapshot& snapshot) noexcept {
32 const uint32_t seq = m_seq.load(std::memory_order_relaxed);
33 m_seq.store(seq + 1, std::memory_order_relaxed);
34 std::atomic_thread_fence(std::memory_order_release);
35 m_frame.store(snapshot.frame, std::memory_order_relaxed);
36 m_loopStart.store(snapshot.loop.start, std::memory_order_relaxed);
37 m_loopLength.store(snapshot.loop.length, std::memory_order_relaxed);
38 m_timeline.store(snapshot.loop.timeline, std::memory_order_relaxed);
39 m_epoch.store(snapshot.epoch, std::memory_order_relaxed);
40 m_seq.store(seq + 2, std::memory_order_release);
41 }
42
44 bool hasPublished() const noexcept { return m_seq.load(std::memory_order_acquire) != 0; }
45
47 bool read(ScheduleSnapshot& out) const noexcept {
48 for (int attempt = 0; attempt < 64; ++attempt) {
49 const uint32_t before = m_seq.load(std::memory_order_acquire);
50 if (before == 0) {
51 return false;
52 }
53 if ((before & 1u) != 0) {
54 continue;
55 }
56 ScheduleSnapshot snapshot;
57 snapshot.frame = m_frame.load(std::memory_order_relaxed);
58 snapshot.loop.start = m_loopStart.load(std::memory_order_relaxed);
59 snapshot.loop.length = m_loopLength.load(std::memory_order_relaxed);
60 snapshot.loop.timeline = m_timeline.load(std::memory_order_relaxed);
61 snapshot.epoch = m_epoch.load(std::memory_order_relaxed);
62 std::atomic_thread_fence(std::memory_order_acquire);
63 if (m_seq.load(std::memory_order_relaxed) == before) {
64 out = snapshot;
65 return true;
66 }
67 }
68 return false;
69 }
70
71private:
72 std::atomic<uint32_t> m_seq{0};
73 std::atomic<uint64_t> m_frame{0};
74 std::atomic<uint64_t> m_loopStart{0};
75 std::atomic<uint64_t> m_loopLength{0};
76 std::atomic<bool> m_timeline{false};
77 std::atomic<uint16_t> m_epoch{0};
78};
