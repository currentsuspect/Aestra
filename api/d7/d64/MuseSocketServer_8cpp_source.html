48// Every accepted socket is non-blocking so no send()/recv() can park the IO thread.
49bool setNonBlocking(SocketHandle socket) {
50#ifdef _WIN32
51 u_long enable = 1;
52 return ::ioctlsocket(socket, FIONBIO, &enable) == 0;
53#else
54 const int flags = ::fcntl(socket, F_GETFL, 0);
55 return flags >= 0 && ::fcntl(socket, F_SETFL, flags | O_NONBLOCK) == 0;
56#endif
57}

59// True when the last socket call failed only because it would have blocked.
60bool wouldBlock() {
61#ifdef _WIN32
62 return ::WSAGetLastError() == WSAEWOULDBLOCK;
63#else
64 return errno == EAGAIN || errno == EWOULDBLOCK || errno == EINTR;
65#endif
66}

98struct MuseSocketServer::Impl {
99 SocketHandle listenSocket = kInvalidSocket;
100 uint16_t boundPort = 0;
101 std::thread ioThread;
102 std::atomic<bool> running{false};
103
104 struct Client {
105 SocketHandle socket = kInvalidSocket;
106 std::string readBuffer;
107 std::string writeBuffer; // IO-thread only
108 };
109
110 // IO-thread state
111 std::unordered_map<uint64_t, Client> clients;
112 uint64_t nextClientId = 1;
113
114 // Cross-thread queues
115 std::mutex mutex;
116 std::deque<std::pair<uint64_t, std::string>> inbox; // client -> request line
117 std::deque<std::pair<uint64_t, std::string>> outbox; // client -> response line
118
119 void ioLoop() {
120 while (running.load(std::memory_order_relaxed)) {
121 // Backpressure (#1271): while the inbox is full, stop reading from
122 // every client. Unread bytes stay in the kernel receive buffer, so
123 // a flooding client is slowed by TCP rather than by our memory.
124 bool inboxHasRoom;
125 {
126 std::lock_guard<std::mutex> lock(mutex);
127 inboxHasRoom = inbox.size() < kMaxInboxRequests;
128 }
129
130 fd_set readSet;
131 fd_set writeSet;
132 FD_ZERO(&readSet);
133 FD_ZERO(&writeSet);
134 FD_SET(listenSocket, &readSet);
135 SocketHandle maxFd = listenSocket;
136 for (const auto& [clientId, client] : clients) {
137 if (inboxHasRoom) FD_SET(client.socket, &readSet);
138 // Wake as soon as a client with pending output can take more.
139 if (!client.writeBuffer.empty()) FD_SET(client.socket, &writeSet);
140 if (client.socket > maxFd) maxFd = client.socket;
141 }
142
143 // Short timeout so queued responses and stop() are picked up
144 // promptly even when no traffic arrives. Every socket is
145 // non-blocking, so nothing below can park the thread and stop()
146 // is always observed within one timeout.
147 timeval timeout{};
148 timeout.tv_usec = 50 * 1000;
149 const int ready =
150 ::select(static_cast<int>(maxFd + 1), &readSet, &writeSet, nullptr, &timeout);
151 if (!running.load(std::memory_order_relaxed)) break;
152
153 if (ready > 0 && FD_ISSET(listenSocket, &readSet)) {
154 sockaddr_in remote{};
155#ifdef _WIN32
156 int remoteLen = sizeof(remote);
157#else
158 socklen_t remoteLen = sizeof(remote);
159#endif
160 SocketHandle accepted =
161 ::accept(listenSocket, reinterpret_cast<sockaddr*>(&remote), &remoteLen);
162 if (accepted != kInvalidSocket) {
163 if (setNonBlocking(accepted)) {
164 Client client;
165 client.socket = accepted;
166 clients.emplace(nextClientId++, std::move(client));
167 } else {
168 closeSocket(accepted);
169 }
170 }
171 }
172
173 std::vector<uint64_t> disconnected;
174 for (auto& [clientId, client] : clients) {
175 if (inboxHasRoom && ready > 0 && FD_ISSET(client.socket, &readSet)) {
176 char chunk[4096];
177 const auto received = ::recv(client.socket, chunk, sizeof(chunk), 0);
178 if (received == 0 || (received < 0 && !wouldBlock())) {
179 disconnected.push_back(clientId);
180 continue;
181 }
182 if (received > 0) client.readBuffer.append(chunk, static_cast<size_t>(received));
183 }
184
185 // Move complete lines to the inbox while it has room; any left
186 // over stay in readBuffer (bounded below) for a later pass.
187 size_t newline;
188 while ((newline = client.readBuffer.find('\n')) != std::string::npos) {
189 {
190 std::lock_guard<std::mutex> lock(mutex);
191 if (inbox.size() >= kMaxInboxRequests) break;
192 std::string line = client.readBuffer.substr(0, newline);
193 client.readBuffer.erase(0, newline + 1);
194 if (!line.empty() && line.back() == '\r') line.pop_back();
195 if (!line.empty()) inbox.emplace_back(clientId, std::move(line));
196 }
197 }
198 // A runaway line without a newline must not grow forever.
199 if (client.readBuffer.size() > kMaxLineBytes) {
200 disconnected.push_back(clientId);
201 }
202 }
203
204 // Deliver queued responses.
205 {
206 std::deque<std::pair<uint64_t, std::string>> toWrite;
207 {
208 std::lock_guard<std::mutex> lock(mutex);
209 toWrite.swap(outbox);
210 }
211 for (auto& [clientId, response] : toWrite) {
212 auto it = clients.find(clientId);
213 if (it == clients.end()) continue; // client already gone
214 it->second.writeBuffer += response;
215 // A client that will not read must not grow memory forever.
216 if (it->second.writeBuffer.size() > kMaxClientWriteBytes) {
217 disconnected.push_back(clientId);
218 }
219 }
220 }
221 for (auto& [clientId, client] : clients) {
222 while (!client.writeBuffer.empty()) {
223 const auto sent = ::send(client.socket, client.writeBuffer.data(),
224#ifdef _WIN32
225 static_cast<int>(client.writeBuffer.size()),
226#else
227 client.writeBuffer.size(),
228#endif
229 kSendFlags);
230 if (sent < 0 && wouldBlock()) break; // keep the rest for next time
231 if (sent <= 0) {
232 disconnected.push_back(clientId);
233 break;
234 }
235 client.writeBuffer.erase(0, static_cast<size_t>(sent));
236 }
237 }
238
239 for (uint64_t clientId : disconnected) {
240 auto it = clients.find(clientId);
241 if (it != clients.end()) {
242 closeSocket(it->second.socket);
243 clients.erase(it);
244 }
245 }
246 }
247
248 for (auto& [clientId, client] : clients) {
249 closeSocket(client.socket);
250 }
251 clients.clear();
252 }
253};

261bool MuseSocketServer::start(uint16_t port, std::string& outError) {
262 if (m_impl->running.load(std::memory_order_relaxed)) {
263 outError = "already running";
264 return false;
265 }
266 if (!initSockets(outError)) return false;
267
268 m_impl->listenSocket = ::socket(AF_INET, SOCK_STREAM, 0);
269 if (m_impl->listenSocket == kInvalidSocket) {
270 outError = "socket() failed";
271 return false;
272 }
273
274 // Loopback only: Muse is a local control surface, never a network service.
275 sockaddr_in address{};
276 address.sin_family = AF_INET;
277 address.sin_port = htons(port);
278 address.sin_addr.s_addr = htonl(INADDR_LOOPBACK);
279
280 int reuse = 1;
281 ::setsockopt(m_impl->listenSocket, SOL_SOCKET, SO_REUSEADDR,
282 reinterpret_cast<const char*>(&reuse), sizeof(reuse));
283
284 if (::bind(m_impl->listenSocket, reinterpret_cast<sockaddr*>(&address), sizeof(address)) !=
285 0) {
286 outError = "bind to 127.0.0.1:" + std::to_string(port) + " failed";
287 closeSocket(m_impl->listenSocket);
288 m_impl->listenSocket = kInvalidSocket;
289 return false;
290 }
291 if (::listen(m_impl->listenSocket, 4) != 0) {
292 outError = "listen failed";
293 closeSocket(m_impl->listenSocket);
294 m_impl->listenSocket = kInvalidSocket;
295 return false;
296 }
297
298 sockaddr_in bound{};
299#ifdef _WIN32
300 int boundLen = sizeof(bound);
301#else
302 socklen_t boundLen = sizeof(bound);
303#endif
304 if (::getsockname(m_impl->listenSocket, reinterpret_cast<sockaddr*>(&bound), &boundLen) ==
305 0) {
306 m_impl->boundPort = ntohs(bound.sin_port);
307 } else {
308 m_impl->boundPort = port;
309 }
310
311 m_impl->running.store(true, std::memory_order_relaxed);
312 m_impl->ioThread = std::thread([impl = m_impl.get()]() { impl->ioLoop(); });
313 Log::info("[MuseSocket] listening on 127.0.0.1:" + std::to_string(m_impl->boundPort));
314 return true;
315}

317void MuseSocketServer::stop() {
318 if (!m_impl->running.exchange(false, std::memory_order_relaxed)) return;
319 if (m_impl->ioThread.joinable()) m_impl->ioThread.join();
320 closeSocket(m_impl->listenSocket);
321 m_impl->listenSocket = kInvalidSocket;
322}

332size_t MuseSocketServer::processPending(MuseService& service) {
333 // At most kMaxRequestsPerFrame per call; the rest wait for the next frame,
334 // so a backlog can never monopolise one frame. Order per client is kept.
335 std::deque<std::pair<uint64_t, std::string>> requests;
336 {
337 std::lock_guard<std::mutex> lock(m_impl->mutex);
338 while (!m_impl->inbox.empty() && requests.size() < kMaxRequestsPerFrame) {
339 requests.push_back(std::move(m_impl->inbox.front()));
340 m_impl->inbox.pop_front();
341 }
342 }
343 for (auto& [clientId, line] : requests) {
344 std::string response = service.handleRequest(line);
345 response += '\n';
346 std::lock_guard<std::mutex> lock(m_impl->mutex);
347 m_impl->outbox.emplace_back(clientId, std::move(response));
348 }
349 return requests.size();
350}
