64struct MuseSocketServer::Impl {
65 SocketHandle listenSocket = kInvalidSocket;
66 uint16_t boundPort = 0;
67 std::thread ioThread;
68 std::atomic<bool> running{false};
69
70 struct Client {
71 SocketHandle socket = kInvalidSocket;
72 std::string readBuffer;
73 std::string writeBuffer; // IO-thread only
74 };
75
76 // IO-thread state
77 std::unordered_map<uint64_t, Client> clients;
78 uint64_t nextClientId = 1;
79
80 // Cross-thread queues
81 std::mutex mutex;
82 std::deque<std::pair<uint64_t, std::string>> inbox; // client -> request line
83 std::deque<std::pair<uint64_t, std::string>> outbox; // client -> response line
84
85 void ioLoop() {
86 while (running.load(std::memory_order_relaxed)) {
87 fd_set readSet;
88 FD_ZERO(&readSet);
89 FD_SET(listenSocket, &readSet);
90 SocketHandle maxFd = listenSocket;
91 for (const auto& [clientId, client] : clients) {
92 FD_SET(client.socket, &readSet);
93 if (client.socket > maxFd) maxFd = client.socket;
94 }
95
96 // Short timeout so queued responses and stop() are picked up
97 // promptly even when no traffic arrives.
98 timeval timeout{};
99 timeout.tv_usec = 50 * 1000;
100 const int ready =
101 ::select(static_cast<int>(maxFd + 1), &readSet, nullptr, nullptr, &timeout);
102 if (!running.load(std::memory_order_relaxed)) break;
103
104 if (ready > 0 && FD_ISSET(listenSocket, &readSet)) {
105 sockaddr_in remote{};
106#ifdef _WIN32
107 int remoteLen = sizeof(remote);
108#else
109 socklen_t remoteLen = sizeof(remote);
110#endif
111 SocketHandle accepted =
112 ::accept(listenSocket, reinterpret_cast<sockaddr*>(&remote), &remoteLen);
113 if (accepted != kInvalidSocket) {
114 Client client;
115 client.socket = accepted;
116 clients.emplace(nextClientId++, std::move(client));
117 }
118 }
119
120 std::vector<uint64_t> disconnected;
121 if (ready > 0) {
122 for (auto& [clientId, client] : clients) {
123 if (!FD_ISSET(client.socket, &readSet)) continue;
124 char chunk[4096];
125 const auto received = ::recv(client.socket, chunk, sizeof(chunk), 0);
126 if (received <= 0) {
127 disconnected.push_back(clientId);
128 continue;
129 }
130 client.readBuffer.append(chunk, static_cast<size_t>(received));
131
132 // A runaway line without a newline must not grow forever.
133 constexpr size_t kMaxLineBytes = 1 << 20;
134 size_t newline;
135 while ((newline = client.readBuffer.find('\n')) != std::string::npos) {
136 std::string line = client.readBuffer.substr(0, newline);
137 client.readBuffer.erase(0, newline + 1);
138 if (!line.empty() && line.back() == '\r') line.pop_back();
139 if (line.empty()) continue;
140 std::lock_guard<std::mutex> lock(mutex);
141 inbox.emplace_back(clientId, std::move(line));
142 }
143 if (client.readBuffer.size() > kMaxLineBytes) {
144 disconnected.push_back(clientId);
145 }
146 }
147 }
148
149 // Deliver queued responses.
150 {
151 std::deque<std::pair<uint64_t, std::string>> toWrite;
152 {
153 std::lock_guard<std::mutex> lock(mutex);
154 toWrite.swap(outbox);
155 }
156 for (auto& [clientId, response] : toWrite) {
157 auto it = clients.find(clientId);
158 if (it == clients.end()) continue; // client already gone
159 it->second.writeBuffer += response;
160 }
161 }
162 for (auto& [clientId, client] : clients) {
163 while (!client.writeBuffer.empty()) {
164 const auto sent = ::send(client.socket, client.writeBuffer.data(),
165#ifdef _WIN32
166 static_cast<int>(client.writeBuffer.size()),
167#else
168 client.writeBuffer.size(),
169#endif
170 0);
171 if (sent <= 0) {
172 disconnected.push_back(clientId);
173 break;
174 }
175 client.writeBuffer.erase(0, static_cast<size_t>(sent));
176 }
177 }
178
179 for (uint64_t clientId : disconnected) {
180 auto it = clients.find(clientId);
181 if (it != clients.end()) {
182 closeSocket(it->second.socket);
183 clients.erase(it);
184 }
185 }
186 }
187
188 for (auto& [clientId, client] : clients) {
189 closeSocket(client.socket);
190 }
191 clients.clear();
192 }
193};

201bool MuseSocketServer::start(uint16_t port, std::string& outError) {
202 if (m_impl->running.load(std::memory_order_relaxed)) {
203 outError = "already running";
204 return false;
205 }
206 if (!initSockets(outError)) return false;
207
208 m_impl->listenSocket = ::socket(AF_INET, SOCK_STREAM, 0);
209 if (m_impl->listenSocket == kInvalidSocket) {
210 outError = "socket() failed";
211 return false;
212 }
213
214 // Loopback only: Muse is a local control surface, never a network service.
215 sockaddr_in address{};
216 address.sin_family = AF_INET;
217 address.sin_port = htons(port);
218 address.sin_addr.s_addr = htonl(INADDR_LOOPBACK);
219
220 int reuse = 1;
221 ::setsockopt(m_impl->listenSocket, SOL_SOCKET, SO_REUSEADDR,
222 reinterpret_cast<const char*>(&reuse), sizeof(reuse));
223
224 if (::bind(m_impl->listenSocket, reinterpret_cast<sockaddr*>(&address), sizeof(address)) !=
225 0) {
226 outError = "bind to 127.0.0.1:" + std::to_string(port) + " failed";
227 closeSocket(m_impl->listenSocket);
228 m_impl->listenSocket = kInvalidSocket;
229 return false;
230 }
231 if (::listen(m_impl->listenSocket, 4) != 0) {
232 outError = "listen failed";
233 closeSocket(m_impl->listenSocket);
234 m_impl->listenSocket = kInvalidSocket;
235 return false;
236 }
237
238 sockaddr_in bound{};
239#ifdef _WIN32
240 int boundLen = sizeof(bound);
241#else
242 socklen_t boundLen = sizeof(bound);
243#endif
244 if (::getsockname(m_impl->listenSocket, reinterpret_cast<sockaddr*>(&bound), &boundLen) ==
245 0) {
246 m_impl->boundPort = ntohs(bound.sin_port);
247 } else {
248 m_impl->boundPort = port;
249 }
250
251 m_impl->running.store(true, std::memory_order_relaxed);
252 m_impl->ioThread = std::thread([impl = m_impl.get()]() { impl->ioLoop(); });
253 Log::info("[MuseSocket] listening on 127.0.0.1:" + std::to_string(m_impl->boundPort));
254 return true;
255}

257void MuseSocketServer::stop() {
258 if (!m_impl->running.exchange(false, std::memory_order_relaxed)) return;
259 if (m_impl->ioThread.joinable()) m_impl->ioThread.join();
260 closeSocket(m_impl->listenSocket);
261 m_impl->listenSocket = kInvalidSocket;
262}

272size_t MuseSocketServer::processPending(MuseService& service) {
273 std::deque<std::pair<uint64_t, std::string>> requests;
274 {
275 std::lock_guard<std::mutex> lock(m_impl->mutex);
276 requests.swap(m_impl->inbox);
277 }
278 for (auto& [clientId, line] : requests) {
279 std::string response = service.handleRequest(line);
280 response += '\n';
281 std::lock_guard<std::mutex> lock(m_impl->mutex);
282 m_impl->outbox.emplace_back(clientId, std::move(response));
283 }
284 return requests.size();
285}
