21ChannelStrip::ChannelStrip(std::shared_ptr<Track> track, TrackManager* trackManager)
22 : m_track(track)
23 , m_trackManager(trackManager)
24{
25 // Create volume fader (vertical slider)
26 m_volumeFader = std::make_shared<AestraUI::Fader>();
27 m_volumeFader->setValue(m_track ? m_track->getVolume() : 0.8f);
28 m_volumeFader->setOnValueChange([this](double value) {
29 if (m_track && m_trackManager) {
30 float newVol = static_cast<float>(value);
31 float oldVol = m_track->getVolume();
32 // Don't set volume manually — let the command do it
33 if (std::abs(newVol - oldVol) > 0.0001f) {
34 m_trackManager->getCommandHistory().pushAndExecute(
35 std::make_shared<SetVolumeCommand>(*m_track, newVol));
36 Aestra::Log::info("[MixerView] Vol cmd: " + std::to_string(oldVol) + " -> " + std::to_string(newVol));
37 }
38 }
39 });
40 addChild(m_volumeFader);
41
42 // Create pan knob (horizontal slider for now)
43 m_panKnob = std::make_shared<AestraUI::PanKnob>();
44 m_panKnob->setValue(0.0f); // Center
45 m_panKnob->setOnValueChange([this](double value) {
46 if (m_track && m_trackManager) {
47 float newPan = static_cast<float>(value);
48 float oldPan = m_track->getPan();
49 if (std::abs(newPan - oldPan) > 0.0001f) {
50 m_trackManager->getCommandHistory().pushAndExecute(
51 std::make_shared<SetPanCommand>(*m_track, newPan));
52 }
53 }
54 });
55 addChild(m_panKnob);
56
57 // Create mute button
58 m_muteButton = std::make_shared<AestraUI::MuteButton>();
59 m_muteButton->setOnToggle([this](bool toggled) {
60 if (m_track && m_trackManager) {
61 m_trackManager->getCommandHistory().pushAndExecute(
62 std::make_shared<SetMuteCommand>(*m_track, toggled));
63 }
64 });
65 // Initialize state
66 if (m_track) m_muteButton->setOn(m_track->isMuted());
67 addChild(m_muteButton);
68
69 // Create solo button
70 m_soloButton = std::make_shared<AestraUI::SoloButton>();
71 m_soloButton->setOnToggle([this](bool toggled) {
72 if (m_track && m_trackManager) {
73 bool newSolo = toggled;
74
75 // If enabling solo, clear all other solos first (exclusive solo)
76 if (newSolo) {
77 m_trackManager->clearAllSolos();
78 }
79
80 m_trackManager->getCommandHistory().pushAndExecute(
81 std::make_shared<SetSoloCommand>(*m_track, newSolo));
82 }
83 });
84 // Initialize state
85 if (m_track) m_soloButton->setOn(m_track->isSoloed());
86 addChild(m_soloButton);
87
88 layoutControls();
89}

99void ChannelStrip::onRender(AestraUI::NUIRenderer& renderer) {
100 AESTRA_ZONE("ChannelStrip_Render");
101 auto& theme = AestraUI::NUIThemeManager::getInstance();
102 auto bounds = getBounds();
103
104 // Determine Base Color
105 AestraUI::NUIColor baseColor = theme.getColor("backgroundSecondary");
106 bool isMaster = false;
107
108 if (m_track) {
109 // Master Track Check (Name-based "MASTER" or ID 0)
110 std::string name = m_track->getName();
111 std::transform(name.begin(), name.end(), name.begin(), ::toupper);
112
113 if (name == "MASTER" || m_track->getChannelId() == 0) {
114 baseColor = theme.getColor("primary"); // Purple
115 isMaster = true;
116 } else {
117 // Priority: Use Playlist Lane Color (Visual Source of Truth)
118 bool colorFound = false;
119
120 // Try mapped lane
121 if (m_laneId.isValid() && m_trackManager) {
122 const auto* lane = m_trackManager->getPlaylistModel().getLane(m_laneId);
123 // Ensure alpha is fully opaque for the base color extraction
124 if (lane && lane->colorRGBA != 0) {
125 baseColor = AestraUI::NUIColor::fromHex(lane->colorRGBA).withAlpha(1.0f);
126 colorFound = true;
127 }
128 }
129
130 // Fallback to internal track color
131 if (!colorFound) {
132 uint32_t c = m_track->getColor();
133 if (c != 0) {
134 baseColor = AestraUI::NUIColor::fromHex(c).withAlpha(1.0f);
135 }
136 }
137 }
138 }
139
140 // 1. Frosted Glass Background (Tinted) - INCREASED OPACITY FOR VISIBILITY
141 // Top: Lighter tint, Bottom: Darker tint
142 float alphaTop = isMaster ? 0.30f : 0.20f;
143 float alphaBottom = isMaster ? 0.15f : 0.10f;
144
145 AestraUI::NUIColor bgTop = baseColor.withAlpha(alphaTop);
146 AestraUI::NUIColor bgBottom = baseColor.withAlpha(alphaBottom);
147
148 // Mix with deep void to ensure it's not too washed out
149 renderer.fillRectGradient(bounds, bgTop, bgBottom, true);
150
151 // 2. Glass Border (Tinted)
152 float borderAlpha = isMaster ? 0.9f : 0.6f;
153 auto borderColor = baseColor.withAlpha(borderAlpha);
154 renderer.strokeRect(bounds, 1.0f, borderColor);
155
156 // Track name at bottom
157 if (m_track) {
158 auto textColor = theme.getColor("textSecondary"); // Less distracting name
159 std::string trackName = m_track->getName();
160 float textY = bounds.y + bounds.height - 25.0f;
161 // Center text
162 renderer.drawTextCentered(trackName, AestraUI::NUIRect(bounds.x, textY, bounds.width, 20.0f), 12.0f, textColor);
163 }
164
165 // Level meter (Neon Segmented)
166 if (m_track) {
167 float meterW = 6.0f;
168 float meterRightPad = 8.0f;
169 float meterX = bounds.x + bounds.width - meterRightPad - meterW;
170 float meterY = bounds.y + 40.0f; // Start below top buttons
171 float meterH = bounds.height - 80.0f; // Leave space
172
173 // Meter background (Dark Slot)
174 AestraUI::NUIRect meterBg(meterX, meterY, meterW, meterH);
175 renderer.fillRoundedRect(meterBg, 2.0f, AestraUI::NUIColor(0.05f, 0.05f, 0.08f, 0.8f));
176
177 // Get current level (Mock/Placeholder)
178 float level = m_track->getVolume(); // Simple mapping
179
180 // Draw segments
181 int numSegments = 24;
182 float segH = (meterH - 2.0f) / numSegments;
183 float segGap = 1.0f;
184
185 int activeSegments = static_cast<int>(level * numSegments);
186
187 for (int i = 0; i < numSegments; ++i) {
188 // Index 0 is bottom
189 float y = meterY + meterH - ((i + 1) * segH);
190
191 bool isActive = i < activeSegments;
192 AestraUI::NUIColor segColor;
193
194 if (isActive) {
195 float n = static_cast<float>(i) / numSegments;
196 if (n < 0.7f) segColor = AestraUI::NUIColor::fromHex(0x00ffaa); // Cyan
197 else if (n < 0.9f) segColor = AestraUI::NUIColor::fromHex(0xffaa00); // Amber
198 else segColor = AestraUI::NUIColor::fromHex(0xff3333); // Red
199 segColor = segColor.withAlpha(0.9f);
200 } else {
201 segColor = AestraUI::NUIColor(0.2f, 0.2f, 0.2f, 0.3f);
202 }
203
204 renderer.fillRect(AestraUI::NUIRect(meterX + 1.0f, y + segGap, meterW - 2.0f, segH - segGap), segColor);
205 }
206 }
207
208 // Render child controls
209 renderChildren(renderer);
210}

221void ChannelStrip::layoutControls() {
222 auto bounds = getBounds();
223 float padding = 6.0f;
224 float controlWidth = bounds.width - 2 * padding;
225
226 // Meter takes right side
227 float meterWidth = 14.0f; // total meter area
228 controlWidth -= meterWidth; // reduce checks
229
230 float buttonHeight = 22.0f;
231 float y = bounds.y + padding;
232
233 // Group: Mute/Solo
234 if (m_muteButton && m_soloButton) {
235 float btnW = (controlWidth - padding) / 2.0f;
236 m_muteButton->setBounds(AestraUI::NUIRect(bounds.x + padding, y, btnW, buttonHeight));
237 m_soloButton->setBounds(AestraUI::NUIRect(bounds.x + padding + btnW + padding, y, btnW, buttonHeight));
238 y += buttonHeight + padding;
239 }
240
241 // Pan Knob
242 if (m_panKnob) {
243 float panHeight = 24.0f; // Compact
244 m_panKnob->setBounds(AestraUI::NUIRect(bounds.x + padding, y, controlWidth, panHeight));
245 y += panHeight + padding;
246 }
247
248 // Fader (Vertical, centered in remaining space)
249 if (m_volumeFader) {
250 float bottomMargin = 30.0f; // Name
251 float faderHeight = bounds.height - (y - bounds.y) - bottomMargin;
252 // Center fader in control area
253 m_volumeFader->setBounds(AestraUI::NUIRect(bounds.x + padding, y, controlWidth, faderHeight));
254 }
255}

297void MixerView::refreshChannels() {
298 // Remove old channel strips from parent before clearing
299 for (auto& strip : m_channelStrips) {
300 removeChild(strip);
301 }
302
303 // Clear existing channel strips
304 m_channelStrips.clear();
305
306 if (!m_trackManager) return;
307
308 // THREAD-SAFE: Get a snapshot of channels to avoid race with Audio Thread
309 auto channelsSnapshot = m_trackManager->getChannelsSnapshot();
310 auto& playlist = m_trackManager->getPlaylistModel();
311
312 // Create channel strip for each track
313 // Assume 1:1 mapping between Playlist Lanes and non-skipped Mixer Channels for now
314 // (excluding Master which is special case ID 0)
315 size_t laneIndex = 0;
316
317 for (const auto& track : channelsSnapshot) {
318 if (track && track->getName() != "Preview") { // Skip preview track
319 auto channelStrip = std::make_shared<ChannelStrip>(std::shared_ptr<MixerChannel>(track, [](MixerChannel*){}), m_trackManager.get());
320
321 // Map to playlist lane if it's a regular track
322 if (track->getChannelId() != 0) {
323 if (laneIndex < playlist.getLaneCount()) {
324 channelStrip->setPlaylistLaneID(playlist.getLaneId(laneIndex));
325 laneIndex++;
326 }
327 }
328
329 channelStrip->setPlatformBridge(m_platformBridge);
330 m_channelStrips.push_back(channelStrip);
331 addChild(channelStrip);
332 }
333 }
334
335 layoutChannels();
336 Log::info("Mixer: Created " + std::to_string(m_channelStrips.size()) + " channel strips");
337}
