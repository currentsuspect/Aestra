41std::shared_ptr<const PolyphaseFilterBank> SampleRateConverter::getSharedFilterBank(SRCQuality quality) {
42 auto& cache = getFilterCache();
43 std::lock_guard<std::mutex> lock(cache.mutex);
44
45 auto it = cache.entries.find(quality);
46 if (it != cache.entries.end()) {
47 if (auto shared = it->second.lock()) {
48 return shared;
49 }
50 }
51
52 // Not in cache or expired, create a heap-owned bank. configure() is already
53 // documented as non-RT-safe, and shared cache entries need normal lifetime
54 // management instead of leaking arena-backed allocations after expiry.
55 auto* rawBank = static_cast<PolyphaseFilterBank*>(::operator new(sizeof(PolyphaseFilterBank)));
56 AESTRA_MEMORY_ALLOC(sizeof(PolyphaseFilterBank));
57 auto newBank = std::shared_ptr<PolyphaseFilterBank>(new (rawBank) PolyphaseFilterBank(),
58 destroySharedFilterBank);
59
60 // We use a temporary SRC instance to generate the bank with a standard upsampling cutoff (0.98)
61 SampleRateConverter temp;
62 temp.m_ratio = 1.01; // Fake upsampling ratio to trigger standard cutoff
63 temp.generateFilterBank(quality);
64
65 // Copy generated coeffs to the shared bank
66 *newBank = temp.m_localFilterBank;
67
68 cache.entries[quality] = newBank;
69 return newBank;
70}

78#ifdef AESTRA_HAS_SSE
79// SSE dot product: processes 4 floats at a time
80float dotProductSSE(const float* a, const float* b, uint32_t n) noexcept {
81 __m128 sum = _mm_setzero_ps();
82
83 // Process 4 elements at a time
84 uint32_t i = 0;
85 for (; i + 4 <= n; i += 4) {
86 __m128 va = _mm_loadu_ps(a + i);
87 __m128 vb = _mm_load_ps(b + i); // Aligned load for coefficients
88 sum = _mm_add_ps(sum, _mm_mul_ps(va, vb));
89 }
90
91 // Horizontal sum: sum[0] + sum[1] + sum[2] + sum[3]
92 __m128 shuf = _mm_shuffle_ps(sum, sum, _MM_SHUFFLE(2, 3, 0, 1));
93 sum = _mm_add_ps(sum, shuf);
94 shuf = _mm_movehl_ps(shuf, sum);
95 sum = _mm_add_ss(sum, shuf);
96 float result = _mm_cvtss_f32(sum);
97
98 // Handle remaining elements (scalar)
99 for (; i < n; ++i) {
100 result += a[i] * b[i];
101 }
102
103 return result;
104}
105#endif
106
107#ifdef AESTRA_HAS_AVX
108// AVX dot product: processes 8 floats at a time
109float dotProductAVX(const float* a, const float* b, uint32_t n) noexcept {
110 __m256 sum = _mm256_setzero_ps();
111
112 uint32_t i = 0;
113 for (; i + 8 <= n; i += 8) {
114 __m256 va = _mm256_loadu_ps(a + i);
115 __m256 vb = _mm256_load_ps(b + i); // Aligned load for coefficients
116 sum = _mm256_add_ps(sum, _mm256_mul_ps(va, vb));
117 }
118
119 // Reduce 8-wide sum to scalar
120 __m128 low = _mm256_castps256_ps128(sum);
121 __m128 high = _mm256_extractf128_ps(sum, 1);
122 __m128 sum128 = _mm_add_ps(low, high);
123
124 __m128 shuf = _mm_shuffle_ps(sum128, sum128, _MM_SHUFFLE(2, 3, 0, 1));
125 sum128 = _mm_add_ps(sum128, shuf);
126 shuf = _mm_movehl_ps(shuf, sum128);
127 sum128 = _mm_add_ss(sum128, shuf);
128 float result = _mm_cvtss_f32(sum128);
129
130 for (; i < n; ++i) {
131 result += a[i] * b[i];
132 }
133
134 return result;
135}
136#endif
137
138// Scalar fallback dot product
139#if defined(_MSC_VER)
140#define AESTRA_RESTRICT __restrict
141#else
142#define AESTRA_RESTRICT __restrict__
143#endif
144
145// Scalar dot product with optional SSE4.1 target for auto-vectorization on GCC/Clang.
146#if defined(__GNUC__) && (defined(__x86_64__) || defined(__i386__))
147__attribute__((target("sse4.1")))
148#endif
149float dotProductScalar(const float* AESTRA_RESTRICT a, const float* AESTRA_RESTRICT b, uint32_t n) noexcept {
150 float sum = 0.0f;
151 uint32_t i = 0;
152
153 // Unroll by 8 for better instruction-level parallelism
154 for (; i + 7 < n; i += 8) {
155 sum += a[i] * b[i] + a[i + 1] * b[i + 1] +
156 a[i + 2] * b[i + 2] + a[i + 3] * b[i + 3] +
157 a[i + 4] * b[i + 4] + a[i + 5] * b[i + 5] +
158 a[i + 6] * b[i + 6] + a[i + 7] * b[i + 7];
159 }
160
161 // Remainder
162 for (; i < n; ++i) {
163 sum += a[i] * b[i];
164 }
165 return sum;
166}

176void SampleRateConverter::configure(uint32_t srcRate, uint32_t dstRate, uint32_t channels, SRCQuality quality) {
177 // Validate parameters
178 if (srcRate == 0 || dstRate == 0) {
179 Log::warning("SampleRateConverter: Invalid sample rate (0)");
180 return;
181 }
182 if (channels == 0 || channels > SRCConstants::MAX_CHANNELS) {
183 Log::warning("SampleRateConverter: Invalid channel count: " + std::to_string(channels) +
184 " (max: " + std::to_string(SRCConstants::MAX_CHANNELS) + ")");
185 return;
186 }
187
188 m_srcRate = srcRate;
189 m_dstRate = dstRate;
190 m_channels = channels;
191 m_quality = quality;
192 m_ratio = static_cast<double>(dstRate) / static_cast<double>(srcRate);
193 m_currentRatio = m_ratio;
194 m_targetRatio = m_ratio;
195 m_ratioSmoothFrames = 0;
196 m_ratioSmoothTotal = 0;
197
198 // Check for passthrough mode (no conversion needed)
199 m_isPassthrough = (srcRate == dstRate);
200
201 // Initialize history buffer
202 m_history.init(channels);
203 m_historyFilled = 0;
204 m_srcPosition = 0.0;
205 m_nextOutputSrcPos = 0.0; // [FIX] Init phase
206
207 // Generate/Fetch Filter Bank
208 if (!m_isPassthrough) {
209 if (m_ratio >= 1.0) {
210 // Upsampling: use shared cache (fixed cutoff)
211 m_sharedFilterBank = getSharedFilterBank(quality);
212 } else {
213 // Downsampling: need instance-owned filter with specific cutoff
214 m_sharedFilterBank = nullptr;
215 generateFilterBank(quality);
216 }
217 } else {
218 m_sharedFilterBank = nullptr;
219 }
220
221 m_configured = true;
222
223 Log::info("SampleRateConverter configured: " + std::to_string(srcRate) + " -> " + std::to_string(dstRate) +
224 " Hz, " + std::to_string(channels) + " channels, quality=" + std::to_string(static_cast<int>(quality)) +
225 (m_isPassthrough ? " (passthrough)" : "") + (hasSIMD() ? " [SIMD]" : " [Scalar]"));
226}

252void SampleRateConverter::generateFilterBank(SRCQuality quality) {
253 m_localFilterBank.clear();
254
255 // Determine number of taps based on quality
256 uint32_t numTaps = 0;
257 double kaiserBeta = SRCConstants::KAISER_BETA_DEFAULT;
258
259 switch (quality) {
260 case SRCQuality::Linear:
261 numTaps = 2;
262 kaiserBeta = 2.0; // Very narrow window
263 break;
264 case SRCQuality::Cubic:
265 numTaps = 4;
266 kaiserBeta = 4.0;
267 break;
268 case SRCQuality::Sinc8:
269 numTaps = 8;
270 kaiserBeta = 6.0;
271 break;
272 case SRCQuality::Sinc16:
273 numTaps = 16;
274 kaiserBeta = 8.0;
275 break;
276 case SRCQuality::Sinc64:
277 numTaps = 64;
278 kaiserBeta = 10.0;
279 break;
280 }
281
282 m_localFilterBank.numTaps = numTaps;
283 m_localFilterBank.halfTaps = numTaps / 2;
284
285 const double halfTaps = static_cast<double>(numTaps) / 2.0;
286
287 // Cutoff frequency for anti-aliasing:
288 // When downsampling (ratio < 1): limit to output Nyquist = ratio * Nyquist
289 // When upsampling (ratio > 1): no limiting needed (already below output Nyquist)
290 // Apply 0.95 factor for transition band (prevents ringing at Nyquist)
291 double cutoff = 1.0;
292 if (m_ratio < 1.0) {
293 // Downsampling: need to filter out frequencies above output Nyquist
294 cutoff = m_ratio * 0.95;
295 } else {
296 // Upsampling: full bandwidth, slight rolloff for smoothness
297 cutoff = 0.98;
298 }
299
300 // Generate coefficients for each fractional phase
301 for (uint32_t phase = 0; phase < SRCConstants::POLYPHASE_PHASES; ++phase) {
302 // Fractional offset for this phase (0.0 to 1.0)
303 const double frac = static_cast<double>(phase) / static_cast<double>(SRCConstants::POLYPHASE_PHASES);
304
305 double sumWeight = 0.0;
306
307 // Compute sinc * window for each tap
308 for (uint32_t tap = 0; tap < numTaps; ++tap) {
309 // x is the distance from the ideal sample position to this tap
310 // Center the filter: tap 0 is at position -halfTaps, tap numTaps-1 is at +halfTaps
311 const double x = (static_cast<double>(tap) - halfTaps) - frac;
312
313 // Sinc function with cutoff applied
314 double sinc = 1.0;
315 if (std::abs(x) > 1e-10) {
316 const double pix = SRCConstants::PI * x * cutoff;
317 sinc = std::sin(pix) / (SRCConstants::PI * x * cutoff); // sinc(x*cutoff) = sin(pi*x*cutoff)/(pi*x*cutoff)
318 } else {
319 // sinc(cutoff*x) tends to 1 at x == 0; cutoff is applied by
320 // the nonzero argument and the per-phase normalization.
321 sinc = 1.0;
322 }
323
324 // Apply Kaiser window centered in the tap array
325 const double window = kaiserWindow(tap, numTaps, kaiserBeta);
326
327 // Coefficient is sinc(cutoff*x) * window
328 const double coeff = sinc * window;
329 m_localFilterBank.coeffs[phase][tap] = static_cast<float>(coeff);
330 sumWeight += coeff;
331 }
332
333 // Normalize coefficients so they sum to 1.0 (unity gain)
334 if (sumWeight > 1e-10) {
335 const float invSum = static_cast<float>(1.0 / sumWeight);
336 for (uint32_t tap = 0; tap < numTaps; ++tap) {
337 m_localFilterBank.coeffs[phase][tap] *= invSum;
338 }
339 }
340 }
341
342 Log::info("SampleRateConverter: Generated " + std::to_string(numTaps) + "-tap filter with " +
343 std::to_string(SRCConstants::POLYPHASE_PHASES) + " phases, cutoff=" + std::to_string(cutoff));
344}

350double SampleRateConverter::kaiserWindow(double n, double N, double beta) noexcept {
351 // Kaiser window: I0(beta * sqrt(1 - ((n - N/2) / (N/2))^2)) / I0(beta)
352 const double halfN = (N - 1.0) / 2.0;
353 const double ratio = (n - halfN) / halfN;
354 const double arg = beta * std::sqrt(std::max(0.0, 1.0 - ratio * ratio));
355 return bessel_I0(arg) / bessel_I0(beta);
356}
357
358double SampleRateConverter::bessel_I0(double x) noexcept {
359 // Modified Bessel function of first kind, order 0
360 // Using series expansion (converges quickly for typical beta values)
361 double sum = 1.0;
362 double term = 1.0;
363 const double halfX = x / 2.0;
364
365 for (int k = 1; k < 25; ++k) { // 25 terms is plenty for convergence
366 term *= (halfX / static_cast<double>(k));
367 term *= (halfX / static_cast<double>(k));
368 sum += term;
369 if (term < 1e-12 * sum)
370 break; // Converged
371 }
372
373 return sum;
374}

380uint32_t SampleRateConverter::process(const float* input, uint32_t inputFrames, float* output,
381 uint32_t maxOutputFrames) noexcept {
382 if (!m_configured || inputFrames == 0 || maxOutputFrames == 0) {
383 return 0;
384 }
385
386 // Passthrough mode: direct copy
387 if (m_isPassthrough && m_currentRatio == 1.0) {
388 const uint32_t copyFrames = std::min(inputFrames, maxOutputFrames);
389 std::memcpy(output, input, copyFrames * m_channels * sizeof(float));
390 return copyFrames;
391 }
392
393 const PolyphaseFilterBank* bank = getFilterBank();
394 if (!bank)
395 return 0;
396
397 const uint32_t numTaps = bank->numTaps;
398 const uint32_t halfTaps = bank->halfTaps;
399 uint32_t outputFrames = 0;
400
401 const bool useSIMD = m_simdEnabled.load(std::memory_order_relaxed) && hasSIMD();
402
403 // Update ratio smoothing
404 if (m_ratioSmoothFrames > 0 && m_currentRatio != m_targetRatio) {
405 // Linear interpolation toward target
406 double t = 1.0 - (static_cast<double>(m_ratioSmoothFrames) / static_cast<double>(m_ratioSmoothTotal));
407 m_currentRatio = m_ratio + t * (m_targetRatio - m_ratio);
408 m_ratioSmoothFrames = (m_ratioSmoothFrames > inputFrames) ? (m_ratioSmoothFrames - inputFrames) : 0;
409 if (m_ratioSmoothFrames == 0) {
410 m_currentRatio = m_targetRatio;
411 }
412 }
413
414 // Use current (possibly smoothed) ratio
415 const double effectiveRatio = m_currentRatio;
416 const double invRatio = 1.0 / effectiveRatio; // hoisted: constant for entire process() call
417
418 // Hoisted constants (invariant across all frames in this call)
419 const double historySizeD = static_cast<double>(m_history.size);
420 const double historySizeMinusHalfTaps = historySizeD - static_cast<double>(halfTaps);
421 const double halfTapsD = static_cast<double>(halfTaps);
422 const double polyPhaseScale = static_cast<double>(SRCConstants::POLYPHASE_PHASES);
423 const uint32_t historySizeMask = m_history.size - 1;
424
425 // Hoist member variables into locals to reduce memory traffic in the hot loop
426 double srcPosition = m_srcPosition;
427 double nextOutputSrcPos = m_nextOutputSrcPos;
428
429 // Track srcPosition - nextOutputSrcPos as a single variable.
430 // In the output loop, srcPosition is constant and nextOutputSrcPos increases
431 // by invRatio each iteration, so srcNextDiff decreases by invRatio.
432 double srcNextDiff = srcPosition - nextOutputSrcPos;
433
434 // Process input samples — use incrementing pointer to avoid multiply per frame
435 const float* inPtr = input;
436 // Use incrementing output pointer to avoid multiply per output sample
437 float* outPtr = output;
438 for (uint32_t inFrame = 0; inFrame < inputFrames; ++inFrame) {
439 // Inline push for stereo (most common case) — eliminates function call
440 if (m_channels == 2) {
441 const uint32_t wp = m_history.writePos;
442 const uint32_t wp1 = wp + m_history.size;
443 const float l = inPtr[0];
444 const float r = inPtr[1];
445 m_history.data[0][wp] = l;
446 m_history.data[0][wp1] = l;
447 m_history.data[1][wp] = r;
448 m_history.data[1][wp1] = r;
449 m_history.writePos = (wp + 1) & (m_history.size - 1);
450 } else {
451 m_history.push(inPtr);
452 }
453 inPtr += m_channels;
454
455 // Logical "now" position in source stream (in samples)
456 srcPosition += 1.0;
457 srcNextDiff = srcPosition - nextOutputSrcPos;
458
459 // Hoist per-frame constants for the output loop
460 const double srcPosMinusHalfTaps = srcPosition - halfTapsD;
461 const double srcPosMinusHistorySize = srcPosition - historySizeMinusHalfTaps;
462
463 // Track historyPos directly: starts at historySizeD - srcNextDiff,
464 // increases by invRatio each output sample iteration
465 double historyPos = historySizeD - srcNextDiff;
466
467 // Generate output samples while we have enough history
468 while (outputFrames < maxOutputFrames) {
469 // If the next sample is ahead of our current history window (accounting for filter tail), wait.
470 if (nextOutputSrcPos > srcPosMinusHalfTaps) {
471 break;
472 }
473
474 // Re-check: If next sample is WAY behind (seek?), reset it to a safe "current" position.
475 if (nextOutputSrcPos < srcPosMinusHistorySize) {
476 nextOutputSrcPos = srcPosMinusHalfTaps;
477 srcNextDiff = srcPosition - nextOutputSrcPos;
478 historyPos = historySizeD - srcNextDiff;
479 }
480
481 const uint32_t intPos = static_cast<uint32_t>(historyPos);
482
483 // Quantize fractional position to polyphase index
484 // Equivalent to fracPos * 256 but avoids the intermediate subtraction:
485 // historyPos * 256 = intPos * 256 + fracPos * 256
486 // The & 255 mask removes the intPos * 256 term
487 const uint32_t phaseIndex =
488 static_cast<uint32_t>(historyPos * polyPhaseScale + 0.5) &
489 (SRCConstants::POLYPHASE_PHASES - 1);
490
491 // Hoist channel-invariant computations
492 const float* coeffs = bank->coeffs[phaseIndex];
493 // Compute windowIdx directly: (intPos - halfTaps) & mask wraps correctly
494 // for unsigned arithmetic, eliminating the intermediate samplePos0 variable
495 const uint32_t windowIdx = m_history.writePos +
496 ((intPos - halfTaps) & historySizeMask);
497
498 // Precompute output base pointer — use incrementing pointer
499 float* out = outPtr;
500
501 // Generate output for each channel — hoist SIMD dispatch outside the loop
502 // Fast path for stereo (most common case): unroll to 2 independent accumulators
503 if (m_channels == 2) {
504 const float* windowL = &m_history.data[0][windowIdx];
505 const float* windowR = &m_history.data[1][windowIdx];
506#ifdef AESTRA_HAS_AVX
507 if (useSIMD) {
508 out[0] = dotProductAVX(windowL, coeffs, numTaps);
509 out[1] = dotProductAVX(windowR, coeffs, numTaps);
510 } else {
511 out[0] = dotProductScalar(windowL, coeffs, numTaps);
512 out[1] = dotProductScalar(windowR, coeffs, numTaps);
513 }
514#elif defined(AESTRA_HAS_SSE)
515 if (useSIMD) {
516 out[0] = dotProductSSE(windowL, coeffs, numTaps);
517 out[1] = dotProductSSE(windowR, coeffs, numTaps);
518 } else {
519 out[0] = dotProductScalar(windowL, coeffs, numTaps);
520 out[1] = dotProductScalar(windowR, coeffs, numTaps);
521 }
522#else
523 (void)useSIMD;
524 out[0] = dotProductScalar(windowL, coeffs, numTaps);
525 out[1] = dotProductScalar(windowR, coeffs, numTaps);
526#endif
527 } else {
528#ifdef AESTRA_HAS_AVX
529 if (useSIMD) {
530 for (uint32_t ch = 0; ch < m_channels; ++ch) {
531 out[ch] = dotProductAVX(&m_history.data[ch][windowIdx], coeffs, numTaps);
532 }
533 } else {
534 for (uint32_t ch = 0; ch < m_channels; ++ch) {
535 out[ch] = dotProductScalar(&m_history.data[ch][windowIdx], coeffs, numTaps);
536 }
537 }
538#elif defined(AESTRA_HAS_SSE)
539 if (useSIMD) {
540 for (uint32_t ch = 0; ch < m_channels; ++ch) {
541 out[ch] = dotProductSSE(&m_history.data[ch][windowIdx], coeffs, numTaps);
542 }
543 } else {
544 for (uint32_t ch = 0; ch < m_channels; ++ch) {
545 out[ch] = dotProductScalar(&m_history.data[ch][windowIdx], coeffs, numTaps);
546 }
547 }
548#else
549 (void)useSIMD;
550 for (uint32_t ch = 0; ch < m_channels; ++ch) {
551 out[ch] = dotProductScalar(&m_history.data[ch][windowIdx], coeffs, numTaps);
552 }
553#endif
554 }
555
556 ++outputFrames;
557 outPtr += m_channels;
558 nextOutputSrcPos += invRatio;
559 srcNextDiff -= invRatio;
560 historyPos += invRatio;
561 }
562 }
563
564 // Write back hoisted locals to member variables
565 m_srcPosition = srcPosition;
566 m_nextOutputSrcPos = nextOutputSrcPos;
567
568 return outputFrames;
569}

571float SampleRateConverter::interpolateSample(uint32_t channel, uint32_t phaseIndex, int32_t centerPos) const noexcept {
572 const PolyphaseFilterBank* bank = getFilterBank();
573 if (!bank)
574 return 0.0f; // Should not happen if configured
575
576 const uint32_t numTaps = bank->numTaps;
577 const uint32_t halfTaps = bank->halfTaps;
578
579 const int32_t samplePos0 = centerPos - static_cast<int32_t>(halfTaps);
580 const float* window = m_history.getWindow(channel, samplePos0);
581 const float* coeffs = bank->coeffs[phaseIndex];
582
583#ifdef AESTRA_HAS_AVX
584 if (m_simdEnabled.load(std::memory_order_relaxed) && hasSIMD()) {
585 return dotProductAVX(window, coeffs, numTaps);
586 }
587#elif defined(AESTRA_HAS_SSE)
588 if (m_simdEnabled.load(std::memory_order_relaxed) && hasSIMD()) {
589 return dotProductSSE(window, coeffs, numTaps);
590 }
591#endif
592
593 return dotProductScalar(window, coeffs, numTaps);
594}
