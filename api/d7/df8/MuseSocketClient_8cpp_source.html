56bool MuseSocketClient::connect(const std::string& host, uint16_t port, std::string& outError) {
57#ifdef _WIN32
58 static std::once_flag once;
59 std::call_once(once, []() {
60 WSADATA data{};
61 ::WSAStartup(MAKEWORD(2, 2), &data);
62 });
63#endif
64 m_impl->socket = ::socket(AF_INET, SOCK_STREAM, 0);
65 if (m_impl->socket == kInvalidSocket) {
66 outError = "socket() failed";
67 return false;
68 }
69#ifdef SO_NOSIGPIPE
70 {
71 const int enable = 1;
72 ::setsockopt(m_impl->socket, SOL_SOCKET, SO_NOSIGPIPE, &enable, sizeof(enable));
73 }
74#endif
75 sockaddr_in address{};
76 address.sin_family = AF_INET;
77 address.sin_port = htons(port);
78 if (::inet_pton(AF_INET, host.c_str(), &address.sin_addr) != 1) {
79 outError = "invalid host address: " + host;
80 closeSocket(m_impl->socket);
81 m_impl->socket = kInvalidSocket;
82 return false;
83 }
84 if (::connect(m_impl->socket, reinterpret_cast<sockaddr*>(&address), sizeof(address)) != 0) {
85 outError = "connect to " + host + ":" + std::to_string(port) + " failed";
86 closeSocket(m_impl->socket);
87 m_impl->socket = kInvalidSocket;
88 return false;
89 }
90 return true;
91}

93void MuseSocketClient::setReadTimeoutMs(int milliseconds) {
94 if (m_impl->socket == kInvalidSocket || milliseconds < 0) {
95 return;
96 }
97#ifdef _WIN32
98 const DWORD value = static_cast<DWORD>(milliseconds);
99 ::setsockopt(m_impl->socket, SOL_SOCKET, SO_RCVTIMEO,
100 reinterpret_cast<const char*>(&value), sizeof(value));
101#else
102 timeval value{};
103 value.tv_sec = milliseconds / 1000;
104 value.tv_usec = (milliseconds % 1000) * 1000;
105 ::setsockopt(m_impl->socket, SOL_SOCKET, SO_RCVTIMEO, &value, sizeof(value));
106#endif
107}

111// SO_RCVTIMEO expiry looks like any other recv failure, so ask the platform
112// which one it was — a timeout is recoverable information for the caller,
113// a dead peer is not.
114bool lastRecvWasTimeout() {
115#ifdef _WIN32
116 const int code = ::WSAGetLastError();
117 return code == WSAETIMEDOUT;
118#else
119 return errno == EAGAIN || errno == EWOULDBLOCK;
120#endif
121}

125std::string MuseSocketClient::request(const std::string& line, Outcome* outOutcome) {
126 const auto report = [&](Outcome outcome) {
127 if (outOutcome != nullptr) *outOutcome = outcome;
128 };
129 report(Outcome::Ok);
130
131 if (m_impl->socket == kInvalidSocket) {
132 report(Outcome::Disconnected);
133 return "{\"status\": \"execution_error\", \"message\": \"not connected\"}";
134 }
135
136 std::string toSend = line;
137 toSend += '\n';
138 size_t sent = 0;
139 while (sent < toSend.size()) {
140 const auto n = ::send(m_impl->socket, toSend.data() + sent,
141#ifdef _WIN32
142 static_cast<int>(toSend.size() - sent),
143#else
144 toSend.size() - sent,
145#endif
146 kSendFlags);
147 if (n <= 0) {
148 closeSocket(m_impl->socket);
149 m_impl->socket = kInvalidSocket;
150 report(Outcome::Disconnected);
151 return "{\"status\": \"execution_error\", \"message\": \"connection lost on send\"}";
152 }
153 sent += static_cast<size_t>(n);
154 }
155
156 size_t newline;
157 while ((newline = m_impl->readBuffer.find('\n')) == std::string::npos) {
158 char chunk[4096];
159 const auto received = ::recv(m_impl->socket, chunk, sizeof(chunk), 0);
160 if (received <= 0) {
161 // A timeout leaves the connection usable in principle, but this
162 // request is unanswerable — close either way so a later request
163 // cannot read this one's late response as its own.
164 const bool timedOut = (received < 0) && lastRecvWasTimeout();
165 closeSocket(m_impl->socket);
166 m_impl->socket = kInvalidSocket;
167 report(timedOut ? Outcome::TimedOut : Outcome::Disconnected);
168 return timedOut
169 ? "{\"status\": \"execution_error\", \"message\": \"timed out waiting for a response\"}"
170 : "{\"status\": \"execution_error\", \"message\": \"connection lost on recv\"}";
171 }
172 m_impl->readBuffer.append(chunk, static_cast<size_t>(received));
173 }
174 std::string response = m_impl->readBuffer.substr(0, newline);
175 m_impl->readBuffer.erase(0, newline + 1);
176 if (!response.empty() && response.back() == '\r') response.pop_back();
177 return response;
178}
