3// PluginEditorRegistry — the factories, half of P4's dispatch table.
4//
5// The id/name table is header-inline, so a test can check every plugin id without
6// linking this file. What lives here is the part that cannot be checked without a
7// live instance and a platform bridge: the lambdas that actually construct the
8// eleven editors.
9//
10// The factories are paired with registry rows BY EDITOR NAME, not by position.
11// The name comes from the editor's own type (#T below), so a factory cannot sit
12// under the wrong name, and the static_asserts check that the two lists name the
13// same set of editors. Reordering either list is harmless; a row with no factory,
14// or a factory with no row, does not build.

43template <typename EditorT>
44PluginEditorFactory factoryFor() {
45 // The instance is a shared_ptr because every editor takes one and stores it.
46 return [](std::shared_ptr<Aestra::Audio::IPluginInstance> instance, NUIPlatformBridge* bridge)
47 -> std::shared_ptr<AestraPanelWindow> {
48 if (!instance)
49 return nullptr;
50 auto editor = std::make_shared<EditorT>(instance);
51 editor->setPlatformBridge(bridge);
52 return editor;
53 };
54}

82constexpr bool sameName(const char* a, const char* b) {
83 while (*a != '\0' && *a == *b) {
84 ++a;
85 ++b;
86 }
87 return *a == *b;
88}
89
90constexpr bool isEditorName(const char* name) {
91 for (const char* candidate : kEditorNames)
92 if (sameName(candidate, name))
93 return true;
94 return false;
95}
96
97constexpr bool everyRowHasAFactory() {
98 for (const PluginEditorRegistration& r : pluginEditorRegistry)
99 if (!isEditorName(r.editorName))
100 return false;
101 return true;
102}
103
104constexpr bool everyFactoryHasARow() {
105 for (const char* name : kEditorNames) {
106 bool found = false;
107 for (const PluginEditorRegistration& r : pluginEditorRegistry)
108 found = found || sameName(r.editorName, name);
109 if (!found)
110 return false;
111 }
112 return true;
113}
114
115static_assert(everyRowHasAFactory(),
116 "a pluginEditorRegistry row names an editor with no factory -- add it to AESTRA_CORE_PLUGIN_EDITORS");
117static_assert(everyFactoryHasARow(), "an editor factory has no pluginEditorRegistry row");
118static_assert(sameName(pluginEditorRegistry.front().editorName, "GenericPluginEditor"),
119 "the first registry row is the generic fallback");

121struct NamedFactory {
122 const char* name;
123 PluginEditorFactory factory;
124};
125
126constexpr size_t kEditorCount = sizeof(kEditorNames) / sizeof(kEditorNames[0]);
127
128const std::array<NamedFactory, kEditorCount>& pluginEditorFactories() {
129#define AESTRA_EDITOR_FACTORY(T) NamedFactory{#T, factoryFor<T>()},
130 static const std::array<NamedFactory, kEditorCount> kFactories{
131 {AESTRA_CORE_PLUGIN_EDITORS(AESTRA_EDITOR_FACTORY) AESTRA_PREMIUM_PLUGIN_EDITORS(AESTRA_EDITOR_FACTORY)}};
132#undef AESTRA_EDITOR_FACTORY
133 return kFactories;
134}

136const PluginEditorFactory& factoryNamed(const char* editorName) {
137 const auto& factories = pluginEditorFactories();
138 for (const NamedFactory& f : factories)
139 if (std::strcmp(f.name, editorName) == 0)
140 return f.factory;
141 // Unreachable: the static_asserts above prove every registry name has a
142 // factory. Kept as the generic editor rather than a null factory.
143 return factories.front().factory;
144}
