31void AestraContent::applyPanelMode(Audio::ViewType view) {

54void AestraContent::showPanelMenu(Audio::ViewType view, const AestraUI::NUIPoint& at) {
55 const auto* spec = specFor(view);
56 if (!spec || !m_overlayLayer)
57 return;
58 AestraUI::PanelModeMenuActions actions;
59 actions.setMode = [this, view, key = std::string(spec->storeKey)](AestraUI::PanelMode next) {
60 writePanelMode(surfaceStore(), key, next);
61 applyPanelMode(view);
62 };
63 actions.resetPosition = [this, view, spec]() {
64 savePanelPreference(view, defaultSurfacePreference(spec->defaultWidth, spec->defaultHeight,
65 spec->defaultAnchorX, spec->defaultAnchorY));
66 applyPanelPreference(view);
67 setDirty(true);
68 };
69 auto menu = AestraUI::buildPanelModeMenu(panelMode(view), std::move(actions));
70 menu->setOnHide([this, weak = std::weak_ptr<AestraUI::NUIContextMenu>(menu)]() {
71 if (auto shown = weak.lock(); shown && m_overlayLayer)
72 m_overlayLayer->removeChild(shown);
73 });
74 m_overlayLayer->addChild(menu);
75 menu->showAt(at);
76}

78// A contextual panel sees EVERY left press (the overlay forwards them all), including the click on its own
79// toggle button in the transport. So it only notes the press; closeContextualPanels() decides a frame
80// later, after whatever the press did (opening or closing a view) has happened.
81void AestraContent::noteOutsidePress(Audio::ViewType view) {
82 if (m_outsidePressViews.empty())
83 m_outsidePressSerial = m_viewChangeSerial;
84 if (std::find(m_outsidePressViews.begin(), m_outsidePressViews.end(), view) == m_outsidePressViews.end())
85 m_outsidePressViews.push_back(view);
86}
87
88void AestraContent::closeContextualPanels() {
89 if (m_outsidePressViews.empty())
90 return;
91 const auto noted = std::move(m_outsidePressViews);
92 m_outsidePressViews.clear();
93 const auto toClose = contextualViewsToClose(noted, m_viewChangeSerial != m_outsidePressSerial, [this](Audio::ViewType view) {
94 auto panel = panelForView(view);
95 return panel && panel->isVisible() && panel->isContextual();
96 });
97 for (const auto view : toClose)
98 setViewOpen(view, false);
99}

101void AestraContent::gatherPanels() {
102 for (const auto& g : Content::kFloatingPanelGeometry) {
103 auto panel = panelForView(g.view);
104 if (panel && panel->isVisible())
105 applyPanelPreference(g.view);
106 }
107 if (m_pluginController)
108 m_pluginController->gatherEditors();
109 setDirty(true);
110}

112void AestraContent::resetPanelPositions() {
113 for (const auto& g : Content::kFloatingPanelGeometry) {
114 savePanelPreference(g.view, defaultSurfacePreference(g.defaultWidth, g.defaultHeight, g.defaultAnchorX,
115 g.defaultAnchorY));
116 if (auto panel = panelForView(g.view); panel && panel->isVisible())
117 applyPanelPreference(g.view);
118 }
119 if (m_pluginController)
120 m_pluginController->resetEditorPositions();
121 setDirty(true);
122}
