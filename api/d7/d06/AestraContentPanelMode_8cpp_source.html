58void AestraContent::showPanelMenu(Audio::ViewType view, const AestraUI::NUIPoint& at) {
59 const auto* spec = specFor(view);
60 if (!spec || !m_overlayLayer)
61 return;
62 AestraUI::PanelModeMenuActions actions;
63 actions.setMode = [this, view, key = std::string(spec->storeKey)](AestraUI::PanelMode next) {
64 writePanelMode(surfaceStore(), key, next);
65 applyPanelMode(view);
66 };
67 actions.resetPosition = [this, view, spec]() {
68 savePanelPreference(view, defaultSurfacePreference(spec->defaultWidth, spec->defaultHeight,
69 spec->defaultAnchorX, spec->defaultAnchorY, spec->defaultMaximized));
70 applyPanelPreference(view);
71 setDirty(true);
72 };
73 auto menu = AestraUI::buildPanelModeMenu(panelMode(view), std::move(actions));
74 menu->setOnHide([this, weak = std::weak_ptr<AestraUI::NUIContextMenu>(menu)]() {
75 if (auto shown = weak.lock(); shown && m_overlayLayer)
76 m_overlayLayer->removeChild(shown);
77 });
78 m_overlayLayer->addChild(menu);
79 menu->showAt(at);
80}

82// A contextual panel sees EVERY left press (the overlay forwards them all), including the click on its own
83// toggle button in the transport. So it only notes the press; closeContextualPanels() decides a frame
84// later, after whatever the press did (opening or closing a view) has happened.
85void AestraContent::noteOutsidePress(Audio::ViewType view) {
86 if (m_outsidePressViews.empty())
87 m_outsidePressSerial = m_viewChangeSerial;
88 if (std::find(m_outsidePressViews.begin(), m_outsidePressViews.end(), view) == m_outsidePressViews.end())
89 m_outsidePressViews.push_back(view);
90}

92void AestraContent::closeContextualPanels() {
93 if (m_outsidePressViews.empty())
94 return;
95 const auto noted = std::move(m_outsidePressViews);
96 m_outsidePressViews.clear();
97 const auto toClose = contextualViewsToClose(noted, m_viewChangeSerial != m_outsidePressSerial, [this](Audio::ViewType view) {
98 auto panel = panelForView(view);
99 return panel && panel->isVisible() && panel->isContextual();
100 });
101 for (const auto view : toClose)
102 setViewOpen(view, false);
103}

105// A panel button, F-key or open request on a panel that is open but under another one raises it instead of closing
106// it (owner report, 2026-10-18: the Arsenal button closed an Arsenal that sat behind the piano roll). Only a panel
107// nothing overlaps from above counts as in front, so the button still closes the panel you are looking at.
108bool AestraContent::raiseIfCovered(Audio::ViewType view) {
109 auto panel = panelForView(view);
110 if (!panel || !panel->isVisible() || !m_overlayLayer)
111 return false;
112 const auto& layer = m_overlayLayer->getChildren();
113 auto it = std::find(layer.begin(), layer.end(), std::static_pointer_cast<AestraUI::NUIComponent>(panel));
114 if (it == layer.end())
115 return false;
116 const bool covered = std::any_of(std::next(it), layer.end(), [&panel](const auto& above) {
117 return std::dynamic_pointer_cast<Audio::WindowPanel>(above) && above->isVisible() &&
118 above->getBounds().intersects(panel->getBounds());
119 });
120 if (covered) {
121 ++m_viewChangeSerial; // a raise is a view change: the button press must not then close a contextual panel
122 panel->bringToFrontPriority();
123 setDirty(true);
124 }
125 return covered;
126}

130 auto panel = panelForView(g.view);
