59 const float luma = (0.2126f * color.r) + (0.7152f * color.g) + (0.0722f * color.b);
60 const float tonedR = ((color.r - luma) * saturationScale + luma) * brightnessScale;
61 const float tonedG = ((color.g - luma) * saturationScale + luma) * brightnessScale;

77// ---------------------------------------------------------------------------
78// Clip colour contract
79//
80// A clip is a flat, opaque surface in its identity hue — the same hue the mixer
81// strips paint — with its content (waveform, notes, label) drawn as a LIGHT ink
82// on top. Owner direction 2026-09-19 (spec 2 §1): foreground legibility is
83// separated from routing identity. Routing colour still decides the hue; it no
84// longer decides whether the foreground comes out light or dark.
85//
86// The previous contract derived ink by deepening the hue toward black and picked
87// label colour per hue by luminance. That made a routed clip's foreground flip
88// between near-black and near-white depending on which channel it fed, and on
89// the dark hues (purple, indigo) the near-black ink sat on a near-black body.
90//
91// The rule now has two halves that only work together:
92//
93// 1. Every body is capped to one luminance ceiling. Hue and saturation are
94// untouched (one scalar multiplies all three channels), so a clip still
95// reads as its channel's colour — it is simply never bright enough to
96// swallow a light foreground.
97// 2. Every foreground is lifted toward white from the surface it sits on, so
98// label and waveform belong to one family instead of being independently
99// chosen colours.
100//
101// Body and ink live here together because what matters is the *gap* between them
102// and its direction; separate the two derivations and a later edit can quietly
103// collapse or invert it. TimelineClipContrastTest guards both.
104// ---------------------------------------------------------------------------

107inline float clipRelativeLuminance(const NUIColor& color) {
108 const auto channel = [](float v) {
109 v = std::clamp(v, 0.0f, 1.0f);
110 return (v <= 0.03928f) ? (v / 12.92f) : std::pow((v + 0.055f) / 1.055f, 2.4f);
111 };
112 return 0.2126f * channel(color.r) + 0.7152f * channel(color.g) + 0.0722f * channel(color.b);
113}

141inline NUIColor capClipLuminance(const NUIColor& color, float ceiling) {
142 if (clipRelativeLuminance(color) <= ceiling) {
143 return color;
144 }
145 const auto scaled = [&color](float k) {
146 return NUIColor(color.r * k, color.g * k, color.b * k, color.a);
147 };
148 float lo = 0.0f;
149 float hi = 1.0f;
150 for (int i = 0; i < 14; ++i) {
151 const float mid = (lo + hi) * 0.5f;
152 if (clipRelativeLuminance(scaled(mid)) > ceiling) {
153 hi = mid;
154 } else {
155 lo = mid;
156 }
157 }
158 return scaled(lo);
159}

202inline int nearestPaletteIndex(uint32_t argb) {
203 if (argb == 0) return -1;
204 int best = 0;
205 int bestDist = INT_MAX;
206 auto parseARGB = [](uint32_t c, int& r, int& g, int& b) {
207 r = (c >> 16) & 0xFF;
208 g = (c >> 8) & 0xFF;
209 b = c & 0xFF;
210 };
211 int r0, g0, b0;
212 parseARGB(argb, r0, g0, b0);
213 for (int i = 0; i < PALETTE_SIZE; ++i) {
214 int r, g, b;
215 parseARGB(TRACK_PALETTE[i], r, g, b);
216 int dr = r - r0, dg = g - g0, db = b - b0;
217 int dist = dr * dr + dg * dg + db * db;
218 if (dist < bestDist) {
219 bestDist = dist;
220 best = i;
221 }
222 }
223 return (bestDist < 5000) ? best : -1;
224}
