52inline NUIColor restrainDawColor(const NUIColor& color, float brightnessScale, float saturationScale, float alpha) {
53 const float luma = (0.2126f * color.r) + (0.7152f * color.g) + (0.0722f * color.b);
54 const float tonedR = ((color.r - luma) * saturationScale + luma) * brightnessScale;
55 const float tonedG = ((color.g - luma) * saturationScale + luma) * brightnessScale;
56 const float tonedB = ((color.b - luma) * saturationScale + luma) * brightnessScale;
57 return NUIColor(std::clamp(tonedR, 0.0f, 1.0f), std::clamp(tonedG, 0.0f, 1.0f), std::clamp(tonedB, 0.0f, 1.0f),
58 alpha >= 0.0f ? alpha : color.a);
59}

71// ---------------------------------------------------------------------------
72// Clip contrast contract
73//
74// A clip is a surface with audio drawn on it. The body is deliberately toned
75// well down; the waveform keeps the identity hue near full strength and is then
76// lifted toward white. Both halves live here, together, because the thing that
77// matters is not either constant but the *gap* between them — separate the two
78// derivations and a later edit can quietly invert the hierarchy, which is the
79// bug this pairing exists to prevent. TimelineClipContrastTest guards the gap.
80// ---------------------------------------------------------------------------
81
83inline NUIColor clipBodyTone(const NUIColor& identity, bool selected) {
84 return restrainDawColor(identity, selected ? 0.78f : 0.68f, selected ? 0.62f : 0.56f, 1.0f);

103inline int nearestPaletteIndex(uint32_t argb) {
104 if (argb == 0) return -1;
105 int best = 0;
106 int bestDist = INT_MAX;
107 auto parseARGB = [](uint32_t c, int& r, int& g, int& b) {
108 r = (c >> 16) & 0xFF;
109 g = (c >> 8) & 0xFF;
110 b = c & 0xFF;
111 };
112 int r0, g0, b0;
113 parseARGB(argb, r0, g0, b0);
114 for (int i = 0; i < PALETTE_SIZE; ++i) {
115 int r, g, b;
116 parseARGB(TRACK_PALETTE[i], r, g, b);
117 int dr = r - r0, dg = g - g0, db = b - b0;
118 int dist = dr * dr + dg * dg + db * db;
119 if (dist < bestDist) {
120 bestDist = dist;
121 best = i;
122 }
123 }
124 return (bestDist < 5000) ? best : -1;
125}
