3// V82-T2 (canvas-2 board 7): a new project has forty numbered tracks, almost all untouched. They fold into one row
4// under the last track, "31 empty tracks · show", and the header counts them. That row is where a drop makes a new
5// track (TrackManagerUIDragDrop), so dragging onto it still works. Only tracks nobody has touched fold: no clips, no
6// automation, the default "Track N" name, one lane. A renamed or used track always shows.

22void TrackManagerUI::renderFoldedEmptyRow(AestraUI::NUIRenderer& renderer) {
23 m_foldedToggleRect = {};
24 if (m_foldedEmptyTracks == 0 && !m_showEmptyTracks)
25 return;
26 if (m_foldedRowRect.isEmpty())
27 return;
28 auto& theme = AestraUI::NUIThemeManager::getInstance();
29 const float controlsW = theme.getLayoutDimensions().trackControlsWidth;
30 const AestraUI::NUIRect header(m_foldedRowRect.x, m_foldedRowRect.y, controlsW, m_foldedRowRect.height);
31 renderer.drawLine({m_foldedRowRect.x, m_foldedRowRect.bottom() - 0.5f},
32 {m_foldedRowRect.right(), m_foldedRowRect.bottom() - 0.5f}, 1.0f,
33 theme.getColor("border").withAlpha(0.5f));
34
35 const float font = 12.5f;
36 const float x = header.x + 34.0f;
37 const float y = renderer.calculateTextY(header, font);
38 std::string lead;
39 std::string action;
40 if (m_showEmptyTracks) {
41 action = "hide empty tracks";
42 } else {
43 lead = std::to_string(m_foldedEmptyTracks) + (m_foldedEmptyTracks == 1 ? " empty track" : " empty tracks") +
44 " \xc2\xb7 ";
45 action = "show";
46 }
47 renderer.drawText(lead, {x, y}, font, theme.getColor("textMuted"));
48 const float actionX = x + renderer.measureText(lead, font).width;
49 const float actionW = renderer.measureText(action, font).width;
50 renderer.drawText(action, {actionX, y}, font, theme.getColor("textSecondary"));
51 m_foldedToggleRect = {actionX - 4.0f, header.y, actionW + 8.0f, header.height};
52}

54bool TrackManagerUI::handleFoldedRowMouse(const AestraUI::NUIMouseEvent& event) {
55 if (!event.pressed || event.button != AestraUI::NUIMouseButton::Left)
56 return false;
57 if (m_foldedToggleRect.isEmpty() || !m_foldedToggleRect.contains(event.position))
58 return false;
59 m_showEmptyTracks = !m_showEmptyTracks;
60 refreshTracks();
61 layoutTracks();
62 invalidateCache();
63 setDirty(true);
64 return true;
65}
