46void AestraLFOEditor::layoutControls() {
47 const auto b = getBounds();
48 const float contentTop = b.y + AestraPanelWindow::TITLE_BAR_H;
49
50 // Target selector: three segments across the top-left
51 constexpr float kSegW = 62.0f;
52 constexpr float kSegH = 26.0f;
53 for (size_t i = 0; i < m_targetRects.size(); ++i) {
54 m_targetRects[i] =
55 NUIRect(b.x + 28.0f + static_cast<float>(i) * (kSegW + 6.0f), contentTop + 14.0f, kSegW, kSegH);
56 }
57
58 constexpr float kBypassW = 88.0f;
59 constexpr float kBypassH = 26.0f;
60 m_bypassRect = NUIRect(b.right() - 44.0f - kBypassW, contentTop + 14.0f, kBypassW, kBypassH);
61 m_syncRect = NUIRect(m_bypassRect.x - 70.0f - 12.0f, contentTop + 14.0f, 70.0f, kBypassH);
62
63 // Wave selector: six segments on the second row
64 constexpr float kWaveW = 52.0f;
65 constexpr float kWaveH = 24.0f;
66 for (size_t i = 0; i < m_waveRects.size(); ++i) {
67 m_waveRects[i] =
68 NUIRect(b.x + 28.0f + static_cast<float>(i) * (kWaveW + 6.0f), contentTop + 52.0f, kWaveW, kWaveH);
69 }
70
71 // Large rate knob on the left; small knobs to its right
72 const float knobTop = contentTop + 96.0f;
73 m_rateRect = NUIRect(b.x + 44.0f, knobTop, 104.0f, 104.0f);
74 m_depthRect = NUIRect(b.x + 216.0f, knobTop + 16.0f, 64.0f, 64.0f);
75 m_phaseRect = NUIRect(b.x + 330.0f, knobTop + 16.0f, 64.0f, 64.0f);
76 m_smoothRect = NUIRect(b.x + 444.0f, knobTop + 16.0f, 64.0f, 64.0f);
77}

80 if (!m_instance)

185bool AestraLFOEditor::onMouseEvent(const NUIMouseEvent& event) {
186 if (!isVisible())
187 return false;
188 if (AestraPanelWindow::onMouseEvent(event))
189 return true;
190 if (!m_instance)
191 return false;
192
193
194 // Bypass pill
195 if (m_bypassRect.contains(event.position) && event.pressed && event.button == NUIMouseButton::Left) {
196 const float cur = m_instance->getParameter(AestraLFO::kBypass);
197 m_instance->setParameter(AestraLFO::kBypass, cur > 0.5f ? 0.0f : 1.0f);
198 setDirty();
199 return true;
200 }
201 m_bypassHovered = m_bypassRect.contains(event.position);
202
203 // Sync pill
204 if (m_syncRect.contains(event.position) && event.pressed && event.button == NUIMouseButton::Left) {
205 const float cur = m_instance->getParameter(AestraLFO::kSyncMode);
206 m_instance->setParameter(AestraLFO::kSyncMode, cur > 0.5f ? 0.0f : 1.0f);
207 setDirty();
208 return true;
209 }
210
211 // Target / wave segments
212 if (event.pressed && event.button == NUIMouseButton::Left) {
213 for (size_t i = 0; i < m_targetRects.size(); ++i) {
214 if (m_targetRects[i].contains(event.position)) {
215 m_instance->setParameter(AestraLFO::kTarget,
216 AestraLFO::normFromTarget(static_cast<AestraLFO::Target>(i)));
217 setDirty();
218 return true;
219 }
220 }
221 for (size_t i = 0; i < m_waveRects.size(); ++i) {
222 if (m_waveRects[i].contains(event.position)) {
223 m_instance->setParameter(AestraLFO::kWave, AestraLFO::normFromWave(static_cast<AestraLFO::Wave>(i)));
224 setDirty();
225 return true;
226 }
227 }
228 }
229
230 // Knob vertical drag (rotary -> cursor capture): AestraPanelWindow::handleKnobDrag.
231 const bool sync = m_instance->getParameter(AestraLFO::kSyncMode) > 0.5f;
232 const uint32_t rateParam = sync ? AestraLFO::kNoteDivision : AestraLFO::kRateHz;
233 const KnobTarget knobs[] = {
234 {m_rateRect, rateParam},
235 {m_depthRect, AestraLFO::kDepth},
236 {m_phaseRect, AestraLFO::kPhase},
237 {m_smoothRect, AestraLFO::kSmooth},
238 };
239 const auto get = [this](uint32_t id) { return m_instance->getParameter(id); };
240 const auto set = [this](uint32_t id, float v) { m_instance->setParameter(id, v); };
241 if (handleKnobDrag(event, knobs, std::size(knobs), kDragRangePx, get, set))
242 return true;
243
244 return consumeInsideBounds(event);
245}
