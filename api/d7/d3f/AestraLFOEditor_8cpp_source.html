32void drawArc(NUIRenderer& renderer, NUIPoint center, float radius, float startAngle, float endAngle, float thickness,
33 NUIColor color) {
34 if (endAngle < startAngle)
35 std::swap(startAngle, endAngle);
36 if (endAngle - startAngle <= 0.001f)
37 return;
38 std::array<NUIPoint, 49> pts{};
39 const float div = static_cast<float>(pts.size() - 1);
40 for (size_t i = 0; i < pts.size(); ++i) {
41 const float t = static_cast<float>(i) / div;
42 const float a = startAngle + (endAngle - startAngle) * t;
43 pts[i] = {center.x + std::cos(a) * radius, center.y + std::sin(a) * radius};
44 }
45 renderer.drawPolyline(pts.data(), static_cast<int>(pts.size()), thickness, color);
46}

115void AestraLFOEditor::drawKnob(NUIRenderer& renderer, const NUIRect& rect, uint32_t paramId, const char* label,
116 bool large) {
117 auto& theme = NUIThemeManager::getInstance();
118 const float value = m_instance ? m_instance->getParameter(paramId) : 0.0f;
119 const NUIPoint c = rect.center();
120 const float r = rect.width * 0.5f - 4.0f;
121 const float angle = kKnobStart + value * kKnobSweep;
122
123 renderer.fillCircle(c, r + 4.0f, insetSurface());
124 renderer.strokeCircle(c, r + 4.0f, 1.0f, editorInk(0.060f));
125
126 drawArc(renderer, c, r - 3.0f, kKnobStart, kKnobStart + kKnobSweep, large ? 4.0f : 3.0f,
127 editorNeutral(0.199f, 1.0f));
128 drawArc(renderer, c, r - 3.0f, kKnobStart, angle, large ? 4.0f : 3.0f, accent().withAlpha(0.92f));
129
130 const float needleLen = r - (large ? 14.0f : 9.0f);
131 const NUIPoint tip(c.x + std::cos(angle) * needleLen, c.y + std::sin(angle) * needleLen);
132 renderer.drawLine(c, tip, 2.0f, accent().withAlpha(0.85f));
133 renderer.fillCircle(tip, large ? 3.5f : 2.5f, accent());
134
135 const float wellR = r * (large ? 0.34f : 0.28f);
136 renderer.fillCircle(c, wellR, editorNeutral(0.045f, 0.96f));
137 renderer.strokeCircle(c, wellR, 1.2f, accent().withAlpha(0.36f));
138
139 renderer.drawTextCentered(label, NUIRect(rect.x, rect.bottom() + 4.0f, rect.width, 14.0f), 10.5f,
140 theme.getColor("textPrimary").withAlpha(0.95f));
141 const std::string valStr = m_instance ? m_instance->getParameterDisplay(paramId) : "";
142 renderer.drawTextCentered(valStr, NUIRect(rect.x - 14.0f, rect.bottom() + 19.0f, rect.width + 28.0f, 13.0f),
143 large ? 10.0f : 9.0f, accent().withAlpha(0.96f));
144}

226bool AestraLFOEditor::onMouseEvent(const NUIMouseEvent& event) {
227 if (!isVisible())
228 return false;
229 if (AestraPanelWindow::onMouseEvent(event))
230 return true;
231 if (!m_instance)
232 return false;
233
234 const float my = event.position.y;
235
236 // Bypass pill
237 if (m_bypassRect.contains(event.position) && event.pressed && event.button == NUIMouseButton::Left) {
238 const float cur = m_instance->getParameter(AestraLFO::kBypass);
239 m_instance->setParameter(AestraLFO::kBypass, cur > 0.5f ? 0.0f : 1.0f);
240 setDirty();
241 return true;
242 }
243 m_bypassHovered = m_bypassRect.contains(event.position);
244
245 // Sync pill
246 if (m_syncRect.contains(event.position) && event.pressed && event.button == NUIMouseButton::Left) {
247 const float cur = m_instance->getParameter(AestraLFO::kSyncMode);
248 m_instance->setParameter(AestraLFO::kSyncMode, cur > 0.5f ? 0.0f : 1.0f);
249 setDirty();
250 return true;
251 }
252
253 // Target / wave segments
254 if (event.pressed && event.button == NUIMouseButton::Left) {
255 for (size_t i = 0; i < m_targetRects.size(); ++i) {
256 if (m_targetRects[i].contains(event.position)) {
257 m_instance->setParameter(AestraLFO::kTarget,
258 AestraLFO::normFromTarget(static_cast<AestraLFO::Target>(i)));
259 setDirty();
260 return true;
261 }
262 }
263 for (size_t i = 0; i < m_waveRects.size(); ++i) {
264 if (m_waveRects[i].contains(event.position)) {
265 m_instance->setParameter(AestraLFO::kWave, AestraLFO::normFromWave(static_cast<AestraLFO::Wave>(i)));
266 setDirty();
267 return true;
268 }
269 }
270 }
271
272 // Knob vertical drag
273 if (m_draggingParam >= 0) {
274 if (event.released) {
275 m_draggingParam = -1;
276 return true;
277 }
278 if (event.button == NUIMouseButton::None) {
279 const float delta = (m_dragStartY - my) / kDragRangePx;
280 m_instance->setParameter(static_cast<uint32_t>(m_draggingParam),
281 std::clamp(m_dragStartValue + delta, 0.0f, 1.0f));
282 setDirty();
283 return true;
284 }
285 }
286 if (event.pressed && event.button == NUIMouseButton::Left) {
287 const bool sync = m_instance->getParameter(AestraLFO::kSyncMode) > 0.5f;
288 const uint32_t rateParam = sync ? AestraLFO::kNoteDivision : AestraLFO::kRateHz;
289 const struct {
290 NUIRect rect;
291 uint32_t param;
292 } knobs[] = {
293 {m_rateRect, rateParam},
294 {m_depthRect, AestraLFO::kDepth},
295 {m_phaseRect, AestraLFO::kPhase},
296 {m_smoothRect, AestraLFO::kSmooth},
297 };
298 for (const auto& k : knobs) {
299 if (k.rect.contains(event.position)) {
300 m_draggingParam = static_cast<int>(k.param);
301 m_dragStartY = my;
302 m_dragStartValue = m_instance->getParameter(k.param);
303 return true;
304 }
305 }
306 }
307
308 return consumeInsideBounds(event);
309}
