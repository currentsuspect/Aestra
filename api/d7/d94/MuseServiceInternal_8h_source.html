4// The seam between MuseService and its verb families.
5//
6// MuseService.cpp was one 2200-line if-chain. Tests/Guards/file_size_baseline.txt
7// pinned it at its exact line count, so it could never grow again — which is the
8// guard doing its job: a hub that every new verb has to be threaded through is a
9// hub nobody can review. The verb bodies now live one family per translation
10// unit, and this header is what they share.
11//
12// The move is mechanical. Each family handler re-declares the three envelope
13// helpers handleRequest() used to declare — makeOk / finish / makeError — with
14// the names and signatures the moved code already used, and binds `id`, `verb`,
15// `request`, `m_trackManager`, `m_engine` and `m_projectLoadReport` to locals of
16// the same names. A verb body therefore still reads exactly as it did before,
17// which is what makes "this refactor changed no behaviour" a claim you can
18// check rather than take on trust.
19//
20// What moved here rather than into a family file: the helpers more than one
21// family calls, plus the verb classification, because handleRequest and the
22// batch verb must agree on it. There is one copy of each, as before.

81class ResponseEnvelope {
82public:
83 ResponseEnvelope(double id, std::string verb)
84 : m_id(id), m_verb(std::move(verb)), m_start(std::chrono::steady_clock::now()) {}
85
86 JSON ok() const {
87 JSON response = JSON::object();
88 response.set("id", JSON(m_id));
89 response.set("status", JSON("ok"));
90 response.set("verb", JSON(m_verb));
91 return response;
92 }
93
94 std::string finish(JSON& response) const {
95 const auto end = std::chrono::steady_clock::now();
96 response.set("executionMs",
97 JSON(std::chrono::duration<double, std::milli>(end - m_start).count()));
98 return response.toString();
99 }
100
101private:
102 double m_id;
103 std::string m_verb;
104 std::chrono::steady_clock::time_point m_start;
105};

124// Render a JSON number the way the flag validators expect to re-parse it:
125// integral values without a fractional tail ("142", not "142.000000").
126inline std::string numberToFlagString(double value) {
127 if (std::isfinite(value) && value == std::floor(value) &&
128 std::abs(value) < 9.007199254740992e15) {
129 char buf[32];
130 std::snprintf(buf, sizeof(buf), "%.0f", value);
131 return buf;
132 }
133 char buf[64];
134 std::snprintf(buf, sizeof(buf), "%.9g", value);
135 return buf;
136}

148// JSON numbers are doubles; an id must be a non-negative integer that a
149// double represents exactly, or the cast would silently target the wrong
150// object.
151inline bool numberToId(double value, uint64_t& out) {
152 constexpr double kMaxExactJsonInteger = 9007199254740991.0; // 2^53 - 1
153 if (!std::isfinite(value) || value < 0.0 || value != std::floor(value) ||
154 value > kMaxExactJsonInteger) {
155 return false;
156 }
157 out = static_cast<uint64_t>(value);
158 return true;
159}

161// JSON args -> flag map for the schema/registry path. Returns false with
162// outError set when a value has a type the flag grammar cannot carry.
163inline bool jsonArgsToFlags(JSON& args, std::unordered_map<std::string, std::string>& outFlags,
164 std::string& outError) {
165 // Non-const asObject() returns a copy of the map (the const overload
166 // returns an empty static — a known footgun).
167 for (auto& entry : args.asObject()) {
168 const std::string& key = entry.first;
169 JSON& value = entry.second;
170 if (value.isString()) {
171 outFlags[key] = value.asString();
172 } else if (value.isNumber()) {
173 outFlags[key] = numberToFlagString(value.asNumber());
174 } else if (value.isBool()) {
175 outFlags[key] = value.asBool() ? "true" : "false";
176 } else {
177 outError = "arg '" + key + "' must be a string, number, or bool";
178 return false;
179 }
180 }
181 return true;
182}

197// --- Verb classification -----------------------------------------------------

203// Query verbs MuseService answers directly (mutations live in MuseGrammar).
204inline bool isQueryVerb(const std::string& verb) {
205 return verb == "get_transport" || verb == "list_tracks" || verb == "list_clips" || verb == "get_session_state" ||
206 verb == "list_units" || verb == "get_pattern" || verb == "list_patterns" || verb == "list_plugins" ||
207 verb == "get_effects" || verb == "list_samples" || verb == "get_meters" || verb == "get_schema" ||
208 verb == "get_capabilities" || verb == "get_audio_health" || verb == "get_project_load_report" ||
209 verb == "get_routing_graph" || verb == "get_latency_report";
210}

212// The few queries that take arguments; every other query rejects them.
213inline bool queryTakesArgs(const std::string& verb) {
214 return verb == "get_pattern" || verb == "get_effects" || verb == "list_samples";
215}

217// Service actions: handled here like queries, but they do work (render a
218// file, run a batch) rather than read state. render_pattern is not routed
219// through CommandHistory — a bounce is not an undoable project edit; batch
220// pushes one CommandTransaction so the whole group is a single undo step.
221// verify_render is here rather than in the grammar table for the same reason:
222// it renders, and a render is not an undoable project edit either.
223inline bool isActionVerb(const std::string& verb) {
224 return verb == "render_pattern" || verb == "render_song" || verb == "batch" ||
225 verb == "undo" || verb == "redo" || verb == "verify_render";
226}

228inline bool isKnownVerb(const std::string& verb) {
229 if (isQueryVerb(verb) || isActionVerb(verb)) return true;
230 for (const auto& cmd : MuseGrammar::allCommands()) {
231 if (cmd.verb == verb) return true;
232 }
233 return false;
234}
