54inline std::optional<FitToBarsResult> computeFitToBars(
55 double sourceRegionSeconds, double bpm, int bars, float pitchSemitones = 0.0f) {
56 if (!(sourceRegionSeconds > 0.0) || !std::isfinite(sourceRegionSeconds) ||
57 !(bpm > 0.0) || !std::isfinite(bpm) || bars <= 0) {
58 return std::nullopt;
59 }
60
61 FitToBarsResult result;
62 result.durationBeats = static_cast<double>(bars) * FIT_BARS_BEATS_PER_BAR;
63 result.durationSeconds = result.durationBeats * 60.0 / bpm;
64 if (!(result.durationSeconds > 0.0) || !std::isfinite(result.durationSeconds)) {
65 return std::nullopt;
66 }
67
68 // Effective varispeed required to fill the span, then decomposed into the
69 // STORED base rate. effectiveVarispeed() clamps BOTH factors to the
70 // envelope, so the base must stay inside [0.25, 4] too: a base like 8.0
71 // at -24 st would render at clamp(8.0) x 0.25 = 1.0, not the requested
72 // 2.0. Either clamp means content cannot exactly fill the span and is
73 // reported on rateClamped.
74 const double effectiveRate = sourceRegionSeconds / result.durationSeconds;
75 const bool effectiveClamped = effectiveRate < 0.25 || effectiveRate > 4.0;
76 const double baseRate = effectiveRate /
77 static_cast<double>(ClipEdits::playbackRateFromSemitones(pitchSemitones));
78 const bool baseClamped = baseRate < 0.25 || baseRate > 4.0;
79 result.rateClamped = effectiveClamped || baseClamped;
80 result.playbackRate = static_cast<float>(std::clamp(baseRate, 0.25, 4.0));
81 return result;
82}
