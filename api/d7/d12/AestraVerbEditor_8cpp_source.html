47void drawVerbArc(NUIRenderer& renderer,
48 NUIPoint center,
49 float radius,
50 float startAngle,
51 float endAngle,
52 float thickness,
53 NUIColor color) {
54 if (endAngle < startAngle) std::swap(startAngle, endAngle);
55 if (endAngle - startAngle <= 0.001f) return;
56 std::array<NUIPoint, 49> points{};
57 const float pointsDivisor = points.size() > 1 ? static_cast<float>(points.size() - 1) : 1.0f;
58 for (size_t i = 0; i < points.size(); ++i) {
59 const float t = static_cast<float>(i) / pointsDivisor;
60 const float angle = startAngle + (endAngle - startAngle) * t;
61 points[i] = {center.x + std::cos(angle) * radius, center.y + std::sin(angle) * radius};
62 }
63 renderer.drawPolyline(points.data(), static_cast<int>(points.size()), thickness, color);
64}

66float protectedLeftDockEdge(const NUIComponent* root, const NUIComponent* self, const NUIRect& parentBounds) {
67 if (!root || parentBounds.width <= 0.0f || parentBounds.height <= 0.0f) return parentBounds.x;
68 float edge = parentBounds.x;
69 const auto scan = [&](const auto& recurse, const NUIComponent* node) -> void {
70 if (!node || node == self || !node->isVisible()) return;
71 const auto b = node->getBounds();
72 const bool leftDocked = b.x <= parentBounds.x + 8.0f;
73 const bool railSized = b.width >= 96.0f && b.width <= std::min(620.0f, parentBounds.width * 0.46f);
74 const bool verticallySignificant = b.height >= parentBounds.height * 0.40f && b.bottom() >= parentBounds.y + 120.0f;
75 if (leftDocked && railSized && verticallySignificant) {
76 edge = std::max(edge, b.right());
77 }
78 for (const auto& child : node->getChildren()) {
79 recurse(recurse, child.get());
80 }
81 };
82 scan(scan, root);
83 return edge;
84}

239 {"Pre Sync", kPredelaySync, 0.0f}, {"Mod Char", kModCharacter, 0.0f}
240 };
241 for (const auto& meta : metas) {
242 KnobControl k;
243 k.label = meta.label;
244 k.paramId = meta.id;
245 k.defaultValue = meta.defaultValue;
246 auto slider = std::make_shared<NUISlider>();
247 slider->setStyle(NUISlider::Style::Rotary);
248 slider->setRange(0.0, 1.0);
249 slider->setValue(std::clamp(getParamValue(meta.id), 0.0f, 1.0f));
250 slider->setPlatformBridge(getPlatformBridge());
251 slider->setOnValueChange([this, paramId = meta.id](double value) {
252 if (m_instance) {
253 updateParameter(paramId, static_cast<float>(std::clamp(value, 0.0, 1.0)));
254 repaint();
255 }
256 });
257 k.slider = slider;
258 addChild(slider);
259 m_knobs.push_back(k);
260 }
261 layoutControls();

445 auto* parent = getParent();
446 if (!parent) return;
447 const NUIRect parentBounds = parent->getBounds();
448 if (parentBounds.width <= 1.0f || parentBounds.height <= 1.0f) return;
449 constexpr float kSafeMargin = 14.0f;
450 constexpr float kChromeReserve = 56.0f;
451 constexpr float kMinSafeWidth = 640.0f;
452 constexpr float kMinSafeHeight = 500.0f;

454 ? kChromeReserve : 0.0f;

456 parentBounds.y + topReserve + kSafeMargin,

461 const float dockEdge = protectedLeftDockEdge(root, this, parentBounds);
462 if (dockEdge > usable.x) {

469 float targetH = std::min(kWinH, usable.height);

1097void AestraVerbEditor::loadUserPresets() {
1098 m_userPresets.clear();
1099 const std::string dir = getUserPresetDir();
1100 namespace fs = std::filesystem;
1101 if (!fs::exists(dir) || !fs::is_directory(dir)) return;
1102 for (const auto& entry : fs::directory_iterator(dir)) {
1103 if (!entry.is_regular_file()) continue;
1104 const auto ext = entry.path().extension().string();
1105 if (ext != ".aeverb") continue;
1106 const std::string json = Aestra::File::readAllText(entry.path().string());
1107 if (json.empty()) continue;
1108 auto parsed = Aestra::JSON::parse(json);
1109 if (!parsed.isObject()) continue;
1110 PresetButton p;
1111 p.label = parsed.has("name") ? parsed["name"].asString() : entry.path().stem().string();
1112 p.tooltip = parsed.has("tooltip") ? parsed["tooltip"].asString() : "User preset";
1113 p.mode = parsed.has("mode") ? parsed["mode"].asInt() : 0;
1114 p.size = parsed.has("size") ? static_cast<float>(parsed["size"].asNumber()) : 0.5f;
1115 p.decay = parsed.has("decay") ? static_cast<float>(parsed["decay"].asNumber()) : 0.5f;
1116 p.damping = parsed.has("damping") ? static_cast<float>(parsed["damping"].asNumber()) : 0.5f;
1117 p.diffusion = parsed.has("diffusion") ? static_cast<float>(parsed["diffusion"].asNumber()) : 0.7f;
1118 p.modRate = parsed.has("modRate") ? static_cast<float>(parsed["modRate"].asNumber()) : 0.4f;
1119 // Legacy fallback stays 0.14 (the pre-retune default) so an old user
1120 // preset saved without a modDepth field loads exactly as it sounded when
1121 // created. New presets persist modDepth explicitly; only fresh instances
1122 // use the calmer 0.07 default.
1123 p.modDepth = parsed.has("modDepth") ? static_cast<float>(parsed["modDepth"].asNumber()) : 0.14f;
1124 p.width = parsed.has("width") ? static_cast<float>(parsed["width"].asNumber()) : 0.68f;
1125 p.mix = parsed.has("mix") ? static_cast<float>(parsed["mix"].asNumber()) : 0.36f;
1126 p.attack = parsed.has("attack") ? static_cast<float>(parsed["attack"].asNumber()) : 0.0f;
1127 p.shape = parsed.has("shape") ? static_cast<float>(parsed["shape"].asNumber()) : 0.5f;
1128 p.predelaySync = parsed.has("predelaySync") ? parsed["predelaySync"].asInt() : 0;
1129 p.modCharacter = parsed.has("modCharacter") ? parsed["modCharacter"].asInt() : 0;
1130 m_userPresets.push_back(p);

1223bool AestraVerbEditor::onMouseEvent(const NUIMouseEvent& event) {
1224 if (!isVisible()) return false;
1225 // Forward events to knob sliders (AestraPanelWindow doesn't dispatch to children)
1226 // Always forward drag/release to the actively-dragging slider so it can stop
1227 for (auto& k : m_knobs) {
1228 if (!k.slider) continue;
1229 const auto sb = k.slider->getBounds();
1230 const bool hasBounds = sb.width >= 16.0f && sb.height >= 16.0f;
1231 if (!hasBounds) continue;
1232 const bool overKnob = sb.contains({event.position.x, event.position.y});
1233 const bool dragging = k.slider->isDragging();
1234 if (overKnob || dragging) {
1235 if (k.slider->onMouseEvent(event)) return true;
1236 }
1237 }
1238 if (AestraPanelWindow::onMouseEvent(event)) return true;
1239 auto b = getBounds();
1240 const bool contains = b.contains(event.position);
1241 auto b2 = getBounds();
1242 const float contentX = editorContentX(b2);
1243 const float contentW = b2.width - (contentX - b2.x) - kPad;
1244 const float mainY = b2.y + 128.0f;
1245 const auto updateHoverState = [&]() {
1246 const int ch = contains ? hitTestCategory(event.position.x, event.position.y) : -1;
1247 const int ddh = contains ? hitTestDropdown(event.position.x, event.position.y) : -1;
1248 const int ph = contains ? hitTestPreset(event.position.x, event.position.y) : -1;
1249 const bool mixHovered = contains && hitTestMix(event.position.x, event.position.y);
1250 const bool bypassHovered = contains && hitTestBypass(event.position.x, event.position.y);
1251 const bool freezeHovered = contains && hitTestFreeze(event.position.x, event.position.y);
1252 int navDir = 0;
1253 const bool navHovered = contains && hitTestPresetNav(event.position.x, event.position.y, navDir);
1254 const bool navPrevH = navHovered && navDir == -1;
1255 const bool navNextH = navHovered && navDir == 1;
1256 const bool saveHovered = contains && event.position.x >= m_saveBounds.x && event.position.x <= m_saveBounds.right() &&
1257 event.position.y >= m_saveBounds.y && event.position.y <= m_saveBounds.bottom();
1258 bool isA = false;
1259 const bool abHovered = contains && hitTestAB(event.position.x, event.position.y, isA);
1260 const bool abHA = abHovered && isA;
1261 const bool abHB = abHovered && !isA;
1262 const bool mixLockH = contains && hitTestMixLock(event.position.x, event.position.y);
1263 if (ch == m_hoveredCategory && ddh == m_hoveredDropdownItem && ph == m_hoveredPreset && mixHovered == m_mixHovered &&
1264 bypassHovered == m_bypassHovered && freezeHovered == m_freezeHovered &&
1265 navPrevH == m_navPrevHovered && navNextH == m_navNextHovered &&
1266 saveHovered == m_saveHovered && abHA == m_abHoveredA && abHB == m_abHoveredB &&
1267 mixLockH == m_mixLockHovered) return;
1268 m_hoveredCategory = ch;
1269 m_hoveredDropdownItem = ddh;
1270 m_hoveredPreset = ph;
1271 m_mixHovered = mixHovered;
1272 m_bypassHovered = bypassHovered;
1273 m_freezeHovered = freezeHovered;
1274 m_navPrevHovered = navPrevH;
1275 m_navNextHovered = navNextH;
1276 m_saveHovered = saveHovered;
1277 m_abHoveredA = abHA;
1278 m_abHoveredB = abHB;
1279 m_mixLockHovered = mixLockH;
1280 for (size_t i = 0; i < m_categoryPills.size(); ++i) m_categoryPills[i].hovered = (static_cast<int>(i) == ch);
1281 for (size_t i = 0; i < m_dropdownItems.size(); ++i) m_dropdownItems[i].hovered = (static_cast<int>(i) == ddh);
1282 for (size_t i = 0; i < m_presets.size(); ++i) m_presets[i].hovered = (static_cast<int>(i) == ph);
1283 setDirty(true);
1284 };
1285 if (!isDraggingWindow() && !m_draggingMix) updateHoverState();
1286 if (event.released) {
1287 if (m_pressedCategory != -1 || m_pressedPreset != -1) {
1288 m_pressedCategory = -1;
1289 m_pressedPreset = -1;
1290 setDirty(true);
1291 }
1292 }
1293 if (!contains && !isDraggingWindow() && !m_draggingMix) {
1294 if (m_hoveredCategory != -1 || m_hoveredDropdownItem != -1 || m_hoveredPreset != -1 || m_mixHovered ||
1295 m_bypassHovered || m_freezeHovered || m_navPrevHovered || m_navNextHovered ||
1296 m_saveHovered || m_abHoveredA || m_abHoveredB || m_mixLockHovered) {
1297 m_hoveredCategory = -1;
1298 m_hoveredDropdownItem = -1;
1299 m_hoveredPreset = -1;
1300 m_mixHovered = false;
1301 m_bypassHovered = false;
1302 m_freezeHovered = false;
1303 m_navPrevHovered = false;
1304 m_navNextHovered = false;
1305 m_saveHovered = false;
1306 m_abHoveredA = false;
1307 m_abHoveredB = false;
1308 m_mixLockHovered = false;
1309 for (auto& pill : m_categoryPills) pill.hovered = false;
1310 for (auto& item : m_dropdownItems) item.hovered = false;
1311 for (auto& preset : m_presets) preset.hovered = false;
1312 setDirty(true);
1313 }
1314 if (m_dropdownOpen) {
1315 m_dropdownOpen = false;
1316 setDirty(true);
1317 }
1318 return false;
1319 }
1320 if (event.pressed && event.button == NUIMouseButton::Left) {
1321 // Category pill click
1322 const int catIdx = hitTestCategory(event.position.x, event.position.y);
1323 if (catIdx >= 0 && catIdx < kCategoryCount && m_categoryPills[catIdx].enabled) {
1324 m_pressedCategory = catIdx;
1325 m_focusedCategory = catIdx;
1326 m_focusedPreset = -1;
1327 m_mixFocused = false;
1328 m_selectedCategory = catIdx;
1329 m_dropdownItems.clear();
1330 int modes[3] = {};
1331 const int count = modesInCategory(catIdx, modes);
1332 for (int i = 0; i < count; ++i) {
1333 static const char* modeNames[] = {
1334 "Room", "Hall", "Plate", "Cathedral", "Chamber",
1335 "Bright Hall", "Ambience", "Scoring", "Smooth Plate"
1336 };
1337 m_dropdownItems.push_back({modeNames[modes[i]], modes[i], {}, false});
1338 }
1339 // Select first mode in category
1340 if (count > 0) {
1341 updateParameter(kMode, static_cast<float>(modes[0]) / static_cast<float>(kModeCount - 1));
1342 }
1343 m_dropdownOpen = true;
1344 layoutControls();
1345 setDirty(true);
1346 return true;
1347 }
1348 // Dropdown interaction
1349 const int ddIdx = hitTestDropdown(event.position.x, event.position.y);
1350 if (ddIdx == -2) {
1351 // Clicked dropdown button — toggle open (only if there are items)
1352 if (!m_dropdownItems.empty()) m_dropdownOpen = !m_dropdownOpen;
1353 setDirty(true);
1354 return true;
1355 }
1356 if (ddIdx >= 0 && m_dropdownOpen && ddIdx < static_cast<int>(m_dropdownItems.size())) {
1357 // Clicked dropdown item
1358 const auto& item = m_dropdownItems[ddIdx];
1359 updateParameter(kMode, static_cast<float>(item.mode) / static_cast<float>(kModeCount - 1));
1360 m_dropdownOpen = false;
1361 setDirty(true);
1362 return true;
1363 }
1364 if (m_dropdownOpen && !m_dropdownButtonBounds.contains({event.position.x, event.position.y}) &&
1365 !m_dropdownListBounds.contains({event.position.x, event.position.y})) {
1366 m_dropdownOpen = false;
1367 setDirty(true);
1368 }
1369 // Preset click
1370 const int presetIdx = hitTestPreset(event.position.x, event.position.y);
1371 if (presetIdx >= 0) {
1372 m_pressedPreset = presetIdx;
1373 m_focusedPreset = presetIdx;
1374 m_focusedCategory = -1;
1375 m_mixFocused = false;
1376 applyPreset(m_presets[static_cast<size_t>(presetIdx)]);
1377 return true;
1378 }
1379 if (hitTestMix(event.position.x, event.position.y)) {
1380 m_draggingMix = true;
1381 m_mixFocused = true;
1382 m_focusedCategory = -1;
1383 m_focusedPreset = -1;
1384 updateParameter(kMix, (event.position.x - m_mixTrack.x) / std::max(1.0f, m_mixTrack.width));
1385 return true;
1386 }
1387 if (hitTestBypass(event.position.x, event.position.y)) {
1388 updateParameter(kBypass, getParamValue(kBypass) > 0.5f ? 0.0f : 1.0f);
1389 return true;
1390 }
1391 if (hitTestFreeze(event.position.x, event.position.y)) {
1392 updateParameter(kFreeze, getParamValue(kFreeze) > 0.5f ? 0.0f : 1.0f);
1393 return true;
1394 }
1395 int navDir = 0;
1396 if (hitTestPresetNav(event.position.x, event.position.y, navDir)) {
1397 const float visibleH = 580.0f - (88.0f - 54.0f);
1398 const int visibleRows = std::max(1, static_cast<int>(visibleH / (50.0f + 6.0f)));
1399 const int maxScroll = std::max(0, static_cast<int>(m_presets.size()) - visibleRows);
1400 m_presetScroll = std::max(0, std::min(m_presetScroll + navDir, maxScroll));
1401 layoutControls();
1402 setDirty(true);
1403 return true;
1404 }
1405 bool isA = false;
1406 if (hitTestAB(event.position.x, event.position.y, isA)) {
1407 if (isA) {
1408 if (m_activeAB == 1) {
1409 for (uint32_t i = 0; i < kParamCount; ++i) updateParameter(i, m_abA.params[i]);
1410 m_activeAB = 0;
1411 } else {
1412 for (uint32_t i = 0; i < kParamCount; ++i) m_abA.params[i] = getParamValue(i);
1413 m_abA.valid = true;
1414 m_activeAB = 1;
1415 }
1416 } else {
1417 if (m_activeAB == 2) {
1418 for (uint32_t i = 0; i < kParamCount; ++i) updateParameter(i, m_abB.params[i]);
1419 m_activeAB = 0;
1420 } else {
1421 for (uint32_t i = 0; i < kParamCount; ++i) m_abB.params[i] = getParamValue(i);
1422 m_abB.valid = true;
1423 m_activeAB = 2;
1424 }
1425 }
1426 setDirty(true);
1427 return true;
1428 }
1429 if (hitTestMixLock(event.position.x, event.position.y)) {
1430 m_mixLocked = !m_mixLocked;
1431 setDirty(true);
1432 return true;
1433 }
1434 if (m_saveHovered) {
1435 saveUserPreset("User " + std::to_string(m_userPresets.size() + 1));
1436 return true;
1437 }
1438 }
1439 if (event.pressed && event.button == NUIMouseButton::Left && event.doubleClick) {
1440 for (auto& k : m_knobs) {
1441 if (k.slider && k.slider->getBounds().contains({event.position.x, event.position.y})) {
1442 updateParameter(k.paramId, k.defaultValue);
1443 if (k.slider) k.slider->setValue(k.defaultValue);
1444 setDirty(true);
1445 return true;
1446 }
1447 }
1448 }
1449 if (std::abs(event.wheelDelta) > 0.001f && contains) {
1450 const auto b3 = getBounds();
1451 const float stripLeft = b3.x + kPad - 5.0f;
1452 const float stripRight = b3.x + kPad + presetColumnWidth(b3.width) + 5.0f;
1453 const float stripTop = b3.y + 54.0f;
1454 const float stripBottom = stripTop + b3.height - 54.0f;
1455 if (event.position.x >= stripLeft && event.position.x <= stripRight &&
1456 event.position.y >= stripTop && event.position.y <= stripBottom) {
