33void drawVerbArc(NUIRenderer& renderer,
34 NUIPoint center,
35 float radius,
36 float startAngle,
37 float endAngle,
38 float thickness,
39 NUIColor color) {
40 if (endAngle < startAngle) std::swap(startAngle, endAngle);
41 if (endAngle - startAngle <= 0.001f) return;
42
43 std::array<NUIPoint, 49> points{};
44 const float pointsDivisor = points.size() > 1 ? static_cast<float>(points.size() - 1) : 1.0f;
45 for (size_t i = 0; i < points.size(); ++i) {
46 const float t = static_cast<float>(i) / pointsDivisor;
47 const float angle = startAngle + (endAngle - startAngle) * t;
48 points[i] = {center.x + std::cos(angle) * radius, center.y + std::sin(angle) * radius};
49 }
50 renderer.drawPolyline(points.data(), static_cast<int>(points.size()), thickness, color);
51}

53float protectedLeftDockEdge(const NUIComponent* root, const NUIComponent* self, const NUIRect& parentBounds) {
54 if (!root || parentBounds.width <= 0.0f || parentBounds.height <= 0.0f) return parentBounds.x;
55
56 float edge = parentBounds.x;
57 const auto scan = [&](const auto& recurse, const NUIComponent* node) -> void {
58 if (!node || node == self || !node->isVisible()) return;
59 const auto b = node->getBounds();
60 const bool leftDocked = b.x <= parentBounds.x + 8.0f;
61 const bool railSized = b.width >= 96.0f && b.width <= std::min(620.0f, parentBounds.width * 0.46f);
62 const bool verticallySignificant = b.height >= parentBounds.height * 0.40f && b.bottom() >= parentBounds.y + 120.0f;
63 if (leftDocked && railSized && verticallySignificant) {
64 edge = std::max(edge, b.right());
65 }
66 for (const auto& child : node->getChildren()) {
67 recurse(recurse, child.get());
68 }
69 };
70
71 scan(scan, root);
72 return edge;
73}

154std::string AestraVerbEditor::formatParameterValue(uint32_t paramId) const {
155 const float v = getParamValue(paramId);
156 std::ostringstream out;
157 switch (paramId) {
158 case kSize:
159 out << std::fixed << std::setprecision(2) << (0.1f + v * 1.9f) << "x";
160 return out.str();
161 case kDecay:
162 {
163 const float seconds = 0.3f + v * 9.7f;
164 if (seconds < 10.0f) {
165 out << std::fixed << std::setprecision(1) << seconds << "s";
166 } else {
167 out << static_cast<int>(std::round(seconds)) << "s";
168 }
169 return out.str();
170 }
171 case kDamping:
172 case kDiffusion:
173 case kWidth:
174 case kMix:
175 out << static_cast<int>(std::round(v * 100.0f)) << "%";
176 return out.str();
177 case kPredelay:
178 out << static_cast<int>(std::round(v * 500.0f)) << "ms";
179 return out.str();
180 case kModRate:
181 out << std::fixed << std::setprecision(2) << (v * 2.0f) << "x";
182 return out.str();
183 case kModDepth:
184 out << std::fixed << std::setprecision(1) << (v * 8.0f) << "smp";
185 return out.str();
186 default:
187 return m_instance ? m_instance->getParameterDisplay(paramId) : "0";
188 }

239 if (k.slider) k.slider->setBounds(knobRect);
240 continue;
241 }
242
243 float x = contentX;
244 float y = mainY;
245 float w = stackW;
246 switch (k.paramId) {
247 case kPredelay: y = mainY + 88.0f; break;
248 case kSize: y = mainY + 158.0f; break;
249 case kDamping: x = rightX; y = mainY + 0.0f; w = rightW; break;

252 case kModDepth: x = rightX; y = mainY + 174.0f; w = rightW; break;
253 case kWidth: x = rightX; y = mainY + 232.0f; w = rightW; break;

255 }
256 if (x == rightX) {
257 switch (k.paramId) {
258 case kDamping: y = mainY + 0.0f; break;
259 case kDiffusion: y = mainY + 52.0f; break;
260 case kModRate: y = mainY + 104.0f; break;
261 case kModDepth: y = mainY + 156.0f; break;
262 case kWidth: y = mainY + 208.0f; break;
263 default: break;
264 }
265 }
266 k.bounds = NUIRect(x, y, w, 50.0f);
267 const NUIRect knobRect = NUIRect(x + 6.0f, y + 2.0f, smallKnob, smallKnob);
268 if (k.slider) k.slider->setBounds(knobRect);

283void AestraVerbEditor::enforceBoundsInParent(bool recenterWhenPossible) {
284 auto* parent = getParent();
285 if (!parent) return;
286
287 const NUIRect parentBounds = parent->getBounds();
288 if (parentBounds.width <= 1.0f || parentBounds.height <= 1.0f) return;
289
290 constexpr float kSafeMargin = 14.0f;
291 constexpr float kChromeReserve = 56.0f;
292 constexpr float kMinSafeWidth = 640.0f;
293 constexpr float kMinSafeHeight = 500.0f;
294
295 const float topReserve = parentBounds.y <= 1.0f && parentBounds.height > kMinSafeHeight + kChromeReserve + kSafeMargin * 2.0f
296 ? kChromeReserve
297 : 0.0f;
298
299 NUIRect usable(parentBounds.x + kSafeMargin,
300 parentBounds.y + topReserve + kSafeMargin,
301 std::max(1.0f, parentBounds.width - kSafeMargin * 2.0f),
302 std::max(1.0f, parentBounds.height - topReserve - kSafeMargin * 2.0f));
303
304 const NUIComponent* root = parent;
305 while (root && root->getParent()) {
306 root = root->getParent();
307 }
308 const float dockEdge = protectedLeftDockEdge(root, this, parentBounds);
309 if (dockEdge > usable.x) {
310 const float shift = std::min(usable.width - 1.0f, dockEdge + kSafeMargin - usable.x);
311 usable.x += shift;
312 usable.width = std::max(1.0f, usable.width - shift);
313 }
314
315 auto current = getBounds();
316 float targetW = std::min(kWinW, usable.width);
317 float targetH = std::min(kWinH, usable.height);
318 if (usable.width >= kMinSafeWidth) targetW = std::max(kMinSafeWidth, targetW);
319 if (usable.height >= kMinSafeHeight) targetH = std::max(kMinSafeHeight, targetH);
320
321 const bool canFullyCenter = usable.width >= targetW && usable.height >= targetH;
322 float targetX = current.x;
323 float targetY = current.y;
324 if (recenterWhenPossible && canFullyCenter) {
325 targetX = usable.x + (usable.width - targetW) * 0.5f;
326 targetY = usable.y + (usable.height - targetH) * 0.5f;
327 } else {
328 targetX = std::clamp(current.x, usable.x, std::max(usable.x, usable.right() - targetW));
329 targetY = std::clamp(current.y, usable.y, std::max(usable.y, usable.bottom() - targetH));
330 }
331
332 if (std::abs(current.x - targetX) > 0.5f || std::abs(current.y - targetY) > 0.5f ||
333 std::abs(current.width - targetW) > 0.5f || std::abs(current.height - targetH) > 0.5f) {
334 setBounds(std::round(targetX), std::round(targetY), std::round(targetW), std::round(targetH));
335 }
336}

457 auto& theme = NUIThemeManager::getInstance();
458 const NUIRect knobRect = k.slider ? k.slider->getBounds() : NUIRect();

622void AestraVerbEditor::onUpdate(double deltaTime) {
623 AestraPanelWindow::onUpdate(deltaTime);
624 if (auto* parent = getParent()) {
625 const auto parentBounds = parent->getBounds();
626 const bool parentChanged = !m_haveParentSnapshot ||
627 std::abs(parentBounds.x - m_lastParentBounds.x) > 0.5f ||
628 std::abs(parentBounds.y - m_lastParentBounds.y) > 0.5f ||
629 std::abs(parentBounds.width - m_lastParentBounds.width) > 0.5f ||
630 std::abs(parentBounds.height - m_lastParentBounds.height) > 0.5f;
631 if (parentChanged && !isDraggingWindow()) {
632 m_lastParentBounds = parentBounds;
633 m_haveParentSnapshot = true;
634 enforceBoundsInParent(!m_userPositioned);
635 layoutControls();
636 } else if (!m_haveParentSnapshot) {
637 m_lastParentBounds = parentBounds;
638 m_haveParentSnapshot = true;
639 }
640 }
641 const float modRate = getParamValue(kModRate);
642 const float modDepth = getParamValue(kModDepth);
643 m_visualPhase += static_cast<float>(deltaTime) * (0.55f + modRate * 2.2f + modDepth * 1.6f);
644 if (m_visualPhase > kTwoPi * 8.0f) {
645 m_visualPhase = std::fmod(m_visualPhase, kTwoPi * 8.0f);
646 }
647 const float targetModePosition = static_cast<float>(std::clamp(static_cast<int>(std::round(getParamValue(kMode) * 2.0f)), 0, 2));
648 const float modeDiff = targetModePosition - m_modeIndicatorPosition;
649 if (std::abs(modeDiff) > 0.001f) {
650 const float step = modeDiff * 12.0f * static_cast<float>(deltaTime);
651 // Clamp step magnitude to avoid overshooting the target
652 m_modeIndicatorPosition += std::copysign(std::min(std::abs(step), std::abs(modeDiff)), modeDiff);
653 if (std::abs(targetModePosition - m_modeIndicatorPosition) < 0.001f) {
654 m_modeIndicatorPosition = targetModePosition;
655 }
656 setDirty(true);
657 }
658 m_visualDirtyAccum += static_cast<float>(deltaTime);
659 if (m_visualDirtyAccum >= 1.0f / 30.0f) {
660 m_visualDirtyAccum = 0.0f;
661 setDirty(true);
662 }
663}

746bool AestraVerbEditor::onMouseEvent(const NUIMouseEvent& event) {
747 if (!isVisible()) return false;
748
749 // Let base class handle title bar / close / drag first
750 if (AestraPanelWindow::onMouseEvent(event)) {
751 return true;
752 }
753
754 auto b = getBounds();
755 const bool contains = b.contains(event.position);
756 const auto updateHoverState = [&]() {
757 const int mh = contains ? hitTestMode(event.position.x, event.position.y) : -1;
758 const int ph = contains ? hitTestPreset(event.position.x, event.position.y) : -1;
759 const bool mixHovered = contains && hitTestMix(event.position.x, event.position.y);
760 if (mh == m_hoveredMode && ph == m_hoveredPreset && mixHovered == m_mixHovered) {
761 return;
762 }
763 m_hoveredMode = mh;
764 m_hoveredPreset = ph;
765 m_mixHovered = mixHovered;
766 for (size_t i = 0; i < m_modes.size(); ++i) m_modes[i].hovered = (static_cast<int>(i) == mh);
767 for (size_t i = 0; i < m_presets.size(); ++i) m_presets[i].hovered = (static_cast<int>(i) == ph);
768 setDirty(true);
769 };
770 if (!isDraggingWindow() && !m_draggingMix) {
771 updateHoverState();
772 }
773 if (event.released) {
774 if (m_pressedMode != -1 || m_pressedPreset != -1) {
775 m_pressedMode = -1;
776 m_pressedPreset = -1;
777 setDirty(true);
778 }
779 }
780 if (!contains && !isDraggingWindow() && !m_draggingMix) {
781 if (m_hoveredMode != -1 || m_hoveredPreset != -1 || m_mixHovered) {
782 m_hoveredMode = -1;
783 m_hoveredPreset = -1;
784 m_mixHovered = false;
785 for (auto& mode : m_modes) mode.hovered = false;
786 for (auto& preset : m_presets) preset.hovered = false;
787 setDirty(true);
788 }
789 return false;
790 }
791
792 if (event.pressed && event.button == NUIMouseButton::Left) {
793 const int modeIdx = hitTestMode(event.position.x, event.position.y);
794 if (modeIdx >= 0) {
795 const auto mode = m_modes[static_cast<size_t>(modeIdx)].mode;
796 m_pressedMode = mode;
797 m_focusedMode = mode;
798 m_focusedPreset = -1;
799 m_mixFocused = false;
800 updateParameter(kMode, static_cast<float>(mode) / 2.0f);
801 return true;
802 }
803 const int presetIdx = hitTestPreset(event.position.x, event.position.y);
804 if (presetIdx >= 0) {
805 m_pressedPreset = presetIdx;
806 m_focusedPreset = presetIdx;
807 m_focusedMode = -1;
808 m_mixFocused = false;
809 applyPreset(m_presets[static_cast<size_t>(presetIdx)]);
810 return true;
811 }
812 if (hitTestMix(event.position.x, event.position.y)) {
813 m_draggingMix = true;
814 m_mixFocused = true;
815 m_focusedMode = -1;
816 m_focusedPreset = -1;
817 updateParameter(kMix, (event.position.x - m_mixTrack.x) / std::max(1.0f, m_mixTrack.width));
818 return true;
819 }
820 }
821
822 if (m_draggingMix) {
823 if (event.released && event.button == NUIMouseButton::Left) {
824 m_draggingMix = false;
825 setDirty(true);
826 return true;
827 }
828 updateParameter(kMix, (event.position.x - m_mixTrack.x) / std::max(1.0f, m_mixTrack.width));
829 return true;
830 }
831
832 if (!event.pressed && !event.released) {
833 updateHoverState();
834 }
835 return contains;
836}
