2// AestraFilter V1 — creative resonant filter with envelope-follower modulation.
3//
4// Topology-preserving-transform (ZDF) state-variable filter (Cytomic/Simper
5// form) so cutoff can be modulated per sample without zipper noise or
6// instability. An input peak follower drives the cutoff bipolarly (up to
7// ±4 octaves), which is what turns a static filter into an envelope shaper:
8// auto-wah upward sweeps, reverse "duck" filtering, dynamic brightness.
9//
10// Control-rate architecture: parameters are read and smoothed once per
11// 8-sample control block, transcendental coefficient recomputes (pow/exp/tan)
12// are gated on smoothed-target movement, and the TPT coefficient set is
13// linearly interpolated across the block. The per-sample path is the envelope
14// follower, the drive tanh and the SVF arithmetic.
15//
16// Signal path (per channel, per sample):
17// in -> drive (transparent-at-zero tanh blend) -> ZDF SVF (LP/BP/HP)
18// -> mix with dry (zero latency — no compensation needed)
19//
20// The envelope is derived from the pre-drive stereo peak so drive changes
21// never re-tune the sweep.

90 void process(const float* const* inputs, float** outputs, uint32_t numInputChannels, uint32_t numOutputChannels,
91 uint32_t numFrames, const MidiBuffer* midiInput = nullptr, MidiBuffer* midiOutput = nullptr) override {
92 (void)midiInput;
93 (void)midiOutput;
94
95 if (!m_active.load(std::memory_order_relaxed) || m_params[kBypass].load(std::memory_order_relaxed) > 0.5f) {
96 copyOrClear(inputs, outputs, numInputChannels, numOutputChannels, numFrames);
97 m_wasBypassed = true;
98 return;
99 }
100 if (m_wasBypassed) {
101 // While bypassed the integrators and envelope stopped evolving;
102 // re-enabling must not restore stale resonance/envelope state.
103 m_ic1[0] = m_ic1[1] = 0.0f;
104 m_ic2[0] = m_ic2[1] = 0.0f;
105 m_envelope = 0.0f;
106 snapSmoothedParams();
107 m_coeffsDirty = true;
108 snapTypeGains();
109 m_wasBypassed = false;
110 }
111
112 const uint32_t channels = std::min<uint32_t>(2, numOutputChannels);
113 const bool stereo = channels >= 2;
114 const float sr = static_cast<float>(m_sampleRate);
115 const float maxCutoff = 0.45f * sr;
116
117 for (uint32_t blockStart = 0; blockStart < numFrames; blockStart += kCtrlInterval) {
118 const uint32_t blockEnd = std::min(blockStart + kCtrlInterval, numFrames);
119 const uint32_t blockLen = blockEnd - blockStart;
120
121 // ── Control block: smooth params, gated coefficient recomputes ──
122 m_cutoffSmoothed += (getParameter(kCutoff) - m_cutoffSmoothed) * m_smoothCoeff;
123 m_resoSmoothed += (getParameter(kReso) - m_resoSmoothed) * m_smoothCoeff;
124 m_driveSmoothed += (getParameter(kDrive) - m_driveSmoothed) * m_smoothCoeff;
125 m_envAmtSmoothed += (getParameter(kEnvAmount) - m_envAmtSmoothed) * m_smoothCoeff;
126 m_attackSmoothed += (getParameter(kEnvAttack) - m_attackSmoothed) * m_smoothCoeff;
127 m_releaseSmoothed += (getParameter(kEnvRelease) - m_releaseSmoothed) * m_smoothCoeff;
128 m_mixSmoothed += (getParameter(kMix) - m_mixSmoothed) * m_smoothCoeff;
129
130 if (m_coeffsDirty || std::abs(m_resoSmoothed - m_resoCached) > 1.0e-5f) {
131 m_resoCached = m_resoSmoothed;
132 m_k = 1.0f / qFromNorm(m_resoCached);
133 }
134 if (m_coeffsDirty || std::abs(m_driveSmoothed - m_driveCached) > 1.0e-5f) {
135 m_driveCached = m_driveSmoothed;
136 m_driveGain = dbToLinear(driveDbFromNorm(m_driveCached));
137 m_driveMakeup = 1.0f / std::sqrt(m_driveGain);
138 }
139 if (m_coeffsDirty || std::abs(m_attackSmoothed - m_attackCached) > 1.0e-4f) {
140 m_attackCached = m_attackSmoothed;
141 m_attackCoeff = 1.0f - std::exp(-1.0f / (attackMsFromNorm(m_attackCached) * 0.001f * sr));
142 }
143 if (m_coeffsDirty || std::abs(m_releaseSmoothed - m_releaseCached) > 1.0e-4f) {
144 m_releaseCached = m_releaseSmoothed;
145 m_releaseCoeff = 1.0f - std::exp(-1.0f / (releaseMsFromNorm(m_releaseCached) * 0.001f * sr));
146 }
147 if (m_coeffsDirty || std::abs(m_cutoffSmoothed - m_cutoffCached) > 1.0e-5f) {
148 m_cutoffCached = m_cutoffSmoothed;
149 m_baseCutoffHz = cutoffHzFromNorm(m_cutoffCached);
150 }
151
152 // Envelope-modulated TPT coefficient target for this block; the
153 // set is linearly interpolated across the block so per-sample cost
154 // is three adds instead of exp2 + tan.
155 const float envAmount = 2.0f * m_envAmtSmoothed - 1.0f;
156 const float modOctaves = envAmount * kEnvModOctaves * std::min(1.0f, m_envelope);
157 const float fc = std::clamp(m_baseCutoffHz * std::exp2(modOctaves), 20.0f, maxCutoff);
158 const float g = std::tan(3.14159265358979323846f * fc / sr);
159 const float a1t = 1.0f / (1.0f + g * (g + m_k));
160 const float a2t = g * a1t;
161 const float a3t = g * a2t;
162 float da1, da2, da3;
163 if (m_coeffsDirty) {
164 m_a1 = a1t;
165 m_a2 = a2t;
166 m_a3 = a3t;
167 da1 = da2 = da3 = 0.0f;
168 m_coeffsDirty = false;
169 } else {
170 const float inv = 1.0f / static_cast<float>(blockLen);
171 da1 = (a1t - m_a1) * inv;
172 da2 = (a2t - m_a2) * inv;
173 da3 = (a3t - m_a3) * inv;
174 }
175
176 // Type is stepped; each of LP/BP/HP carries a gain that continuously
177 // slews toward a one-hot target. A plain two-state crossfade drops
178 // the in-progress blend when a switch interrupts a fade (rapid
179 // LP->BP->HP), which can click; per-type gains morph through any
180 // interruption because the current audible blend is the state.
181 const uint32_t selType =
182 static_cast<uint32_t>(typeFromNorm(m_params[kType].load(std::memory_order_relaxed)));
183
184 const float wet = m_mixSmoothed;
185 const float dry = 1.0f - wet;
186
187 for (uint32_t i = blockStart; i < blockEnd; ++i) {
188 m_a1 += da1;
189 m_a2 += da2;
190 m_a3 += da3;
191
192 const float inL = sanitizeSample(readInput(inputs, numInputChannels, 0, i));
193 const float inR = stereo ? sanitizeSample(readInput(inputs, numInputChannels, 1, i)) : inL;
194
195 // ── Envelope follower (pre-drive stereo peak) ──
196 const float peak = std::max(std::abs(inL), std::abs(inR));
197 const float envCoeff = (peak > m_envelope) ? m_attackCoeff : m_releaseCoeff;
198 m_envelope += (peak - m_envelope) * envCoeff;
199
200 // Continuous per-type gain slew toward the one-hot selection.
201 for (uint32_t t = 0; t < kTypeCount; ++t) {
202 const float target = (t == selType) ? 1.0f : 0.0f;
203 m_typeGain[t] += (target - m_typeGain[t]) * m_typeCoeff;
204 }
205
206 for (uint32_t ch = 0; ch < channels; ++ch) {
207 const float x = (ch == 0) ? inL : inR;
208 // Drive: transparent at 0, tanh soft clip as it rises.
209 // 1/sqrt(G) makeup keeps perceived loudness roughly level.
210 const float driven = x + m_driveCached * (std::tanh(m_driveGain * x) * m_driveMakeup - x);
211
212 float& ic1 = m_ic1[ch];
213 float& ic2 = m_ic2[ch];
214 const float v3 = driven - ic2;
215 const float v1 = m_a1 * ic1 + m_a2 * v3;
216 const float v2 = ic2 + m_a2 * ic1 + m_a3 * v3;
217 ic1 = 2.0f * v1 - ic1;
218 ic2 = 2.0f * v2 - ic2;
219
220 const float lp = v2;
221 const float bp = v1;
222 const float hp = driven - m_k * v1 - v2;
223 const float filtered =
224 m_typeGain[kTypeLowPass] * lp + m_typeGain[kTypeBandPass] * bp + m_typeGain[kTypeHighPass] * hp;
225
226 if (outputs[ch])
227 outputs[ch][i] = x * dry + filtered * wet;
228 }
229
230 if (!stereo && numOutputChannels > 1 && outputs[1] && outputs[0]) {
231 outputs[1][i] = outputs[0][i];
232 }
233 for (uint32_t ch = 2; ch < numOutputChannels; ++ch) {
234 if (outputs[ch])
235 outputs[ch][i] = 0.0f;
236 }
237 }
238 }
239
240 m_envLevel.store(std::min(1.0f, m_envelope), std::memory_order_relaxed);
241 }

273 std::string getParameterDisplay(uint32_t id) const override {
274 if (id >= kParamCount)
275 return "";
276 const float v = getParameter(id);
277 char buf[32];
278 switch (id) {
279 case kType:
280 switch (typeFromNorm(v)) {
281 case kTypeLowPass:
282 return "Low Pass";
283 case kTypeBandPass:
284 return "Band Pass";
285 default:
286 return "High Pass";
287 }
288 case kCutoff: {
289 const float hz = cutoffHzFromNorm(v);
290 if (hz >= 1000.0f) {
291 std::snprintf(buf, sizeof(buf), "%.2f kHz", hz / 1000.0f);
292 } else {
293 std::snprintf(buf, sizeof(buf), "%.0f Hz", hz);
294 }
295 return buf;
296 }
297 case kReso:
298 std::snprintf(buf, sizeof(buf), "Q %.2f", qFromNorm(v));
299 return buf;
300 case kDrive:
301 std::snprintf(buf, sizeof(buf), "%.1f dB", driveDbFromNorm(v));
302 return buf;
303 case kEnvAmount:
304 std::snprintf(buf, sizeof(buf), "%+d%%", static_cast<int>(std::round((2.0f * v - 1.0f) * 100.0f)));
305 return buf;
306 case kEnvAttack:
307 std::snprintf(buf, sizeof(buf), "%.1f ms", attackMsFromNorm(v));
308 return buf;
309 case kEnvRelease:
310 std::snprintf(buf, sizeof(buf), "%.0f ms", releaseMsFromNorm(v));
311 return buf;
312 case kMix:
313 return std::to_string(static_cast<int>(std::round(v * 100.0f))) + "%";
314 case kBypass:
315 return v > 0.5f ? "ON" : "OFF";
316 default:
317 return "";
318 }

321 std::vector<uint8_t> saveState() const override {
322 struct Blob {
323 uint32_t magic = kStateMagic;
324 uint32_t version = 1;
325 float params[kParamCount] = {};
326 } blob;
327 for (uint32_t i = 0; i < kParamCount; ++i) {
328 blob.params[i] = getParameter(i);
329 }
330 const auto* data = reinterpret_cast<const uint8_t*>(&blob);
331 return {data, data + sizeof(blob)};
332 }

334 bool loadState(const std::vector<uint8_t>& state) override {
335 if (state.size() < sizeof(uint32_t) * 2)
336 return false;
337 uint32_t magic = 0;
338 std::memcpy(&magic, state.data(), sizeof(magic));
339 if (magic != kStateMagic)
340 return false;
341 struct Blob {
342 uint32_t magic;
343 uint32_t version;
344 float params[kParamCount];
345 };
346 if (state.size() < sizeof(Blob))
347 return false;
348 Blob blob{};
349 std::memcpy(&blob, state.data(), sizeof(blob));
350 if (blob.version < 1 || blob.version > 1)
351 return false;
352 // Validate the entire decoded set before mutating anything. setParameter
353 // silently drops non-finite values, so applying in place would leave a
354 // half-updated state while still reporting success — fail atomically.
355 for (uint32_t i = 0; i < kParamCount; ++i) {
356 if (!std::isfinite(blob.params[i]) || blob.params[i] < 0.0f || blob.params[i] > 1.0f)
357 return false;
358 }
359 for (uint32_t i = 0; i < kParamCount; ++i) {
360 setParameter(i, blob.params[i]);
361 }
362 return true;
363 }

375 // Resonant ring-out is sample-rate-relative: a 2-pole resonator's -60 dB
376 // decay is ~ln(1000)*Q/(pi*f0), so the worst case here (20 Hz cutoff,
377 // Q 10) rings ~1.1 s. A fixed 4096 samples truncated that ring in
378 // offline bounce and drifted with sample rate — report ~1.2 s scaled to
379 // the current rate.
380 return static_cast<uint32_t>(m_sampleRate * 1.2);

426 // Snap the per-type gains to the current one-hot selection so activation
427 // and bypass->active transitions do not morph the filter type.
428 void snapTypeGains() {
429 const uint32_t sel = static_cast<uint32_t>(typeFromNorm(getParameter(kType)));
430 for (uint32_t t = 0; t < kTypeCount; ++t)
431 m_typeGain[t] = (t == sel) ? 1.0f : 0.0f;
432 }

434 void resetRuntimeState() {
435 m_ic1[0] = m_ic1[1] = 0.0f;
436 m_ic2[0] = m_ic2[1] = 0.0f;
437 m_envelope = 0.0f;
438 m_wasBypassed = false;
439 m_coeffsDirty = true;
440 const float sr = static_cast<float>(m_sampleRate);
441 // Smoothing runs once per control block, so the coefficient is scaled
442 // to the block interval to keep the 2 ms time constant.
443 m_smoothCoeff = 1.0f - std::exp(-static_cast<float>(kCtrlInterval) / (0.002f * sr));
444 m_typeCoeff = 1.0f - std::exp(-1.0f / (0.003f * sr)); // ~3 ms type morph (per sample)
445 snapTypeGains();
446 m_envLevel.store(0.0f, std::memory_order_relaxed);
447 }

460 // ── Utility ──
461 static void copyOrClear(const float* const* inputs, float** outputs, uint32_t numInputChannels,
462 uint32_t numOutputChannels, uint32_t numFrames) {
463 for (uint32_t ch = 0; ch < numOutputChannels; ++ch) {
464 if (outputs[ch] && ch < numInputChannels && inputs[ch]) {
465 // Skip the copy when the host renders in place (out == in):
466 // memcpy on aliased buffers is undefined and the data is already
467 // where it needs to be.
468 if (outputs[ch] != inputs[ch])
469 std::memcpy(outputs[ch], inputs[ch], numFrames * sizeof(float));
470 } else if (outputs[ch]) {
471 std::memset(outputs[ch], 0, numFrames * sizeof(float));
472 }
473 }
474 }

477 if (channel >= channels || !inputs[channel]) {
