41NUIColor parameterAccent(uint32_t parameterId) {
42 switch (parameterId) {
43 case Drift::kGrain:
44 case Drift::kTexture:
45 case Drift::kSpread:
46 return grainAccent();
47 case Drift::kMotion:
48 case Drift::kMotionRate:
49 return motionAccent();
50 default:
51 return pitchAccent();

55void drawArc(NUIRenderer& renderer, NUIPoint center, float radius, float startAngle, float endAngle, float thickness,
56 NUIColor color) {
57 if (endAngle <= startAngle + 0.001f)
58 return;
59 std::array<NUIPoint, 49> points{};
60 for (size_t i = 0; i < points.size(); ++i) {
61 const float t = static_cast<float>(i) / static_cast<float>(points.size() - 1);
62 const float angle = startAngle + (endAngle - startAngle) * t;
63 points[i] = {center.x + std::cos(angle) * radius, center.y + std::sin(angle) * radius};
64 }
65 renderer.drawPolyline(points.data(), static_cast<int>(points.size()), thickness, color);
66}

83void AestraDriftEditor::buildControls() {
84 m_knobs.clear();
85 if (!m_instance)
86 return;
87 struct Meta {
88 uint32_t id;
89 const char* label;
90 const char* subtitle;
91 };
92 constexpr Meta meta[] = {
93 {Drift::kGrain, "GRAIN", "Window length"},
94 {Drift::kTexture, "TEXTURE", "Overlap color"},
95 {Drift::kFine, "FINE", "Cents"},
96 {Drift::kSpread, "SPREAD", "Stereo phase"},
97 {Drift::kMotion, "MOTION", "Optional drift"},
98 {Drift::kMotionRate, "RATE", "Motion speed"},
99 {Drift::kOutput, "OUTPUT", "Final level"},
100 };
101 const auto parameters = m_instance->getParameters();
102 for (const auto& item : meta) {
103 if (item.id >= parameters.size())
104 continue;
105 m_knobs.push_back({item.id, item.label, item.subtitle, parameters[item.id].defaultValue, {}, {}});
106 }
107 layoutControls();
108}

110void AestraDriftEditor::layoutControls() {
111 const auto bounds = getBounds();
112 const float top = bounds.y + AestraPanelWindow::TITLE_BAR_H + kPad;
113 const float bottomBarHeight = 52.0f;
114 const float mainHeight = std::max(0.0f, bounds.bottom() - kPad - bottomBarHeight - top);
115 const float contentWidth = std::max(0.0f, bounds.width - kPad * 2.0f);
116 const float gap = 12.0f;
117 const float pitchWidth = std::clamp(contentWidth * 0.43f, 286.0f, 310.0f);
118 m_pitchPanelBounds = {bounds.x + kPad, top, pitchWidth, mainHeight};
119 m_texturePanelBounds = {m_pitchPanelBounds.right() + gap, top, contentWidth - pitchWidth - gap, mainHeight};
120
121 const float wheelSize = std::min(188.0f, m_pitchPanelBounds.width - 56.0f);
122 m_pitchWheelBounds = {m_pitchPanelBounds.center().x - wheelSize * 0.5f, m_pitchPanelBounds.y + 48.0f, wheelSize,
123 wheelSize};
124
125 const float intervalGap = 5.0f;
126 const float intervalPad = 12.0f;
127 const float intervalWidth =
128 (m_pitchPanelBounds.width - intervalPad * 2.0f - intervalGap * 6.0f) / static_cast<float>(kIntervals.size());
129 for (size_t i = 0; i < m_intervalBounds.size(); ++i) {
130 m_intervalBounds[i] = {m_pitchPanelBounds.x + intervalPad +
131 static_cast<float>(i) * (intervalWidth + intervalGap),
132 m_pitchPanelBounds.bottom() - 42.0f, intervalWidth, 28.0f};
133 }
134
135 const float controlPad = 10.0f;
136 const float controlGap = 8.0f;
137 const float controlTop = m_texturePanelBounds.y + 40.0f;
138 const float cellWidth = (m_texturePanelBounds.width - controlPad * 2.0f - controlGap * 3.0f) / 4.0f;
139 const float cellHeight = (m_texturePanelBounds.bottom() - controlTop - controlPad - controlGap) / 2.0f;
140 for (size_t i = 0; i < m_knobs.size(); ++i) {
141 const size_t row = i < 4 ? 0 : 1;
142 const size_t column = i < 4 ? i : i - 4;
143 const float rowOffset = row == 1 ? (cellWidth + controlGap) * 0.5f : 0.0f;
144 auto& control = m_knobs[i];
145 control.bounds = {m_texturePanelBounds.x + controlPad + rowOffset +
146 static_cast<float>(column) * (cellWidth + controlGap),
147 controlTop + static_cast<float>(row) * (cellHeight + controlGap), cellWidth, cellHeight};
148 const float knobSize = std::min(68.0f, cellWidth - 22.0f);
149 control.knobBounds = {control.bounds.center().x - knobSize * 0.5f, control.bounds.y + 12.0f, knobSize,
150 knobSize};
151 }
152
153 const float bottomY = m_pitchPanelBounds.bottom() + 10.0f;
154 m_mixBounds = {bounds.x + kPad, bottomY, contentWidth - 112.0f, 42.0f};
155 m_bypassBounds = {m_mixBounds.right() + 10.0f, bottomY, 102.0f, 42.0f};
156}

172void AestraDriftEditor::drawPitchPanel(NUIRenderer& renderer) {
173 auto& theme = NUIThemeManager::getInstance();
174 renderer.fillRoundedRect(m_pitchPanelBounds, 16.0f, panelSurface());
175 renderer.strokeRoundedRect(m_pitchPanelBounds, 16.0f, 1.0f, pitchAccent().withAlpha(0.34f));
176 renderer.fillRoundedRect({m_pitchPanelBounds.x + 12.0f, m_pitchPanelBounds.y + 13.0f, 4.0f, 16.0f}, 2.0f,
177 pitchAccent());
178 renderer.drawText("PITCH", {m_pitchPanelBounds.x + 24.0f, m_pitchPanelBounds.y + 14.0f}, 10.0f,
179 theme.getColor("textPrimary"));
180 renderer.drawText("COARSE SHIFT", {m_pitchPanelBounds.right() - 92.0f, m_pitchPanelBounds.y + 15.0f}, 8.0f,
181 theme.getColor("textSecondary").withAlpha(0.62f));
182 drawPitchWheel(renderer);
183 renderer.drawText("INTERVALS", {m_pitchPanelBounds.x + 14.0f, m_pitchPanelBounds.bottom() - 58.0f}, 7.5f,
184 theme.getColor("textSecondary").withAlpha(0.58f));
185
186 const float currentSemitones = -12.0f + m_instance->getParameter(Drift::kPitch) * 24.0f;
187 for (size_t i = 0; i < m_intervalBounds.size(); ++i) {
188 const bool active = std::fabs(currentSemitones - static_cast<float>(kIntervals[i])) < 0.25f;
189 const bool hovered = static_cast<int>(i) == m_hoveredInterval;
190 renderer.fillRoundedRect(m_intervalBounds[i], 7.0f,
191 active ? pitchAccent().withAlpha(0.28f)
192 : hovered ? pitchAccent().withAlpha(0.13f)
193 : insetSurface());
194 renderer.strokeRoundedRect(m_intervalBounds[i], 7.0f, 1.0f,
195 active || hovered ? pitchAccent().withAlpha(0.56f)
196 : editorInk(0.05f));
197 const std::string label =
198 kIntervals[i] > 0 ? "+" + std::to_string(kIntervals[i]) : std::to_string(kIntervals[i]);
199 renderer.drawTextCentered(label, m_intervalBounds[i], 9.0f,
200 active ? pitchAccent() : theme.getColor("textSecondary"));
201 }

204void AestraDriftEditor::drawPitchWheel(NUIRenderer& renderer) {
205 auto& theme = NUIThemeManager::getInstance();
206 const float value = m_instance->getParameter(Drift::kPitch);
207 const float angle = kWheelStart + value * kWheelSweep;
208 const NUIPoint center = m_pitchWheelBounds.center();
209 const float radius = m_pitchWheelBounds.width * 0.5f;
210
211 renderer.fillCircle(center, radius, insetSurface());
212 renderer.strokeCircle(center, radius, 1.0f,
213 m_pitchHovered || m_draggingPitch ? pitchAccent().withAlpha(0.54f)
214 : editorInk(0.08f));
215 drawArc(renderer, center, radius - 11.0f, kWheelStart, kWheelStart + kWheelSweep, 6.0f,
216 editorInk(0.09f));
217 drawArc(renderer, center, radius - 11.0f, kWheelStart, angle, 6.0f, pitchAccent().withAlpha(0.94f));
218 for (int semitones : kIntervals) {
219 const float t = (static_cast<float>(semitones) + 12.0f) / 24.0f;
220 const float tickAngle = kWheelStart + t * kWheelSweep;
221 const NUIPoint outer{center.x + std::cos(tickAngle) * (radius - 20.0f),
222 center.y + std::sin(tickAngle) * (radius - 20.0f)};
223 const NUIPoint inner{center.x + std::cos(tickAngle) * (radius - 29.0f),
224 center.y + std::sin(tickAngle) * (radius - 29.0f)};
225 renderer.drawLine(inner, outer, semitones == 0 ? 2.0f : 1.0f,
226 semitones == 0 ? pitchAccent().withAlpha(0.70f)
227 : theme.getColor("textSecondary").withAlpha(0.36f));
228 }
229 const NUIPoint needle{center.x + std::cos(angle) * (radius - 35.0f), center.y + std::sin(angle) * (radius - 35.0f)};
230 renderer.drawLine(center, needle, 2.3f, pitchAccent().withAlpha(0.88f));
231 renderer.fillCircle(center, 28.0f, editorNeutral(NUIColor(0.048f, 0.045f, 0.060f, 1.0f)));
232 renderer.strokeCircle(center, 28.0f, 1.0f, pitchAccent().withAlpha(0.30f));
233 renderer.drawTextCentered(pitchValueString(), {center.x - 34.0f, center.y - 11.0f, 68.0f, 22.0f}, 18.0f,
234 pitchAccent());
235 renderer.drawTextCentered("SEMITONES", {center.x - 42.0f, center.y + 17.0f, 84.0f, 11.0f}, 7.0f,
236 theme.getColor("textSecondary").withAlpha(0.54f));
237}

258void AestraDriftEditor::drawKnob(NUIRenderer& renderer, const KnobControl& control, int index) {
259 auto& theme = NUIThemeManager::getInstance();
260 const bool hovered = m_hoveredKnob == index;
261 const bool dragging = m_draggingKnob == index;
262 const bool rateInactive =
263 control.parameterId == Drift::kMotionRate && m_instance->getParameter(Drift::kMotion) < 0.001f;
264 const NUIColor color = parameterAccent(control.parameterId);
265 renderer.fillRoundedRect(control.bounds, 10.0f,
266 hovered || dragging ? editorNeutral(NUIColor(0.070f, 0.064f, 0.086f, 0.98f)) : insetSurface());
267 renderer.strokeRoundedRect(control.bounds, 10.0f, 1.0f,
268 hovered || dragging ? color.withAlpha(0.48f) : editorInk(0.045f));
269 const float value = m_instance->getParameter(control.parameterId);
270 const NUIPoint center = control.knobBounds.center();
271 const float radius = control.knobBounds.width * 0.5f;
272 const float angle = kWheelStart + value * kWheelSweep;
273 renderer.fillCircle(center, radius, editorNeutral(NUIColor(0.030f, 0.029f, 0.039f, 1.0f)));
274 renderer.strokeCircle(center, radius, 1.0f, editorInk(0.07f));
275 drawArc(renderer, center, radius - 6.0f, kWheelStart, kWheelStart + kWheelSweep, 3.5f,
276 editorInk(0.08f));
277 drawArc(renderer, center, radius - 6.0f, kWheelStart, angle, 3.5f, color.withAlpha(rateInactive ? 0.30f : 0.92f));
278 const NUIPoint needle{center.x + std::cos(angle) * (radius - 14.0f), center.y + std::sin(angle) * (radius - 14.0f)};
279 renderer.drawLine(center, needle, 1.8f, color.withAlpha(rateInactive ? 0.28f : 0.86f));
280 renderer.fillCircle(center, 4.5f, editorNeutral(NUIColor(0.070f, 0.066f, 0.086f, 1.0f)));
281 renderer.fillCircle(center, 1.8f, color.withAlpha(rateInactive ? 0.30f : 0.90f));
282
283 const float labelY = control.knobBounds.bottom() + 5.0f;
284 renderer.drawTextCentered(control.label, {control.bounds.x, labelY, control.bounds.width, 12.0f}, 8.5f,
285 theme.getColor("textPrimary").withAlpha(rateInactive ? 0.42f : 0.90f));
286 renderer.drawTextCentered(m_instance->getParameterDisplay(control.parameterId),
287 {control.bounds.x, labelY + 14.0f, control.bounds.width, 11.0f}, 8.0f,
288 color.withAlpha(rateInactive ? 0.32f : 0.92f));
289 renderer.drawTextCentered(control.subtitle, {control.bounds.x, labelY + 28.0f, control.bounds.width, 10.0f}, 7.0f,
290 theme.getColor("textSecondary").withAlpha(rateInactive ? 0.28f : 0.50f));

333bool AestraDriftEditor::onMouseEvent(const NUIMouseEvent& event) {
334 if (!isVisible())
335 return false;
336 if (AestraPanelWindow::onMouseEvent(event))
337 return true;
338 if (!m_instance)
339 return false;
340
341 const NUIRect mixTrack(m_mixBounds.x + 70.0f, m_mixBounds.y + 7.0f, m_mixBounds.width - 132.0f,
342 m_mixBounds.height - 14.0f);
343 if (event.released) {
344 const bool wasDragging = m_draggingPitch || m_draggingMix || m_draggingKnob >= 0;
345 m_draggingPitch = false;
346 m_draggingMix = false;
347 m_draggingKnob = -1;
348 if (wasDragging)
349 return true;
350 }
351
352 if (event.type == NUIMouseEventType::DoubleClick || event.doubleClick) {
353 if (m_pitchWheelBounds.contains(event.position)) {
354 setParameter(Drift::kPitch, 0.5f);
355 return true;
356 }
357 const int knob = hitTestKnob(event.position);
358 if (knob >= 0) {
359 setParameter(m_knobs[static_cast<size_t>(knob)].parameterId,
360 m_knobs[static_cast<size_t>(knob)].defaultValue);
361 return true;
362 }
363 }
364
365 if (event.pressed && event.button == NUIMouseButton::Left) {
366 if (m_bypassBounds.contains(event.position)) {
367 setParameter(Drift::kBypass, m_instance->getParameter(Drift::kBypass) > 0.5f ? 0.0f : 1.0f);
368 return true;
369 }
370 for (size_t i = 0; i < m_intervalBounds.size(); ++i) {
371 if (m_intervalBounds[i].contains(event.position)) {
372 setParameter(Drift::kPitch, (static_cast<float>(kIntervals[i]) + 12.0f) / 24.0f);
373 return true;
374 }
375 }
376 if (m_pitchWheelBounds.contains(event.position)) {
377 m_draggingPitch = true;
378 m_dragStartY = event.position.y;
379 m_dragStartValue = m_instance->getParameter(Drift::kPitch);
380 return true;
381 }
382 if (mixTrack.contains(event.position)) {
383 m_draggingMix = true;
384 setParameter(Drift::kMix, (event.position.x - mixTrack.x) / mixTrack.width);
385 return true;
386 }
387 const int knob = hitTestKnob(event.position);
388 if (knob >= 0) {
389 m_draggingKnob = knob;
390 m_dragStartY = event.position.y;
391 m_dragStartValue = m_instance->getParameter(m_knobs[static_cast<size_t>(knob)].parameterId);
392 return true;
393 }
394 }
395
396 if (event.type == NUIMouseEventType::Move || event.button == NUIMouseButton::None) {
397 if (m_draggingPitch) {
398 setParameter(Drift::kPitch, m_dragStartValue + (m_dragStartY - event.position.y) / kDragRangePixels);
399 return true;
400 }
401 if (m_draggingMix) {
402 setParameter(Drift::kMix, (event.position.x - mixTrack.x) / mixTrack.width);
403 return true;
404 }
405 if (m_draggingKnob >= 0) {
406 const auto& knob = m_knobs[static_cast<size_t>(m_draggingKnob)];
407 setParameter(knob.parameterId, m_dragStartValue + (m_dragStartY - event.position.y) / kDragRangePixels);
408 return true;
409 }
410 m_pitchHovered = m_pitchWheelBounds.contains(event.position);
411 m_bypassHovered = m_bypassBounds.contains(event.position);
412 m_hoveredKnob = hitTestKnob(event.position);
413 m_hoveredInterval = -1;
414 for (size_t i = 0; i < m_intervalBounds.size(); ++i) {
415 if (m_intervalBounds[i].contains(event.position)) {
416 m_hoveredInterval = static_cast<int>(i);
417 break;
418 }
419 }
420 setDirty();
421 }
422
423 if (event.type == NUIMouseEventType::Scroll) {
424 if (m_pitchWheelBounds.contains(event.position)) {
425 setParameter(Drift::kPitch, m_instance->getParameter(Drift::kPitch) + event.wheelDelta * 0.01f);
426 return true;
427 }
428 if (mixTrack.contains(event.position)) {
429 setParameter(Drift::kMix, m_instance->getParameter(Drift::kMix) + event.wheelDelta * 0.01f);
430 return true;
431 }
432 const int knob = hitTestKnob(event.position);
433 if (knob >= 0) {
434 const uint32_t id = m_knobs[static_cast<size_t>(knob)].parameterId;
435 setParameter(id, m_instance->getParameter(id) + event.wheelDelta * 0.01f);
436 return true;
437 }
438 }
439
440 return consumeInsideBounds(event);
441}
