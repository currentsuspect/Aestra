14void CommandHistory::pushAndExecute(std::shared_ptr<ICommand> cmd) {
15 if (!cmd)
16 return;
17
18 // If a transaction is active, add to the transaction instead of executing
19 {
20 std::lock_guard<std::mutex> lock(m_mutex);
21 if (m_activeTransaction && m_transactionNestingLevel > 0) {
22 m_activeTransaction->add(cmd);
23 return;
24 }
25 }
26
27 // Execute BEFORE acquiring lock to prevent deadlock if command
28 // triggers callbacks that re-enter CommandHistory
29 try {
30 cmd->execute();
31 } catch (const std::exception& e) {
32 std::cerr << "Command execution failed: " << e.what() << std::endl;
33 return;
34 }
35 if (!cmd->isUndoable()) {
36 return;
37 }
38
39 bool stateChanged = false;
40 {
41 std::lock_guard<std::mutex> lock(m_mutex);
42
43 // Add to undo stack
44 m_undoStack.push_back(cmd);
45
46 // Clear redo stack on new action
47 m_redoStack.clear();
48
49 // Trim history if needed
50 trimHistory();
51
52 stateChanged = true;
53 }
54
55 // Notify listeners AFTER releasing lock to prevent deadlock
56 // if callback queries CommandHistory state
57 if (stateChanged) {
58 for (const auto& cb : m_onStateChangedCallbacks) { if (cb) cb(); }
59 }
60}

62bool CommandHistory::pushExecuted(std::shared_ptr<ICommand> cmd) {
63 if (!cmd || !cmd->isUndoable())
64 return false;
65
66 bool recorded = false;
67 {
68 // Transaction check and stack insertion under one lock so a
69 // transaction beginning on another thread cannot slip between them.
70 std::lock_guard<std::mutex> lock(m_mutex);
71 if (m_activeTransaction && m_transactionNestingLevel > 0) {
72 // Deferred-execution transactions would re-run this command at
73 // commit; refuse rather than execute it twice.
74 std::cerr << "CommandHistory::pushExecuted: not valid inside an active transaction"
75 << std::endl;
76 } else {
77 m_undoStack.push_back(cmd);
78 m_redoStack.clear();
79 trimHistory();
80 recorded = true;
81 }
82 }
83
84 if (recorded) {
85 for (const auto& cb : m_onStateChangedCallbacks) { if (cb) cb(); }
86 }
87 return recorded;
88}

98void CommandHistory::commitTransaction() {
99 std::shared_ptr<CommandTransaction> transactionToCommit;
100 {
101 std::lock_guard<std::mutex> lock(m_mutex);
102 if (m_transactionNestingLevel <= 0) {
103 std::cerr << "CommandHistory::commitTransaction: No active transaction" << std::endl;
104 return;
105 }
106 m_transactionNestingLevel--;
107 if (m_transactionNestingLevel == 0) {
108 transactionToCommit = m_activeTransaction;
109 m_activeTransaction.reset();
110 }
111 }
112
113 if (transactionToCommit && transactionToCommit->isValid()) {
114 // Execute the transaction (this executes all commands in it)
115 try {
116 transactionToCommit->execute();
117 } catch (const std::exception& e) {
118 std::cerr << "Transaction execution failed: " << e.what() << std::endl;
119 return;
120 }
121
122 bool stateChanged = false;
123 {
124 std::lock_guard<std::mutex> lock(m_mutex);
125 m_undoStack.push_back(transactionToCommit);
126 m_redoStack.clear();
127 trimHistory();
128 stateChanged = true;
129 }
130
131 if (stateChanged) { for (const auto& cb : m_onStateChangedCallbacks) { if (cb) cb(); } }
132 }
133}

152bool CommandHistory::undo() {
153 std::shared_ptr<ICommand> cmd;
154
155 std::lock_guard<std::mutex> opLock(m_operationMutex);
156
157 // Phase 1: Pop from undo stack under lock
158 {
159 std::lock_guard<std::mutex> lock(m_mutex);
160 if (m_undoStack.empty())
161 return false;
162 cmd = m_undoStack.back();
163 m_undoStack.pop_back();
164 }
165
166 // Phase 2: Execute undo outside lock to prevent deadlock
167 // if the command's undo triggers code that re-enters CommandHistory
168 bool success = false;
169 try {
170 cmd->undo();
171 success = true;
172 } catch (const std::exception& e) {
173 std::cerr << "Command undo failed: " << e.what() << std::endl;
174 std::lock_guard<std::mutex> lock(m_mutex);
175 m_undoStack.push_back(cmd);
176 } catch (...) {
177 std::cerr << "Command undo failed with unknown exception" << std::endl;
178 std::lock_guard<std::mutex> lock(m_mutex);
179 m_undoStack.push_back(cmd);
180 }
181
182 // Phase 3: Push to redo stack under lock
183 if (success) {
184 std::lock_guard<std::mutex> lock(m_mutex);
185 m_redoStack.push_back(cmd);
186 }
187
188 // Phase 4: Notify listeners outside lock
189 if (success) {
190 for (const auto& cb : m_onStateChangedCallbacks) { if (cb) cb(); }
191 }
192 return success;
193}

291void CommandHistory::trimHistory() {
292 if (m_maxHistorySize == 0 && m_maxHistoryMemory == 0)
293 return;
294
295 // Trim by count first
296 if (m_maxHistorySize > 0 && m_undoStack.size() > m_maxHistorySize) {
297 size_t excess = m_undoStack.size() - m_maxHistorySize;
298 m_undoStack.erase(m_undoStack.begin(), m_undoStack.begin() + excess);
299 }
300
301 // Then trim by memory if still over limit
302 if (m_maxHistoryMemory > 0) {
303 trimHistoryByMemory();
304 }
305}

307void CommandHistory::trimHistoryByMemory() {
308 if (m_maxHistoryMemory == 0)
309 return;
310
311 while (!m_undoStack.empty() && calculateMemoryUsage() > m_maxHistoryMemory) {
312 // Remove oldest commands until under memory limit
313 m_undoStack.erase(m_undoStack.begin());
314 }

327void CommandHistory::undoTo(int targetIndex) {
328 std::vector<std::shared_ptr<ICommand>> cmds;
329
330 std::lock_guard<std::mutex> opLock(m_operationMutex);
331
332 // Phase 1: Pop all commands to undo under lock
333 {
334 std::lock_guard<std::mutex> lock(m_mutex);
335 while (static_cast<int>(m_undoStack.size()) > targetIndex && !m_undoStack.empty()) {
336 cmds.push_back(m_undoStack.back());
337 m_undoStack.pop_back();
338 }
339 }
340
341 // Phase 2: Execute all undos outside lock
342 std::vector<std::shared_ptr<ICommand>> successfulUndos;
343 size_t nextIndex = 0;
344 try {
345 for (; nextIndex < cmds.size(); ++nextIndex) {
346 cmds[nextIndex]->undo();
347 successfulUndos.push_back(cmds[nextIndex]);
348 }
349 } catch (const std::exception& e) {
350 std::cerr << "Command undoTo failed: " << e.what() << std::endl;
351 } catch (...) {
352 std::cerr << "Command undoTo failed with unknown exception" << std::endl;
353 }
354
355 // Phase 3: Move only successful undos to redo and restore the rest.
356 {
357 std::lock_guard<std::mutex> lock(m_mutex);
358 for (size_t i = cmds.size(); i > nextIndex; --i) {
359 m_undoStack.push_back(cmds[i - 1]);
360 }
361 for (auto& cmd : successfulUndos) {
362 m_redoStack.push_back(cmd);
363 }
364 }
365
366 // Phase 4: Notify listeners outside lock
367 if (!successfulUndos.empty()) {
368 for (const auto& cb : m_onStateChangedCallbacks) { if (cb) cb(); }
369 }
370}
