55void ArsenalSceneMenu::show(const Request& request) {
56 if (!request.tracks || !request.owner || request.id == 0) return;
57 cancel();
58 const Audio::PatternID id{request.id};
59 auto& tracks = *request.tracks;
60
61 m_menu = std::make_shared<AestraUI::NUIContextMenu>();
62 m_menu->setOnHide([this]() { detach(m_menu); });
63
64 m_menu->addItem("Rename", [this, request]() { beginRename(request); });
65
66 m_menu->addItem("Duplicate", [request, id]() {
67 auto copy = std::make_shared<Audio::DuplicatePatternCommand>(request.tracks->getPatternManager(), id);
68 request.tracks->getCommandHistory().pushAndExecute(copy);
69 if (request.changed) request.changed(copy->copyId().value);
70 });
71
72 std::string why;
73 const bool allowed = canDelete(request.tracks->getPlaylistModel().isPatternUsed(id), request.scenes.size(), &why);
74 auto del = std::make_shared<AestraUI::NUIContextMenuItem>(allowed ? "Delete" : why);
75 del->setDestructive(true);
76 del->setEnabled(allowed);
77 del->setOnClick([request, id]() {
78 // Pick the neighbour before the pattern is gone; only move the selection when the active one is the one deleted.
79 const uint64_t next = request.activeId == request.id ? ArsenalSceneMenu::selectAfterDelete(request.scenes, request.id) : 0;
80 auto remove = std::make_shared<Audio::DeletePatternCommand>(request.tracks->getPatternManager(), id);
81 request.tracks->getCommandHistory().pushAndExecute(remove);
82 if (request.changed) request.changed(next);
83 });
84 m_menu->addItem(del);
85
86 AestraUI::NUIComponent* root = rootOf(request.owner);
87 if (!root) root = request.owner;
88 root->addChild(m_menu);
89 m_menu->setOwner(request.owner->weak_from_this()); // the menu outlives the panel's callbacks, so it drops them with it
90 m_menu->showAt(request.at);
91 root->repaint();
92}

94void ArsenalSceneMenu::beginRename(const Request& request) {
95 const auto* pattern = request.tracks->getPatternManager().getPattern(Audio::PatternID{request.id});
96 if (!pattern || request.chip.width <= 0.0f) return;
97 finishRename(false);
98 m_retired.reset();
99
100 auto& theme = AestraUI::NUIThemeManager::getInstance();
101 m_input = std::make_shared<AestraUI::NUITextInput>(pattern->name);
102 m_input->setBounds(request.chip);
103 m_input->setTextColor(theme.getColor("textPrimary"));
104 m_input->setBackgroundColor(theme.getColor("inputBgDefault"));
105 m_input->setBorderColor(theme.getColor("inputBorderFocus"));
106 m_input->setBorderWidth(1.0f);
107 m_input->setBorderRadius(3.0f);
108 m_input->setFocusedBorderColor(theme.getColor("accentPrimary"));
109 m_input->setOnReturnKey([this]() { finishRename(true); });
110 m_input->setOnEscapeKey([this]() { finishRename(false); });
111 m_input->setOnFocusLost([this]() { finishRename(true); });
112
113 m_inputOwner = request.owner;
114 m_tracks = request.tracks;
115 m_renameId = request.id;
116 m_changed = request.changed;
117 request.owner->addChild(m_input);
118 m_input->setFocused(true);
119 m_input->setCaretPosition(static_cast<int>(pattern->name.length()));
120 m_input->selectAll();
121 request.owner->repaint();
122}

124void ArsenalSceneMenu::finishRename(bool commit) {
125 if (!m_input) return;
126 // Take the box out first: committing (or losing focus) can arrive from inside the box's own handler.
127 auto input = std::move(m_input);
128 m_input.reset();
129 const std::string text = input->getText();
130 if (m_inputOwner) {
131 m_inputOwner->removeChild(input);
132 m_inputOwner->repaint();
133 }
134 m_retired = std::move(input);
135
136 if (!commit || !m_tracks || text.empty()) return;
137 const Audio::PatternID id{m_renameId};
138 const auto* pattern = m_tracks->getPatternManager().getPattern(id);
139 if (!pattern || pattern->name == text) return;
140 auto rename = std::make_shared<Audio::RenamePatternCommand>(m_tracks->getPatternManager(), id, text);
141 m_tracks->getCommandHistory().pushAndExecute(rename);
142 if (m_changed) m_changed(0);
143}
