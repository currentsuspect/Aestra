15AudioGraph AudioGraphBuilder::buildFromTrackManager(TrackManager& trackManager) {
16 AudioGraph graph;
17 const size_t channelCount = trackManager.getChannelCount();
18 graph.tracks.reserve(channelCount);
19
20 // Create track render states for all mixer channels
21 bool anySoloFound = false;
22 for (size_t i = 0; i < channelCount; ++i) {
23 auto channel = trackManager.getChannel(i);
24 if (!channel)
25 continue;
26
27 TrackRenderState trackState;
28 trackState.trackId = channel->getChannelId();
29 trackState.trackIndex = static_cast<uint32_t>(i);
30 trackState.volume = channel->getVolume();
31 trackState.pan = channel->getPan();
32 trackState.mute = channel->isMuted();
33 trackState.solo = channel->isSoloed();
34 if (trackState.solo)
35 anySoloFound = true;
36 trackState.isSoloSafe = channel->isSoloSafe();
37
38 // Copy Routing
39 trackState.mainOutputId = channel->getMainOutputId();
40 trackState.sends = channel->getSends();
41 trackState.effectChainSnapshot = channel->getEffectChainSnapshot();
42
43 graph.tracks.push_back(std::move(trackState));
44 }
45 graph.anySolo = anySoloFound;
46
47 // Populate clips from PlaylistModel
48 const auto& playlist = trackManager.getPlaylistModel();
49 const auto& patterns = trackManager.getPatternManager();
50 const auto& sources = trackManager.getSourceManager();
51
52 auto snapshot = playlist.buildRuntimeSnapshot(patterns, sources);
53 if (snapshot) {
54 uint64_t maxEndSample = 0;
55 double projectSampleRate = playlist.getProjectSampleRate();
56
57 // Map snapshot lanes to mixer tracks.
58 // In the current implementation, lane index usually corresponds to mixer channel index.
59 for (size_t laneIdx = 0; laneIdx < snapshot->lanes.size(); ++laneIdx) {
60 if (laneIdx >= graph.tracks.size()) {
61 AESTRA_DEBUG_ONLY(std::cerr << "[AudioGraphBuilder] Warning: snapshot has more lanes (" << snapshot->lanes.size()
62 << ") than mixer tracks (" << graph.tracks.size() << "). Extra lanes dropped.\n");
63 break;
64 }
65
66 auto& trackState = graph.tracks[laneIdx];
67 const auto& laneInfo = snapshot->lanes[laneIdx];
68
69 for (const auto& clipInfo : laneInfo.clips) {
70 if (!clipInfo.isValid())
71 continue;
72
73 ClipRenderState clip;
74 // ClipRenderState.bufferOwner holds a shared_ptr<AudioBufferData>
75 // that extends the buffer's lifetime independently of SourceManager.
76 // audioData is extracted from the same buffer as a raw pointer for hot-path use.
77
78 if (clipInfo.isAudio()) {
79 clip.audioData = clipInfo.audioData->interleavedData.data();
80 clip.bufferOwner = clipInfo.sharedAudioData;
81 clip.totalFrames = clipInfo.audioData->numFrames;
82 clip.sourceSampleRate = static_cast<double>(clipInfo.sourceSampleRate);
83 clip.channels = clipInfo.sourceChannels;
84 }
85
86 clip.startSample = clipInfo.startTime;
87 clip.endSample = clipInfo.getEndTime();
88
89 // Convert sourceStart (Project Rate) to sampleOffset (Source Rate)
90 // Use double precision to prevent audio popping due to sub-sample drift
91 if (projectSampleRate > 0.0 && clip.sourceSampleRate > 0.0) {
92 clip.sampleOffset =
93 static_cast<double>(clipInfo.sourceStart) * (clip.sourceSampleRate / projectSampleRate);
94 } else {
95 clip.sampleOffset = static_cast<double>(clipInfo.sourceStart);
96 }
97
98 clip.gain = clipInfo.gainLinear;
99 clip.pan = clipInfo.pan;
100
101 if (clip.endSample > maxEndSample) {
102 maxEndSample = clip.endSample;
103 }
104
105 trackState.clips.push_back(std::move(clip));
106 }
107 trackState.automationCurves = std::move(laneInfo.automationCurves);
108 }
109
110 graph.timelineEndSample = maxEndSample;
111 graph.bpm = snapshot->bpm;
112 }
113
114 return graph;
115}
