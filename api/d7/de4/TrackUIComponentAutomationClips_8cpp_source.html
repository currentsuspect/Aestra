22bool TrackUIComponent::handleAutomationClipMouse(const AestraUI::NUIMouseEvent& event,
23 const AestraUI::NUIRect& bounds, float gridStartX) {
24 AutomationClipLayer::Hooks hooks;
25 hooks.snapBeat = [this](double beat) { return snapBeatToGrid(std::max(0.0, beat)); };
26 hooks.captureMouse = [this](bool on) {
27 if (auto* parent = dynamic_cast<TrackManagerUI*>(getParent()))
28 if (auto* window = parent->getPlatformWindow())
29 window->setMouseCapture(on);
30 };
31 hooks.changed = [this]() {
32 setDirty(true);
33 repaint();
34 if (m_onCacheInvalidationCallback)
35 m_onCacheInvalidationCallback();
36 };
37 int edited = -1;
38 const bool handled = m_clipLayer.onMouse(event, *m_trackManager, m_laneId,
39 {bounds, gridStartX, m_pixelsPerBeat, m_timelineScrollOffset,
40 m_editedAutomationTarget},
41 hooks, edited);
42 if (edited >= 0) {
43 if (const auto* lane = m_trackManager->getPlaylistModel().getLane(m_laneId))
44 if (edited < static_cast<int>(lane->automationCurves.size()))
45 setEditedAutomationTarget(lane->automationCurves[static_cast<size_t>(edited)].getAutomationTarget());
46 }
47 return handled;
48}
