43bool PianoRollControlPanel::onMouseEvent(const NUIMouseEvent& event) {
44 // Note: If dropdown consumes event, we return true above.
45 // Else check sidebar/velocity area.
46
47 auto b = getBounds();
48
49 // CRITICAL FIX: Ignore events outside bounds unless we are already dragging
50 if (!b.contains(event.position) && !isDragging_) return false;
51
52 constexpr float sidebarW = 76.0f;
53 // Note Layer Interaction (Velocity)
54 // We assume clicks in content area are for velocity
55 // COPIED FROM OLD LOGIC
56 auto layer = noteLayer_.lock();
57 if (layer) {
58 auto b = getBounds();
59 // Sidebar: clicking it flips the lane between velocity and pan.
60 if (!isDragging_ && event.position.x < b.x + sidebarW) {
61 if (event.pressed && event.button == NUIMouseButton::Left) {
62 laneMode_ = (laneMode_ == LaneMode::Velocity) ? LaneMode::Pan : LaneMode::Velocity;
63 repaint();
64 return true;
65 }
66 return NUIComponent::onMouseEvent(event);
67 }
68
69 const bool panMode = laneMode_ == LaneMode::Pan;
70 float localX = event.position.x - b.x + scrollX_ - sidebarW;
71
72 // Find the note whose velocity stem sits under a given lane-x, preferring
73 // a selected one. Shared by the initial press and the sweep-drag below.
74 auto noteUnderX = [&](float lx) -> int {
75 const auto& notes = layer->getNotes();
76 int found = -1;
77 for (int i = 0; i < static_cast<int>(notes.size()); ++i) {
78 if (notes[i].isDeleted) continue;
79 float nStart = static_cast<float>(notes[i].startBeat * pixelsPerBeat_);
80 float nEnd = static_cast<float>((notes[i].startBeat + notes[i].durationBeats) * pixelsPerBeat_);
81 if (nEnd < nStart + 6.0f) nEnd = nStart + 6.0f; // lollipop head radius
82 const bool hitHead = std::abs(nStart - lx) < 10.0f;
83 const bool hitBody = (lx >= nStart - 2.0f && lx <= nEnd + 2.0f);
84 if (hitHead || hitBody) {
85 found = i; // last (topmost) wins by default
86 if (notes[i].selected) return i; // but a selected note takes priority
87 }
88 }
89 return found;
90 };
91
92 if (event.pressed && event.button == NUIMouseButton::Left) {
93 int foundIdx = noteUnderX(localX);
94
95 // Map the cursor height to the lane's value: velocity is unipolar
96 // from the lane floor; pan is bipolar around the centre line
97 // (top = right, bottom = left).
98 const auto laneValueAtY = [&](float y) -> float {
99 const float availH = std::max(1.0f, b.height - 28.0f);
100 const float bottomY = b.bottom() - 8.0f;
101 if (panMode) {
102 const float centerY = bottomY - availH * 0.5f;
103 return std::clamp((centerY - y) / std::max(1.0f, availH * 0.5f), -1.0f, 1.0f);
104 }
105 return velocityFromPanelPosition(y, bottomY, availH);
106 };
107
108 if (foundIdx != -1) {
109 isDragging_ = true;
110 hoveringNoteIndex_ = foundIdx;
111 dragStartPos_ = event.position;
112 dragUndoSnapshot_ = layer->getNotes(); // baseline for one undo step
113
114 // Set the value immediately based on click Y (Global)
115 const float newValue = laneValueAtY(event.position.y);
116
117 auto modNotes = layer->getNotes();
118 if (panMode) modNotes[foundIdx].pan = newValue;
119 else modNotes[foundIdx].velocity = newValue;
120
121 // Single Edit Only (Batch removed)
122
123 layer->setNotes(modNotes);
124 repaint();
125 return true;
126 }
127 }
128 else if (isDragging_) {
129 // Drag logic
130 if (event.released) {
131 isDragging_ = false;
132 hoveringNoteIndex_ = -1;
133 // Fold the whole lane drag into a single undo step.
134 if (!dragUndoSnapshot_.empty()) {
135 layer->pushExternalEdit(dragUndoSnapshot_, panMode ? "Pan" : "Velocity");
136 dragUndoSnapshot_.clear();
137 }
138 return true;
139 }
140
141 // Sweep-paint: as the cursor moves across the lane, the note under it
142 // takes the cursor's height, so dragging draws a ramp/curve.
143 // When between stems, keep painting the last note so quick sweeps
144 // don't drop out.
145 const float availH = std::max(1.0f, b.height - 28.0f);
146 const float bottomY = b.bottom() - 8.0f;
147 float newValue;
148 if (panMode) {
149 const float centerY = bottomY - availH * 0.5f;
150 newValue = std::clamp((centerY - event.position.y) / std::max(1.0f, availH * 0.5f),
151 -1.0f, 1.0f);
152 } else {
153 newValue = velocityFromPanelPosition(event.position.y, bottomY, availH);
154 }
155
156 const int swept = noteUnderX(localX);
157 if (swept != -1) hoveringNoteIndex_ = swept;
158
159 auto modNotes = layer->getNotes();
160 if (hoveringNoteIndex_ >= 0 && static_cast<size_t>(hoveringNoteIndex_) < modNotes.size()) {
161 if (panMode) modNotes[hoveringNoteIndex_].pan = newValue;
162 else modNotes[hoveringNoteIndex_].velocity = newValue;
163 layer->setNotes(modNotes);
164 repaint();
165 }
166 return true;
167 }
168 }
169
170 return NUIComponent::onMouseEvent(event);
171}

173void PianoRollControlPanel::onRender(NUIRenderer& renderer) {
174 auto b = getBounds();
175 auto& themeManager = NUIThemeManager::getInstance();
176
177 const auto panelBg = themeManager.getColor("backgroundSecondary").darkened(0.025f);
178 const auto border = themeManager.getColor("border").withAlpha(0.52f);
179 renderer.fillRect(b, panelBg);
180 renderer.drawLine(NUIPoint(b.x, b.y), NUIPoint(b.right(), b.y), 1.0f, border);
181
182 // Sidebar Area (Left)
183 constexpr float sidebarW = 76.0f;
184 NUIRect sidebarRect(b.x, b.y, sidebarW, b.height);
185
186 renderer.fillRect(sidebarRect, themeManager.getColor("backgroundPrimary").withAlpha(0.72f));
187 renderer.drawLine(NUIPoint(sidebarRect.right(), sidebarRect.y),
188 NUIPoint(sidebarRect.right(), sidebarRect.bottom()),
189 1.0f,
190 border);
191
192 // Stacked mode tabs — the active lane reads accent, the other dim; a click
193 // anywhere on the sidebar swaps them (handled in onMouseEvent).
194 const bool panMode = laneMode_ == LaneMode::Pan;
195 const float labelSize = themeManager.getFontSize("xs");
196 const auto activeColor = themeManager.getColor("accentPrimary").withAlpha(0.95f);
197 const auto inactiveColor = themeManager.getColor("textSecondary").withAlpha(0.42f);
198 const auto velDim = renderer.measureText("VELOCITY", labelSize);
199 renderer.drawText("VELOCITY",
200 NUIPoint(b.x + (sidebarW - velDim.width) * 0.5f, b.y + 13.0f),
201 labelSize,
202 panMode ? inactiveColor : activeColor);
203 const auto panDim = renderer.measureText("PAN", labelSize);
204 renderer.drawText("PAN",
205 NUIPoint(b.x + (sidebarW - panDim.width) * 0.5f, b.y + 31.0f),
206 labelSize,
207 panMode ? activeColor : inactiveColor);
208 const char* rangeText = panMode ? "L - C - R" : "MIDI 0 - 127";
209 const auto rangeDim = renderer.measureText(rangeText, 8.0f);
210 renderer.drawText(rangeText,
211 NUIPoint(b.x + (sidebarW - rangeDim.width) * 0.5f, b.y + 49.0f),
212 8.0f,
213 themeManager.getColor("textSecondary").withAlpha(0.48f));
214
215 auto layer = noteLayer_.lock();
216 if (!layer || !isVisible()) return;
217
218 // Content Area Clip
219 NUIRect contentRect(b.x + sidebarW, b.y, b.width - sidebarW, b.height);
220 renderer.setClipRect(contentRect);
221
222 const float startX = contentRect.x;
223 renderTimelineGrid(renderer, contentRect, startX, contentRect.right(), scrollX_, pixelsPerBeat_, beatsPerBar_,
224 themeManager.getCurrentTheme().textPrimary);
225
226 const float availH = std::max(1.0f, b.height - 28.0f);
227 const float bottomY = b.bottom() - 8.0f;
228 const float topY = bottomY - availH;
229 const auto guideColor = themeManager.getColor("border").withAlpha(0.11f);
230 renderer.drawLine(NUIPoint(contentRect.x, topY), NUIPoint(contentRect.right(), topY), 1.0f, guideColor);
231 renderer.drawLine(NUIPoint(contentRect.x, topY + availH * 0.5f),
232 NUIPoint(contentRect.right(), topY + availH * 0.5f),
233 1.0f,
234 guideColor.withAlpha(0.72f));
235 renderer.drawLine(NUIPoint(contentRect.x, bottomY), NUIPoint(contentRect.right(), bottomY), 1.0f, guideColor);
236
237 const float patternEndX = startX + static_cast<float>(layer->getTotalDurationBeats() * pixelsPerBeat_) - scrollX_;
238 if (patternEndX < contentRect.right()) {
239 const float shadeX = std::max(contentRect.x, patternEndX);
240 renderer.fillRect(NUIRect(shadeX, contentRect.y, contentRect.right() - shadeX, contentRect.height),
241 NUIThemeManager::getInstance().getCurrentTheme().textPrimary.withAlpha(0.14f));
242 }
243 if (patternEndX >= contentRect.x && patternEndX <= contentRect.right()) {
244 renderer.drawLine(NUIPoint(patternEndX, contentRect.y),
245 NUIPoint(patternEndX, contentRect.bottom()),
246 2.0f,
247 themeManager.getColor("accentPrimary").withAlpha(0.58f));
248 }
249
250 // 2. Render the lane stems: velocity rises from the floor; pan hangs off
251 // the centre line (up = right, down = left).
252 const auto& notes = layer->getNotes();
253 auto velColorBase = themeManager.getColor("accentPrimary").lightened(0.05f);
254 const float centerY = bottomY - availH * 0.5f;
255
256 for (size_t noteIndex = 0; noteIndex < notes.size(); ++noteIndex) {
257 const auto& n = notes[noteIndex];
258 if (n.isDeleted && n.animationScale < 0.01f) continue;
259
260 float x = startX + static_cast<float>(n.startBeat * pixelsPerBeat_) - scrollX_;
261
262 // Skip if out of view
263 if (x > b.x + b.width) continue;
264
265 const float normalizedVelocity = std::clamp(n.velocity, 0.0f, 1.0f);
266 float y;
267 float stemTop;
268 float stemH;
269 float intensity; // drives the stem alpha
270 if (panMode) {
271 const float pv = std::clamp(n.pan, -1.0f, 1.0f);
272 y = centerY - pv * availH * 0.5f;
273 stemTop = std::min(y, centerY);
274 stemH = std::max(2.0f, std::abs(y - centerY));
275 intensity = std::abs(pv);
276 } else {
277 const float h = velocityToPanelHeight(normalizedVelocity, availH);
278 y = bottomY - h;
279 stemTop = y;
280 stemH = std::max(2.0f, h);
281 intensity = normalizedVelocity;
282 }
283
284 float alpha = 0.48f + intensity * 0.48f;
285 auto col = velColorBase.withAlpha(alpha);
286 if (n.selected) col = themeManager.getColor("accentSecondary").withAlpha(0.92f);
287
288 renderer.fillRoundedRect(NUIRect(x - 2.0f, stemTop, 4.0f, stemH), 2.0f, col.withAlpha(alpha * 0.78f));
289
290 const float handleSize = n.selected ? 7.0f : 6.0f;
291 NUIRect handleRect(x - handleSize * 0.5f, y - handleSize * 0.5f, handleSize, handleSize);
292 renderer.fillRoundedRect(handleRect, 2.0f, col);
293 renderer.strokeRoundedRect(handleRect,
294 2.0f,
295 1.0f,
296 NUIColor::white().withAlpha(n.selected ? 0.48f : 0.20f));
297
298 float w = static_cast<float>(n.durationBeats * pixelsPerBeat_);
299 if (w > 4.0f) {
300 renderer.drawLine(NUIPoint(x, y), NUIPoint(x + w, y), n.selected ? 2.0f : 1.0f, col.withAlpha(0.64f));
301 }
302
303 if (isDragging_ && static_cast<int>(noteIndex) == hoveringNoteIndex_) {
304 std::string value;
305 if (panMode) {
306 const int panSteps = static_cast<int>(std::lround(std::abs(n.pan) * 100.0f));
307 if (panSteps == 0) value = "C";
308 else value = (n.pan < 0.0f ? "L " : "R ") + std::to_string(panSteps);
309 } else {
310 value = std::to_string(static_cast<int>(std::lround(normalizedVelocity * 127.0f)));
311 }
312 const float fontSize = 9.0f;
313 const auto textSize = renderer.measureText(value, fontSize);
314 const float bubbleWidth = textSize.width + 10.0f;
315 const float bubbleY = y - 18.0f >= contentRect.y ? y - 18.0f : y + 8.0f;
316 const NUIRect bubble(x + 7.0f, bubbleY, bubbleWidth, 16.0f);
317 renderer.fillRoundedRect(bubble, 4.0f, NUIColor::black().withAlpha(0.92f));
318 renderer.strokeRoundedRect(bubble, 4.0f, 1.0f, col.withAlpha(0.82f));
319 renderer.drawText(value,
320 NUIPoint(bubble.x + 5.0f, bubble.y + (bubble.height - textSize.height) * 0.5f),
321 fontSize,
322 NUIColor::white().withAlpha(0.92f));
323 }
324 }
325
326 renderer.clearClipRect();
327}
