14bool placeClipAsOneUndoStep(TrackManager& trackManager, const PlaylistLaneID& laneId, const ClipInstance& clip,
15 const std::string& newLaneName, const std::string& stepName) {
16 auto& playlist = trackManager.getPlaylistModel();
17
18 // Each member is executed and validated against the state its predecessor produced,
19 // then the whole is adopted as one executed transaction (as a drop does).
20 std::shared_ptr<CreateTrackWithLaneCommand> laneCommand;
21 PlaylistLaneID target = laneId;
22 if (!target.isValid()) {
23 laneCommand = std::make_shared<CreateTrackWithLaneCommand>(trackManager, newLaneName);
24 laneCommand->execute();
25 target = laneCommand->getLaneId();
26 if (!target.isValid()) {
27 return false;
28 }
29 }
30
31 ClipInstance placed = clip; // the id is fixed here so success can be checked against it
32 if (!placed.id.isValid()) {
33 placed.id = ClipInstanceID::generate();
34 }
35 auto clipCommand = std::make_shared<AddClipCommand>(playlist, target, placed);
36 clipCommand->execute();
37 if (!playlist.getClip(placed.id)) {
38 if (laneCommand) {
39 laneCommand->undo();
40 }
41 return false;
42 }
43
44 auto transaction = std::make_shared<CommandTransaction>(stepName);
45 if (laneCommand) {
46 transaction->add(laneCommand);
47 }
48 transaction->add(clipCommand);
49 transaction->markExecuted();
50 return trackManager.getCommandHistory().pushExecuted(transaction);
51}
