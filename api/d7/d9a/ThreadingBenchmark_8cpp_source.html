27struct Stats {
28 DurationNs median{0};
29 DurationNs p95{0};
30 DurationNs p99{0};
31 DurationNs min{0};
32 DurationNs max{0};
33 double mean{0};
34 double stddev{0};
35 int xruns{0}; // operations exceeding 10× median
36 int missedDeadlines{0}; // operations exceeding a hard deadline
37 int samples{0};
38
39 DurationNs deadlineNs{0}; // configured deadline for this benchmark
40};
41
42void computeStats(Stats& s, std::vector<DurationNs>& samples, DurationNs deadlineNs = 0) {
43 s.samples = static_cast<int>(samples.size());
44 s.deadlineNs = deadlineNs;
45 if (samples.empty()) return;
46
47 std::sort(samples.begin(), samples.end());
48 s.min = samples.front();
49 s.max = samples.back();
50 s.median = samples[samples.size() / 2];
51 s.p95 = samples[static_cast<size_t>(samples.size() * 0.95)];
52 s.p99 = samples[static_cast<size_t>(samples.size() * 0.99)];
53
54 double sum = 0;
55 for (auto v : samples) sum += v;
56 s.mean = sum / samples.size();
57
58 double sqSum = 0;
59 for (auto v : samples) sqSum += (v - s.mean) * (v - s.mean);
60 s.stddev = std::sqrt(sqSum / samples.size());
61
62 DurationNs xrunThreshold = s.median > 0 ? s.median * 10 : 10'000'000; // 10ms fallback
63 for (auto v : samples) {
64 if (v > xrunThreshold) s.xruns++;
65 if (deadlineNs > 0 && v > deadlineNs) s.missedDeadlines++;
66 }
67}

91
92Stats benchRingBufferLatency(int iterations = 100'000) {
93 LockFreeRingBuffer<int, 1024> buffer;
94 std::vector<DurationNs> samples;
95 samples.reserve(iterations);
96
97 int pushVal = 42;
98 int popVal = 0;
99 for (int i = 0; i < iterations; ++i) {
100 auto t0 = Clock::now();
101 if (!buffer.push(pushVal)) { /* should not happen in empty buffer */ }
102 if (!buffer.pop(popVal)) { /* should not happen */ }
103 auto t1 = Clock::now();
104 samples.push_back(std::chrono::duration_cast<std::chrono::nanoseconds>(t1 - t0).count());
105 }
106
107 Stats s;
108 // Deadline: 1μs per push+pop pair is generous for lock-free ops
109 computeStats(s, samples, 1'000);
110 return s;
111}

113// =============================================================================
114// Benchmark: Ring Buffer Contended (producer + consumer)
115// =============================================================================
116
117Stats benchRingBufferContention(int items = 100'000) {
118 LockFreeRingBuffer<int, 1024> buffer;
119 std::vector<DurationNs> pushSamples;
120 std::vector<DurationNs> popSamples;
121 pushSamples.reserve(items);
122 popSamples.reserve(items);
123 std::atomic<bool> producerDone{false};
124 int consumed = 0;
125
126 auto producer = [&]() {
127 for (int i = 0; i < items; ++i) {
128 auto t0 = Clock::now();
129 while (!buffer.push(i)) {
130 // spin on full — count the total time including backoff
131 std::this_thread::yield();
132 }
133 auto t1 = Clock::now();
134 pushSamples.push_back(std::chrono::duration_cast<std::chrono::nanoseconds>(t1 - t0).count());
135 }
136 producerDone = true;
137 };
138
139 auto consumer = [&]() {
140 int value = 0;
141 while (!producerDone || !buffer.isEmpty()) {
142 auto t0 = Clock::now();
143 if (buffer.pop(value)) {
144 auto t1 = Clock::now();
145 popSamples.push_back(std::chrono::duration_cast<std::chrono::nanoseconds>(t1 - t0).count());
146 consumed++;
147 } else {
148 std::this_thread::yield();
149 }
150 }
151 };
152
153 std::thread t1(producer);
154 std::thread t2(consumer);
155 t1.join();
156 t2.join();
157
158 Stats s;
159 // Deadline: 10μs for a push under contention
160 computeStats(s, pushSamples, 10'000);
161 return s;
162}

165// Benchmark: ThreadPool Task Dispatch Latency
166// =============================================================================
167
168Stats benchThreadPoolDispatch(int tasks = 10'000) {
169 ThreadPool pool(4);
170 std::vector<DurationNs> roundTrip;
171 roundTrip.reserve(tasks);
172 std::atomic<int> completed{0};
173
174 // Warm up
175 for (int i = 0; i < 100; ++i) {
176 (void)pool.enqueue([]{});
177 }
178 std::this_thread::sleep_for(std::chrono::milliseconds(10));
179
180 // Measure enqueue → task start latency
181 for (int i = 0; i < tasks; ++i) {
182 auto t0 = Clock::now();
183 (void)pool.enqueue([&roundTrip, t0, &completed]() {
184 auto t1 = Clock::now();
185 roundTrip.push_back(
186 std::chrono::duration_cast<std::chrono::nanoseconds>(t1 - t0).count());
187 completed.fetch_add(1, std::memory_order_release);
188 });
189 }
190
191 // Wait for completion
192 while (completed.load(std::memory_order_acquire) < tasks) {
193 std::this_thread::yield();
194 }
195
196 Stats s;
197 // Deadline: 10ms round-trip for a trivial task (includes mutex + condvar wakeup)
198 computeStats(s, roundTrip, 10'000'000);
199 return s;
200}

202// =============================================================================
203// Benchmark: Barrier Synchronization Latency
204// =============================================================================
205// Measures the cost of N threads reaching a barrier and the coordinator
206// waiting for all of them. Workers do minimal work (a counter increment).
207
208Stats benchBarrierSync(int iterations = 10'000, int numWorkers = 4) {
209 std::vector<DurationNs> samples;
210 samples.reserve(iterations);
211
212 Barrier barrier(numWorkers);
213 std::atomic<int> iterationCounter{0};
214 std::atomic<bool> stopWorkers{false};
215 std::atomic<int> startIteration{-1}; // workers wait for this to increment
216
217 std::vector<std::thread> workers;
218 for (int t = 0; t < numWorkers; ++t) {
219 workers.emplace_back([&, t]() {
220 int lastIter = -1;
221 while (!stopWorkers.load(std::memory_order_acquire)) {
222 // Wait for next iteration
223 int target = lastIter + 1;
224 while (startIteration.load(std::memory_order_acquire) < target) {
225 std::this_thread::yield();
226 }
227 lastIter = target;
228 // Minimal work: increment shared counter
229 iterationCounter.fetch_add(1, std::memory_order_relaxed);
230 // Signal barrier
231 barrier.signal();
232 }
233 });
234 }
235
236 for (int iter = 0; iter < iterations; ++iter) {
237 // Reset barrier for this round
238 barrier.reset(numWorkers);
239 auto t0 = Clock::now();
240 // Signal workers to start
241 startIteration.store(iter, std::memory_order_release);
242 // Wait for all workers to reach barrier
243 barrier.wait();
244 auto t1 = Clock::now();
245
246 samples.push_back(
247 std::chrono::duration_cast<std::chrono::nanoseconds>(t1 - t0).count());
248
249 // Wait until counter confirms all workers completed
250 while (iterationCounter.load(std::memory_order_acquire) < (iter + 1) * numWorkers) {
251 std::this_thread::yield();
252 }
253 }
254
255 stopWorkers.store(true, std::memory_order_release);
256 startIteration.store(2000000000, std::memory_order_release); // unblock workers
257 for (auto& w : workers) w.join();
258
259 Stats s;
260 // Deadline: 50μs for barrier sync (spin-wait, workers do 1 atomic op)
261 computeStats(s, samples, 50'000);
262 return s;
263}

267// =============================================================================
268
269Stats benchSpinLockContention(int iterations = 50'000, int threads = 4) {
270 SpinLock lock;
271 std::vector<DurationNs> samples;
272 samples.reserve(iterations);
273 std::atomic<int> totalOps{0};
274
275 std::vector<std::thread> workers;
276 for (int t = 0; t < threads; ++t) {
277 workers.emplace_back([&]() {
278 for (int i = 0; i < iterations / threads; ++i) {
279 auto t0 = Clock::now();
280 lock.lock();
281 // Critical section: trivial increment
282 totalOps.fetch_add(1, std::memory_order_relaxed);
283 lock.unlock();
284 auto t1 = Clock::now();
285 samples.push_back(
286 std::chrono::duration_cast<std::chrono::nanoseconds>(t1 - t0).count());
287 }
288 });
289 }
290
291 for (auto& w : workers) w.join();
292
293 Stats s;
294 // Deadline: 5μs to acquire + critical section under contention
295 computeStats(s, samples, 5'000);
296 return s;
297}
