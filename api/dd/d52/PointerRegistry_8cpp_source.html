37bool PointerRegistry::validateAll() {
38#if AESTRA_DEBUG_POINTERS
39 bool allPassed = true;
40 int passed = 0;
41 int failed = 0;
42
43 Log::info("[PointerRegistry] Validating " + std::to_string(expectations().size()) + " expectations...");
44
45 for (const auto& exp : expectations()) {
46 bool ok = false;
47 std::ostringstream oss;
48
49 if (exp.isNullCheck) {
50 ok = (exp.ptrA != nullptr);
51 oss << " " << (ok ? "✓" : "✗") << " " << exp.name << ": ";
52 if (ok) {
53 oss << "0x" << std::hex << reinterpret_cast<uintptr_t>(exp.ptrA);
54 } else {
55 oss << "NULL (FAILED)";
56 }
57 } else {
58 ok = (exp.ptrA == exp.ptrB);
59 oss << " " << (ok ? "✓" : "✗") << " " << exp.name << ": ";
60 oss << "0x" << std::hex << reinterpret_cast<uintptr_t>(exp.ptrA);
61 if (!ok) {
62 oss << " != 0x" << reinterpret_cast<uintptr_t>(exp.ptrB) << " (MISMATCH!)";
63 }
64 }
65
66 if (ok) {
67 passed++;
68 Log::info(oss.str());
69 } else {
70 failed++;
71 allPassed = false;
72 Log::warning(oss.str());
73 }
74 }
75
76 Log::info("[PointerRegistry] Result: " + std::to_string(passed) + " passed, " + std::to_string(failed) + " failed");
77
78 return allPassed;
79#else
80 return true; // Always passes in Release
81#endif
82}
