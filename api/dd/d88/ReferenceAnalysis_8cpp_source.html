21// One 2nd-order Butterworth section (RBJ), applied twice for a 4th-order slope.
22struct Section2 {
23 double b0 = 0, b1 = 0, b2 = 0, a1 = 0, a2 = 0, z1 = 0, z2 = 0;
24 Section2(double rate, double hz, bool high) {
25 const double w0 = 2.0 * kPi * hz / rate, c = std::cos(w0), alpha = std::sin(w0) / (2.0 * 0.7071067811865476);
26 const double a0 = 1.0 + alpha;
27 b1 = (high ? -(1.0 + c) : (1.0 - c)) / a0;
28 b0 = (high ? (1.0 + c) / 2.0 : (1.0 - c) / 2.0) / a0;
29 b2 = b0;
30 a1 = -2.0 * c / a0;
31 a2 = (1.0 - alpha) / a0;
32 }
33 double process(double x) {
34 const double y = b0 * x + z1;
35 z1 = b1 * x - a1 * y + z2;
36 z2 = b2 * x - a2 * y;
37 return y;
38 }
39};
40struct Cross {
41 Section2 a, b;
42 Cross(double rate, double hz, bool high) : a(rate, hz, high), b(rate, hz, high) {}
43 double process(double x) { return b.process(a.process(x)); }
44};

46// Splits one signal into the five analysis bands, one sample at a time.
47struct BandSplitter {
48 std::vector<Cross> lows, highs;
49 size_t edges = 0;
50 explicit BandSplitter(double rate) {
51 for (double edge : AudioAnalysis::kBandEdgesHz) {
52 if (edge >= rate * 0.5) break;
53 lows.emplace_back(rate, edge, false);
54 highs.emplace_back(rate, edge, true);
55 ++edges;
56 }
57 }
58 void split(double x, double (&out)[AudioAnalysis::kBandCount]) {
59 for (double& v : out) v = 0.0;
60 double rest = x;
61 for (size_t e = 0; e < edges; ++e) {
62 out[e] = lows[e].process(rest);
63 rest = highs[e].process(rest);
64 }
65 out[edges] = rest;
66 }
67};

76StereoDetail analyzeStereoDetail(const float* x, uint64_t frames, uint32_t channels, uint32_t rate) {
77 StereoDetail out;
78 if (!x || channels != 2 || frames == 0 || rate == 0) return out;
79
80 BandSplitter mid(rate), side(rate);
81 double midE[AudioAnalysis::kBandCount] = {}, sideE[AudioAnalysis::kBandCount] = {};
82 double m[AudioAnalysis::kBandCount], s[AudioAnalysis::kBandCount];
83
84 const uint64_t window = std::max<uint64_t>(1, static_cast<uint64_t>(std::llround(rate * StereoDetail::kWindowSeconds)));
85 double ll = 0, rr = 0, lr = 0;
86 for (uint64_t f = 0; f < frames; ++f) {
87 const double l = sanitize(x[f * 2]), r = sanitize(x[f * 2 + 1]);
88 mid.split((l + r) * 0.5, m);
89 side.split((l - r) * 0.5, s);
90 for (size_t b = 0; b < AudioAnalysis::kBandCount; ++b) {
91 midE[b] += m[b] * m[b];
92 sideE[b] += s[b] * s[b];
93 }
94 ll += l * l; rr += r * r; lr += l * r;
95 if ((f + 1) % window == 0 || f + 1 == frames) {
96 const bool full = (f + 1) % window == 0;
97 if (full || ll > 0.0) {
98 const double c = (ll > 0.0 && rr > 0.0) ? std::clamp(lr / std::sqrt(ll * rr), -1.0, 1.0) : 0.0;
99 if (full) {
100 const double startSec = static_cast<double>(out.correlationOverTime.size()) * StereoDetail::kWindowSeconds;
101 out.correlationOverTime.push_back(static_cast<float>(c));
102 if (ll > 0.0 && rr > 0.0 && c < StereoDetail::kMonoProblemCorrelation) out.monoProblemStartsSec.push_back(startSec);
103 }
104 }
105 ll = rr = lr = 0.0;
106 }
107 }
108 for (size_t b = 0; b < AudioAnalysis::kBandCount; ++b) {
109 const double total = midE[b] + sideE[b];
110 out.bandWidth[b] = total > 0.0 ? sideE[b] / total : 0.0;
111 }
112 return out;
113}

115std::vector<Section> detectSections(const float* x, uint64_t frames, uint32_t channels, uint32_t rate) {
116 std::vector<Section> sections;
117 if (!x || frames == 0 || channels == 0 || rate == 0) return sections;
118
119 // Features per half-second block: level and the five band shares of the mono mix.
120 const uint64_t block = std::max<uint64_t>(1, rate / 2);
121 const size_t blocks = static_cast<size_t>((frames + block - 1) / block);
122 struct Feature { double v[1 + AudioAnalysis::kBandCount]; };
123 std::vector<Feature> features(blocks);
124 std::vector<double> blockPeak(blocks, 0.0), blockEnergy(blocks, 0.0);
125
126 BandSplitter splitter(rate);
127 double bands[AudioAnalysis::kBandCount];
128 double bandE[AudioAnalysis::kBandCount] = {};
129 for (uint64_t f = 0; f < frames; ++f) {
130 double mono = 0.0, peak = 0.0;
131 for (uint32_t c = 0; c < channels; ++c) {
132 const double v = sanitize(x[f * channels + c]);
133 mono += v; peak = std::max(peak, std::fabs(v));
134 }
135 mono /= channels;
136 const size_t b = static_cast<size_t>(f / block);
137 blockPeak[b] = std::max(blockPeak[b], peak);
138 blockEnergy[b] += mono * mono;
139 splitter.split(mono, bands);
140 for (size_t k = 0; k < AudioAnalysis::kBandCount; ++k) bandE[k] += bands[k] * bands[k];
141 if ((f + 1) % block == 0 || f + 1 == frames) {
142 const uint64_t n = (f % block) + 1;
143 double total = 0.0;
144 for (double e : bandE) total += e;
145 features[b].v[0] = toDb(std::sqrt(blockEnergy[b] / static_cast<double>(n))) / 20.0; // ~ scaled so 20 dB = 1
146 for (size_t k = 0; k < AudioAnalysis::kBandCount; ++k) features[b].v[1 + k] = total > 0.0 ? bandE[k] / total : 0.0;
147 for (double& e : bandE) e = 0.0;
148 }
149 }
150
151 // Novelty: how different the four blocks after a point are from the four before it.
152 constexpr size_t kSide = 4;
153 constexpr double kThreshold = 0.35;
154 constexpr size_t kMinSpacing = 8; // blocks (4 s)
155 std::vector<double> novelty(blocks, 0.0);
156 for (size_t i = kSide; i + kSide <= blocks; ++i) {
157 double dist = 0.0;
158 for (size_t k = 0; k < 1 + AudioAnalysis::kBandCount; ++k) {
159 double before = 0.0, after = 0.0;
160 for (size_t j = 0; j < kSide; ++j) { before += features[i - 1 - j].v[k]; after += features[i + j].v[k]; }
161 const double d = (after - before) / kSide;
162 dist += d * d;
163 }
164 novelty[i] = std::sqrt(dist);
165 }
166 std::vector<size_t> cuts;
167 for (size_t i = kSide; i + kSide <= blocks; ++i) {
168 if (novelty[i] < kThreshold) continue;
169 const size_t lo = i >= kSide ? i - kSide : 0, hi = std::min(blocks, i + kSide + 1);
170 if (novelty[i] < *std::max_element(novelty.begin() + static_cast<long>(lo), novelty.begin() + static_cast<long>(hi))) continue;
171 if (!cuts.empty() && i - cuts.back() < kMinSpacing) continue;
172 cuts.push_back(i);
173 }
174
175 const double blockSec = static_cast<double>(block) / rate;
176 const double totalSec = static_cast<double>(frames) / rate;
177 std::vector<size_t> edges{0};
178 edges.insert(edges.end(), cuts.begin(), cuts.end());
179 edges.push_back(blocks);
180 for (size_t e = 0; e + 1 < edges.size(); ++e) {
181 Section s;
182 s.startSec = static_cast<double>(edges[e]) * blockSec;
183 s.endSec = e + 2 == edges.size() ? totalSec : static_cast<double>(edges[e + 1]) * blockSec;
184 double energy = 0.0, peak = 0.0;
185 uint64_t n = 0;
186 for (size_t b = edges[e]; b < edges[e + 1]; ++b) {
187 energy += blockEnergy[b];
188 peak = std::max(peak, blockPeak[b]);
189 n += std::min<uint64_t>(block, frames - static_cast<uint64_t>(b) * block);
190 }
191 s.rmsDb = n ? toDb(std::sqrt(energy / static_cast<double>(n))) : AudioAnalysis::kSilenceDb;
192 s.peakDb = toDb(peak);
193 s.crestDb = (s.peakDb > AudioAnalysis::kSilenceDb && s.rmsDb > AudioAnalysis::kSilenceDb) ? s.peakDb - s.rmsDb : 0.0;
194 sections.push_back(s);
195 }
196 return sections;
197}

234// In-place iterative radix-2 FFT of n (a power of two) complex values.
235void fft(std::vector<double>& re, std::vector<double>& im) {
236 const size_t n = re.size();
237 for (size_t i = 1, j = 0; i < n; ++i) {
238 size_t bit = n >> 1;
239 for (; j & bit; bit >>= 1) j ^= bit;
240 j ^= bit;
241 if (i < j) { std::swap(re[i], re[j]); std::swap(im[i], im[j]); }
242 }
243 for (size_t len = 2; len <= n; len <<= 1) {
244 const double ang = -2.0 * kPi / static_cast<double>(len);
245 const double wr = std::cos(ang), wi = std::sin(ang);
246 for (size_t i = 0; i < n; i += len) {
247 double cr = 1.0, ci = 0.0;
248 for (size_t k = 0; k < len / 2; ++k) {
249 const size_t a = i + k, b = i + k + len / 2;
250 const double tr = re[b] * cr - im[b] * ci, ti = re[b] * ci + im[b] * cr;
251 re[b] = re[a] - tr; im[b] = im[a] - ti;
252 re[a] += tr; im[a] += ti;
253 const double nr = cr * wr - ci * wi;
254 ci = cr * wi + ci * wr;
255 cr = nr;
256 }
257 }
258 }

262Spectrum analyzeSpectrum(const float* x, uint64_t frames, uint32_t channels, uint32_t rate) {
263 Spectrum out;
264 out.levelDb.fill(AudioAnalysis::kSilenceDb);
265 constexpr size_t N = 8192;
266 if (!x || channels == 0 || rate == 0 || frames < N) return out;
267 std::vector<double> window(N);
268 for (size_t i = 0; i < N; ++i) window[i] = 0.5 - 0.5 * std::cos(2.0 * kPi * static_cast<double>(i) / (N - 1));
269 // Windows are spread evenly, at most about 300, so a long song costs the same as a short one.
270 const uint64_t maxStart = frames - N;
271 const size_t windows = static_cast<size_t>(std::min<uint64_t>(300, maxStart / (N / 2) + 1));
272 std::vector<double> power(N / 2 + 1, 0.0), re(N), im(N);
273 for (size_t w = 0; w < windows; ++w) {
274 const uint64_t start = windows > 1 ? maxStart * w / (windows - 1) : 0;
275 for (size_t i = 0; i < N; ++i) {
276 double mono = 0.0;
277 for (uint32_t c = 0; c < channels; ++c) mono += sanitize(x[(start + i) * channels + c]);
278 re[i] = mono / channels * window[i];
279 im[i] = 0.0;
280 }
281 fft(re, im);
282 for (size_t k = 0; k <= N / 2; ++k) power[k] += re[k] * re[k] + im[k] * im[k];
283 }
284 // Each band is the power in its third-octave range (edges a sixth of an octave either side of the centre),
285 // as a level per Hz so wide bands do not read louder than narrow ones.
286 const double binHz = static_cast<double>(rate) / N;
287 const auto& centers = Spectrum::centersHz();
288 for (size_t b = 0; b < centers.size(); ++b) {
289 const double lo = centers[b] / std::pow(2.0, 1.0 / 6.0), hi = centers[b] * std::pow(2.0, 1.0 / 6.0);
290 if (hi >= rate * 0.5) continue;
291 double sum = 0.0;
292 size_t n = 0;
293 for (size_t k = static_cast<size_t>(std::ceil(lo / binHz)); k < N / 2 && static_cast<double>(k) * binHz < hi; ++k) { sum += power[k]; ++n; }
294 if (n > 0 && sum > 0.0) out.levelDb[b] = 10.0 * std::log10(sum / static_cast<double>(n) / static_cast<double>(windows));
295 }
296 return out;
297}
