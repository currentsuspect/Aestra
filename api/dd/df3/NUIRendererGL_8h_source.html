250 struct KerningCacheKey {
251 FT_Face face = nullptr;
252 uint32_t previousGlyph = 0;
253 uint32_t currentGlyph = 0;
254
255 bool operator==(const KerningCacheKey& other) const {
256 return face == other.face &&
257 previousGlyph == other.previousGlyph &&
258 currentGlyph == other.currentGlyph;
259 }
260 };
261
262 struct KerningCacheKeyHash {
263 size_t operator()(const KerningCacheKey& key) const noexcept {
264 size_t hash = std::hash<uintptr_t>{}(reinterpret_cast<uintptr_t>(key.face));
265 hash ^= std::hash<uint32_t>{}(key.previousGlyph) + 0x9e3779b9u + (hash << 6u) + (hash >> 2u);
266 hash ^= std::hash<uint32_t>{}(key.currentGlyph) + 0x9e3779b9u + (hash << 6u) + (hash >> 2u);
267 return hash;
268 }
269 };

418 // Text measurement cache (LRU-style with max entries)
419 struct TextMeasurementKey {
420 std::string text;
421 float fontSize;
422 bool operator==(const TextMeasurementKey& o) const {
423 return text == o.text && std::abs(fontSize - o.fontSize) < 0.01f;
424 }
425 };
426 struct TextMeasurementKeyHash {
427 size_t operator()(const TextMeasurementKey& k) const {
428 size_t h1 = std::hash<std::string>{}(k.text);
429 size_t h2 = std::hash<float>{}(k.fontSize);
430 return h1 ^ (h2 << 1);
431 }
432 };
