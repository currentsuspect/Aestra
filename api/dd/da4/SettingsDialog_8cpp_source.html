45void SettingsDialog::show() {
46 m_visible = true;
47 setVisible(true);
48
49 // Ensure we fill the parent window (Modal Overlay)
50 if (getParent()) {
51 auto parentBounds = getParent()->getBounds();
52 if (parentBounds.width > 0 && parentBounds.height > 0) {
53 setBounds(parentBounds); // Force resize to parent
54 }
55 }
56
57 // Center on screen
58 auto componentBounds = getBounds();
59 if (componentBounds.width > 0 && componentBounds.height > 0) {
60 m_dialogBounds.x = componentBounds.x + (componentBounds.width - m_dialogBounds.width) / 2;
61 m_dialogBounds.y = componentBounds.y + (componentBounds.height - m_dialogBounds.height) / 2;
62 }
63
64 layoutComponents();
65 if (m_activePage) m_activePage->onShow();
66}

75void SettingsDialog::addPage(std::shared_ptr<ISettingsPage> page) {
76 if (!page) return;

97void SettingsDialog::setActivePage(const std::string& pageID) {
98 auto it = m_pages.find(pageID);
99 if (it != m_pages.end()) {
100 if (m_activePage) m_activePage->onHide();
101
102 m_activePage = it->second;
103 m_activePageId = pageID;
104 m_activePage->onShow();
105
106 // Update selection state
107 for (auto& item : m_sidebarItems) {
108 item.active = (item.id == pageID);
109 }
110 layoutComponents();
111 }

114void SettingsDialog::layoutComponents() {
115 if (!m_visible) return;
116
117 float padding = 20.0f;
118 float sidebarWidth = 220.0f;
119 float footerHeight = 60.0f;
120 float titleHeight = 32.0f;
121
122 // Sidebar
123 m_sidebarBounds = AestraUI::NUIRect(
124 m_dialogBounds.x,
125 m_dialogBounds.y + titleHeight,
126 sidebarWidth,
127 m_dialogBounds.height - titleHeight
128 );
129
130 // Content Area
131 m_contentBounds = AestraUI::NUIRect(
132 m_dialogBounds.x + sidebarWidth,
133 m_dialogBounds.y + titleHeight,
134 m_dialogBounds.width - sidebarWidth,
135 m_dialogBounds.height - titleHeight - footerHeight
136 );
137
138 // Close button — matches AestraPanelWindow chrome (24x24 at right-28, top+4)
139 m_closeButtonBounds = AestraUI::NUIRect(
140 m_dialogBounds.x + m_dialogBounds.width - 28,
141 m_dialogBounds.y + 4,
142 24, 24
143 );
144
145 // Sidebar items layout
146 float itemY = m_sidebarBounds.y + padding;
147 float itemHeight = 36.0f;
148 for (auto& item : m_sidebarItems) {
149 item.bounds = AestraUI::NUIRect(m_dialogBounds.x, itemY, sidebarWidth, itemHeight);
150 itemY += itemHeight;
151 }
152
153 // Footer buttons
154 float buttonWidth = 100.0f;
155 float buttonHeight = 32.0f;
156 float buttonY = m_dialogBounds.y + m_dialogBounds.height - 46.0f;
157 float rightX = m_dialogBounds.x + m_dialogBounds.width - padding;
158
159 m_okButton->setBounds(AestraUI::NUIRect(rightX - buttonWidth, buttonY, buttonWidth, buttonHeight));
160 m_cancelButton->setBounds(AestraUI::NUIRect(rightX - buttonWidth*2 - 10, buttonY, buttonWidth, buttonHeight));
161 m_applyButton->setBounds(AestraUI::NUIRect(rightX - buttonWidth*3 - 20, buttonY, buttonWidth, buttonHeight));
162
163 // Active Page
164 if (m_activePage) {
165 m_activePage->setBounds(m_contentBounds);
166 // Pages usually need to layout their own internal components
167 // We'll rely on onResize propagating the bounds change
168 }

194void SettingsDialog::onRender(AestraUI::NUIRenderer& renderer) {
195 if (!m_visible) return;
196
197 auto& theme = AestraUI::NUIThemeManager::getInstance();
198
199 // 1. Dimmed Background Overlay
200 renderer.fillRect(getBounds(), AestraUI::NUIColor(0, 0, 0, 0.5f));
201
202 // 2. Dialog Window Shadow
203 // (Simple drop shadow simulation if renderer supports it, or just dark rect)
204
205 // 3. Dialog Window Background - FORCE correct key to avoid purple fallback
206 // Use rounded rect with standard radius (e.g. 8px or theme radius)
207 float radius = 8.0f; // Standard rounded corner radius
208 renderer.fillRoundedRect(m_dialogBounds, radius, theme.getColor("backgroundPrimary"));
209 renderer.strokeRoundedRect(m_dialogBounds, radius, 1.0f, theme.getColor("borderSubtle"));
210
211 // 4. Title bar background (unified 32px chrome)
212 AestraUI::NUIRect titleBar(m_dialogBounds.x, m_dialogBounds.y, m_dialogBounds.width, 32.0f);
213 renderer.fillRect(titleBar, theme.getColor("backgroundPrimary"));
214 renderer.drawLine({titleBar.x, titleBar.bottom() - 0.5f},
215 {titleBar.right(), titleBar.bottom() - 0.5f},
216 0.5f, theme.getColor("borderSubtle"));
217
218 // 5. Title text — vertically centered via font metrics
219 float titleBaseline = renderer.calculateTextY(titleBar, 12.0f);
220 renderer.drawText("Settings", {m_dialogBounds.x + 12.0f, titleBaseline}, 12.0f,
221 theme.getColor("textSecondary"));
222
223 // 6. Close button — line-drawn X matching AestraPanelWindow chrome
224 if (m_closeButtonHovered) {
225 renderer.fillRoundedRect(m_closeButtonBounds, 6.0f, AestraUI::NUIColor(0xff, 0xd9, 0x5f).withAlpha(0.22f));
226 }
227 float cx = m_closeButtonBounds.x + m_closeButtonBounds.width * 0.5f;
228 float cy = m_closeButtonBounds.y + m_closeButtonBounds.height * 0.5f;
229 float d = 5.0f;
230 AestraUI::NUIColor xColor = m_closeButtonHovered
231 ? AestraUI::NUIColor(0xff, 0xe5, 0x73)
232 : theme.getColor("textDisabled");
233 renderer.drawLine({cx - d, cy - d}, {cx + d, cy + d}, 1.5f, xColor);
234 renderer.drawLine({cx + d, cy - d}, {cx - d, cy + d}, 1.5f, xColor);
235
236 // 7. Sidebar Background
237 renderer.fillRect(m_sidebarBounds, theme.getColor("backgroundSecondary"));
238 renderer.fillRect(AestraUI::NUIRect(m_sidebarBounds.x + m_sidebarBounds.width - 1, m_sidebarBounds.y, 1, m_sidebarBounds.height),
239 theme.getColor("divider"));
240
241 // 8. Sidebar Items
242 for (const auto& item : m_sidebarItems) {
243 AestraUI::NUIColor bg = item.active ? theme.getColor("primary").withAlpha(0.2f) :
244 (item.hovered ? theme.getColor("list.hover") : AestraUI::NUIColor(0,0,0,0));
245
246 if (item.active || item.hovered) {
247 renderer.fillRect(item.bounds, bg);
248 }
249
250 // Active indicator strip
251 if (item.active) {
252 renderer.fillRect(AestraUI::NUIRect(item.bounds.x, item.bounds.y, 3, item.bounds.height), theme.getColor("primary"));
253 }
254
255 renderer.drawText(item.title, AestraUI::NUIPoint(item.bounds.x + 20, item.bounds.y + 8), 14.0f,
256 item.active ? theme.getColor("textSelect") : theme.getColor("text"));
257 }
258
259 // 9. Footer Divider
260 float footerY = m_dialogBounds.y + m_dialogBounds.height - 60.0f;
261 renderer.fillRect(AestraUI::NUIRect(m_dialogBounds.x + 220, footerY, m_dialogBounds.width - 220, 1),
262 theme.getColor("divider"));
263
264 // 10. Blink Effect
265 if (m_blinkAnimation > 0.0f) {
266 renderer.strokeRoundedRect(m_dialogBounds, radius, 2.0f, AestraUI::NUIColor(1.0f, 1.0f, 1.0f, m_blinkAnimation * 0.5f));
267 }
268
269 // Render Children (Page content + Buttons)
270 // We only want to render the active page
271 if (m_activePage) {
272 m_activePage->onRender(renderer);
273 }
274
275 // Render footer buttons
276 m_applyButton->onRender(renderer);
277 m_cancelButton->onRender(renderer);
278 m_okButton->onRender(renderer);
279}

281bool SettingsDialog::onMouseEvent(const AestraUI::NUIMouseEvent& event) {
282 if (!m_visible) return false;
283
284 // Footer buttons interaction
285 if (m_applyButton->onMouseEvent(event)) return true;
286 if (m_cancelButton->onMouseEvent(event)) return true;
287 if (m_okButton->onMouseEvent(event)) return true;
288
289 // Active Page interaction
290 if (m_activePage && m_activePage->onMouseEvent(event)) return true;
291
292 // Sidebar interaction
293 bool sidebarHovered = false;
294 for (auto& item : m_sidebarItems) {
295 if (item.bounds.contains(event.position)) {
296 if (!item.hovered) {
297 item.hovered = true;
298 setDirty(true);
299 }
300 sidebarHovered = true;
301
302 if (event.pressed && event.button == AestraUI::NUIMouseButton::Left) {
303 setActivePage(item.id);
304 return true;
305 }
306 } else {
307 if (item.hovered) {
308 item.hovered = false;
309 setDirty(true);
310 }
311 }
312 }
313
314 // Close button
315 bool closeHovered = m_closeButtonBounds.contains(event.position);
316 if (closeHovered != m_closeButtonHovered) {
317 m_closeButtonHovered = closeHovered;
318 setDirty(true);
319 }
320
321 if (event.pressed && event.button == AestraUI::NUIMouseButton::Left) {
322 if (closeHovered) {
323 hide();
324 return true;
325 }
326
327 // Click outside dialog?
328 if (!m_dialogBounds.contains(event.position)) {
329 m_blinkAnimation = 1.0f;
330 setDirty(true);
331 return true;
332 }
333 }
334
335 return true; // Consume all events to stay modal
336}
