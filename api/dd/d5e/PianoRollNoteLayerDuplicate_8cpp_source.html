12PianoRollNoteLayer::duplicateLikeFl(const std::optional<std::pair<double, double>>& ruler) {
13 namespace Dup = PianoRollDuplicate;
14 // The step follows the snap grid; with snap off it is the selection's own length.
15 const double snapStep = snap_ == SnapGrid::None ? 0.0 : MusicTheory::getSnapDuration(snap_);
16 const auto plan = Dup::plan(notes_, ruler, snapStep);
17 if (!plan) return std::nullopt;
18
19 const auto made = Dup::copies(notes_, *plan);
20 // The pattern has to hold the copies, and in range mode the highlight that moves onto them.
21 double needed = Dup::endOf(made);
22 if (plan->rangeMode) needed = std::max(needed, plan->to.second);
23 if (needed > 0.0 && onNeedPatternEnd_) onNeedPatternEnd_(needed);
24
25 if (!made.empty()) {
26 for (auto& n : notes_) n.selected = false;
27 notes_.insert(notes_.end(), made.begin(), made.end());
28 commitNotes(); // one step for the whole press
29 }
30 repaint();
31 return plan->rangeMode ? std::optional<std::pair<double, double>>(plan->to) : std::nullopt;
32}
