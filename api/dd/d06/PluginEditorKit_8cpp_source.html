15void drawArc(NUIRenderer& renderer, NUIPoint center, float radius, float startAngle, float endAngle, float thickness,
16 NUIColor color) {
17 if (endAngle < startAngle)
18 std::swap(startAngle, endAngle);
19 if (endAngle - startAngle <= 0.001f)
20 return;
21 std::array<NUIPoint, 49> pts{};
22 const float div = static_cast<float>(pts.size() - 1);
23 for (size_t i = 0; i < pts.size(); ++i) {
24 const float t = static_cast<float>(i) / div;
25 const float a = startAngle + (endAngle - startAngle) * t;
26 pts[i] = {center.x + std::cos(a) * radius, center.y + std::sin(a) * radius};
27 }
28 renderer.drawPolyline(pts.data(), static_cast<int>(pts.size()), thickness, color);
29}

31void drawKnob(NUIRenderer& renderer, const NUIRect& rect, float value, const char* label, const std::string& valueText,
32 NUIColor accent, NUIColor insetSurface, const KnobLook& look) {
33 auto& theme = NUIThemeManager::getInstance();
34 const NUIPoint c = rect.center();
35 const float r = rect.width * 0.5f - 4.0f;
36 const float angle = kKnobStart + value * kKnobSweep;
37 const float arcWidth = look.large ? 4.0f : 3.0f;
38
39 renderer.fillCircle(c, r + 4.0f, insetSurface);
40 renderer.strokeCircle(c, r + 4.0f, 1.0f, editorInk(0.060f));
41
42 drawArc(renderer, c, r - 3.0f, kKnobStart, kKnobStart + kKnobSweep, arcWidth, editorNeutral(0.199f, 1.0f));
43 if (look.bipolar) {
44 // Fill from the top-centre detent toward the value. Always the thin arc:
45 // that is what every bipolar knob drew before the kit existed.
46 const float centerAngle = kKnobStart + 0.5f * kKnobSweep;
47 drawArc(renderer, c, r - 3.0f, std::min(centerAngle, angle), std::max(centerAngle, angle), 3.0f,
48 accent.withAlpha(0.92f));
49 } else {
50 drawArc(renderer, c, r - 3.0f, kKnobStart, angle, arcWidth, accent.withAlpha(0.92f));
51 }
52
53 const float needleLen = r - (look.large ? 14.0f : look.smallNeedleInset);
54 const NUIPoint tip(c.x + std::cos(angle) * needleLen, c.y + std::sin(angle) * needleLen);
55 renderer.drawLine(c, tip, 2.0f, accent.withAlpha(0.85f));
56 renderer.fillCircle(tip, look.large ? 3.5f : 2.5f, accent);
57
58 const float wellR = r * (look.large ? 0.34f : look.smallWellRatio);
59 renderer.fillCircle(c, wellR, editorNeutral(0.045f, 0.96f));
60 renderer.strokeCircle(c, wellR, 1.2f, accent.withAlpha(0.36f));
61
62 renderer.drawTextCentered(label, NUIRect(rect.x, rect.bottom() + 4.0f, rect.width, 14.0f), 10.5f,
63 theme.getColor("textPrimary").withAlpha(0.95f));
64 const float pad = look.valueLabelPad;
65 renderer.drawTextCentered(valueText, NUIRect(rect.x - pad, rect.bottom() + 19.0f, rect.width + pad * 2.0f, 13.0f),
66 look.large ? 10.0f : 9.0f, accent.withAlpha(0.96f));
67}
