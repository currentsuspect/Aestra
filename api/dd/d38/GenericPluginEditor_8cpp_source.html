28void GenericPluginEditor::buildParameterWidgets() {
29 if (!m_instance) return;
30
31 m_parameters.clear();
32
33 auto allParams = m_instance->getParameters();
34
35 for (const auto& info : allParams) {
36 // Skip read-only or bypass parameters
37 if (info.isReadOnly || info.isBypass) continue;
38
39 ParameterWidget widget;
40 widget.parameterId = info.id;
41 widget.name = info.name;
42 widget.shortName = info.shortName.empty() ? info.name : info.shortName;
43 widget.normalizedValue = m_instance->getParameter(info.id);
44 widget.minValue = info.minValue;
45 widget.maxValue = info.maxValue;
46 widget.stepCount = info.stepCount;
47
48 m_parameters.push_back(widget);
49 }
50
51 layoutParameters();
52}

54void GenericPluginEditor::layoutParameters() {
55 float y = AestraPanelWindow::TITLE_BAR_H + PADDING - m_scrollOffset;
56
57 for (auto& p : m_parameters) {

62 x += LABEL_WIDTH + PADDING;

64 // Slider
65 p.sliderBounds = NUIRect(x, y + PARAMETER_HEIGHT * 0.35f,
66 SLIDER_WIDTH, PARAMETER_HEIGHT * 0.3f);
67 x += SLIDER_WIDTH + PADDING;
68
69 // Value display
70 p.valueBounds = NUIRect(x, y, VALUE_WIDTH, PARAMETER_HEIGHT);
71
72 y += PARAMETER_HEIGHT + 6.0f;
73 }

80void GenericPluginEditor::drawContent(NUIRenderer& renderer, const NUIRect& contentRect) {
81 auto bounds = getBounds();
82 auto& theme = NUIThemeManager::getInstance();
83
84 if (m_parameters.empty()) {
85 std::string msg = "No editable parameters";
86 renderer.drawText(msg,
87 {bounds.x + bounds.width * 0.5f - 60,
88 bounds.y + AestraPanelWindow::TITLE_BAR_H
89 + (bounds.height - AestraPanelWindow::TITLE_BAR_H) * 0.5f},
90 11.0f, theme.getColor("textSecondary").withAlpha(0.5f));
91 } else {
92 float viewportTop = AestraPanelWindow::TITLE_BAR_H;
93 float viewportBottom = viewportTop + contentRect.height;
94
95 for (size_t i = 0; i < m_parameters.size(); ++i) {
96 const auto& p = m_parameters[i];
97
98 // Skip if parameter is outside viewport
99 if (p.labelBounds.y + PARAMETER_HEIGHT < viewportTop ||
100 p.labelBounds.y > viewportBottom) {
101 continue;
102 }
103
104 bool hovered = (static_cast<int>(i) == m_hoveredParameter);
105 drawParameter(renderer, p, hovered);
106 }
107 }

110void GenericPluginEditor::drawParameter(NUIRenderer& renderer, const ParameterWidget& p, bool hovered) {
111 auto& theme = NUIThemeManager::getInstance();
112 auto bounds = getBounds();
113
114 float offsetX = bounds.x;
115 float offsetY = bounds.y;
116
117 NUIRect rowRect(offsetX + p.labelBounds.x - 6.0f,
118 offsetY + p.labelBounds.y + 1.0f,
119 LABEL_WIDTH + SLIDER_WIDTH + VALUE_WIDTH + PADDING * 3.0f + 12.0f,
120 PARAMETER_HEIGHT - 2.0f);
121 renderer.fillRoundedRect(rowRect, 9.0f,
122 hovered || p.isDragging ? editorNeutral(0.181f, 0.92f)
123 : editorNeutral(0.111f, 0.74f));
124 renderer.strokeRoundedRect(rowRect, 9.0f, 1.0f,
125 hovered || p.isDragging ? theme.getColor("accentPrimary").withAlpha(0.30f)
126 : editorInk(0.05f));
127
128 float labelY = offsetY + p.sliderBounds.y + (p.sliderBounds.height * 0.5f) - 5.0f;
129 renderer.drawText(p.shortName,
130 {offsetX + p.labelBounds.x, labelY},
131 11.0f, theme.getColor("textPrimary").withAlpha(0.90f));
132
133 float trackH = 8.0f;
134 float trackY = offsetY + p.sliderBounds.y + (p.sliderBounds.height - trackH) * 0.5f;
135 NUIRect track(offsetX + p.sliderBounds.x, trackY,
136 p.sliderBounds.width, trackH);
137
138 renderer.fillRoundedRect(track, 3.0f, editorNeutral(0.039f, 0.78f));
139
140 float fillWidth = track.width * p.normalizedValue;
141 if (fillWidth > 0) {
142 NUIRect fillRect(track.x, track.y, fillWidth, track.height);
143 NUIColor accent = theme.getColor("accentPrimary");
144
145 if (hovered || p.isDragging) {
146 renderer.fillRoundedRect(fillRect, 3.0f, accent.withAlpha(0.95f));
147 renderer.fillRoundedRect({fillRect.x, fillRect.y - 1, fillRect.width, fillRect.height + 2},
148 4.0f, accent.withAlpha(0.22f));
149 } else {
150 renderer.fillRoundedRect(fillRect, 3.0f, accent.withAlpha(0.78f));
151 }
152 }
153
154 float thumbSize = 12.0f;
155 float thumbX = track.x + fillWidth - (thumbSize * 0.5f);
156 float thumbY = track.center().y - (thumbSize * 0.5f);
157 thumbX = std::clamp(thumbX, track.x - thumbSize * 0.5f, track.right() - thumbSize * 0.5f);
158 NUIRect thumbRect(thumbX, thumbY, thumbSize, thumbSize);
159
160 if (hovered || p.isDragging) {
161 renderer.fillRoundedRect(thumbRect, thumbSize * 0.5f, editorInk(1.0f));
162 renderer.strokeRoundedRect(thumbRect, thumbSize * 0.5f, 1.5f, theme.getColor("accentPrimary"));
163 } else {
164 float idleSize = 8.0f;
165 float offset = (thumbSize - idleSize) * 0.5f;
166 NUIRect idleThumb = {thumbRect.x + offset, thumbRect.y + offset, idleSize, idleSize};
167 renderer.fillRoundedRect(idleThumb, idleSize * 0.5f, NUIColor(0.9f, 0.9f, 0.9f, 0.9f));
168 }
169
170 std::string valueStr = formatParameterValue(p);
171 renderer.drawText(valueStr,
172 {offsetX + p.valueBounds.x, labelY},
173 11.0f, theme.getColor("textPrimary"));
174}

176bool GenericPluginEditor::onMouseEvent(const NUIMouseEvent& event) {
177 if (!isVisible()) return false;
178
179 // Let base class handle title bar / close button / window drag first
180 if (AestraPanelWindow::onMouseEvent(event)) {
181 return true;
182 }
183
184 auto bounds = getBounds();
185 const bool isDraggingParameter = std::any_of(
186 m_parameters.begin(), m_parameters.end(),
187 [](const ParameterWidget& p) { return p.isDragging; });
188
189 if (!bounds.contains(event.position) && !isDraggingWindow() && !isDraggingParameter) {
190 return false;
191 }
192
193 // Mouse wheel scrolling
194 if (event.wheelDelta != 0) {
195 m_scrollOffset -= event.wheelDelta * 15.0f;
196
197 // Clamp scroll
198 float contentHeight = m_parameters.size() * PARAMETER_HEIGHT + PADDING * 2;
199 float viewHeight = bounds.height - AestraPanelWindow::TITLE_BAR_H;
200 m_scrollOffset = std::clamp(m_scrollOffset, 0.0f, std::max(0.0f, contentHeight - viewHeight));
201
202 layoutParameters();
203 repaint();
204 return true;
205 }
206
207 // Update hover (only check parameters in content area)
208 int paramIndex = hitTestParameter(event.position.x, event.position.y);
209 if (paramIndex != m_hoveredParameter) {
210 m_hoveredParameter = paramIndex;
211 repaint();
212 }
213
214 // Handle slider dragging
215 if (event.button == NUIMouseButton::Left) {
216 if (event.pressed && paramIndex >= 0) {
217 // Start dragging
218 m_parameters[paramIndex].isDragging = true;
219 m_parameters[paramIndex].dragStartX = event.position.x;
220 m_parameters[paramIndex].dragStartValue = m_parameters[paramIndex].normalizedValue;
221 return true;
222 } else if (!event.pressed) {
223 // Stop all dragging
224 for (auto& p : m_parameters) {
225 p.isDragging = false;
226 }
227 return true;
228 }
229 }
230
231 // Update dragging parameters
232 for (size_t i = 0; i < m_parameters.size(); ++i) {
233 if (m_parameters[i].isDragging) {
234 float deltaX = event.position.x - m_parameters[i].dragStartX;
235 float deltaValue = deltaX / SLIDER_WIDTH;
236 float newValue = std::clamp(m_parameters[i].dragStartValue + deltaValue, 0.0f, 1.0f);
237
238 updateParameterValue(static_cast<int>(i), newValue);
239 repaint();
240 return true;
241 }
242 }
243
244 return true; // Always consume events within window
245}

247int GenericPluginEditor::hitTestParameter(float x, float y) {
248 auto bounds = getBounds();
249
250 // Convert global mouse position to local coordinates
251 float localX = x - bounds.x;
252 float localY = y - bounds.y;
253
254 for (size_t i = 0; i < m_parameters.size(); ++i) {
255 const auto& p = m_parameters[i];
256 // Check if mouse is over slider area (both in local coords now)
257 if (p.sliderBounds.contains(NUIPoint(localX, localY))) {
258 return static_cast<int>(i);
259 }
260 }
261 return -1;
262}

264void GenericPluginEditor::updateParameterValue(int paramIndex, float normalizedValue) {
265 if (paramIndex < 0 || paramIndex >= static_cast<int>(m_parameters.size())) return;
266
267 auto& p = m_parameters[paramIndex];
268
269 // Handle stepped parameters
270 if (p.stepCount > 0) {
271 float stepped = std::round(normalizedValue * p.stepCount) / p.stepCount;
272 p.normalizedValue = stepped;
273 } else {
274 p.normalizedValue = normalizedValue;
275 }
276
277 // Update plugin parameter
278 if (m_instance) {
279 m_instance->setParameter(p.parameterId, p.normalizedValue);
280 }
281}

283std::string GenericPluginEditor::formatParameterValue(const ParameterWidget& p) {
284 // Try to get display string from plugin first
285 if (m_instance) {
286 std::string display = m_instance->getParameterDisplay(p.parameterId);
287 if (!display.empty()) {
288 return display;
289 }
290 }
291
292 // Fallback: format the actual value
293 float actualValue = p.minValue + p.normalizedValue * (p.maxValue - p.minValue);
294
295 std::ostringstream oss;
296 oss << std::fixed << std::setprecision(2) << actualValue;
297 return oss.str();
298}
