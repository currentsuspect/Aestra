56void AestraApp::createNewProject() {
57 if (m_content && m_content->getTrackManager()) m_content->getTrackManager()->stop();
58 // Leave the old session: discards its unsaved/unreferenced takes.
59 cleanupUnreferencedRecordings();
60 if (m_content) m_content->resetToDefaultProject();
61 clearProjectLoadReport();
62 m_documentState.startUntitled(autosavePathOrEmpty());
63 reinitAutosaveManager();
64 syncRecordingProjectPath(m_content, m_documentState.canonicalPath());
65 m_lastWindowTitle.clear();
66 updateWindowTitle();
67 Log::info("New project created");
68}

70ProjectSerializer::LoadResult AestraApp::openProjectFromPath(const std::string& path) {
71 // Old session's keeper path: captured BEFORE the load, so discarded-take
72 // cleanup (which runs only on a successful switch) keeps exactly the previous
73 // project's recordings. The old session's redo history may still require its
74 // WAVs if the new project fails to load, so cleanup is not unconditional.
75 const std::string oldKeeperPath = m_documentState.canonicalPath();
76 auto result = loadProjectFromPath(path);
77 if (result.ok) {
78 cleanupUnreferencedRecordings(oldKeeperPath);
79 } else {
80 Log::error("Failed to load project: " + path + " (" + result.errorMessage + ")");
81 }
82 return result;
83}

96 // --- project.save -------------------------------------------------------
97 //
98 // No path means "save where this project already lives". An untitled project
99 // has nowhere to live, and that is reported rather than guessed: inventing a
100 // path in the user's home directory is how an agent ends up scattering
101 // projects where nobody will find them.
102 {
103 HostVerbSpec spec;
104 spec.name = "project.save";
105 spec.domain = HostVerbDomain::Project;
106 spec.affinity = HostThreadAffinity::HostUiThread;
107 spec.mutates = true;
108 spec.description =
109 "Save the project to disk. With 'path', write there and make it the "
110 "project's canonical location. Without one, save to the current "
111 "canonical path, which fails for a project that has never been saved. "
112 "Returns the path written and whether it is now the canonical path.";
113 HostVerbArg path;
114 path.name = "path";
115 path.type = FlagType::String;
116 path.required = false;
117 path.description = "Absolute or relative project file path (.aes). Omit to re-save in place.";
118 spec.args.push_back(path);
119
120 registerOrLog(std::move(spec), [this](const JSON& args) -> HostVerbResult {
121 const bool wantsPath = args.has("path") && args["path"].isString() &&
122 !args["path"].asString().empty();
123 const std::string target = wantsPath ? args["path"].asString()
124 : m_documentState.canonicalPath();
125
126 if (target.empty()) {
127 return HostVerbResult::failure(
128 "no_canonical_path",
129 "this project has never been saved, so there is nowhere to save it; "
130 "pass an explicit 'path'");
131 }
132
133 // A pathless save is Save, not Save As, so it inherits Save's refusal.
134 // requiresSaveAs() is true when the canonical file is empty OR when the
135 // project loaded with an integrity Mismatch and protectCanonicalFromOverwrite()
136 // was set — in which case the app deliberately refuses to overwrite the
137 // original on disk, because it is not the project that was just opened.
138 // Ordinary Save checks this in AestraApp::saveProject(); without the same
139 // check here the verb was SAFER for a human than for an agent, which is
140 // exactly backwards. An explicit 'path' is the Save As route and stays
141 // allowed: writing elsewhere cannot destroy the protected original.
142 if (!wantsPath && m_documentState.requiresSaveAs()) {
143 return HostVerbResult::failure(
144 "save_as_required",
145 "this project must not overwrite its canonical file "
146 "(it loaded with an integrity mismatch, so the file on disk is not "
147 "the project now open); pass an explicit 'path' to save elsewhere");
148 }
149
150 // establishCanonical only when the caller named a path: re-saving in
151 // place must not re-point the document at itself, which is the
152 // caller's decision to make, not a side effect of a verb.
153 const bool established = saveProjectToPath(target, wantsPath);
154 if (!established) {
155 return HostVerbResult::failure("save_failed",
156 "the project could not be written to " + target);
157 }
158
159 JSON result = JSON::object();
160 result.set("path", JSON(m_documentState.canonicalPath()));
161 result.set("untitled", JSON(m_documentState.canonicalPath().empty()));
162 return HostVerbResult::success(result);
163 });
164 }
165
166 // --- project.open -------------------------------------------------------
167 //
168 // A failed load still produces a full diagnostic report — missing samples,
169 // missing plugins, plugins that rejected their state — and that report is
170 // published rather than returned here, because HostVerbResult::failure
171 // carries no payload. An agent that gets `load_failed` should then call
172 // get_project_load_report, which is the verb that exists to answer "what
173 // actually went wrong".
174 {
175 HostVerbSpec spec;
176 spec.name = "project.open";
177 spec.domain = HostVerbDomain::Project;
178 spec.affinity = HostThreadAffinity::HostUiThread;
179 spec.mutates = true;
180 spec.description =
181 "Open a project file, replacing the current session. Returns the load "
182 "report on success. On failure the reason is in the error and the full "
183 "diagnostics (missing samples, missing or state-rejecting plugins) are "
184 "available from get_project_load_report.";
185 HostVerbArg path;
186 path.name = "path";
187 path.type = FlagType::String;
188 path.required = true;
189 path.description = "Path to an Aestra project file (.aes).";
190 spec.args.push_back(path);
191
192 registerOrLog(std::move(spec), [this, &service](const JSON& args) -> HostVerbResult {
193 const std::string path = args["path"].asString();
194
195 std::error_code ec;
196 if (!std::filesystem::exists(path, ec)) {
197 // Record the attempt before returning. loadProjectFromPath() is what
198 // normally clears and publishes the report, and this branch never
199 // reaches it — so without this, get_project_load_report would go on
200 // describing some EARLIER load, and an agent that opened a typo'd path
201 // would be told the previous project's load had succeeded.
202 const std::string reason =
203 ec ? "could not read " + path + " (" + ec.message() + ")"
204 : "no such project file: " + path;
205 ProjectSerializer::LoadResult missing;
206 missing.errorMessage = reason;
207 service.setProjectLoadReport(
208 makeMuseProjectLoadReport(missing, MuseProjectLoadOrigin::Canonical));
209 return HostVerbResult::failure("no_such_file", reason);
210 }
211
212 const auto result = openProjectFromPath(path);
213 if (!result.ok) {
214 return HostVerbResult::failure("load_failed", result.errorMessage);
215 }
216
217 JSON payload = makeMuseProjectLoadReport(result, MuseProjectLoadOrigin::Canonical);
218 payload.set("path", JSON(m_documentState.canonicalPath()));
219 return HostVerbResult::success(payload);
220 });
221 }
222
223 // --- project.new --------------------------------------------------------
224 {
225 HostVerbSpec spec;
226 spec.name = "project.new";
227 spec.domain = HostVerbDomain::Project;
228 spec.affinity = HostThreadAffinity::HostUiThread;
229 spec.mutates = true;
230 spec.description =
231 "Discard the current session and start an empty project. Anything not "
232 "saved is lost, exactly as it is when a person picks File > New Project.";
233 registerOrLog(std::move(spec), [this](const JSON&) -> HostVerbResult {
234 const std::string previous = m_documentState.canonicalPath();
235 createNewProject();
236
237 JSON result = JSON::object();
238 // An untitled project has an empty canonical path, and saying so
239 // plainly is more useful than reporting "" as if it were a location.
240 result.set("previousPath", JSON(previous));
241 result.set("untitled", JSON(true));
242 return HostVerbResult::success(result);
243 });
244 }
