57// Extracted from the File menu's inline lambdas verbatim. The menu items now
58// call these, so the agent and the human run the same sequence.
59
60// New and Open used to replace the session without asking, while Quit asked
61// (FD-50). The same dialog now guards all three. Don't Save keeps the autosave:
62// unlike Quit, the work is only being left, and Versions can still reach it.
63void AestraApp::confirmLeaveProject(const std::string& what, std::function<void()> proceed) {
64 auto trackManager = m_content ? m_content->getTrackManager() : nullptr;
65 if (!trackManager || !trackManager->isModified() || !m_windowManager) {
66 proceed();
67 return;
68 }
69 m_autoSaveManager.forceAutosave(); // recoverable even if the app dies with the dialog open
70 auto dialog = m_windowManager->getConfirmationDialog();
71 if (!dialog) {
72 dialog = std::make_shared<Aestra::ConfirmationDialog>();
73 m_windowManager->setConfirmationDialog(dialog);
74 }
75 if (auto* root = m_windowManager->getRootComponent()) {
76 dialog->setBounds(root->getBounds());
77 root->removeChild(dialog);
78 root->addChild(dialog);
79 }
80 const std::string name = m_documentState.displayName();
81 dialog->show("Unsaved Changes", "Save \"" + name + "\" before " + what + "?",
82 [this, proceed = std::move(proceed)](Aestra::DialogResponse response) {
83 if (response == Aestra::DialogResponse::Save) {
84 saveCurrentProject([proceed](bool saved) {
85 if (saved) proceed();
86 });
87 } else if (response == Aestra::DialogResponse::DontSave) {
88 proceed();
89 }
90 });
91}

93void AestraApp::createNewProject() {
94 if (m_content && m_content->getTrackManager()) m_content->getTrackManager()->stop();
95 // Leave the old session: discards its unsaved/unreferenced takes.
96 cleanupUnreferencedRecordings();
97 if (m_content) m_content->resetToDefaultProject();
98 clearProjectLoadReport();
99 // The tempo lives in four places (engine, playlist, the clock notes are
100 // scheduled by, the transport field); a new project resets all of them, the
101 // way applyLoadedProject() sets them from a file. The playhead goes home.
102 constexpr double tempo = Aestra::Audio::kNewProjectTempo;
103 if (m_audioController && m_audioController->getEngine()) {
104 m_audioController->getEngine()->setBPM(static_cast<float>(tempo));
105 m_audioController->getEngine()->setGlobalSamplePos(0);
106 }
107 if (m_content) {
108 m_content->setPluginTempo(static_cast<float>(tempo));
109 if (auto trackManager = m_content->getTrackManager()) {
110 trackManager->getPlaylistModel().setBPM(tempo);
111 trackManager->getTimelineClock().setTempo(tempo);
112 trackManager->getPatternPlaybackEngine().rewindScheduledInstances();
113 trackManager->setPosition(0.0);
114 trackManager->setPlayStartPosition(0.0);
115 trackManager->setModified(false);
116 }
117 if (auto transportBar = m_content->getTransportBar()) {
118 transportBar->setTempo(static_cast<float>(tempo));
119 transportBar->setPosition(0.0);
120 }
121 m_content->refreshProjectViews(); // mixer, pattern browser and Arsenal show the empty project
122 }
123 m_documentState.startUntitled(autosavePathOrEmpty());
124 reinitAutosaveManager();
125 syncRecordingProjectPath(m_content, m_documentState.canonicalPath());
126 m_lastWindowTitle.clear();
127 updateWindowTitle();
128 Log::info("New project created");
129}

131ProjectSerializer::LoadResult AestraApp::openProjectFromPath(const std::string& path) {
132 // Old session's keeper path: captured BEFORE the load, so discarded-take
133 // cleanup (which runs only on a successful switch) keeps exactly the previous
134 // project's recordings. The old session's redo history may still require its
135 // WAVs if the new project fails to load, so cleanup is not unconditional.
136 const std::string oldKeeperPath = m_documentState.canonicalPath();
137 auto result = loadProjectFromPath(path);
138 if (result.ok) {
139 cleanupUnreferencedRecordings(oldKeeperPath);
140 } else {
141 Log::error("Failed to load project: " + path + " (" + result.errorMessage + ")");
142 }
143 return result;
144}

148// Save a Copy: the file picked gets the project as it is now, and the document stays where it was (same file, same
149// saved or unsaved state, same autosave). Choosing the open file itself is an ordinary Save.
150void AestraApp::saveProjectCopy() {
151 Aestra::IPlatformUtils::SaveFileDialogOptions options;
152 options.title = "Save a Copy";
153 options.filter =
154 std::string("Aestra Project\0*.aes\0All Files\0*.*\0", sizeof("Aestra Project\0*.aes\0All Files\0*.*\0") - 1);
155 const std::string current = m_documentState.canonicalPath();
156 if (!current.empty()) {
157 const std::filesystem::path p(current);
158 options.defaultPath = (p.parent_path() / (p.stem().string() + " copy.aes")).string();
159 }
160 options.defaultExtension = "aes";
161
162 const bool started =
163 pickFileAsync([options](Aestra::IPlatformUtils& utils) { return utils.saveFileDialog(options); },
164 [this](const std::string& pickedPath) {
165 if (pickedPath.empty())
166 return;
167 const bool ok = pickedPath == m_documentState.canonicalPath()
168 ? saveProject()
169 : saveProjectToPath(pickedPath, false);
170 if (ok)
171 Log::info("Project copy saved: " + pickedPath);
172 else
173 Log::warning("Could not save a copy to " + pickedPath);
174 });
175 if (!started)
176 Log::warning("Cannot save a copy: a file picker is already open or the platform layer is unavailable");
177}

188 // --- project.save -------------------------------------------------------
189 //
190 // No path means "save where this project already lives". An untitled project
191 // has nowhere to live, and that is reported rather than guessed: inventing a
192 // path in the user's home directory is how an agent ends up scattering
193 // projects where nobody will find them.
194 {
195 HostVerbSpec spec;
196 spec.name = "project.save";
197 spec.domain = HostVerbDomain::Project;
198 spec.affinity = HostThreadAffinity::HostUiThread;
199 spec.mutates = true;
200 spec.description = "Save the project to disk. With 'path', write there and make it the "
201 "project's canonical location. Without one, save to the current "
202 "canonical path, which fails for a project that has never been saved. "
203 "Returns the path written and whether it is now the canonical path. "
204 "With 'copy': true, write 'path' as a copy and stay on the open project.";
205 HostVerbArg path;
206 path.name = "path";
207 path.type = FlagType::String;
208 path.required = false;
209 path.description = "Absolute or relative project file path (.aes). Omit to re-save in place.";
210 spec.args.push_back(path);
211 HostVerbArg copy;
212 copy.name = "copy";
213 copy.type = FlagType::Bool;
214 copy.required = false;
215 copy.description = "Save a Copy: write 'path' and keep the open project where it is.";
216 spec.args.push_back(copy);
217
218 registerOrLog(std::move(spec), [this](const JSON& args) -> HostVerbResult {
219 const bool wantsPath = args.has("path") && args["path"].isString() &&
220 !args["path"].asString().empty();
221 const std::string target = wantsPath ? args["path"].asString()
222 : m_documentState.canonicalPath();
223
224 if (target.empty()) {
225 return HostVerbResult::failure(
226 "no_canonical_path",
227 "this project has never been saved, so there is nowhere to save it; "
228 "pass an explicit 'path'");
229 }
230
231 // A pathless save is Save, not Save As, so it inherits Save's refusal.
232 // requiresSaveAs() is true when the canonical file is empty OR when the
233 // project loaded with an integrity Mismatch and protectCanonicalFromOverwrite()
234 // was set — in which case the app deliberately refuses to overwrite the
235 // original on disk, because it is not the project that was just opened.
236 // Ordinary Save checks this in AestraApp::saveProject(); without the same
237 // check here the verb was SAFER for a human than for an agent, which is
238 // exactly backwards. An explicit 'path' is the Save As route and stays
239 // allowed: writing elsewhere cannot destroy the protected original.
240 if (!wantsPath && m_documentState.requiresSaveAs()) {
241 return HostVerbResult::failure(
242 "save_as_required",
243 "this project must not overwrite its canonical file "
244 "(it loaded with an integrity mismatch, so the file on disk is not "
245 "the project now open); pass an explicit 'path' to save elsewhere");
246 }
247
248 // copy:true is Save a Copy: write the named file, stay on the open one.
249 const bool copy = args.has("copy") && args["copy"].isBool() && args["copy"].asBool();
250 if (copy && (!wantsPath || target == m_documentState.canonicalPath())) {
251 return HostVerbResult::failure("copy_needs_path",
252 "a copy needs an explicit 'path' other than the open project's file");
253 }
254 // establishCanonical only when the caller named a path: re-saving in
255 // place must not re-point the document at itself, which is the
256 // caller's decision to make, not a side effect of a verb.
257 const bool established = saveProjectToPath(target, wantsPath && !copy);
258 if (!established) {
259 return HostVerbResult::failure("save_failed",
260 "the project could not be written to " + target);
261 }
262
263 JSON result = JSON::object();
264 result.set("path", JSON(m_documentState.canonicalPath()));
265 if (copy)
266 result.set("copy", JSON(target));
267 result.set("untitled", JSON(m_documentState.canonicalPath().empty()));
268 return HostVerbResult::success(result);
269 });
270 }
271
272 // --- project.open -------------------------------------------------------
273 //
274 // A failed load still produces a full diagnostic report — missing samples,
275 // missing plugins, plugins that rejected their state — and that report is
276 // published rather than returned here, because HostVerbResult::failure
277 // carries no payload. An agent that gets `load_failed` should then call
278 // get_project_load_report, which is the verb that exists to answer "what
279 // actually went wrong".
280 {
281 HostVerbSpec spec;
282 spec.name = "project.open";
283 spec.domain = HostVerbDomain::Project;
284 spec.affinity = HostThreadAffinity::HostUiThread;
285 spec.mutates = true;
286 spec.description =
287 "Open a project file, replacing the current session. Returns the load "
288 "report on success. On failure the reason is in the error and the full "
289 "diagnostics (missing samples, missing or state-rejecting plugins) are "
290 "available from get_project_load_report.";
291 HostVerbArg path;
292 path.name = "path";
293 path.type = FlagType::String;
294 path.required = true;
295 path.description = "Path to an Aestra project file (.aes).";
296 spec.args.push_back(path);
297
298 registerOrLog(std::move(spec), [this, &service](const JSON& args) -> HostVerbResult {
299 const std::string path = args["path"].asString();
300
301 std::error_code ec;
302 if (!std::filesystem::exists(path, ec)) {
303 // Record the attempt before returning. loadProjectFromPath() is what
304 // normally clears and publishes the report, and this branch never
305 // reaches it — so without this, get_project_load_report would go on
306 // describing some EARLIER load, and an agent that opened a typo'd path
307 // would be told the previous project's load had succeeded.
308 const std::string reason =
309 ec ? "could not read " + path + " (" + ec.message() + ")"
310 : "no such project file: " + path;
311 ProjectSerializer::LoadResult missing;
312 missing.errorMessage = reason;
313 service.setProjectLoadReport(
314 makeMuseProjectLoadReport(missing, MuseProjectLoadOrigin::Canonical));
315 return HostVerbResult::failure("no_such_file", reason);
316 }
317
318 const auto result = openProjectFromPath(path);
319 if (!result.ok) {
320 return HostVerbResult::failure("load_failed", result.errorMessage);
321 }
322
323 JSON payload = makeMuseProjectLoadReport(result, MuseProjectLoadOrigin::Canonical);
324 payload.set("path", JSON(m_documentState.canonicalPath()));
325 return HostVerbResult::success(payload);
326 });
327 }
328
329 // --- project.new --------------------------------------------------------
330 {
331 HostVerbSpec spec;
332 spec.name = "project.new";
333 spec.domain = HostVerbDomain::Project;
334 spec.affinity = HostThreadAffinity::HostUiThread;
335 spec.mutates = true;
336 spec.description =
337 "Discard the current session and start an empty project. Anything not "
338 "saved is lost, exactly as it is when a person picks File > New Project.";
339 registerOrLog(std::move(spec), [this](const JSON&) -> HostVerbResult {
340 const std::string previous = m_documentState.canonicalPath();
341 createNewProject();
342
343 JSON result = JSON::object();
344 // An untitled project has an empty canonical path, and saying so
345 // plainly is more useful than reporting "" as if it were a location.
346 result.set("previousPath", JSON(previous));
347 result.set("untitled", JSON(true));
348 return HostVerbResult::success(result);
349 });
350 }
