78std::vector<float> buildPreviewWaveform(const std::vector<float>& samples, uint32_t channels, size_t targetSize = 128) {
79 std::vector<float> waveform(targetSize, 0.0f);
80 if (samples.empty() || channels == 0 || targetSize == 0) {
81 return waveform;
82 }
83
84 const size_t totalFrames = samples.size() / channels;
85 if (totalFrames == 0) {
86 return waveform;
87 }
88
89 const double framesPerBin = static_cast<double>(totalFrames) / static_cast<double>(targetSize);
90 for (size_t bin = 0; bin < targetSize; ++bin) {
91 const size_t startFrame = static_cast<size_t>(static_cast<double>(bin) * framesPerBin);
92 const size_t endFrame =
93 std::min(totalFrames, static_cast<size_t>(static_cast<double>(bin + 1) * framesPerBin) + 1);
94
95 float peak = 0.0f;
96 for (size_t frame = startFrame; frame < endFrame; ++frame) {
97 float sum = 0.0f;
98 for (uint32_t ch = 0; ch < channels; ++ch) {
99 sum += std::abs(samples[frame * channels + ch]);
100 }
101 peak = std::max(peak, sum / static_cast<float>(channels));
102 }
103 waveform[bin] = std::min(1.0f, peak);
104 }
105
106 return waveform;
107}

109std::vector<float> buildEditorWaveform(const std::vector<float>& samples, uint32_t channels, uint32_t sampleLength,
110 size_t buckets = 1024) {
111 std::vector<float> waveform;
112 if (samples.empty() || channels == 0 || sampleLength == 0 || buckets == 0) {
113 return waveform;
114 }
115
116 const size_t framesPerBucket = std::max<size_t>(1, static_cast<size_t>(sampleLength) / buckets);
117 waveform.reserve(buckets * 2);
118 for (size_t bucket = 0; bucket < buckets; ++bucket) {
119 const size_t startFrame = bucket * framesPerBucket;
120 const size_t endFrame = std::min<size_t>(static_cast<size_t>(sampleLength), startFrame + framesPerBucket);
121 if (startFrame >= endFrame) {
122 waveform.push_back(0.0f);
123 waveform.push_back(0.0f);
124 continue;
125 }
126
127 float minV = 1.0f;
128 float maxV = -1.0f;
129 for (size_t frame = startFrame; frame < endFrame; ++frame) {
130 const size_t idx = frame * channels;
131 const float mono = (channels > 1) ? 0.5f * (samples[idx] + samples[idx + 1]) : samples[idx];
132 minV = std::min(minV, mono);
133 maxV = std::max(maxV, mono);
134 }
135 waveform.push_back(maxV);
136 waveform.push_back(minV);
137 }
138 return waveform;
139}

1219 | (static_cast<uint64_t>(ch->isSoloed()) << 1)
1220 | (static_cast<uint64_t>(ch->isArmed()) << 2)
1221 | (static_cast<uint64_t>(ch->isMonitoringEnabled()) << 3));
