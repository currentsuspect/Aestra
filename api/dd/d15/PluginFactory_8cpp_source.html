21void InProcessPluginFactory::createPluginAsync(const PluginInfo& info,
22 std::function<void(PluginInstancePtr)> callback) {
23 // Current In-Process implementation acts synchronously "simulating" async completion
24 // This allows the API to be async-ready without forcing a thread context switch
25 // that might be unsafe for VST3 initialization (which often requires Main Thread).
26
27 PluginInstancePtr instance = nullptr;
28
29 try {
30 switch (info.format) {
31 case PluginFormat::VST3:
32 instance = createVST3Instance(info);
33 break;
34 case PluginFormat::CLAP:
35 instance = createCLAPInstance(info);
36 break;
37 case PluginFormat::Internal:
38 instance = createInternalInstance(info);
39 break;
40 default:
41 Log::error("Unknown plugin format for: " + info.name);
42 break;
43 }
44 } catch (const std::exception& e) {
45 Log::error("Exception during plugin creation: " + std::string(e.what()));
46 } catch (...) {
47 Log::error("Unknown exception during plugin creation");
48 }
49
50 if (callback) {
51 callback(instance);
52 }
53}

87void OutOfProcessPluginFactory::createPluginAsync(const PluginInfo& info,
88 std::function<void(PluginInstancePtr)> callback) {
89 PluginInstancePtr instance = nullptr;
90
91 if (info.format == PluginFormat::Internal) {
92 Log::warning("[PluginHost] Internal plugin requested through out-of-process factory: " + info.id);
93 } else {
94 auto proxy = std::make_shared<OutOfProcessPluginInstance>(info, m_hostExecutablePath);
95 if (proxy->load()) {
96 instance = std::move(proxy);
97 }
98 }
99
100 if (callback) {
101 callback(instance);
102 }
103}
