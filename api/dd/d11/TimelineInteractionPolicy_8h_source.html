30class TimelineTrackSelection {
31public:
32 void apply(const PlaylistLaneID& laneId, TrackSelectionIntent intent) {
33 if (!laneId.isValid()) {
34 return;
35 }
36
37 if (intent == TrackSelectionIntent::Replace) {
38 m_laneIds.clear();
39 } else if (intent == TrackSelectionIntent::Toggle) {
40 if (m_laneIds.erase(laneId) != 0) {
41 return;
42 }
43 }
44 m_laneIds.insert(laneId);
45 }
46
47 void clear() { m_laneIds.clear(); }
48 bool contains(const PlaylistLaneID& laneId) const { return m_laneIds.find(laneId) != m_laneIds.end(); }
49 size_t size() const { return m_laneIds.size(); }
50 bool empty() const { return m_laneIds.empty(); }
51
52 void selectAll(const std::vector<PlaylistLaneID>& laneIds) {
53 m_laneIds.clear();
54 for (const auto& laneId : laneIds) {
55 if (laneId.isValid()) {
56 m_laneIds.insert(laneId);
57 }
58 }
59 }
60
61 void retainOnly(const std::vector<PlaylistLaneID>& validLaneIds) {
62 const std::unordered_set<PlaylistLaneID> valid(validLaneIds.begin(), validLaneIds.end());
63 for (auto it = m_laneIds.begin(); it != m_laneIds.end();) {
64 it = valid.find(*it) == valid.end() ? m_laneIds.erase(it) : std::next(it);
65 }
66 }
67
68private:
69 std::unordered_set<PlaylistLaneID> m_laneIds;
70};
