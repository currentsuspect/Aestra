9// One atomic word owns admission to RT-visible buffer storage. The high bit is
10// reserved for a control-thread configuration transaction; the remaining bits
11// count callbacks that have already entered processBlock().
12inline constexpr uint32_t kBufferConfigOwnedBit = uint32_t{1} << 31;
13inline constexpr uint32_t kProcessBlockCountMask = ~kBufferConfigOwnedBit;
14
15inline bool tryEnterProcessBlock(std::atomic<uint32_t>& state) noexcept {
16 uint32_t observed = state.load(std::memory_order_acquire);
17 for (;;) {
18 if ((observed & kBufferConfigOwnedBit) != 0) {
19 return false;
20 }
21 if ((observed & kProcessBlockCountMask) == kProcessBlockCountMask) {
22 return false;
23 }
24 if (state.compare_exchange_weak(observed, observed + 1, std::memory_order_acquire,
25 std::memory_order_relaxed)) {
26 return true;
27 }
28 }
29}
30
31inline void leaveProcessBlock(std::atomic<uint32_t>& state) noexcept {
32 state.fetch_sub(1, std::memory_order_release);
33}
34
35inline bool tryBeginBufferConfig(std::atomic<uint32_t>& state) noexcept {
36 uint32_t expected = 0;
37 return state.compare_exchange_strong(expected, kBufferConfigOwnedBit, std::memory_order_acq_rel,
38 std::memory_order_acquire);
39}
40
41inline void endBufferConfig(std::atomic<uint32_t>& state) noexcept {
42 state.store(0, std::memory_order_release);
43}
