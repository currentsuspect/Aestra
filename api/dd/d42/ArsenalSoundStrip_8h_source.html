33 static bool isAudioFile(const std::string& name) {
34 const auto dot = name.find_last_of('.');
35 if (dot == std::string::npos) return false;
36 std::string ext = name.substr(dot + 1);
37 for (auto& c : ext) c = static_cast<char>(std::tolower(static_cast<unsigned char>(c)));
38 return ext == "wav" || ext == "flac" || ext == "mp3" || ext == "ogg" || ext == "aif" || ext == "aiff";
39 }

42 static std::vector<std::string> siblings(const std::string& path) {
43 std::vector<std::string> out;
44 std::error_code ec;
45 const std::filesystem::path dir = std::filesystem::path(path).parent_path();
46 if (dir.empty() || !std::filesystem::is_directory(dir, ec)) return out;
47 for (const auto& entry : std::filesystem::directory_iterator(dir, std::filesystem::directory_options::skip_permission_denied, ec)) {
48 if (out.size() >= kMaxSiblings) break;
49 std::error_code e2;
50 if (entry.is_regular_file(e2) && isAudioFile(entry.path().filename().string())) out.push_back(entry.path().string());
51 }
52 std::sort(out.begin(), out.end(), [](const std::string& a, const std::string& b) { return lower(a) < lower(b); });
53 return out;
54 }

66 static std::string stepped(const std::vector<std::string>& list, int index, int delta, bool enterFromOutside = false) {
67 if (list.empty()) return {};
68 if (index < 0) return enterFromOutside ? list[delta > 0 ? 0 : list.size() - 1] : std::string();
69 if (list.size() < 2) return {};
70 const int n = static_cast<int>(list.size());
71 return list[static_cast<size_t>(((index + delta) % n + n) % n)];
72 }
