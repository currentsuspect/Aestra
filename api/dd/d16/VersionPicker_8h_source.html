36inline Chip chip(const TakeManager::Manifest& manifest) {
37 Chip c;
38 const auto* open = manifest.ok ? manifest.activeTake() : nullptr;
39 if (!open) return c;
40 const auto lanes = VersionLanes::names(manifest.takes);
41 const std::string branch = open->branch.empty() ? "Main" : open->branch;
42 c.lane = static_cast<int>(std::find(lanes.begin(), lanes.end(), branch) - lanes.begin());
43 c.branch = VersionLanes::label(branch);
44 c.version = nameOf(*open);
45 return c;
46}

49inline std::vector<Entry> entries(const TakeManager::Manifest& manifest) {
50 std::vector<Entry> out;
51 if (!manifest.ok) return out;
52 const auto lanes = VersionLanes::names(manifest.takes);
53 const auto* open = manifest.activeTake();
54 const std::string openBranch = open ? (open->branch.empty() ? "Main" : open->branch) : "";
55 for (size_t i = 0; i < lanes.size(); ++i) {
56 const TakeManager::TakeEntry* newest = nullptr;
57 for (const auto& v : manifest.takes) {
58 if ((v.branch.empty() ? "Main" : v.branch) != lanes[i] || v.kind == TakeManager::TakeEntry::Kind::Autosave) continue;
59 if (!newest || v.createdAtMs >= newest->createdAtMs) newest = &v;
60 }
61 if (!newest) continue;
62 out.push_back({VersionLanes::label(lanes[i]), static_cast<int>(i), newest->id, nameOf(*newest), lanes[i] == openBranch});
63 }
64 return out;
65}
