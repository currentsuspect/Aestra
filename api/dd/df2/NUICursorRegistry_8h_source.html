2// NUICursorRegistry — the single canonical source of cursor artwork.
3//
4// Every interaction cursor in the app resolves through this registry so the
5// same interaction uses the same asset regardless of which editor or component
6// renders it. Prefer these SVG glyphs over ad-hoc drawLine/vector cursor
7// artwork.
8//
9// - nuiCursorSvg(style): the overlay cursor glyphs (drawn by the app's custom
10// cursor renderer). White fill, near-black outline and a soft drop shadow, so
11// every glyph reads on dark, mid and light surfaces alike.
12// - nuiTrimResizeCursorSvg(): a tintable horizontal-stretch glyph for
13// components that paint their own cursor (e.g. the track trim edge), using
14// currentColor so NUIIcon::setColor drives the tone.
15//
16// Glyphs mean what they say:
17// Hand — a pointing hand: something clickable.
18// Grab — an open hand: something draggable, hovered.
19// Grabbing — the same hand closed: something being dragged right now.
20//
21// Hotspots (24x24 viewBox; the renderer offsets each glyph so the hotspot sits
22// on the pointer — keep AestraWindowManager::renderCustomCursor in step):
23// Arrow (2, 2) · Hand fingertip (9, 2) · Pencil tip (3, 21) · Eraser edge (6, 19) ·
24// everything else centred at (12, 12). nuiCursorHotspot() is the table the renderer reads.

125inline std::string nuiCursorSvgForSurface(NUICursorStyle style, bool darkSurface) {
126 const char* svg = nuiCursorSvg(style);
127 if (svg == nullptr) {
128 return {};
129 }
130 std::string out(svg);
131 if (!darkSurface) {
132 return out;
133 }
134 // Exact quoted tokens only, via a placeholder, so no other colour can be touched.
135 const auto replaceAll = [&out](const std::string& from, const std::string& to) {
136 for (size_t pos = out.find(from); pos != std::string::npos; pos = out.find(from, pos + to.size())) {
137 out.replace(pos, from.size(), to);
138 }
139 };
140 replaceAll("\"#fff\"", "\"@INK@\"");
141 replaceAll("\"#141416\"", "\"#fff\"");
142 replaceAll("\"@INK@\"", "\"#141416\"");
143 return out;
144}
