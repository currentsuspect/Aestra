52void MuteButton::onRender(NUIRenderer& renderer)
53{
54 auto& theme = NUIThemeManager::getInstance();
55 auto b = getBounds();
56 const auto& props = theme.getCurrentTheme();
57 NUIColor base = theme.getColor("muted");
58 auto colors = resolveControlColors(props, {true, isHovered(), false, isOn(), isFocused()});
59 NUIColor bg = colors.background;
60 NUIColor border = colors.border;
61 NUIColor text = colors.text;
62
63 if (isOn()) {
64 bg = base.withAlpha(0.16f);
65 border = base.withAlpha(0.52f);
66 text = theme.getColor("textPrimary");
67 }
68
69 renderer.fillRoundedRect(b, props.radiusM, bg);
70 renderer.strokeRoundedRect(b, props.radiusM, colors.borderWidth, border);
71 renderer.drawTextCentered("M", b, props.fontSizeM, text);
72}

113 NUIColor base = theme.getColor("armed");
114 auto colors = resolveControlColors(props, {true, isHovered(), false, isOn(), isFocused()});
115 NUIColor bg = colors.background;
116 NUIColor border = colors.border;

134 } else {
135 renderer.drawTextCentered("R", b, props.fontSizeM, text);

169UIMixerSend::UIMixerSend()
170 : m_accentColor(NUIThemeManager::getInstance().getColor("accentPrimary"))
171{
172 destSelector_ = std::make_shared<UIItemSelector>();
173 destSelector_->setAccentColor(m_accentColor);
174
175 // Forward selection changes
176 destSelector_->setOnSelectionChanged([this](int index) {
177 if (onDestChanged_ && index >= 0 && index < static_cast<int>(destinations_.size())) {
178 onDestChanged_(destinations_[index].first);
179 }
180 });
181
182 levelKnob_ = std::make_shared<UIMixerKnob>(UIMixerKnobType::Send);
183 levelKnob_->setAccentColor(m_accentColor);
184 levelKnob_->setValue(0.7f); // Unity-ish
185 levelKnob_->onValueChanged = [this](float v) {
186 if (onLevelChanged_) onLevelChanged_(v);
187 };
188
189 m_modeControl = std::make_shared<NUISegmentedControl>(std::vector<std::string>{"Pre", "Post"});
190 m_modeControl->setCornerRadius(8.0f);
191 m_modeControl->setAccentColor(m_accentColor);
192 m_modeControl->setSelectedIndex(1, false);
193 m_modeControl->setOnSelectionChanged([this](size_t index) {
194 const bool post = (index == 1);
195 if (m_postFader != post) {
196 m_postFader = post;
197 if (m_onPostFaderChanged) {
198 m_onPostFaderChanged(post);
199 }
200 repaint();
201 }
202 });
203
204 m_sendTypeControl = std::make_shared<NUISegmentedControl>(std::vector<std::string>{"Audio", "SC"});
205 m_sendTypeControl->setCornerRadius(8.0f);
206 m_sendTypeControl->setAccentColor(m_accentColor);
207 m_sendTypeControl->setSelectedIndex(0, false);
208 m_sendTypeControl->setOnSelectionChanged([this](size_t index) {
209 const bool scOnly = (index == 1);
210 if (m_sidechainOnly != scOnly) {
211 m_sidechainOnly = scOnly;
212 if (m_onSidechainModeChanged) {
213 m_onSidechainModeChanged(scOnly);
214 }
215 repaint();
216 }
217 });
218
219 deleteButton_ = std::make_shared<NUIButton>("");
220 deleteButton_->setStyle(NUIButton::Style::Secondary); // Visible border/bg
221
222 auto trashIcon = NUIIcon::createTrashIcon();
223 trashIcon->setIconSize(14, 14);
224 trashIcon->setBounds({3, 3, 14, 14});
225 trashIcon->setColor(NUIColor::white()); // Force white icon
226
227 deleteButton_->addChild(trashIcon);
228 // Red-ish background for visibility/danger
229 deleteButton_->setBackgroundColor(NUIColor::fromHex(0x502020));
230 deleteButton_->setBorderEnabled(true);
231
232 deleteButton_->setOnClick([this]() {
233 if (onDelete_) onDelete_();
234 });
235
236 addChild(destSelector_);
237 addChild(levelKnob_);
238 addChild(m_modeControl);
239 addChild(m_sendTypeControl);
240 addChild(deleteButton_);
241}

261void UIMixerSend::onRender(NUIRenderer& renderer)
262{
263 auto b = getBounds();
264 auto& theme = NUIThemeManager::getInstance();
265 const float radius = 14.0f;
266
267 renderer.drawShadow(b, 0.0f, 4.0f, 12.0f, NUIColor(0, 0, 0, 0.10f));
268 renderer.fillRoundedRect(b, radius, theme.getColor("backgroundSecondary").withAlpha(0.42f));
269 renderer.strokeRoundedRect(b, radius, 1.0f, theme.getColor("borderSubtle").withAlpha(0.36f));
270 renderer.strokeRoundedRect({b.x + 1.0f, b.y + 1.0f, b.width - 2.0f, b.height - 2.0f},
271 radius - 1.0f,
272 1.0f,
273 NUIColor::white().withAlpha(0.02f));
274
275 const NUIRect indexChip{b.x + 10.0f, b.y + 8.0f, 22.0f, 16.0f};
276 renderer.fillRoundedRect(indexChip, 8.0f, m_accentColor.withAlpha(0.12f));
277 renderer.strokeRoundedRect(indexChip, 8.0f, 1.0f, m_accentColor.withAlpha(0.20f));
278 renderer.drawTextCentered(std::to_string(index_ + 1), indexChip, 9.0f, theme.getColor("textSecondary").withAlpha(0.94f));
279
280 const char* sendKind = m_muted ? "Muted" : (m_sidechainOnly ? "SC" : "Audio");
281 const float kindW = renderer.measureText(sendKind, 8.0f).width + 12.0f;
282 const NUIRect kindChip{b.right() - kindW - 10.0f, b.y + 8.0f, kindW, 15.0f};
283 renderer.fillRoundedRect(kindChip,
284 7.5f,
285 m_muted ? theme.getColor("warning").withAlpha(0.12f)
286 : theme.getColor("backgroundPrimary").withAlpha(0.26f));
287 renderer.strokeRoundedRect(kindChip,
288 7.5f,
289 1.0f,
290 m_muted ? theme.getColor("warning").withAlpha(0.22f)
291 : theme.getColor("borderSubtle").withAlpha(0.18f));
292 renderer.drawTextCentered(sendKind, kindChip, 8.0f, theme.getColor("textSecondary").withAlpha(0.92f));
293
294 char levelBuf[32];
295 const float level = std::max(0.0001f, levelKnob_ ? levelKnob_->getValue() : 1.0f);
296 const float db = 20.0f * std::log10(level);
297 if (db <= -59.9f) {
298 std::snprintf(levelBuf, sizeof(levelBuf), "%s", "-inf dB");
299 } else {
300 std::snprintf(levelBuf, sizeof(levelBuf), "%.1f dB", db);
301 }
302
303 const float knobSize = 26.0f;
304 const float deleteBtnSize = 20.0f;
305 const float modeW = 64.0f;
306 const float typeW = 64.0f;
307 const NUIRect knobRect{b.right() - 42.0f, b.y + 48.0f, knobSize, knobSize};
308 const NUIRect comboRect{b.x + 12.0f, b.y + 52.0f, b.width - 66.0f, 22.0f};
309 const NUIRect deleteRect{b.x + 12.0f, b.bottom() - 30.0f, deleteBtnSize, deleteBtnSize};
310 const NUIRect modeGroupRect{deleteRect.right() + 10.0f, b.bottom() - 31.0f, modeW, 24.0f};
311 const NUIRect typeGroupRect{modeGroupRect.right() + 8.0f, b.bottom() - 31.0f, typeW, 24.0f};
312 const NUIRect modeRect{modeGroupRect.x, modeGroupRect.y + 8.0f, modeW, 16.0f};
313 const NUIRect typeRect{typeGroupRect.x, typeGroupRect.y + 8.0f, typeW, 16.0f};
314
315 renderer.drawText("SEND", {indexChip.right() + 8.0f, b.y + 11.0f}, 8.5f,
316 theme.getColor("textSecondary").withAlpha(0.74f));
317 const float levelW = renderer.measureText(levelBuf, 8.0f).width;
318 renderer.drawText(levelBuf,
319 {knobRect.x + knobRect.width * 0.5f - levelW * 0.5f, b.y + 27.0f},
320 8.0f, theme.getColor("textPrimary").withAlpha(0.86f));
321
322 renderer.drawText("Destination", {comboRect.x, b.y + 38.0f}, 7.5f, theme.getColor("textSecondary").withAlpha(0.76f));
323 if (m_modeEditable) {
324 renderer.drawText("Tap", {modeGroupRect.x, modeGroupRect.y - 10.0f}, 7.25f,
325 theme.getColor("textSecondary").withAlpha(0.76f));
326 }
327 renderer.drawText("Send", {typeGroupRect.x, typeGroupRect.y - 10.0f}, 7.25f,
328 theme.getColor("textSecondary").withAlpha(0.76f));
329
330 levelKnob_->setBounds(knobRect);
331 destSelector_->setBounds(comboRect);
332 m_modeControl->setVisible(m_modeEditable);
333 m_modeControl->setBounds(modeRect);
334 m_sendTypeControl->setBounds(typeRect);
335 deleteButton_->setBounds(deleteRect);
336
337 renderChildren(renderer);
338}
