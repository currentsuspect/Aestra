29uint64_t nowMs() {
30 const auto now = std::chrono::system_clock::now().time_since_epoch();
31 return static_cast<uint64_t>(std::chrono::duration_cast<std::chrono::milliseconds>(now).count());
32}

34std::string boundedStringOr(const Aestra::JSON& object, const char* key, const std::string& fallback) {
35 if (!object.has(key) || !object[key].isString()) {
36 return fallback;
37 }
38 const std::string& value = object[key].asString();
39 if (value.size() > TAKE_MAX_STRING_BYTES) {
40 return value.substr(0, TAKE_MAX_STRING_BYTES);
41 }
42 return value;
43}
44
45uint64_t boundedUint64Or(const Aestra::JSON& object, const char* key, uint64_t fallback) {
46 if (!object.has(key) || !object[key].isNumber()) {
47 return fallback;
48 }
49 const double value = object[key].asNumber();
50 if (!std::isfinite(value) || value < 0.0 || value > static_cast<double>(std::numeric_limits<uint64_t>::max())) {
51 return fallback;
52 }
53 return static_cast<uint64_t>(value);
54}

56std::string sanitizeIdPart(const std::string& input) {
57 if (input.empty()) {
58 return {};
59 }
60
61 std::string out;
62 out.reserve(input.size());
63 for (char c : input) {
64 const unsigned char uc = static_cast<unsigned char>(c);
65 if (std::isalnum(uc) || c == '-' || c == '_') {
66 out.push_back(c);
67 } else {
68 out.push_back('_');
69 }
70 }
71 return out.empty() ? "take" : out;
72}

101std::filesystem::path resolveManifestSnapshotPath(const std::string& projectPath, const std::string& snapshotPath) {
102 namespace fs = std::filesystem;
103 fs::path path(snapshotPath);
104 if (path.is_absolute()) {
105 return {};
106 }
107 const fs::path takesDir = fs::path(TakeManager::getTakesDirectory(projectPath)).lexically_normal();
108 path = (takesDir / path).lexically_normal();
109
110 std::string takesDirStr = takesDir.generic_string();
111 std::string pathStr = path.generic_string();
112#ifdef _WIN32
113 std::transform(takesDirStr.begin(), takesDirStr.end(), takesDirStr.begin(),
114 [](unsigned char c) { return static_cast<char>(std::tolower(c)); });
115 std::transform(pathStr.begin(), pathStr.end(), pathStr.begin(),
116 [](unsigned char c) { return static_cast<char>(std::tolower(c)); });
117#endif
118 while (takesDirStr.size() > 1 && takesDirStr.back() == '/') {
119 takesDirStr.pop_back();
120 }
121 if (pathStr.size() > 2 && pathStr.compare(pathStr.size() - 2, 2, "/.") == 0) {
122 pathStr.resize(pathStr.size() - 2);
123 }
124 while (pathStr.size() > 1 && pathStr.back() == '/') {
125 pathStr.pop_back();
126 }
127 if (pathStr == takesDirStr) {
128 return {};
129 }
130 if (pathStr != takesDirStr && (pathStr.size() <= takesDirStr.size() ||
131 pathStr.compare(0, takesDirStr.size(), takesDirStr) != 0 ||
132 pathStr[takesDirStr.size()] != '/')) {
133 return {};
134 }
135 return path;
136}

146Aestra::JSON takeToJson(const TakeManager::TakeEntry& take) {
147 Aestra::JSON json = Aestra::JSON::object();
148 json.set("id", Aestra::JSON(take.id));
149 json.set("name", Aestra::JSON(take.name));
150 json.set("parentId", Aestra::JSON(take.parentId));
151 json.set("snapshotPath", Aestra::JSON(take.snapshotPath));
152 json.set("createdAtMs", Aestra::JSON(static_cast<double>(take.createdAtMs)));
153 json.set("updatedAtMs", Aestra::JSON(static_cast<double>(take.updatedAtMs)));
154 json.set("branch", Aestra::JSON(take.branch));
155 json.set("kind", Aestra::JSON(take.kind == TakeManager::TakeEntry::Kind::Autosave ? "autosave" : "saved"));
156 if (!take.bounce.empty()) json.set("bounce", Aestra::JSON(take.bounce));
157 if (!take.notes.empty()) {
158 Aestra::JSON notes = Aestra::JSON::array();
159 for (const auto& n : take.notes) {
160 Aestra::JSON item = Aestra::JSON::object();
161 item.set("atMs", Aestra::JSON(static_cast<double>(n.atMs)));
162 item.set("text", Aestra::JSON(n.text));
163 notes.push(item);
164 }
165 json.set("notes", notes);
166 }
167 if (!take.steps.empty()) {
168 Aestra::JSON steps = Aestra::JSON::array();
169 for (const auto& st : take.steps) {
170 Aestra::JSON item = Aestra::JSON::object();
171 item.set("label", Aestra::JSON(st.label));
172 item.set("commandType", Aestra::JSON(st.commandType));
173 item.set("atMs", Aestra::JSON(static_cast<double>(st.atMs)));
174 steps.push(item);
175 }
176 json.set("steps", steps);
177 }
178 return json;
179}

181Aestra::JSON manifestToJson(const TakeManager::Manifest& manifest) {
182 Aestra::JSON root = Aestra::JSON::object();
183 root.set("version", Aestra::JSON(static_cast<double>(TAKE_MANIFEST_VERSION)));
184 root.set("projectPath", Aestra::JSON(std::filesystem::path(manifest.projectPath).generic_string()));
185 root.set("activeTakeId", Aestra::JSON(manifest.activeTakeId));
186
187 Aestra::JSON takes = Aestra::JSON::array();
188 for (const auto& take : manifest.takes) {
189 takes.push(takeToJson(take));
190 }
191 root.set("takes", takes);
192 return root;

288TakeManager::Manifest TakeManager::loadManifest(const std::string& projectPath) {
289 namespace fs = std::filesystem;
290
291 Manifest manifest;
292 manifest.projectPath = projectPath;
293
294 if (projectPath.empty()) {
295 return makeErrorManifest(projectPath, "Project path is empty");
296 }
297
298 const fs::path manifestPath(getManifestPath(projectPath));
299 std::error_code ec;
300 if (!fs::exists(manifestPath, ec) || ec) {
301 return makeErrorManifest(projectPath, "No Takes manifest");
302 }
303
304 const auto size = fs::file_size(manifestPath, ec);
305 if (ec || size > TAKE_MAX_MANIFEST_BYTES) {
306 return makeErrorManifest(projectPath, "Takes manifest is too large or unreadable");
307 }
308
309 std::ifstream in(manifestPath, std::ios::binary);
310 if (!in) {
311 return makeErrorManifest(projectPath, "Could not open Takes manifest");
312 }
313
314 const std::string contents((std::istreambuf_iterator<char>(in)), std::istreambuf_iterator<char>());
315 Aestra::JSON root = Aestra::JSON::parse(contents);
316 if (!root.isObject()) {
317 return makeErrorManifest(projectPath, "Takes manifest is not valid JSON");
318 }
319
320 const int version = [&]() -> int {
321 if (!root.has("version") || !root["version"].isNumber()) {
322 return 0;
323 }
324 const double v = root["version"].asNumber();
325 if (!std::isfinite(v) || v < 1.0 || v > static_cast<double>(std::numeric_limits<int>::max())) {
326 return 0;
327 }
328 return static_cast<int>(v);
329 }();
330 if (version < 1 || version > TAKE_MANIFEST_VERSION) {
331 return makeErrorManifest(projectPath, "Unsupported Takes manifest version");
332 }
333
334 manifest.activeTakeId = boundedStringOr(root, "activeTakeId", "");
335
336 if (!root.has("takes") || !root["takes"].isArray()) {
337 return makeErrorManifest(projectPath, "Takes manifest is missing takes array");
338 }
339
340 const Aestra::JSON& takesJson = root["takes"];
341 if (takesJson.size() > TAKE_MAX_ENTRIES) {
342 return makeErrorManifest(projectPath, "Takes manifest exceeds maximum take count");
343 }
344
345 for (size_t i = 0; i < takesJson.size(); ++i) {
346 const Aestra::JSON& item = takesJson[i];
347 if (!item.isObject()) {
348 continue;
349 }
350
351 TakeEntry take;
352 take.id = sanitizeIdPart(boundedStringOr(item, "id", ""));
353 take.name = boundedStringOr(item, "name", take.id.empty() ? "Take" : take.id);
354 take.parentId = sanitizeIdPart(boundedStringOr(item, "parentId", ""));
355 take.snapshotPath = boundedStringOr(item, "snapshotPath", makeSnapshotFilename(take.id));
356 take.createdAtMs = boundedUint64Or(item, "createdAtMs", 0);
357 take.updatedAtMs = boundedUint64Or(item, "updatedAtMs", take.createdAtMs);
358 take.active = !take.id.empty() && take.id == manifest.activeTakeId;
359 take.branch = boundedStringOr(item, "branch", "");
360 take.kind = boundedStringOr(item, "kind", "saved") == "autosave" ? TakeEntry::Kind::Autosave
361 : TakeEntry::Kind::Saved;
362 take.bounce = boundedStringOr(item, "bounce", "");
363 if (item.has("notes") && item["notes"].isArray()) {
364 const Aestra::JSON& notes = item["notes"];
365 for (size_t n = 0; n < notes.size() && n < TAKE_MAX_ENTRIES; ++n) {
366 if (!notes[n].isObject()) continue;
367 take.notes.push_back({boundedUint64Or(notes[n], "atMs", 0), boundedStringOr(notes[n], "text", "")});
368 }
369 }
370 if (item.has("steps") && item["steps"].isArray()) {
371 const Aestra::JSON& steps = item["steps"];
372 for (size_t n = 0; n < steps.size() && n < TAKE_MAX_STEPS; ++n) {
373 if (!steps[n].isObject()) continue;
374 take.steps.push_back({boundedStringOr(steps[n], "label", ""), boundedStringOr(steps[n], "commandType", ""),
375 boundedUint64Or(steps[n], "atMs", 0)});
376 }
377 }
378
379 if (!take.id.empty()) {
380 manifest.takes.push_back(std::move(take));
381 }
382 }
383
384 if (manifest.takes.empty()) {
385 return makeErrorManifest(projectPath, "Takes manifest contains no valid takes");
386 }
387
388 // Migration (SPEC-017): a take saved before versions has no branch. A top-level take is on Main; a take's
389 // first child continues its branch; any later child starts a branch named after itself. In memory only;
390 // the next save writes the branches out.
391 for (auto& take : manifest.takes) {
392 if (!take.branch.empty()) continue;
393 if (take.parentId.empty()) { take.branch = "Main"; continue; }
394 const TakeEntry* parent = manifest.findTake(take.parentId);
395 bool firstChild = true;
396 for (const auto& other : manifest.takes) {
397 if (&other == &take) break;
398 if (other.parentId == take.parentId) { firstChild = false; break; }
399 }
400 take.branch = (parent && firstChild && !parent->branch.empty()) ? parent->branch
401 : (parent && firstChild) ? "Main" : take.name;
402 }
403
404 if (!manifest.findTake(manifest.activeTakeId)) {
405 manifest.activeTakeId = manifest.takes.front().id;
406 }
407 for (auto& take : manifest.takes) {
408 take.active = take.id == manifest.activeTakeId;
409 }
410
411 manifest.ok = true;
412 return manifest;
413}

415TakeManager::Result TakeManager::ensureManifest(const std::string& projectPath,
416 const std::string& currentProjectContents,
417 const std::string& initialTakeName) {
418 std::string error;
419 if (!ensureDirectoryForProject(projectPath, error)) {
420 return makeResult(false, error, makeErrorManifest(projectPath, error));
421 }
422
423 Manifest existing = loadManifest(projectPath);
424 if (existing.ok) {
425 bool changed = false;
426 if (!existing.findTake(existing.activeTakeId) && !existing.takes.empty()) {
427 existing.activeTakeId = existing.takes.front().id;
428 changed = true;
429 }
430 for (auto& take : existing.takes) {
431 take.active = take.id == existing.activeTakeId;
432 }
433 if (changed && !saveManifest(existing, error)) {
434 return makeResult(false, error, existing);
435 }
436 return makeResult(true, "", existing, existing.activeTake() ? *existing.activeTake() : TakeEntry{});
437 }
438
439 // Only bootstrap a new manifest when the file simply doesn't exist.
440 // Any other load error (corrupt JSON, too large, etc.) is a real failure.
441 if (existing.errorMessage != "No Takes manifest") {
442 return makeResult(false, existing.errorMessage, existing);
443 }
444
445 Manifest manifest;
446 manifest.ok = true;
447 manifest.projectPath = projectPath;
448
449 TakeEntry mainTake;
450 mainTake.id = "main";
451 mainTake.name = initialTakeName.empty() ? "Main" : initialTakeName;
452 mainTake.snapshotPath = makeSnapshotFilename(mainTake.id);
453 mainTake.createdAtMs = nowMs();
454 mainTake.updatedAtMs = mainTake.createdAtMs;
455 mainTake.active = true;
456
457 manifest.activeTakeId = mainTake.id;
458 manifest.takes.push_back(mainTake);
459
460 if (!writeSnapshot(projectPath, mainTake, currentProjectContents, error)) {
461 return makeResult(false, error, manifest);
462 }
463 if (!saveManifest(manifest, error)) {
464 return makeResult(false, error, manifest);
465 }
466
467 Aestra::Log::info("[Takes] Initialized Takes for project: " + projectPath);
468 return makeResult(true, "", manifest, mainTake);
469}

471TakeManager::Result TakeManager::saveActiveTake(const std::string& projectPath,
472 const std::string& currentProjectContents) {
473 Result ensured = ensureManifest(projectPath, currentProjectContents);
474 if (!ensured.ok) {
475 return ensured;
476 }
477
478 Manifest manifest = ensured.manifest;
479 TakeEntry* active = nullptr;
480 for (auto& take : manifest.takes) {
481 if (take.id == manifest.activeTakeId) {
482 active = &take;
483 break;
484 }
485 }
486 if (!active) {
487 return makeResult(false, "No active take", manifest);
488 }
489
490 std::string error;
491 active->updatedAtMs = nowMs();
492 active->active = true;
493 if (!writeSnapshot(projectPath, *active, currentProjectContents, error)) {
494 return makeResult(false, error, manifest, *active);
495 }
496 if (!saveManifest(manifest, error)) {
497 return makeResult(false, error, manifest, *active);
498 }
499
500 return makeResult(true, "", manifest, *active);
501}

508TakeManager::Result TakeManager::createVersion(const std::string& projectPath, const std::string& currentProjectContents,
509 const std::string& name, const std::string& parentId,
510 const std::string& branch, TakeEntry::Kind kind,
511 const std::vector<TakeEntry::Step>& steps) {
512 Result ensured = ensureManifest(projectPath, currentProjectContents);
513 if (!ensured.ok) {
514 return ensured;
515 }
516
517 Manifest manifest = ensured.manifest;
518 if (manifest.takes.size() >= TAKE_MAX_ENTRIES) {
519 return makeResult(false, "Maximum number of takes reached (" + std::to_string(TAKE_MAX_ENTRIES) + ")", manifest);
520 }
521 const std::string effectiveParentId = parentId.empty() ? manifest.activeTakeId : sanitizeIdPart(parentId);
522
523 TakeEntry take;
524 take.id = makeTakeId();
525 take.name = name.empty() ? ("Take " + std::to_string(manifest.takes.size() + 1)) : name;
526 take.parentId = effectiveParentId;
527 take.snapshotPath = makeSnapshotFilename(take.id);
528 take.createdAtMs = nowMs();
529 take.updatedAtMs = take.createdAtMs;
530 take.active = true;
531 take.kind = kind;
532 take.steps = steps.size() > TAKE_MAX_STEPS ? std::vector<TakeEntry::Step>(steps.begin(), steps.begin() + TAKE_MAX_STEPS) : steps;
533 if (!branch.empty()) {
534 take.branch = branch.substr(0, TAKE_MAX_STRING_BYTES); // a named branch starts here
535 } else if (const TakeEntry* parent = manifest.findTake(effectiveParentId); parent && !parent->branch.empty()) {
536 take.branch = parent->branch; // continue the parent's branch
537 } else {
538 take.branch = "Main";
539 }
540
541 for (auto& existing : manifest.takes) {
542 existing.active = false;
543 }
544 manifest.activeTakeId = take.id;
545
546 std::string error;
547 if (!writeSnapshot(projectPath, take, currentProjectContents, error)) {
548 return makeResult(false, error, manifest, take);
549 }
550
551 manifest.takes.push_back(take);
552 if (!saveManifest(manifest, error)) {
553 return makeResult(false, error, manifest, take);
554 }
555
556 Aestra::Log::info("[Takes] Created take '" + take.name + "' for project: " + projectPath);
557 return makeResult(true, "", manifest, take);
558}

560TakeManager::Result TakeManager::setVersionSteps(const std::string& projectPath, const std::string& takeId,
561 const std::vector<TakeEntry::Step>& steps) {
562 Manifest manifest = loadManifest(projectPath);
563 if (!manifest.ok) return makeResult(false, manifest.errorMessage, manifest);
564 const std::string id = sanitizeIdPart(takeId);
565 for (auto& take : manifest.takes) {
566 if (take.id != id) continue;
567 take.steps = steps.size() > TAKE_MAX_STEPS ? std::vector<TakeEntry::Step>(steps.begin(), steps.begin() + TAKE_MAX_STEPS) : steps;
568 take.updatedAtMs = nowMs();
569 std::string error;
570 if (!saveManifest(manifest, error)) return makeResult(false, error, manifest, take);
571 return makeResult(true, "", manifest, take);
572 }
573 return makeResult(false, "Version not found: " + takeId, manifest);
574}

597TakeManager::Result TakeManager::addVersionNote(const std::string& projectPath, const std::string& takeId,

599 Manifest manifest = loadManifest(projectPath);
600 if (!manifest.ok) return makeResult(false, manifest.errorMessage, manifest);

607 std::string error;

609 return makeResult(true, "", manifest, take);
610 }
611 return makeResult(false, "Version not found: " + takeId, manifest);

614TakeManager::Result TakeManager::setActiveTake(const std::string& projectPath, const std::string& takeId) {
615 Manifest manifest = loadManifest(projectPath);
616 if (!manifest.ok) {
617 return makeResult(false, manifest.errorMessage, manifest);
618 }
619
620 const std::string sanitizedId = sanitizeIdPart(takeId);
621 const TakeEntry* selected = manifest.findTake(sanitizedId);
622 if (!selected) {
623 return makeResult(false, "Take not found: " + takeId, manifest);
624 }
625
626 manifest.activeTakeId = sanitizedId;
627 TakeEntry selectedCopy = *selected;
628 for (auto& take : manifest.takes) {
629 take.active = take.id == manifest.activeTakeId;
630 if (take.active) {

673 target->updatedAtMs = nowMs();

676 if (!saveManifest(manifest, error)) {
677 return makeResult(false, error, manifest, *target);
678 }
679 return makeResult(true, "", manifest, *target);

682TakeManager::Result TakeManager::duplicateTake(const std::string& projectPath, const std::string& takeId,
683 const std::string& newName, const std::string& branch) {
684 namespace fs = std::filesystem;
685
686 Manifest manifest = loadManifest(projectPath);
687 if (!manifest.ok) {
688 return makeResult(false, manifest.errorMessage, manifest);
689 }
690 if (manifest.takes.size() >= TAKE_MAX_ENTRIES) {
691 return makeResult(false, "Maximum number of takes reached (" + std::to_string(TAKE_MAX_ENTRIES) + ")", manifest);
692 }
693
694 const std::string sanitizedId = sanitizeIdPart(takeId);
695 const TakeEntry* source = manifest.findTake(sanitizedId);
696 if (!source) {
697 return makeResult(false, "Take not found: " + takeId, manifest);
698 }
699
700 const std::string sourceSnapshot = resolveSnapshotPath(projectPath, *source);
701 std::error_code ec;
702 if (sourceSnapshot.empty() || !fs::exists(sourceSnapshot, ec) || ec) {
703 return makeResult(false, "Source take snapshot is missing: " + source->id, manifest);
704 }
705
706 TakeEntry copy;
707 copy.id = makeTakeId();
708 copy.name = newName.empty() ? (source->name + " Copy") : newName;
709 copy.parentId = source->id;
710 copy.snapshotPath = makeSnapshotFilename(copy.id);
711 copy.createdAtMs = nowMs();
712 copy.updatedAtMs = copy.createdAtMs;
713 copy.active = false;
714 copy.branch = branch.empty() ? source->branch : branch.substr(0, TAKE_MAX_STRING_BYTES);
715 copy.steps = source->steps; // the copy is the same state, so it carries the same steps
716
717 const std::string copySnapshot = resolveSnapshotPath(projectPath, copy);
718 if (copySnapshot.empty()) {
719 return makeResult(false, "Could not resolve duplicate snapshot path", manifest);
720 }
721 fs::copy_file(sourceSnapshot, copySnapshot, fs::copy_options::overwrite_existing, ec);
722 if (ec) {
723 return makeResult(false, "Could not copy take snapshot: " + ec.message(), manifest);
724 }
725
726 const std::string sourceName = source->name; // push_back below may reallocate and dangle `source`
727 manifest.takes.push_back(copy);
728 std::string error;
729 if (!saveManifest(manifest, error)) {
730 fs::remove(copySnapshot, ec); // best effort: don't leave an orphan snapshot
731 return makeResult(false, error, manifest, copy);
732 }
733
734 Aestra::Log::info("[Takes] Duplicated take '" + sourceName + "' as '" + copy.name + "'");
735 return makeResult(true, "", manifest, copy);
736}

738TakeManager::Result TakeManager::deleteTake(const std::string& projectPath, const std::string& takeId) {
739 namespace fs = std::filesystem;
740
741 Manifest manifest = loadManifest(projectPath);
742 if (!manifest.ok) {
743 return makeResult(false, manifest.errorMessage, manifest);
744 }
745
746 const std::string sanitizedId = sanitizeIdPart(takeId);
747 const TakeEntry* target = manifest.findTake(sanitizedId);
748 if (!target) {
749 return makeResult(false, "Take not found: " + takeId, manifest);
750 }
751 if (sanitizedId == manifest.activeTakeId) {
752 return makeResult(false, "Cannot delete the active take — switch to another take first", manifest, *target);
753 }
754
755 TakeEntry deleted = *target;
756 const std::string snapshotPath = resolveSnapshotPath(projectPath, deleted);
757
758 // Children keep their history by moving up to the deleted take's parent.
759 for (auto& take : manifest.takes) {
760 if (take.parentId == deleted.id) {
761 take.parentId = deleted.parentId;
762 }
763 }
764 manifest.takes.erase(std::remove_if(manifest.takes.begin(), manifest.takes.end(),
765 [&deleted](const TakeEntry& take) { return take.id == deleted.id; }),
766 manifest.takes.end());
767
768 std::string error;
769 if (!saveManifest(manifest, error)) {
770 return makeResult(false, error, manifest, deleted);
771 }
772
773 // Snapshot removal is best-effort: an orphan file is harmless, a manifest
774 // pointing at a deleted file is not — so the manifest write comes first.
775 if (!snapshotPath.empty()) {
776 std::error_code ec;
777 fs::remove(snapshotPath, ec);
778 }
779
780 Aestra::Log::info("[Takes] Deleted take '" + deleted.name + "'");
781 return makeResult(true, "", manifest, deleted);
782}
