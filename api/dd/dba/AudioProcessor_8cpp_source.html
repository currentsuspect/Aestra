37void AudioProcessor::handleCommand(const AudioCommandMessage& message) {
38 switch (message.command) {
39 case AudioCommand::SetGain: {
40 m_gain.store(message.value1, std::memory_order_release);
41 break;
42 }
43
44 case AudioCommand::SetPan: {
45 float clampedPan = (message.value1 < -1.0f) ? -1.0f : (message.value1 > 1.0f) ? 1.0f : message.value1;
46 m_pan.store(clampedPan, std::memory_order_release);
47 break;
48 }
49
50 case AudioCommand::Mute: {
51 m_muted.store(true, std::memory_order_release);
52 break;
53 }
54
55 case AudioCommand::Unmute: {
56 m_muted.store(false, std::memory_order_release);
57 break;
58 }
59
60 case AudioCommand::Reset: {
61 m_gain.store(1.0f, std::memory_order_release);
62 m_pan.store(0.0f, std::memory_order_release);
63 m_muted.store(false, std::memory_order_release);
64 break;
65 }
66
67 default:
68 break;
69 }
70}

110void TestToneGenerator::process(float* outputBuffer, const float* inputBuffer, uint32_t numFrames, double streamTime) {
111 (void)inputBuffer;
112 (void)streamTime;
113
114 // Process any pending commands
115 processCommands();
116
117 // Get current parameters (atomic reads)
118 float gain = m_gain.load(std::memory_order_acquire);
119 float pan = m_pan.load(std::memory_order_acquire);
120 bool muted = m_muted.load(std::memory_order_acquire);
121 double frequency = m_frequency.load(std::memory_order_acquire);
122 double phase = m_phase.load(std::memory_order_acquire);
123
124 // Calculate pan gains (constant power panning)
125 float leftGain = std::cos((pan + 1.0f) * 3.14159265358979323846 * 0.25f);
126 float rightGain = std::sin((pan + 1.0f) * 3.14159265358979323846 * 0.25f);
127
128 // Generate audio
129 for (uint32_t i = 0; i < numFrames; ++i) {
130 float sample = 0.0f;
131
132 if (!muted) {
133 // Generate sine wave
134 sample = static_cast<float>(std::sin(2.0 * 3.14159265358979323846 * phase) * gain * 0.3);
135
136 // Advance phase
137 phase += frequency / m_sampleRate;
138 if (phase >= 1.0) {
139 phase -= 1.0;
140 }
141 }
142
143 // Apply panning and write to output (stereo)
144 outputBuffer[i * 2 + 0] = sample * leftGain; // Left
145 outputBuffer[i * 2 + 1] = sample * rightGain; // Right
146 }
147
148 // Store phase back (atomic write)
149 m_phase.store(phase, std::memory_order_release);
150}
