2// AestraOTT V1 — 3-band upward + downward compressor ("over-the-top" style).
3//
4// The input is split into low/mid/high with 4th-order Linkwitz-Riley
5// crossovers (two cascaded Butterworth TPT-SVF stages per slope). The low
6// band passes through an LR4 allpass at the high crossover so all three
7// bands stay phase-aligned and sum flat when no gain is applied.
8//
9// Each band has a stereo-linked peak follower and a symmetric gain computer
10// that pulls the band level toward a fixed -18 dBFS target: loud bands are
11// compressed down, quiet bands are boosted up. Upward gain fades to zero
12// below -50 dB so silence and noise floors are never dragged up. Depth
13// scales the computed gain (in dB) toward unity — the wet control lives in
14// the gain domain, not a dry/wet mix, so there is no phase-cancelling
15// recombination with the allpass-rotated wet path.
16//
17// Signal path (per sample):
18// in -> input gain -> LR4 split (low/mid/high)
19// -> per-band gain computer (up+down toward -18 dB, scaled by Depth)
20// -> per-band trim -> sum -> output gain

94 void process(const float* const* inputs, float** outputs, uint32_t numInputChannels, uint32_t numOutputChannels,
95 uint32_t numFrames, const MidiBuffer* midiInput = nullptr, MidiBuffer* midiOutput = nullptr) override {
96 (void)midiInput;
97 (void)midiOutput;
98
99 if (!m_active.load(std::memory_order_relaxed) || m_params[kBypass].load(std::memory_order_relaxed) > 0.5f) {
100 copyOrClear(inputs, outputs, numInputChannels, numOutputChannels, numFrames);
101 return;
102 }
103
104 const uint32_t channels = std::min<uint32_t>(2, numOutputChannels);
105 const bool stereo = channels >= 2;
106 const float sr = static_cast<float>(m_sampleRate);
107
108 // Parameters are read/smoothed and all dB<->linear transforms run once
109 // per control block; per-band gains are linearly interpolated across
110 // the block. The per-sample path is the crossover SVFs, the peak
111 // followers and multiply-adds — no atomics or transcendentals.
112 for (uint32_t blockStart = 0; blockStart < numFrames; blockStart += kCtrlBlock) {
113 const uint32_t blockEnd = std::min(blockStart + kCtrlBlock, numFrames);
114 const uint32_t blockLen = blockEnd - blockStart;
115
116 m_depthSmoothed += (getParameter(kDepth) - m_depthSmoothed) * m_smoothCoeff;
117 m_timeSmoothed += (getParameter(kTime) - m_timeSmoothed) * m_smoothCoeff;
118 m_inGainSmoothed += (getParameter(kInGain) - m_inGainSmoothed) * m_smoothCoeff;
119 m_outGainSmoothed += (getParameter(kOutGain) - m_outGainSmoothed) * m_smoothCoeff;
120 m_xoverLowSmoothed += (getParameter(kXoverLow) - m_xoverLowSmoothed) * m_smoothCoeff;
121 m_xoverHighSmoothed += (getParameter(kXoverHigh) - m_xoverHighSmoothed) * m_smoothCoeff;
122 for (uint32_t b = 0; b < kBandCount; ++b) {
123 m_bandTrimSmoothed[b] += (getParameter(kLowGain + b) - m_bandTrimSmoothed[b]) * m_smoothCoeff;
124 }
125
126 // Crossover coefficients: recompute only when the smoothed target
127 // actually moved — tan() is not free and xovers are near-static.
128 if (std::abs(m_xoverLowSmoothed - m_xoverLowCached) > 1.0e-5f || m_coeffsDirty) {
129 m_xoverLowCached = m_xoverLowSmoothed;
130 m_coeffLow = makeCoeff(xoverLowHzFromNorm(m_xoverLowCached), sr);
131 }
132 if (std::abs(m_xoverHighSmoothed - m_xoverHighCached) > 1.0e-5f || m_coeffsDirty) {
133 m_xoverHighCached = m_xoverHighSmoothed;
134 m_coeffHigh = makeCoeff(xoverHighHzFromNorm(m_xoverHighCached), sr);
135 }
136 // Attack/release coefficients: same gating, driven by the Time macro.
137 if (std::abs(m_timeSmoothed - m_timeCached) > 1.0e-4f || m_coeffsDirty) {
138 m_timeCached = m_timeSmoothed;
139 updateTimeCoeffs(sr);
140 }
141
142 const float inGain = dbToLinear(bipolarDbFromNorm(m_inGainSmoothed, 24.0f));
143 const float outGain = dbToLinear(bipolarDbFromNorm(m_outGainSmoothed, 24.0f));
144 const float depth = m_depthSmoothed;
145
146 // ── Per-band gain targets from the current envelopes; the applied
147 // gain ramps linearly to the target across the block. Envelopes
148 // move over milliseconds, so a 16-sample ramp tracks them closely. ──
149 float dGain[kBandCount];
150 for (uint32_t b = 0; b < kBandCount; ++b) {
151 const float envDb = linearToDb(m_env[b]);
152 float compDb = (kTargetDb - envDb) * kRatioAmount;
153 if (compDb > 0.0f) {
154 // Upward gain: gate away near the noise floor, cap the boost.
155 const float upScale =
156 std::clamp((envDb - kUpGateLowDb) / (kUpGateHighDb - kUpGateLowDb), 0.0f, 1.0f);
157 compDb = std::min(compDb * upScale, kMaxUpDb);
158 }
159 const float trimDb = bipolarDbFromNorm(m_bandTrimSmoothed[b], 12.0f);
160 const float target = dbToLinear(depth * compDb + trimDb);
161 m_bandGainDb[b].store(depth * compDb, std::memory_order_relaxed);
162 if (m_coeffsDirty) {
163 m_bandGain[b] = target;
164 dGain[b] = 0.0f;
165 } else {
166 dGain[b] = (target - m_bandGain[b]) / static_cast<float>(blockLen);
167 }
168 }
169 m_coeffsDirty = false;
170
171 for (uint32_t i = blockStart; i < blockEnd; ++i) {
172 const float inL = sanitizeSample(readInput(inputs, numInputChannels, 0, i)) * inGain;
173 const float inR = (stereo ? sanitizeSample(readInput(inputs, numInputChannels, 1, i)) : inL) * inGain;
174
175 // ── LR4 band split (low band allpass-corrected at the high xover) ──
176 float band[kBandCount][2];
177 for (uint32_t ch = 0; ch < 2; ++ch) {
178 const float x = (ch == 0) ? inL : inR;
179 float lowPath, highPath;
180 m_xover1[ch].split(x, m_coeffLow, lowPath, highPath);
181 float apLo, apHi;
182 m_apCorrect[ch].split(lowPath, m_coeffHigh, apLo, apHi);
183 band[kBandLow][ch] = apLo + apHi; // LR4 allpass at fHigh
184 m_xover2[ch].split(highPath, m_coeffHigh, band[kBandMid][ch], band[kBandHigh][ch]);
185 }
186
187 float outL = 0.0f;
188 float outR = 0.0f;
189 for (uint32_t b = 0; b < kBandCount; ++b) {
190 const float peak = std::max(std::abs(band[b][0]), std::abs(band[b][1]));
191 float& env = m_env[b];
192 const float coeff = (peak > env) ? m_attackCoeff[b] : m_releaseCoeff[b];
193 env += (peak - env) * coeff;
194 if (env != env || env < 1.0e-12f)
195 env = 0.0f; // NaN check + denormal flush
196
197 m_bandGain[b] += dGain[b];
198 outL += band[b][0] * m_bandGain[b];
199 outR += band[b][1] * m_bandGain[b];
200 }
201
202 if (outputs[0])
203 outputs[0][i] = flushDenormal(outL * outGain);
204 if (numOutputChannels > 1 && outputs[1])
205 outputs[1][i] = flushDenormal((stereo ? outR : outL) * outGain);
206 for (uint32_t ch = 2; ch < numOutputChannels; ++ch) {
207 if (outputs[ch])
208 outputs[ch][i] = 0.0f;
209 }
210 }
211 }
212 }

246 if (id >= kParamCount)

248 const float v = getParameter(id);

256 case kInGain:
257 case kOutGain:
258 std::snprintf(buf, sizeof(buf), "%+.1f dB", bipolarDbFromNorm(v, 24.0f));

260 case kLowGain:
261 case kMidGain:
262 case kHighGain:
263 std::snprintf(buf, sizeof(buf), "%+.1f dB", bipolarDbFromNorm(v, 12.0f));
264 return buf;
265 case kXoverLow:
266 std::snprintf(buf, sizeof(buf), "%.0f Hz", xoverLowHzFromNorm(v));
267 return buf;
268 case kXoverHigh: {
269 const float hz = xoverHighHzFromNorm(v);
270 std::snprintf(buf, sizeof(buf), "%.2f kHz", hz / 1000.0f);
271 return buf;
272 }
273 case kBypass:
274 return v > 0.5f ? "ON" : "OFF";
275 default:
276 return "";

280 std::vector<uint8_t> saveState() const override {
281 struct Blob {
282 uint32_t magic = kStateMagic;
283 uint32_t version = 1;
284 float params[kParamCount] = {};
285 } blob;
286 for (uint32_t i = 0; i < kParamCount; ++i) {
287 blob.params[i] = getParameter(i);
288 }
289 const auto* data = reinterpret_cast<const uint8_t*>(&blob);
290 return {data, data + sizeof(blob)};
291 }

293 bool loadState(const std::vector<uint8_t>& state) override {
294 if (state.size() < sizeof(uint32_t) * 2)
295 return false;
296 uint32_t magic = 0;
297 std::memcpy(&magic, state.data(), sizeof(magic));
298 if (magic != kStateMagic)
299 return false;
300 struct Blob {
301 uint32_t magic;
302 uint32_t version;
303 float params[kParamCount];
304 };
305 if (state.size() < sizeof(Blob))
306 return false;
307 Blob blob{};
308 std::memcpy(&blob, state.data(), sizeof(blob));
309 if (blob.version < 1 || blob.version > 1)
310 return false;
311 // Validate the entire decoded set before mutating anything. setParameter
312 // silently drops non-finite values, so applying in place would leave a
313 // half-updated state while still reporting success — fail atomically.
314 for (uint32_t i = 0; i < kParamCount; ++i) {
315 if (!std::isfinite(blob.params[i]) || blob.params[i] < 0.0f || blob.params[i] > 1.0f)
316 return false;
317 }
318 for (uint32_t i = 0; i < kParamCount; ++i) {
319 setParameter(i, blob.params[i]);
320 }
321 return true;
322 }

356 // 0.1x .. 10x, log, center = 1x
357 return 0.1f * std::pow(100.0f, std::clamp(value, 0.0f, 1.0f));

362 }

365 // 60 Hz .. 500 Hz, log

374private:
375 static constexpr uint32_t kCtrlBlock = 16; // control-rate interval for param/gain updates

377 // ── TPT (ZDF) SVF stage, Butterworth damping — building block for LR4 ──
378 struct SvfCoeff {
379 float a1 = 0.0f;
380 float a2 = 0.0f;
381 float a3 = 0.0f;
382 float k = 1.41421356f; // 1/Q for Butterworth
383 };
384
385 struct SvfStage {
386 float ic1 = 0.0f;
387 float ic2 = 0.0f;
388 void reset() { ic1 = ic2 = 0.0f; }
389 void step(float x, const SvfCoeff& c, float& lp, float& hp) {
390 const float v3 = x - ic2;
391 const float v1 = c.a1 * ic1 + c.a2 * v3;
392 const float v2 = ic2 + c.a2 * ic1 + c.a3 * v3;
393 ic1 = 2.0f * v1 - ic1;
394 ic2 = 2.0f * v2 - ic2;
395 lp = v2;
396 hp = x - c.k * v1 - v2;
397 }
398 };
399
400 // LR4 crossover: LP4 = two cascaded Butterworth LP2, HP4 = two cascaded
401 // Butterworth HP2. LP4 + HP4 sums to a 4th-order allpass (flat magnitude).
402 struct CrossoverLR4 {
403 SvfStage first;
404 SvfStage lowSecond;
405 SvfStage highSecond;
406 void reset() {
407 first.reset();
408 lowSecond.reset();
409 highSecond.reset();
410 }
411 void split(float x, const SvfCoeff& c, float& low, float& high) {
412 float lp1, hp1;
413 first.step(x, c, lp1, hp1);
414 float unusedHp, unusedLp;
415 lowSecond.step(lp1, c, low, unusedHp);
416 highSecond.step(hp1, c, unusedLp, high);
417 }
418 };

420 static SvfCoeff makeCoeff(float fc, float sr) {
421 SvfCoeff c;
422 // At extremely low sample rates 0.45*sr can fall below the 20 Hz floor,
423 // which would make std::clamp's bounds invalid (lo > hi). Keep lo <= hi.
424 const float hi = 0.45f * sr;
425 const float clamped = std::clamp(fc, std::min(20.0f, hi), hi);
426 const float g = std::tan(3.14159265358979323846f * clamped / sr);
427 c.a1 = 1.0f / (1.0f + g * (g + c.k));
428 c.a2 = g * c.a1;
429 c.a3 = g * c.a2;
430 return c;
431 }
432
433 void updateTimeCoeffs(float sr) {
434 // Base times per band (ms): lows breathe slower, highs snap faster.
435 static constexpr float kAttackMs[kBandCount] = {24.0f, 10.0f, 3.0f};
436 static constexpr float kReleaseMs[kBandCount] = {240.0f, 120.0f, 60.0f};
437 const float mult = timeMultFromNorm(m_timeCached);
438 for (uint32_t b = 0; b < kBandCount; ++b) {
439 m_attackCoeff[b] = 1.0f - std::exp(-1.0f / (kAttackMs[b] * mult * 0.001f * sr));
440 m_releaseCoeff[b] = 1.0f - std::exp(-1.0f / (kReleaseMs[b] * mult * 0.001f * sr));
441 }
442 }

444 void resetRuntimeState() {
445 for (uint32_t ch = 0; ch < 2; ++ch) {
446 m_xover1[ch].reset();
447 m_xover2[ch].reset();
448 m_apCorrect[ch].reset();
449 }
450 for (uint32_t b = 0; b < kBandCount; ++b) {
451 m_env[b] = 0.0f;
452 m_bandGainDb[b].store(0.0f, std::memory_order_relaxed);
453 }
454 const float sr = static_cast<float>(m_sampleRate);
455 // Smoothing runs once per control block, so the coefficient is scaled
456 // to the block interval to keep the 2 ms time constant.
457 m_smoothCoeff = 1.0f - std::exp(-static_cast<float>(kCtrlBlock) / (0.002f * sr));
458 m_coeffsDirty = true;
459 }

474 // ── Utility ──
475 static void copyOrClear(const float* const* inputs, float** outputs, uint32_t numInputChannels,
476 uint32_t numOutputChannels, uint32_t numFrames) {
477 for (uint32_t ch = 0; ch < numOutputChannels; ++ch) {
478 if (outputs[ch] && ch < numInputChannels && inputs[ch]) {
479 // Skip the copy when the host renders in place (out == in):
480 // memcpy on aliased buffers is undefined and the data is already
481 // where it needs to be.
482 if (outputs[ch] != inputs[ch])
483 std::memcpy(outputs[ch], inputs[ch], numFrames * sizeof(float));
484 } else if (outputs[ch]) {
485 std::memset(outputs[ch], 0, numFrames * sizeof(float));
486 }
487 }
488 }

491 if (channel >= channels || !inputs[channel]) {
