55NUIRect UIParamBar::barRect() const {
56 const auto b = getBounds();
57 if (m_look == Look::Bar)
58 return NUIRect{b.x, b.y, b.width, b.height};
59 const float x = b.x + (m_label.empty() ? 0.0f : kLabelW);

63void UIParamBar::onRender(NUIRenderer& renderer) {
64 const auto b = getBounds();
65 if (b.isEmpty())
66 return;
67 auto& theme = NUIThemeManager::getInstance();
68 const NUIColor text = theme.getColor("textPrimary");
69 const NUIColor muted = theme.getColor("textSecondary");
70 const NUIColor track = theme.getColor("border").withAlpha(0.45f);
71 const NUIColor fill = (m_hovered || m_dragging) ? text : text.withAlpha(0.82f);
72
73 NUIRect bar = barRect();
74 if (m_look == Look::Bar) {
75 // Thin bar centred in the bounds: 6 px, with the marker and centre tick taller than it.
76 bar = NUIRect{b.x, b.y + b.height * 0.5f - 3.0f, b.width, 6.0f};
77 } else {
78 if (!m_label.empty())
79 renderer.drawText(m_label, {b.x, b.y + (b.height - kFont) * 0.5f - 1.0f}, kFont, muted.withAlpha(0.9f));
80 const std::string valueText = m_formatter ? m_formatter(m_value) : std::to_string(m_value);
81 const float w = renderer.measureText(valueText, kFont).width;
82 renderer.drawText(valueText, {b.right() - w, b.y + (b.height - kFont) * 0.5f - 1.0f}, kFont, text.withAlpha(0.95f));
83 }
84
85 renderer.fillRect(bar, track);
86 const float t = (m_value - m_min) / (m_max - m_min);
87 const float x = bar.x + t * bar.width;
88 if (m_bipolar) {
89 const float mid = bar.x + bar.width * 0.5f;
90 renderer.fillRect(NUIRect{std::min(mid, x), bar.y, std::max(1.0f, std::fabs(x - mid)), bar.height},
91 fill.withAlpha(0.35f));
92 renderer.fillRect(NUIRect{mid - 0.5f, bar.y - 3.0f, 1.0f, bar.height + 6.0f}, muted.withAlpha(0.55f));
93 } else {
94 renderer.fillRect(NUIRect{bar.x, bar.y, std::max(0.0f, x - bar.x), bar.height}, fill.withAlpha(0.35f));
95 }
96 // The marker: a 2 px line taller than the bar, so the value reads even where the fill is faint.
97 renderer.fillRect(NUIRect{std::clamp(x - 1.0f, bar.x, bar.right() - 2.0f), bar.y - 2.0f, 2.0f, bar.height + 4.0f}, fill);
98}

118bool UIParamBar::onMouseEvent(const NUIMouseEvent& event) {
119 if (!isVisible() || !isEnabled())
120 return false;
121 const auto b = getBounds();
122 const bool inside = b.contains(event.position);
123 if (!m_dragging && event.button == NUIMouseButton::None) {
124 if (inside != m_hovered) {
125 m_hovered = inside;
126 repaint();
127 }
128 return false;
129 }
130 if (!inside && !m_dragging)
131 return false;
132
133 if (event.pressed && event.button == NUIMouseButton::Left) {
134 if (m_clickTracker.registerPress() || event.doubleClick) {
135 applyUserValue(m_default);
136 if (onValueCommitted)
137 onValueCommitted(m_value);
138 return true;
139 }
140 m_dragging = true;
141 m_dragStartX = event.position.x;
142 m_dragStartValue = m_value;
143 repaint();
144 return true;
145 }
146 if (m_dragging && event.released && event.button == NUIMouseButton::Left) {
147 m_dragging = false;
148 repaint();
149 if (onValueCommitted)
150 onValueCommitted(m_value);
151 return true;
152 }
153 if (m_dragging && event.button == NUIMouseButton::None) {
154 const float width = std::max(8.0f, barRect().width);
155 const float fine = (event.modifiers & NUIModifiers::Shift) ? 0.1f : 1.0f;
156 applyUserValue(m_dragStartValue + (event.position.x - m_dragStartX) / width * (m_max - m_min) * fine);
157 return true;
158 }
159 return false;
160}
