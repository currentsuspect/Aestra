10// Platform::setCurrentThreadPriority implementation for Linux
11bool Platform::setCurrentThreadPriority(ThreadPriority priority) {
12 if (priority == ThreadPriority::Normal) {
13 // Normal priority: SCHED_OTHER, nice 0
14 struct sched_param param;
15 param.sched_priority = 0;
16 if (pthread_setschedparam(pthread_self(), SCHED_OTHER, &param) == 0) {
17 setpriority(PRIO_PROCESS, 0, 0); // Reset nice value
18 return true;
19 }
20 return false;
21 }
22
23 if (priority == ThreadPriority::Low) {
24 // Low: SCHED_OTHER, nice 10
25 setpriority(PRIO_PROCESS, 0, 10);
26 return true;
27 }
28
29 if (priority == ThreadPriority::High) {
30 // High: nice -10 (might need CAP_SYS_NICE for < 0, but -10 usually requires it)
31 // Try nice first
32 if (setpriority(PRIO_PROCESS, 0, -10) == 0)
33 return true;
34
35 std::cerr << "Warning: Failed to set High thread priority (EPERM?)." << std::endl;
36 return false;
37 }
38
39 if (priority == ThreadPriority::RealtimeAudio) {
40 // Realtime: SCHED_FIFO or SCHED_RR
41 // This usually requires higher privileges or rtlimits setup
42 int policy = SCHED_FIFO;
43 int min_prio = sched_get_priority_min(policy);
44 int max_prio = sched_get_priority_max(policy);
45
46 struct sched_param param;
47 // Conservative RT priority
48 param.sched_priority = min_prio + 10;
49 if (param.sched_priority > max_prio)
50 param.sched_priority = max_prio;
51
52 if (pthread_setschedparam(pthread_self(), policy, &param) == 0) {
53 return true;
54 } else {
55 std::cerr << "Warning: Failed to set Realtime thread priority (needs CAP_SYS_NICE or RT limits)."
56 << std::endl;
57 // Fallback to high priority nice
58 setpriority(PRIO_PROCESS, 0, -15);
59 return false;
60 }
61 }
62
63 return true;
64}
