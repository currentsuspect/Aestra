310class MidiBuffer {
311public:
312 MidiBuffer() = default;
313
314 static constexpr size_t kMaxEvents = 1024;
315
316 // Explicit copy logic since atomic deletes default copy/move
317 MidiBuffer(const MidiBuffer& other) {
318 // Not strictly thread safe to copy other while it's being written,
319 // but meaningful for vector resize if done from a safe thread.
320 // Copy events
321 uint32_t count = other.m_eventCount.load(std::memory_order_acquire);
322 m_eventCount.store(count, std::memory_order_release);
323 std::memcpy(m_events, other.m_events, sizeof(Event) * kMaxEvents);
324 }
325
326 MidiBuffer& operator=(const MidiBuffer& other) {
327 if (this != &other) {
328 uint32_t count = other.m_eventCount.load(std::memory_order_acquire);
329 m_eventCount.store(count, std::memory_order_release);
330 std::memcpy(m_events, other.m_events, sizeof(Event) * kMaxEvents);
331 }
332 return *this;
333 }
334
335 // Move is same as copy for atomic + POD array
336 MidiBuffer(MidiBuffer&& other) noexcept {
337 uint32_t count = other.m_eventCount.load(std::memory_order_acquire);
338 m_eventCount.store(count, std::memory_order_release);
339 std::memcpy(m_events, other.m_events, sizeof(Event) * kMaxEvents);
340 // Reset other not strictly needed but good practice
341 other.m_eventCount.store(0, std::memory_order_release);
342 }
343
344 struct Event {
345 uint32_t sampleOffset;
346 uint8_t data[4];
347 uint8_t size;
348 };
349
350 void addEvent(uint32_t sampleOffset, const uint8_t* data, uint8_t size) {
351 if (size <= 3 && data != nullptr) {
352 uint32_t index = m_eventCount.load(std::memory_order_acquire);
353 while (index < kMaxEvents) {
354 const uint32_t desired = index + 1;
355 if (!m_eventCount.compare_exchange_weak(index, desired,
356 std::memory_order_acq_rel,
357 std::memory_order_acquire)) {
358 continue;
359 }
360 Event& e = m_events[index];
361 e.sampleOffset = sampleOffset;
362 e.size = size;
363 std::memset(e.data, 0, sizeof(e.data));
364 std::memcpy(e.data, data, size);
365 return;
366 }
367 }
368 }
369
370 void clear() { m_eventCount.store(0, std::memory_order_release); }
371 bool isEmpty() const { return m_eventCount.load(std::memory_order_acquire) == 0; }
372 size_t getEventCount() const {
373 size_t count = m_eventCount.load(std::memory_order_acquire);
374 return count > kMaxEvents ? kMaxEvents : count;
375 }
376 const Event& getEvent(size_t index) const {
377 assert(index < kMaxEvents);
378 return m_events[index];
379 }
380
384 void addNoteOn(uint8_t channel, uint8_t pitch, uint8_t velocity, uint32_t sampleOffset) {
385 uint8_t data[3] = {static_cast<uint8_t>(0x90 | ((channel - 1) & 0x0F)), pitch, velocity};
386 addEvent(sampleOffset, data, 3);
387 }
388
392 void addNoteOff(uint8_t channel, uint8_t pitch, uint8_t velocity, uint32_t sampleOffset) {
393 uint8_t data[3] = {static_cast<uint8_t>(0x80 | ((channel - 1) & 0x0F)), pitch, velocity};
394 addEvent(sampleOffset, data, 3);
395 }
396
397 // Iterator support
398 const Event* begin() const { return &m_events[0]; }
399 const Event* end() const { return &m_events[getEventCount()]; }
400
401private:
402 Event m_events[kMaxEvents];
403 std::atomic<uint32_t> m_eventCount{0};
404};
