352class MidiBuffer {
353public:
354 MidiBuffer() = default;
355
356 static constexpr size_t kMaxEvents = 1024;
357
358 // Explicit copy logic since atomic deletes default copy/move
359 MidiBuffer(const MidiBuffer& other) {
360 // Not strictly thread safe to copy other while it's being written,
361 // but meaningful for vector resize if done from a safe thread.
362 // Copy events
363 uint32_t count = other.m_eventCount.load(std::memory_order_acquire);
364 m_eventCount.store(count, std::memory_order_release);
365 std::memcpy(m_events, other.m_events, sizeof(Event) * kMaxEvents);
366 }
367
368 MidiBuffer& operator=(const MidiBuffer& other) {
369 if (this != &other) {
370 uint32_t count = other.m_eventCount.load(std::memory_order_acquire);
371 m_eventCount.store(count, std::memory_order_release);
372 std::memcpy(m_events, other.m_events, sizeof(Event) * kMaxEvents);
373 }
374 return *this;
375 }
376
377 // Move is same as copy for atomic + POD array
378 MidiBuffer(MidiBuffer&& other) noexcept {
379 uint32_t count = other.m_eventCount.load(std::memory_order_acquire);
380 m_eventCount.store(count, std::memory_order_release);
381 std::memcpy(m_events, other.m_events, sizeof(Event) * kMaxEvents);
382 // Reset other not strictly needed but good practice
383 other.m_eventCount.store(0, std::memory_order_release);
384 }
385
386 struct Event {
387 uint32_t sampleOffset;
388 uint8_t data[4];
389 uint8_t size;
393 uint32_t noteId;
394 };
395
396 void addEvent(uint32_t sampleOffset, const uint8_t* data, uint8_t size, uint32_t noteId = 0) {
397 if (size <= 3 && data != nullptr) {
398 uint32_t index = m_eventCount.load(std::memory_order_acquire);
399 while (index < kMaxEvents) {
400 const uint32_t desired = index + 1;
401 if (!m_eventCount.compare_exchange_weak(index, desired,
402 std::memory_order_acq_rel,
403 std::memory_order_acquire)) {
404 continue;
405 }
406 Event& e = m_events[index];
407 e.sampleOffset = sampleOffset;
408 e.size = size;
409 e.noteId = noteId;
410 std::memset(e.data, 0, sizeof(e.data));
411 std::memcpy(e.data, data, size);
412 return;
413 }
414 }
415 }
416
417 void clear() { m_eventCount.store(0, std::memory_order_release); }
418 bool isEmpty() const { return m_eventCount.load(std::memory_order_acquire) == 0; }
419 size_t getEventCount() const {
420 size_t count = m_eventCount.load(std::memory_order_acquire);
421 return count > kMaxEvents ? kMaxEvents : count;
422 }
423 const Event& getEvent(size_t index) const {
424 assert(index < kMaxEvents);
425 return m_events[index];
426 }
427
431 void addNoteOn(uint8_t channel, uint8_t pitch, uint8_t velocity, uint32_t sampleOffset) {
432 uint8_t data[3] = {static_cast<uint8_t>(0x90 | ((channel - 1) & 0x0F)), pitch, velocity};
433 addEvent(sampleOffset, data, 3);
434 }
435
439 void addNoteOff(uint8_t channel, uint8_t pitch, uint8_t velocity, uint32_t sampleOffset) {
440 uint8_t data[3] = {static_cast<uint8_t>(0x80 | ((channel - 1) & 0x0F)), pitch, velocity};
441 addEvent(sampleOffset, data, 3);
442 }
443
444 // Iterator support
445 const Event* begin() const { return &m_events[0]; }
446 const Event* end() const { return &m_events[getEventCount()]; }
447
448private:
449 Event m_events[kMaxEvents];
450 std::atomic<uint32_t> m_eventCount{0};
451};
