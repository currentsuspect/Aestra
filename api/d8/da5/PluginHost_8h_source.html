358class MidiBuffer {
359public:
360 MidiBuffer() = default;
361
362 static constexpr size_t kMaxEvents = 1024;
363
364 // Explicit copy logic since atomic deletes default copy/move
365 MidiBuffer(const MidiBuffer& other) {
366 // Not strictly thread safe to copy other while it's being written,
367 // but meaningful for vector resize if done from a safe thread.
368 // Copy events
369 uint32_t count = other.m_eventCount.load(std::memory_order_acquire);
370 m_eventCount.store(count, std::memory_order_release);
371 std::memcpy(m_events, other.m_events, sizeof(Event) * kMaxEvents);
372 }
373
374 MidiBuffer& operator=(const MidiBuffer& other) {
375 if (this != &other) {
376 uint32_t count = other.m_eventCount.load(std::memory_order_acquire);
377 m_eventCount.store(count, std::memory_order_release);
378 std::memcpy(m_events, other.m_events, sizeof(Event) * kMaxEvents);
379 }
380 return *this;
381 }
382
383 // Move is same as copy for atomic + POD array
384 MidiBuffer(MidiBuffer&& other) noexcept {
385 uint32_t count = other.m_eventCount.load(std::memory_order_acquire);
386 m_eventCount.store(count, std::memory_order_release);
387 std::memcpy(m_events, other.m_events, sizeof(Event) * kMaxEvents);
388 // Reset other not strictly needed but good practice
389 other.m_eventCount.store(0, std::memory_order_release);
390 }
391
392 struct Event {
393 uint32_t sampleOffset;
394 uint8_t data[4];
395 uint8_t size;
399 uint32_t noteId;
400 };
401
402 void addEvent(uint32_t sampleOffset, const uint8_t* data, uint8_t size, uint32_t noteId = 0) {
403 if (size <= 3 && data != nullptr) {
404 uint32_t index = m_eventCount.load(std::memory_order_acquire);
405 while (index < kMaxEvents) {
406 const uint32_t desired = index + 1;
407 if (!m_eventCount.compare_exchange_weak(index, desired,
408 std::memory_order_acq_rel,
409 std::memory_order_acquire)) {
410 continue;
411 }
412 Event& e = m_events[index];
413 e.sampleOffset = sampleOffset;
414 e.size = size;
415 e.noteId = noteId;
416 std::memset(e.data, 0, sizeof(e.data));
417 std::memcpy(e.data, data, size);
418 return;
419 }
420 }
421 }
422
423 void clear() { m_eventCount.store(0, std::memory_order_release); }
424 bool isEmpty() const { return m_eventCount.load(std::memory_order_acquire) == 0; }
425 size_t getEventCount() const {
426 size_t count = m_eventCount.load(std::memory_order_acquire);
427 return count > kMaxEvents ? kMaxEvents : count;
428 }
429 const Event& getEvent(size_t index) const {
430 assert(index < kMaxEvents);
431 return m_events[index];
432 }
433
437 void addNoteOn(uint8_t channel, uint8_t pitch, uint8_t velocity, uint32_t sampleOffset) {
438 uint8_t data[3] = {static_cast<uint8_t>(0x90 | ((channel - 1) & 0x0F)), pitch, velocity};
439 addEvent(sampleOffset, data, 3);
440 }
441
445 void addNoteOff(uint8_t channel, uint8_t pitch, uint8_t velocity, uint32_t sampleOffset) {
446 uint8_t data[3] = {static_cast<uint8_t>(0x80 | ((channel - 1) & 0x0F)), pitch, velocity};
447 addEvent(sampleOffset, data, 3);
448 }
449
450 // Iterator support
451 const Event* begin() const { return &m_events[0]; }
452 const Event* end() const { return &m_events[getEventCount()]; }
453
454private:
455 Event m_events[kMaxEvents];
456 std::atomic<uint32_t> m_eventCount{0};
457};
