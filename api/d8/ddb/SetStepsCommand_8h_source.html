27class SetStepsCommand : public ICommand {
28public:
29 SetStepsCommand(PatternManager& patternManager, PatternID patternId,
30 std::vector<MidiNote> rowNotes, uint64_t unitId, int pitch,
31 double rowLengthBeats)
32 : m_patternManager(patternManager), m_patternId(patternId),
33 m_rowNotes(std::move(rowNotes)), m_unitId(unitId), m_pitch(pitch),
34 m_rowLengthBeats(rowLengthBeats) {}
35
36 void execute() override {
37 if (m_executed) return;
38 m_replacedNotes.clear();
39
40 m_patternManager.applyPatch(m_patternId, [this](PatternSource& pattern) {
41 if (!pattern.isMidi()) return;
42 auto& notes = std::get<MidiPayload>(pattern.payload).notes;
43
44 // Capture and remove the entire existing row for this identity.
45 for (const MidiNote& note : notes) {
46 if (note.unitId == m_unitId && note.pitch == m_pitch) {
47 m_replacedNotes.push_back(note);
48 }
49 }
50 notes.erase(std::remove_if(notes.begin(), notes.end(),
51 [this](const MidiNote& n) {
52 return n.unitId == m_unitId && n.pitch == m_pitch;
53 }),
54 notes.end());
55
56 for (const MidiNote& note : m_rowNotes) {
57 notes.push_back(note);
58 }
59
60 m_previousLengthBeats = pattern.lengthBeats;
61 if (m_rowLengthBeats > pattern.lengthBeats) {
62 pattern.lengthBeats = m_rowLengthBeats;
63 }
64 });
65
66 m_executed = true;
67 }
68
69 void undo() override {
70 if (!m_executed) return;
71
72 m_patternManager.applyPatch(m_patternId, [this](PatternSource& pattern) {
73 if (!pattern.isMidi()) return;
74 auto& notes = std::get<MidiPayload>(pattern.payload).notes;
75
76 notes.erase(std::remove_if(notes.begin(), notes.end(),
77 [this](const MidiNote& n) {
78 return n.unitId == m_unitId && n.pitch == m_pitch;
79 }),
80 notes.end());
81 for (const MidiNote& note : m_replacedNotes) {
82 notes.push_back(note);
83 }
84 pattern.lengthBeats = m_previousLengthBeats;
85 });
86
87 m_executed = false;
88 }
89
90 void redo() override {
91 if (m_executed) return;
92 execute();
93 }
94
95 std::string getName() const override { return "Set Steps"; }
96 size_t getSizeInBytes() const override {
97 return sizeof(*this) + (m_rowNotes.size() + m_replacedNotes.size()) * sizeof(MidiNote);
98 }
99 bool changesProjectState() const override { return true; }
100
101private:
102 PatternManager& m_patternManager;
103 PatternID m_patternId;
104 std::vector<MidiNote> m_rowNotes;
105 uint64_t m_unitId;
106 int m_pitch;
107 double m_rowLengthBeats;
108
109 // Recorded by execute() for undo
110 std::vector<MidiNote> m_replacedNotes;
111 double m_previousLengthBeats = 0.0;
112 bool m_executed = false;
113};
