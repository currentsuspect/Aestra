50 m_patternManager.applyPatch(m_patternId, [this, semitones](PatternSource& pattern) {
51 if (!pattern.isMidi()) return;
52 for (MidiNote& note : std::get<MidiPayload>(pattern.payload).notes) {
53 if (m_unitId != 0 && note.unitId != m_unitId) continue;
54 note.pitch += semitones;
