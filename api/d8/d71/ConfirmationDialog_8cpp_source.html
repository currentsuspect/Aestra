58void ConfirmationDialog::handleResponse(DialogResponse response) {
59 m_response = response;
60
61 std::string responseStr;
62 switch (response) {
63 case DialogResponse::Save: responseStr = "Save"; break;
64 case DialogResponse::DontSave: responseStr = "Don't Save"; break;
65 case DialogResponse::Cancel: responseStr = "Cancel"; break;
66 case DialogResponse::Confirm: responseStr = m_confirmLabel; break;
67 default: responseStr = "Unknown"; break;
68 }
69 Log::info("[ConfirmationDialog] User selected: " + responseStr);
70
71 hide();
72
73 if (m_callback) {
74 m_callback(response);
75 }
76}

94void ConfirmationDialog::calculateLayout() {
95 AestraUI::NUIRect parentBounds = getBounds();
96 const auto& theme = AestraUI::NUIThemeManager::getInstance().getCurrentTheme();
97
98 // Dialog dimensions
99 const float dialogWidth = 400.0f;
100 const float buttonHeight = theme.layout.dialogActionHeight;
101 const float buttonSpacing = theme.spacingS;
102 const float margin = theme.spacingL;
103 // Sized to its content (SPEC 3 §3.2): title bar, the one-line message band, the divider
104 // and the buttons. It was a fixed 172 + title bar: 172 was sized when the title was drawn
105 // inside the body, so once the title moved into the shared title bar the message sat at
106 // the top of an empty band.
107 constexpr float kMessageBandH = 24.0f;
108 constexpr float kDividerGap = 16.0f; // divider sits this far above the buttons
109 const float dialogHeight =
110 AestraUI::AestraPanelWindow::TITLE_BAR_H + margin + kMessageBandH + margin + kDividerGap + buttonHeight + margin;
111
112 // Center dialog in parent (rounded so 1px strokes/text stay crisp).
113 m_dialogRect.x = std::round(parentBounds.x + (parentBounds.width - dialogWidth) / 2.0f);
114 m_dialogRect.y = std::round(parentBounds.y + (parentBounds.height - dialogHeight) / 2.0f);
115 m_dialogRect.width = dialogWidth;
116 m_dialogRect.height = dialogHeight;
117 m_messageCentreY = m_dialogRect.y + AestraUI::AestraPanelWindow::TITLE_BAR_H + margin + kMessageBandH * 0.5f;
118
119 // Right-aligned button row, primary rightmost — modern convention:
120 // Save flow: [ Cancel ] [ Don't Save ] [ Save ].
121 // Confirm flow: [ Cancel ] [ Confirm ].
122 const float cancelWidth = 84.0f;
123 const float confirmWidth = 110.0f;
124 const float saveWidth = 96.0f;
125 const float dontSaveWidth = 104.0f;
126 const float totalWidth = m_isConfirmMode
127 ? cancelWidth + confirmWidth + buttonSpacing
128 : cancelWidth + dontSaveWidth + saveWidth + buttonSpacing * 2.0f;
129
130 const float buttonY = m_dialogRect.y + dialogHeight - margin - buttonHeight;
131 const float startX = m_dialogRect.right() - margin - totalWidth;
132
133 m_cancelButtonRect = {startX, buttonY, cancelWidth, buttonHeight};
134 if (m_isConfirmMode) {
135 // The Don't Save slot does not exist in confirm mode; keep it empty so
136 // no stale rect aliases a live button.
137 m_dontSaveButtonRect = AestraUI::NUIRect{};
138 m_saveButtonRect = {m_cancelButtonRect.right() + buttonSpacing, buttonY, confirmWidth, buttonHeight};
139 } else {
140 m_dontSaveButtonRect = {m_cancelButtonRect.right() + buttonSpacing, buttonY, dontSaveWidth, buttonHeight};
141 m_saveButtonRect = {m_dontSaveButtonRect.right() + buttonSpacing, buttonY, saveWidth, buttonHeight};
142 }
143}

145void ConfirmationDialog::onRender(AestraUI::NUIRenderer& renderer) {
146 if (!m_isVisible) {
147 return;
148 }
149
150 calculateLayout();
151 const auto& theme = AestraUI::NUIThemeManager::getInstance().getCurrentTheme();
152
153 // Dim the app behind the modal.
154 const AestraUI::NUIRect parentBounds = getBounds();
155 renderer.fillRect(parentBounds, theme.overlay);
156
157 // Soft drop shadow (offsetX, offsetY, blur, color).
158 renderer.drawShadow(m_dialogRect, 0.0f, theme.spacingS, theme.spacingL, theme.shadow);
159
160 // Surface and title bar come from the shared dialog chrome (spec 2 §5), so
161 // this reads as an Aestra panel rather than as a generic alert box. The
162 // title moves into the title bar; the "unsaved" dot stays with the message,
163 // which is where the state it marks actually is.
164 AestraUI::DialogChrome chrome;
165 chrome.panel = m_dialogRect;
166 chrome.title = m_title;
167 chrome.showClose = false; // Cancel is the dismissal; two would be ambiguous.
168 const AestraUI::NUIRect content = AestraUI::drawDialogChrome(renderer, chrome);
169
170 const float padX = content.x + theme.spacingL;
171 // Message and its dot share one centre line: the text optically by cap height, the dot
172 // geometrically (it used a hand-rolled messageY + 5).
173 const float messageY = std::round(renderer.calculateOpticalTextY(m_messageCentreY, theme.fontSizeM));
174 const float dotR = 4.0f;
175 if (!m_isConfirmMode) {
176 renderer.fillCircle({padX + dotR, m_messageCentreY}, dotR, theme.warning);
177 }
178 renderer.drawText(m_message,
179 AestraUI::NUIPoint(m_isConfirmMode ? padX : padX + dotR * 2.0f + 10.0f, messageY),
180 theme.fontSizeM, theme.textSecondary);
181
182 // Divider above the button row.
183 const float dividerY = std::round(m_saveButtonRect.y - 16.0f); // kDividerGap in calculateLayout
184 renderer.drawLine({m_dialogRect.x + theme.spacingM, dividerY},
185 {m_dialogRect.right() - theme.spacingM, dividerY},
186 theme.layout.dividerWidth, theme.borderSubtle);
187
188 // --- Buttons ---
189 // Cancel: ghost (border only).
190 const bool cancelPressed = m_pressedButton == DialogResponse::Cancel;
191 renderer.fillRoundedRect(m_cancelButtonRect, theme.radiusM,
192 cancelPressed ? theme.pressed : (m_cancelHovered ? theme.hover : theme.buttonBgDefault));
193 renderer.strokeRoundedRect(m_cancelButtonRect, theme.radiusM, theme.layout.dividerWidth,
194 theme.borderStrong);
195 renderer.drawTextCentered("Cancel", m_cancelButtonRect, theme.fontSizeM,
196 m_cancelHovered ? theme.textPrimary : theme.textSecondary);
197
198 if (m_isConfirmMode) {
199 // Confirm: primary fill, mirrors the Save slot.
200 const bool confirmPressed = m_pressedButton == DialogResponse::Confirm;
201 renderer.fillRoundedRect(m_saveButtonRect, theme.radiusM,
202 confirmPressed ? theme.primaryPressed
203 : (m_saveHovered ? theme.primaryHover : theme.primary));
204 renderer.drawTextCentered(m_confirmLabel, m_saveButtonRect, theme.fontSizeM, theme.textOnPrimary);
205 } else {
206 // Don't Save: subtle filled + border.
207 const bool dontSavePressed = m_pressedButton == DialogResponse::DontSave;
208 renderer.fillRoundedRect(m_dontSaveButtonRect, theme.radiusM,
209 dontSavePressed ? theme.pressed
210 : (m_dontSaveHovered ? theme.hover : theme.buttonBgDefault));
211 renderer.strokeRoundedRect(m_dontSaveButtonRect, theme.radiusM, theme.layout.dividerWidth,
212 theme.borderStrong);
213 renderer.drawTextCentered("Don't Save", m_dontSaveButtonRect, theme.fontSizeM, theme.error);
214
215 const bool savePressed = m_pressedButton == DialogResponse::Save;
216 renderer.fillRoundedRect(m_saveButtonRect, theme.radiusM,
217 savePressed ? theme.primaryPressed : (m_saveHovered ? theme.primaryHover : theme.primary));
218 renderer.drawTextCentered("Save", m_saveButtonRect, theme.fontSizeM, theme.textOnPrimary);
219 }
220
221 // Keyboard focus highlight: a faint accent ring around the focused button
222 // (Left/Right move it, Enter activates it).
223 const AestraUI::NUIRect focusRect = m_isConfirmMode
224 ? (m_focusIndex == 0 ? m_cancelButtonRect : m_saveButtonRect)
225 : (m_focusIndex == 0) ? m_cancelButtonRect : (m_focusIndex == 1) ? m_dontSaveButtonRect : m_saveButtonRect;
226 AestraUI::NUIRect ring = focusRect;
227 ring.x -= 2.0f;
228 ring.y -= 2.0f;
229 ring.width += 4.0f;
230 ring.height += 4.0f;
231 renderer.strokeRoundedRect(ring, theme.radiusL, 1.5f, theme.focusRing);
232}

234bool ConfirmationDialog::onMouseEvent(const AestraUI::NUIMouseEvent& event) {
235 if (!m_isVisible) {
236 return false;
237 }
238
239 calculateLayout();
240
241 float mouseX = event.position.x;
242 float mouseY = event.position.y;
243
244 // Update hover states; hovering also moves keyboard focus so the two stay in sync.
245 m_cancelHovered = m_cancelButtonRect.contains(mouseX, mouseY);
246 m_dontSaveHovered = !m_isConfirmMode && m_dontSaveButtonRect.contains(mouseX, mouseY);
247 m_saveHovered = m_saveButtonRect.contains(mouseX, mouseY);
248 if (m_isConfirmMode) {
249 if (m_cancelHovered) m_focusIndex = 0;
250 else if (m_saveHovered) m_focusIndex = 1;
251 } else if (m_cancelHovered) m_focusIndex = 0;
252 else if (m_dontSaveHovered) m_focusIndex = 1;
253 else if (m_saveHovered) m_focusIndex = 2;
254
255 // Arm on press, fire on release. Consuming BOTH the press and the release while
256 // the dialog is visible prevents the release from clicking through to the app
257 // behind once the dialog hides (the old code responded on press, so the release
258 // reached whatever was underneath).
259 if (event.button == AestraUI::NUIMouseButton::Left) {
260 if (event.pressed) {
261 if (m_saveHovered) m_pressedButton = m_isConfirmMode ? DialogResponse::Confirm : DialogResponse::Save;
262 else if (m_dontSaveHovered) m_pressedButton = DialogResponse::DontSave;
263 else if (m_cancelHovered) m_pressedButton = DialogResponse::Cancel;
264 else if (!m_dialogRect.contains(mouseX, mouseY)) m_pressedButton = DialogResponse::Cancel; // click-outside
265 else m_pressedButton = DialogResponse::None;
266 return true;
267 }
268 if (event.released) {
269 const DialogResponse armed = m_pressedButton;
270 m_pressedButton = DialogResponse::None;
271 // Only fire if the release lands on the same button that was pressed.
272 const bool overArmed = m_isConfirmMode
273 ? (armed == DialogResponse::Confirm && m_saveHovered) ||
274 (armed == DialogResponse::Cancel &&
275 (m_cancelHovered || !m_dialogRect.contains(mouseX, mouseY)))
276 : (armed == DialogResponse::Save && m_saveHovered) ||
277 (armed == DialogResponse::DontSave && m_dontSaveHovered) ||
278 (armed == DialogResponse::Cancel &&
279 (m_cancelHovered || !m_dialogRect.contains(mouseX, mouseY)));
280 if (armed != DialogResponse::None && overArmed) {
281 handleResponse(armed);
282 }
283 return true;
284 }
285 }
286
287 // Consume all mouse events while dialog is visible
288 return true;
289}

291bool ConfirmationDialog::onKeyEvent(const AestraUI::NUIKeyEvent& event) {
292 if (!m_isVisible) {
293 return false;
294 }
295
296 if (event.pressed) {
297 // Escape = Cancel (regardless of focus).
298 if (event.keyCode == AestraUI::NUIKeyCode::Escape) {
299 handleResponse(DialogResponse::Cancel);
300 return true;
301 }
302
303 // Left/Right move the focus highlight across the button row.
304 if (event.keyCode == AestraUI::NUIKeyCode::Left) {
305 const int buttonCount = m_isConfirmMode ? 2 : 3;
306 m_focusIndex = (m_focusIndex - 1 + buttonCount) % buttonCount; // wrap left
307 return true;
308 }
309 if (event.keyCode == AestraUI::NUIKeyCode::Right) {
310 const int buttonCount = m_isConfirmMode ? 2 : 3;
311 m_focusIndex = (m_focusIndex + 1) % buttonCount; // wrap right
312 return true;
313 }
314
315 // Enter/Tab-less activate: fire the focused button.
316 if (event.keyCode == AestraUI::NUIKeyCode::Enter) {
317 handleResponse(responseForFocus(m_focusIndex));
318 return true;
319 }
320 }
321
322 // Consume all key events while dialog is visible
323 return true;
324}
