2// The audio of one block, shared between the app and the plugin helper (V82-H7 follow-up).
3//
4// Hex over the pipe cost two characters per byte each way (8 KB out and 8 KB back for 512 stereo frames), read one byte
5// per syscall, and the block came back late often enough to play dry. Now the samples and the MIDI sit in one shared
6// region; the pipe only carries a short PROCESSSHM line and an "OK" back. The pipe stays the doorbell on purpose: it
7// keeps the timeouts, the crash detection and the order of commands exactly as they were, and its write/read pair is
8// the memory barrier, so the region needs no atomics.
9//
10// The app makes the region and owns the sizes; the helper only maps it. Nothing the helper writes into it is trusted:
11// the app reads back only its own sample count. Layout: header | input floats | output floats | MIDI bytes.

44 static size_t bytesFor(size_t maxSamples, size_t maxMidiBytes) {
45 return kHeaderBytes + (2 * maxSamples * sizeof(float)) + maxMidiBytes;
46 }

49 bool create(const std::string& name, size_t maxSamples, size_t maxMidiBytes) {
50 close();
51 const size_t bytes = bytesFor(maxSamples, maxMidiBytes);
52#ifdef _WIN32
53 m_mapping = CreateFileMappingA(INVALID_HANDLE_VALUE, nullptr, PAGE_READWRITE, 0, static_cast<DWORD>(bytes),
54 name.c_str());
55 if (!m_mapping || GetLastError() == ERROR_ALREADY_EXISTS) {
56 close();
57 return false;
58 }
59#else
60 const int fd = shm_open(name.c_str(), O_CREAT | O_EXCL | O_RDWR, 0600);
61 if (fd < 0)
62 return false;
63 m_name = name;
64 if (ftruncate(fd, static_cast<off_t>(bytes)) != 0) {
65 ::close(fd);
66 close();
67 return false;
68 }
69 m_fd = fd;
70#endif
71 if (!map(bytes)) {
72 close();
73 return false;
74 }
75 auto* h = header();
76 h->magic = kMagic;
77 h->version = kVersion;
78 h->maxSamples = static_cast<uint32_t>(maxSamples);
79 h->maxMidiBytes = static_cast<uint32_t>(maxMidiBytes);
80 setSizes(maxSamples, maxMidiBytes);
81 return true;
82 }
83
85 bool open(const std::string& name, size_t maxSamples, size_t maxMidiBytes) {
86 close();
87 const size_t bytes = bytesFor(maxSamples, maxMidiBytes);
88#ifdef _WIN32
89 m_mapping = OpenFileMappingA(FILE_MAP_ALL_ACCESS, FALSE, name.c_str());
90 if (!m_mapping)
91 return false;
92#else
93 const int fd = shm_open(name.c_str(), O_RDWR, 0600);
94 if (fd < 0)
95 return false;
96 struct stat st {};
97 if (fstat(fd, &st) != 0 || static_cast<size_t>(st.st_size) < bytes) {
98 ::close(fd);
99 return false;
100 }
101 m_fd = fd;
102#endif
103 if (!map(bytes)) {
104 close();
105 return false;
106 }
107 const auto* h = header();
108 if (h->magic != kMagic || h->version != kVersion || h->maxSamples != maxSamples ||
109 h->maxMidiBytes != maxMidiBytes) {
110 close();
111 return false;
112 }
113 setSizes(maxSamples, maxMidiBytes);
114 return true;
115 }

128 void close() {
129 unlinkName();
130#ifdef _WIN32
131 if (m_base)
132 UnmapViewOfFile(m_base);
133 if (m_mapping)
134 CloseHandle(m_mapping);
135 m_mapping = nullptr;
136#else
137 if (m_base)
138 munmap(m_base, m_bytes);
139 if (m_fd >= 0)
140 ::close(m_fd);
141 m_fd = -1;
142#endif
143 m_base = nullptr;
144 m_bytes = 0;
145 m_maxSamples = 0;
146 m_maxMidiBytes = 0;
147 }

152 float* input() { return reinterpret_cast<float*>(bytes() + kHeaderBytes); }
153 float* output() { return input() + m_maxSamples; }
154 uint8_t* midi() { return reinterpret_cast<uint8_t*>(output() + m_maxSamples); }

168private:
169 struct Header {
170 uint32_t magic;
171 uint32_t version;
172 uint32_t maxSamples;

175 static constexpr size_t kHeaderBytes = 64; // keeps the samples cache-line aligned
176
177 bool map(size_t bytes) {
178#ifdef _WIN32
179 m_base = MapViewOfFile(m_mapping, FILE_MAP_ALL_ACCESS, 0, 0, bytes);
180#else
181 void* p = mmap(nullptr, bytes, PROT_READ | PROT_WRITE, MAP_SHARED, m_fd, 0);
182 m_base = p == MAP_FAILED ? nullptr : p;
183 // The mapping outlives the descriptor, and a descriptor left open in the helper is one more handle untrusted
184 // plugin code could reach (the sandbox test checks the helper holds only 0, 1 and 2).
185 ::close(m_fd);
186 m_fd = -1;
187#endif
188 m_bytes = m_base ? bytes : 0;
189 return m_base != nullptr;
190 }
191 void setSizes(size_t maxSamples, size_t maxMidiBytes) {
192 m_maxSamples = maxSamples;
193 m_maxMidiBytes = maxMidiBytes;
194 }
195 Header* header() { return reinterpret_cast<Header*>(m_base); }
196 uint8_t* bytes() { return static_cast<uint8_t*>(m_base); }
197
198 void* m_base = nullptr;
199 size_t m_bytes = 0;
200 size_t m_maxSamples = 0;
201 size_t m_maxMidiBytes = 0;
202#ifdef _WIN32
203 HANDLE m_mapping = nullptr;
204#else
205 int m_fd = -1;
206 std::string m_name;
207#endif
