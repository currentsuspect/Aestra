20RecoveryInfo::Info AestraApp::recoveryInfoFor(const std::string& originalProjectPath, const std::vector<std::string>& candidates) {
21 std::vector<uint64_t> times;
22 for (const auto& path : candidates) {
23 std::error_code ec;
24 const auto ftime = std::filesystem::last_write_time(path, ec);
25 if (ec) continue;
26 const auto sys = std::chrono::time_point_cast<std::chrono::milliseconds>(ftime - std::filesystem::file_time_type::clock::now() + std::chrono::system_clock::now());
27 times.push_back(static_cast<uint64_t>(sys.time_since_epoch().count()));
28 }
29 const auto manifest = originalProjectPath.empty() ? TakeManager::Manifest{} : TakeManager::loadManifest(originalProjectPath);
30 return RecoveryInfo::build(manifest, std::move(times));
31}

34static void archiveAutosave(const std::string& autosavePath, const std::string& projectPath) {
35 if (projectPath.empty() || !std::filesystem::exists(autosavePath)) return;
36 const std::filesystem::path dir = ProjectSerializer::getHistoryDirectory(projectPath);
37 std::error_code ec;
38 std::filesystem::create_directories(dir, ec);
39 const auto now = std::chrono::system_clock::to_time_t(std::chrono::system_clock::now());
40 std::tm tm{};
41#ifdef _WIN32
42 localtime_s(&tm, &now);
43#else
44 localtime_r(&now, &tm);
45#endif
46 char stamp[32];
47 std::strftime(stamp, sizeof stamp, "%Y%m%d_%H%M%S", &tm);
48 std::filesystem::copy_file(autosavePath, dir / (std::filesystem::path(projectPath).stem().string() + "_rescued_" + stamp + ".aes"),
49 std::filesystem::copy_options::overwrite_existing, ec);
50 if (ec) Log::warning("[Recovery] Could not keep the autosave as a rescued session: " + ec.message());
51}

53void AestraApp::onRecoveryResponse(Aestra::RecoveryResponse response, const std::string& autosavePath, const std::string& projectPath,
54 bool hasRequestedProject, const std::string& originalProjectPath,
55 const std::vector<std::string>& recoveryCandidates) {
56 m_recoveryHandled = true;
57 m_pendingAutosavePath.clear();
58 if (response == Aestra::RecoveryResponse::Recover || response == Aestra::RecoveryResponse::RecoverAsAlt) {
59 beginProjectLoadAttempt();
60 auto selected = ProjectSerializer::loadFirstValid(recoveryCandidates, m_content ? m_content->getTrackManager() : nullptr, originalProjectPath);
61 auto result = std::move(selected.result);
62 recordProjectLoadAttempt(result, ProjectLoadSource::Recovery);
63 if (result.ok) {
64 result = applyLoadedProject(selected.loadedPath, ProjectLoadSource::Recovery, originalProjectPath, std::move(result));
65 if (m_content && m_content->getTrackManager()) m_content->getTrackManager()->markModified();
66 m_autoSaveManager.markDirty();
67 updateWindowTitle();
68 Log::info("[Recovery] Autosave recovered successfully from: " + selected.loadedPath);
69 // As an alt: the recovered state becomes its own branch, and the saved branch is left as it was.
70 if (response == Aestra::RecoveryResponse::RecoverAsAlt && !m_documentState.canonicalPath().empty()) {
71 if (!createTakeFromCurrentProject(true)) Log::warning("[Recovery] Recovered, but could not save it as an alt");
72 refreshVersionChip();
73 }
74 } else {
75 Log::error("[Recovery] Failed to load autosave");
76 if (m_content) m_content->resetToDefaultProject();
77 clearProjectLoadReport();
78 m_documentState.startUntitled(autosavePathOrEmpty());
79 reinitAutosaveManager();
80 syncRecordingProjectPath(m_content, m_documentState.canonicalPath());
81 }
82 } else if (response == Aestra::RecoveryResponse::Discard || response == Aestra::RecoveryResponse::OpenLastSaved) {
83 if (response == Aestra::RecoveryResponse::OpenLastSaved) archiveAutosave(autosavePath, originalProjectPath);
84 std::error_code ec;
85 std::filesystem::remove(autosavePath, ec);
86 std::filesystem::remove(getRecoveryMarkerPath(autosavePath), ec);
87 if (ec) Log::warning("[Recovery] Failed to remove autosave: " + ec.message());
88 else Log::info("[Recovery] Autosave discarded");
89 if (hasRequestedProject) {
90 if (!loadProjectFromPath(projectPath).ok) Log::error("Failed to load requested project after discarding recovery: " + projectPath);
91 } else {
92 if (m_content) m_content->resetToDefaultProject();
93 clearProjectLoadReport();
94 m_documentState.startUntitled(autosavePathOrEmpty());
95 reinitAutosaveManager();
96 syncRecordingProjectPath(m_content, m_documentState.canonicalPath());
97 updateWindowTitle();
98 }
99 }
