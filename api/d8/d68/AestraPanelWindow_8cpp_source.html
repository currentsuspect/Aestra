43float AestraPanelWindow::knobDragStep(const NUIMouseEvent& event, float rangePx) const
44{
45 // Shift = surgical precision (quarter speed). Held state is read per frame
46 // so it can be toggled mid-drag. Default (no modifier) speed is unchanged.
47 constexpr float kFineDragScale = 0.25f;
48 const float fine = (event.modifiers & NUIModifiers::Shift) ? kFineDragScale : 1.0f;
49 return (-event.delta.y / rangePx) * fine;
50}

52bool AestraPanelWindow::handleKnobDrag(const NUIMouseEvent& event, const KnobTarget* targets, size_t count,
53 float rangePx, const std::function<float(uint32_t)>& get,
54 const std::function<void(uint32_t, float)>& set) {
55 if (m_knobDragParam >= 0) {
56 // Only the button that grabbed the knob lets go of it. A synthetic
57 // release (focus loss, cancelled capture) always does, so nothing is
58 // left captured.
59 if (event.released && (event.button == NUIMouseButton::Left || event.synthetic)) {
60 endKnobCapture();
61 m_knobDragParam = -1;
62 return true;
63 }
64 if (event.button == NUIMouseButton::None) {
65 // Service-owned frame delta (up = increase), accumulated into the
66 // current value, so total travel matches an absolute mapping.
67 const uint32_t id = static_cast<uint32_t>(m_knobDragParam);
68 set(id, std::clamp(get(id) + knobDragStep(event, rangePx), 0.0f, 1.0f));
69 setDirty();
70 return true;
71 }
72 }
73 if (event.pressed && event.button == NUIMouseButton::Left) {
74 for (size_t i = 0; i < count; ++i) {
75 const NUIRect& r = targets[i].rect;
76 if (r.contains(event.position)) {
77 m_knobDragParam = static_cast<int>(targets[i].paramId);
78 beginKnobCapture({r.x + r.width * 0.5f, r.y + r.height * 0.5f}, event.position);
79 return true;
80 }
81 }
82 }
83 return false;
84}

133void AestraPanelWindow::drawTitleBar(NUIRenderer& renderer)
134{
135 auto b = getBounds();
136 if (b.isEmpty()) return;
137
138 // Bottom border only — body fillRoundedRect already covered the top area
139 renderer.drawLine({b.x, b.y + TITLE_BAR_H - 0.5f},
140 {b.right(), b.y + TITLE_BAR_H - 0.5f},
141 0.5f, m_border);
142
143 // Title text — 12px, left-aligned at x=12, vertically centered via font metrics
144 if (!m_title.empty()) {
145 NUIRect textRect{b.x, b.y, b.width, TITLE_BAR_H};
146 float titleBaseline = renderer.calculateTextY(textRect, 12.0f);
147 renderer.drawText(m_title,
148 {b.x + 12.0f, titleBaseline},
149 12.0f, m_textSecondary);
150 }
151
152 // Badge pill (right of title)
153 if (!m_badgeText.empty()) {
154 auto& theme = NUIThemeManager::getInstance();
155 float badgeW = renderer.measureText(m_badgeText, 11.0f).width + 20.0f;
156 float titleW = m_title.empty() ? 0.0f : renderer.measureText(m_title, 12.0f).width;
157 NUIRect badgeRect{b.x + 12.0f + titleW + 8.0f,
158 b.y + (TITLE_BAR_H - 18.0f) * 0.5f,
159 badgeW, 18.0f};
160 renderer.fillRoundedRect(badgeRect, 9.0f, theme.getColor("success").withAlpha(0.18f));
161 renderer.strokeRoundedRect(badgeRect, 9.0f, 1.0f, theme.getColor("success").withAlpha(0.42f));
162 renderer.drawTextCentered(m_badgeText, badgeRect, 11.0f, theme.getColor("success"));
163 }
164
165 // Close button — right-aligned at x=width-28
166 NUIRect closeBtn{b.right() - 28.0f, b.y + 4.0f, 24.0f, 24.0f};
167 if (m_closeHovered || m_closePressed) {
168 renderer.fillRoundedRect(closeBtn, 6.0f, m_closeHover.withAlpha(m_closePressed ? 0.35f : 0.22f));
169 }
170 float cx = closeBtn.x + closeBtn.width * 0.5f;
171 float cy = closeBtn.y + closeBtn.height * 0.5f;
172 float d = 5.0f;
173 NUIColor xColor = m_closeHovered ? m_closeActive : m_textTertiary;
174 renderer.drawLine({cx - d, cy - d}, {cx + d, cy + d}, 1.5f, xColor);
175 renderer.drawLine({cx + d, cy - d}, {cx - d, cy + d}, 1.5f, xColor);
176}

178void AestraPanelWindow::onRender(NUIRenderer& renderer)
179{
180 auto b = getBounds();
181 if (b.isEmpty()) return;
182
183 // Window body
184 renderer.fillRoundedRect(b, kRadius, m_bg);
185
186 // Content area (clipped, global coordinates)
187 NUIRect content = getContentRect();
188 if (!content.isEmpty()) {
189 renderer.setClipRect(content);
190 drawContent(renderer, content);
191 renderer.clearClipRect();
192 }
193
194 // Title bar on top of content
195 drawTitleBar(renderer);
196
197 // Border stroke on top of everything
198 renderer.strokeRoundedRect(b, kRadius, 1.0f, m_border);
199}

201bool AestraPanelWindow::onMouseEvent(const NUIMouseEvent& event)
202{
203 auto b = getBounds();
204
205 // Close on outside click
206 if (m_closeOnOutsideClick && event.pressed && event.button == NUIMouseButton::Left
207 && !b.contains(event.position) && !m_isDraggingWindow) {
208 if (m_onClose) m_onClose();
209 return false;
210 }
211
212 // Close button interaction
213 bool overClose = hitTestCloseButton(event.position.x, event.position.y);
214 if (event.pressed && event.button == NUIMouseButton::Left && overClose) {
215 m_closePressed = true;
216 setDirty(true);
217 return true;
218 }
219 if (event.released && event.button == NUIMouseButton::Left && m_closePressed) {
220 m_closePressed = false;
221 setDirty(true);
222 if (overClose && m_onClose) {
223 m_onClose();
224 }
225 return true;
226 }
227 if (overClose != m_closeHovered) {
228 m_closeHovered = overClose;
229 setDirty(true);
230 }
231
232 if (event.pressed && event.button == NUIMouseButton::Right && m_onMenuRequested && !overClose &&
233 hitTestTitleBar(event.position.x, event.position.y)) {
234 m_onMenuRequested(event.position);
235 return true;
236 }
237
238 // Window drag from title bar (not when docked)
239 if (event.pressed && event.button == NUIMouseButton::Left && !m_locked &&
240 hitTestTitleBar(event.position.x, event.position.y) && !overClose) {
241 m_isDraggingWindow = true;
242 m_dragStartPos = event.position;
243 m_windowStartPos = {b.x, b.y};
244 return true;
245 }
246 if (m_isDraggingWindow) {
247 if (event.released && event.button == NUIMouseButton::Left) {
248 if (m_enforceParentBounds) {
249 m_userPositioned = true;
250 }
251 onDragEnd();
252 if (m_onWindowDragEnd) {
253 m_onWindowDragEnd();
254 }
255 m_isDraggingWindow = false;
256 return true;
257 }
258 float dx = event.position.x - m_dragStartPos.x;
259 float dy = event.position.y - m_dragStartPos.y;
260 float newX = m_windowStartPos.x + dx;
261 float newY = m_windowStartPos.y + dy;
262 if (m_enforceParentBounds) {
263 if (auto* parent = getParent()) {
264 const auto pb = parent->getBounds();
265 const float margin = 14.0f;
266 newX = std::clamp(newX, pb.x + margin, std::max(pb.x + margin, pb.right() - margin - b.width));
267 newY = std::clamp(newY, pb.y + margin, std::max(pb.y + margin, pb.bottom() - margin - b.height));
268 }
269 }
270 setBounds(NUIRect(newX, newY, b.width, b.height));
271 return true;
272 }
273
274 return false;
275}

277void AestraPanelWindow::onUpdate(double deltaTime)
278{
279 NUIComponent::onUpdate(deltaTime);
280
281 if (!m_enforceParentBounds || isDraggingWindow()) return;
282
283 if (auto* parent = getParent()) {
284 const auto pb = parent->getBounds();
285 const bool parentChanged = !m_haveParentSnapshot ||
286 std::abs(pb.x - m_lastParentBounds.x) > 0.5f ||
287 std::abs(pb.y - m_lastParentBounds.y) > 0.5f ||
288 std::abs(pb.width - m_lastParentBounds.width) > 0.5f ||
289 std::abs(pb.height - m_lastParentBounds.height) > 0.5f;
290 if (parentChanged) {
291 m_lastParentBounds = pb;
292 m_haveParentSnapshot = true;
293 enforceBoundsInParent();
294 } else if (!m_haveParentSnapshot) {
295 m_lastParentBounds = pb;
296 m_haveParentSnapshot = true;
297 }
298 }

301void AestraPanelWindow::enforceBoundsInParent(float safeMargin)
302{
303 auto* parent = getParent();
304 if (!parent) return;
305
306 const NUIRect pb = parent->getBounds();
307 if (pb.isEmpty()) return;
308
309 auto b = getBounds();
310 float minX = pb.x + safeMargin;
311 float minY = pb.y + safeMargin;
312 float maxX = std::max(minX, pb.right() - safeMargin - b.width);
313 float maxY = std::max(minY, pb.bottom() - safeMargin - b.height);
314 float targetX = std::clamp(b.x, minX, maxX);
315 float targetY = std::clamp(b.y, minY, maxY);
316
317 if (std::abs(b.x - targetX) > 0.5f || std::abs(b.y - targetY) > 0.5f) {
318 setPosition(targetX, targetY);
319 }
320}
