41float AestraPanelWindow::knobDragStep(const NUIMouseEvent& event, float rangePx) const
42{
43 // Shift = surgical precision (quarter speed). Held state is read per frame
44 // so it can be toggled mid-drag. Default (no modifier) speed is unchanged.
45 constexpr float kFineDragScale = 0.25f;
46 const float fine = (event.modifiers & NUIModifiers::Shift) ? kFineDragScale : 1.0f;
47 return (-event.delta.y / rangePx) * fine;
48}

97void AestraPanelWindow::drawTitleBar(NUIRenderer& renderer)
98{
99 auto b = getBounds();
100 if (b.isEmpty()) return;
101
102 // Bottom border only — body fillRoundedRect already covered the top area
103 renderer.drawLine({b.x, b.y + TITLE_BAR_H - 0.5f},
104 {b.right(), b.y + TITLE_BAR_H - 0.5f},
105 0.5f, m_border);
106
107 // Title text — 12px, left-aligned at x=12, vertically centered via font metrics
108 if (!m_title.empty()) {
109 NUIRect textRect{b.x, b.y, b.width, TITLE_BAR_H};
110 float titleBaseline = renderer.calculateTextY(textRect, 12.0f);
111 renderer.drawText(m_title,
112 {b.x + 12.0f, titleBaseline},
113 12.0f, m_textSecondary);
114 }
115
116 // Badge pill (right of title)
117 if (!m_badgeText.empty()) {
118 auto& theme = NUIThemeManager::getInstance();
119 float badgeW = renderer.measureText(m_badgeText, 11.0f).width + 20.0f;
120 float titleW = m_title.empty() ? 0.0f : renderer.measureText(m_title, 12.0f).width;
121 NUIRect badgeRect{b.x + 12.0f + titleW + 8.0f,
122 b.y + (TITLE_BAR_H - 18.0f) * 0.5f,
123 badgeW, 18.0f};
124 renderer.fillRoundedRect(badgeRect, 9.0f, theme.getColor("success").withAlpha(0.18f));
125 renderer.strokeRoundedRect(badgeRect, 9.0f, 1.0f, theme.getColor("success").withAlpha(0.42f));
126 renderer.drawTextCentered(m_badgeText, badgeRect, 11.0f, theme.getColor("success"));
127 }
128
129 // Close button — right-aligned at x=width-28
130 NUIRect closeBtn{b.right() - 28.0f, b.y + 4.0f, 24.0f, 24.0f};
131 if (m_closeHovered || m_closePressed) {
132 renderer.fillRoundedRect(closeBtn, 6.0f, m_closeHover.withAlpha(m_closePressed ? 0.35f : 0.22f));
133 }
134 float cx = closeBtn.x + closeBtn.width * 0.5f;
135 float cy = closeBtn.y + closeBtn.height * 0.5f;
136 float d = 5.0f;
137 NUIColor xColor = m_closeHovered ? m_closeActive : m_textTertiary;
138 renderer.drawLine({cx - d, cy - d}, {cx + d, cy + d}, 1.5f, xColor);
139 renderer.drawLine({cx + d, cy - d}, {cx - d, cy + d}, 1.5f, xColor);
140}

142void AestraPanelWindow::onRender(NUIRenderer& renderer)
143{
144 auto b = getBounds();
145 if (b.isEmpty()) return;
146
147 // Window body
148 renderer.fillRoundedRect(b, kRadius, m_bg);
149
150 // Content area (clipped, global coordinates)
151 NUIRect content = getContentRect();
152 if (!content.isEmpty()) {
153 renderer.setClipRect(content);
154 drawContent(renderer, content);
155 renderer.clearClipRect();
156 }
157
158 // Title bar on top of content
159 drawTitleBar(renderer);
160
161 // Border stroke on top of everything
162 renderer.strokeRoundedRect(b, kRadius, 1.0f, m_border);
163}

165bool AestraPanelWindow::onMouseEvent(const NUIMouseEvent& event)
166{
167 auto b = getBounds();
168
169 // Close on outside click
170 if (m_closeOnOutsideClick && event.pressed && event.button == NUIMouseButton::Left
171 && !b.contains(event.position) && !m_isDraggingWindow) {
172 if (m_onClose) m_onClose();
173 return false;
174 }
175
176 // Close button interaction
177 bool overClose = hitTestCloseButton(event.position.x, event.position.y);
178 if (event.pressed && event.button == NUIMouseButton::Left && overClose) {
179 m_closePressed = true;
180 setDirty(true);
181 return true;
182 }
183 if (event.released && event.button == NUIMouseButton::Left && m_closePressed) {
184 m_closePressed = false;
185 setDirty(true);
186 if (overClose && m_onClose) {
187 m_onClose();
188 }
189 return true;
190 }
191 if (overClose != m_closeHovered) {
192 m_closeHovered = overClose;
193 setDirty(true);
194 }
195
196 // Window drag from title bar
197 if (event.pressed && event.button == NUIMouseButton::Left && hitTestTitleBar(event.position.x, event.position.y) && !overClose) {
198 m_isDraggingWindow = true;
199 m_dragStartPos = event.position;
200 m_windowStartPos = {b.x, b.y};
201 return true;
202 }
203 if (m_isDraggingWindow) {
204 if (event.released && event.button == NUIMouseButton::Left) {
205 if (m_enforceParentBounds) {
206 m_userPositioned = true;
207 }
208 onDragEnd();
209 m_isDraggingWindow = false;
210 return true;
211 }
212 float dx = event.position.x - m_dragStartPos.x;
213 float dy = event.position.y - m_dragStartPos.y;
214 float newX = m_windowStartPos.x + dx;
215 float newY = m_windowStartPos.y + dy;
216 if (m_enforceParentBounds) {
217 if (auto* parent = getParent()) {
218 const auto pb = parent->getBounds();
219 const float margin = 14.0f;
220 newX = std::clamp(newX, pb.x + margin, std::max(pb.x + margin, pb.right() - margin - b.width));
221 newY = std::clamp(newY, pb.y + margin, std::max(pb.y + margin, pb.bottom() - margin - b.height));
222 }
223 }
224 setBounds(NUIRect(newX, newY, b.width, b.height));
225 return true;
226 }
227
228 return false;
229}

231void AestraPanelWindow::onUpdate(double deltaTime)
232{
233 NUIComponent::onUpdate(deltaTime);
234
235 if (!m_enforceParentBounds || isDraggingWindow()) return;
236
237 if (auto* parent = getParent()) {
238 const auto pb = parent->getBounds();
239 const bool parentChanged = !m_haveParentSnapshot ||
240 std::abs(pb.x - m_lastParentBounds.x) > 0.5f ||
241 std::abs(pb.y - m_lastParentBounds.y) > 0.5f ||
242 std::abs(pb.width - m_lastParentBounds.width) > 0.5f ||
243 std::abs(pb.height - m_lastParentBounds.height) > 0.5f;
244 if (parentChanged) {
245 m_lastParentBounds = pb;
246 m_haveParentSnapshot = true;
247 enforceBoundsInParent();
248 } else if (!m_haveParentSnapshot) {
249 m_lastParentBounds = pb;
250 m_haveParentSnapshot = true;
251 }
252 }
253}

255void AestraPanelWindow::enforceBoundsInParent(float safeMargin)
256{
257 auto* parent = getParent();
258 if (!parent) return;
259
260 const NUIRect pb = parent->getBounds();
261 if (pb.isEmpty()) return;
262
263 auto b = getBounds();
264 float minX = pb.x + safeMargin;
265 float minY = pb.y + safeMargin;
266 float maxX = std::max(minX, pb.right() - safeMargin - b.width);
267 float maxY = std::max(minY, pb.bottom() - safeMargin - b.height);
268 float targetX = std::clamp(b.x, minX, maxX);
269 float targetY = std::clamp(b.y, minY, maxY);
270
271 if (std::abs(b.x - targetX) > 0.5f || std::abs(b.y - targetY) > 0.5f) {
272 setPosition(targetX, targetY);
273 }
274}
