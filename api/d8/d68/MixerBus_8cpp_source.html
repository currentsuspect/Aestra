17void MixerBus::process(float* buffer, uint32_t numFrames) {
18 // Load targets (atomic)
19 const float targetGain = m_gain.load(std::memory_order_acquire);
20 const float targetPan = m_pan.load(std::memory_order_acquire);
21 const float targetWidth = m_width.load(std::memory_order_acquire);
22 const bool muted = m_muted.load(std::memory_order_acquire);
23
24 // If muted, clear buffer and reset smoothing state to prevent jumps on unmute
25 if (muted) {
26 clear(buffer, numFrames);
27 m_currentGain = targetGain;
28 m_currentPan = targetPan;
29 m_currentWidth = targetWidth;
30 return;
31 }
32
33 // Handle mono vs stereo
34 if (m_numChannels == 1) {
35 // Mono: just apply gain with smoothing (Width ignored for mono source)
36 const float gainStart = m_currentGain;
37 const float gainEnd = targetGain;
38 const float gainDelta = (gainEnd - gainStart) / static_cast<float>(numFrames);
39
40 float current = gainStart;
41 for (uint32_t i = 0; i < numFrames; ++i) {
42 buffer[i] *= current;
43 current += gainDelta;
44 }
45 } else if (m_numChannels == 2) {
46 // Stereo: Apply Width (M/S), then Pan, then Gain
47 const float panStart = m_currentPan;
48 const float panEnd = targetPan;
49 const float gainStart = m_currentGain;
50 const float gainEnd = targetGain;
51 const float widthStart = m_currentWidth;
52 const float widthEnd = targetWidth;
53
54 const float panDelta = (panEnd - panStart) / static_cast<float>(numFrames);
55 const float gainDelta = (gainEnd - gainStart) / static_cast<float>(numFrames);
56 const float widthDelta = (widthEnd - widthStart) / static_cast<float>(numFrames);
57
58 float currentPan = panStart;
59 float currentGain = gainStart;
60 float currentWidth = widthStart;
61
62 for (uint32_t i = 0; i < numFrames; ++i) {
63 const uint32_t leftIdx = i * 2;
64 const uint32_t rightIdx = i * 2 + 1;
65
66 float L = buffer[leftIdx];
67 float R = buffer[rightIdx];
68
69 // 1. Stereo Width (Mid-Side Matrix)
70 if (currentWidth != 1.0f) {
71 // Encode M/S
72 float mid = 0.5f * (L + R);
73 float side = 0.5f * (L - R);
74
75 // Process Side
76 side *= currentWidth;
77
78 // Decode L/R
79 L = mid + side;
80 R = mid - side;
81 }
82
83 // 2. Panning
84 float leftPanGain, rightPanGain;
85 calculatePanGains(currentPan, leftPanGain, rightPanGain);
86
87 L *= leftPanGain;
88 R *= rightPanGain;
89
90 // 3. Master Gain
91 L *= currentGain;
92 R *= currentGain;
93
94 buffer[leftIdx] = L;
95 buffer[rightIdx] = R;
96
97 currentPan += panDelta;
98 currentGain += gainDelta;
99 currentWidth += widthDelta;
100 }
101
102 // Calculate Phase Correlation for this block
103 float sumLR = 0.0f;
104 float sumLL = 0.0f;
105 float sumRR = 0.0f;
106
107 // Optimization: Use a stride to save CPU? Correlation changes slowly.
108 // Let's check every sample for accuracy for now (SIMD would be better later).
109 for (uint32_t i = 0; i < numFrames; ++i) {
110 float l = buffer[i * 2];
111 float r = buffer[i * 2 + 1];
112 sumLR += l * r;
113 sumLL += l * l;
114 sumRR += r * r;
115 }
116
117 float correlation = 0.0f;
118 const float denominator = std::sqrt(sumLL * sumRR);
119
120 if (denominator > 1e-9f) {
121 correlation = sumLR / denominator;
122 } else if (sumLL < 1e-9f && sumRR < 1e-9f) {
123 // Silence = perfectly correlated (technically undefined but 0 or 1 is fine, 0 is safer UI)
124 correlation = 0.0f;
125 }
126
127 // Simple smoothing for the UI readout (1-pole LPF)
128 // alpha ~ 0.1 for fast response but not jittery
129 float prev = m_lastCorrelation.load(std::memory_order_relaxed);
130 correlation = prev * 0.9f + correlation * 0.1f;
131
132 m_lastCorrelation.store(correlation, std::memory_order_relaxed);
133 }
134
135 // Snap to target at end of block
136 m_currentGain = targetGain;
137 m_currentPan = targetPan;
138 m_currentWidth = targetWidth;
139}

141void MixerBus::mixInto(float* output, const float* input, uint32_t numFrames) {
142 // Load targets (atomic)
143 const float targetGain = m_gain.load(std::memory_order_acquire);
144 const float targetPan = m_pan.load(std::memory_order_acquire);
145 const bool muted = m_muted.load(std::memory_order_acquire);
146
147 // If muted, don't mix anything
148 if (muted) {
149 m_currentGain = targetGain;
150 m_currentPan = targetPan;
151 return;
152 }
153
154 // Handle mono vs stereo
155 if (m_numChannels == 1) {
156 const float gainStart = m_currentGain;
157 const float gainEnd = targetGain;
158 const float gainDelta = (gainEnd - gainStart) / static_cast<float>(numFrames);
159
160 float current = gainStart;
161 for (uint32_t i = 0; i < numFrames; ++i) {
162 output[i] += input[i] * current;
163 current += gainDelta;
164 }
165 } else if (m_numChannels == 2) {
166 // Stereo: mix with gain and pan smoothing
167 const float panStart = m_currentPan;
168 const float panEnd = targetPan;
169 const float gainStart = m_currentGain;
170 const float gainEnd = targetGain;
171
172 const float panDelta = (panEnd - panStart) / static_cast<float>(numFrames);
173 const float gainDelta = (gainEnd - gainStart) / static_cast<float>(numFrames);
174
175 float currentPan = panStart;
176 float currentGain = gainStart;
177
178 for (uint32_t i = 0; i < numFrames; ++i) {
179 float leftGain, rightGain;
180 calculatePanGains(currentPan, leftGain, rightGain);
181
182 leftGain *= currentGain;
183 rightGain *= currentGain;
184
185 const uint32_t leftIdx = i * 2;
186 const uint32_t rightIdx = i * 2 + 1;
187
188 output[leftIdx] += input[leftIdx] * leftGain;
189 output[rightIdx] += input[rightIdx] * rightGain;
190
191 currentPan += panDelta;
192 currentGain += gainDelta;
193 }
194 }
195
196 // Snap to target
197 m_currentGain = targetGain;
198 m_currentPan = targetPan;
199}

252void SimpleMixer::process(float* masterOutput, const float** inputs, uint32_t numFrames) {
253 // Clear master output
254 std::memset(masterOutput, 0, numFrames * 2 * sizeof(float)); // Assume stereo master
255
256 // Check if any bus is soloed
257 bool anySoloed = false;
258 for (const auto& bus : m_buses) {
259 if (bus->isSoloed()) {
260 anySoloed = true;
261 break;
262 }
263 }
264
265 // Mix each bus into master
266 for (size_t i = 0; i < m_buses.size(); ++i) {
267 auto& bus = m_buses[i];
268
269 // Skip if muted
270 if (bus->isMuted()) {
271 continue;
272 }
273
274 // If any bus is soloed, only process soloed buses
275 if (anySoloed && !bus->isSoloed()) {
276 continue;
277 }
278
279 // Mix this bus into master
280 if (inputs[i] != nullptr) {
281 bus->mixInto(masterOutput, inputs[i], numFrames);
282 }
283 }
284}
