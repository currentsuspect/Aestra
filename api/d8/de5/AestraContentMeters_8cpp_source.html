3// The transport bar's DSP load (DspLoadReadout): the last audio callback's time over its buffer's duration, read
4// from the engine's telemetry each UI frame. Relaxed atomic loads only; the audio thread writes them.
5#include "MidiInputService.h"
6#include "AestraContent.h"
7#include "AudioEngine.h"
8#include "TransportBar.h"

10void AestraContent::updateDspLoadReadout(double dt) {
11 if (!m_transportBar)
12 return;
13 if (!m_audioEngine) {
14 m_transportBar->setDspLoad(-1.0f);
15 return;
16 }
17 auto& telemetry = m_audioEngine->telemetry();
18 const float percent = Aestra::DspLoadReadout::percentOf(telemetry.lastCallbackNs.load(std::memory_order_relaxed),
19 telemetry.getCallbackBudgetNs());
20 m_transportBar->setDspLoad(m_dspLoad.update(percent, dt));
21}

23// #456: MidiInputService::refreshPorts() re-scans and opens newly connected ports, but nothing called it, so a
24// controller plugged in after startup was never picked up. The UI thread is a control thread, so it polls here, at a
25// low rate: a rescan enumerates the system's MIDI ports, which is cheap but not free.
26void AestraContent::updateFrameServices(double dt) {
27 updateDspLoadReadout(dt);
28 constexpr double kMidiPortPollSeconds = 2.0;
29 m_midiPortPollSeconds += dt;
30 if (m_midiPortPollSeconds >= kMidiPortPollSeconds) {
31 m_midiPortPollSeconds = 0.0;
32 if (m_midiInput) m_midiInput->refreshPorts();
33 }
34}
