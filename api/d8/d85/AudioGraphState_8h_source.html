22struct SmoothedParamD {
23 double current{1.0};
24 double target{1.0};
25 double step{0.0};
26 uint32_t samplesRemaining{0};
27
29 inline double next() {
30 if (samplesRemaining > 0) {
31 current += step;
32 --samplesRemaining;
33 if (samplesRemaining == 0) {
34 current = target;
35 step = 0.0;
36 }
37 }
38 return current;
39 }
40
42 void setTarget(double t) {
43 // Sanitize target to prevent NaN/Inf propagation
44 // Fall back to current value, then to 0.0, to guarantee finite output
45 if (std::isfinite(t)) {
46 target = t;
47 } else if (std::isfinite(current)) {
48 target = current;
49 } else {
50 target = 0.0;
51 }
52 }
53
58 void beginRamp(uint32_t samples) {
59 // Sanitize both current and target before computing step
60 // Mirror setTarget behavior: fall back to the other finite value to avoid clicks
61 if (!std::isfinite(current)) current = std::isfinite(target) ? target : 0.0;
62 if (!std::isfinite(target)) target = current;
63
64 if (samples == 0 || current == target) {
65 current = target;
66 step = 0.0;
67 samplesRemaining = 0;
68 return;
69 }
70 samplesRemaining = samples;
71 step = (target - current) / static_cast<double>(samples);
72
73 // Validate step is finite, otherwise bypass ramp
74 if (!std::isfinite(step)) {
75 step = 0.0;
76 samplesRemaining = 0;
77 current = target;
78 }
79 }
81 void snap() {
82 // Sanitize target before assigning — same contract as setTarget()/beginRamp()
83 if (!std::isfinite(target)) target = std::isfinite(current) ? current : 0.0;
84 current = target;
85 step = 0.0;
86 samplesRemaining = 0;
87 }
88};

244
245 // Track the last active send count so renderGraph() can detect when
246 // sends are added/removed without resizing the pre-sized vectors.
247 size_t lastActiveSendCount{0};
248
249 // === Plugin Delay Compensation ===
250 uint32_t pluginLatencySamples{0}; // Total latency from effect chain
251
252 // Compensation ring (stereo interleaved), allocated ONLY for tracks that
253 // actually carry a nonzero delay: the 128 KiB ring lives inside
254 // CompensationRingDescriptor and is allocated on the FIRST nonzero-delay
255 // publication. The tiny slot below (~200 B) is pre-created on TrackRTState
256 // construction so the shared pointer is immutable for the RT thread.
257 //
258 // This was a std::array<float, 32768> — 128 KiB embedded BY VALUE in every
259 // TrackRTState. m_trackState.reserve(kMaxTracks) therefore asked for a
260 // single 512.8 MiB allocation up front (131,281 B * 4096), which was ~98%
261 // untouched under demand paging until the audio stream's
262 // mlockall(MCL_CURRENT|MCL_FUTURE) faulted every page in and pinned it —
263 // 520 MiB resident and unswappable on a 3.68 GiB machine (#727). Physical
264 // cost is now proportional to the tracks that need compensation and to the
265 // delay they actually require; the 4096 logical track ceiling is unchanged.
266 //
267 // Keeping this struct small also matters for iteration: it dropped from
268 // ~128 KiB to ~200 B.
269 //
270 // The ring + delay are published as an immutable CompensationRingDescriptor
271 // behind an atomic pointer on the slot (see TrackCompensationState): RT
272 // reads {published, capacity, delay, generation} as one consistent tuple,
273 // and generations are retired only after RT acknowledges them.
274 //
275 // The slot itself is pre-created when the TrackRTState is constructed and
276 // NEVER reassigned afterward, so `compensation.get()` is immutable for the
277 // RT thread: the whole 128 KiB ring stays lazy (behind `published`, which
278 // control fills on the first nonzero-delay publication). This removes the
279 // first-use assignment entirely, so control can never write this unique_ptr
280 // while an RT block is reading it (CodeRabbit, PR #730).
281 std::unique_ptr<TrackCompensationState> compensation{std::make_unique<TrackCompensationState>()};
282
283 // PDC v2 (P4b.2/P4b.3): per-outgoing-edge compensation state. Populated
284 // off-RT by AudioEngine::calculateLatencyCompensation() from
285 // SolvedLatencyTopology::edges; consumed RT-side in processBlock.
286 //
287 // Stored by unique_ptr because EdgeDelayState contains std::atomic members
288 // (non-movable). Lazily allocated by the apply pass when an edge actually
289 // needs compensation. A nullptr means "no compensation for this edge".
290 std::unique_ptr<EdgeDelayState> mainOutEdgeDelay;
291 std::vector<std::unique_ptr<EdgeDelayState>> sendEdgeDelays;
292};
