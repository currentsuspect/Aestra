42std::string getCurrentTimeString() {
43 auto now = std::time(nullptr);
44 auto tm = *std::localtime(&now);
45 std::stringstream ss;
46 ss << std::put_time(&tm, "Synced · %H:%M");
47 return ss.str();

110inline void drawMembershipIcon(AestraUI::NUIRenderer& renderer, const char* name, float cx, float cy, float size, const AestraUI::NUIColor& color) {
111 auto icon = NUIICON(name);
112 icon->setIconSize(size, size);
113 icon->setColor(color);
114 icon->setBounds(AestraUI::NUIRect(cx - size * 0.5f, cy - size * 0.5f, size, size));
115 icon->onRender(renderer);
116}

118// Draw a leading icon + label as one row, both vertically centred on centreY.
119// calculateTextY() centres the full line box against a zero-height rect at
120// centreY, exactly how the SYNC-card rows position their icon (at the row centre)
121// and label — those sit level to sub-pixel. Prefer this over hand-tuned Y offsets
122// like `cy + 2`, which measured ~5px low against the geometrically centred icon.
123inline void drawIconLabel(AestraUI::NUIRenderer& renderer, const char* iconName,
124 const std::string& label, float iconCx, float textX,
125 float centreY, float iconSize, float fontSize,
126 const AestraUI::NUIColor& iconColor,
127 const AestraUI::NUIColor& textColor) {
128 drawMembershipIcon(renderer, iconName, iconCx, centreY, iconSize, iconColor);
129 const AestraUI::NUIRect rowRect(textX, centreY, 0.0f, 0.0f);
130 renderer.drawText(label, {textX, renderer.calculateTextY(rowRect, fontSize)}, fontSize, textColor);
131}

134std::string serviceStatusMessage(Aestra::License::AccountServiceStatus status) {
135 switch (status) {
136 case Aestra::License::AccountServiceStatus::Success:
137 return "Synced now.";
138 case Aestra::License::AccountServiceStatus::Unauthorized:
139 return "Session invalid or expired. Sign in again.";
140 case Aestra::License::AccountServiceStatus::SyncUnavailable:
141 return "Sync unavailable. Last valid local membership was preserved.";
142 case Aestra::License::AccountServiceStatus::RejectedSignature:
143 return "Server response was rejected because the signed lease was invalid.";
144 case Aestra::License::AccountServiceStatus::InvalidResponse:
145 return "Server response was malformed. Last valid local membership was preserved.";
146 case Aestra::License::AccountServiceStatus::CacheWriteFailed:
147 return "Unable to update local account cache.";
148 case Aestra::License::AccountServiceStatus::NotConfigured:
149 default:
150 return "Account API is not configured.";
151 }
152}

257void MembershipSettingsPage::refreshDisplay() {
258#if defined(AESTRA_HAS_LICENSE_GATE) && AESTRA_HAS_LICENSE_GATE
259 Aestra::License::EntitlementStore entitlements;
260 Aestra::License::AccountSession accountSession(m_accountCache, entitlements);
261 Aestra::License::MembershipViewModel viewModel(accountSession, entitlements);
262 const Aestra::License::MembershipViewState state = viewModel.current();
263
264 m_accountLabel->setText(state.accountLabel);
265 m_tierLabel->setText(compactTierLabel(state.tierLabel));
266 m_statusLabel->setText(state.statusLabel);
267 m_verificationLabel->setText(std::string("Verification: ") +
268 (state.verified ? "Verified signed lease" : "Core or unverified local state"));
269 m_syncLabel->setText(std::string("Sync: ") + (state.offline ? "Local/offline cache" : "Online"));
270 m_lastRefreshLabel->setText("Last refresh: " + m_lastRefreshValue);
271 m_detailLabel->setText(state.detailMessage);
272 m_featuresTitleLabel->setText("Features");
273
274 std::vector<std::string> rows;
275 rows.reserve(m_featureLabels.size());
276 rows.push_back("Available");
277 for (const Aestra::License::MembershipFeatureRow& feature : state.features) {
278 if (feature.enabled) {
279 rows.push_back("+ " + feature.label);
280 }
281 }
282
283 bool hasLocked = false;
284 for (const Aestra::License::MembershipFeatureRow& feature : state.features) {
285 if (!feature.enabled) {
286 if (!hasLocked) {
287 if (rows.size() < m_featureLabels.size()) {
288 rows.push_back("Locked");
289 }
290 hasLocked = true;
291 }
292 if (rows.size() < m_featureLabels.size()) {
293 rows.push_back("- " + feature.label);
294 }
295 }
296 }
297
298 if (hasLocked && rows.size() < m_featureLabels.size()) {
299 rows.push_back("Requires verified membership.");
300 }
301
302 for (std::size_t i = 0; i < m_featureLabels.size(); ++i) {
303 if (i >= rows.size()) {
304 m_featureLabels[i]->setText("");
305 m_featureLabels[i]->setVisible(false);
306 continue;
307 }
308 m_featureLabels[i]->setText(rows[i]);
309 const bool heading = rows[i] == "Available" || rows[i] == "Locked";
310 const bool note = rows[i] == "Requires verified membership.";
311 m_featureLabels[i]->setFontSize(AestraUI::NUIThemeManager::getInstance().getFontSize(heading ? "m" : "s"));
312 m_featureLabels[i]->setTextColor(
313 AestraUI::NUIThemeManager::getInstance()
314 .getColor("textPrimary")
315 .withAlpha(heading ? 0.92f : (note ? 0.66f : 0.76f)));
316 m_featureLabels[i]->setVisible(true);
317 }
318
319 m_signedIn = state.signedIn;
320 m_canSignOut = state.canSignOut;
321 m_refreshButton->setEnabled(state.canRefresh);
322 m_signOutButton->setEnabled(state.canSignOut);
323 const bool showSignIn = !state.signedIn;
324 m_signInTitleLabel->setVisible(showSignIn);
325 m_emailInput->setVisible(showSignIn);
326 m_codeInput->setVisible(showSignIn);
327 m_startLoginButton->setVisible(showSignIn);
328 m_verifyLoginButton->setVisible(showSignIn);
329 m_startLoginButton->setEnabled(showSignIn);
330 m_verifyLoginButton->setEnabled(showSignIn && !m_pendingChallengeId.empty());
331#else
332 m_accountLabel->setText("Signed out");
333 m_tierLabel->setText("Core");
334 m_statusLabel->setText("License services unavailable");
335 m_verificationLabel->setText("Verification: Core or unverified local state");
336 m_syncLabel->setText("Sync: Local/offline cache");
337 m_lastRefreshLabel->setText("Last refresh: Not refreshed this session.");
338 m_detailLabel->setText("Core access remains available.");
339 m_featuresTitleLabel->setText("Features");
340 for (std::size_t i = 0; i < m_featureLabels.size(); ++i) {
341 m_featureLabels[i]->setText(i == 0 ? "+ Core DAW" : "");
342 m_featureLabels[i]->setFontSize(AestraUI::NUIThemeManager::getInstance().getFontSize("s"));
343 m_featureLabels[i]->setTextColor(
344 AestraUI::NUIThemeManager::getInstance().getColor("textPrimary").withAlpha(0.76f));
345 m_featureLabels[i]->setVisible(i == 0);
346 }
347 m_signedIn = false;
348 m_canSignOut = false;
349 m_refreshButton->setEnabled(false);
350 m_signOutButton->setEnabled(false);
351 m_signInTitleLabel->setVisible(false);
352 m_emailInput->setVisible(false);
353 m_codeInput->setVisible(false);
354 m_startLoginButton->setVisible(false);
355 m_verifyLoginButton->setVisible(false);
356#endif
357 layoutComponents();
358 setDirty(true);
359}

361void MembershipSettingsPage::refreshAccount() {
362 if (m_isRefreshing) return;
363 m_isRefreshing = true;
364 m_lastRefreshFailed = false;
365 m_actionErrorMessage.clear();
366 repaint();
367
368#if defined(AESTRA_HAS_LICENSE_GATE) && AESTRA_HAS_LICENSE_GATE
369 if (!m_accountService) {
370 m_lastRefreshValue = "Failed to sync";
371 m_lastRefreshFailed = true;
372 m_actionErrorMessage = "Could not reach license server. Try again later.";
373 m_actionErrorTimer = 6.0f;
374 } else {
375 m_lastRefreshValue = "Syncing…";
376 repaint();
377 const Aestra::License::AccountServiceResult result = m_accountService->refreshEntitlements();
378 if (result.status == Aestra::License::AccountServiceStatus::Success) {
379 m_lastRefreshValue = getCurrentTimeString();
380 m_lastRefreshFailed = false;
381 } else if (result.status == Aestra::License::AccountServiceStatus::RejectedSignature ||
382 result.status == Aestra::License::AccountServiceStatus::Unauthorized) {
383 // Lease invalid/revoked: treat as sign-out
384 m_lastRefreshValue = "Failed to sync";
385 m_lastRefreshFailed = true;
386 m_actionErrorMessage = "Your license could not be verified. Please sign in again.";
387 m_actionErrorTimer = 6.0f;
388 signOut();
389 if (m_onSignOutConfirmed) m_onSignOutConfirmed();
390 m_isRefreshing = false;
391 refreshDisplay();
392 return;
393 } else {
394 m_lastRefreshValue = "Failed to sync";
395 m_lastRefreshFailed = true;
396 m_actionErrorMessage = "Could not reach license server. Try again later.";
397 m_actionErrorTimer = 6.0f;
398 }
399 }
400#else
401 m_lastRefreshValue = "Failed to sync";
402 m_lastRefreshFailed = true;
403 m_actionErrorMessage = "Could not reach license server. Try again later.";
404 m_actionErrorTimer = 6.0f;
405#endif
406 m_isRefreshing = false;
407 refreshDisplay();
408}

410void MembershipSettingsPage::startLogin() {
411#if defined(AESTRA_HAS_LICENSE_GATE) && AESTRA_HAS_LICENSE_GATE
412 const std::string email = m_emailInput ? m_emailInput->getText() : "";
413 if (!m_accountService) {
414 m_signInMessage = "Account service is unavailable.";
415 } else if (email.empty()) {
416 m_signInMessage = "Email is required.";
417 } else {
418 const Aestra::License::AccountLoginStartServiceResult result = m_accountService->loginStart(email);
419 if (result.status == Aestra::License::AccountServiceStatus::Success && !result.challengeId.empty()) {
420 m_pendingLoginEmail = email;
421 m_pendingChallengeId = result.challengeId;
422 m_signInMessage = "Login code sent. Enter it below.";
423 } else {
424 m_pendingChallengeId.clear();
425 m_signInMessage = serviceStatusMessage(result.status);
426 }

608bool MembershipSettingsPage::onMouseEvent(const AestraUI::NUIMouseEvent& event) {
609 float mx = event.position.x;
610 float my = event.position.y;
611
612 // Sign-out confirmation overlay takes priority
613 if (m_showingSignOutConfirm) {
614 // Update hover states based on current position
615 bool oldCancel = m_confirmCancelHovered;
616 bool oldSignOut = m_confirmSignOutHovered;
617 m_confirmCancelHovered = m_confirmCancelBtnBounds.contains(mx, my);
618 m_confirmSignOutHovered = m_confirmSignOutBtnBounds.contains(mx, my);
619 if (oldCancel != m_confirmCancelHovered || oldSignOut != m_confirmSignOutHovered) {
620 repaint();
621 }
622
623 if (event.button == AestraUI::NUIMouseButton::Left && event.pressed) {
624 m_confirmPressedButton = m_confirmCancelHovered ? 0
625 : m_confirmSignOutHovered ? 1
626 : !m_confirmDialogBounds.contains(mx, my) ? 2
627 : -1;
628 repaint();
629 return true;
630 }
631 if (event.button == AestraUI::NUIMouseButton::Left && event.released) {
632 const int armedButton = m_confirmPressedButton;
633 m_confirmPressedButton = -1;
634 if ((armedButton == 0 && m_confirmCancelHovered) ||
635 (armedButton == 2 && !m_confirmDialogBounds.contains(mx, my))) {
636 m_showingSignOutConfirm = false;
637 repaint();
638 return true;
639 }
640 if (armedButton == 1 && m_confirmSignOutHovered) {
641 m_showingSignOutConfirm = false;
642 signOut();
643 if (m_onSignOutConfirmed) m_onSignOutConfirmed();
644 return true;
645 }
646 repaint();
647 return true;
648 }
649 return true; // Consume all mouse events while modal is open
650 }
651
652 if (AestraUI::NUIComponent::onMouseEvent(event))
653 return true;
654 if (m_form.onMouseWheel(event)) {
655 repaint();
656 return true;
657 }
658 return false;
659}
