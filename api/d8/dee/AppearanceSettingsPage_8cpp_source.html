18void AppearanceSettingsPage::createUI() {
19 m_themeLabel = std::make_shared<AestraUI::NUILabel>();
20 m_themeLabel->setText("UI Theme:");
21 m_themeLabel->setFontSize(AestraUI::NUIThemeManager::getInstance().getFontSize("m"));
22 addChild(m_themeLabel);
23
24 m_themeDropdown = std::make_shared<AestraUI::NUIDropdown>();
25 m_themeDropdown->addItem("Aestra Dark (Default)", 0);
26 m_themeDropdown->addItem("Aestra Light", 1);
27 m_themeDropdown->addItem("High Contrast Dark", 2);
28 m_themeDropdown->setOnSelectionChanged([this](int index) {
29 if (!m_syncingSelection && index >= 0 && index < static_cast<int>(kThemeNames.size()))
30 selectTheme(kThemeNames[static_cast<size_t>(index)]);
31 });
32 addChild(m_themeDropdown);
33
34 onShow();
35}

37void AppearanceSettingsPage::onShow() {
38 auto& manager = AestraUI::NUIThemeManager::getInstance();
39 m_originalTheme = manager.getActiveTheme();

44 m_syncingSelection = false;

47void AppearanceSettingsPage::selectTheme(const std::string& themeName) {
48 auto& manager = AestraUI::NUIThemeManager::getInstance();
49 if (!manager.setActiveTheme(themeName))
50 return;
51 m_pendingTheme = themeName;
52 m_dirty = m_pendingTheme != m_originalTheme;
53}

55void AppearanceSettingsPage::applyChanges() {
56 auto& preferences = Preferences::instance();
57 preferences.theme = m_pendingTheme;

59 m_originalTheme = m_pendingTheme;
60 m_dirty = false;

63void AppearanceSettingsPage::cancelChanges() {

72int AppearanceSettingsPage::indexForTheme(const std::string& themeName) const {
73 for (size_t i = 0; i < kThemeNames.size(); ++i) {
74 if (themeName == kThemeNames[i])
75 return static_cast<int>(i);
76 }
77 return 0;
