4// V82-H3: what a plugin's editor needs from the thread that owns its window. On Linux a plugin editor is not self
5// driving: it asks the host to call it when a file descriptor becomes readable (X11 connection, a socket) and every N
6// milliseconds (animation, meters). VST3 does that through Linux::IRunLoop, CLAP through its posix-fd-support and
7// timer-support extensions. Both end up here: this class only remembers who asked for what; the thread that owns the
8// window (the sandbox helper's GUI thread) asks it which descriptors to poll and when the next timer is due, and calls
9// back into it. No SDK, no X11, so it is testable on its own.
10//
11// Registration may come from any thread; handlers are called OUTSIDE the lock, so a handler can add or remove
12// registrations (a timer that cancels itself is routine).
13

28 int addFd(int fd, std::function<void()> onReadable) {
29 std::lock_guard<std::mutex> lock(m_mutex);
30 const int token = ++m_nextToken;
31 m_fds.push_back({token, fd, std::move(onReadable)});
32 return token;
33 }
34 void removeFd(int token) {
35 std::lock_guard<std::mutex> lock(m_mutex);
36 m_fds.erase(std::remove_if(m_fds.begin(), m_fds.end(), [&](const FdEntry& e) { return e.token == token; }),
37 m_fds.end());
38 }
39
41 int addTimer(std::chrono::milliseconds interval, std::function<void()> onTimer, Clock::time_point now = Clock::now()) {
42 std::lock_guard<std::mutex> lock(m_mutex);
43 const int token = ++m_nextToken;
44 const auto step = std::max(interval, std::chrono::milliseconds(1));
45 m_timers.push_back({token, step, now + step, std::move(onTimer)});
46 return token;
47 }
48 void removeTimer(int token) {
49 std::lock_guard<std::mutex> lock(m_mutex);
50 m_timers.erase(std::remove_if(m_timers.begin(), m_timers.end(), [&](const TimerEntry& e) { return e.token == token; }),
51 m_timers.end());
52 }
53
55 std::vector<int> fds() const {
56 std::lock_guard<std::mutex> lock(m_mutex);
57 std::vector<int> out;
58 for (const auto& entry : m_fds)
59 if (std::find(out.begin(), out.end(), entry.fd) == out.end()) out.push_back(entry.fd);
60 return out;
61 }
62
64 int dispatchFd(int fd) {
65 std::vector<std::function<void()>> due;
66 {
67 std::lock_guard<std::mutex> lock(m_mutex);
68 for (const auto& entry : m_fds)
69 if (entry.fd == fd) due.push_back(entry.handler);
70 }
71 for (auto& handler : due) handler();
72 return static_cast<int>(due.size());
73 }
74
76 int msUntilNextTimer(Clock::time_point now = Clock::now()) const {
77 std::lock_guard<std::mutex> lock(m_mutex);
78 if (m_timers.empty()) return -1;
79 auto soonest = m_timers.front().due;
80 for (const auto& timer : m_timers) soonest = std::min(soonest, timer.due);
81 if (soonest <= now) return 0;
82 const auto ms = std::chrono::duration_cast<std::chrono::milliseconds>(soonest - now).count();
83 return static_cast<int>(ms) + 1; // round up: waking a hair early would spin instead of firing
84 }
85
88 int fireDueTimers(Clock::time_point now = Clock::now()) {
89 std::vector<std::pair<int, std::function<void()>>> due;
90 {
91 std::lock_guard<std::mutex> lock(m_mutex);
92 for (auto& timer : m_timers) {
93 if (timer.due > now) continue;
94 due.emplace_back(timer.token, timer.handler);
95 timer.due = now + timer.interval;
96 }
97 }
98 for (auto& entry : due) {
99 { // skip one that an earlier handler in this pass has just removed
100 std::lock_guard<std::mutex> lock(m_mutex);
101 const bool stillThere = std::any_of(m_timers.begin(), m_timers.end(),
102 [&](const TimerEntry& t) { return t.token == entry.first; });
103 if (!stillThere) continue;
104 }
105 entry.second();
106 }
107 return static_cast<int>(due.size());
108 }
109
110 bool empty() const {
111 std::lock_guard<std::mutex> lock(m_mutex);
112 return m_fds.empty() && m_timers.empty();
113 }
114
115private:
116 struct FdEntry {
117 int token;
118 int fd;
119 std::function<void()> handler;
120 };
121 struct TimerEntry {
122 int token;
123 std::chrono::milliseconds interval;
124 Clock::time_point due;
125 std::function<void()> handler;
126 };
127 mutable std::mutex m_mutex;
128 std::vector<FdEntry> m_fds;
129 std::vector<TimerEntry> m_timers;
130 int m_nextToken = 0;
131};
