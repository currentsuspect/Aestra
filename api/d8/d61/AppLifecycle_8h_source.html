128 bool transitionTo(AppState newState) {
129 AppState expected = m_state.load(std::memory_order_acquire);
130
131 // Validate transition
132 if (!isValidTransition(expected, newState)) {
133 return false;
134 }
135
136 // Atomic transition
137 if (m_state.compare_exchange_strong(expected, newState,
138 std::memory_order_acq_rel)) {
139 // Notify listeners (on current thread)
140 notifyStateChange(expected, newState);
141 return true;
142 }
143
144 return false; // Another thread changed state
145 }

152 size_t addStateChangeCallback(StateChangeCallback callback) {
153 std::lock_guard<std::mutex> lock(m_callbackMutex);
154 size_t id = ++m_nextCallbackId;
155 m_callbacks.push_back({id, std::move(callback)});
156 return id;
157 }

162 void removeStateChangeCallback(size_t id) {
163 std::lock_guard<std::mutex> lock(m_callbackMutex);
164 m_callbacks.erase(
165 std::remove_if(m_callbacks.begin(), m_callbacks.end(),
166 [id](const auto& cb) { return cb.first == id; }),
167 m_callbacks.end()
168 );
169 }

174 void reset() {
175 m_state.store(AppState::Created, std::memory_order_release);
176 std::lock_guard<std::mutex> lock(m_callbackMutex);
177 m_callbacks.clear();
178 m_nextCallbackId = 0;
179 }

184 bool isValidTransition(AppState from, AppState to) const {
185 switch (from) {
186 case AppState::Created:
187 return to == AppState::Initializing;
188 case AppState::Initializing:
189 return to == AppState::Running || to == AppState::InitFailed;
190 case AppState::InitFailed:
191 return to == AppState::ShuttingDown;
192 case AppState::Running:
193 return to == AppState::ShuttingDown;
194 case AppState::ShuttingDown:
195 return to == AppState::Terminated;
196 case AppState::Terminated:
197 return false; // Terminal state
198 default:
199 return false;
200 }
201 }

203 void notifyStateChange(AppState oldState, AppState newState) {
204 std::vector<StateChangeCallback> callbacksCopy;
205 {
206 std::lock_guard<std::mutex> lock(m_callbackMutex);
207 for (const auto& cb : m_callbacks) {
208 callbacksCopy.push_back(cb.second);
209 }
210 }
211
212 for (const auto& callback : callbacksCopy) {
213 callback(oldState, newState);
214 }
215 }

217 std::atomic<AppState> m_state;
218 std::mutex m_callbackMutex;

233class LifecycleGuard {
