117bool MixerChannel::resyncEngineState() {
118 if (reportRealtimeMisuse("MixerChannel::resyncEngineState")) return false;
119 m_effectChain.refreshLatency(); // a plugin's own latency can change by itself: re-solve delay compensation (V82-H2)
120 if (!isStateDirty()) {
121 return true;
122 }
123 if (!m_commandSink || m_channelId == 0) {
124 return false;
125 }
126 const float volume = m_volume.load(std::memory_order_relaxed);
127 const float pan = m_pan.load(std::memory_order_relaxed);
128 const bool mute = m_muted.load(std::memory_order_relaxed);
129 const bool solo = m_soloed.load(std::memory_order_relaxed);
130 AudioQueueCommand cmd{};
131 cmd.channelId = m_channelId;
132 cmd.type = AudioQueueCommandType::SetTrackVolume;
133 cmd.value1 = volume;
134 if (!m_commandSink(cmd)) {
135 return false;
136 }
137 cmd.type = AudioQueueCommandType::SetTrackPan;
138 cmd.value1 = pan;
139 if (!m_commandSink(cmd)) {
140 return false;
141 }
142 cmd.type = AudioQueueCommandType::SetTrackMute;
143 cmd.value1 = mute ? 1.0f : 0.0f;
144 if (!m_commandSink(cmd)) {
145 return false;
146 }
147 cmd.type = AudioQueueCommandType::SetTrackSolo;
148 cmd.value1 = solo ? 1.0f : 0.0f;
149 if (!m_commandSink(cmd)) {
150 return false;
151 }
152 m_stateDirty.store(false, std::memory_order_relaxed);
153 return true;
154}

164std::vector<AudioRoute> MixerChannel::getSends() const {
165 // B-005: getSends() must never be called from the audio callback (RT) thread.
166 // Calling from RT can cause lock contention or deadlock. Sends are only
167 // accessed during graph build (main thread) or UI serialization.
168 std::lock_guard<std::mutex> lock(m_sendMutex);
169 return m_sends;
170}

173AudioRoute sanitizeRoute(const AudioRoute& route) {
174 AudioRoute sanitized = route;
175 sanitized.gain = std::isfinite(route.gain) ? std::clamp(route.gain, 0.0f, 4.0f) : 0.0f;
176 sanitized.pan = std::isfinite(route.pan) ? std::clamp(route.pan, -1.0f, 1.0f) : 0.0f;
177 return sanitized;
178}

181void MixerChannel::addSend(const AudioRoute& route) {

191void MixerChannel::insertSend(int index, const AudioRoute& route) {
192 if (reportRealtimeMisuse("MixerChannel::insertSend")) return;
193 std::lock_guard<std::mutex> lock(m_sendMutex);
194 AudioRoute sanitized = sanitizeRoute(route);
195 if (sanitized.sendId == 0) {
196 sanitized.sendId = m_nextSendId++;
197 }
198 const size_t clampedIndex = std::min(static_cast<size_t>(std::max(index, 0)), m_sends.size());
199 m_sends.insert(m_sends.begin() + static_cast<ptrdiff_t>(clampedIndex), sanitized);
200}

202void MixerChannel::setSend(uint64_t sendId, const AudioRoute& route) {
203 if (reportRealtimeMisuse("MixerChannel::setSend")) return;
204
205 std::lock_guard<std::mutex> lock(m_sendMutex);
206 const int index = findSendIndexLocked(sendId);
207 if (index < 0) {
208 return;
209 }
210 AudioRoute sanitized = sanitizeRoute(route);
211 // Identity is owned by the channel: the replacement keeps the sendId.
212 sanitized.sendId = m_sends[static_cast<size_t>(index)].sendId;
213 m_sends[static_cast<size_t>(index)] = sanitized;
214}

216void MixerChannel::replaceSends(const std::vector<AudioRoute>& routes) {
217 if (reportRealtimeMisuse("MixerChannel::replaceSends")) return;
218
219 std::lock_guard<std::mutex> lock(m_sendMutex);
220 m_sends.clear();
221 m_sends.reserve(routes.size());
222 for (const auto& route : routes) {
223 AudioRoute sanitized = sanitizeRoute(route);
224 if (sanitized.sendId == 0) {
225 sanitized.sendId = m_nextSendId++;
226 }
227 m_sends.push_back(sanitized);
228 }
229}

231void MixerChannel::removeSend(uint64_t sendId) {
232 if (reportRealtimeMisuse("MixerChannel::removeSend")) return;
233
234 std::lock_guard<std::mutex> lock(m_sendMutex);
235 const int index = findSendIndexLocked(sendId);
236 if (index >= 0) {
237 m_sends.erase(m_sends.begin() + index);
238 }
239}

241int MixerChannel::findSendIndex(uint64_t sendId) const {
242 std::lock_guard<std::mutex> lock(m_sendMutex);
243 return findSendIndexLocked(sendId);

246int MixerChannel::findSendIndexLocked(uint64_t sendId) const {
247 for (size_t i = 0; i < m_sends.size(); ++i) {
248 if (m_sends[i].sendId == sendId) {
249 return static_cast<int>(i);
250 }
251 }
252 return -1;
253}
