106void MixerChannel::processAudio(float* outputBuffer, uint32_t numFrames, double streamTime, double outputSampleRate) {
107 if (!outputBuffer || numFrames == 0)
108 return;
109 if (m_muted.load())
110 return;
111
112 // In v3.0, MixerChannel processes its internal bus/effects chain.
113 // The TrackManager orchestration handles mixing clip data into appropriate channel buffers.
114 if (m_mixerBus) {
115 m_mixerBus->process(outputBuffer, numFrames);
116 }
117
118 // Process through insert effect chain (if any plugins loaded)
119 // Pass 3: Use snapshot for RT-safety when available
120 auto snapshot = m_effectChainSnapshot.load(std::memory_order_acquire);
121 if (snapshot && snapshot->getActiveSlotCount() > 0) {
122 // Audio buffer is interleaved stereo (LRLRLRLR...)
123 // Plugins expect planar format (LL...LL, RR...RR)
124 // So we need to de-interleave -> process -> re-interleave
125
126 if (m_leftChannelBuf.size() < numFrames || m_rightChannelBuf.size() < numFrames ||
127 m_dryChannelBuf.size() < static_cast<size_t>(numFrames) * 2)
128 return;
129
130 for (uint32_t i = 0; i < numFrames; ++i) {
131 m_leftChannelBuf[i] = outputBuffer[i * 2];
132 m_rightChannelBuf[i] = outputBuffer[i * 2 + 1];
133 }
134
135 float* channels[2] = {m_leftChannelBuf.data(), m_rightChannelBuf.data()};
136
137 snapshot->process(channels, 2, numFrames, nullptr, 0, m_dryChannelBuf.data());
138
139 for (uint32_t i = 0; i < numFrames; ++i) {
140 outputBuffer[i * 2] = m_leftChannelBuf[i];
141 outputBuffer[i * 2 + 1] = m_rightChannelBuf[i];
142 }
143 } else if (m_effectChain.getActiveSlotCount() > 0) {
144 // Fallback: direct processing when no snapshot set (should not happen in normal operation)
145 if (m_leftChannelBuf.size() < numFrames || m_rightChannelBuf.size() < numFrames)
146 return;
147
148 for (uint32_t i = 0; i < numFrames; ++i) {
149 m_leftChannelBuf[i] = outputBuffer[i * 2];
150 m_rightChannelBuf[i] = outputBuffer[i * 2 + 1];
151 }
152
153 float* channels[2] = {m_leftChannelBuf.data(), m_rightChannelBuf.data()};
154
155 m_effectChain.process(channels, 2, numFrames);
156
157 for (uint32_t i = 0; i < numFrames; ++i) {
158 outputBuffer[i * 2] = m_leftChannelBuf[i];
159 outputBuffer[i * 2 + 1] = m_rightChannelBuf[i];
160 }
161 }

164std::vector<AudioRoute> MixerChannel::getSends() const {
165 // B-005: getSends() must never be called from the audio callback (RT) thread.
166 // Calling from RT can cause lock contention or deadlock. Sends are only
167 // accessed during graph build (main thread) or UI serialization.
168 std::lock_guard<std::mutex> lock(m_sendMutex);
169 return m_sends;

173AudioRoute sanitizeRoute(const AudioRoute& route) {
174 AudioRoute sanitized = route;
175 sanitized.gain = std::isfinite(route.gain) ? std::clamp(route.gain, 0.0f, 4.0f) : 0.0f;
176 sanitized.pan = std::isfinite(route.pan) ? std::clamp(route.pan, -1.0f, 1.0f) : 0.0f;
177 return sanitized;
178}

181void MixerChannel::addSend(const AudioRoute& route) {
182 std::lock_guard<std::mutex> lock(m_sendMutex);

190void MixerChannel::insertSend(int index, const AudioRoute& route) {
191 std::lock_guard<std::mutex> lock(m_sendMutex);
192 AudioRoute sanitized = sanitizeRoute(route);
193 if (sanitized.sendId == 0) {
194 sanitized.sendId = m_nextSendId++;
195 }
196 const size_t clampedIndex = std::min(static_cast<size_t>(std::max(index, 0)), m_sends.size());
197 m_sends.insert(m_sends.begin() + static_cast<ptrdiff_t>(clampedIndex), sanitized);
198}

200void MixerChannel::setSend(uint64_t sendId, const AudioRoute& route) {
201 std::lock_guard<std::mutex> lock(m_sendMutex);
202 const int index = findSendIndexLocked(sendId);
203 if (index < 0) {
204 return;
205 }
206 AudioRoute sanitized = sanitizeRoute(route);
207 // Identity is owned by the channel: the replacement keeps the sendId.
208 sanitized.sendId = m_sends[static_cast<size_t>(index)].sendId;
209 m_sends[static_cast<size_t>(index)] = sanitized;
210}

212void MixerChannel::replaceSends(const std::vector<AudioRoute>& routes) {
213 std::lock_guard<std::mutex> lock(m_sendMutex);
214 m_sends.clear();
215 m_sends.reserve(routes.size());
216 for (const auto& route : routes) {
217 AudioRoute sanitized = sanitizeRoute(route);
218 if (sanitized.sendId == 0) {
219 sanitized.sendId = m_nextSendId++;
220 }
221 m_sends.push_back(sanitized);
222 }
223}

225void MixerChannel::removeSend(uint64_t sendId) {
226 std::lock_guard<std::mutex> lock(m_sendMutex);
227 const int index = findSendIndexLocked(sendId);
228 if (index >= 0) {
229 m_sends.erase(m_sends.begin() + index);
230 }
231}

233int MixerChannel::findSendIndex(uint64_t sendId) const {
234 std::lock_guard<std::mutex> lock(m_sendMutex);
235 return findSendIndexLocked(sendId);
236}

238int MixerChannel::findSendIndexLocked(uint64_t sendId) const {
239 for (size_t i = 0; i < m_sends.size(); ++i) {
240 if (m_sends[i].sendId == sendId) {
241 return static_cast<int>(i);
242 }
243 }
244 return -1;
245}
