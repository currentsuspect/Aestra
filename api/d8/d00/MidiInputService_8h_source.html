75 static std::optional<TranslatedEvent> translateMessage(const uint8_t* bytes, size_t size) {
76 if (bytes == nullptr || size < 3) {
77 return std::nullopt;
78 }
79 const uint8_t status = bytes[0] & 0xF0;
80 if (status != 0x90 && status != 0x80) {
81 return std::nullopt; // only note messages ride the live path for now
82 }
83 const uint8_t note = bytes[1] & 0x7F;
84 const uint8_t velocity = bytes[2] & 0x7F;
85 uint8_t outStatus = bytes[0];
86 // Note-on with velocity 0 is a note-off in disguise (very common on
87 // controllers). Normalize so downstream never needs the special case.
88 if (status == 0x90 && velocity == 0) {
89 outStatus = static_cast<uint8_t>(0x80 | (bytes[0] & 0x0F));
90 }
91 return TranslatedEvent{outStatus, note, velocity};
92 }
