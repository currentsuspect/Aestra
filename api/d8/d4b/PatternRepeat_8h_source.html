3// How a timeline clip repeats its pattern (FD-50, V82-T3). A clip reads source beats [sourceStart, sourceEnd) of
4// an endless repetition of its pattern: source beat s plays pattern beat s mod length. A clip no longer than its
5// pattern plays exactly what it did before, so only clips running past their pattern change.
6//
7// One definition for the scheduler (PatternPlaybackEngine), the clip's mini piano roll and dashed repeat lines
8// (TrackUIComponent / TimelineClipPaint) and the right-edge drag, so what is drawn, dragged and heard agree.

21struct Range {
22 int64_t first = 0;
23 int64_t last = -1;
24 bool empty() const { return last < first; }

29inline bool repeats(double patternLengthBeats, double sourceEndBeat) {
30 return patternLengthBeats > 0.0 && std::isfinite(patternLengthBeats) && std::isfinite(sourceEndBeat);
31}
32
34inline Range inSource(double patternLengthBeats, double sourceStartBeat, double sourceEndBeat) {
35 if (!repeats(patternLengthBeats, sourceEndBeat) || !(sourceEndBeat > sourceStartBeat)) return {0, 0};
36 constexpr double kEpsilon = 1.0e-9;
37 const auto first = static_cast<int64_t>(std::floor(std::max(0.0, sourceStartBeat) / patternLengthBeats));
38 const auto last = static_cast<int64_t>(std::ceil(sourceEndBeat / patternLengthBeats - kEpsilon)) - 1;
39 return {first, std::max(first, last)};
40}
41
45inline Range inWindow(double patternLengthBeats, double sourceStartBeat, double sourceEndBeat,
46 double patternStartBeat, double fromBeat, double toBeat, double reachBeats) {
47 Range r = inSource(patternLengthBeats, sourceStartBeat, sourceEndBeat);
48 if (!repeats(patternLengthBeats, sourceEndBeat)) return r;
49 const double lo = std::floor((fromBeat - patternStartBeat - reachBeats) / patternLengthBeats);
50 const double hi = std::floor((toBeat - patternStartBeat) / patternLengthBeats);
51 if (lo > static_cast<double>(r.first)) r.first = static_cast<int64_t>(lo);
52 if (hi < static_cast<double>(r.last)) r.last = static_cast<int64_t>(hi);
53 return r;
54}

57inline uint64_t passSeed(uint64_t pass, int64_t repeat) {
58 constexpr uint64_t kGoldenRatio64 = 0x9E3779B97F4A7C15ULL; // spreads consecutive repeats across the hash
59 return pass + (static_cast<uint64_t>(repeat) * kGoldenRatio64);
60}

66inline double snapEndToRepeat(double patternLengthBeats, double patternStartBeat, double clipStartBeat,
67 double proposedEndBeat) {
68 if (!(patternLengthBeats > 0.0) || !std::isfinite(patternLengthBeats)) return proposedEndBeat;
69 constexpr double kEpsilon = 1.0e-6;
70 const double firstK = std::floor((clipStartBeat - patternStartBeat) / patternLengthBeats + kEpsilon) + 1.0;
71 const double firstBoundary = patternStartBeat + firstK * patternLengthBeats;
72 if (proposedEndBeat < firstBoundary - kEpsilon) return proposedEndBeat;
73 const double k = std::max(firstK, std::round((proposedEndBeat - patternStartBeat) / patternLengthBeats));
74 return patternStartBeat + k * patternLengthBeats;
75}
