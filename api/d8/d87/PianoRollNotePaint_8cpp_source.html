11void PianoRollNoteLayer::paintNote(NUIRenderer& renderer, const NUIRect& r, const MidiNote& n, bool hovered, bool sounding,
12 float rawWidth) {
13 PianoRollLook::NoteState state;
14 state.selected = n.selected;
15 state.hovered = hovered;
16 state.sounding = sounding;
17 const NUIColor fill = PianoRollLook::noteFill(noteColor_, state);
18
19 // Flat and opaque, like a clip: no shadow, no gradient, no alpha by velocity (the velocity lane says that).
20 renderer.fillRoundedRect(r, 2.0f, fill);
21 if (n.selected) renderer.strokeRoundedRect(r, 2.0f, 1.0f, NUIColor::white().withAlpha(0.95f));
22
23 // Selected notes keep their resize grips; the edge under the pointer goes full white.
24 if (n.selected || (hovered && (hoverOnRightEdge_ || hoverOnLeftEdge_))) {
25 const NUIColor idleGrip = NUIColor::white().withAlpha(0.9f);
26 const NUIColor activeGrip = NUIColor::white();
27 if (n.selected || hoverOnRightEdge_) {
28 const NUIColor grip = (hovered && hoverOnRightEdge_) ? activeGrip : idleGrip;
29 renderer.fillRect(NUIRect(r.right() - 3.0f, r.y + 3.0f, 2.0f, r.height - 6.0f), grip);
30 }
31 if (n.selected || hoverOnLeftEdge_) {
32 const NUIColor grip = (hovered && hoverOnLeftEdge_) ? activeGrip : idleGrip;
33 renderer.fillRect(NUIRect(r.x + 1.0f, r.y + 3.0f, 2.0f, r.height - 6.0f), grip);
34 }
35 }
36
37 // The pitch name inside, when there is room for it.
38 constexpr float kMinWidthForLabel = 22.0f;
39 if (rawWidth >= kMinWidthForLabel) {
40 const std::string label = MusicTheory::getPitchName(n.pitch);
41 constexpr float kFontSize = 10.0f;
42 const auto measured = renderer.measureText(label, kFontSize);
43 const float labelX = r.x + 6.0f;
44 if (labelX + measured.width < r.right() - 4.0f)
45 renderer.drawText(label, NUIPoint(labelX, r.y + (r.height - measured.height) * 0.5f), kFontSize,
46 PianoRollLook::noteLabelColor(fill));
47 }
48}
