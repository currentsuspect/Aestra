25std::string lower(std::string s) {
26 for (auto& c : s) c = static_cast<char>(std::tolower(static_cast<unsigned char>(c)));

30std::string fit(NUIRenderer& r, const std::string& text, float size, float maxW) {
31 if (r.measureText(text, size).width <= maxW) return text;
32 std::string s = text;
33 while (!s.empty() && r.measureText(s + "\xE2\x80\xA6", size).width > maxW) s.pop_back();
34 return s + "\xE2\x80\xA6";

37std::string formatDb(float db) {
38 if (db <= -100.0f) return "-inf dB";
39 char buf[24];
40 std::snprintf(buf, sizeof(buf), "%+.1f dB", static_cast<double>(db));
41 std::string s = buf;
42 if (s[0] == '-') s = "\xE2\x88\x92" + s.substr(1); // a real minus sign
43 if (db == 0.0f) s = "0.0 dB";
44 return s;
45}

51void dashedRect(NUIRenderer& r, const NUIRect& rc, const NUIColor& c) {
52 auto dash = [&](NUIPoint a, NUIPoint b) {
53 const float len = std::hypot(b.x - a.x, b.y - a.y);
54 for (float t = 0; t < len; t += 8.0f) {
55 const float t2 = std::min(len, t + 4.0f);
56 r.drawLine({a.x + (b.x - a.x) * t / len, a.y + (b.y - a.y) * t / len},
57 {a.x + (b.x - a.x) * t2 / len, a.y + (b.y - a.y) * t2 / len}, 1.0f, c);
58 }
59 };
60 dash({rc.x, rc.y}, {rc.right(), rc.y});
61 dash({rc.right(), rc.y}, {rc.right(), rc.bottom()});
62 dash({rc.right(), rc.bottom()}, {rc.x, rc.bottom()});
63 dash({rc.x, rc.bottom()}, {rc.x, rc.y});
64}

67void UIRoutingMap::drawWire(NUIRenderer& r, NUIPoint a, NUIPoint b, float thickness, const NUIColor& c, bool dashed,
68 bool vertical) {
69 // Horizontal tangents for outputs and sends; a sidechain drops onto the top of its node, so it ends vertical.
70 const float dx = std::max(40.0f, std::abs(b.x - a.x) * 0.5f);
71 const NUIPoint c1{a.x + dx, a.y};
72 const NUIPoint c2 = vertical ? NUIPoint{b.x, b.y - std::max(60.0f, std::abs(b.y - a.y) * 0.6f)}
73 : NUIPoint{b.x - dx, b.y};
74 constexpr int kSteps = 40;
75 NUIPoint pts[kSteps + 1];
76 for (int i = 0; i <= kSteps; ++i) {
77 const float t = static_cast<float>(i) / kSteps, u = 1.0f - t;
78 pts[i] = {u * u * u * a.x + 3 * u * u * t * c1.x + 3 * u * t * t * c2.x + t * t * t * b.x,
79 u * u * u * a.y + 3 * u * u * t * c1.y + 3 * u * t * t * c2.y + t * t * t * b.y};
80 }
81 if (std::abs(a.y - b.y) < 0.5f && !vertical) {
82 if (!dashed) {
83 r.drawLine(a, b, thickness, c);
84 return;
85 }
86 }
87 if (!dashed) {
88 r.drawPolyline(pts, kSteps + 1, thickness, c);
89 return;
90 }
91 for (int i = 0; i < kSteps; i += 2) r.drawLine(pts[i], pts[i + 1], thickness, c);
92}

261void UIRoutingMap::drawEdges(NUIRenderer& r, const Geometry& g) {
262 const bool tracing = m_selectedId != 0 && !m_patching;
263 for (const auto& e : m_edges) {
264 const Node* s = nodeById(e.source);
265 if (!s) continue;
266 const Node* t = e.target == RoutingGraph::kMaster ? nullptr : nodeById(e.target);
267 if (e.target != RoutingGraph::kMaster && !t) continue;
268 const NUIPoint a = outputPort(*s);
269 NUIPoint bpt = t ? (e.type == Edge::Sidechain ? sidechainPort(*t) : inputPort(*t))
270 : NUIPoint{g.rail.x, railPortY(a.y)};
271
272 bool inChain = false;
273 if (tracing) {
274 const bool up = m_trace.upstream.count(e.source) && t && m_trace.upstream.count(e.target);
275 const bool down = m_trace.downstream.count(e.source) > 0 && !m_trace.listening.count(e.source);
276 inChain = up || down;
277 }
278 NUIColor c = e.type == Edge::Output ? m_textSecondary.withAlpha(0.7f)
279 : e.type == Edge::Send ? m_textMuted.withAlpha(0.55f)
280 : m_sidechain.withAlpha(0.85f);
281 float th = e.type == Edge::Output ? 1.4f : 1.2f;
282 if (s->muted) c = c.withAlpha(c.a * 0.5f);
283 if (tracing) {
284 if (inChain) {
285 c = e.type == Edge::Sidechain ? m_sidechain : m_text;
286 th = 2.0f;
287 } else {
288 c = c.withAlpha(c.a * 0.3f);
289 }
290 }
291 if (m_patching) c = c.withAlpha(c.a * 0.5f);
292 drawWire(r, a, bpt, th, c, e.type == Edge::Sidechain, e.type == Edge::Sidechain);
293
294 if (inChain && e.type == Edge::Send && m_zoom >= 0.5f) {
295 // The level sits beside the wire on the side away from its neighbours: above a level or rising
296 // wire, below a falling one.
297 const std::string lbl = formatDb(e.levelDb);
298 const float w = r.measureText(lbl, 9.5f).width + 12.0f;
299 const float lx = a.x + (bpt.x - a.x) * 0.45f - w * 0.5f;
300 const float ly = bpt.y > a.y + 1.0f ? std::max(a.y, bpt.y) + 8.0f : std::min(a.y, bpt.y) - 26.0f;
301 const NUIRect box{lx, ly, w, 18.0f};
302 r.fillRoundedRect(box, 2.0f, m_node);
303 r.strokeRoundedRect(box, 2.0f, 1.0f, m_border);
304 r.drawTextCentered(lbl, box, 9.5f, m_textSecondary);
305 }
306 if (m_signalFlow && e.type != Edge::Sidechain && !s->muted && s->peakDb > -60.0f) {
307 for (int k = 0; k < 3; ++k) {
308 const float t01 = std::fmod(m_flowPhase + k / 3.0f, 1.0f);
309 const float dx = std::max(40.0f, std::abs(bpt.x - a.x) * 0.5f);
310 const NUIPoint c1{a.x + dx, a.y}, c2{bpt.x - dx, bpt.y};
311 const float u = 1.0f - t01;
312 const NUIPoint q{u * u * u * a.x + 3 * u * u * t01 * c1.x + 3 * u * t01 * t01 * c2.x + t01 * t01 * t01 * bpt.x,
313 u * u * u * a.y + 3 * u * u * t01 * c1.y + 3 * u * t01 * t01 * c2.y + t01 * t01 * t01 * bpt.y};
314 r.fillCircle(q, 2.5f, m_live.withAlpha(0.9f));
315 }
316 }
317 }

320void UIRoutingMap::drawNodes(NUIRenderer& r, const Geometry&) {
321 const bool tracing = m_selectedId != 0 && !m_patching;
322 const std::string q = lower(m_search);
323 bool patchTargetMaster = false;
324 const uint32_t patchTo = m_patching ? patchTarget(m_patchMouse, patchTargetMaster) : kNoOutput;
325 for (size_t i = 0; i < m_nodes.size(); ++i) {
326 const Node& n = m_nodes[i];
327 const NUIRect rc = nodeRect(n);
328 const NUIColor col = argb(n.color);
329 if (m_zoom < 0.5f) {
330 // Far: colour blocks only.
331 r.fillRect(rc, col.withAlpha(tracing && !m_trace.contains(n.id) ? 0.3f : 0.9f));
332 if (n.id == m_selectedId) r.strokeRect(rc, 1.5f, m_accent);
333 continue;
334 }
335 const bool refused = m_patching && refuses(n.id);
336 r.fillRoundedRect(rc, 2.0f, m_node);
337 r.strokeRoundedRect(rc, 2.0f, 1.0f, static_cast<int>(i) == m_hoverNode && !m_patching ? m_textMuted : m_border);
338 if (refused) {
339 // Hatched: this one would loop.
340 r.setClipRect(rc);
341 for (float x = rc.x - rc.height; x < rc.right(); x += 8.0f)
342 r.drawLine({x, rc.bottom()}, {x + rc.height, rc.y}, 1.0f, m_textMuted.withAlpha(0.18f));
343 r.clearClipRect();
344 r.setClipRect(geometry().canvas);
345 }
346
347 const float z = m_zoom;
348 const float cy = rc.y + rc.height * 0.5f;
349 r.fillRect({rc.x + 14.0f * z, cy - 5.0f * z, 10.0f * z, 10.0f * z}, col);
350 const float font = std::max(8.0f, 12.0f * z);
351 // The role tag needs room beside the name; below 75 % it would collide, so only LOOP survives.
352 const char* tag = refused ? "LOOP"
353 : z < 0.75f ? ""
354 : n.role == RoutingGraph::Role::Bus ? "BUS"
355 : n.role == RoutingGraph::Role::Return ? "RETURN"
356 : "";
357 const float tagFont = std::max(7.0f, 8.5f * z);
358 const float tagW = *tag ? r.measureText(tag, tagFont).width : 0.0f;
359 r.drawText(fit(r, n.label, font, rc.width - 44.0f * z - tagW), {rc.x + 30.0f * z, cy - font * 0.6f}, font,
360 n.muted ? m_textMuted : m_text);
361 if (*tag)
362 r.drawText(tag, {rc.right() - 12.0f * z - tagW, cy - tagFont * 0.6f}, tagFont,
363 refused ? m_refuse : m_textMuted);
364
365 // Ports: the output lights with level.
366 const NUIPoint in = inputPort(n), out = outputPort(n);
367 r.fillCircle(in, 3.5f * z, m_bg);
368 r.strokeCircle(in, 3.5f * z, 1.0f, m_textMuted);
369 const bool lit = i < m_lit.size() && m_lit[i];
370 r.fillCircle(out, 3.5f * z, lit ? m_live : m_bg);
371 r.strokeCircle(out, 3.5f * z, 1.0f, lit ? m_live : m_textMuted);
372
373 bool sidechained = false;
374 for (const auto& e : m_edges) sidechained |= e.type == Edge::Sidechain && e.target == n.id;
375 if (sidechained) {
376 const NUIPoint d = sidechainPort(n);
377 const float s = 4.5f * z;
378 r.fillRect({d.x - 1, d.y - 1, 2, 2}, m_sidechain);
379 r.drawLine({d.x, d.y - s}, {d.x + s, d.y}, 1.2f, m_sidechain);
380 r.drawLine({d.x + s, d.y}, {d.x, d.y + s}, 1.2f, m_sidechain);
381 r.drawLine({d.x, d.y + s}, {d.x - s, d.y}, 1.2f, m_sidechain);
382 r.drawLine({d.x - s, d.y}, {d.x, d.y - s}, 1.2f, m_sidechain);
383 }
384
385 if (tracing && !m_trace.contains(n.id)) r.fillRoundedRect(rc, 2.0f, m_bg.withAlpha(0.62f));
386 if (refused) r.fillRoundedRect(rc, 2.0f, m_bg.withAlpha(0.4f));
387 if (n.id == m_selectedId && !m_patching)
388 r.strokeRoundedRect({rc.x - 1, rc.y - 1, rc.width + 2, rc.height + 2}, 2.0f, 1.5f, m_accent);
389 if (m_patching && n.id == m_patchSource)
390 r.strokeRoundedRect({rc.x - 1, rc.y - 1, rc.width + 2, rc.height + 2}, 2.0f, 1.5f, m_accent);
391 if (m_patching && n.id == patchTo && !refused) r.strokeRoundedRect(rc, 2.0f, 1.5f, m_text);
392 if (!q.empty() && lower(n.label).find(q) != std::string::npos)
