117 inline float sampleLinear(const float* data, int64_t numFrames, uint32_t numChannels, double position,
118 uint32_t channel) const noexcept {
119 int64_t idx0 = static_cast<int64_t>(position);
120 int64_t idx1 = idx0 + 1;
121
122 if (idx1 >= numFrames)
123 idx1 = numFrames - 1;
124
125 float frac = static_cast<float>(position - idx0);
126
127 float s0 = data[idx0 * numChannels + channel];
128 float s1 = data[idx1 * numChannels + channel];
129
130 return s0 + frac * (s1 - s0);
131 }

133 inline void sampleLinearStereo(const float* data, int64_t numFrames, double position, float& outL,
134 float& outR) const noexcept {
135 int64_t idx0 = static_cast<int64_t>(position);
136 int64_t idx1 = idx0 + 1;
137
138 if (idx1 >= numFrames)
139 idx1 = numFrames - 1;
140
141 float frac = static_cast<float>(position - idx0);
142
143 float l0 = data[idx0 * 2];
144 float l1 = data[idx1 * 2];
145 float r0 = data[idx0 * 2 + 1];
146 float r1 = data[idx1 * 2 + 1];
147
148 outL = l0 + frac * (l1 - l0);
149 outR = r0 + frac * (r1 - r0);
150 }

156 inline float sampleCubic(const float* data, int64_t numFrames, uint32_t numChannels, double position,
157 uint32_t channel) const noexcept {
158 int64_t idx = static_cast<int64_t>(position);
159 double frac = position - static_cast<double>(idx);
160
161 // Get 4 sample points with bounds clamping
162 int64_t i0 = (idx > 0) ? idx - 1 : 0;
163 int64_t i1 = idx;
164 int64_t i2 = (idx + 1 < numFrames) ? idx + 1 : numFrames - 1;
165 int64_t i3 = (idx + 2 < numFrames) ? idx + 2 : numFrames - 1;
166
167 float s0 = data[i0 * numChannels + channel];
168 float s1 = data[i1 * numChannels + channel];
169 float s2 = data[i2 * numChannels + channel];
170 float s3 = data[i3 * numChannels + channel];
171
172 // Catmull-Rom coefficients
173 float frac2 = static_cast<float>(frac * frac);
174 float frac3 = frac2 * static_cast<float>(frac);
175
176 float c0 = -0.5f * frac3 + frac2 - 0.5f * static_cast<float>(frac);
177 float c1 = 1.5f * frac3 - 2.5f * frac2 + 1.0f;
178 float c2 = -1.5f * frac3 + 2.0f * frac2 + 0.5f * static_cast<float>(frac);
179 float c3 = 0.5f * frac3 - 0.5f * frac2;
180
181 return s0 * c0 + s1 * c1 + s2 * c2 + s3 * c3;
182 }

210 uint32_t channel) const noexcept {
211 // For mono or channel extraction, we need to build a stereo view
212 // This is less efficient than the stereo path, but maintains quality
213 float outL, outR;
214
215 if (numChannels >= 2) {
216 Interpolators::Sinc64Turbo::interpolate(data, numFrames, position, outL, outR);
217 return (channel == 0) ? outL : outR;
218 } else {
219 // Mono: duplicate to stereo for Sinc64
220 // Note: This is a fallback path, stereo is more common
221 return sampleCubic(data, numFrames, numChannels, position, channel);
222 }
