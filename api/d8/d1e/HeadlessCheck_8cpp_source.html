103bool readText(const std::string& path, std::string& out) {
104 std::ifstream in(path, std::ios::binary);
105 if (!in) {
106 return false;
107 }
108 std::ostringstream ss;
109 ss << in.rdbuf();
110 out = ss.str();
111 return true;

123bool readExpectation(const std::string& path, RenderExpectation& out, std::string& error) {
124 std::string text;
125 if (!readText(path, text)) {
126 error = "cannot read " + path;
127 return false;
128 }
129 bool consumedAll = false;
130 const JSON root = JSON::parseStrict(text, consumedAll);
131 if (!root.isObject() || !consumedAll || !root.has("frames") || !root.has("peakDbfs") || !root.has("rmsDbfs")) {
132 error = path + " is not a render golden (want frames, peakDbfs, rmsDbfs, and optionally hash)";
133 return false;
134 }
135 // No "hash": a cross-platform golden, levels only (see RenderExpectation).
136 out.hasHash = root.has("hash");
137 if (out.hasHash && !hashFromHex(root["hash"].asString(), out.hash64)) {
138 error = path + ": hash must be 16 lower-case hex digits";
139 return false;
140 }
141 out.frames = static_cast<uint64_t>(root["frames"].asNumber());
142 out.peakDbfs = root["peakDbfs"].asNumber();
143 out.rmsDbfs = root["rmsDbfs"].asNumber();
144 // Loudness pins (V8-S8) are optional: a golden from before them still checks.
145 out.hasLoudness = root.has("integratedLufs") && root.has("truePeakDbtp");
146 if (out.hasLoudness) {
147 out.integratedLufs = root["integratedLufs"].asNumber();
148 out.truePeakDbtp = root["truePeakDbtp"].asNumber();
149 }
150 return true;
151}

166// Prints the report, writes it if asked, and returns the exit code.
167int finish(JSON& report, const std::vector<std::string>& failures, const HeadlessCheckOptions& options, int code) {
168 report.set("ok", JSON(code == kPass));
169 report.set("failures", stringArray(failures));
170 const std::string text = report.toString(2);
171 std::cout << text << "\n";
172 if (!options.reportPath.empty() && !writeText(options.reportPath, text)) {
173 std::cerr << "AestraHeadless --check: cannot write report " << options.reportPath << "\n";
174 return kCouldNotCheck;
175 }
176 for (const auto& f : failures) {
177 std::cerr << "FAIL " << f << "\n";
178 }
179 return code;
180}

184int runHeadlessCheck(const HeadlessCheckOptions& options) {
185 namespace fs = std::filesystem;
186 JSON report = JSON::object();
187 std::vector<std::string> failures;
188 const std::string projectPath = absolutePath(options.projectPath);
189 report.set("project", JSON(projectPath));
190
191 // --- Load, the way the app does ------------------------------------------
192 auto trackManager = std::make_shared<TrackManager>();
193 trackManager->setOutputSampleRate(static_cast<double>(options.sampleRate));
194 trackManager->setInputSampleRate(static_cast<double>(options.sampleRate));
195 trackManager->setInputChannelCount(0);
196 ProjectSerializer::LoadResult load = ProjectSerializer::load(projectPath, trackManager);
197
198 JSON loadJson = JSON::object();
199 loadJson.set("ok", JSON(load.ok));
200 loadJson.set("tempo", JSON(load.tempo));
201 loadJson.set("missingAssets", stringArray(load.missingAssets));
202 loadJson.set("missingPlugins", pluginArray(load.missingPlugins));
203 loadJson.set("unreadablePluginState", pluginArray(load.unreadablePluginState));
204 if (!load.errorMessage.empty()) {
205 loadJson.set("error", JSON(load.errorMessage));
206 }
207 report.set("load", loadJson);
208 if (!load.ok) {
209 failures.push_back("project did not load: " + load.errorMessage);
210 return finish(report, failures, options, kCouldNotCheck);
211 }
212 // A missing asset or plugin renders as silence where it should sound, so
213 // the render would be checked against the wrong music. Fail unless asked.
214 if (!options.allowMissing) {
215 if (!load.missingAssets.empty()) {
216 failures.push_back(std::to_string(load.missingAssets.size()) + " missing asset(s)");
217 }
218 if (!load.missingPlugins.empty()) {
219 failures.push_back(std::to_string(load.missingPlugins.size()) + " missing plugin(s)");
220 }
221 if (!load.unreadablePluginState.empty()) {
222 failures.push_back(std::to_string(load.unreadablePluginState.size()) +
223 " plugin(s) rejected their saved state");
224 }
225 }
226
227 // --- Render, the way Export does -----------------------------------------
228 // Float_32: no dither, so the samples on disk are the samples rendered and
229 // the hash can be bit-exact across runs.
230 const bool keep = !options.outPath.empty();
231 const std::string wavPath =
232 keep ? absolutePath(options.outPath)
233 : (fs::temp_directory_path() / ("aestra_check_" + std::to_string(std::hash<std::string>{}(projectPath)) +
234 ".wav"))
235 .string();
236 struct TempCleanup {
237 std::string path;
238 ~TempCleanup() {
239 if (!path.empty()) {
240 std::error_code ec;
241 std::filesystem::remove(path, ec);
242 }
243 }
244 } cleanup{keep ? std::string() : wavPath};
245
246 AudioEngine engine;
247 OfflineRenderRequest request;
248 request.outputPath = wavPath;
249 request.sampleRate = options.sampleRate;
250 request.bitDepth = AudioExporter::BitDepth::Float_32;
251 request.tempoBpm = load.tempo;
252 const AudioExporter::Result render = renderProjectOffline(engine, *trackManager, request);
253 if (!render.ok()) {
254 failures.push_back("render failed: " + render.errorMessage);
255 return finish(report, failures, options, kCouldNotCheck);
256 }
257 if (keep) {
258 report.set("output", JSON(wavPath));
259 }
260
261 // --- Read back and summarise ---------------------------------------------
262 std::vector<float> samples;
263 uint32_t fileRate = 0;
264 uint32_t fileChannels = 0;
265 if (!decodeAudioFile(wavPath, samples, fileRate, fileChannels) || fileChannels == 0) {
266 failures.push_back("cannot read back the render: " + wavPath);
267 return finish(report, failures, options, kCouldNotCheck);
268 }
269 const RenderSummary summary = summarizeRender(samples, fileChannels, fileRate);
270 report.set("summary", summaryJson(summary, render.maxTruePeakdBTP));
271 const AudioAnalysis analysis = analyzeAudio(samples, fileChannels, fileRate);
272 report.set("analysis", analysisJson(analysis));
273
274 if (summary.nonFiniteSamples > 0) {
275 failures.push_back(std::to_string(summary.nonFiniteSamples) + " NaN/Inf sample(s)");
276 }
277 if (summary.silent && !options.allowSilence) {
278 failures.push_back("the render is silent");
279 }
280
281 // --- Golden ----------------------------------------------------------------
282 if (!options.writeExpectPath.empty()) {
283 if (!writeText(options.writeExpectPath, expectationJson(expectationFrom(summary, analysis)).toString(2))) {
284 failures.push_back("cannot write golden " + options.writeExpectPath);
285 return finish(report, failures, options, kCouldNotCheck);
286 }
287 report.set("wroteExpect", JSON(options.writeExpectPath));
288 }
289 if (!options.expectPath.empty()) {
290 RenderExpectation expected;
291 std::string error;
292 if (!readExpectation(options.expectPath, expected, error)) {
293 failures.push_back(error);
294 return finish(report, failures, options, kCouldNotCheck);
295 }
296 const auto diffs = compareToExpectation(summary, analysis, expected);
297 report.set("expect", JSON(options.expectPath));
298 report.set("expectDiffs", stringArray(diffs));
299 for (const auto& d : diffs) {
300 failures.push_back("golden: " + d);
301 }
302 }
303
304 return finish(report, failures, options, failures.empty() ? kPass : kFailed);
305}
