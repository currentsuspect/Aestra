2// AestraTransient V1 — feed-forward transient shaper (envelope-follower attack/decay
3// gain scaling). Zero latency: output[i] depends only on inputs up to i, no
4// lookahead, no reported latency, no PDC.
5//
6// Signal path (mono-linked detector, per-channel gain application):
7// in -> |x| -> fast envelope (peaks) + slow envelope (body)
8// -> normalized difference -> tanh-bounded gain computer
9// -> smoothed gain -> output trim -> mix with dry
10//
11// The fast envelope tracks onsets within ~1 ms while the slow envelope holds the
12// steady body, so their normalized difference is positive on an attack portion
13// and negative on a decay portion. The Attack amount scales the gain on the
14// attack portion, the Sustain amount scales it on the decay portion; both are
15// bipolar around neutral.

63 // Detector time constants (seconds). Fixed by design: the workhorse brief
64 // exposes only the two amounts, not the detector tuning.
65 static constexpr float kFastAttackSec = 0.001f;
66 static constexpr float kFastReleaseSec = 0.008f;
67 static constexpr float kSlowAttackSec = 0.025f;
68 static constexpr float kSlowReleaseSec = 0.120f;
69 static constexpr float kMaxSwingDb = 12.0f;
70 static constexpr float kGainSmoothSec = 0.002f;
71 static constexpr float kEnvFloor = 1.0e-6f;

95 void process(const float* const* inputs, float** outputs, uint32_t numInputChannels, uint32_t numOutputChannels,
96 uint32_t numFrames, const MidiBuffer* midiInput = nullptr, MidiBuffer* midiOutput = nullptr) AESTRA_RT_NONBLOCKING override {
97 (void)midiInput;
98 (void)midiOutput;
99
100 // Zero reported latency means internal bypass is a plain copy.
101 if (!m_active.load(std::memory_order_relaxed) || isBypassed()) {
102 copyOrClear(inputs, outputs, numInputChannels, numOutputChannels, numFrames);
103 m_wasBypassed = true;
104 return;
105 }
106 if (m_wasBypassed) {
107 // Flush stale envelope/gain state from before the bypass so
108 // reactivation does not emit it.
109 resetRuntimeState();
110 snapSmoothedParams();
111 m_wasBypassed = false;
112 }
113
114 const uint32_t channels = std::min<uint32_t>(2, numOutputChannels);
115 const bool stereo = channels >= 2;
116
117 float blockInputPeak = 0.0f;
118 float blockOutputPeak = 0.0f;
119
120 float fast = m_fast;
121 float slow = m_slow;
122 float gainSm = m_gainSmoothed;
123
124 for (uint32_t blockStart = 0; blockStart < numFrames; blockStart += kCtrlBlock) {
125 const uint32_t blockEnd = std::min(blockStart + kCtrlBlock, numFrames);
126
127 // Smoothing coefficient derives from the actual segment length so
128 // the 2 ms time constant holds for partial segments and short
129 // host callbacks alike (block-size independent).
130 const float segSmoothCoeff = 1.0f - std::exp(-static_cast<float>(blockEnd - blockStart) /
131 (0.002f * static_cast<float>(m_sampleRate)));
132 m_attackSmoothed += (getParameter(kAttack) - m_attackSmoothed) * segSmoothCoeff;
133 m_sustainSmoothed += (getParameter(kSustain) - m_sustainSmoothed) * segSmoothCoeff;
134 m_outputSmoothed += (getParameter(kOutput) - m_outputSmoothed) * segSmoothCoeff;
135 m_mixSmoothed += (getParameter(kMix) - m_mixSmoothed) * segSmoothCoeff;
136
137 const float amtA = bipolarFromNorm(m_attackSmoothed);
138 const float amtS = bipolarFromNorm(m_sustainSmoothed);
139 const float outGain = dbToLinear(outputDbFromNorm(m_outputSmoothed));
140 const float wet = std::clamp(m_mixSmoothed, 0.0f, 1.0f);
141 const float dry = 1.0f - wet;
142
143 for (uint32_t i = blockStart; i < blockEnd; ++i) {
144 const float inL = sanitizeSample(readInput(inputs, numInputChannels, 0, i));
145 const float inR = stereo ? sanitizeSample(readInput(inputs, numInputChannels, 1, i)) : inL;
146 blockInputPeak = std::max(blockInputPeak, std::max(std::abs(inL), std::abs(inR)));
147
148 const float abs = std::max(std::abs(inL), std::abs(inR));
149 fast += (abs - fast) * (abs > fast ? m_fastAttackCoeff : m_fastReleaseCoeff);
150 slow += (abs - slow) * (abs > slow ? m_slowAttackCoeff : m_slowReleaseCoeff);
151
152 // Normalized envelope difference: positive on an attack
153 // portion, negative on a decay portion, bounded via tanh.
154 const float diff = (fast - slow) / std::max(slow, kEnvFloor);
155 const float shaped = std::tanh(diff);
156 const float gainDb = (diff >= 0.0f ? amtA * shaped : amtS * (-shaped)) * kMaxSwingDb;
157 const float gain = dbToLinear(gainDb);
158
159 // One-pole smoothing of the applied gain prevents zipper noise
160 // when the gain computer jumps between attack and decay.
161 gainSm += (gain - gainSm) * m_gainSmoothCoeff;
162 if (!std::isfinite(gainSm) || gainSm < 1.0e-12f)
163 gainSm = 1.0f; // NaN guard + denormal flush; unity is the safe state
164
165 const float outL = flushDenormal((inL * gainSm * outGain) * wet + inL * dry);
166 const float outR = flushDenormal((inR * gainSm * outGain) * wet + inR * dry);
167 blockOutputPeak = std::max(blockOutputPeak, std::max(std::abs(outL), std::abs(outR)));
168
169 if (numOutputChannels > 0 && outputs[0])
170 outputs[0][i] = outL;
171 if (numOutputChannels > 1 && outputs[1])
172 outputs[1][i] = stereo ? outR : outL;
173 for (uint32_t ch = 2; ch < numOutputChannels; ++ch) {
174 if (outputs[ch])
175 outputs[ch][i] = sanitizeSample(readInput(inputs, numInputChannels, ch, i));
176 }
177 }
178 }
179
180 m_fast = flushEnvelope(fast);
181 m_slow = flushEnvelope(slow);
182 m_gainSmoothed = gainSm;
183 m_inputLevel.store(blockInputPeak, std::memory_order_relaxed);
184 m_outputLevel.store(blockOutputPeak, std::memory_order_relaxed);
185 }

196 std::string getParameterDisplay(uint32_t id) const override {
197 if (id >= kParamCount)
198 return "";
199 const float v = getParameter(id);
200 char buf[32];
201 switch (id) {
202 case kAttack: {
203 const float amt = bipolarFromNorm(v);
204 std::snprintf(buf, sizeof(buf), "%+.0f%%", amt * 100.0f);
205 return buf;
206 }
207 case kSustain: {
208 const float amt = bipolarFromNorm(v);
209 std::snprintf(buf, sizeof(buf), "%+.0f%%", amt * 100.0f);
210 return buf;
211 }
212 case kOutput:
213 std::snprintf(buf, sizeof(buf), "%+.1f dB", outputDbFromNorm(v));
214 return buf;
215 case kMix:
216 return std::to_string(static_cast<int>(std::round(v * 100.0f))) + "%";
217 case kBypass:
218 return v > 0.5f ? "ON" : "OFF";
219 default:
220 return "";
221 }

237 // ── Parameter mapping (public: shared with editor/tests) ──
238 static float bipolarFromNorm(float value) { return std::clamp(value, 0.0f, 1.0f) * 2.0f - 1.0f; }

240 static float outputDbFromNorm(float value) { return -12.0f + std::clamp(value, 0.0f, 1.0f) * 24.0f; }

245 void resetRuntimeState() {
246 m_fast = 0.0f;
247 m_slow = 0.0f;
248 m_gainSmoothed = 1.0f;
249 m_wasBypassed = false;
250 const float sr = static_cast<float>(m_sampleRate);
251 // Per-sample one-pole coefficients from the fixed detector constants.
252 m_fastAttackCoeff = 1.0f - std::exp(-1.0f / std::max(1.0f, sr * kFastAttackSec));
253 m_fastReleaseCoeff = 1.0f - std::exp(-1.0f / std::max(1.0f, sr * kFastReleaseSec));
254 m_slowAttackCoeff = 1.0f - std::exp(-1.0f / std::max(1.0f, sr * kSlowAttackSec));
255 m_slowReleaseCoeff = 1.0f - std::exp(-1.0f / std::max(1.0f, sr * kSlowReleaseSec));
256 m_gainSmoothCoeff = 1.0f - std::exp(-1.0f / std::max(1.0f, sr * kGainSmoothSec));
257 m_inputLevel.store(0.0f, std::memory_order_relaxed);
258 m_outputLevel.store(0.0f, std::memory_order_relaxed);
259 }

268 static float flushEnvelope(float value) { return (!std::isfinite(value) || value < 1.0e-12f) ? 0.0f : value; }

270 // ── Utility ──
271 static void copyOrClear(const float* const* inputs, float** outputs, uint32_t numInputChannels,
272 uint32_t numOutputChannels, uint32_t numFrames) {
273 for (uint32_t ch = 0; ch < numOutputChannels; ++ch) {
274 if (outputs[ch] && ch < numInputChannels && inputs[ch]) {
275 if (inputs[ch] != outputs[ch]) {
276 std::memcpy(outputs[ch], inputs[ch], numFrames * sizeof(float));
277 }
278 } else if (outputs[ch]) {
279 std::memset(outputs[ch], 0, numFrames * sizeof(float));
280 }
281 }
282 }

284 static float readInput(const float* const* inputs, uint32_t channels, uint32_t channel, uint32_t frame) {
285 if (channel >= channels || !inputs[channel]) {
286 return (channel == 1 && channels > 0 && inputs[0]) ? inputs[0][frame] : 0.0f;
287 }
288 return inputs[channel][frame];
289 }

291 static float sanitizeSample(float sample) {
292 if (!std::isfinite(sample))
293 return 0.0f;
294 return std::clamp(sample, -16.0f, 16.0f);
295 }

297 static float dbToLinear(float db) { return std::exp(db * 0.11512925464970229f); }
298
299 static float flushDenormal(float value) { return std::abs(value) < 1.0e-20f ? 0.0f : value; }
