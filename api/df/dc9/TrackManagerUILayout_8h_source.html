4// V8-X2b phase 4: the timeline's regions and track rows, resolved once by the
5// layout system instead of re-derived by hand in each TrackManagerUI file.
6//
7// Before this, layoutTracks(), the drop preview, both "place on timeline"
8// paths, three hit tests and two scroll-range calculations each restated
9// "time band height + index * (row height + spacing) - scroll", and two of
10// them carried a comment saying they MUST match layoutTracks() exactly. That
11// agreement was kept by hand. Everything here answers in the component's own
12// LOCAL space (origin at its top-left); TrackManagerUI's bounds are
13// window-absolute, and the one conversion between the two is localToWindow()
14// at the call site — the convention the TrackManagerUIToolbar.cpp:186 comment
15// used to explain in prose.
16//
17// Header-only and free of widget dependencies, like TrackManagerUIMath.h, so
18// a test can include it without the UI targets that AESTRA_CI=ON disables.

47inline TimelineLayout resolveTimelineLayout(float width, float height, float trackControlsWidth,
48 AestraUI::Layout::NUILayoutRecorder* recorder = nullptr) {
49 using namespace AestraUI::Layout;
50 const NUILocalRect component(0.0f, 0.0f, width, height);
51 const auto name = [recorder](const char* scope, const char* from) {
52 if (recorder) {
53 recorder->scope(scope, from);
54 }
55 };
56
57 name("bands", "");
58 const auto bands = splitVertical(component, kTimelineTimeBandHeight, recorder);
59 name("body", "bands.trailing");
60 const auto body = splitHorizontal(bands.trailing, width - kTimelineScrollbarWidth, recorder);
61
62 name("bandColumns", "bands.leading");
63 const auto bandColumns = splitHorizontal(bands.leading, width - kTimelineScrollbarWidth, recorder);
64 name("plane", "bandColumns.leading");
65 const auto plane = splitHorizontal(bandColumns.leading, trackControlsWidth, recorder);
66 name("minimapRow", "plane.trailing");
67 const auto minimapRow = splitVertical(plane.trailing, kTimelineMinimapHeight, recorder);
68
69 TimelineLayout layout;
70 layout.minimap = minimapRow.leading;
71 layout.scrollbar = body.trailing;
72 layout.rows = body.leading;
73 return layout;
74}
