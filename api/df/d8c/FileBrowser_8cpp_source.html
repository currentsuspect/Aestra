66float computeNavigationWidth(float browserWidth) {
67 if (browserWidth <= kCompactNavStartWidth) {
68 return std::min(browserWidth, kCompactNavWidth);
69 }
70
71 const float expandedWidth = std::clamp(browserWidth * 0.34f, 118.0f, 188.0f);
72 if (browserWidth >= kExpandedNavStartWidth) {
73 return expandedWidth;
74 }
75
76 const float expandedAtBreakpoint = kExpandedNavStartWidth * 0.34f;
77 const float progress = (browserWidth - kCompactNavStartWidth) / (kExpandedNavStartWidth - kCompactNavStartWidth);
78 return kCompactNavWidth + progress * (expandedAtBreakpoint - kCompactNavWidth);
79}

99// Greedy word wrap for placeholder hint copy, which is longer than the list
100// strip at narrow browser widths.
101static std::vector<std::string> wrapHintLines(NUIRenderer& renderer, const std::string& hint, float fontSize,
102 float maxWidth) {
103 std::vector<std::string> lines;
104 std::string remaining = hint;
105 while (!remaining.empty()) {
106 std::string line = remaining;
107 std::string rest;
108 while (renderer.measureText(line, fontSize).width > maxWidth) {
109 const size_t space = line.find_last_of(' ');
110 if (space == std::string::npos) break; // single unbreakable word
111 rest = line.substr(space + 1) + (rest.empty() ? "" : " " + rest);
112 line = line.substr(0, space);
113 }
114 lines.push_back(line);
115 remaining = rest;
116 }
117 return lines;
118}

120// Greedy wrap leaves a short orphan last line ("...appear in / the browser").
121// Once the line count is known, re-wrap near the average width so the lines
122// come out visually balanced; keep the greedy result if that would add a line.
123static std::vector<std::string> wrapHintBalanced(NUIRenderer& renderer, const std::string& hint, float fontSize,
124 float maxWidth) {
125 auto lines = wrapHintLines(renderer, hint, fontSize, maxWidth);
126 if (lines.size() < 2) return lines;
127 const float total = renderer.measureText(hint, fontSize).width;
128 const float target = (total / static_cast<float>(lines.size())) * 1.2f;
129 auto balanced = wrapHintLines(renderer, hint, fontSize, std::clamp(target, maxWidth * 0.5f, maxWidth));
130 return balanced.size() == lines.size() ? balanced : lines;
131}

133std::string ellipsizeMiddle(NUIRenderer& renderer, const std::string& text, float fontSize, float maxWidth) {
134 constexpr const char* kEllipsis = "...";
135
136 if (text.empty()) return text;
137 if (maxWidth <= 0.0f) return "";
138
139 // Check full string first (common case)
140 if (renderer.measureText(text, fontSize).width <= maxWidth) {
141 return text;
142 }
143
144 // Measure prefix (first 60%) and suffix (last 40%) to maintain context
145 size_t prefixLen = static_cast<size_t>(text.size() * 0.6);
146 size_t suffixLen = text.size() - prefixLen;
147
148 std::string prefix = text.substr(0, prefixLen);
149 std::string suffix = text.substr(text.size() - suffixLen);
150
151 // Shrink suffix first, then prefix
152 while (!suffix.empty() && renderer.measureText(prefix + kEllipsis + suffix, fontSize).width > maxWidth) {
153 suffix = suffix.substr(1);
154 }
155 while (!prefix.empty() && renderer.measureText(prefix + kEllipsis + suffix, fontSize).width > maxWidth) {
156 prefix = prefix.substr(0, prefix.size() - 1);
157 }
158
159 return prefix + kEllipsis + suffix;
160}

162static int parseBpmFromFilename(const std::string& name) {
163 // Matches: "kick_120bpm", "loop 128 BPM", "90bpm_hat", "140_bpm_loop"
164 for (size_t i = 0; i + 2 < name.size(); ++i) {
165 if (!std::isdigit(static_cast<unsigned char>(name[i]))) continue;
166 size_t start = i;
167 while (i < name.size() && std::isdigit(static_cast<unsigned char>(name[i]))) ++i;
168 size_t end = i;
169 if (end - start < 2 || end - start > 3) continue;
170 std::string numStr = name.substr(start, end - start);
171 int bpm = std::stoi(numStr);
172 if (bpm < 60 || bpm > 300) continue;
173 // Check for "bpm" nearby (before or after the number)
174 std::string context = name.substr(start > 4 ? start - 4 : 0,
175 std::min(name.size() - start, end + 5));
176 std::string lower;
177 for (char c : context) lower += static_cast<char>(std::tolower(static_cast<unsigned char>(c)));
178 if (lower.find("bpm") != std::string::npos) return bpm;
179 }
180 return 0;
181}

183std::string ellipsizeEnd(NUIRenderer& renderer, const std::string& text, float fontSize, float maxWidth) {
184 if (text.empty()) return text;
185 if (maxWidth <= 0.0f) return "";
186
187 if (renderer.measureText(text, fontSize).width <= maxWidth) return text;
188
189 constexpr const char* kEllipsis = "...";
190 const float ellipsisW = renderer.measureText(kEllipsis, fontSize).width;
191 if (ellipsisW >= maxWidth) return text;
192
193 static constexpr size_t MIN_VISIBLE = 24;
194 const size_t startChars = std::min(MIN_VISIBLE, text.size());
195 std::string minCandidate = text.substr(0, startChars) + kEllipsis;
196 if (renderer.measureText(minCandidate, fontSize).width > maxWidth) return minCandidate;
197
198 int low = static_cast<int>(startChars);
199 int high = static_cast<int>(text.size());
200 std::string best = minCandidate;
201
202 while (low <= high) {
203 int mid = low + (high - low) / 2;
204 std::string candidate = text.substr(0, mid) + kEllipsis;
205 if (renderer.measureText(candidate, fontSize).width <= maxWidth) {
206 best = candidate;
207 low = mid + 1;
208 } else {
209 high = mid - 1;
210 }
211 }
212
213 return best;
214}

239bool isPathUnderRoot(const std::filesystem::path& candidatePath, const std::filesystem::path& rootPath) {
240 const std::string candidate = normalizedPathForCompare(candidatePath);
241 std::string root = normalizedPathForCompare(rootPath);
242 if (root.empty()) return true;
243
244 // Allow exact match.
245 if (candidate == root) return true;
246
247 // Ensure `root/` prefix match.
248 if (root.back() != '/') root.push_back('/');
249 if (candidate.size() < root.size()) return false;
250 return candidate.compare(0, root.size(), root) == 0;
251}

253std::string resolveExistingDirectoryPath(const std::string& requestedPath, const std::string& rootPath) {
254 namespace fs = std::filesystem;
255
256 std::error_code ec;
257 fs::path root = rootPath.empty() ? fs::path() : fs::path(rootPath);
258 fs::path candidate = requestedPath.empty() ? root : fs::path(requestedPath);
259
260 if (!root.empty() && !isPathUnderRoot(candidate, root)) {
261 candidate = root;
262 }
263
264 while (!candidate.empty()) {
265 if (fs::exists(candidate, ec) && fs::is_directory(candidate, ec)) {
266 if (!root.empty() && !isPathUnderRoot(candidate, root)) {
267 break;
268 }
269 return canonicalOrNormalized(candidate).string();
270 }
271
272 const fs::path parent = candidate.parent_path();
273 if (parent.empty() || parent == candidate) {
274 break;
275 }
276 candidate = parent;
277 }
278
279 if (!root.empty() && fs::exists(root, ec) && fs::is_directory(root, ec)) {
280 return canonicalOrNormalized(root).string();
281 }
282
283 if (!root.empty()) {
284 return {};
285 }
286
287 return requestedPath;
288}

587void FileBrowser::scanWorkerLoop() {
588 while (true) {
589 ScanTask task;
590 {
591 std::unique_lock<std::mutex> lock(scanMutex_);
592 scanCv_.wait(lock, [&]() {
593 return scanStop_.load(std::memory_order_acquire) || !scanTasks_.empty();
594 });
595
596 if (scanStop_.load(std::memory_order_acquire) && scanTasks_.empty()) {
597 return;
598 }
599
600 task = std::move(scanTasks_.front());
601 scanTasks_.pop_front();
602 }
603
604 const uint64_t currentGen = scanGeneration_.load(std::memory_order_acquire);
605 if (task.generation != currentGen) {
606 continue;
607 }
608
609 ScanResult result;
610 result.kind = task.kind;
611 result.path = task.path;
612 result.depth = task.depth;
613 result.generation = task.generation;
614 result.items = scanDirectory(task.path, task.depth, task.showHidden, task.generation, result.error);
615
616 {
617 std::lock_guard<std::mutex> lock(scanMutex_);
618 scanResults_.push_back(std::move(result));
619 }
620 }

664std::vector<FileItem> FileBrowser::scanDirectory(const std::string& path, int depth, bool showHidden,
665 uint64_t generation, std::string& error) const {
666 std::vector<FileItem> items;
667 error.clear();
668 try {
669 const std::filesystem::path dir(path);
670 const auto options = std::filesystem::directory_options::skip_permission_denied;
671 std::error_code iterEc;
672 std::filesystem::directory_iterator it(dir, options, iterEc);
673 if (iterEc) {
674 error = iterEc.message();
675 Log::warning(std::string("[FileBrowser] Scan failed for ") + path + ": " + iterEc.message());
676 return items;
677 }
678
679 for (; it != std::filesystem::directory_iterator(); it.increment(iterEc)) {
680 if (scanStop_.load(std::memory_order_acquire) ||
681 generation != scanGeneration_.load(std::memory_order_acquire)) {
682 break;
683 }
684
685 if (iterEc) {
686 error = iterEc.message();
687 Log::warning(std::string("[FileBrowser] Scan iteration failed for ") + path + ": " + iterEc.message());
688 break;
689 }
690
691 const auto& entry = *it;
692
693 const std::string name = entry.path().filename().string();
694 if (!showHidden && !name.empty() && name[0] == '.') {
695 continue;
696 }
697
698 const std::string entryPath = entry.path().string();
699
700 // --- SMART FILTER APPLIED HERE ---
701 // If it's not a directory and not in our whitelist, skip it.
702 std::error_code dirEc;
703 const bool isDir = entry.is_directory(dirEc);
704 if (dirEc) continue;
705
706 if (!isDir && !FileFilter::isAllowed(entryPath)) {
707 continue; // Whitelist filter
708 }
709
710 FileType type = FileFilter::getType(entryPath, isDir);
711 size_t size = 0;
712 std::string lastModified;
713
714 if (!isDir) {
715 std::error_code sizeEc;
716 size = static_cast<size_t>(entry.file_size(sizeEc));
717 if (sizeEc) size = 0;
718 }
719
720 FileItem item(name, entryPath, type, isDir, size, lastModified);
721 item.depth = depth;
722 if (!isDir && (type == FileType::AudioFile || type == FileType::MusicFile ||
723 type == FileType::WavFile || type == FileType::Mp3File ||
724 type == FileType::FlacFile)) {
725 item.detectedBpm = parseBpmFromFilename(name);
726 }
727 items.push_back(std::move(item));
728 }
729 } catch (const std::exception& e) {
730 error = e.what();
731 Log::warning(std::string("[FileBrowser] Scan failed for ") + path + ": " + e.what());
732 }
733
734 return items;
735}

737FileItem* FileBrowser::findItemByPath(const std::string& path) {
738 std::function<FileItem*(std::vector<FileItem>&)> findRecursive = [&](std::vector<FileItem>& items) -> FileItem* {
739 for (auto& item : items) {
740 if (item.path == path) return &item;
741 if (!item.children.empty()) {
742 if (auto* found = findRecursive(item.children)) return found;
743 }
744 }
745 return nullptr;
746 };

748 return findRecursive(rootItems_);

751void FileBrowser::processScanResults() {
752 std::deque<ScanResult> results;
753 {
754 std::lock_guard<std::mutex> lock(scanMutex_);
755 if (scanResults_.empty()) return;
756 results.swap(scanResults_);
757 }
758
759 const uint64_t currentGen = scanGeneration_.load(std::memory_order_acquire);
760 bool didUpdate = false;
761
762 for (auto& result : results) {
763 if (result.generation != currentGen) continue;
764
765 if (result.kind == ScanKind::Root) {
766 scanningRoot_ = false;
767 scanError_ = std::move(result.error);
768
769 rootItems_ = std::move(result.items);
770 sortFiles();
771 updateDisplayList();
772
773 if (isFilterActive()) {
774 applyFilter();
775 } else {
776 filteredFiles_.clear();
777 viewDirty_ = true;
778 if (!pendingSelectionPath_.empty()) {
779 const std::string restoredPath = pendingSelectionPath_;
780 pendingSelectionPath_.clear();
781 selectFile(restoredPath);
782 } else if (!displayItems_.empty()) {
783 selectedIndex_ = 0;
784 selectedFile_ = displayItems_[0];
785 selectedIndices_.clear();
786 selectedIndices_.push_back(0);
787 lastShiftSelectIndex_ = 0;
788 } else {
789 clearSelection();
790 }
791 updateScrollbarVisibility();
792 invalidateCache();
793 }
794
795 didUpdate = true;
796 continue;
797 }
798
799 if (result.kind == ScanKind::Folder) {
800 if (FileItem* folder = findItemByPath(result.path)) {
801 folder->children = std::move(result.items);
802 folder->hasLoadedChildren = result.error.empty();
803 folder->isLoadingChildren = false;
804
805 if (!result.error.empty()) {
806 FileItem placeholder("Folder unavailable — collapse and reopen to retry", "",
807 FileType::Unknown, false, 0, "");
808 placeholder.depth = folder->depth + 1;
809 placeholder.isPlaceholder = true;
810 folder->children.push_back(std::move(placeholder));
811 }
812
813 std::stable_sort(folder->children.begin(), folder->children.end(),
814 [this](const FileItem& a, const FileItem& b) { return compareFileItems(a, b); });
815
816 updateDisplayList();
817 if (isFilterActive()) {
818 applyFilter();
819 } else {
820 updateScrollbarVisibility();
821 invalidateCache();
822 }
823 didUpdate = true;
824 }
825 }
826 }
827
828 if (didUpdate) {
829 updateScrollbarVisibility();
830 }
831}

1450void FileBrowser::onRender(NUIRenderer& renderer) {
1451 AESTRA_ZONE("FileBrowser_Render");
1452
1453 if (!isVisible()) return;
1454
1455 NUIRect bounds = getBounds();
1456 if (bounds.isEmpty()) return;
1457
1458 // Specialized views (Plugins, Patterns, future browser modes) share this
1459 // geometry even when the File Browser's static FBO cache is reused.
1460 updateContentViews();

1462 // FBO Caching Logic
1463 auto* renderCache = renderer.getRenderCache();
1464 if (!renderCache || !renderCache->isEnabled()) {
1465 // Fallback: Immediate render

1467 renderHoverOverlays(renderer);

1474
1475 // Cache size matches the component bounds
1476 AestraUI::NUISize cacheSize(static_cast<int>(bounds.width), static_cast<int>(bounds.height));

1478 // Get/Create Cache
1479 // We use a shared_ptr<void> member to hold the cache reference if NUI supports it,
1480 // or just look it up. TrackManagerUI uses getOrCreateCache returning SharedRenderCache.

1690 lastMousePos_ = event.position;
1691 NUIRect bounds = getBounds();

1693 auto& themeManager = NUIThemeManager::getInstance();
1694 float itemHeight = BROWSER_LIST_ROW_H;
1695
1696 const BrowserLayout browserLayout = computeBrowserLayout();
1697 float listY = browserLayout.list.y;
1698 float listHeight = browserLayout.list.height;
1699
1700 // Update scrollbar track height ensuring it is fresh for this event
1701 scrollbarTrackHeight_ = listHeight;
1702
1703 // If a click happens outside the search input, drop focus so shortcuts/navigation work normally.
1704 if (searchInput_ && event.pressed && event.button == NUIMouseButton::Left) {
1705 if (searchInput_->isFocused() && !searchInput_->getBounds().contains(event.position)) {
1706 searchInput_->setFocused(false);
1707 }
1708 }
1709
1710 if (handleChromeMouse(event, browserLayout)) {
1711 return true;
1712 }
1713
1714 // Claim keyboard focus when interacting with the file browser (so it can own arrow-key navigation).
1715 if (event.pressed && event.button == NUIMouseButton::Left) {
1716 if (bounds.contains(event.position)) {
1717 // If the click is on the search input, it will take focus itself.
1718 if (!searchInput_ || !searchInput_->getBounds().contains(event.position)) {
1719 setFocused(true);
1720 }
1721 }
1722 }
1723
1724 if (handleActiveDragMouse(event)) {
1725 return true;
1726 }
1727
1728 // Popup menu handling (context + dropdowns)
1729 if (NUIComponent::onMouseEvent(event)) {
1730 return true;
1731 }
1732 if (popupMenu_ && popupMenu_->isVisible() &&
1733 event.pressed && (event.button == NUIMouseButton::Left || event.button == NUIMouseButton::Right) &&
1734 !popupMenu_->getBounds().contains(event.position)) {
1735 hidePopupMenu();

1737 }
1738
1739 if (handleDragInitiation(event, view)) {
1740 return true;
1741 }
1742
1743 // If we're dragging the scrollbar, handle mouse events even outside bounds
1744 if (isDraggingScrollbar_) {
1745 // Always check scrollbar events if we're dragging
1746 if (handleScrollbarMouseEvent(event)) {
1747 return true;
1748 }
1749 }
1750
1751 // Check if mouse is within bounds
1752 bool mouseInside = bounds.contains(event.position.x, event.position.y);
1753
1754 if (handleNavigationMouseEvent(event, browserLayout)) {
1755 return true;
1756 }
1757
1758 if (handleWheelScroll(event, browserLayout, mouseInside, view)) {
1759 return true;
1760 }
1761

1763 if (!mouseInside && !isDraggingScrollbar_) {
1764 bool dirty = false;
1765 if (hoveredIndex_ != -1) {
1766 hoveredIndex_ = -1;
1767 dirty = true;
1768 }
1769 if (dirty)
1770 setDirty(true); // hover overlay only — no cache rebuild
1771 return false;
1772 }
1773
1774 // Scrollbar visibility for this event (list overflow)
1775 const float contentHeight = view.size() * itemHeight;
1776 const float maxScroll = std::max(0.0f, contentHeight - scrollbarTrackHeight_);
1777 const bool needsScrollbar = maxScroll > 0.0f;
1778
1779 updateBreadcrumbHover(event);
1780
1781 // Breadcrumb interaction
1782 if (handleBreadcrumbMouseEvent(event)) {
1783 return true;
1784 }
1785
1786 // Check scrollbar events if scrollbar is needed (but not dragging - handled above)
1787 if (needsScrollbar && !view.empty() && !isDraggingScrollbar_) {
1788 if (handleScrollbarMouseEvent(event)) {
1789 return true;
1790 }
1791 }
1792
1793 if (handleListMouse(event, view, browserLayout)) {
1794 return true;
1795 }
1796
1797 if (mouseInside && event.pressed && event.button == NUIMouseButton::Right) {
1798 hidePopupMenu();
1799 return true;
1800 }
1801
1802 return false;
1803}

1879 float dy = event.position.y - dragStartPos_.y;

1882 if (dist >= dragManager.getDragThreshold()) {

1885 if (!dragFile->isDirectory && FileFilter::isAllowed(dragFile->path)) {
1886 AestraUI::DragData dragData;

1893 dragData.displayName = dragFile->name;
1894 dragData.accentColor = NUIThemeManager::getInstance().getColor("accentPrimary");
1895 dragData.previewWidth = 150.0f;
1896 dragData.previewHeight = 30.0f;
1897
1898 dragManager.beginDrag(dragData, dragStartPos_, this);
1899 isDraggingFile_ = true;
1900 dragPotential_ = false;
1901 return true;
1902 }
1903 dragPotential_ = false;
1904 dragSourceIndex_ = -1;
1905 return true;
1906 }
1907 }
1908
1909 // Cancel drag potential on mouse release
1910 if (!event.pressed && event.button == NUIMouseButton::Left) {

1912 dragSourceIndex_ = -1;
1913 }

1918 const float itemHeight = BROWSER_LIST_ROW_H;

1923 if (navOverflow > 0.0f) {

1926 invalidateCache();

1928 }

1946 // Clamp target scroll offset
1947 targetScrollOffset_ = std::max(0.0f, std::min(targetScrollOffset_, maxScroll));
1948

1950 return true; // Consume the wheel event
1951 }

2893 bool FileBrowser::compareFileItems(const FileItem& a, const FileItem& b) const {
2894 // Priority: Search Score (descending) -> Folders First -> Name/etc (stable tie-break)
2895
2896 if (searchInput_ && !searchInput_->getText().empty()) {
2897 if (a.searchScore != b.searchScore) {
2898 return a.searchScore > b.searchScore; // Higher score first
2899 }
2900 // If scores equal, fall through to standard sort for stability
2901 }
2902
2903 if (a.isDirectory != b.isDirectory) {
2904 return a.isDirectory > b.isDirectory; // folders first
2905 }
2906
2907 const auto tieBreak = [&]() {
2908 if (a.name != b.name) {
2909 return sortAscending_ ? (a.name < b.name) : (a.name > b.name);
2910 }
2911 return a.path < b.path;
2912 };
2913
2914 switch (sortMode_) {
2915 case SortMode::Name:
2916 return tieBreak();
2917 case SortMode::Type:
2918 if (a.type != b.type) return sortAscending_ ? (a.type < b.type) : (a.type > b.type);
2919 return tieBreak();
2920 case SortMode::Size:
2921 if (a.size != b.size) return sortAscending_ ? (a.size < b.size) : (a.size > b.size);
2922 return tieBreak();
2923 case SortMode::Modified:
2924 if (a.lastModified != b.lastModified) {
2925 return sortAscending_ ? (a.lastModified < b.lastModified) : (a.lastModified > b.lastModified);
2926 }
2927 return tieBreak();
2928 }
2929 return tieBreak();
2930 }

3690bool FileBrowser::handleScrollbarMouseEvent(const NUIMouseEvent& event) {
3691 const BrowserLayout browserLayout = computeBrowserLayout();
3692 float scrollbarX = browserLayout.list.right() - scrollbarWidth_ - 2.0f;

3694
3695 // Use the member variable scrollbarTrackHeight_ for consistency
3696 // It's set in onResize() and used for thumb calculation
3697
3698 const auto& view = getActiveView();
3699
3700 // If we're dragging, continue dragging regardless of mouse position
3701 if (isDraggingScrollbar_) {
3702 scrollbarFadeTimer_ = 0.0f;
3703 scrollbarOpacity_ = 1.0f;
3704 // Stop dragging on mouse release (anywhere, not just in scrollbar area)
3705 if (!event.pressed && event.button == NUIMouseButton::Left) {
3706 isDraggingScrollbar_ = false;

3708 }
3709
3710 // Continue dragging even if mouse is outside scrollbar area
3711 float deltaY = event.position.y - dragStartY_;
3712 float scrollRatio = deltaY / scrollbarTrackHeight_;

3717 targetScrollOffset_ = dragStartScrollOffset_ + (scrollRatio * maxScroll);
3718 scrollOffset_ = targetScrollOffset_; // Instant during drag, no lerp
3719
3720 // Clamp scroll offset
3721 scrollOffset_ = std::max(0.0f, std::min(scrollOffset_, maxScroll));
3722 targetScrollOffset_ = scrollOffset_;
3723
3724 invalidateCache();
3725 return true;
3726 }
3727
3728 // Check if mouse is over scrollbar area (with padding for easier clicking)
3729 bool inScrollbarArea = (event.position.x >= scrollbarX - 10 &&
3730 event.position.x <= scrollbarX + scrollbarWidth_ + 10 &&
3731 event.position.y >= scrollbarY - 10 &&
3732 event.position.y <= scrollbarY + scrollbarTrackHeight_ + 10);
3733
3734 if (scrollbarHovered_ != inScrollbarArea) {
3735 scrollbarHovered_ = inScrollbarArea;
3736 invalidateCache();
3737 }
3738
3739 // Hovering the scrollbar should reveal it (auto-hide UX)
3740 if (inScrollbarArea) {
3741 scrollbarFadeTimer_ = 0.0f;
3742 if (scrollbarOpacity_ < 1.0f) {
3743 scrollbarOpacity_ = 1.0f;
3744 invalidateCache();
3745 }
3746 }
3747
3748 if (!inScrollbarArea) {
3749 return false;
3750 }
3751
3752
3753 if (event.pressed && event.button == NUIMouseButton::Left) {
3754 scrollbarFadeTimer_ = 0.0f;
3755 scrollbarOpacity_ = 1.0f;
3756 // Check if clicking on thumb or track (with padding)
3757 float thumbAbsoluteY = scrollbarY + scrollbarThumbY_;
3758 if (event.position.y >= thumbAbsoluteY - 10 &&
3759 event.position.y <= thumbAbsoluteY + scrollbarThumbHeight_ + 10) {
3760 // Start dragging thumb
3761 isDraggingScrollbar_ = true;
3762 dragStartY_ = event.position.y;
3763 dragStartScrollOffset_ = scrollOffset_;
3764 } else {
3765 // Click on track - jump to position
3766 float relativeY = event.position.y - scrollbarY;
3767 float scrollRatio = relativeY / scrollbarTrackHeight_;
3768 float itemHeight = BROWSER_LIST_ROW_H;
3769 float maxScroll = std::max(0.0f, (view.size() * itemHeight) - scrollbarTrackHeight_);
3770 // Direct jump to clicked position - set both for instant response

3775 scrollOffset_ = std::max(0.0f, std::min(scrollOffset_, maxScroll));

4153
4154 if (isExtensionSearch) {
4155 // Rule 1: Extension Match (ends_with)
4156 if (hay.length() >= needle.length()) {
4157 matchesSearch = (hay.compare(hay.length() - needle.length(), needle.length(), needle) == 0);
4158 score = 1000; // High score for exact extension
4159 } else {

4163 else if (isSubstringSearch) {
4164 // Rule 2: Substring-ish (all chars in order, contiguous ideally)
4165 // For "kick.wav" seeking "kick.wav" -> exact substring
4166 size_t foundPos = hay.find(needle);
4167 matchesSearch = (foundPos != std::string::npos);

4170 }
4171 }
4172 else {
4173 // Rule 3: Fuzzy Subsequence with Scoring
4174 // -1 gap, +10 start, +5 start of word, +5 contiguous
4175 // Penalty: -len/10
4176
4177 size_t nIdx = 0;
4178 size_t hIdx = 0;
4179 int gapPenalty = 0;
4180 int bonuses = 0;
4181 int contiguousRun = 0;
4182 bool firstCharMatched = false;
4183
4184 // Track start of match for scoring
4185 int firstMatchIdx = -1;
4186
4187 while (nIdx < needle.length() && hIdx < hay.length()) {
4188 if (needle[nIdx] == hay[hIdx]) {
4189 if (firstMatchIdx == -1) firstMatchIdx = static_cast<int>(hIdx);
4190
4191 // Start of string bonus
4192 if (hIdx == 0) bonuses += 10;
4193
4194 // Start of word bonus (check prev char for separator)
4195 if (hIdx > 0) {
4196 char prev = hay[hIdx - 1];
4197 if (prev == '_' || prev == '-' || prev == ' ' || prev == '.') {
4198 bonuses += 5;
4199 }
4200 }
4201
4202 // Contiguous bonus
4203 if (nIdx > 0 && hIdx > 0 && needle[nIdx-1] == hay[hIdx-1]) { // Logic check: actually just checking if we matched prev loop
4204 // This logic is slightly flawed for "contiguous in haystack", simplistic approach:
4205 contiguousRun++;
4206 if (contiguousRun > 0) bonuses += 5;
4207 } else {
4208 contiguousRun = 0;
4209 }
4210
4211 nIdx++;
4212 } else {
4213 // Gap
4214 if (firstMatchIdx != -1) gapPenalty -= 1; // Only penalize gaps inside the match span?
4215 // Or simple: penalize every skipped char
4216 }
4217 hIdx++; // Always advance haystack
4218 }
4219
4220 matchesSearch = (nIdx == needle.length()); // Found all chars
4221
4222 if (matchesSearch) {
4223 // Simple fuzzy score calculation re-pass or simplification
4224 // Since the above verification loop is greedy, it might not find optimal alignment.
4225 // For UI responsiveness, greedy is usually fine.
4226
4227 // Add penalty for total length to prefer shorter files
4228 int lengthPenalty = static_cast<int>(hay.length()) / 10;
4229
4230 score = bonuses + gapPenalty - lengthPenalty;
4231 }
4232 }
4233 }
4234
4235 bool matchesTag = true;
4236 if (matchesTag && hasTagFilter) {
4237 matchesTag = hasTag(item->path, activeTagFilter_);
4238 }
4239 bool matchesType = !hasQuickFilter || matchesQuickFilter(*item);
4240
4241 if (matchesSearch && matchesTag && matchesType) {
4242 item->searchScore = score;
4243 filteredFiles_.push_back(item);
4244 }
4245 }
4246
4247 sortFiles(); // Will use searchScore if query is active
4248
4249 selectedFile_ = nullptr;
4250 selectedIndex_ = -1;
4251 selectedIndices_.clear();
4252 updateScrollbarVisibility();
4253 viewDirty_ = true;
4254 invalidateCache();
