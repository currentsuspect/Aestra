35 std::string fitLabel(NUIRenderer& renderer, const std::string& text, float fontSize, float maxWidth)
36 {
37 if (renderer.measureText(text, fontSize).width <= maxWidth) return text;
38 std::string clipped = text;
39 const std::string ellipsis = "...";
40 while (!clipped.empty() &&
41 renderer.measureText(clipped + ellipsis, fontSize).width > maxWidth) {
42 clipped.pop_back();
43 }
44 return clipped.empty() ? ellipsis : clipped + ellipsis;
45 }

93void UIRoutingMap::rebuildGraph() {
94 // Preserve existing node positions so user-dragged layouts survive rebuilds
95 std::unordered_map<uint32_t, NUIPoint> savedPositions;
96 for (const auto& n : m_nodes) {
97 savedPositions[n.id] = {n.x, n.y};
98 }
99
100 m_nodes.clear();
101 m_edges.clear();
102
103 if (!m_viewModel) return;
104
105 // Minimap: focused view (selected track + its destinations only)
106 if (m_mode == Mode::Minimap) {
107 int32_t selId = m_viewModel->getSelectedChannelId();
108 const Aestra::ChannelViewModel* selected = nullptr;
109 if (selId >= 0) {
110 selected = m_viewModel->getChannelById(static_cast<uint32_t>(selId));
111 }
112 rebuildFocusedGraph(selected);
113 autoLayout();
114 computeWorldBounds();
115 return;
116 }
117
118 size_t channelCount = m_viewModel->getChannelCount();
119 const Aestra::ChannelViewModel* master = m_viewModel->getMaster();
120
121 // Master node (id = 0)
122 if (master) {
123 Node masterNode;
124 masterNode.id = 0;
125 masterNode.type = Node::Master;
126 masterNode.label = "MASTER";
127 masterNode.color = 0xFF808080;
128 masterNode.insertCount = 0;
129 m_nodes.push_back(masterNode);
130 }
131
132 // Track nodes
133 for (size_t i = 0; i < channelCount; ++i) {
134 const auto* ch = m_viewModel->getChannelByIndex(i);
135 if (!ch || ch->id == 0) continue; // skip master if it snuck in
136
137 Node node;
138 node.id = ch->id;
139 node.type = Node::Track;
140 node.label = ch->name.empty() ? ("Track " + std::to_string(i + 1)) : ch->name;
141 node.color = paletteIndexToARGB(ch->trackColorIndex);
142 node.insertCount = ch->fxCount;
143 node.muted = ch->muted;
144 node.soloed = ch->soloed;
145 node.peakDb = std::max(ch->smoothedPeakL, ch->smoothedPeakR);
146 if (m_viewModel) {
147 node.hasRoutingWarning = !m_viewModel->getRoutingWarning(ch->id).empty();
148 }
149 for (size_t in = 0; in < ch->inserts.size() && node.insertNames.size() < 4; ++in) {
150 if (!ch->inserts[in].isEmpty && !ch->inserts[in].name.empty()) {
151 node.insertNames.push_back(ch->inserts[in].name);
152 }
153 }
154 m_nodes.push_back(node);
155
156 // Main path edge
157 uint32_t destId = ch->mainOutputId;
158 if (destId == 0xFFFFFFFFu) destId = 0;
159 if (ch->masterSendEnabled || destId != 0) {
160 Edge edge;
161 edge.sourceNodeId = ch->id;
162 edge.targetNodeId = destId;
163 edge.type = Edge::MainPath;
164 m_edges.push_back(edge);
165 }
166
167 // Send edges
168 for (size_t s = 0; s < ch->sends.size(); ++s) {
169 const auto& send = ch->sends[s];
170 uint32_t sendTargetId = send.targetId;
171 if (sendTargetId == 0xFFFFFFFFu) sendTargetId = 0;
172
173 Edge edge;
174 edge.sourceNodeId = ch->id;
175 edge.targetNodeId = sendTargetId;
176 edge.type = send.sidechainOnly ? Edge::SidechainPath : Edge::SendPath;
177 edge.sendLevelDb = gainToDb(send.gain);
178 edge.sendIndex = static_cast<int>(s);
179 m_edges.push_back(edge);
180 }
181 }
182
183 autoLayout();
184
185 // Restore user-dragged positions for nodes that existed before
186 bool hasNewNodes = false;
187 for (auto& node : m_nodes) {
188 auto it = savedPositions.find(node.id);
189 if (it != savedPositions.end()) {
190 node.x = it->second.x;
191 node.y = it->second.y;
192 } else {
193 hasNewNodes = true; // at least one node didn't exist before
194 }
195 }
196
197 computeWorldBounds();
198 if (m_mode == Mode::FullPanel && hasNewNodes) {
199 m_fitPending = true;
200 }

226 node.muted = selected->muted;

231 }

233 if (!selected->inserts[in].isEmpty && !selected->inserts[in].name.empty()) {

269 tn.color = paletteIndexToARGB(target->trackColorIndex);
270 tn.insertCount = target->fxCount;
271 tn.muted = target->muted;
272 tn.soloed = target->soloed;
273 tn.peakDb = std::max(target->smoothedPeakL, target->smoothedPeakR);
274 if (m_viewModel && targetId != 0) {
275 tn.hasRoutingWarning = !m_viewModel->getRoutingWarning(targetId).empty();
276 }
277 for (size_t in = 0; in < target->inserts.size() && tn.insertNames.size() < 4; ++in) {
278 if (!target->inserts[in].isEmpty && !target->inserts[in].name.empty()) {
279 tn.insertNames.push_back(target->inserts[in].name);
280 }
281 }
282 m_nodes.push_back(tn);
283 }
284 }
285
286 Edge edge;
287 edge.sourceNodeId = selected->id;
288 edge.targetNodeId = targetId;
289 edge.type = send.sidechainOnly ? Edge::SidechainPath : Edge::SendPath;
290 edge.sendLevelDb = gainToDb(send.gain);
291 m_edges.push_back(edge);

360void UIRoutingMap::autoLayout() {
361 if (m_mode == Mode::Minimap) {
362 // Compact representation: tracks left column, master right, square-ish layout
363 // Adapts node spacing to track count so 20+ tracks remain readable.
364 std::vector<Node*> tracks;
365 Node* masterNode = nullptr;
366 for (auto& n : m_nodes) {
367 if (n.type == Node::Master) masterNode = &n;
368 else tracks.push_back(&n);
369 }
370
371 // Use small uniform spacing; renderMinimap() rescales to fit bounds.
372 float trackW = kMinimapTrackW;
373 float trackH = kMinimapTrackH;
374 float masterW = kMinimapMasterW;
375 float masterH = kMinimapMasterH;
376 float gap = 100.0f;
377 float spacing = 6.0f;
378
379 for (size_t i = 0; i < tracks.size(); ++i) {
380 tracks[i]->x = 0.0f;
381 tracks[i]->y = static_cast<float>(i) * (trackH + spacing);
382 tracks[i]->w = trackW;
383 tracks[i]->h = trackH;
384 tracks[i]->outputX = trackW;
385 tracks[i]->outputY = trackH * 0.5f;
386 tracks[i]->inputX = 0.0f;
387 tracks[i]->inputY = trackH * 0.5f;
388 }
389
390 if (masterNode) {
391 masterNode->x = trackW + gap;
392 // Center master vertically against track column
393 float colHeight = tracks.empty() ? masterH
394 : tracks.size() * trackH + (tracks.size() - 1) * spacing;
395 masterNode->y = colHeight * 0.5f - masterH * 0.5f;
396 masterNode->w = masterW;
397 masterNode->h = masterH;
398 masterNode->inputX = 0.0f;
399 masterNode->inputY = masterH * 0.5f;
400 masterNode->outputX = masterW;
401 masterNode->outputY = masterH * 0.5f;
402 }
403 } else {
404 // Full panel: flow-ordered columns instead of a rigid index grid.
405 // Tracks that other tracks feed (send/bus targets) gravitate toward the
406 // column nearest the master — horizontal position follows signal flow.
407 // Partial columns are vertically centred against the tallest one, and
408 // alternate columns get a half-row stagger so cross-column wires pass
409 // between nodes instead of colliding head-on.
410 std::vector<Node*> tracks;
411 Node* masterNode = nullptr;
412 for (auto& n : m_nodes) {
413 if (n.type == Node::Master) masterNode = &n;
414 else tracks.push_back(&n);
415 }
416
417 // Bus-like = target of at least one track-to-track edge (send or route).
418 std::unordered_map<uint32_t, int> inDegree;
419 for (const auto& e : m_edges) {
420 if (e.targetNodeId != 0) ++inDegree[e.targetNodeId];
421 }
422 std::stable_sort(tracks.begin(), tracks.end(), [&](const Node* a, const Node* b) {
423 const bool aBus = inDegree.find(a->id) != inDegree.end();
424 const bool bBus = inDegree.find(b->id) != inDegree.end();
425 return !aBus && bBus; // pure sources first, bus-like last (stable)
426 });
427
428 const int trackCount = static_cast<int>(tracks.size());
429 // Cap rows per column so vertical extent stays reasonable
430 constexpr int kMaxRowsPerColumn = 14;
431 int numColumns = std::max(1, (trackCount + kMaxRowsPerColumn - 1) / kMaxRowsPerColumn);
432 int rowsPerColumn = std::max(1, (trackCount + numColumns - 1) / numColumns);
433
434 float trackW = kTrackNodeW;
435 float trackH = kTrackNodeH;
436 // Breathing room: a bit looser than the old grid in both axes
437 float vSpacing = (numColumns > 1) ? 18.0f : kTrackSpacing;
438 float colStride = trackW + 88.0f;
439 const float rowStride = trackH + vSpacing;
440 const float fullColH = rowsPerColumn * trackH + (rowsPerColumn - 1) * vSpacing;
441
442 for (int i = 0; i < trackCount; ++i) {
443 const int col = i / rowsPerColumn;
444 const int row = i % rowsPerColumn;
445 const int rowsInCol = std::min(rowsPerColumn, trackCount - col * rowsPerColumn);
446 const float colH = rowsInCol * trackH + (rowsInCol - 1) * vSpacing;
447 const float yBase = (fullColH - colH) * 0.5f; // centre partial columns
448 const float stagger = (col % 2 == 1) ? rowStride * 0.5f : 0.0f; // brick offset
449 tracks[i]->x = static_cast<float>(col) * colStride;
450 tracks[i]->y = yBase + stagger + static_cast<float>(row) * rowStride;
451 tracks[i]->w = trackW;
452 tracks[i]->h = trackH;
453 tracks[i]->outputX = trackW;
454 tracks[i]->outputY = trackH * 0.5f;
455 tracks[i]->inputX = 0.0f;
456 tracks[i]->inputY = trackH * 0.5f;
457 }
458
459 if (masterNode) {
460 // Master sits to the right of the last column
461 float lastColX = static_cast<float>(numColumns - 1) * colStride;
462 masterNode->x = lastColX + trackW + 140.0f;
463 masterNode->y = fullColH * 0.5f - kMasterNodeH * 0.5f;
464 masterNode->w = kMasterNodeW;
465 masterNode->h = kMasterNodeH;
466 masterNode->inputX = 0.0f;
467 masterNode->inputY = kMasterNodeH * 0.5f;
468 masterNode->outputX = kMasterNodeW;
469 masterNode->outputY = kMasterNodeH * 0.5f;
470 }
471 }

474void UIRoutingMap::updateEdgeAttachments() {
475 // Spread incoming wires along each target node's left edge instead of
476 // terminating them all on a single pin — at 50 tracks the master turned
477 // into a solid grey fan. Attachments are ordered by the source's vertical
478 // position so wires arrive sorted and don't cross each other at the node.
479 std::unordered_map<uint32_t, const Node*> byId;
480 byId.reserve(m_nodes.size());
481 for (const auto& n : m_nodes) byId[n.id] = &n;
482
483 std::unordered_map<uint32_t, std::vector<size_t>> incoming;
484 for (size_t i = 0; i < m_edges.size(); ++i) {
485 incoming[m_edges[i].targetNodeId].push_back(i);
486 }
487
488 for (auto& [targetId, edgeIdxs] : incoming) {
489 auto dstIt = byId.find(targetId);
490 if (dstIt == byId.end()) continue;
491 const Node* dst = dstIt->second;
492
493 if (edgeIdxs.size() < 2) {
494 for (size_t i : edgeIdxs) m_edges[i].dstPinYOffset = -1.0f;
495 continue;
496 }
497
498 auto sourceY = [&](const Edge& e) {
499 auto it = byId.find(e.sourceNodeId);
500 return (it != byId.end()) ? it->second->y + it->second->outputY : 0.0f;
501 };
502 std::sort(edgeIdxs.begin(), edgeIdxs.end(), [&](size_t a, size_t b) {
503 const float ya = sourceY(m_edges[a]);
504 const float yb = sourceY(m_edges[b]);
505 if (ya != yb) return ya < yb;
506 return a < b; // stable tiebreak for equal heights
507 });
508
509 const float pad = dst->h * 0.16f;
510 const float span = dst->h - pad * 2.0f;
511 const float count = static_cast<float>(edgeIdxs.size());
512 for (size_t k = 0; k < edgeIdxs.size(); ++k) {
513 m_edges[edgeIdxs[k]].dstPinYOffset =
514 pad + span * ((static_cast<float>(k) + 0.5f) / count);
515 }
516 }
517}

523void UIRoutingMap::onUpdate(double deltaTime) {
524 if (!m_viewModel) return;
525
526 // Animate live-wire pulse (cycles 0..1 at ~1.5 Hz base)
527 m_livePulsePhase += static_cast<float>(deltaTime) * 1.5f;
528 if (m_livePulsePhase > 1.0f) m_livePulsePhase -= 1.0f;
529
530 // Search bar caret blink (~1 Hz)
531 if (m_searchFocused) {
532 m_searchCaretTimer += static_cast<float>(deltaTime);
533 if (m_searchCaretTimer >= 0.5f) {
534 m_searchCaretTimer -= 0.5f;
535 m_searchCaretVisible = !m_searchCaretVisible;
536 repaint();
537 }
538 } else {
539 m_searchCaretTimer = 0.0f;
540 m_searchCaretVisible = true;
541 }
542
543 // Cache global solo state for edge dimming
544 m_anySoloed = false;
545 for (const auto& node : m_nodes) {
546 if (node.soloed) { m_anySoloed = true; break; }
547 }
548
549 // Rebuild graph if channel count or selection changed
550 size_t channelCount = m_viewModel->getChannelCount();
551 int32_t selectedId = m_viewModel->getSelectedChannelId();
552 // Normalize -1 (no selection) to sentinel 0xFFFFFFFFu for stable comparison
553 uint32_t selectedIdU = (selectedId < 0) ? 0xFFFFFFFFu : static_cast<uint32_t>(selectedId);
554 if (m_graphDirty ||
555 m_lastChannelCount != static_cast<uint32_t>(channelCount) ||
556 m_lastSelectedId != selectedIdU) {
557 rebuildGraph();
558 m_graphDirty = false;
559 m_lastChannelCount = static_cast<uint32_t>(channelCount);
560 m_lastSelectedId = selectedIdU;
561 repaint();
562 } else {
563 // Just refresh live meters / mute / solo each frame without rebuild
564 refreshLiveState();
565 repaint();
566 }
567
568 // Auto-fit on first show or after resize
569 if (m_mode == Mode::FullPanel && m_fitPending && !m_nodes.empty()) {
570 fitToView();
571 repaint();
572 }
573}

595void UIRoutingMap::onRender(NUIRenderer& renderer) {
596 // Recompute per frame: attachment order depends on node positions, which
597 // change during drags. O(E log E) on ≤ a few hundred edges — negligible.

1377void UIRoutingMap::drawBezier(NUIRenderer& renderer, const NUIPoint& a, const NUIPoint& b,
1378 float thickness, const NUIColor& color, bool dashed) {
1379 // Cubic bezier with a short horizontal lead/tail so the curve leaves the source
1380 // and enters the target perpendicular-ish, but heads toward target Y quickly
1381 // (so it avoids passing through intermediate same-row nodes).
1382 float dx = std::abs(b.x - a.x);
1383 float lead = std::min(dx * 0.35f, 40.0f * m_zoom + 20.0f);
1384 float cp1x = a.x + lead;
1385 float cp1y = a.y;
1386 float cp2x = b.x - lead;
1387 float cp2y = b.y;
1388
1389 // Draw as polyline with adaptive subdivision
1390 constexpr int kSegments = 24;
1391 std::vector<NUIPoint> points;
1392 points.reserve(kSegments + 1);
1393
1394 for (int i = 0; i <= kSegments; ++i) {
1395 float t = static_cast<float>(i) / static_cast<float>(kSegments);
1396 float u = 1.0f - t;
1397 float tt = t * t;
1398 float uu = u * u;
1399 float uuu = uu * u;
1400 float ttt = tt * t;
1401
1402 float x = uuu * a.x + 3.0f * uu * t * cp1x + 3.0f * u * tt * cp2x + ttt * b.x;
1403 float y = uuu * a.y + 3.0f * uu * t * cp1y + 3.0f * u * tt * cp2y + ttt * b.y;
1404 points.push_back({x, y});
1405 }
1406
1407 if (dashed) {
1408 // Draw dashed: 4px on, 4px off
1409 const float dashOn = 4.0f;
1410 const float dashOff = 4.0f;
1411 float accumulated = 0.0f;
1412 bool drawing = true;
1413
1414 for (size_t i = 1; i < points.size(); ++i) {
1415 float segDx = points[i].x - points[i - 1].x;
1416 float segDy = points[i].y - points[i - 1].y;
1417 float segLen = std::sqrt(segDx * segDx + segDy * segDy);
1418
1419 float segStart = accumulated;
1420 float segEnd = accumulated + segLen;
1421 float pos = segStart;
1422
1423 while (pos < segEnd - 0.5f) {
1424 float dashPhase = std::fmod(pos, dashOn + dashOff);
1425 bool inDash = dashPhase < dashOn;
1426
1427 float dashEnd = pos + (inDash ? (dashOn - dashPhase) : (dashOn + dashOff - dashPhase));
1428 dashEnd = std::min(dashEnd, segEnd);
1429
1430 if (inDash) {
1431 float t0 = (pos - segStart) / segLen;
1432 float t1 = (dashEnd - segStart) / segLen;
1433 NUIPoint p0{points[i - 1].x + segDx * t0, points[i - 1].y + segDy * t0};
1434 NUIPoint p1{points[i - 1].x + segDx * t1, points[i - 1].y + segDy * t1};
1435 renderer.drawLine(p0, p1, thickness, color);
1436 }
1437
1438 pos = dashEnd;
1439 }
1440
1441 accumulated = segEnd;
1442 }
1443 } else {
1444 renderer.drawPolyline(points.data(), static_cast<int>(points.size()), thickness, color);
1445 }
1446}

1488int UIRoutingMap::hitTestEdge(const NUIPoint& p) const {
1489 // Simple distance-to-line-segment test on the bezier polyline
1490 constexpr float kHitDist = 6.0f;
1491 constexpr int kSegments = 12;
1492
1493 for (size_t e = 0; e < m_edges.size(); ++e) {
1494 const auto& edge = m_edges[e];
1495 const Node* src = nullptr;
1496 const Node* dst = nullptr;
1497 for (const auto& n : m_nodes) {
1498 if (n.id == edge.sourceNodeId) src = &n;
1499 if (n.id == edge.targetNodeId) dst = &n;
1500 }
1501 if (!src || !dst) continue;
1502
1503 float ax, ay, bx, by;
1504 if (m_mode == Mode::FullPanel) {
1505 NUIRect bounds = getBounds();
1506 float canvasY = bounds.y + 44.0f;
1507 ax = bounds.x + (src->x + src->outputX) * m_zoom + m_cameraX;
1508 ay = canvasY + (src->y + src->outputY) * m_zoom + m_cameraY;
1509 bx = bounds.x + (dst->x + dst->inputX) * m_zoom + m_cameraX;
1510 by = canvasY + edgeTargetY(edge, *dst) * m_zoom + m_cameraY;
1511 } else {
1512 ax = m_minimapOffsetX + (src->x + src->outputX - m_minimapCenterX) * m_minimapScale;
1513 ay = m_minimapOffsetY + (src->y + src->outputY - m_minimapCenterY) * m_minimapScale;
1514 bx = m_minimapOffsetX + (dst->x + dst->inputX - m_minimapCenterX) * m_minimapScale;
1515 by = m_minimapOffsetY + (edgeTargetY(edge, *dst) - m_minimapCenterY) * m_minimapScale;
1516 }
1517
1518 float dx = std::abs(bx - ax);
1519 float cp1x = ax + dx * 0.5f;
1520 float cp1y = ay;
1521 float cp2x = bx - dx * 0.5f;
1522 float cp2y = by;
1523
1524 for (int i = 0; i < kSegments; ++i) {
1525 float t0 = static_cast<float>(i) / kSegments;
1526 float t1 = static_cast<float>(i + 1) / kSegments;
1527
1528 auto bezierPoint = [&](float t) -> NUIPoint {
1529 float u = 1.0f - t;
1530 float tt = t * t;
1531 float uu = u * u;
1532 float uuu = uu * u;
1533 float ttt = tt * t;
1534 float x = uuu * ax + 3.0f * uu * t * cp1x + 3.0f * u * tt * cp2x + ttt * bx;
1535 float y = uuu * ay + 3.0f * uu * t * cp1y + 3.0f * u * tt * cp2y + ttt * by;
1536 return {x, y};
1537 };
1538
1539 NUIPoint a0 = bezierPoint(t0);
1540 NUIPoint b0 = bezierPoint(t1);
1541
1542 // Distance from point p to line segment a0-b0
1543 float lx = b0.x - a0.x;
1544 float ly = b0.y - a0.y;
1545 float lenSq = lx * lx + ly * ly;
1546 float t = (lenSq > 0.0f) ? std::clamp(((p.x - a0.x) * lx + (p.y - a0.y) * ly) / lenSq, 0.0f, 1.0f) : 0.0f;
1547 float closestX = a0.x + t * lx;
1548 float closestY = a0.y + t * ly;
1549 float distSq = (p.x - closestX) * (p.x - closestX) + (p.y - closestY) * (p.y - closestY);
1550 if (distSq <= kHitDist * kHitDist) {
1551 return static_cast<int>(e);
1552 }
1553 }
1554 }
1555 return -1;
1556}

1558int UIRoutingMap::hitTestOutputPort(const NUIPoint& p) const {
1559 for (size_t i = 0; i < m_nodes.size(); ++i) {
1560 const auto& node = m_nodes[i];
1561 if (node.type == Node::Master) continue; // master has no output port
1562 float nx, ny, nw, nh;
1563 float scale;
1564 if (m_mode == Mode::FullPanel) {
1565 NUIRect bounds = getBounds();
1566 float canvasY = bounds.y + 44.0f;
1567 nx = bounds.x + node.x * m_zoom + m_cameraX;
1568 ny = canvasY + node.y * m_zoom + m_cameraY;
1569 nw = node.w * m_zoom;
1570 nh = node.h * m_zoom;
1571 scale = m_zoom;
1572 } else {
1573 nx = m_minimapOffsetX + (node.x - m_minimapCenterX) * m_minimapScale;
1574 ny = m_minimapOffsetY + (node.y - m_minimapCenterY) * m_minimapScale;
1575 nw = node.w * m_minimapScale;
1576 nh = node.h * m_minimapScale;
1577 scale = m_minimapScale;
1578 }
1579 float portCx = nx + nw;
1580 float portCy = ny + nh * 0.5f;
1581 float portR = std::max(3.5f, kPortRadius * scale);
1582 float dx = p.x - portCx;
1583 float dy = p.y - portCy;
1584 if (dx * dx + dy * dy <= portR * portR + 16.0f) {
1585 return static_cast<int>(i);
1586 }
1587 }
1588 return -1;
1589}
