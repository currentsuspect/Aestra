48NUIPoint bezierAt(NUIPoint a, NUIPoint b, float t) {
49 const float dx = std::max(40.0f, std::abs(b.x - a.x) * 0.5f);
50 const NUIPoint c1{a.x + dx, a.y}, c2{b.x - dx, b.y};
51 const float u = 1.0f - t;
52 return {u * u * u * a.x + 3 * u * u * t * c1.x + 3 * u * t * t * c2.x + t * t * t * b.x,
53 u * u * u * a.y + 3 * u * u * t * c1.y + 3 * u * t * t * c2.y + t * t * t * b.y};
54}

100size_t UIRoutingMap::routingSignature() const {
101 if (!m_viewModel) return 0;
102 size_t h = 1469598103934665603ull;
103 auto mix = [&h](size_t v) { h = (h ^ v) * 1099511628211ull; };
104 for (size_t i = 0; i < m_viewModel->getChannelCount(); ++i) {
105 const auto* ch = m_viewModel->getChannelByIndex(i);
106 if (!ch) continue;
107 mix(ch->id);
108 mix(ch->mainOutputId);
109 mix(ch->masterSendEnabled ? 1 : 2);
110 mix(static_cast<size_t>(ch->role));
111 mix(static_cast<size_t>(ch->trackColorIndex + 7));
112 mix(std::hash<std::string>{}(ch->name));
113 mix(static_cast<size_t>(ch->fxCount));
114 for (const auto& s : ch->sends) {
115 mix(s.sendId);
116 mix(s.targetId);
117 mix(s.sidechainOnly ? 3 : 4);
118 mix(s.postFader ? 5 : 6);
119 mix(static_cast<size_t>(std::lround(gainToDb(s.gain) * 10.0f) + 2000));
120 }
121 }
122 mix(m_allTracks ? 11 : 12);
123 return h == 0 ? 1 : h;
124}

126void UIRoutingMap::rebuildGraph() {
127 m_channels.clear();
128 m_nodes.clear();
129 m_edges.clear();
130 m_direct.clear();
131 if (!m_viewModel) return;
132
133 std::unordered_map<uint32_t, const Aestra::ChannelViewModel*> byId;
134 for (size_t i = 0; i < m_viewModel->getChannelCount(); ++i) {
135 const auto* ch = m_viewModel->getChannelByIndex(i);
136 if (!ch || ch->id == 0) continue;
137 byId[ch->id] = ch;
138 RoutingGraph::Channel c;
139 c.id = ch->id;
140 c.output = ch->mainOutputId == kNoOutput ? RoutingGraph::kMaster : ch->mainOutputId;
141 c.outputAudible = c.output != RoutingGraph::kMaster || ch->masterSendEnabled;
142 c.role = roleOf(*ch);
143 for (const auto& s : ch->sends)
144 c.sends.push_back({s.targetId == kNoOutput ? RoutingGraph::kMaster : s.targetId, s.sidechainOnly});
145 m_channels.push_back(std::move(c));
146 }
147
148 m_plan = RoutingGraph::plan(m_channels, m_allTracks);
149 for (const auto& p : m_plan.nodes) {
150 const auto* ch = byId[p.id];
151 Node n;
152 n.id = p.id;
153 n.role = roleOf(*ch);
154 n.label = channelDisplayName(ch->id, ch->name);
155 n.color = paletteIndexToARGB(ch->trackColorIndex);
156 n.column = p.column;
157 n.x = static_cast<float>(p.column) * kColStride;
158 n.y = p.row * kRowPitch;
159 n.muted = ch->muted;
160 n.soloed = ch->soloed;
161 n.peakDb = std::max(ch->smoothedPeakL, ch->smoothedPeakR);
162 m_nodes.push_back(n);
163
164 const uint32_t out = ch->mainOutputId == kNoOutput ? RoutingGraph::kMaster : ch->mainOutputId;
165 if (out != RoutingGraph::kMaster || ch->masterSendEnabled) m_edges.push_back({p.id, out, Edge::Output});
166 for (const auto& s : ch->sends) {
167 Edge e;
168 e.source = p.id;
169 e.target = s.targetId == kNoOutput ? RoutingGraph::kMaster : s.targetId;
170 e.type = s.sidechainOnly ? Edge::Sidechain : Edge::Send;
171 e.levelDb = gainToDb(s.gain);
172 e.postFader = s.postFader;
173 e.sendId = s.sendId;
174 m_edges.push_back(e);
175 }
176 }
177 for (uint32_t id : m_plan.direct) {
178 const auto* ch = byId[id];
179 m_direct.push_back({id, channelDisplayName(ch->id, ch->name), paletteIndexToARGB(ch->trackColorIndex)});
180 }
181 if (m_selectedId != 0 && !byId.count(m_selectedId)) m_selectedId = 0;
182 m_trace = m_selectedId ? RoutingGraph::trace(m_channels, m_selectedId) : RoutingGraph::Trace{};
183 m_lit.assign(m_nodes.size(), false);
184 m_hoverNode = m_hoverEdge = m_hoverDirect = -1;
185}

193void UIRoutingMap::onUpdate(double deltaTime) {
194 if (!m_viewModel) return;
195
196 const int32_t vmSel = m_viewModel->getSelectedChannelId();
197 const size_t sig = routingSignature();
198 if (sig != m_signature) {
199 m_signature = sig;
200 rebuildGraph();
201 repaint();
202 }
203 if (vmSel != m_lastVmSelection) {
204 m_lastVmSelection = vmSel;
205 selectChannel(vmSel > 0 ? static_cast<uint32_t>(vmSel) : 0);
206 }
207 refreshLiveState();
208
209 if (m_signalFlow) {
210 bool any = false;
211 for (bool l : m_lit) any |= l;
212 if (any) {
213 m_flowPhase = std::fmod(m_flowPhase + static_cast<float>(deltaTime) * 0.8f, 1.0f);
214 repaint();
215 }
216 }
217 if (m_fitPending && !getBounds().isEmpty()) {
218 fitToView();
219 repaint();
220 }
221}

223void UIRoutingMap::refreshLiveState() {
224 if (!m_viewModel) return;
225 bool changed = false;
226 if (m_lit.size() != m_nodes.size()) m_lit.assign(m_nodes.size(), false);
227 for (size_t i = 0; i < m_nodes.size(); ++i) {
228 auto& n = m_nodes[i];
229 const auto* ch = m_viewModel->getChannelById(n.id);
230 if (!ch) continue;
231 n.peakDb = std::max(ch->smoothedPeakL, ch->smoothedPeakR);
232 const bool lit = n.peakDb > -60.0f && !ch->muted;
233 if (lit != m_lit[i] || n.muted != ch->muted || n.soloed != ch->soloed) changed = true;
234 m_lit[i] = lit;
235 n.muted = ch->muted;
236 n.soloed = ch->soloed;
237 }
238 if (const auto* m = m_viewModel->getMaster()) {
239 if (std::abs(m->smoothedPeakL - m_masterPeakL) > 0.5f || std::abs(m->smoothedPeakR - m_masterPeakR) > 0.5f) {
240 m_masterPeakL = m->smoothedPeakL;
241 m_masterPeakR = m->smoothedPeakR;
242 changed = true;
243 }
244 }
245 if (changed) repaint();
246}

251UIRoutingMap::Geometry UIRoutingMap::geometry() const {
252 const NUIRect b = getBounds();
253 Geometry g;
254 g.header = {b.x, b.y, b.width, kHeaderH};
255 const bool list = !m_allTracks && !m_direct.empty();
256 g.list = {b.x, b.y + kHeaderH, list ? kListW : 0.0f, std::max(0.0f, b.height - kHeaderH)};
257 const float cx = g.list.right();
258 const float detailH = (m_selectedId != 0 && !m_patching) ? kDetailH : 0.0f;
259 const float fullH = std::max(0.0f, b.bottom() - g.header.bottom());
260 g.detail = {cx, b.bottom() - detailH, std::max(0.0f, b.right() - cx), detailH};
261 g.canvas = {cx, g.header.bottom(), std::max(0.0f, b.right() - cx), std::max(0.0f, fullH - detailH)};
262 g.rail = {g.canvas.right() - kRailInset - kRailW, g.canvas.y + 16.0f, kRailW,
263 std::max(40.0f, g.canvas.height - 16.0f - 24.0f)};
264 return g;
265}

267void UIRoutingMap::fitToView() {
268 m_fitPending = false;
269 const Geometry g = geometry();
270 m_cameraX = kOriginX;
271 m_cameraY = kOriginY;
272 m_zoom = 1.0f;
273 if (m_nodes.empty()) return;
274 float maxX = 0.0f, maxY = 0.0f;
275 for (const auto& n : m_nodes) {
276 maxX = std::max(maxX, n.x + kNodeW);
277 maxY = std::max(maxY, n.y + kNodeH);
278 }
279 const float availW = g.rail.x - g.canvas.x - kOriginX - 64.0f;
280 const float availH = g.canvas.height - kOriginY - 48.0f;
281 // Never below 65 %: names stay readable and the rest pans.
282 if (availW > 0 && availH > 0) m_zoom = std::clamp(std::min({1.0f, availW / maxX, availH / maxY}), 0.65f, 1.0f);
283}

285void UIRoutingMap::clampCamera() {
286 const Geometry g = geometry();
287 float maxX = kNodeW, maxY = kNodeH;
288 for (const auto& n : m_nodes) {
289 maxX = std::max(maxX, n.x + kNodeW);
290 maxY = std::max(maxY, n.y + kNodeH);
291 }
292 // Keep at least 80 px of the graph on screen in each direction.
293 m_cameraX = std::clamp(m_cameraX, 80.0f - maxX * m_zoom, std::max(80.0f, g.canvas.width - 80.0f));
294 m_cameraY = std::clamp(m_cameraY, 80.0f - maxY * m_zoom, std::max(80.0f, g.canvas.height - 80.0f));
295}

334int UIRoutingMap::hitNode(const NUIPoint& p) const {
335 if (!geometry().canvas.contains(p)) return -1;
336 for (int i = static_cast<int>(m_nodes.size()) - 1; i >= 0; --i)
337 if (nodeRect(m_nodes[i]).contains(p)) return i;
338 return -1;
339}
340
341int UIRoutingMap::hitOutputPort(const NUIPoint& p) const {
342 if (!geometry().canvas.contains(p)) return -1;
343 for (int i = 0; i < static_cast<int>(m_nodes.size()); ++i) {
344 const NUIPoint o = outputPort(m_nodes[i]);
345 if (std::abs(p.x - o.x) <= kPortHit && std::abs(p.y - o.y) <= kPortHit) return i;
346 }
347 return -1;
348}
349
350int UIRoutingMap::hitDirect(const NUIPoint& p) const {
351 const Geometry g = geometry();
352 if (g.list.width <= 0 || !g.list.contains(p)) return -1;
353 const float y0 = g.list.y + kListHeadH - m_listScroll;
354 if (p.y < g.list.y + kListHeadH) return -1;
355 const int i = static_cast<int>(std::floor((p.y - y0) / kListRowH));
356 return (i >= 0 && i < static_cast<int>(m_direct.size())) ? i : -1;
357}
358
359int UIRoutingMap::hitEdge(const NUIPoint& p) const {
360 const Geometry g = geometry();
361 if (!g.canvas.contains(p)) return -1;
362 for (int i = 0; i < static_cast<int>(m_edges.size()); ++i) {
363 const Edge& e = m_edges[i];
364 if (e.type == Edge::Output) continue; // only sends carry a menu
365 const Node* s = nodeById(e.source);
366 const Node* t = nodeById(e.target);
367 if (!s || !t) continue;
368 const NUIPoint a = outputPort(*s);
369 const NUIPoint b = e.type == Edge::Sidechain ? sidechainPort(*t) : inputPort(*t);
370 for (int k = 0; k <= 24; ++k) {
371 const NUIPoint q = bezierAt(a, b, static_cast<float>(k) / 24.0f);
372 if (std::abs(q.x - p.x) < 5.0f && std::abs(q.y - p.y) < 5.0f) return i;
373 }
374 }
375 return -1;
376}

390void UIRoutingMap::beginPatch(uint32_t source, const NUIPoint& from, const NUIPoint& mouse) {
391 m_patching = true;
392 m_patchSource = source;
393 m_patchStartScreen = from;
394 m_patchMouse = mouse;
395 m_patchRefused.clear();
396 if (m_viewModel) {
397 // Anything not offered as a destination would loop (the view model asks the engine's routing authority).
398 std::unordered_set<uint32_t> ok;
399 for (const auto& d : m_viewModel->getAvailableDestinations(source)) ok.insert(d.id);
400 for (const auto& c : m_channels)
401 if (c.id != source && !ok.count(c.id)) m_patchRefused.push_back(c.id);
402 }
403 repaint();
404}

406uint32_t UIRoutingMap::patchTarget(const NUIPoint& mouse, bool& isMaster) const {
407 isMaster = overRail(mouse);
408 if (isMaster) return RoutingGraph::kMaster;
409 const int i = hitNode(mouse);
410 if (i < 0 || m_nodes[i].id == m_patchSource) return kNoOutput;
411 return m_nodes[i].id;
412}
413
414void UIRoutingMap::finishPatch(const NUIPoint& mouse) {
415 bool isMaster = false;
416 const uint32_t target = patchTarget(mouse, isMaster);
417 const uint32_t source = m_patchSource;
418 const bool alt = (static_cast<int>(m_patchMods) & static_cast<int>(NUIModifiers::Alt)) != 0;
419 const bool shift = (static_cast<int>(m_patchMods) & static_cast<int>(NUIModifiers::Shift)) != 0;
420 m_patching = false;
421 m_patchSource = 0;
422 if (target != kNoOutput && !(target != RoutingGraph::kMaster && refuses(target))) {
423 if (isMaster || (!alt && !shift)) {
424 if (m_onRerouteMain) m_onRerouteMain(source, target);
425 } else if (m_onAddSend) {
426 m_onAddSend(source, target, shift);
427 }
428 m_signature = 0; // rebuild from the new routing next frame
429 }
430 m_patchRefused.clear();
431 repaint();
432}

524bool UIRoutingMap::onMouseEvent(const NUIMouseEvent& event) {
525 const NUIPoint& p = event.position;
526 if (!getBounds().contains(p) && !m_patching && !m_panning) return false;
527 const Geometry g = geometry();
528 const bool left = event.button == NUIMouseButton::Left;
529
530 if (event.type == NUIMouseEventType::Move || event.pressed) {
531 const int hn = hitNode(p);
532 const int hd = hitDirect(p);
533 const int he = hn < 0 ? hitEdge(p) : -1;
534 int hb = -1;
535 for (int i = 0; i < 5; ++i)
536 if (m_buttons[i].contains(p)) hb = i;
537 if (hn != m_hoverNode || hd != m_hoverDirect || he != m_hoverEdge || hb != m_hoverButton) {
538 m_hoverNode = hn;
539 m_hoverDirect = hd;
540 m_hoverEdge = he;
541 m_hoverButton = hb;
542 repaint();
543 }
544 }
545
546 if (event.type == NUIMouseEventType::Move) {
547 if (m_patching) {
548 m_patchMouse = p;
549 m_patchMods = event.modifiers;
550 repaint();
551 return true;
552 }
553 if (m_pressNode >= 0 && !m_pressOnNode) {
554 // A press on a Direct row turns into a patch once it moves.
555 if (std::abs(p.x - m_pressPos.x) + std::abs(p.y - m_pressPos.y) > 5.0f) {
556 const auto& row = m_direct[static_cast<size_t>(m_pressNode)];
557 const float y = g.list.y + kListHeadH - m_listScroll + m_pressNode * kListRowH + kListRowH * 0.5f;
558 m_pressNode = -1;
559 beginPatch(row.id, {g.list.right() - 8.0f, y}, p);
560 m_patchMods = event.modifiers;
561 }
562 return true;
563 }
564 if (m_panning) {
565 m_cameraX = m_panCamX + (p.x - m_panStart.x);
566 m_cameraY = m_panCamY + (p.y - m_panStart.y);
567 clampCamera();
568 repaint();
569 return true;
570 }
571 return false;
572 }
573
574 if (event.type == NUIMouseEventType::Scroll) {
575 if (g.list.contains(p)) {
576 const float content = static_cast<float>(m_direct.size()) * kListRowH;
577 const float view = g.list.height - kListHeadH;
578 m_listScroll = std::clamp(m_listScroll - event.wheelDelta * 30.0f, 0.0f, std::max(0.0f, content - view));
579 repaint();
580 return true;
581 }
582 if (g.canvas.contains(p)) {
583 const float old = m_zoom;
584 m_zoom = std::clamp(m_zoom * (1.0f + event.wheelDelta * 0.08f), 0.35f, 2.0f);
585 const float wx = (p.x - g.canvas.x - m_cameraX) / old;
586 const float wy = (p.y - g.canvas.y - m_cameraY) / old;
587 m_cameraX = p.x - g.canvas.x - wx * m_zoom;
588 m_cameraY = p.y - g.canvas.y - wy * m_zoom;
589 clampCamera();
590 repaint();
591 return true;
592 }
593 return true;
594 }
595
596 if (event.pressed && event.button == NUIMouseButton::Right) {
597 if (m_patching) {
598 m_patching = false;
599 m_patchRefused.clear();
600 repaint();
601 return true;
602 }
603 for (const auto& [rect, sendId] : m_sendRows)
604 if (rect.contains(p)) {
605 openSendMenu(p, m_selectedId, sendId);
606 return true;
607 }
608 if (m_hoverNode >= 0) {
609 openNodeMenu(p, m_nodes[m_hoverNode].id);
610 return true;
611 }
612 if (m_hoverEdge >= 0) {
613 const Edge& e = m_edges[m_hoverEdge];
614 openSendMenu(p, e.source, e.sendId);
615 return true;
616 }
617 return true;
618 }
619
620 if (event.pressed && left) {
621 switch (m_hoverButton) {
622 case 0: setShowAllTracks(false); return true;
623 case 1: setShowAllTracks(true); return true;
624 case 2: m_signalFlow = !m_signalFlow; repaint(); return true;
625 case 3:
626 m_searchOpen = !m_searchOpen;
627 m_search.clear();
628 if (m_searchOpen) setFocused(true);
629 repaint();
630 return true;
631 case 4:
632 if (m_onCollapse) m_onCollapse();
633 return true;
634 default: break;
635 }
636 if (g.header.contains(p)) return true;
637 if (g.detail.height > 0 && g.detail.contains(p)) {
638 if (m_outputBox.contains(p)) openOutputMenu({m_outputBox.x, m_outputBox.bottom()}, m_selectedId);
639 else if (m_addSendRect.contains(p)) openAddSendMenu({m_addSendRect.x, m_addSendRect.bottom()}, m_selectedId);
640 else
641 for (const auto& [rect, sendId] : m_sendRows)
642 if (rect.contains(p)) openSendMenu(p, m_selectedId, sendId);
643 return true;
644 }
645 if (m_hoverDirect >= 0) {
646 m_pressNode = m_hoverDirect;
647 m_pressOnNode = false;
648 m_pressPos = p;
649 return true;
650 }
651 const int port = hitOutputPort(p);
652 if (port >= 0) {
653 beginPatch(m_nodes[port].id, outputPort(m_nodes[port]), p);
654 m_patchMods = event.modifiers;
655 return true;
656 }
657 if (m_hoverNode >= 0) {
658 if ((static_cast<int>(event.modifiers) & static_cast<int>(NUIModifiers::Shift)) != 0) {
659 if (m_onNodeSoloToggle) m_onNodeSoloToggle(m_nodes[m_hoverNode].id);
660 return true;
661 }
662 m_pressNode = m_hoverNode;
663 m_pressOnNode = true;
664 m_pressPos = p;
665 return true;
666 }
667 if (g.canvas.contains(p)) {
668 m_panning = true;
669 m_panStart = p;
670 m_panCamX = m_cameraX;
671 m_panCamY = m_cameraY;
672 return true;
673 }
674 return true;
675 }
676
677 if (event.released && left) {
678 if (m_patching) {
679 m_patchMods = event.modifiers;
680 finishPatch(p);
681 return true;
682 }
683 if (m_pressNode >= 0) {
684 const uint32_t id = m_pressOnNode ? m_nodes[static_cast<size_t>(m_pressNode)].id
685 : m_direct[static_cast<size_t>(m_pressNode)].id;
686 m_pressNode = -1;
687 selectChannel(id);
688 if (m_onNodeSelected) m_onNodeSelected(id);
689 return true;
690 }
691 if (m_panning) {
692 m_panning = false;
693 if (std::abs(p.x - m_panStart.x) + std::abs(p.y - m_panStart.y) < 4.0f) selectChannel(0);
694 return true;
695 }
696 }
697 return getBounds().contains(p);
698}

700bool UIRoutingMap::onKeyEvent(const NUIKeyEvent& event) {
701 if (m_patching) {
702 // Holding or letting go of Alt / Shift changes what the drop will do; show it at once.
703 m_patchMods = event.modifiers;
704 repaint();
705 }
706 if (!event.pressed) return false;
707
708 if (m_searchOpen) {
709 if (event.keyCode == NUIKeyCode::Escape) {
710 m_searchOpen = false;
711 m_search.clear();
712 repaint();
713 return true;
714 }
715 if (event.keyCode == NUIKeyCode::Backspace) {
716 if (!m_search.empty()) m_search.pop_back();
717 repaint();
718 return true;
719 }
720 if (event.keyCode == NUIKeyCode::Enter) {
721 const std::string q = lower(m_search);
722 for (const auto& n : m_nodes)
723 if (!q.empty() && lower(n.label).find(q) != std::string::npos) {
724 selectChannel(n.id);
725 if (m_onNodeSelected) m_onNodeSelected(n.id);
726 break;
727 }
728 m_searchOpen = false;
729 m_search.clear();
730 repaint();
731 return true;
732 }
733 if (event.character >= 32 && event.character < 127) {
734 m_search.push_back(event.character);
735 repaint();
736 return true;
737 }
738 }
739
740 if (event.keyCode == NUIKeyCode::Escape) {
741 if (m_patching) {
742 m_patching = false;
743 m_patchRefused.clear();
744 repaint();
745 return true;
746 }
747 if (m_selectedId != 0) {
748 selectChannel(0);
749 return true;
750 }
751 if (m_onCollapse) m_onCollapse();
752 return true;
753 }
754 if (event.keyCode == NUIKeyCode::F) {
755 m_fitPending = true;
756 repaint();
757 return true;
758 }
759 return false;
760}
