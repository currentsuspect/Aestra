22class InterleavedFifo {
23public:
24 void reset(size_t capacityFrames, uint32_t channels) {
25 m_channels = std::max<uint32_t>(1, channels);
26 m_capacity = std::max<size_t>(1, capacityFrames);
27 m_data.assign(m_capacity * m_channels, 0.0f);
28 clear();
29 }
30 void clear() noexcept {
31 m_read = 0;
32 m_frames = 0;
33 }
34 size_t frames() const noexcept { return m_frames; }
35 size_t space() const noexcept { return m_capacity - m_frames; }
36
38 size_t pushPlanar(const float* const* channels, uint32_t numChannels, size_t frames) noexcept {
39 frames = std::min(frames, space());
40 for (size_t f = 0; f < frames; ++f) {
41 float* dst = slot(m_frames + f);
42 for (uint32_t c = 0; c < m_channels; ++c) {
43 dst[c] = (channels && c < numChannels && channels[c]) ? channels[c][f] : 0.0f;
44 }
45 }
46 m_frames += frames;
47 return frames;
48 }
50 size_t pushInterleaved(const float* data, size_t frames) noexcept {
51 frames = std::min(frames, space());
52 for (size_t f = 0; f < frames; ++f) {
53 std::memcpy(slot(m_frames + f), data + f * m_channels, m_channels * sizeof(float));
54 }
55 m_frames += frames;
56 return frames;
57 }
59 size_t pushSilence(size_t frames) noexcept {
60 frames = std::min(frames, space());
61 for (size_t f = 0; f < frames; ++f) {
62 std::memset(slot(m_frames + f), 0, m_channels * sizeof(float));
63 }
64 m_frames += frames;
65 return frames;
66 }
68 void popInterleaved(float* dst, size_t frames) noexcept {
69 frames = std::min(frames, m_frames);
70 for (size_t f = 0; f < frames; ++f) {
71 std::memcpy(dst + f * m_channels, slot(f), m_channels * sizeof(float));
72 }
73 discard(frames);
74 }
76 void popPlanar(float* const* outputs, uint32_t numOutputs, size_t frames) noexcept {
77 frames = std::min(frames, m_frames);
78 for (uint32_t c = 0; c < numOutputs; ++c) {
79 if (!outputs || !outputs[c]) {
80 continue;
81 }
82 for (size_t f = 0; f < frames; ++f) {
83 outputs[c][f] = c < m_channels ? slot(f)[c] : 0.0f;
84 }
85 }
86 discard(frames);
87 }
88 void discard(size_t frames) noexcept {
89 frames = std::min(frames, m_frames);
90 m_read = (m_read + frames) % m_capacity;
91 m_frames -= frames;
92 }
93
94private:
95 float* slot(size_t offsetFromRead) noexcept {
96 return m_data.data() + ((m_read + offsetFromRead) % m_capacity) * m_channels;
97 }
98 std::vector<float> m_data;
99 size_t m_capacity{1};
100 uint32_t m_channels{1};
101 size_t m_read{0};
102 size_t m_frames{0};
103};

120 bool push(uint64_t frame, const uint8_t* data) noexcept {
121 if (m_count == kCapacity) {
122 return false;
123 }
124 Event& e = m_events[m_count++];
125 e.frame = frame;
126 std::memcpy(e.data, data, 3);
127 return true;
128 }
130 size_t takeChunk(uint64_t chunkStart, uint64_t chunkEnd, uint8_t* out, size_t outCapacityBytes) noexcept {
131 size_t bytes = 0;
132 size_t kept = 0;
133 for (size_t i = 0; i < m_count; ++i) {
134 const Event& e = m_events[i];
135 if (e.frame < chunkEnd) {
136 if (bytes + 8 <= outCapacityBytes) {
137 const uint32_t offset = static_cast<uint32_t>(e.frame > chunkStart ? e.frame - chunkStart : 0);
138 std::memcpy(out + bytes, &offset, sizeof(offset));
139 out[bytes + 4] = 3;
140 std::memcpy(out + bytes + 5, e.data, 3);
141 bytes += 8;
142 }
143 } else {
144 m_events[kept++] = e;
145 }
146 }
147 m_count = kept;
148 return bytes;
149 }
150 void clear() noexcept { m_count = 0; }
151
152private:
153 Event m_events[kCapacity]{};
154 size_t m_count{0};
155};
