78std::vector<double> spectrumShape(const Audio::Spectrum& s, std::vector<bool>& valid) {
79 const auto& c = Audio::Spectrum::centersHz();
80 double sum = 0.0;
81 int n = 0;
82 for (size_t b = 0; b < Audio::Spectrum::kBands; ++b)
83 if (c[b] >= 63.0 && c[b] <= 8000.0 && s.levelDb[b] > -150.0) { sum += s.levelDb[b]; ++n; }
84 std::vector<double> out(Audio::Spectrum::kBands, 0.0);
85 valid.assign(Audio::Spectrum::kBands, false);
86 if (n == 0) return out;
87 const double mean = sum / n;
88 for (size_t b = 0; b < Audio::Spectrum::kBands; ++b)
89 if (s.levelDb[b] > -150.0) { out[b] = s.levelDb[b] - mean; valid[b] = true; }
90 return out;
91}

94std::vector<NUIRect> sectionBlocks(const NUIRect& lane, const std::vector<Audio::Section>& sections, double seconds) {
95 std::vector<NUIRect> out;
96 if (seconds <= 0.0) return out;
97 for (const auto& s : sections) {
98 const float x0 = lane.x + lane.width * static_cast<float>(s.startSec / seconds);
99 const float x1 = lane.x + lane.width * static_cast<float>(std::min(s.endSec, seconds) / seconds);
100 out.push_back({x0 + 1.0f, lane.y, std::max(2.0f, x1 - x0 - 2.0f), lane.height});
101 }
102 return out;
103}

114void AuditionHub::renderItemTab(NUIRenderer& r, const AuditionHubLayout::ItemFrame& p, const AuditionCompare::Side* mix) {
115 const Ink c = ink();
116 m_tabContentH = 0.0f;
117 if (!mix) {
118 r.drawTextCentered(m_sideShared->working.load() > 0 ? "Measuring..." : "Could not read this file", p.panel, 12.0f, c.muted.withAlpha(0.7f));
119 return;
120 }
121 // A tab is as tall as it needs; it scrolls inside the body (a laptop screen leaves little room under the hero).
122 const NUIRect panel{p.panel.x, p.panel.y - m_tabScroll, p.panel.width, 4000.0f};
123 r.setClipRect(p.body);
124 float bottom = panel.y;
125 const AuditionCompare::Side* ref = m_result ? &m_result->ref : nullptr;
126 const double refGain = m_result ? m_result->referenceGainDb + m_nudgeDb : 0.0;
127 const std::string mixName = m_item.kind == AuditionLibrary::Item::Kind::Reference ? m_item.title : (m_item.badge.empty() ? "Mix" : "Mix " + m_item.badge);
128 const std::string refName = m_ref ? AuditionLibrary::displayName(*m_ref) : std::string("Reference");
129 const bool current = m_hasNow && m_now.path == m_item.path && m_engine;
130 const double position = current ? m_engine->getPositionSeconds() : -1.0;
131
132 switch (m_tab) {
133 case ItemTab::Tonal: {
134 // The curve: each track's third-octave balance, as a shape (the level difference is taken out).
135 heading(r, "TONAL BALANCE", panel.x, panel.y, c);
136 float lx = panel.x + r.measureText("TONAL BALANCE", 9.5f).width + 24.0f;
137 lx = legend(r, lx, panel.y - 2.0f, c.mix, mixName, true);
138 if (ref) legend(r, lx, panel.y - 2.0f, kRef, refName, true);
139 const NUIRect g{panel.x, panel.y + 26.0f, panel.width, 220.0f};
140 r.fillRect(g, c.panel);
141 std::vector<bool> mixValid, refValid;
142 const auto mixShape = spectrumShape(mix->spectrum, mixValid);
143 const auto refShape = ref ? spectrumShape(ref->spectrum, refValid) : std::vector<double>{};
144 double maxAbs = 6.0;
145 for (size_t b = 0; b < mixShape.size(); ++b) if (mixValid[b]) maxAbs = std::max(maxAbs, std::fabs(mixShape[b]));
146 for (size_t b = 0; b < refShape.size(); ++b) if (refValid[b]) maxAbs = std::max(maxAbs, std::fabs(refShape[b]));
147 const double range = std::min(36.0, std::ceil(maxAbs / 6.0) * 6.0);
148 const auto& centers = Audio::Spectrum::centersHz();
149 const double lo = std::log10(20.0), hi = std::log10(16000.0);
150 const auto xAt = [&](double hz) { return g.x + g.width * static_cast<float>((std::log10(hz) - lo) / (hi - lo)); };
151 const auto yAt = [&](double db) { return g.y + g.height * 0.5f - static_cast<float>(db / range) * (g.height * 0.5f - 8.0f); };
152 r.fillRect({g.x, yAt(0.0), g.width, 1.0f}, c.rule);
153 for (double hz : {50.0, 100.0, 250.0, 500.0, 1000.0, 2500.0, 5000.0, 10000.0}) {
154 const float x = xAt(hz);
155 r.fillRect({x, g.y, 1.0f, g.height}, c.rule.withAlpha(0.5f));
156 const std::string label = hz >= 1000.0 ? fmt(hz == 2500.0 ? "%.1fk" : "%.0fk", hz / 1000.0) : fmt("%.0f", hz);
157 r.drawText(label, {x - r.measureText(label, 9.5f).width * 0.5f, g.bottom() + 6.0f}, 9.5f, c.muted.withAlpha(0.7f));
158 }
159 r.drawText(fmt("+%.0f dB", range), {g.x + 6.0f, g.y + 4.0f}, 9.5f, c.muted.withAlpha(0.6f));
160 r.drawText(fmt("-%.0f dB", range), {g.x + 6.0f, g.bottom() - 16.0f}, 9.5f, c.muted.withAlpha(0.6f));
161 const auto points = [&](const std::vector<double>& shape) {
162 std::vector<NUIPoint> pts;
163 for (size_t b = 0; b < shape.size(); ++b) pts.push_back({xAt(centers[b]), yAt(std::clamp(shape[b], -range, range))});
164 return pts;
165 };
166 if (ref) plot(r, points(refShape), refValid, 1.6f, kRef);
167 plot(r, points(mixShape), mixValid, 2.2f, c.mix);
168
169 // The five bands: mix minus reference, or (alone) the mix's share of the energy.
170 const float by = g.bottom() + 40.0f;
171 heading(r, ref ? "MIX AGAINST THE REFERENCE, PER BAND" : "SHARE OF THE ENERGY, PER BAND", panel.x, by, c);
172 const float colW = panel.width / 5.0f, barH = 44.0f, mid = by + 24.0f + barH;
173 std::vector<std::string> notes;
174 for (size_t b = 0; b < 5; ++b) {
175 const float x = panel.x + colW * static_cast<float>(b), cx = x + colW * 0.5f;
176 if (ref) {
177 const double d = m_result->tonalDeltaDb[b];
178 const float h = std::min(barH, static_cast<float>(std::fabs(d) / 6.0) * barH);
179 const bool big = std::fabs(d) >= 1.5;
180 r.fillRect({cx - 14.0f, mid, 28.0f, 1.0f}, c.rule);
181 r.fillRect({cx - 10.0f, d > 0 ? mid - h : mid, 20.0f, std::max(1.0f, h)}, big ? kRef : c.muted.withAlpha(0.5f));
182 const std::string v = fmt("%+.1f", d);
183 r.drawText(v, {cx - r.measureText(v, 12.0f).width * 0.5f, mid + barH + 8.0f}, 12.0f, big ? kRef : c.text);
184 if (big) notes.push_back(std::string(d > 0 ? "more " : "less ") + kBandNames[b] + fmt(" (%+.1f dB)", d));
185 } else {
186 const double share = mix->analysis.bandShare[b];
187 const float h = static_cast<float>(share) * barH * 2.0f;
188 r.fillRect({cx - 10.0f, mid + barH - h, 20.0f, std::max(1.0f, h)}, c.mixFill);
189 const std::string v = fmt("%.0f%%", share * 100.0);
190 r.drawText(v, {cx - r.measureText(v, 12.0f).width * 0.5f, mid + barH + 8.0f}, 12.0f, c.text);
191 }
192 r.drawText(kBandNames[b], {cx - r.measureText(kBandNames[b], 11.0f).width * 0.5f, mid + barH + 28.0f}, 11.0f, c.muted);
193 r.drawText(kBandRanges[b], {cx - r.measureText(kBandRanges[b], 9.5f).width * 0.5f, mid + barH + 44.0f}, 9.5f, c.muted.withAlpha(0.7f));
194 }
195 std::string summary;
196 if (!ref) summary = "Pick a reference above to see where your mix has more or less than it.";
197 else if (notes.empty()) summary = "Within 1.5 dB of the reference in every band.";
198 else {
199 summary = "Your mix has ";
200 for (size_t i = 0; i < notes.size(); ++i) summary += (i ? (i + 1 == notes.size() ? " and " : ", ") : "") + notes[i];
201 summary += " than the reference.";
202 }
203 r.drawText(summary, {panel.x, mid + barH + 72.0f}, 11.5f, c.muted);
204 bottom = mid + barH + 90.0f;
205 break;
206 }
207
208 case ItemTab::Dynamics: {
209 // Short-term loudness over time: the reference as it is heard, after the level match.
210 heading(r, "SHORT-TERM LOUDNESS", panel.x, panel.y, c);
211 float lx = panel.x + r.measureText("SHORT-TERM LOUDNESS", 9.5f).width + 24.0f;
212 lx = legend(r, lx, panel.y - 2.0f, c.mix, mixName, true);
213 if (ref) legend(r, lx, panel.y - 2.0f, kRef, refName + fmt(", level-matched %+.1f dB", refGain), true);
214 const NUIRect g{panel.x, panel.y + 26.0f, panel.width, 200.0f};
215 r.fillRect(g, c.panel);
216 const auto& a = mix->analysis.shortTermLufs;
217 const std::vector<float> none;
218 const auto& b = ref ? ref->analysis.shortTermLufs : none;
219 double top = -200.0, low = 0.0;
220 const auto scan = [&](const std::vector<float>& v, double gain) {
221 for (float x : v) if (x > -70.0f) { top = std::max(top, x + gain); low = std::min(low, x + gain); }
222 };
223 scan(a, 0.0);
224 scan(b, refGain);
225 if (top < -150.0) { r.drawTextCentered("Too quiet or too short to measure", g, 12.0f, c.muted); break; }
226 top = std::ceil((top + 1.0) / 3.0) * 3.0;
227 low = std::max(top - 36.0, std::floor((low - 1.0) / 3.0) * 3.0);
228 const double span = std::max(mix->seconds, ref ? ref->seconds : 0.0);
229 const auto xAt = [&](double sec) { return g.x + g.width * static_cast<float>(std::clamp(sec / std::max(0.001, span), 0.0, 1.0)); };
230 const auto yAt = [&](double lufs) { return g.bottom() - 6.0f - (g.height - 12.0f) * static_cast<float>((std::clamp(lufs, low, top) - low) / (top - low)); };
231 for (double v = top; v >= low; v -= 6.0) {
232 r.fillRect({g.x, yAt(v), g.width, 1.0f}, c.rule.withAlpha(0.5f));
233 r.drawText(fmt("%.0f", v), {g.x + 6.0f, yAt(v) + 2.0f}, 9.0f, c.muted.withAlpha(0.6f));
234 }
235 const auto series = [&](const std::vector<float>& v, double gain, std::vector<bool>& ok) {
236 std::vector<NUIPoint> pts;
237 ok.clear();
238 for (size_t i = 0; i < v.size(); ++i) { pts.push_back({xAt(0.1 * static_cast<double>(i)), yAt(v[i] + gain)}); ok.push_back(v[i] > -70.0f); }
239 return pts;
240 };
241 std::vector<bool> okA, okB;
242 if (ref) plot(r, series(b, refGain, okB), okB, 1.4f, kRef);
243 plot(r, series(a, 0.0, okA), okA, 2.0f, c.mix);
244 if (position >= 0.0) r.fillRect({xAt(position), g.y, 1.5f, g.height}, NUIColor::white());
245
246 // The four numbers that describe the dynamics, mix against reference.
247 const float cy = g.bottom() + 24.0f, cw = (panel.width - 36.0f) / 4.0f;
248 struct Card { const char* name; double a, b; const char* unit; };
249 const auto& A = mix->analysis;
250 const Card cards[] = {
251 {"Integrated", A.integratedLufs, ref ? ref->analysis.integratedLufs : 0.0, "LUFS"},
252 {"Loudness range", A.loudnessRangeLu, ref ? ref->analysis.loudnessRangeLu : 0.0, "LU"},
253 {"PLR", Audio::peakToLoudnessRatioDb(A), ref ? Audio::peakToLoudnessRatioDb(ref->analysis) : 0.0, "dB"},
254 {"True peak", A.truePeakDbtp, ref ? ref->analysis.truePeakDbtp : 0.0, "dBTP"}};
255 for (int i = 0; i < 4; ++i) {
256 const NUIRect card{panel.x + static_cast<float>(i) * (cw + 12.0f), cy, cw, 70.0f};
257 r.fillRect(card, c.panel);
258 r.drawText(cards[i].name, {card.x + 12.0f, card.y + 10.0f}, 10.0f, c.muted);
259 const std::string av = cards[i].a > -150.0 ? fmt("%.1f", cards[i].a) : std::string("-");
260 r.drawText(av, {card.x + 12.0f, card.y + 30.0f}, 22.0f, c.mix);
261 float x = card.x + 12.0f + r.measureText(av, 22.0f).width + 8.0f;
262 if (ref) {
263 r.drawText("vs", {x, card.y + 40.0f}, 10.0f, c.muted);
264 x += r.measureText("vs", 10.0f).width + 8.0f;
265 const std::string bv = cards[i].b > -150.0 ? fmt("%.1f", cards[i].b) : std::string("-");
266 r.drawText(bv, {x, card.y + 30.0f}, 22.0f, kRef);
267 x += r.measureText(bv, 22.0f).width + 6.0f;
268 }
269 r.drawText(cards[i].unit, {x, card.y + 40.0f}, 10.0f, c.muted.withAlpha(0.8f));
270 }
271
272 // Each section's level and crest (peak over RMS: how much the peaks stand out), mix and reference side by side.
273 const float ty = cy + 96.0f;
274 bottom = ty;
275 {
276 const float cols[] = {panel.x, panel.x + panel.width * 0.22f, panel.x + panel.width * 0.40f, panel.x + panel.width * 0.62f, panel.x + panel.width * 0.80f};
277 const char* heads[] = {"SECTION", "MIX RMS", "MIX CREST", "REF RMS", "REF CREST"};
278 for (int i = 0; i < (ref ? 5 : 3); ++i) r.drawText(heads[i], {cols[i], ty}, 9.5f, i >= 3 ? kRef.withAlpha(0.8f) : c.muted.withAlpha(0.8f));
279 const size_t rows = std::max(mix->sections.size(), ref ? ref->sections.size() : size_t{0});
280 float y = ty + 20.0f;
281 for (size_t i = 0; i < rows; ++i, y += 24.0f) {
282 r.fillRect({panel.x, y - 4.0f, panel.width, 1.0f}, c.rule);
283 r.drawText(std::to_string(i + 1), {cols[0], y + 2.0f}, 11.5f, c.muted);
284 if (i < mix->sections.size()) {
285 const auto& s = mix->sections[i];
286 r.drawText(mmss(s.startSec), {cols[0] + 24.0f, y + 2.0f}, 11.5f, c.muted.withAlpha(0.8f));
287 r.drawText(levelDb(s.rmsDb), {cols[1], y + 2.0f}, 11.5f, c.mix);
288 r.drawText(s.rmsDb > -90.0 ? fmt("%.1f dB", s.crestDb) : std::string("-"), {cols[2], y + 2.0f}, 11.5f, c.mix);
289 }
290 if (ref && i < ref->sections.size()) {
291 const auto& s = ref->sections[i];
292 r.drawText(levelDb(s.rmsDb > -90.0 ? s.rmsDb + refGain : s.rmsDb), {cols[3], y + 2.0f}, 11.5f, kRef);
293 r.drawText(s.rmsDb > -90.0 ? fmt("%.1f dB", s.crestDb) : std::string("-"), {cols[4], y + 2.0f}, 11.5f, kRef);
294 }
295 }
296 bottom = y + 8.0f;
297 }
298 break;
299 }
300
301 case ItemTab::Stereo: {
302 if (mix->stereo.correlationOverTime.empty()) {
303 r.drawTextCentered("This file is mono: it has no stereo image to measure.", p.panel, 12.0f, c.muted);
304 break;
305 }
306 // Width per band: side energy as a share of the whole, drawn out from the centre.
307 heading(r, "WIDTH BY BAND", panel.x, panel.y, c);
308 float lx = panel.x + r.measureText("WIDTH BY BAND", 9.5f).width + 24.0f;
309 lx = legend(r, lx, panel.y - 2.0f, c.mix, mixName, false);
310 if (ref) legend(r, lx, panel.y - 2.0f, kRef, refName, false);
311 const float labelW = 110.0f, valueW = 96.0f;
312 const NUIRect track{panel.x + labelW, panel.y + 28.0f, panel.width - labelW - valueW, 5.0f * 30.0f};
313 const float cx = track.x + track.width * 0.5f;
314 r.fillRect({cx, track.y - 4.0f, 1.0f, track.height + 4.0f}, c.rule);
315 for (size_t b = 0; b < 5; ++b) {
316 const float y = track.y + 30.0f * static_cast<float>(b);
317 r.drawText(kBandNames[b], {panel.x, y + 4.0f}, 11.5f, c.muted);
318 r.fillRect({track.x, y + 12.0f, track.width, 1.0f}, c.rule.withAlpha(0.5f));
319 const float wm = static_cast<float>(mix->stereo.bandWidth[b]) * track.width * 0.5f;
320 r.fillRect({cx - wm, y + 6.0f, wm * 2.0f, 6.0f}, c.mixFill);
321 std::string v = fmt("%.2f", mix->stereo.bandWidth[b]);
322 if (ref && !ref->stereo.correlationOverTime.empty()) {
323 const float wr = static_cast<float>(ref->stereo.bandWidth[b]) * track.width * 0.5f;
324 r.fillRect({cx - wr, y + 15.0f, wr * 2.0f, 3.0f}, kRef);
325 v += " / " + fmt("%.2f", ref->stereo.bandWidth[b]);
326 }
327 r.drawText(v, {track.right() + 16.0f, y + 4.0f}, 11.0f, c.text);
328 }
329 r.drawText("0 = mono, 1 = all side. Low end usually sits near the centre.", {panel.x, track.bottom() + 4.0f}, 10.0f, c.muted.withAlpha(0.75f));
330
331 // Correlation over time: below zero the two sides cancel when they are summed to mono.
332 const float gy = track.bottom() + 44.0f;
333 heading(r, "CORRELATION OVER TIME", panel.x, gy, c);
334 const NUIRect g{panel.x, gy + 22.0f, panel.width, 120.0f};
335 r.fillRect(g, c.panel);
336 const double span = std::max(mix->seconds, ref ? ref->seconds : 0.0);
337 const auto xAt = [&](double sec) { return g.x + g.width * static_cast<float>(std::clamp(sec / std::max(0.001, span), 0.0, 1.0)); };
338 const auto yAt = [&](double v) { return g.y + g.height * 0.5f - static_cast<float>(std::clamp(v, -1.0, 1.0)) * (g.height * 0.5f - 6.0f); };
339 r.fillRect({g.x, yAt(0.0), g.width, 1.0f}, c.rule);
340 r.drawText("+1", {g.x + 4.0f, g.y + 2.0f}, 9.0f, c.muted.withAlpha(0.6f));
341 r.drawText("-1", {g.x + 4.0f, g.bottom() - 14.0f}, 9.0f, c.muted.withAlpha(0.6f));
342 const double win = Audio::StereoDetail::kWindowSeconds;
343 for (double start : mix->stereo.monoProblemStartsSec)
344 r.fillRect({xAt(start), g.y, std::max(2.0f, xAt(start + win) - xAt(start)), g.height}, kWarn.withAlpha(0.25f));
345 const auto series = [&](const std::vector<float>& v) {
346 std::vector<NUIPoint> pts;
347 for (size_t i = 0; i < v.size(); ++i) pts.push_back({xAt((static_cast<double>(i) + 0.5) * win), yAt(v[i])});
348 return pts;
349 };
350 if (ref && !ref->stereo.correlationOverTime.empty()) {
351 const auto pts = series(ref->stereo.correlationOverTime);
352 plot(r, pts, std::vector<bool>(pts.size(), true), 1.2f, kRef.withAlpha(0.8f));
353 }

416 if (m_tab != ItemTab::Sections || !(event.pressed && event.button == NUIMouseButton::Left) || !m_engine) return false;

423 for (size_t i = 0; i < blocks.size(); ++i) {
424 if (!blocks[i].contains(event.position)) continue;

431 }
432 return false;
433 };
