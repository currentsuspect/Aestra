4// V82-H3 / H8: the sandbox helper's plugin window (Linux, X11; Windows, a Win32 window; VST3 and CLAP). It opens the X11 display, makes the window,
5// attaches the plugin's view and then runs the X connection plus whatever descriptors and timers the plugin registered
6// through its run loop.
7// The app drives it with EDITOR OPEN / CLOSE / STATE; closing the window with the title-bar button is noticed on STATE.
8//
9// Everything here runs on the helper's command thread, between commands (see HelperLineReader.h): no thread of its own.

59// Xlib's default error handler exits the process. A plugin's editor makes plenty of requests we do not control (a bad
60// atom, a window it already destroyed), and none of them is worth losing the user's session over: log it and go on.
61inline int ignoreXError(Display*, XErrorEvent* e) {
62 static std::atomic<int> logged{0};
63 if (logged++ < 5) // a plugin can repeat the same one every frame
64 std::fprintf(stderr, "AestraPluginHost: X error %d (request %d) ignored\n", e->error_code, e->request_code);
65 return 0;
66}

76 bool open(EditorBackend backend, const std::string& title, int& width, int& height) {
77 close();
78 if (!backend.open) return false;
79 m_display = XOpenDisplay(nullptr);
80 if (!m_display) return false;
81 XSetErrorHandler(ignoreXError);
82 m_screen = DefaultScreen(m_display);
83 m_wmDelete = XInternAtom(m_display, "WM_DELETE_WINDOW", False);
84 m_backend = std::move(backend);
85
86 auto resizeWindow = [this](int w, int h) {
87 if (!m_window) return;
88 XResizeWindow(m_display, m_window, static_cast<unsigned>(w), static_cast<unsigned>(h));
89 XFlush(m_display);
90 m_width = w;
91 m_height = h;
92 };
93 auto makeWindow = [this, &title](int w, int h, bool resizable) -> std::uintptr_t {
94 m_window = XCreateSimpleWindow(m_display, RootWindow(m_display, m_screen), 0, 0, static_cast<unsigned>(w),
95 static_cast<unsigned>(h), 0, 0, BlackPixel(m_display, m_screen));
96 if (!m_window) return 0;
97 XStoreName(m_display, m_window, title.c_str());
98 XSetWMProtocols(m_display, m_window, &m_wmDelete, 1);
99 XSelectInput(m_display, m_window, StructureNotifyMask | KeyPressMask | KeyReleaseMask | FocusChangeMask);
100 if (!resizable) {
101 XSizeHints hints{};
102 hints.flags = PMinSize | PMaxSize;
103 hints.min_width = hints.max_width = w;
104 hints.min_height = hints.max_height = h;
105 XSetWMNormalHints(m_display, m_window, &hints);
106 }
107 XMapWindow(m_display, m_window);
108 XFlush(m_display);
109 m_width = w;
110 m_height = h;
111 return static_cast<std::uintptr_t>(m_window);
112 };
113
114 if (!m_backend.open(makeWindow, m_pump, resizeWindow)) {
115 teardown(false);
116 return false;
117 }
118 width = m_width;
119 height = m_height;
120 return true;
121 }
122
124 bool isOpen() const { return m_display != nullptr; }
125 int width() const { return m_width; }
126 int height() const { return m_height; }
127
128 void close() { teardown(true); }
129
132 void pollFds(std::vector<pollfd>& out, int& timeoutMs) {
133 if (!m_display) return;
134 out.push_back({ConnectionNumber(m_display), POLLIN, 0});
135 for (const int fd : m_pump.fds()) out.push_back({fd, POLLIN, 0});
136 int wait = m_pump.msUntilNextTimer();
137 if (wait < 0 || wait > 30) wait = 30; // also how soon the user closing the window is noticed
138 if (timeoutMs < 0 || wait < timeoutMs) timeoutMs = wait;
139 }
140
142 void service() {
143 if (!m_display) return;
144 while (m_display && XPending(m_display) > 0) {
145 XEvent event;
146 XNextEvent(m_display, &event);
147 if (event.type == ClientMessage && static_cast<Atom>(event.xclient.data.l[0]) == m_wmDelete) {
148 teardown(true); // the user closed the window
149 return;
150 }
151 if (event.type == ConfigureNotify && event.xconfigure.window == m_window && m_backend.canResize() &&
152 (event.xconfigure.width != m_width || event.xconfigure.height != m_height)) {
153 // The user dragged the window edge: tell the plugin its view has a new size.
154 m_width = event.xconfigure.width;
155 m_height = event.xconfigure.height;
156 m_backend.resize(m_width, m_height);
157 }
158 }
159 std::vector<pollfd> ready;
160 for (const int fd : m_pump.fds()) ready.push_back({fd, POLLIN, 0});
161 if (!ready.empty() && poll(ready.data(), static_cast<nfds_t>(ready.size()), 0) > 0)
162 for (const auto& p : ready)
163 if (p.revents & (POLLIN | POLLHUP)) m_pump.dispatchFd(p.fd);
164 m_pump.fireDueTimers();
165 if (m_backend.sync) m_backend.sync(); // what the processor reported, to the window
166 }
167
168private:
169 void teardown(bool closeBackend) {
170 if (!m_display) return;
171 if (closeBackend && m_backend.close) m_backend.close();
172 if (m_window) XDestroyWindow(m_display, m_window);
173 XCloseDisplay(m_display);
174 m_display = nullptr;
175 m_window = 0;
176 m_backend = {};
177 }
178
179 EditorBackend m_backend;
180 Aestra::Audio::EditorLoopPump m_pump;
181 Display* m_display = nullptr;
182 Window m_window = 0;
183 Atom m_wmDelete = 0;
184 int m_screen = 0;
185 int m_width = 0;
186 int m_height = 0;
187};

212 bool open(EditorBackend backend, const std::string& title, int& width, int& height) {
213 close();
214 if (!backend.open || !registerClass()) return false;
215 m_backend = std::move(backend);
216 m_closeRequested = false;

219 auto resizeWindow = [this](int w, int h) {
220 if (!m_window) return;
221 RECT r{0, 0, w, h};
222 AdjustWindowRect(&r, static_cast<DWORD>(GetWindowLongPtrW(m_window, GWL_STYLE)), FALSE);
223 m_ignoreSize = true; // our own resize, not the user's
224 SetWindowPos(m_window, nullptr, 0, 0, r.right - r.left, r.bottom - r.top,
225 SWP_NOMOVE | SWP_NOZORDER | SWP_NOACTIVATE);
226 m_ignoreSize = false;
227 m_width = w;
228 m_height = h;
229 };
230 auto makeWindow = [this, &title](int w, int h, bool resizable) -> std::uintptr_t {
231 const DWORD style = resizable ? WS_OVERLAPPEDWINDOW
232 : (WS_OVERLAPPEDWINDOW & ~(WS_THICKFRAME | WS_MAXIMIZEBOX));
233 RECT r{0, 0, w, h};
234 AdjustWindowRect(&r, style, FALSE);
235 const std::wstring wideTitle = widen(title);
236 m_window = CreateWindowExW(0, kClassName, wideTitle.c_str(), style, CW_USEDEFAULT, CW_USEDEFAULT,
237 r.right - r.left, r.bottom - r.top, nullptr, nullptr,
238 GetModuleHandleW(nullptr), nullptr);
239 if (!m_window) return 0;
240 SetWindowLongPtrW(m_window, GWLP_USERDATA, reinterpret_cast<LONG_PTR>(this));
241 ShowWindow(m_window, SW_SHOWNORMAL);
242 SetForegroundWindow(m_window); // the user asked for it: bring it up, not behind Aestra
243 m_width = w;
244 m_height = h;
245 return reinterpret_cast<std::uintptr_t>(m_window);
246 };
247
248 if (!m_backend.open(makeWindow, m_pump, resizeWindow)) {
249 teardown(false);
250 return false;
251 }
252 width = m_width;
253 height = m_height;
254 return true;
255 }

272 void service() {
273 if (!m_window) return;
274 if (m_closeRequested) {
275 teardown(true); // the user closed the window
276 return;
277 }
278 if (m_sizePending) {
279 m_sizePending = false;
280 if (m_backend.canResize && m_backend.canResize() && (m_pendingW != m_width || m_pendingH != m_height)) {
281 m_width = m_pendingW;
282 m_height = m_pendingH;
283 m_backend.resize(m_width, m_height);
284 }
285 }
286 m_pump.fireDueTimers();
287 if (m_backend.sync) m_backend.sync();
288 }

293 static std::wstring widen(const std::string& utf8) {
294 const int n = MultiByteToWideChar(CP_UTF8, 0, utf8.c_str(), -1, nullptr, 0);
295 if (n <= 0) return L"Plugin";
296 std::wstring out(static_cast<size_t>(n), L'\0');

302 static bool registerClass() {
303 static const bool registered = [] {
304 WNDCLASSEXW wc{};
305 wc.cbSize = sizeof wc;
306 wc.lpfnWndProc = &EditorHost::windowProc;
307 wc.hInstance = GetModuleHandleW(nullptr);
308 wc.hCursor = LoadCursor(nullptr, IDC_ARROW); // IDC_ARROW follows UNICODE; LoadCursor matches it

312 }();

316 static LRESULT CALLBACK windowProc(HWND hwnd, UINT msg, WPARAM wParam, LPARAM lParam) {
317 auto* self = reinterpret_cast<EditorHost*>(GetWindowLongPtrW(hwnd, GWLP_USERDATA));
318 if (self) {
319 if (msg == WM_CLOSE) {
320 self->m_closeRequested = true; // torn down in service(), after the plugin lets go of the window
321 return 0;
322 }
323 if (msg == WM_SIZE && wParam != SIZE_MINIMIZED && !self->m_ignoreSize) {
324 self->m_pendingW = LOWORD(lParam);
325 self->m_pendingH = HIWORD(lParam);
326 self->m_sizePending = true;
327 }
328 }
329 return DefWindowProcW(hwnd, msg, wParam, lParam);
330 }

336 DestroyWindow(m_window);

338 m_backend = {};
339 }
