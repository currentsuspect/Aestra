339 explicit EffectChainSnapshot(const std::array<EffectSlot, MAX_SLOTS>& slots) {
340 for (size_t i = 0; i < MAX_SLOTS; ++i) {
341 m_slots[i].plugin = slots[i].plugin;
342 m_slots[i].bypassed = slots[i].bypassed.load(std::memory_order_acquire);
343 m_slots[i].dryWetMix = slots[i].dryWetMix.load(std::memory_order_acquire);
344 m_slots[i].faultState = slots[i].faultState;
345 }
346 m_slotCount = MAX_SLOTS;
347 }
348
349 size_t slotCount() const noexcept { return m_slotCount; }
350
351 const EffectChainSnapshotSlot& slot(size_t index) const noexcept {
352 return m_slots[index];
353 }
354
355 const std::array<EffectChainSnapshotSlot, MAX_SLOTS>& slots() const noexcept {
356 return m_slots;
357 }
358
359 size_t getActiveSlotCount() const noexcept {
360 size_t count = 0;
361 for (size_t i = 0; i < MAX_SLOTS; ++i) {
362 if (!m_slots[i].isEmpty()) {
363 ++count;
364 }
365 }
366 return count;
367 }

385
386private:
387 std::array<EffectChainSnapshotSlot, MAX_SLOTS> m_slots{};
