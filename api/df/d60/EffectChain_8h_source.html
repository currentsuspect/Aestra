33struct EffectSlot {
34 PluginInstancePtr plugin;
35 std::atomic<bool> bypassed{false};
36 std::atomic<float> dryWetMix{1.0f};
37 std::shared_ptr<EffectSlotFaultState> faultState{std::make_shared<EffectSlotFaultState>()};

39 // Missing-plugin placeholder (#647).
40 //
41 // When loadState() cannot instantiate a plugin — it is not installed, the
42 // scan cache is stale, or its id changed — the slot retains the plugin's
43 // identity and its opaque state here instead of silently becoming empty.
44 // saveState() re-emits the retained record verbatim, so a load/save cycle on
45 // a machine missing the plugin does not destroy it.
46 //
47 // Non-RT only. process() never reads these; the RT predicate is isEmpty(),
48 // which stays `plugin == nullptr` precisely so a placeholder is skipped by
49 // the audio path without a null check.
50 std::string missingPluginId;
51 std::vector<uint8_t> missingPluginState;
52
53 // Last-good state cache for crash preservation (#931).
54 //
55 // saveState() on a dead helper returns empty, which would cement the crash
56 // into the project on the next save. So every successful non-empty
57 // saveState() refreshes this cache, and a crashed occupant is emitted as a
58 // #647-style record carrying the cached blob instead of the empty one.
59 //
60 // Keyed by instanceId: a replaced occupant mints a fresh id, so a stale
61 // cache can never attach to a new plugin. Non-RT only, like the
62 // missing-plugin record above. saveState() stays logically const — this is
63 // memoization, hence mutable.
64 mutable std::vector<uint8_t> lastGoodPluginState;
65 mutable uint64_t lastGoodInstanceId{0};
66
68 bool isEmpty() const { return plugin == nullptr; }
69
71 bool hasMissingPlugin() const { return plugin == nullptr && !missingPluginId.empty(); }
72
75 bool isOccupied() const { return plugin != nullptr || !missingPluginId.empty(); }
76
77 // Permanent identity for whatever occupies this slot (#667).
78 //
79 // The slot INDEX is a position; this is an identity. Automation must address
80 // the plugin instance a curve was drawn for, never the position that instance
81 // happened to occupy — reordering a chain is a rearrangement, never a
82 // retarget. movePlugin/swapPlugins carry this value with the plugin, so a
83 // reorder cannot silently re-point anything that refers to it.
84 //
85 // Zero means "nothing here". Minted by insertPlugin() and by loadState() for
86 // slots restored from a pre-v2 chain that predates identity; cleared by
87 // removePlugin(). A missing-plugin placeholder (#647) keeps its id, because
88 // the slot still stands for the same plugin — automation addressed to a
89 // plugin that failed to load must survive the round trip exactly as the
90 // placeholder itself does.
91 //
92 // Minted process-wide rather than per-chain: a plugin dragged from one
93 // channel strip to another keeps its identity, and per-chain counters would
94 // collide across channels.
95 uint64_t instanceId{0};
96
97 void clearMissingPlugin() {
98 missingPluginId.clear();
99 missingPluginState.clear();
100 missingPluginState.shrink_to_fit();
101 }
102};

211 // project of ten or fewer slots is still saved as v2 and stays loadable by
212 // older builds; an older build meeting a v3 payload refuses it cleanly (the
213 // unknown-version path) instead of misreading it.

555class EffectChainSnapshot {
556public:
557 EffectChainSnapshot() = default;
558
561 explicit EffectChainSnapshot(const EffectChain& chain) {
562 const size_t count = chain.slotCount();
563 m_slots.resize(count);
564 for (size_t i = 0; i < count; ++i) {
565 const EffectSlot* slot = chain.getSlot(i);
566 m_slots[i].plugin = slot->plugin;
567 m_slots[i].bypassed = slot->bypassed.load(std::memory_order_acquire);
568 m_slots[i].dryWetMix = slot->dryWetMix.load(std::memory_order_acquire);
569 m_slots[i].faultState = slot->faultState;
570 m_slots[i].instanceId = slot->instanceId;
571 }
572 }
573
581 size_t findSlotByInstanceId(uint64_t instanceId) const noexcept {
582 if (instanceId == 0) {
583 return EffectChain::kNoSlot;
584 }
585 for (size_t i = 0; i < m_slots.size(); ++i) {
586 if (m_slots[i].instanceId == instanceId) {
587 return i;
588 }
589 }
590 return EffectChain::kNoSlot;
591 }
592
593 size_t slotCount() const noexcept { return m_slots.size(); }
594
595 const EffectChainSnapshotSlot& slot(size_t index) const noexcept {
596 return m_slots[index];
597 }
598
599 const std::vector<EffectChainSnapshotSlot>& slots() const noexcept {
600 return m_slots;
601 }
602
603 size_t getActiveSlotCount() const noexcept {
604 size_t count = 0;
605 for (const auto& slot : m_slots) {
606 if (!slot.isEmpty()) {
607 ++count;
608 }
609 }
610 return count;
611 }
