32struct EffectSlot {
33 PluginInstancePtr plugin;
34 std::atomic<bool> bypassed{false};
35 std::atomic<float> dryWetMix{1.0f};
36 std::shared_ptr<EffectSlotFaultState> faultState{std::make_shared<EffectSlotFaultState>()};
37
38 // Missing-plugin placeholder (#647).
39 //
40 // When loadState() cannot instantiate a plugin — it is not installed, the
41 // scan cache is stale, or its id changed — the slot retains the plugin's
42 // identity and its opaque state here instead of silently becoming empty.
43 // saveState() re-emits the retained record verbatim, so a load/save cycle on
44 // a machine missing the plugin does not destroy it.
45 //
46 // Non-RT only. process() never reads these; the RT predicate is isEmpty(),
47 // which stays `plugin == nullptr` precisely so a placeholder is skipped by
48 // the audio path without a null check.
49 std::string missingPluginId;
50 std::vector<uint8_t> missingPluginState;
51
53 bool isEmpty() const { return plugin == nullptr; }
54
56 bool hasMissingPlugin() const { return plugin == nullptr && !missingPluginId.empty(); }
57
60 bool isOccupied() const { return plugin != nullptr || !missingPluginId.empty(); }
61
62 // Permanent identity for whatever occupies this slot (#667).
63 //
64 // The slot INDEX is a position; this is an identity. Automation must address
65 // the plugin instance a curve was drawn for, never the position that instance
66 // happened to occupy — reordering a chain is a rearrangement, never a
67 // retarget. movePlugin/swapPlugins carry this value with the plugin, so a
68 // reorder cannot silently re-point anything that refers to it.
69 //
70 // Zero means "nothing here". Minted by insertPlugin() and by loadState() for
71 // slots restored from a pre-v2 chain that predates identity; cleared by
72 // removePlugin(). A missing-plugin placeholder (#647) keeps its id, because
73 // the slot still stands for the same plugin — automation addressed to a
74 // plugin that failed to load must survive the round trip exactly as the
75 // placeholder itself does.
76 //
77 // Minted process-wide rather than per-chain: a plugin dragged from one
78 // channel strip to another keeps its identity, and per-chain counters would
79 // collide across channels.
80 uint64_t instanceId{0};
81
82 void clearMissingPlugin() {
83 missingPluginId.clear();
84 missingPluginState.clear();
85 missingPluginState.shrink_to_fit();
86 }
87};

485 m_slots[i].bypassed = slots[i].bypassed.load(std::memory_order_acquire);
486 m_slots[i].dryWetMix = slots[i].dryWetMix.load(std::memory_order_acquire);
487 m_slots[i].faultState = slots[i].faultState;
488 m_slots[i].instanceId = slots[i].instanceId;
489 }
490 m_slotCount = MAX_SLOTS;
491 }
492
500 size_t findSlotByInstanceId(uint64_t instanceId) const noexcept {
501 if (instanceId == 0) {
502 return MAX_SLOTS;
503 }
504 for (size_t i = 0; i < MAX_SLOTS; ++i) {
505 if (m_slots[i].instanceId == instanceId) {
506 return i;
507 }
508 }
509 return MAX_SLOTS;
510 }
511
512 size_t slotCount() const noexcept { return m_slotCount; }
513
514 const EffectChainSnapshotSlot& slot(size_t index) const noexcept {
515 return m_slots[index];
516 }
517

519 return m_slots;
520 }
521
522 size_t getActiveSlotCount() const noexcept {
523 size_t count = 0;
524 for (size_t i = 0; i < MAX_SLOTS; ++i) {
525 if (!m_slots[i].isEmpty()) {
