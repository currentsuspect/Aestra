36 Result readLine(std::string& out, long timeoutMs) {
37#ifdef _WIN32
38 (void)timeoutMs;
39 if (std::getline(std::cin, out)) return Result::Line;
40 return Result::Eof;
41#else
42 for (;;) {
43 const auto newline = m_pending.find('\n');
44 if (newline != std::string::npos) {
45 out = m_pending.substr(0, newline);
46 m_pending.erase(0, newline + 1);
47 if (!out.empty() && out.back() == '\r') out.pop_back();
48 return Result::Line;
49 }
50 pollfd descriptor{m_fd, POLLIN, 0};
51 const int ready = ::poll(&descriptor, 1, timeoutMs < 0 ? -1 : static_cast<int>(timeoutMs));
52 if (ready == 0) return Result::TimedOut;
53 if (ready < 0) return Result::Eof;
54 char chunk[512];
55 const auto got = ::read(m_fd, chunk, sizeof(chunk));
56 if (got <= 0) {
57 if (!m_pending.empty()) { // final line without a trailing newline
58 out = m_pending;
59 m_pending.clear();
60 return Result::Line;
61 }
62 return Result::Eof;
63 }
64 m_pending.append(chunk, static_cast<size_t>(got));
65 }
66#endif
67 }
68
69private:

71 std::string m_pending;
72};
