18std::vector<float> generateWaveformFromAudio(const std::vector<float>& samples,
19 uint32_t numChannels,
20 size_t targetSize = 256) {
21 std::vector<float> waveform(targetSize, 0.0f);
22 if (samples.empty() || numChannels == 0) return waveform;
23
24 size_t totalFrames = samples.size() / numChannels;
25 float framesPerBin = static_cast<float>(totalFrames) / targetSize;
26
27 for (size_t bin = 0; bin < targetSize; ++bin) {
28 size_t startFrame = static_cast<size_t>(bin * framesPerBin);
29 size_t endFrame = static_cast<size_t>((bin + 1) * framesPerBin);
30 endFrame = std::min(endFrame, totalFrames);
31
32 float maxAmp = 0.0f;
33 for (size_t frame = startFrame; frame < endFrame; ++frame) {
34 float sum = 0.0f;
35 for (uint32_t ch = 0; ch < numChannels; ++ch) {
36 sum += std::abs(samples[frame * numChannels + ch]);
37 }
38 maxAmp = std::max(maxAmp, sum / numChannels);
39 }
40 waveform[bin] = std::min(1.0f, maxAmp);
41 }
42
43 return waveform;
44}

46std::string truncateToWidth(NUIRenderer& renderer, const std::string& text, float fontSize, float maxWidth) {
47 if (maxWidth <= 0.0f) return "";
48 if (renderer.measureText(text, fontSize).width <= maxWidth) {
49 return text;
50 }
51
52 static const std::string ellipsis = "...";
53 if (renderer.measureText(ellipsis, fontSize).width > maxWidth) {
54 return "";
55 }
56
57 auto trimLastUtf8Codepoint = [](std::string& value) {
58 if (value.empty()) {
59 return;
60 }
61 size_t index = value.size();
62 do {
63 --index;
64 } while (index > 0 && (static_cast<unsigned char>(value[index]) & 0xC0u) == 0x80u);
65 value.erase(index);
66 };
67
68 std::string candidate = text;
69 while (!candidate.empty()) {
70 trimLastUtf8Codepoint(candidate);
71 const std::string attempt = candidate + ellipsis;
72 if (renderer.measureText(attempt, fontSize).width <= maxWidth) {
73 return attempt;
74 }
75 }
76 return "";
77}

106 {
107 std::lock_guard<std::mutex> lock(waveformMutex_);
108 waveformData_.clear();
109 }
110
111 currentGeneration_++;
112
113 pendingWaveformPath_.clear();
114 pendingWaveformFileSize_ = 0;
115 pendingWaveformDelay_ = 0.0;
116 waveformQueued_ = false;
117
118 isLoading_ = false;
119 isWaveformLoading_ = false;
120
121 if (hasCurrentFile_ && !currentFile_.isDirectory) {
122 std::string ext = std::filesystem::path(currentFile_.path).extension().string();
123 std::transform(ext.begin(), ext.end(), ext.begin(), ::tolower);
124
125 if (ext == ".wav" || ext == ".mp3" || ext == ".flac" || ext == ".ogg" ||
126 ext == ".aif" || ext == ".aiff" || ext == ".m4a" || ext == ".mp4") {
127 pendingWaveformPath_ = currentFile_.path;
128 pendingWaveformFileSize_ = currentFile_.size;
129 pendingWaveformDelay_ = 0.18;
130 waveformQueued_ = true;
131 }
132 }
133 setDirty(true);

180void FilePreviewPanel::onUpdate(double deltaTime) {
181 NUIComponent::onUpdate(deltaTime);
182
183 if (waveformJustCompleted_.exchange(false)) {
184 setDirty(true);
185 }
186
187 if (isLoading_ || isWaveformLoading_) {
188 loadingAnimationTime_ += static_cast<float>(deltaTime);
189 setDirty(true);
190 }
191
192 if (!waveformQueued_ || !isVisible() || isWaveformLoading_ || pendingWaveformPath_.empty()) {
193 return;
194 }
195
196 pendingWaveformDelay_ -= deltaTime;
197 if (pendingWaveformDelay_ <= 0.0) {
198 waveformQueued_ = false;
199 generateWaveform(pendingWaveformPath_, pendingWaveformFileSize_);
200 }

203void FilePreviewPanel::generateWaveform(const std::string& path, size_t fileSize) {
204 isWaveformLoading_ = true;
205 loadingAnimationTime_ = 0.0f;
206 uint64_t gen = currentGeneration_.load();
207 std::weak_ptr<NUIComponent> weakSelf = weak_from_this();
208
209 std::thread([weakSelf, path, gen]() {
210 auto self = std::dynamic_pointer_cast<FilePreviewPanel>(weakSelf.lock());
211 if (!self) return;
212
213 self->waveformWorker(path, gen);
214 }).detach();

217void FilePreviewPanel::waveformWorker(const std::string& path, uint64_t generation) {
218 // Every exit must clear isWaveformLoading_ — a stale-generation early
219 // return that leaves it set pins the spinner on forever and blocks
220 // onUpdate() from ever dequeuing the next waveform job.
221 const auto finishStale = [this]() {
222 std::lock_guard<std::mutex> lock(waveformMutex_);
223 isWaveformLoading_ = false;
224 waveformJustCompleted_.store(true);
225 };
226
227 if (generation != currentGeneration_.load(std::memory_order_acquire)) {
228 finishStale();
229 return;
230 }
231
232 std::vector<float> audioData;
233 uint32_t sampleRate = 0;
234 uint32_t numChannels = 0;
235
236 constexpr uint64_t kPreviewMaxFrames = 48000 * 24;
237 constexpr double kPreviewMaxSeconds = static_cast<double>(kPreviewMaxFrames) / 48000.0;
238 bool success = Aestra::Audio::decodeAudioPreview(path, audioData, sampleRate, numChannels, kPreviewMaxSeconds);
239
240 if (generation != currentGeneration_.load(std::memory_order_acquire)) {
241 finishStale();
242 return;
243 }
244
245 if (success && !audioData.empty()) {
246 std::vector<float> waveform = generateWaveformFromAudio(audioData, numChannels, 1024);
247
248 std::lock_guard<std::mutex> lock(waveformMutex_);
249 if (generation == currentGeneration_.load(std::memory_order_acquire)) {
250 waveformData_ = std::move(waveform);
251 isWaveformLoading_ = false;
252 waveformJustCompleted_.store(true);
253 }
254 } else {
255 std::lock_guard<std::mutex> lock(waveformMutex_);
256 if (generation == currentGeneration_.load(std::memory_order_acquire)) {
257 isWaveformLoading_ = false;
258 waveformJustCompleted_.store(true);
259 }
260 }

263void FilePreviewPanel::onRender(NUIRenderer& renderer) {
264 auto& theme = NUIThemeManager::getInstance();
265 NUIRect bounds = getBounds();
266 if (bounds.isEmpty()) return;
267
268 const bool hasSelection = hasCurrentFile_;
269 NUIColor bgColor = theme.getColor("backgroundPrimary").darkened(0.02f);
270 NUIColor borderColor = theme.getColor("borderSubtle").withAlpha(0.62f);
271 NUIColor accent = theme.getColor("accentPrimary");
272
273 // Background fill
274 renderer.fillRect(bounds, bgColor);
275 // Top separator line
276 renderer.drawLine(NUIPoint(bounds.x, bounds.y), NUIPoint(bounds.right(), bounds.y), 1.0f, borderColor);
277
278 // === EMPTY STATE ===
279 if (!hasSelection) {
280 float centerX = bounds.x + bounds.width * 0.5f;
281 float centerY = bounds.y + bounds.height * 0.5f;
282
283 float iconSize = 32.0f;
284 if (fileIcon_) {
285 fileIcon_->setBounds(NUIRect(centerX - iconSize * 0.5f, centerY - iconSize * 0.5f - 10.0f, iconSize, iconSize));
286 fileIcon_->setColor(theme.getColor("textSecondary").withAlpha(0.18f));
287 fileIcon_->onRender(renderer);
288 }
289
290 std::string emptyText = "Select a file to preview";
291 float fontSize = theme.getFontSize("m");
292 auto size = renderer.measureText(emptyText, fontSize);
293 renderer.drawText(emptyText,
294 NUIPoint(centerX - size.width * 0.5f, centerY + 18.0f),
295 fontSize, theme.getColor("textSecondary").withAlpha(0.42f));
296 return;
297 }
298
299 // === FOLDER STATE ===
300 if (currentFile_.isDirectory) {
301 float iconSize = 26.0f;
302 float padding = 14.0f;
303 float startX = bounds.x + 16.0f;
304 float centerY = bounds.y + bounds.height * 0.5f;
305
306 if (folderIcon_) {
307 NUIRect iconRect(startX, centerY - iconSize * 0.5f, iconSize, iconSize);
308 folderIcon_->setBounds(iconRect);
309 folderIcon_->setColor(accent.withAlpha(0.75f));
310 folderIcon_->onRender(renderer);
311 }
312
313 float textX = startX + iconSize + padding;
314 float textMaxWidth = bounds.width - (startX + iconSize + padding + 20.0f);
315 float totalTextHeight = 14.0f + 3.0f + 10.0f;
316 float textStartY = centerY - (totalTextHeight * 0.5f);
317
318 std::string name = currentFile_.name;
319 if (name.length() > 35) {
320 name = name.substr(0, 32) + "...";
321 }
322
323 renderer.drawText(name, NUIPoint(textX, textStartY + 2.0f), theme.getFontSize("l"), theme.getColor("textPrimary").withAlpha(0.88f));
324 renderer.drawText("Folder", NUIPoint(textX, textStartY + 14.0f + 5.0f), theme.getFontSize("xs"), theme.getColor("textSecondary").withAlpha(0.55f));
325 return;
326 }
327
328 // === AUDIO FILE STATE ===
329 const float padL = 10.0f;
330 const float padR = 10.0f;
331 const float scrubberH = 3.0f;
332 const float padB = 8.0f; // bottom padding below scrubber
333 const float playBtnSize = 26.0f;
334 const float playBtnX = bounds.x + padL;
335 const float centerY = bounds.y + bounds.height * 0.5f;
336
337 // Play button bounds
338 playButtonBounds_ = NUIRect(playBtnX, centerY - playBtnSize * 0.5f, playBtnSize, playBtnSize);
339
340 // The text and waveform share a lane to the right of the play button.
341 const float textX = playButtonBounds_.right() + 8.0f;
342 const float textMaxWidth = std::max(0.0f, bounds.right() - padR - textX);
343 scrubberBounds_ = NUIRect(textX, bounds.bottom() - padB - scrubberH, textMaxWidth, scrubberH);
344
345 float progress = 0.0f;
346 if (duration_ > 0.0) {
347 progress = std::clamp(static_cast<float>(playheadPosition_ / duration_), 0.0f, 1.0f);
348 }
349
350 // -- Quiet waveform lane. Purple is reserved for played progress. --
351 {
352 std::lock_guard<std::mutex> lock(waveformMutex_);
353 if (!waveformData_.empty() && textMaxWidth > 0.0f) {
354 const NUIRect waveformRect(textX, bounds.y + 31.0f, textMaxWidth, 22.0f);
355 const NUIColor waveformFill = theme.getColor("textSecondary").withAlpha(0.10f);
356 const NUIColor waveformPlayed = accent.withAlpha(0.48f);
357 const float playedX = waveformRect.x + waveformRect.width * progress;
358 const float wfY = waveformRect.y + waveformRect.height * 0.5f;
359 const float maxAmp = waveformRect.height * 0.42f;
360 const float samplesPerPixel = static_cast<float>(waveformData_.size()) / waveformRect.width;
361
362 if (samplesPerPixel > 0.0f) {
363 for (float x = 0; x < waveformRect.width; x += 2.0f) {
364 int startSample = static_cast<int>(x * samplesPerPixel);
365 int endSample = static_cast<int>((x + 2.0f) * samplesPerPixel);
366 startSample = std::clamp(startSample, 0, (int)waveformData_.size() - 1);
367 endSample = std::clamp(endSample, startSample + 1, (int)waveformData_.size());
368
369 float amplitude = 0.0f;
370 for (int i = startSample; i < endSample; ++i) {
371 amplitude = std::max(amplitude, waveformData_[i]);
372 }
373
374 float barHeight = std::max(1.0f, amplitude * maxAmp * 2.0f);
375 float yStart = wfY - barHeight * 0.5f;
376 // Keyed off the playhead, not isPlaying_, so pausing keeps the
377 // accent in step with the scrubber fill below.
378 const bool played = duration_ > 0.0 && (waveformRect.x + x) < playedX;
379 renderer.drawLine(
380 NUIPoint(waveformRect.x + x, yStart),
381 NUIPoint(waveformRect.x + x, yStart + barHeight),
382 1.0f, played ? waveformPlayed : waveformFill
383 );
384 }
385 }
386 }
387 }
388
389 // -- Play button --
390 NUIColor btnBg = isPlaying_ ? accent.withAlpha(0.22f) : NUIColor::transparent();
391 NUIColor btnBorder = isPlaying_
392 ? accent.withAlpha(0.20f)
393 : theme.getColor("border").withAlpha(0.35f);
394 if (!isPlaying_) {
395 // Subtle hover ring could be added here if hover state tracked
396 btnBorder = theme.getColor("border").withAlpha(0.45f);
397 }
398
399 renderer.fillRoundedRect(playButtonBounds_, playBtnSize * 0.5f, btnBg);
400 renderer.strokeRoundedRect(playButtonBounds_, playBtnSize * 0.5f, 1.0f, btnBorder);
401
402 auto& icon = isPlaying_ ? stopIcon_ : playIcon_;
403 if (icon) {
404 float iconSize = 13.0f;
405 float iconX = std::floor(playButtonBounds_.x + (playButtonBounds_.width - iconSize) * 0.5f);
406 float iconY = std::floor(playButtonBounds_.y + (playButtonBounds_.height - iconSize) * 0.5f);
407
408 icon->setBounds(NUIRect(iconX, iconY, iconSize, iconSize));
409 icon->setColor(isPlaying_ ? NUIColor::white() : theme.getColor("textPrimary").withAlpha(0.85f));
410 icon->onRender(renderer);
411 }
412
413 // -- Loading spinner geometry. Declared up here so the filename lane can
414 // reserve the space instead of running underneath the spinner. --
415 const bool loading = isLoading_ || isWaveformLoading_;
416 const float spinnerR = 6.0f;
417 const float spinnerCX = bounds.right() - 28.0f;
418 const float spinnerLeft = spinnerCX - (spinnerR + 2.0f);
419
420 // -- File info. The row already establishes that this is audio, so the
421 // redundant glyph gives its width back to the filename. --
422 const float infoTopY = bounds.y + 8.0f;
423 const float nameX = bounds.x + padL;
424 const float nameRight = loading ? spinnerLeft - 6.0f : bounds.right() - padR;
425 const float nameMaxW = std::max(0.0f, nameRight - nameX);
426 const float nameFont = theme.getFontSize("m");
427 std::string displayName = truncateToWidth(renderer, currentFile_.name, nameFont, nameMaxW);
428 renderer.drawText(displayName, NUIPoint(nameX, infoTopY), nameFont, theme.getColor("textPrimary").withAlpha(0.92f));
429
430 // -- Scrubber / Progress bar --
431 float trackY = scrubberBounds_.y + scrubberBounds_.height * 0.5f;
432
433 // Track background
434 renderer.fillRoundedRect(scrubberBounds_, scrubberH * 0.5f, theme.getColor("border").withAlpha(0.22f));
435
436 // Progress fill
437 float fillW = scrubberBounds_.width * progress;
438 if (fillW > 0.5f) {
439 NUIRect fillRect(scrubberBounds_.x, scrubberBounds_.y, fillW, scrubberBounds_.height);
440 renderer.fillRoundedRect(fillRect, scrubberH * 0.5f, accent.withAlpha(0.78f));
441 }
442
443 // Playhead indicator
444 if (duration_ > 0.0) {
445 float playheadX = scrubberBounds_.x + progress * scrubberBounds_.width;
446 float indicatorH = 7.0f;
447 renderer.drawLine(
448 NUIPoint(playheadX, trackY - indicatorH * 0.5f),
449 NUIPoint(playheadX, trackY + indicatorH * 0.5f),
450 2.0f, accent.withAlpha(0.92f)
451 );
452 }
453
454 // -- Loading spinner overlay (geometry hoisted above the filename lane) --
455 if (loading) {
456 float spinnerCY = infoTopY + 6.0f;
457 int segments = 8;
458 float angleOffset = loadingAnimationTime_ * 5.0f;
459 for (int i = 0; i < segments; ++i) {
460 float a = angleOffset + (i * 2.0f * 3.14159265f / segments);
461 float alpha = (1.0f - static_cast<float>(i) / segments) * 0.8f;
462 float x1 = spinnerCX + std::cos(a) * (spinnerR - 2.0f);
463 float y1 = spinnerCY + std::sin(a) * (spinnerR - 2.0f);
464 float x2 = spinnerCX + std::cos(a) * (spinnerR + 2.0f);
465 float y2 = spinnerCY + std::sin(a) * (spinnerR + 2.0f);
466 renderer.drawLine(NUIPoint(x1, y1), NUIPoint(x2, y2), 1.8f, accent.withAlpha(alpha));
467 }
468 }

471bool FilePreviewPanel::onMouseEvent(const NUIMouseEvent& event) {
472 const auto bounds = getBounds();
473 if (!bounds.contains(event.position)) return false;
474
475 if (!hasCurrentFile_ || currentFile_.isDirectory) {
476 if (event.pressed && event.button == NUIMouseButton::Right) {
477 return true;
478 }
479 return false;
480 }
481
482 auto seekFromPosition = [&](const NUIPoint& pos) {
483 if (duration_ <= 0.0 || scrubberBounds_.width <= 0.0f) return;
484 float relativeX = pos.x - scrubberBounds_.x;
485 float progress = std::clamp(relativeX / scrubberBounds_.width, 0.0f, 1.0f);
486 double seekTime = progress * duration_;
487 if (onSeek_) onSeek_(seekTime);
488 };
489
490 if (isSeekDragging_) {
491 if (event.released) {
492 seekFromPosition(event.position);
493 isSeekDragging_ = false;
494 setDirty(true);
495 return true;
496 }
497 if (event.type == NUIMouseEventType::Drag) {
498 seekFromPosition(event.position);
499 setDirty(true);
500 return true;
501 }
502 }
503
504 if (event.pressed && event.button == NUIMouseButton::Left) {
505 if (playButtonBounds_.contains(event.position)) {
506 if (isPlaying_) {
507 if (onStop_) onStop_();
508 } else {
509 if (onPlay_) onPlay_(currentFile_);
510 }
511 return true;
512 }
513
514 if (scrubberBounds_.contains(event.position) && duration_ > 0.0) {
515 isSeekDragging_ = true;
516 seekFromPosition(event.position);
517 setDirty(true);
518 return true;
519 }
520 }
521
522 if (event.pressed && event.button == NUIMouseButton::Right) {
523 return true;
524 }
525
526 return false;
527}
