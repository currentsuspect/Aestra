18std::vector<float> generateWaveformFromAudio(const std::vector<float>& samples,
19 uint32_t numChannels,
20 size_t targetSize = 256) {
21 std::vector<float> waveform(targetSize, 0.0f);
22 if (samples.empty() || numChannels == 0) return waveform;
23
24 size_t totalFrames = samples.size() / numChannels;
25 float framesPerBin = static_cast<float>(totalFrames) / targetSize;
26
27 for (size_t bin = 0; bin < targetSize; ++bin) {
28 size_t startFrame = static_cast<size_t>(bin * framesPerBin);
29 size_t endFrame = static_cast<size_t>((bin + 1) * framesPerBin);
30 endFrame = std::min(endFrame, totalFrames);
31
32 float maxAmp = 0.0f;
33 for (size_t frame = startFrame; frame < endFrame; ++frame) {
34 float sum = 0.0f;
35 for (uint32_t ch = 0; ch < numChannels; ++ch) {
36 sum += std::abs(samples[frame * numChannels + ch]);
37 }
38 maxAmp = std::max(maxAmp, sum / numChannels);
39 }
40 waveform[bin] = std::min(1.0f, maxAmp);
41 }
42
43 return waveform;
44}

46std::string truncateToWidth(NUIRenderer& renderer, const std::string& text, float fontSize, float maxWidth) {
47 if (maxWidth <= 0.0f) return "";
48 if (renderer.measureText(text, fontSize).width <= maxWidth) {
49 return text;
50 }
51
52 static const std::string ellipsis = "...";
53 if (renderer.measureText(ellipsis, fontSize).width > maxWidth) {
54 return "";
55 }
56
57 auto trimLastUtf8Codepoint = [](std::string& value) {
58 if (value.empty()) {
59 return;
60 }
61 size_t index = value.size();
62 do {
63 --index;
64 } while (index > 0 && (static_cast<unsigned char>(value[index]) & 0xC0u) == 0x80u);
65 value.erase(index);
66 };
67
68 std::string candidate = text;
69 while (!candidate.empty()) {
70 trimLastUtf8Codepoint(candidate);
71 const std::string attempt = candidate + ellipsis;
72 if (renderer.measureText(attempt, fontSize).width <= maxWidth) {
73 return attempt;
74 }
75 }
76 return "";
77}

79std::string formatTimeShort(double seconds) {
80 if (seconds <= 0.0) {
81 return "0:00";
82 }
83 const int total = static_cast<int>(std::round(seconds));
84 const int mins = total / 60;
85 const int secs = total % 60;
86 char buffer[16];
87 std::snprintf(buffer, sizeof(buffer), "%d:%02d", mins, secs);
88 return std::string(buffer);
89}

91std::string fileExtensionUpper(const std::string& path) {
92 std::string ext = std::filesystem::path(path).extension().string();
93 std::transform(ext.begin(), ext.end(), ext.begin(), ::toupper);
94 if (!ext.empty() && ext[0] == '.') ext = ext.substr(1);
95 return ext;
96}

118void FilePreviewPanel::setFile(const FileItem* file) {
119 hasCurrentFile_ = file != nullptr;
120 m_currentFilePath.clear();
121 m_currentFileBpm = 0;
122 if (file) {
123 currentFile_ = *file;
124 m_currentFilePath = file->path;
125 m_currentFileBpm = file->detectedBpm;
126 } else {
127 currentFile_ = FileItem();
128 }
129 {
130 std::lock_guard<std::mutex> lock(waveformMutex_);
131 waveformData_.clear();
132 }
133
134 currentGeneration_++;
135
136 pendingWaveformPath_.clear();
137 pendingWaveformFileSize_ = 0;
138 pendingWaveformDelay_ = 0.0;
139 waveformQueued_ = false;
140
141 isLoading_ = false;
142 isWaveformLoading_ = false;
143
144 if (hasCurrentFile_ && !currentFile_.isDirectory) {
145 std::string ext = std::filesystem::path(currentFile_.path).extension().string();
146 std::transform(ext.begin(), ext.end(), ext.begin(), ::tolower);
147
148 if (ext == ".wav" || ext == ".mp3" || ext == ".flac" || ext == ".ogg" ||
149 ext == ".aif" || ext == ".aiff" || ext == ".m4a" || ext == ".mp4") {
150 pendingWaveformPath_ = currentFile_.path;
151 pendingWaveformFileSize_ = currentFile_.size;
152 pendingWaveformDelay_ = 0.18;
153 waveformQueued_ = true;
154 }
155 }
156 setDirty(true);
157}

211void FilePreviewPanel::onUpdate(double deltaTime) {
212 NUIComponent::onUpdate(deltaTime);
213
214 if (waveformJustCompleted_.exchange(false)) {
215 setDirty(true);
216 }
217
218 if (isLoading_ || isWaveformLoading_) {
219 loadingAnimationTime_ += static_cast<float>(deltaTime);
220 setDirty(true);
221 }
222
223 if (!waveformQueued_ || !isVisible() || isWaveformLoading_ || pendingWaveformPath_.empty()) {
224 return;
225 }
226
227 pendingWaveformDelay_ -= deltaTime;
228 if (pendingWaveformDelay_ <= 0.0) {
229 waveformQueued_ = false;
230 generateWaveform(pendingWaveformPath_, pendingWaveformFileSize_);
231 }

234void FilePreviewPanel::generateWaveform(const std::string& path, size_t fileSize) {
235 isWaveformLoading_ = true;
236 loadingAnimationTime_ = 0.0f;
237 uint64_t gen = currentGeneration_.load();
238 std::weak_ptr<NUIComponent> weakSelf = weak_from_this();
239
240 std::thread([weakSelf, path, gen]() {
241 auto self = std::dynamic_pointer_cast<FilePreviewPanel>(weakSelf.lock());
242 if (!self) return;
243
244 self->waveformWorker(path, gen);
245 }).detach();
246}

248void FilePreviewPanel::waveformWorker(const std::string& path, uint64_t generation) {
249 if (generation != currentGeneration_.load(std::memory_order_acquire)) return;
250
251 std::vector<float> audioData;
252 uint32_t sampleRate = 0;
253 uint32_t numChannels = 0;
254
255 constexpr uint64_t kPreviewMaxFrames = 48000 * 24;
256 constexpr double kPreviewMaxSeconds = static_cast<double>(kPreviewMaxFrames) / 48000.0;
257 bool success = Aestra::Audio::decodeAudioPreview(path, audioData, sampleRate, numChannels, kPreviewMaxSeconds);
258
259 if (generation != currentGeneration_.load(std::memory_order_acquire)) return;
260
261 if (success && !audioData.empty()) {
262 std::vector<float> waveform = generateWaveformFromAudio(audioData, numChannels, 1024);
263
264 std::lock_guard<std::mutex> lock(waveformMutex_);
265 if (generation == currentGeneration_.load(std::memory_order_acquire)) {
266 waveformData_ = std::move(waveform);
267 isWaveformLoading_ = false;
268 waveformJustCompleted_.store(true);
269 }
270 } else {
271 std::lock_guard<std::mutex> lock(waveformMutex_);
272 if (generation == currentGeneration_.load(std::memory_order_acquire)) {
273 isWaveformLoading_ = false;
274 waveformJustCompleted_.store(true);
275 }
276 }

564 m_bpmSyncEnabled ? NUIColor(0.204f, 0.835f, 0.600f, 0.9f) : theme.getColor("textSecondary").withAlpha(0.35f));
565 }

567 // -- Loading spinner overlay (small, near play button) --
568 bool loading = isLoading_ || isWaveformLoading_;
569
570 if (loading) {

572 float spinnerCY = infoTopY + 6.0f;

574 int segments = 8;

580 float y1 = spinnerCY + std::sin(a) * (spinnerR - 2.0f);
581 float x2 = spinnerCX + std::cos(a) * (spinnerR + 2.0f);
582 float y2 = spinnerCY + std::sin(a) * (spinnerR + 2.0f);
583 renderer.drawLine(NUIPoint(x1, y1), NUIPoint(x2, y2), 1.8f, accent.withAlpha(alpha));
584 }

588bool FilePreviewPanel::onMouseEvent(const NUIMouseEvent& event) {
589 const auto bounds = getBounds();
590 if (!bounds.contains(event.position)) return false;
591
592 if (!hasCurrentFile_ || currentFile_.isDirectory) {
593 if (event.pressed && event.button == NUIMouseButton::Right) {
594 return true;
595 }
596 return false;
597 }
598
599 if (event.type == NUIMouseEventType::Move) {
600 if (stopButtonBounds_.contains(event.position)) {
601 showRemoteTooltip("Stop preview", event.position, this);
602 } else if (loopButtonBounds_.contains(event.position)) {
603 showRemoteTooltip(m_loopEnabled ? "Loop: On" : "Loop: Off", event.position, this);
604 } else if (bpmSyncButtonBounds_.contains(event.position)) {
605 showRemoteTooltip(m_bpmSyncEnabled ? "BPM Sync: On" : "BPM Sync: Off", event.position, this);
606 }
607 }
608
609 auto seekFromPosition = [&](const NUIPoint& pos) {
610 if (duration_ <= 0.0 || scrubberBounds_.width <= 0.0f) return;
611 float relativeX = pos.x - scrubberBounds_.x;
612 float progress = std::clamp(relativeX / scrubberBounds_.width, 0.0f, 1.0f);
613 double seekTime = progress * duration_;
614 if (onSeek_) onSeek_(seekTime);
615 };
616
617 if (isSeekDragging_) {
618 if (event.released) {
619 seekFromPosition(event.position);
620 isSeekDragging_ = false;
621 setDirty(true);
622 return true;
623 }
624 if (event.type == NUIMouseEventType::Drag) {
625 seekFromPosition(event.position);
626 setDirty(true);
627 return true;
628 }
629 }
630
631 if (event.pressed && event.button == NUIMouseButton::Left) {
632 if (playButtonBounds_.contains(event.position)) {
633 if (isPlaying_) {
634 if (onStop_) onStop_();
635 } else {
636 if (onPlay_) onPlay_(currentFile_);
637 }
638 return true;
639 }
640
641 if (scrubberBounds_.contains(event.position) && duration_ > 0.0) {
642 isSeekDragging_ = true;
643 seekFromPosition(event.position);
644 setDirty(true);
645 return true;
646 }
647
648 // Stop button
649 if (stopButtonBounds_.contains(event.position)) {
650 if (onStop_) onStop_();
651 playheadPosition_ = 0.0;
652 setDirty(true);
653 return true;
654 }
655
656 // Loop button
657 if (loopButtonBounds_.contains(event.position)) {
658 m_loopEnabled = !m_loopEnabled;
659 setDirty(true);
660 return true;
661 }
662
663 // BPM sync button
664 if (bpmSyncButtonBounds_.contains(event.position)) {
665 m_bpmSyncEnabled = !m_bpmSyncEnabled;
666 setDirty(true);
667 return true;
668 }
669 }
670
671 if (event.pressed && event.button == NUIMouseButton::Right) {
672 return true;
673 }
674
675 return false;
676}
