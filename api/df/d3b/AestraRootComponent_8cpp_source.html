8bool AestraRootComponent::onKeyEvent(const NUIKeyEvent& event) {
9 // 0. Ctrl+S — save project (highest priority, before any panel)
10 if (event.pressed && (event.modifiers & NUIModifiers::Ctrl) &&
11 event.keyCode == NUIKeyCode::S && m_rootSaveCallback) {
12 m_rootSaveCallback();
13 return true;
14 }
15
16 // 1. Global app-level shortcuts and key releases go first. Releases must
17 // reach musical typing even if focus moved after note-on.
18 const bool globalFirst = event.released || event.keyCode == NUIKeyCode::Space ||
19 (event.modifiers & NUIModifiers::Ctrl);
20 if (globalFirst && m_rootContent) {
21 if (m_rootContent->onKeyEvent(event)) {
22 return true;
23 }
24 }
25
26 // 2. Dispatch to focused component
27 if (auto focused = AestraUI::NUIComponent::getFocusedComponent()) {
28 if (focused->onKeyEvent(event)) {
29 return true;
30 }
31 }
32
33 // 3. Non-global fallback after the focused control declined the key.
34 if (!globalFirst && m_rootContent && m_rootContent->onKeyEvent(event)) {
35 return true;
36 }
37
38 // 4. Fallback: F12 HUD toggle
39 if (event.pressed) {
40 if (event.keyCode == NUIKeyCode::F12) {
41 if (m_rootUnifiedHUD) {
42 m_rootUnifiedHUD->setVisible(!m_rootUnifiedHUD->isVisible());
43 return true;
44 }
45 }
46 }
47
48 return false;
49}
