15void NotificationToast::onRender(NUIRenderer& renderer)
16{
17 if (text_.empty())
18 return;
19
20 // Fade out over the toast's final moments instead of vanishing.
21 constexpr double kFadeSeconds = 0.35;
22 const double remaining = duration_ - elapsed_;
23 const float alpha = remaining >= kFadeSeconds
24 ? 1.0f
25 : std::max(0.0f, static_cast<float>(remaining / kFadeSeconds));
26
27 auto& theme = NUIThemeManager::getInstance();
28 const NUIRect bounds = getBounds();
29 constexpr float kFontSize = 12.0f;
30 const NUISize textSize = renderer.measureText(text_, kFontSize);
31 const float pillW = std::min(bounds.width, textSize.width + 32.0f);
32 const float pillH = std::min(bounds.height, 28.0f);
33 const NUIRect pill(bounds.x + (bounds.width - pillW) * 0.5f,
34 bounds.y + (bounds.height - pillH) * 0.5f, pillW, pillH);
35 renderer.fillRoundedRect(pill, pillH * 0.5f, theme.getColor("surfaceTertiary").withAlpha(0.92f * alpha));
36 renderer.strokeRoundedRect(pill, pillH * 0.5f, 1.0f, theme.getColor("accentPrimary").withAlpha(0.55f * alpha));
37 renderer.drawTextCentered(text_, pill, kFontSize, theme.getColor("textPrimary").withAlpha(alpha));
38}
