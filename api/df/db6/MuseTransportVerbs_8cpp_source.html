2// transport.* host verbs: play, stop, pause, seek and loop the way the app's
3// own transport does them (V82-M1, FD-53).
4//
5// The legacy `play`/`stop` verbs start and stop the audio engine directly. In the
6// running app that is not what the Play button does: the button goes through
7// AestraContent's focus-aware transport, which also moves the playhead, honours
8// the count-in and picks the Arsenal or the Timeline. An agent driving the app
9// for a screen recording got meters but a frozen playhead. These verbs call the
10// same functions the transport bar and the ruler call, so the two cannot drift.

47HostVerbSpec transportSpec(const char* name, bool mutates, std::string description) {
48 HostVerbSpec spec;
49 spec.name = name;
50 spec.domain = HostVerbDomain::Transport;
51 spec.affinity = HostThreadAffinity::HostUiThread;
52 // Playhead, loop and play state are session navigation, not project edits:
53 // the UI does not put them on the undo stack either.
54 spec.mutates = mutates;
55 spec.description = std::move(description);
56 return spec;
57}

67// The two transports a play can mean. Audition has its own player and RoutingMap
68// has none, so neither is a valid target for "play from beat N".
69bool parsePlayTarget(const std::string& name, ViewFocus& out) {
70 if (name == "timeline") {
71 out = ViewFocus::Timeline;
72 return true;
73 }
74 if (name == "arsenal") {
75 out = ViewFocus::Arsenal;
76 return true;
77 }
78 return false;
79}

81double tempoBpm(Audio::TrackManager& tm) {
82 return std::max(1.0, tm.getTimelineClock().getCurrentTempo());
83}

85// The playhead in beats, read the way the focused surface reads it: the Arsenal
86// maps seconds straight through the tempo, the Timeline through the playlist's
87// tempo map.
88double positionBeats(::AestraContent& content, Audio::TrackManager& tm) {
89 const double seconds = std::max(0.0, tm.getPosition());
90 if (content.getViewFocus() == ViewFocus::Arsenal) {
91 return seconds * tempoBpm(tm) / 60.0;
92 }
93 return tm.getPlaylistModel().secondsToBeats(seconds);
94}

96// Move the playhead as the ruler click (Timeline) or the progress-header scrub
97// (Arsenal) does. Both end the same way: the UI cue AND the engine's sample
98// position move. A ruler drag reaches the engine through the app loop's
99// per-frame scrub sync while the button is held; a one-shot seek has no held
100// button, so it moves the engine itself — without that, a seek while playing
101// is overwritten by the engine position on the next frame.
102// Beats past this are not a session position (about 2.5 hours at 400 BPM);
103// refusing them also keeps the sample-position cast below well defined.
104constexpr double kMaxSeekBeat = 60000.0;
105
106// Why a seek to @p beat cannot happen in the current focus, or empty when it can.
107std::string seekRefusal(::AestraContent& content, double beat) {
108 const ViewFocus focus = content.getViewFocus();
109 if (focus != ViewFocus::Timeline && focus != ViewFocus::Arsenal) {
110 return std::string("the ") + WorkspaceFocusModel::workspaceFocusName(focus) +
111 " workspace has no beat transport; use transport.play {in: timeline|arsenal}";
112 }
113 if (!std::isfinite(beat) || beat < 0.0 || beat > kMaxSeekBeat) {
114 return "beat must be between 0 and " + std::to_string(static_cast<int>(kMaxSeekBeat));
115 }
116 return {};
117}

119void seekTo(::AestraContent& content, Audio::TrackManager& tm, Audio::AudioEngine* engine, double beat) {
120 const double seconds = content.getViewFocus() == ViewFocus::Arsenal
121 ? std::max(0.0, beat * 60.0 / tempoBpm(tm))
122 : std::max(0.0, tm.getPlaylistModel().beatToSeconds(beat));
123 tm.setPosition(seconds);
124 tm.setPlayStartPosition(seconds);
125 if (engine) {
126 // seekRefusal bounds the beat, but a tempo map can still stretch it:
127 // only convert a sample position that fits, never cast past uint64_t.
128 const double samples = seconds * static_cast<double>(engine->getSampleRate());
129 if (std::isfinite(samples) && samples >= 0.0 && samples < 9.0e18) {
130 engine->setGlobalSamplePos(static_cast<uint64_t>(samples));
131 }
132 }

135JSON transportState(::AestraContent& content, Audio::TrackManager& tm) {
136 JSON result = JSON::object();
137 result.set("playing", JSON(tm.isPlaying()));
138 result.set("countInPending", JSON(tm.isCountInPending()));
139 result.set("positionBeats", JSON(positionBeats(content, tm)));
140 result.set("positionSeconds", JSON(std::max(0.0, tm.getPosition())));
141 result.set("bpm", JSON(tempoBpm(tm)));
142 result.set("focus", JSON(std::string(WorkspaceFocusModel::workspaceFocusName(content.getViewFocus()))));
143 JSON loop = JSON::object();
144 if (auto ui = content.getTrackManagerUI()) {
145 loop.set("enabled", JSON(ui->isLoopEnabled()));
146 loop.set("startBeat", JSON(ui->getLoopStartBeat()));
147 loop.set("endBeat", JSON(ui->getLoopEndBeat()));
148 }
149 result.set("loop", loop);
150 return result;
151}

159void registerMuseTransportVerbs(Audio::MuseService& service, ::AestraContent& content, ::AestraApp& app) {
160 ::AestraContent* c = &content;
161 ::AestraApp* a = &app;
162
163 // --- transport.state -----------------------------------------------------
164 registerOrLog(service,
165 transportSpec("transport.state", false,
166 "Where the transport is: playing, countInPending, positionBeats, "
167 "positionSeconds, bpm, focus (which transport Play drives) and the "
168 "timeline loop {enabled, startBeat, endBeat}."),
169 [c](const JSON&) {
170 auto tm = c->getTrackManager();
171 if (!tm) return noTimeline();
172 return HostVerbResult::success(transportState(*c, *tm));
173 });
174
175 // --- transport.play ------------------------------------------------------
176 {
177 auto spec = transportSpec(
178 "transport.play", false,
179 "Press Play, exactly as the transport bar does: the playhead moves, the count-in is "
180 "honoured, and the focused workspace decides what plays. 'in' switches to 'timeline' or "
181 "'arsenal' first; 'from' moves the playhead to that beat first. Already playing is a "
182 "success with changed=false.");
183 spec.args.push_back(numberArg("from", false, "beat to start from"));
184 HostVerbArg in;
185 in.name = "in";
186 in.type = Audio::FlagType::String;
187 in.required = false;
188 in.description = "timeline | arsenal";
189 spec.args.push_back(in);
190 registerOrLog(service, std::move(spec), [c, a](const JSON& args) {
191 auto tm = c->getTrackManager();
192 if (!tm) return noTimeline();
193 // Validate everything before changing anything: a refused request
194 // must leave the workspace and playhead as they were.
195 ViewFocus target = c->getViewFocus();
196 if (args.has("in") && !parsePlayTarget(args["in"].asString(), target)) {
197 return HostVerbResult::failure("invalid_args", "'in' must be 'timeline' or 'arsenal'");
198 }
199 const bool hasFrom = args.has("from");
200 const double from = hasFrom ? args["from"].asNumber() : 0.0;
201 if (hasFrom && (!std::isfinite(from) || from < 0.0 || from > kMaxSeekBeat)) {
202 return HostVerbResult::failure("invalid_args", "'from' must be a beat between 0 and " +
203 std::to_string(static_cast<int>(kMaxSeekBeat)));
204 }
205 if (hasFrom && target != ViewFocus::Timeline && target != ViewFocus::Arsenal) {
206 return HostVerbResult::failure("invalid_args", "'from' needs the timeline or the arsenal; pass 'in'");
207 }
208 if (c->getViewFocus() != target) {
209 c->setViewFocus(target);
210 }
211 if (hasFrom) {
212 seekTo(*c, *tm, a->getAudioEngine(), from);
213 }
214 const bool wasPlaying = tm->isPlaying() || tm->isCountInPending();
215 if (!wasPlaying) {
216 c->requestTransportPlay();
217 }
218 JSON result = transportState(*c, *tm);
219 result.set("changed", JSON(!wasPlaying));
220 return HostVerbResult::success(result);
221 });
222 }
223
224 // --- transport.stop / transport.pause ------------------------------------
225 {
226 auto spec = transportSpec("transport.stop", false,
227 "Press Stop. rewind=true is the double-stop: the playhead goes back to 0.");
228 HostVerbArg rewind;
229 rewind.name = "rewind";
230 rewind.type = Audio::FlagType::Bool;
231 rewind.required = false;
232 rewind.description = "return the playhead to beat 0";
233 spec.args.push_back(rewind);
234 registerOrLog(service, std::move(spec), [c](const JSON& args) {
235 auto tm = c->getTrackManager();
236 if (!tm) return noTimeline();
237 const bool rewindToStart = args.has("rewind") && args["rewind"].asBool();
238 c->stopFromCurrentFocus(rewindToStart);
239 return HostVerbResult::success(transportState(*c, *tm));
240 });
241 }
242 registerOrLog(service,
243 transportSpec("transport.pause", false,
244 "Pause: playback stops and the playhead stays where it is, so the next "
245 "transport.play resumes from there."),
246 [c](const JSON&) {
247 auto tm = c->getTrackManager();
248 if (!tm) return noTimeline();
249 c->pauseFromCurrentFocus();
250 return HostVerbResult::success(transportState(*c, *tm));
251 });
252
253 // --- transport.seek ------------------------------------------------------
254 {
255 auto spec = transportSpec("transport.seek", false,
256 "Move the playhead to a beat, as a click on the ruler does. Works while "
257 "playing or stopped; the next Play starts from here.");
258 spec.args.push_back(numberArg("beat", true, "beat position >= 0"));
259 registerOrLog(service, std::move(spec), [c, a](const JSON& args) {
260 auto tm = c->getTrackManager();
261 if (!tm) return noTimeline();
262 const double beat = args["beat"].asNumber();
263 if (const std::string refusal = seekRefusal(*c, beat); !refusal.empty()) {
264 return HostVerbResult::failure("invalid_args", refusal);
265 }
266 seekTo(*c, *tm, a->getAudioEngine(), beat);
267 return HostVerbResult::success(transportState(*c, *tm));
268 });
269 }
270
271 // --- transport.setLoop / transport.clearLoop -----------------------------
272 {
273 auto spec = transportSpec("transport.setLoop", false,
274 "Loop the timeline over [start, end) beats, as dragging that range on "
275 "the ruler does (the loop preset becomes Selection).");
276 spec.args.push_back(numberArg("start", true, "loop start beat >= 0"));
277 spec.args.push_back(numberArg("end", true, "loop end beat, after start"));
278 registerOrLog(service, std::move(spec), [c](const JSON& args) {
279 auto tm = c->getTrackManager();
280 auto ui = c->getTrackManagerUI();
281 if (!tm || !ui) return noTimeline();
282 const double start = args["start"].asNumber();
283 const double end = args["end"].asNumber();
284 if (!std::isfinite(start) || !std::isfinite(end) || start < 0.0 || !(end > start)) {
285 return HostVerbResult::failure("invalid_args", "need 0 <= start < end");
286 }
287 ui->selectLoopRange(start, end);
288 return HostVerbResult::success(transportState(*c, *tm));
289 });
290 }
291 registerOrLog(service,
292 transportSpec("transport.clearLoop", false,
293 "Turn the timeline loop off and clear its range, as a click on the ruler "
294 "without a drag does."),
295 [c](const JSON&) {
296 auto tm = c->getTrackManager();
297 auto ui = c->getTrackManagerUI();
298 if (!tm || !ui) return noTimeline();
299 ui->clearLoopRange();
300 return HostVerbResult::success(transportState(*c, *tm));
301 });
302}
