94 float getValueAtBeat(double beat) const {
95 if (points.empty()) {
96 return defaultValue;
97 }
98
99 // Find surrounding points (points are sorted by beat)
100 const AutomationPoint* prev = nullptr;
101 const AutomationPoint* next = nullptr;
102
103 for (const auto& pt : points) {
104 if (pt.beat <= beat) {
105 prev = &pt;
106 } else {
107 next = &pt;
108 break;
109 }
110 }
111
112 if (!prev && !next) {
113 return defaultValue;
114 }
115 if (!prev) {
116 return next->value;
117 }
118 if (!next) {
119 return prev->value;
120 }
121
122 // Linear interpolation in beat domain
123 const double beatRange = next->beat - prev->beat;
124 if (beatRange <= 0.0) {
125 return prev->value;
126 }
127
128 const double t = (beat - prev->beat) / beatRange;
129 return prev->value + static_cast<float>(t) * (next->value - prev->value);
130 }

152 void addPoint(double beat, float value, double samplesPerBeat, float tension = 0.5f) {
153 AutomationPoint pt;
154 pt.sample = static_cast<uint64_t>(beat * samplesPerBeat);
155 pt.value = value;
156 // beat and curve are what ProjectSerializer persists — leaving them at
157 // their zero defaults made every point created through this API (loader,
158 // UI draw path) serialize at beat 0 on the next save.
159 pt.beat = beat;
160 pt.curve = tension;
161
162 // Insert in sorted order (beat domain — see AutomationPoint docs)
163 auto it = points.begin();
164 while (it != points.end() && it->beat < pt.beat) {
165 ++it;
166 }
167 points.insert(it, pt);
168 }

170 void removePoint(size_t index) {
171 if (index < points.size()) {

174 }
