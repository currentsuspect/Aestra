84 float getValueAtBeat(double beat, double samplesPerBeat) const {
85 if (points.empty()) {
86 return defaultValue;
87 }
88
89 const uint64_t targetSample = static_cast<uint64_t>(beat * samplesPerBeat);
90
91 // Find surrounding points
92 const AutomationPoint* prev = nullptr;
93 const AutomationPoint* next = nullptr;
94
95 for (const auto& pt : points) {
96 if (pt.sample <= targetSample) {
97 prev = &pt;
98 } else {
99 next = &pt;
100 break;
101 }
102 }
103
104 if (!prev && !next) {
105 return defaultValue;
106 }
107 if (!prev) {
108 return next->value;
109 }
110 if (!next) {
111 return prev->value;
112 }
113
114 // Linear interpolation
115 const double sampleRange = static_cast<double>(next->sample - prev->sample);
116 if (sampleRange <= 0.0) {
117 return prev->value;
118 }
119
120 const double t = static_cast<double>(targetSample - prev->sample) / sampleRange;
121 return prev->value + static_cast<float>(t) * (next->value - prev->value);
122 }
