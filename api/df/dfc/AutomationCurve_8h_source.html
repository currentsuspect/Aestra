68inline double automationTensionShape(double t, float tension) {
69 constexpr double kStrength = 6.0; // the bend at |tension| == 1
70 const double k = std::clamp(static_cast<double>(tension), -1.0, 1.0) * kStrength;
71 if (std::abs(k) < 1e-6) {
72 return t;
73 }
74 return (1.0 - std::exp(-k * t)) / (1.0 - std::exp(-k));
75}

178 float getValueAtBeat(double beat) const {
179 if (points.empty()) {
180 return defaultValue;
181 }
182
183 // Find surrounding points (points are sorted by beat)
184 const AutomationPoint* prev = nullptr;
185 const AutomationPoint* next = nullptr;
186
187 for (const auto& pt : points) {
188 if (pt.beat <= beat) {
189 prev = &pt;
190 } else {
191 next = &pt;
192 break;
193 }
194 }
195
196 if (!prev && !next) {
197 return defaultValue;
198 }
199 if (!prev) {
200 return next->value;
201 }
202 if (!next) {
203 return prev->value;
204 }
205
206 // Interpolation in the beat domain, shaped by the segment's tension
207 const double beatRange = next->beat - prev->beat;
208 if (beatRange <= 0.0) {
209 return prev->value;
210 }
211
212 const double t = automationTensionShape((beat - prev->beat) / beatRange, prev->curve);
213 return prev->value + static_cast<float>(t) * (next->value - prev->value);
214 }

236 void addPoint(double beat, float value, double samplesPerBeat, float tension = 0.0f) {
237 AutomationPoint pt;
238 pt.sample = static_cast<uint64_t>(beat * samplesPerBeat);
239 pt.value = value;
240 // beat and curve are what ProjectSerializer persists — leaving them at
241 // their zero defaults made every point created through this API (loader,
242 // UI draw path) serialize at beat 0 on the next save.
243 pt.beat = beat;
244 pt.curve = tension;
245
246 // Insert in sorted order (beat domain — see AutomationPoint docs)
247 auto it = points.begin();
248 while (it != points.end() && it->beat < pt.beat) {
249 ++it;
250 }
251 points.insert(it, pt);
252 }
253
254 void removePoint(size_t index) {
255 if (index < points.size()) {
256 points.erase(points.begin() + index);
257 }
258 }

262 // so sorting by sample could scramble order after edits.
263 std::sort(points.begin(), points.end(),
264 [](const AutomationPoint& a, const AutomationPoint& b) { return a.beat < b.beat; });

293inline bool automationCurvesEqual(const std::vector<AutomationCurve>& a,
294 const std::vector<AutomationCurve>& b) noexcept {
295 if (a.size() != b.size())
296 return false;
297 for (size_t i = 0; i < a.size(); ++i) {
298 const AutomationCurve& x = a[i];
299 const AutomationCurve& y = b[i];
300 if (x.name != y.name || x.target != y.target || x.mixerChannelId != y.mixerChannelId ||
301 x.defaultValue != y.defaultValue || x.muted != y.muted || !automationRegionsEqual(x.regions, y.regions) || x.deviceInstanceId != y.deviceInstanceId || x.paramId != y.paramId) {
302 return false;
303 }
304 if (x.points.size() != y.points.size())
305 return false;
306 for (size_t p = 0; p < x.points.size(); ++p) {
307 if (!automationPointsEqual(x.points[p], y.points[p]))
308 return false;
309 }
310 }
311 return true;
