103 float getValueAtBeat(double beat) const {
104 if (points.empty()) {
105 return defaultValue;
106 }
107
108 // Find surrounding points (points are sorted by beat)
109 const AutomationPoint* prev = nullptr;
110 const AutomationPoint* next = nullptr;
111
112 for (const auto& pt : points) {
113 if (pt.beat <= beat) {
114 prev = &pt;
115 } else {
116 next = &pt;
117 break;
118 }
119 }
120
121 if (!prev && !next) {
122 return defaultValue;
123 }
124 if (!prev) {
125 return next->value;
126 }
127 if (!next) {
128 return prev->value;
129 }
130
131 // Linear interpolation in beat domain
132 const double beatRange = next->beat - prev->beat;
133 if (beatRange <= 0.0) {
134 return prev->value;
135 }
136
137 const double t = (beat - prev->beat) / beatRange;
138 return prev->value + static_cast<float>(t) * (next->value - prev->value);
139 }

161 void addPoint(double beat, float value, double samplesPerBeat, float tension = 0.5f) {
162 AutomationPoint pt;
163 pt.sample = static_cast<uint64_t>(beat * samplesPerBeat);
164 pt.value = value;
165 // beat and curve are what ProjectSerializer persists — leaving them at
166 // their zero defaults made every point created through this API (loader,
167 // UI draw path) serialize at beat 0 on the next save.
168 pt.beat = beat;
169 pt.curve = tension;
170
171 // Insert in sorted order (beat domain — see AutomationPoint docs)
172 auto it = points.begin();
173 while (it != points.end() && it->beat < pt.beat) {
174 ++it;
175 }
176 points.insert(it, pt);
177 }
178
179 void removePoint(size_t index) {
180 if (index < points.size()) {
181 points.erase(points.begin() + index);
182 }
183 }

185 void sortPoints() {
186 // Beat domain: the UI drag path updates beat (not the sample cache),
187 // so sorting by sample could scramble order after edits.

189 [](const AutomationPoint& a, const AutomationPoint& b) { return a.beat < b.beat; });
