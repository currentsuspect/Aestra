121 // Polyphase coefficients, TRANSPOSED.
122 //
123 // m_polyphaseT[tap][phase] is the coefficient of `phase` at `tap`, where
124 // the prototype FIR is h[phase + kOversampleFactor*tap].
125 //
126 // The transposed layout is what makes the hot loop a single vector
127 // accumulation: the four phases read the SAME history, so one 4-wide
128 // accumulator can hold all four phase sums at once, one phase per lane.
129 // Storing the coefficients phase-major would mean four separate 12-tap
130 // dot products, each with its own serial accumulator.
131 //
132 // The tap order is unchanged, so each lane accumulates exactly the same
133 // products in exactly the same order as the scalar version did. The result
134 // is BIT-IDENTICAL, not merely close.
135 alignas(16) std::array<std::array<float, kOversampleFactor>, kTapsPerPhase> m_polyphaseT{};
136 // The hot loop reads four coefficients at a time. The loads use the
137 // unaligned form on purpose: the member IS 16-byte aligned and each inner
138 // array is exactly 16 bytes, so the addresses are aligned in practice —
139 // but making correctness depend on that is a trap, and an unaligned load
140 // costs nothing on any CPU this runs on.
141 static_assert(sizeof(std::array<float, kOversampleFactor>) == 16,
142 "the polyphase coefficient rows must be exactly one vector wide");
143
144 // Per-channel history of the most recent kTapsPerPhase input samples.

162 // Push a sample into a history FIFO (newest first).
163 static inline void pushHistory(std::array<float, kTapsPerPhase>& hist, float sample) noexcept {
164#if defined(AESTRA_DSP_HAS_SSE2)
165 // Shift the newest-first FIFO right by one and write the sample at 0.
166 // Descending four-wide moves: each store reads the four entries one
167 // below it, and those have not been written yet because the copy runs
168 // from the top down.
169 _mm_storeu_ps(&hist[8], _mm_loadu_ps(&hist[7]));
170 _mm_storeu_ps(&hist[4], _mm_loadu_ps(&hist[3]));
171 _mm_storeu_ps(&hist[1], _mm_loadu_ps(&hist[0]));
172#else
173 for (uint32_t i = kTapsPerPhase - 1; i > 0; --i) {
174 hist[i] = hist[i - 1];
175 }
176#endif
177 hist[0] = sample;
178 }
179
180 // Compute the maximum |y| across the 4 polyphase outputs for one channel,
181 // using the current history. Updates the running true peak.
182 inline void updateTruePeakFromHistory(const std::array<float, kTapsPerPhase>& hist,
183 float& runningTruePeak) const noexcept {
184#if defined(AESTRA_DSP_HAS_SSE2)
185 // One accumulator, four lanes: lane p is the sum for polyphase phase p.
186 __m128 acc = _mm_setzero_ps();
187 for (uint32_t tap = 0; tap < kTapsPerPhase; ++tap) {
188 acc = _mm_add_ps(acc, _mm_mul_ps(_mm_set1_ps(hist[tap]),
189 _mm_loadu_ps(&m_polyphaseT[tap][0])));
190 }
191 // Magnitude of each lane, then the largest of the four.
192 //
193 // On non-finite input this still matches the scalar path exactly, and
194 // not by luck: a NaN in the history makes EVERY lane NaN, the vector
195 // maximum can therefore yield a NaN, and the final `peak > running`
196 // comparison is false for a NaN -- so a NaN cannot raise the peak,
197 // which is what the scalar path's per-phase `if (mag > peak)` did.
198 // (A NaN in only ONE lane would need a NaN coefficient, and the
199 // coefficients are generated, never sampled.)
200 const __m128 mag = _mm_andnot_ps(_mm_set1_ps(-0.0f), acc);
201 __m128 t = _mm_max_ps(mag, _mm_movehl_ps(mag, mag));
202 t = _mm_max_ss(t, _mm_shuffle_ps(t, t, 0x55));
203 const float peak = _mm_cvtss_f32(t);
204 if (peak > runningTruePeak) {
205 runningTruePeak = peak;
206 }
207#else
208 for (uint32_t phase = 0; phase < kOversampleFactor; ++phase) {
209 float acc = 0.0f;
210 for (uint32_t tap = 0; tap < kTapsPerPhase; ++tap) {
211 acc += m_polyphaseT[tap][phase] * hist[tap];
212 }
213 const float mag = acc < 0.0f ? -acc : acc;
214 if (mag > runningTruePeak) {
215 runningTruePeak = mag;
216 }
217 }
218#endif
219 }
