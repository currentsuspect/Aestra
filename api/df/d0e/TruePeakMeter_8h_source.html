134 // Push a sample into a history FIFO (newest first), returning nothing.
135 static inline void pushHistory(std::array<float, kTapsPerPhase>& hist, float sample) noexcept {
136 // Shift newer-first FIFO right by one and write sample at index 0.
137 for (uint32_t i = kTapsPerPhase - 1; i > 0; --i) {
138 hist[i] = hist[i - 1];
139 }
140 hist[0] = sample;
141 }
142
143 // Compute the maximum |y| across the 4 polyphase outputs for one channel,
144 // using the current history. Updates the running true peak.
145 inline void updateTruePeakFromHistory(const std::array<float, kTapsPerPhase>& hist,
146 float& runningTruePeak) const noexcept {
147 for (uint32_t phase = 0; phase < kOversampleFactor; ++phase) {
148 float acc = 0.0f;
149 const auto& coeffs = m_polyphase[phase];
150 for (uint32_t tap = 0; tap < kTapsPerPhase; ++tap) {
151 acc += coeffs[tap] * hist[tap];
152 }
153 const float mag = acc < 0.0f ? -acc : acc;
154 if (mag > runningTruePeak) {
155 runningTruePeak = mag;
156 }
157 }
158 }
