43void UIOutputChip::onRender(NUIRenderer& renderer) {
44 const auto b = getBounds();
45 if (b.isEmpty())
46 return;
47 auto& theme = NUIThemeManager::getInstance();
48 const float alpha = m_dimmed ? 0.45f : 1.0f;
49 if (m_compact) {
50 renderer.fillRoundedRect(NUIRect{b.x, b.y, b.width, b.height}, 2.0f, m_color.withAlpha(0.85f * alpha));
51 return;
52 }
53 const NUIRect r{std::floor(b.x) + 0.5f, std::floor(b.y) + 0.5f, std::max(1.0f, std::floor(b.width) - 1.0f),
54 std::max(1.0f, std::floor(b.height) - 1.0f)};
55
56 renderer.fillRoundedRect(r, kRadius, m_color.withAlpha((m_hovered ? 0.24f : 0.14f) * alpha));
57 renderer.strokeRoundedRect(r, kRadius, 1.0f, m_color.withAlpha((m_hovered ? 0.70f : 0.42f) * alpha));
58
59 const NUIRect dot{r.x + 5.0f, r.y + (r.height - kDot) * 0.5f, kDot, kDot};
60 renderer.fillRoundedRect(dot, kDot * 0.5f, m_color.withAlpha(alpha));
61
62 const float textX = dot.right() + 4.0f;
63 const NUIRect textRect{textX, r.y, std::max(1.0f, r.right() - textX - 4.0f), r.height};
64 // The arrow says "goes to"; ellipsis keeps a long bus name inside the pill.
65 std::string text = "\xE2\x86\x92 " + m_label;
66 const float avail = textRect.width;
67 while (text.size() > 4 && renderer.measureText(text, kFont).width > avail)
68 text.pop_back();
69 if (text.size() < m_label.size() + 4)
70 text += "\xE2\x80\xA6";
71 renderer.drawText(text, {textRect.x, textRect.y + (textRect.height - kFont) * 0.5f - 1.0f}, kFont,
72 theme.getColor("textPrimary").withAlpha(0.9f * alpha));
73}

75bool UIOutputChip::onMouseEvent(const NUIMouseEvent& event) {
76 if (!isVisible() || !isEnabled())
77 return false;
78 const bool inside = getBounds().contains(event.position);
79 if (event.button == NUIMouseButton::None) {
80 if (inside != m_hovered) {
81 m_hovered = inside;
82 repaint();
83 }
84 return false;
85 }
86 if (event.pressed && event.button == NUIMouseButton::Left && inside) {
87 m_pressed = true;
88 return true;
89 }
90 if (event.released && event.button == NUIMouseButton::Left) {
91 const bool fire = m_pressed && inside;
92 m_pressed = false;
93 if (fire && onClicked)
94 onClicked();
95 return fire;
96 }
97 return false;
98}
