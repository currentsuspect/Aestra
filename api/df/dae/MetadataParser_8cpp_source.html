18AudioMetadata MetadataParser::parse(const std::string& filePath) {
19 AudioMetadata meta;
20
21 std::string ext = toLowerCase(getFileExtension(filePath));
22
23 if (ext == "mp3") {
24 parseID3v2(filePath, meta);
25 } else if (ext == "flac") {
26 parseFLAC(filePath, meta);
27 } else if (ext == "wav") {
28 parseWAV(filePath, meta);
29 }
30
31 return meta;
32}

51uint32_t MetadataParser::readSynchsafeInt(const uint8_t* data) {
52 // ID3v2 synchsafe: 7 bits per byte, MSB is always 0
53 return ((data[0] & 0x7F) << 21) | ((data[1] & 0x7F) << 14) | ((data[2] & 0x7F) << 7) | (data[3] & 0x7F);
54}

56uint32_t MetadataParser::readBigEndian32(const uint8_t* data) {
57 return (data[0] << 24) | (data[1] << 16) | (data[2] << 8) | data[3];

64std::string MetadataParser::decodeID3String(const uint8_t* data, size_t size, uint8_t encoding) {
65 if (size == 0)
66 return "";
67
68 // Encoding: 0=ISO-8859-1, 1=UTF-16 BOM, 2=UTF-16BE, 3=UTF-8
69 if (encoding == 0 || encoding == 3) {
70 // ASCII-compatible: Just copy, skip null terminators
71 size_t len = size;
72 while (len > 0 && data[len - 1] == 0)
73 len--;
74 return std::string(reinterpret_cast<const char*>(data), len);
75 } else if (encoding == 1) {
76 // UTF-16 with BOM
77 if (size < 2)
78 return "";
79 bool littleEndian = (data[0] == 0xFF && data[1] == 0xFE);
80 // Simple: Just extract ASCII-range characters (good enough for most music)
81 std::string result;
82 for (size_t i = 2; i + 1 < size; i += 2) {
83 uint16_t ch = littleEndian ? (data[i] | (data[i + 1] << 8)) : ((data[i] << 8) | data[i + 1]);
84 if (ch == 0)
85 break;
86 if (ch < 128)
87 result += static_cast<char>(ch);
88 else
89 result += '?'; // Non-ASCII placeholder
90 }
91 return result;
92 } else if (encoding == 2) {
93 // UTF-16BE no BOM
94 std::string result;
95 for (size_t i = 0; i + 1 < size; i += 2) {
96 uint16_t ch = (data[i] << 8) | data[i + 1];
97 if (ch == 0)
98 break;
99 if (ch < 128)
100 result += static_cast<char>(ch);
101 else
102 result += '?';
103 }
104 return result;
105 }
106
107 return "";
108}

114bool MetadataParser::parseID3v2(const std::string& filePath, AudioMetadata& meta) {
115 std::ifstream file(filePath, std::ios::binary);
116 if (!file)
117 return false;
118
119 // Read header
120 uint8_t header[10];
121 file.read(reinterpret_cast<char*>(header), 10);
122 if (!file || file.gcount() < 10)
123 return false;
124
125 // Check "ID3" signature
126 if (header[0] != 'I' || header[1] != 'D' || header[2] != '3') {
127 return false; // No ID3v2 tag
128 }
129
130 uint8_t majorVersion = header[3]; // 3 = ID3v2.3, 4 = ID3v2.4
131 // uint8_t minorVersion = header[4];
132 // uint8_t flags = header[5];
133 uint32_t tagSize = readSynchsafeInt(&header[6]);
134
135 // Guard against heap exhaustion: max 10 MB tag
136 constexpr uint32_t MAX_TAG_SIZE = 10 * 1024 * 1024;
137 if (tagSize > MAX_TAG_SIZE) {
138 return false;
139 }
140
141 // Read entire tag
142 std::vector<uint8_t> tagData(tagSize);
143 file.read(reinterpret_cast<char*>(tagData.data()), tagSize);
144 if (!file || static_cast<uint32_t>(file.gcount()) < tagSize)
145 return false;
146
147 // Parse frames
148 size_t offset = 0;
149 while (offset + 10 <= tagSize) {
150 // Frame header: 4 bytes ID, 4 bytes size, 2 bytes flags
151 std::string frameId(reinterpret_cast<char*>(&tagData[offset]), 4);
152 if (frameId[0] == '\0')
153 break; // Padding
154
155 uint32_t frameSize;
156 if (majorVersion == 4) {
157 frameSize = readSynchsafeInt(&tagData[offset + 4]);
158 } else {
159 frameSize = readBigEndian32(&tagData[offset + 4]);
160 }
161
162 // uint16_t frameFlags = (tagData[offset + 8] << 8) | tagData[offset + 9];
163 offset += 10;
164
165 if (offset + frameSize > tagSize)
166 break;
167
168 const uint8_t* frameData = &tagData[offset];
169
170 // Text frames (TIT2, TPE1, TALB)
171 if (frameId == "TIT2" && frameSize > 1) {
172 meta.title = decodeID3String(frameData + 1, frameSize - 1, frameData[0]);
173 } else if (frameId == "TPE1" && frameSize > 1) {
174 meta.artist = decodeID3String(frameData + 1, frameSize - 1, frameData[0]);
175 } else if (frameId == "TALB" && frameSize > 1) {
176 meta.album = decodeID3String(frameData + 1, frameSize - 1, frameData[0]);
177 } else if (frameId == "APIC" && frameSize > 12) {
178 // Attached Picture
179 uint8_t encoding = frameData[0];
180 size_t pos = 1;
181
182 // MIME type (null-terminated)
183 std::string mimeType;
184 while (pos < frameSize && frameData[pos] != 0) {
185 mimeType += static_cast<char>(frameData[pos++]);
186 }
187 pos++; // Skip null
188
189 if (pos >= frameSize) {
190 offset += frameSize;
191 continue;
192 }
193
194 uint8_t pictureType = frameData[pos++];
195 (void)pictureType; // Could filter for front cover (type 3)
196
197 // Description (null-terminated, encoding-aware)
198 if (encoding == 0 || encoding == 3) {
199 while (pos < frameSize && frameData[pos] != 0)
200 pos++;
201 pos++; // Skip null
202 } else {
203 // UTF-16: double null terminator
204 while (pos + 1 < frameSize && !(frameData[pos] == 0 && frameData[pos + 1] == 0))
205 pos += 2;
206 pos += 2;
207 }
208
209 // Picture data
210 if (pos < frameSize && meta.coverArtData.empty()) {
211 meta.coverArtData.assign(frameData + pos, frameData + frameSize);
212 meta.coverArtMimeType = mimeType.empty() ? "image/jpeg" : mimeType;
213 }
214 }
215
216 offset += frameSize;
217 }
218
219 return true;
220}

226bool MetadataParser::parseFLAC(const std::string& filePath, AudioMetadata& meta) {
227 std::ifstream file(filePath, std::ios::binary);
228 if (!file)
229 return false;
230
231 // Check "fLaC" signature
232 uint8_t sig[4];
233 file.read(reinterpret_cast<char*>(sig), 4);
234 if (!file || sig[0] != 'f' || sig[1] != 'L' || sig[2] != 'a' || sig[3] != 'C') {
235 return false;
236 }
237
238 // Parse metadata blocks
239 bool lastBlock = false;
240 while (!lastBlock && file) {
241 uint8_t blockHeader[4];
242 file.read(reinterpret_cast<char*>(blockHeader), 4);
243 if (!file || file.gcount() < 4)
244 break;
245
246 lastBlock = (blockHeader[0] & 0x80) != 0;
247 uint8_t blockType = blockHeader[0] & 0x7F;
248 uint32_t blockSize = readBigEndian24(&blockHeader[1]);
249
250 if (blockSize == 0)
251 continue;
252
253 // Cap block size at 10MB to prevent corrupted files from huge allocations
254 if (blockSize > 10 * 1024 * 1024)
255 break;
256
257 std::vector<uint8_t> blockData(blockSize);
258 file.read(reinterpret_cast<char*>(blockData.data()), blockSize);
259 if (!file || static_cast<uint32_t>(file.gcount()) < blockSize)
260 break;
261
262 if (blockType == 0) {
263 // STREAMINFO - extract duration
264 if (blockSize >= 18) {
265 // Bytes 10-13 (upper 4 bits) + bytes 14-17: total samples
266 // Sample rate: bytes 10-12 (20 bits)
267 uint32_t sampleRate = (blockData[10] << 12) | (blockData[11] << 4) | (blockData[12] >> 4);
268 uint64_t totalSamples = (static_cast<uint64_t>(blockData[13] & 0x0F) << 32) | (blockData[14] << 24) |
269 (blockData[15] << 16) | (blockData[16] << 8) | blockData[17];
270 if (sampleRate > 0) {
271 meta.durationSeconds = static_cast<double>(totalSamples) / sampleRate;
272 }
273 }
274 } else if (blockType == 4) {
275 // VORBIS_COMMENT
276 if (blockSize < 8)
277 continue;
278
279 size_t pos = 0;
280 // Vendor string (little-endian length)
281 uint32_t vendorLen = blockData[0] | (blockData[1] << 8) | (blockData[2] << 16) | (blockData[3] << 24);
282 // [SEC-RTM-012] Guard against integer overflow: vendorLen must fit within blockSize
283 if (vendorLen > blockSize - 4)
284 continue;
285 pos = 4 + vendorLen;
286 if (pos + 4 > blockSize)
287 continue;
288
289 // Number of comments
290 uint32_t numComments =
291 blockData[pos] | (blockData[pos + 1] << 8) | (blockData[pos + 2] << 16) | (blockData[pos + 3] << 24);
292 pos += 4;
293
294 for (uint32_t i = 0; i < numComments && pos + 4 <= blockSize; ++i) {
295 uint32_t commentLen = blockData[pos] | (blockData[pos + 1] << 8) | (blockData[pos + 2] << 16) |
296 (blockData[pos + 3] << 24);
297 pos += 4;
298 if (pos + commentLen > blockSize)
299 break;
300
301 std::string comment(reinterpret_cast<char*>(&blockData[pos]), commentLen);
302 pos += commentLen;
303
304 // Parse "KEY=VALUE"
305 size_t eq = comment.find('=');
306 if (eq != std::string::npos) {
307 std::string key = toLowerCase(comment.substr(0, eq));
308 std::string value = comment.substr(eq + 1);
309
310 if (key == "title")
311 meta.title = value;
312 else if (key == "artist")
313 meta.artist = value;
314 else if (key == "album")
315 meta.album = value;
316 }
317 }
318 } else if (blockType == 6) {
319 // PICTURE block (cover art)
320 // Structure: type(4) + mimeLen(4) + mime(mimeLen) + descLen(4) + desc(descLen)
321 // + width(4) + height(4) + depth(4) + colors(4) + dataLen(4) + data(dataLen)
322 if (blockSize < 32 || !meta.coverArtData.empty())
323 continue;
324
325 size_t pos = 0;
326
327 // Picture type (4 bytes)
328 if (pos + 4 > blockSize) continue;
329 pos += 4;
330
331 // MIME type length + string
332 if (pos + 4 > blockSize) continue;
333 uint32_t mimeLen = readBigEndian32(&blockData[pos]);
334 pos += 4;
335 if (mimeLen > 256 || pos + mimeLen > blockSize) continue;
336 std::string mimeType(reinterpret_cast<char*>(&blockData[pos]), mimeLen);
337 pos += mimeLen;
338
339 // Description length + string
340 if (pos + 4 > blockSize) continue;
341 uint32_t descLen = readBigEndian32(&blockData[pos]);
342 pos += 4;
343 if (descLen > 4096 || pos + descLen > blockSize) continue;
344 pos += descLen;
345
346 // Width(4) + Height(4) + Depth(4) + Colors(4)
347 if (pos + 16 > blockSize) continue;
348 pos += 16;
349
350 // Image data length
351 if (pos + 4 > blockSize) continue;
352 uint32_t dataLen = readBigEndian32(&blockData[pos]);
353 pos += 4;
354
355 // Validate image data: must have at least a valid JPEG/PNG header
356 if (dataLen < 8 || pos + dataLen > blockSize) continue;
357
358 // Sanity check: first bytes should be JPEG (FF D8 FF) or PNG (89 50 4E 47)
359 const uint8_t* imgData = &blockData[pos];
360 bool isJPEG = (imgData[0] == 0xFF && imgData[1] == 0xD8 && imgData[2] == 0xFF);
361 bool isPNG = (imgData[0] == 0x89 && imgData[1] == 0x50 && imgData[2] == 0x4E && imgData[3] == 0x47);
362 if (!isJPEG && !isPNG) continue;
363
364 meta.coverArtData.assign(imgData, imgData + dataLen);
365 meta.coverArtMimeType = mimeType;
366 }
367 }
368
369 return true;
370}

376bool MetadataParser::parseWAV(const std::string& filePath, AudioMetadata& meta) {
377 std::ifstream file(filePath, std::ios::binary);
378 if (!file)
379 return false;
380
381 uint8_t riffHeader[12];
382 file.read(reinterpret_cast<char*>(riffHeader), 12);
383 if (!file || file.gcount() < 12)
384 return false;
385
386 if (std::memcmp(riffHeader, "RIFF", 4) != 0 || std::memcmp(riffHeader + 8, "WAVE", 4) != 0) {
387 return false;
388 }
389
390 uint16_t numChannels = 0;
391 uint32_t sampleRate = 0;
392 uint16_t bitsPerSample = 0;
393 uint32_t dataSize = 0;
394 bool foundFmt = false;
395 bool foundData = false;
396
397 uint8_t chunkHeader[8];
398 while (file.read(reinterpret_cast<char*>(chunkHeader), 8)) {
399 uint32_t chunkSize = chunkHeader[4] | (chunkHeader[5] << 8) |
400 (chunkHeader[6] << 16) | (chunkHeader[7] << 24);
401 char chunkId[5] = {};
402 std::memcpy(chunkId, chunkHeader, 4);
403
404 if (std::memcmp(chunkHeader, "fmt ", 4) == 0 && !foundFmt) {
405 foundFmt = true;
406 std::vector<uint8_t> fmtData(std::min<uint32_t>(chunkSize, 40));
407 file.read(reinterpret_cast<char*>(fmtData.data()), fmtData.size());
408 if (fmtData.size() >= 16) {
409 numChannels = fmtData[2] | (fmtData[3] << 8);
410 sampleRate = fmtData[4] | (fmtData[5] << 8) | (fmtData[6] << 16) | (fmtData[7] << 24);
411 bitsPerSample = fmtData[14] | (fmtData[15] << 8);
412 }
413 if (chunkSize > fmtData.size()) {
414 file.seekg(chunkSize - fmtData.size(), std::ios::cur);
415 }
416 } else if (std::memcmp(chunkHeader, "data", 4) == 0 && !foundData) {
417 foundData = true;
418 dataSize = chunkSize;
419 break;
420 } else {
421 file.seekg(chunkSize, std::ios::cur);
422 }
423 }
424
425 if (sampleRate > 0 && numChannels > 0 && bitsPerSample > 0) {
426 uint32_t bytesPerSample = (bitsPerSample / 8) * numChannels;
427 if (bytesPerSample > 0) {
428 meta.durationSeconds = static_cast<double>(dataSize) / (sampleRate * bytesPerSample);
429 }
430 }
431
432 return true;
433}
