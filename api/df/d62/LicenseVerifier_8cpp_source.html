63UserProfile loadProfile() {
64 UserProfile profile;
65 // Defaults for open-source builds
66 profile.username = "Guest";
67 profile.tier = "Aestra Core";
68 profile.serial = "CORE-XXXXXXX";
69 profile.signature = "";
70 profile.verified = false;
71
72 const std::string path = getLicenseFilePath();
73 std::ifstream f(path, std::ios::in | std::ios::binary);
74 if (!f.good()) {
75 return profile; // fallback defaults
76 }
77 std::stringstream buffer; buffer << f.rdbuf();
78 try {
79 Aestra::JSON j = Aestra::JSON::parse(buffer.str());
80 profile = parseProfile(j);
81 } catch (const std::exception& e) {
82 Aestra::Log::warning(std::string("[LicenseVerifier] Failed to parse user profile: ") + e.what());
83 }
84 return profile;
85}

91 std::ofstream f(p.string(), std::ios::out | std::ios::binary | std::ios::trunc);
92 if (!f.good()) return false;
93 Aestra::JSON j = serializeProfile(profile);
94 f << j.toString(2);
95 return true;

98 return false;
