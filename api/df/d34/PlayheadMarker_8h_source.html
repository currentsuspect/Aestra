4// One playhead for the timeline, the piano roll and the Arsenal ruler: a flat,
5// filled head (a short bar over a point) and a solid 2 px line. The heads used
6// to be three 1.2 px strokes, which read as a hollow, jagged outline; the line
7// was a 1 px stroke at 72 %. Both are now whole-pixel fills, so they stay crisp
8// at any x and match the flat look of the clips and the Arsenal (v0.8.1).
9//
10// Header-only and renderer-only, so every widget can share it.
11
12#include "../Graphics/NUIRenderer.h"
13
14#include <cmath>
15
16namespace AestraUI::Playhead {
17
18constexpr float kLineWidth = 2.0f;
19constexpr float kHeadWidth = 11.0f; // odd, so the point lands on the line's centre
20constexpr float kHeadBarHeight = 4.0f;
21constexpr float kHeadPointHeight = 5.0f;
22constexpr float kHeadHeight = kHeadBarHeight + kHeadPointHeight;
23
25inline float lineLeft(float x) { return std::floor(x) - 1.0f; }
26
28inline void drawLine(NUIRenderer& renderer, float x, float top, float bottom, const NUIColor& color) {
29 if (bottom <= top) return;
30 renderer.fillRect(NUIRect(lineLeft(x), std::round(top), kLineWidth, std::round(bottom) - std::round(top)), color);
31}
32
38inline void drawHead(NUIRenderer& renderer, float x, float tipY, const NUIColor& color, float scale = 1.0f) {
39 const float cx = lineLeft(x) + kLineWidth * 0.5f;
40 const float halfW = std::round(kHeadWidth * scale) * 0.5f;
41 const float pointH = std::round(kHeadPointHeight * scale);
42 const float barH = std::round(kHeadBarHeight * scale);
43 const float tip = std::round(tipY);
44 const float pointTop = tip - pointH;
45 renderer.fillRect(NUIRect(cx - halfW, pointTop - barH, halfW * 2.0f, barH), color);
46 // The point as a strip from the left edge, through the tip, to the right edge.
47 const NUIPoint top[3] = {{cx - halfW, pointTop}, {cx, pointTop}, {cx + halfW, pointTop}};
48 const NUIPoint bottom[3] = {{cx - halfW, pointTop}, {cx, tip}, {cx + halfW, pointTop}};
49 renderer.fillWaveform(top, bottom, 3, color);
50 // The line's own top half-pixel, so head and line meet without a seam.
51 renderer.fillRect(NUIRect(lineLeft(x), tip - 2.0f, kLineWidth, 2.0f), color);
52}
53
54} // namespace AestraUI::Playhead
