29constexpr double kSourceLineRefreshSeconds = 0.5;

32const std::string& TrackUIComponent::headerSourceLine() {
33 // The standard clock, not the platform's: headless harnesses (UIScenarioTest) paint lanes with no platform layer.
34 const double now =
35 std::chrono::duration<double>(std::chrono::steady_clock::now().time_since_epoch()).count();
36 if (m_headerSourceLineAt >= 0.0 && now - m_headerSourceLineAt < kSourceLineRefreshSeconds) {
37 return m_headerSourceLine;
38 }
39 m_headerSourceLineAt = now;
40 m_headerSourceLine.clear();
41 if (!m_trackManager || m_isNestedLane) return m_headerSourceLine;
42
43 // Lanes carry no mixer channel of their own: what a lane "goes to" is the channel its Arsenal units play
44 // into. When every unit on the lane shares one channel that channel is named (unless it is Master) and its
45 // level drives the header meter; when they split, neither is shown.
46 TrackHeader::LaneSource source;
47 bool sharedTarget = true;
48 bool anyUnit = false;
49 uint32_t target = 0;
50 if (const PlaylistLane* lane = m_trackManager->getPlaylistModel().getLane(m_laneId)) {
51 std::unordered_set<uint64_t> seen;
52 const auto& units = m_trackManager->getUnitManager();
53 for (const auto& clip : lane->clips) {
54 auto pattern = m_trackManager->getPatternManager().getPattern(clip.patternId);
55 if (!pattern) continue;
56 if (pattern->isAudio()) {
57 source.hasAudio = true;
58 continue;
59 }
60 if (!pattern->isMidi()) continue;
61 for (const auto& note : std::get<MidiPayload>(pattern->payload).notes) {
62 if (!seen.insert(note.unitId).second) continue;
63 const UnitInfo* unit = units.getUnit(note.unitId);
64 if (!unit) continue;
65 source.unitNames.push_back(unit->name);
66 if (!anyUnit) target = unit->targetMixerChannelId;
67 else if (unit->targetMixerChannelId != target) sharedTarget = false;
68 anyUnit = true;
69 }
70 }
71 }
72
73 std::string output;
74 m_headerMeterChannelId = 0;
75 if (anyUnit && sharedTarget && target != MASTER_MIXER_CHANNEL_ID) {
76 if (auto channel = m_trackManager->getChannelById(target)) {
77 m_headerMeterChannelId = target; // the meter shows what this lane plays
78 // Each unit usually plays into its own channel of the same name, so that hop says nothing. The line
79 // names the bus the channel feeds, if it feeds one ("Hi Hat - Crisp → Drum Bus"); a channel straight
80 // to Master names nothing.
81 const uint32_t next = channel->getMainOutputId();
82 if (!source.hasAudio && next != 0xFFFFFFFFu && next != MASTER_MIXER_CHANNEL_ID) {
83 if (auto bus = m_trackManager->getChannelById(next)) output = bus->getName();
84 }
85 }
86 }
87 m_headerSourceLine = TrackHeader::sourceLine(source, output);
88 return m_headerSourceLine;
89}

91void TrackUIComponent::paintHeaderIdentity(AestraUI::NUIRenderer& renderer,
92 const AestraUI::NUIRect& controlAreaBounds) {
93 auto& theme = AestraUI::NUIThemeManager::getInstance();
94 const PlaylistLane* lane = m_trackManager ? m_trackManager->getPlaylistModel().getLane(m_laneId) : nullptr;
95
96 // Colour bar: the track's colour at full strength, 3 px, full row height.
97 if (lane && !m_isLoading) {
98 renderer.fillRect({controlAreaBounds.x, controlAreaBounds.y, 3.0f, controlAreaBounds.height},
99 ClipPaint::identityFromARGB(lane->colorRGBA));
100 }
101
102 // Level meter: 3 px at the header's right edge, rising from the bottom, in the mixer's meter colours.
103 (void)headerSourceLine(); // refreshes m_headerMeterChannelId
104 auto meterChannel = (m_trackManager && m_headerMeterChannelId != 0)
105 ? m_trackManager->getChannelById(m_headerMeterChannelId)
106 : nullptr;
107 if (meterChannel && !meterChannel->isMuted()) {
108 auto snapshots = m_trackManager->getMeterSnapshots();
109 auto slots = m_trackManager->getChannelSlotMapShared();
110 if (snapshots && slots) {
111 const uint32_t slot = slots->getSlotIndex(meterChannel->getChannelId());
112 if (slot != ChannelSlotMap::INVALID_SLOT) {
113 const auto readout = snapshots->readSnapshot(slot);
114 const float level = std::clamp((readout.peakL + readout.peakR) * 0.5f, 0.0f, 1.0f);
115 if (level > 0.001f) {
116 const float visual = std::sqrt(level); // perceptual, as the header's old meter
117 const float span = controlAreaBounds.height - 8.0f;
118 const float h = std::max(1.0f, span * visual);
119 const AestraUI::NUIColor color = visual > 0.9f ? theme.getColor("meterCrit")
120 : visual > 0.7f ? theme.getColor("meterWarn")
121 : theme.getColor("meterSafe");
122 renderer.fillRect({controlAreaBounds.right() - 5.0f, controlAreaBounds.bottom() - 4.0f - h, 3.0f, h},
123 color);
124 }
125 }
126 }
127 }
128
129 // What plays it → where it goes, under the name, when the row is tall enough for two lines.
130 if (m_nameLabel && TrackHeader::showsSourceLine(controlAreaBounds.height)) {
131 const std::string& line = headerSourceLine();
132 if (!line.empty()) {
133 const auto name = m_nameLabel->getBounds();
134 const float fontSize = theme.getFontSize("xs");
135 const float maxWidth = std::max(0.0f, name.width);
136 std::string shown = line;
137 if (renderer.measureText(shown, fontSize).width > maxWidth) {
138 // Trim whole UTF-8 characters until the text and an ellipsis fit.
139 const std::string ellipsis = "\u2026";
140 while (!shown.empty() && renderer.measureText(shown + ellipsis, fontSize).width > maxWidth) {
141 unsigned char removed = 0;
142 do { // one whole character: continuation bytes, then its lead byte
143 removed = static_cast<unsigned char>(shown.back());
144 shown.pop_back();
145 } while (!shown.empty() && (removed & 0xC0) == 0x80);
146 }
147 shown = shown.empty() ? std::string{} : shown + ellipsis;
148 }
149 renderer.drawText(shown, AestraUI::NUIPoint(name.x, name.bottom() + 2.0f), fontSize,
150 theme.getColor("textSecondary").withAlpha(m_selected ? 0.72f : 0.52f));
151 }
152 }
153
154 // V82-T1: a small curve after the name when this track's automation lane is hidden but holds points; a click
155 // on it shows the lane again (TrackUIComponentAutomationRow.cpp).
156 m_hiddenAutomationMarkRect = {};
157 if (m_hasHiddenAutomation && m_nameLabel && lane) {
158 const auto name = m_nameLabel->getBounds();
159 const float textWidth = renderer.measureText(m_nameLabel->getText(), m_nameLabel->getFontSize()).width;
160 const float x = std::min(name.x + textWidth + 9.0f, name.right() - 12.0f);
161 const float cy = std::round(name.y + name.height * 0.5f);
162 const AestraUI::NUIPoint wave[] = {{x, cy + 3.0f}, {x + 3.0f, cy + 3.0f}, {x + 6.0f, cy - 3.0f}, {x + 9.0f, cy - 3.0f}};
163 renderer.drawPolyline(wave, 4, 1.2f, ClipPaint::identityFromARGB(lane->colorRGBA).withAlpha(0.9f));
164 m_hiddenAutomationMarkRect = {x - 3.0f, cy - 8.0f, 15.0f, 16.0f};
165 }
166}
