14void RenderAudioClipCommand::execute() {
15 if (m_executed) {
16 return;
17 }
18 const ClipInstance* clip = m_manager.getPlaylistModel().getClip(m_clipId);
19 if (!clip) {
20 return;
21 }
22 if (!m_originalPatternId.isValid()) {
23 m_originalPatternId = clip->patternId;
24 m_originalEdits = clip->edits;
25 }
26
27 // Redo path: the audio was rendered once already, so put the detached
28 // pattern back instead of writing a second identical file.
29 if (m_detachedPattern) {
30 if (!m_manager.getPatternManager().reinsertPattern(m_detachedPattern)) {
31 return;
32 }
33 } else {
34 ClipInstance clipCopy = *clip;
35 auto rendered = renderBuffer(clipCopy);
36 if (!rendered || !rendered->isValid()) {
37 return;
38 }
39
40 ClipRenderService service(m_manager.getSourceManager(), m_manager.getPatternManager());
41 const auto region = service.resolveClipRegion(clipCopy, m_manager.getPlaylistModel().getProjectSampleRate());
42 const std::string baseName = (region.name.empty() ? clipCopy.name : region.name) + " " + renderSuffix();
43
44 const auto result = service.commit(*rendered, m_manager.renderRootDirectory(), baseName, region.lengthBeats,
45 region.mixerChannelId);
46 if (!result.isValid()) {
47 return;
48 }
49 m_renderedPatternId = result.patternId;
50 }
51
52 if (!m_manager.getPlaylistModel().setClipPattern(m_clipId, m_renderedPatternId)) {
53 // Leave no orphan pattern behind if the repoint failed.
54 m_detachedPattern = m_manager.getPatternManager().detachPattern(m_renderedPatternId);
55 return;
56 }
57
58 ClipEdits edits = m_originalEdits;
59 // The rendered source begins exactly at the region that was resolved, so
60 // the slip that selected that region is already baked in. Leaving it set
61 // would apply the offset a second time against the new full-length source.
62 edits.sourceStart = 0.0;
63 adjustEditsAfterRender(edits);
64 // Compare every field: a subclass may reset anything it baked.
65 if (edits != m_originalEdits) {
66 m_editsChanged = m_manager.getPlaylistModel().setClipEdits(m_clipId, edits);
67 }
68 // Same reasoning for the canonical per-clip offset.
69 if (auto* placed = m_manager.getPlaylistModel().getClip(m_clipId)) {
70 m_originalSourceOffset = placed->sourceOffset;
71 m_originalSourceOffsetSeconds = placed->sourceOffsetSeconds;
72 if (placed->sourceOffset != 0.0 || placed->sourceOffsetSeconds != 0.0) {
73 placed->sourceOffset = 0.0;
74 placed->sourceOffsetSeconds = 0.0;
75 m_offsetsCleared = true;
76 }
77 }
78
79 m_manager.markModified();
80 m_manager.requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged);
81 m_executed = true;
82}

84void RenderAudioClipCommand::undo() {
85 if (!m_executed) {
86 return;
87 }
88 if (!m_manager.getPlaylistModel().setClipPattern(m_clipId, m_originalPatternId)) {
89 return;
90 }
91 if (m_editsChanged) {
92 m_manager.getPlaylistModel().setClipEdits(m_clipId, m_originalEdits);
93 m_editsChanged = false;
94 }
95 if (m_offsetsCleared) {
96 if (auto* restored = m_manager.getPlaylistModel().getClip(m_clipId)) {
97 restored->sourceOffset = m_originalSourceOffset;
98 restored->sourceOffsetSeconds = m_originalSourceOffsetSeconds;
99 }
100 m_offsetsCleared = false;
101 }
102 // Detach rather than remove: redo reinserts this exact pattern, and the
103 // rendered file it points at is still on disk.
104 m_detachedPattern = m_manager.getPatternManager().detachPattern(m_renderedPatternId);
105 m_manager.markModified();
106 m_manager.requestAudioGraphRebuild(GraphDirtyReason::TimelineChanged);
107 m_executed = false;
108}

112std::shared_ptr<AudioBufferData> ReverseAudioClipCommand::renderBuffer(const ClipInstance& clip) {
113 ClipRenderService service(m_manager.getSourceManager(), m_manager.getPatternManager());
114 const auto region = service.resolveClipRegion(clip, m_manager.getPlaylistModel().getProjectSampleRate());
115 if (!region.isValid()) {
116 Log::warning("[ReverseAudioClipCommand] Clip has no resolvable audio to reverse.");
117 return nullptr;
118 }
119
120 auto buffer = ClipRenderService::extractRegion(*region.buffer, region.startFrame, region.frameCount);
121 if (!buffer) {
122 return nullptr;
123 }
124 ClipRenderService::reverseInPlace(*buffer);
125 return buffer;
126}

130std::shared_ptr<AudioBufferData> CommitAudioClipEditsCommand::renderBuffer(const ClipInstance& clip) {
131 ClipRenderService service(m_manager.getSourceManager(), m_manager.getPatternManager());
132 const auto region = service.resolveClipRegion(clip, m_manager.getPlaylistModel().getProjectSampleRate());
133 if (!region.isValid()) {
134 Log::warning("[CommitAudioClipEditsCommand] Clip has no resolvable audio to commit.");
135 return nullptr;
136 }
137
138 auto buffer = ClipRenderService::extractRegion(*region.buffer, region.startFrame, region.frameCount);
139 if (!buffer) {
140 return nullptr;
141 }
142
143 // Fades are authored in beats, so they only become frame counts at the
144 // project tempo the user is hearing them at.
145 const double bpm = std::max(1.0, m_manager.getPlaylistModel().getBPM());
146 const double framesPerBeat = (static_cast<double>(buffer->sampleRate) * 60.0) / bpm;
147 const auto toFrames = [framesPerBeat](float beats) -> uint64_t {
148 if (!std::isfinite(beats) || beats <= 0.0f) {
149 return 0;
150 }
151 const double frames = static_cast<double>(beats) * framesPerBeat;
152 return frames > 0.0 ? static_cast<uint64_t>(frames) : 0;
153 };
154
155 ClipRenderService::applyFades(*buffer, toFrames(clip.edits.fadeInBeats), toFrames(clip.edits.fadeOutBeats));
156
157 const float gain = std::isfinite(clip.edits.gainLinear) ? clip.edits.gainLinear : 1.0f;
158 ClipRenderService::applyGain(*buffer, gain);
159
160 return buffer;
161}

163void CommitAudioClipEditsCommand::adjustEditsAfterRender(ClipEdits& edits) const {
164 // These are now part of the audio; leaving them set would apply them twice.
165 edits.gainLinear = 1.0f;
166 edits.fadeInBeats = 0.0f;
167 edits.fadeOutBeats = 0.0f;
168}
