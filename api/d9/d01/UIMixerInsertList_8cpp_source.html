50void UIMixerInsertList::onRender(NUIRenderer& renderer) {
51 const auto b = getBounds();
52 if (b.isEmpty())
53 return;
54 auto& theme = NUIThemeManager::getInstance();
55 const NUIRect r{std::floor(b.x) + 0.5f, std::floor(b.y) + 0.5f, std::max(1.0f, std::floor(b.width) - 1.0f),
56 std::max(1.0f, std::floor(b.height) - 1.0f)};
57 const NUIColor textPrimary = theme.getColor("textPrimary");
58 const NUIColor textSecondary = theme.getColor("textSecondary");
59 const NUIColor accent = theme.getColor("accentPrimary");
60
61 renderer.fillRoundedRect(r, kRadius, theme.getColor("buttonBgDefault").withAlpha(m_hovered ? 0.98f : 0.7f));
62 renderer.strokeRoundedRect(r, kRadius, 1.0f, theme.getColor("border").withAlpha(m_hovered ? 0.5f : 0.26f));
63
64 if (m_rows.empty()) {
65 renderer.drawTextCentered("+ insert", r, kFont, textSecondary.withAlpha(m_hovered ? 0.95f : 0.62f));
66 return;
67 }
68
69 const int capacity = visibleRowCapacity();
70 const bool overflow = static_cast<int>(m_rows.size()) > capacity;
71 const int shown = overflow ? std::max(1, capacity - 1) : static_cast<int>(m_rows.size());
72
73 float y = r.y + kPad;
74 for (int i = 0; i < shown; ++i) {
75 const Row& row = m_rows[static_cast<size_t>(i)];
76 const float alpha = row.bypassed ? 0.38f : 1.0f;
77 // A dot per insert: lit when it is processing, hollow-dim when bypassed.
78 const NUIRect dot{r.x + 6.0f, y + (kRowH - 5.0f) * 0.5f, 5.0f, 5.0f};
79 renderer.fillRoundedRect(dot, 2.5f, row.bypassed ? textSecondary.withAlpha(0.35f) : accent.withAlpha(0.9f));
80
81 const float textX = dot.right() + 5.0f;
82 float rightInset = 5.0f;
83 std::string status;
84 if (i == 0 && !m_status.empty()) {
85 status = m_status;
86 rightInset += renderer.measureText(status, 8.5f).width + 4.0f;
87 }
88 const float avail = std::max(1.0f, r.right() - textX - rightInset);
89 std::string name = row.name;
90 bool cut = false;
91 while (name.size() > 1 && renderer.measureText(name, kFont).width > avail) {
92 name.pop_back();
93 cut = true;
94 }
95 if (cut)
96 name += "\xE2\x80\xA6";
97 const float textY = y + (kRowH - kFont) * 0.5f - 1.0f;
98 renderer.drawText(name, {textX, textY}, kFont, textPrimary.withAlpha(0.92f * alpha));
99 if (row.bypassed) {
100 const float w = std::min(avail, renderer.measureText(name, kFont).width);
101 renderer.fillRect(NUIRect{textX, y + kRowH * 0.5f, w, 1.0f}, textSecondary.withAlpha(0.45f));
102 }
103 if (!status.empty()) {
104 renderer.drawText(status, {r.right() - rightInset + 4.0f, y + (kRowH - 8.5f) * 0.5f - 1.0f}, 8.5f,
105 accent.withAlpha(0.92f));
106 }
107 y += kRowH;
108 }
109 if (overflow) {
110 const int hidden = static_cast<int>(m_rows.size()) - shown;
111 renderer.drawText("+" + std::to_string(hidden) + " more",
112 {r.x + 6.0f, y + (kRowH - kFont) * 0.5f - 1.0f}, kFont, textSecondary.withAlpha(0.7f));
113 }
114}

116bool UIMixerInsertList::onMouseEvent(const NUIMouseEvent& event) {
117 if (!isVisible() || !isEnabled())
118 return false;
119 const auto b = getBounds();
120 const bool inside = b.contains(event.position);
121 if (event.button == NUIMouseButton::None) {
122 if (inside != m_hovered) {
123 m_hovered = inside;
124 if (m_hovered && m_rows.empty())
125 NUIComponent::showRemoteTooltip("Add insert", {b.center().x, b.bottom() + 8.0f}, this);
126 else if (!m_hovered)
127 NUIComponent::hideRemoteTooltip(this);
128 repaint();
129 }
130 return false;
131 }
132 if (event.pressed && event.button == NUIMouseButton::Left && inside) {
133 m_pressed = true;
134 return true;
135 }
136 if (event.released && event.button == NUIMouseButton::Left) {
137 const bool fire = m_pressed && inside;
138 m_pressed = false;
139 if (fire && onClicked)
140 onClicked();
141 return fire;
142 }
143 return false;
144}
