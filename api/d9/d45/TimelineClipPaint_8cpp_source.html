19AestraUI::NUIColor identityFromARGB(uint32_t argb) {
20 const float a = ((argb >> 24) & 0xFF) / 255.0f;
21 const AestraUI::NUIColor color(((argb >> 16) & 0xFF) / 255.0f, ((argb >> 8) & 0xFF) / 255.0f,
22 (argb & 0xFF) / 255.0f, a > 0.0f ? a : 1.0f);
23 const float mx = std::max({color.r, color.g, color.b});
24 const float mn = std::min({color.r, color.g, color.b});
25 if (mx - mn < 0.06f) {
26 return AestraUI::NUIThemeManager::getInstance().getColor("accentPrimary").withAlpha(1.0f);
27 }
28 return color.withAlpha(1.0f);
29}

64void paintMenuGlyph(AestraUI::NUIRenderer& renderer, const AestraUI::NUIRect& slot, bool hot, bool pointed) {
65 if (slot.width <= 0.0f) return;
66 if (pointed) renderer.fillRect(slot, kCanvas.withAlpha(0.18f));
67 // Three 1 px lines, 7 px wide, 3 px apart, snapped to whole pixels and centred in the slot: crisp at every
68 // band height, never touching the band's edges.
69 const float cx = std::round(slot.x + slot.width * 0.5f);
70 const float cy = std::round(slot.y + slot.height * 0.5f);
71 const AestraUI::NUIColor ink = headerLabelInk(pointed ? 0.95f : (hot ? 0.8f : 0.55f));
72 for (int i = -1; i <= 1; ++i) {
73 renderer.fillRect({cx - 3.0f, cy + static_cast<float>(i) * 3.0f - 0.5f, 7.0f, 1.0f}, ink);
74 }
75}

86void paintMiniPianoRoll(AestraUI::NUIRenderer& renderer, const AestraUI::NUIRect& visible,
87 const AestraUI::NUIRect& full, float bandHeight, float pixelsPerBeat,
88 const std::vector<MidiNote>& notes, double patternLengthBeats, double sourceOffsetBeats,
89 double clipDurationBeats, const AestraUI::NUIColor& ink) {
90 if (!(pixelsPerBeat > 0.0f)) return;
91 const float noteAreaY = full.y + bandHeight + 3.0f;
92 const float noteAreaHeight = std::max(4.0f, full.bottom() - 3.0f - noteAreaY);
93 int minPitch = 127;
94 int maxPitch = 0;
95 double reach = std::max(0.0, patternLengthBeats);
96 for (const auto& n : notes) {
97 minPitch = std::min(minPitch, n.pitch);
98 maxPitch = std::max(maxPitch, n.pitch);
99 reach = std::max(reach, n.startBeat + n.durationBeats);
100 }
101 const int pitchRange = std::max(1, maxPitch - minPitch);
102 const float rowHeight = std::clamp(noteAreaHeight / static_cast<float>(pitchRange + 1), 1.0f, 2.0f);
103 const double ppb = pixelsPerBeat;
104
105 // Clip-relative beats of the on-screen part; the pattern starts at -sourceOffset in the same frame.
106 const double fromBeat = (visible.x - full.x) / ppb;
107 const double toBeat = (visible.right() - full.x) / ppb;
108 const bool repeating = patternLengthBeats > 0.0;
109 const PatternRepeat::Range repeats =
110 PatternRepeat::inWindow(patternLengthBeats, sourceOffsetBeats, sourceOffsetBeats + clipDurationBeats,
111 -sourceOffsetBeats, fromBeat, toBeat, reach);
112 for (int64_t r = repeats.first; r <= repeats.last; ++r) {
113 const double repeatOffset = repeating ? static_cast<double>(r) * patternLengthBeats : 0.0;
114 for (const auto& n : notes) {
115 if (repeating && n.startBeat >= patternLengthBeats) continue; // wraps before it: never plays
116 const double inClip = repeatOffset + noteBeatInClip(n.startBeat, sourceOffsetBeats);
117 if (inClip + n.durationBeats <= 0.0 || inClip >= clipDurationBeats) continue;
118 const double start = std::max(0.0, inClip);
119 const double end = std::min(clipDurationBeats, inClip + n.durationBeats);
120 float x = full.x + static_cast<float>(start * ppb);
121 float right = full.x + static_cast<float>(end * ppb) - 1.0f;
122 x = std::max(x, visible.x + 1.0f);
123 right = std::min(right, visible.right());
124 if (right <= x) right = x + 1.0f;
125 if (x >= visible.right()) continue;
126 const float t = maxPitch > minPitch ? static_cast<float>(n.pitch - minPitch) / pitchRange : 0.5f;
127 const float y = noteAreaY + (noteAreaHeight - rowHeight) * (1.0f - t);
128 renderer.fillRect(AestraUI::NUIRect(x, y, right - x, rowHeight), ink.withAlpha(0.95f));
129 }
130 }
131
132 // Lines closer than a few pixels would fill the clip with dashes; the notes already show the rhythm then.
133 constexpr double kMinLineSpacingPx = 4.0;
134 if (!repeating || patternLengthBeats * ppb < kMinLineSpacingPx) return;
135 for (double beat : repeatLinesInClip(patternLengthBeats, sourceOffsetBeats, clipDurationBeats, fromBeat, toBeat)) {
136 const float x = full.x + static_cast<float>(beat * ppb);
137 if (x > visible.x && x < visible.right()) {
138 paintRepeatLine(renderer, std::round(x) + 0.5f, visible.y + bandHeight, visible.bottom(), ink.withAlpha(0.55f));
139 }
140 }
141}
