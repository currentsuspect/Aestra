79 void applyPreset(QualityPreset p) {
80 preset = p;
81 if (p == QualityPreset::Economy) {
82 resampling = ResamplingMode::Fast;
83 dithering = DitheringMode::None;
84 } else if (p == QualityPreset::Balanced) {
85 resampling = ResamplingMode::Medium;
86 dithering = DitheringMode::Triangular;
87 } else if (p == QualityPreset::HighFidelity) {
88 resampling = ResamplingMode::High;
89 // TODO: Implement high-pass dither in live path. Currently aliases to Triangular.
90 dithering = DitheringMode::Triangular;
91 } else if (p == QualityPreset::Mastering) {
92 resampling = ResamplingMode::Perfect;
93 // TODO: Implement noise-shaped dither in live path. Currently aliases to Triangular.
94 dithering = DitheringMode::Triangular;
95 }
96 }
