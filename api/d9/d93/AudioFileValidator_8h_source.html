100 static bool validateFileHeader(const std::string& path) {
101 std::ifstream file(path, std::ios::binary);
102 if (!file.is_open()) {
103 return false;
104 }
105
106 // Read first 16 bytes for header detection (need up to 16 for ASF GUID)
107 uint8_t header[16] = {0};
108 file.read(reinterpret_cast<char*>(header), sizeof(header));
109
110 // Basic check - need at least 4 bytes for any header detection
111 if (file.gcount() < 4) {
112 return false; // File too small
113 }
114
115 // WAV: RIFF....WAVE (requires 12 bytes for header+8 check)
116 if (file.gcount() >= 12 && memcmp(header, "RIFF", 4) == 0 && memcmp(header + 8, "WAVE", 4) == 0) {
117 return true;
118 }
119
120 // MP3: ID3 tag or frame sync (requires 3-4 bytes)
121 if (file.gcount() >= 3 && memcmp(header, "ID3", 3) == 0) {
122 return true; // ID3v2 tag
123 }
124 if (file.gcount() >= 2 && header[0] == 0xFF && (header[1] & 0xE0) == 0xE0) {
125 return true; // MP3 frame sync
126 }
127
128 // FLAC: fLaC (requires 4 bytes)
129 if (file.gcount() >= 4 && memcmp(header, "fLaC", 4) == 0) {
130 return true;
131 }
132
133 // OGG: OggS (requires 4 bytes)
134 if (file.gcount() >= 4 && memcmp(header, "OggS", 4) == 0) {
135 return true;
136 }
137
138 // AIFF: FORM....AIFF or FORM....AIFC (requires 12 bytes for header+8 check)
139 if (file.gcount() >= 12 && memcmp(header, "FORM", 4) == 0) {
140 if (memcmp(header + 8, "AIFF", 4) == 0 || memcmp(header + 8, "AIFC", 4) == 0) {
141 return true;
142 }
143 }
144
145 // M4A/MP4/AAC: ftyp (requires 8 bytes for header+4 check)
146 if (file.gcount() >= 8 && memcmp(header + 4, "ftyp", 4) == 0) {
147 return true;
148 }
149
150 // WMA/ASF: ASF header GUID (requires full 16 bytes)
151 static const uint8_t asfGuid[16] = {0x30, 0x26, 0xB2, 0x75, 0x8E, 0x66, 0xCF, 0x11,
152 0xA6, 0xD9, 0x00, 0xAA, 0x00, 0x62, 0xCE, 0x6C};
153 if (file.gcount() >= 16 && memcmp(header, asfGuid, 16) == 0) { // Check full 16-byte GUID
154 return true;
155 }
156
157 return false;
158 }

165 static std::string getRejectionReason(const std::string& path) {
166 if (!hasValidAudioExtension(path)) {
167 std::string ext = getExtension(path);
168 if (ext.empty()) {
