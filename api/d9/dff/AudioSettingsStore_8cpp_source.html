19// The complete key set. Both the parser and the serializer drive off this, so a

39std::string trim(const std::string& s) {
40 const auto first = s.find_first_not_of(" \t\r\n");
41 if (first == std::string::npos) {
42 return {};
43 }
44 const auto last = s.find_last_not_of(" \t\r\n");
45 return s.substr(first, last - first + 1);
46}

50AudioSettings parseAudioSettings(std::istream& in) {
51 AudioSettings settings;
52
53 std::string line;
54 while (std::getline(in, line)) {
55 const auto eq = line.find('=');
56 if (eq == std::string::npos) {
57 continue;
58 }
59
60 const std::string key = trim(line.substr(0, eq));
61 const std::string value = trim(line.substr(eq + 1));
62 if (key.empty() || value.empty()) {
63 continue;
64 }
65
66 for (const auto& binding : kBindings) {
67 if (key != binding.key) {
68 continue;
69 }
70 // A malformed value leaves the field UNSET rather than defaulting.
71 // Defaulting here would be indistinguishable, downstream, from the
72 // user having chosen the default.
73 try {
74 size_t consumed = 0;
75 const int parsed = std::stoi(value, &consumed);
76 if (consumed == value.size()) {
77 settings.*(binding.field) = parsed;
78 }
79 } catch (const std::exception&) {
80 // leave unset
81 }
82 break;
83 }
84 }
85
86 return settings;
87}

89void serializeAudioSettings(std::ostream& out, const AudioSettings& settings) {
90 for (const auto& binding : kBindings) {
91 const int value = settings.*(binding.field);
92 if (value != AudioSettings::kUnset) {
93 out << binding.key << '=' << value << '\n';
94 }
95 }
96}

98std::string audioSettingsConfigPath() {
99 std::error_code ec;
100 if (auto* utils = Aestra::Platform::getUtils()) {
101 std::filesystem::path appDataDir(utils->getAppDataPath("Aestra"));
102 if (!appDataDir.empty()) {
103 std::filesystem::create_directories(appDataDir, ec);
104 if (!ec) {
105 std::filesystem::path appDataPath = appDataDir / "audio_settings.conf";
106 if (std::filesystem::exists(appDataPath, ec)) {
107 return appDataPath.string();
108 }
109 }
110 }
111 }
112
113 // Legacy location: a config written before the app-data move.
114 std::filesystem::path legacyPath = std::filesystem::current_path() / "audio_settings.conf";
115 if (std::filesystem::exists(legacyPath, ec)) {
116 return legacyPath.string();
117 }
118
119 if (auto* utils = Aestra::Platform::getUtils()) {
120 std::filesystem::path appDataDir(utils->getAppDataPath("Aestra"));
121 if (!appDataDir.empty()) {
122 return (appDataDir / "audio_settings.conf").string();
123 }
124 }
125 return legacyPath.string();
126}

128AudioSettings loadAudioSettings() {
129 std::ifstream file(audioSettingsConfigPath());
130 if (!file.is_open()) {
131 return {}; // all unset — nothing persisted, so nothing to apply
132 }
133 return parseAudioSettings(file);
