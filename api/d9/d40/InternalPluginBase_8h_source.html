85 size_t rows = 0;

282 std::vector<uint8_t> saveState() const override {
283 const uint32_t count = paramSpecCount();
284 std::vector<uint8_t> blob(keyedBlobSize(count));
285 writeHeader(blob.data(), stateMagic(), kStateVersion);
286 const uint32_t entries = count;
287 std::memcpy(blob.data() + sizeof(uint32_t) * 2, &entries, sizeof(entries));
288 uint8_t* cursor = blob.data() + kKeyedHeaderSize;
289 for (const ParamSpec* p = specsBegin(); p != specsEnd(); ++p) {
290 const ParamSpec& spec = *p;
291 const uint32_t id = spec.id;
292 std::memcpy(cursor, &id, sizeof(id));
293 cursor += sizeof(id);
294 const float value = m_params[spec.id].load(std::memory_order_relaxed);
295 std::memcpy(cursor, &value, sizeof(value));
296 cursor += sizeof(value);
297 }
298 return blob;
299 }
332 bool loadState(const std::vector<uint8_t>& state) override {
333 // Header first, and the version check BEFORE the count is read. A v1 blob
334 // is 8 + 4N bytes; read as v2, its first parameter's float bits would be
335 // parsed as the entry count and the load would look successful while
336 // restoring entirely the wrong values. That is the failure this format
337 // exists to prevent, so it has to be impossible rather than unlikely.
338 if (state.size() < kKeyedHeaderSize)
339 return false;
340
341 if (!headerMatches(state.data(), stateMagic(), kStateVersion))
342 return false;
343
344 uint32_t count = 0;
345 std::memcpy(&count, state.data() + sizeof(uint32_t) * 2, sizeof(count));
346 if (count > kMaxSpecParams)
347 return false;
348 // Zero entries for a plugin that HAS parameters is not a shape any writer
349 // produces. It is exactly what a v1 blob whose first parameter is 0.0f
350 // looks like when misread: 0.0f's bits are 0x00000000, so the "count"
351 // parses as 0, which is inside kMaxSpecParams, and any blob of 12+ bytes
352 // satisfies keyedBlobSize(0). Every other guard passes. Refusing it here
353 // means the invariant does not depend SOLELY on the version check
354 // happening to run first.
355 if (count == 0 && paramSpecCount() > 0)
356 return false;
357 if (state.size() < keyedBlobSize(count))
358 return false;
359
360 // Pass 1: classify. Nothing here touches a parameter, so a rejection
361 // anywhere in this loop leaves the instance exactly as it was.
362 std::vector<std::pair<uint32_t, float>> entries;
363 entries.reserve(count);
364 const uint8_t* cursor = state.data() + kKeyedHeaderSize;
365 for (uint32_t i = 0; i < count; ++i) {
366 uint32_t id = 0;
367 float value = 0.0f;
368 std::memcpy(&id, cursor, sizeof(id));
369 cursor += sizeof(id);
370 std::memcpy(&value, cursor, sizeof(value));
371 cursor += sizeof(value);
372 if (id >= kMaxSpecParams)
373 continue; // unknown to this table: a removed parameter, or junk
374 const ParamSpec* spec = findSpec(id);
375 if (!spec || spec->isReadOnly)
376 continue;
377 if (!std::isfinite(value))
378 return false;
379 entries.emplace_back(id, value);
380 }
381
382 // Pass 2: apply.
383 for (const auto& entry : entries) {
384 const ParamSpec* spec = findSpec(entry.first);
385 if (!spec)
386 continue;
387 const float value = std::clamp(entry.second, spec->minValue, spec->maxValue);
388 m_params[entry.first].store(value, std::memory_order_relaxed);
389 onParameterChanged(entry.first, value);
390 }
391 return true;
392 }
