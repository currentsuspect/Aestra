77 // Edge commands carry transitions the audio thread cannot reconstruct from
78 // persistent state (play/stop/seek, count-in start/stop). They must not
79 // disappear silently; state commands (volume/pan/mute/solo) may drop, with
80 // the engine converging on the next update. See #913.
81 static constexpr bool isEdgeCommand(AudioQueueCommandType type) noexcept {
82 return type == AudioQueueCommandType::SetTransportState ||
83 type == AudioQueueCommandType::SetPlaybackContext ||
84 type == AudioQueueCommandType::MetronomeCountInStart ||
85 type == AudioQueueCommandType::MetronomeCountInStop;
86 }
87
88 bool push(const AudioQueueCommand& cmd) {
89 const bool ok = m_queue.push(cmd);
90 if (!ok) {
91 // Drop-newest policy: keep audio thread deterministic; UI can observe drops via telemetry.
92 m_dropped.fetch_add(1, std::memory_order_relaxed);
93 return false;
94 }
95
96 const uint32_t depth = static_cast<uint32_t>(m_queue.size());
97 uint32_t prev = m_maxDepth.load(std::memory_order_relaxed);
98 while (depth > prev &&
99 !m_maxDepth.compare_exchange_weak(prev, depth, std::memory_order_relaxed, std::memory_order_relaxed)) {}
100 return true;
101 }
102
103 // Bounded reliable delivery for edge commands. Non-RT producers only: spins
104 // up to timeoutMs waiting for space, so a transient flood delays the edge
105 // instead of losing it. True means accepted into the FIFO (ordered drain
106 // still applies); false means dropped after the deadline and counted in
107 // both droppedCount() and edgeDroppedCount() — callers must fail loudly.
108 // The 200 ms default covers draining a fully flooded queue at 16 cmds per
109 // ~10 ms block; it only ever engages when the queue is pathologically
110 // full, never on the hot path. Never call from the audio thread: misuse
111 // degrades to a single non-blocking attempt.
112 bool pushReliable(const AudioQueueCommand& cmd, uint32_t timeoutMs = 200) {
113 if (reportRealtimeMisuse("AudioCommandQueue::pushReliable")) {
114 if (m_queue.push(cmd)) {
115 return true;
116 }
117 m_dropped.fetch_add(1, std::memory_order_relaxed);
118 m_edgeDropped.fetch_add(1, std::memory_order_relaxed);
119 return false;
120 }
121 const auto deadline = std::chrono::steady_clock::now() + std::chrono::milliseconds(timeoutMs);
122 for (;;) {
123 if (push(cmd)) {
124 return true;
125 }
126 // push() already counted this attempt in m_dropped; rewind it so a
127 // retried edge is counted exactly once, on final failure below.
128 m_dropped.fetch_sub(1, std::memory_order_relaxed);
129 if (std::chrono::steady_clock::now() >= deadline) {
130 break;
131 }
132 std::this_thread::sleep_for(std::chrono::milliseconds(1));
133 }
134 m_dropped.fetch_add(1, std::memory_order_relaxed);
135 m_edgeDropped.fetch_add(1, std::memory_order_relaxed);
136 return false;
137 }
138
139 bool pop(AudioQueueCommand& outCmd) { return m_queue.pop(outCmd); }
140
141 bool empty() const { return m_queue.isEmpty(); }
142
143 uint32_t approxDepth() const noexcept { return static_cast<uint32_t>(m_queue.size()); }
144
145 uint32_t maxDepth() const noexcept { return m_maxDepth.load(std::memory_order_relaxed); }
146
147 uint64_t droppedCount() const noexcept { return m_dropped.load(std::memory_order_relaxed); }
148
149 uint64_t edgeDroppedCount() const noexcept { return m_edgeDropped.load(std::memory_order_relaxed); }
150
151 static constexpr uint32_t capacity() noexcept {
152 return static_cast<uint32_t>(Aestra::LockFreeRingBuffer<AudioQueueCommand, kQueueCapacity>::capacity());
153 }
154
155private:
156 Aestra::LockFreeRingBuffer<AudioQueueCommand, kQueueCapacity> m_queue;
157 std::atomic<uint64_t> m_dropped{0};
158 std::atomic<uint64_t> m_edgeDropped{0};
159 std::atomic<uint32_t> m_maxDepth{0};
160};
