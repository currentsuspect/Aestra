49class AudioCommandQueue {
50public:
51 static constexpr size_t kQueueCapacity = 1024;
52
53 bool push(const AudioQueueCommand& cmd) {
54 const bool ok = m_queue.push(cmd);
55 if (!ok) {
56 // Drop-newest policy: keep audio thread deterministic; UI can observe drops via telemetry.
57 m_dropped.fetch_add(1, std::memory_order_relaxed);
58 return false;
59 }
60
61 const uint32_t depth = static_cast<uint32_t>(m_queue.size());
62 uint32_t prev = m_maxDepth.load(std::memory_order_relaxed);
63 while (depth > prev &&
64 !m_maxDepth.compare_exchange_weak(prev, depth, std::memory_order_relaxed, std::memory_order_relaxed)) {}
65 return true;
66 }
67
68 bool pop(AudioQueueCommand& outCmd) { return m_queue.pop(outCmd); }
69
70 bool empty() const { return m_queue.isEmpty(); }
71
72 uint32_t approxDepth() const noexcept { return static_cast<uint32_t>(m_queue.size()); }
73
74 uint32_t maxDepth() const noexcept { return m_maxDepth.load(std::memory_order_relaxed); }
75
76 uint64_t droppedCount() const noexcept { return m_dropped.load(std::memory_order_relaxed); }
77
78 static constexpr uint32_t capacity() noexcept {
79 return static_cast<uint32_t>(Aestra::LockFreeRingBuffer<AudioQueueCommand, kQueueCapacity>::capacity());
80 }
81
82private:
83 Aestra::LockFreeRingBuffer<AudioQueueCommand, kQueueCapacity> m_queue;
84 std::atomic<uint64_t> m_dropped{0};
85 std::atomic<uint32_t> m_maxDepth{0};
86};
