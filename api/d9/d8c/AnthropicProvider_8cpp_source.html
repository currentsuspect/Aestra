9ModelResponse AnthropicProvider::complete(const ModelRequest& request) {
10 ModelResponse response;
11
12 JSON body = JSON::object();
13 body.set("model", JSON(request.model.empty() ? m_model : request.model));
14 body.set("max_tokens", JSON(static_cast<double>(request.maxTokens)));
15 if (!request.systemPrompt.empty()) {
16 body.set("system", JSON(request.systemPrompt));
17 }
18
19 if (!request.tools.empty()) {
20 JSON tools = JSON::array();
21 for (const auto& tool : request.tools) {
22 JSON entry = JSON::object();
23 entry.set("name", JSON(tool.name));
24 entry.set("description", JSON(tool.description));
25 entry.set("input_schema", tool.parametersSchema);
26 tools.push(entry);
27 }
28 body.set("tools", tools);
29 }
30
31 // Consecutive tool messages must land in ONE user message of tool_result
32 // blocks — splitting them degrades the model's parallel tool use.
33 JSON messages = JSON::array();
34 size_t i = 0;
35 while (i < request.messages.size()) {
36 const Message& message = request.messages[i];
37 if (message.role == "tool") {
38 JSON content = JSON::array();
39 while (i < request.messages.size() && request.messages[i].role == "tool") {
40 const Message& result = request.messages[i];
41 JSON block = JSON::object();
42 block.set("type", JSON("tool_result"));
43 block.set("tool_use_id", JSON(result.toolCallId));
44 block.set("content", JSON(result.text));
45 if (result.isError) block.set("is_error", JSON(true));
46 content.push(block);
47 ++i;
48 }
49 JSON entry = JSON::object();
50 entry.set("role", JSON("user"));
51 entry.set("content", content);
52 messages.push(entry);
53 continue;
54 }
55
56 JSON entry = JSON::object();
57 entry.set("role", JSON(message.role));
58 if (message.role == "assistant" && !message.toolCalls.empty()) {
59 JSON content = JSON::array();
60 if (!message.text.empty()) {
61 JSON textBlock = JSON::object();
62 textBlock.set("type", JSON("text"));
63 textBlock.set("text", JSON(message.text));
64 content.push(textBlock);
65 }
66 for (const auto& call : message.toolCalls) {
67 JSON toolUse = JSON::object();
68 toolUse.set("type", JSON("tool_use"));
69 toolUse.set("id", JSON(call.id));
70 toolUse.set("name", JSON(call.name));
71 toolUse.set("input", call.arguments);
72 content.push(toolUse);
73 }
74 entry.set("content", content);
75 } else {
76 entry.set("content", JSON(message.text));
77 }
78 messages.push(entry);
79 ++i;
80 }
81 body.set("messages", messages);
82
83 const HttpResponse http = httpPostJson(
84 m_baseUrl + "/v1/messages",
85 {{"x-api-key", m_apiKey}, {"anthropic-version", "2023-06-01"}}, body.toString(),
86 request.timeoutSeconds);
87
88 if (!http.error.empty()) {
89 response.error = "transport: " + http.error;
90 return response;
91 }
92
93 JSON parsed;
94 try {
95 parsed = JSON::parse(http.body);
96 } catch (const std::exception& e) {
97 response.error = "unparseable response: " + std::string(e.what());
98 return response;
99 }
100
101 if (http.status != 200) {
102 std::string detail = http.body;
103 if (parsed.has("error") && parsed["error"].has("message") &&
104 parsed["error"]["message"].isString()) {
105 detail = parsed["error"]["message"].asString();
106 }
107 response.error = "HTTP " + std::to_string(http.status) + ": " + detail;
108 return response;
109 }
110
111 // A 200 with missing or mistyped fields stays inside the error contract —
112 // nothing below may throw past complete().
113 if (parsed.has("content")) {
114 JSON& content = parsed["content"];
115 for (size_t blockIndex = 0; blockIndex < content.size(); ++blockIndex) {
116 JSON& block = content[blockIndex];
117 const std::string type = block.has("type") && block["type"].isString()
118 ? block["type"].asString()
119 : "";
120 if (type == "text") {
121 if (!block.has("text") || !block["text"].isString()) {
122 response.error = "malformed text block (missing string \"text\")";
123 response.stopReason = StopReason::Error;
124 return response;
125 }
126 response.text += block["text"].asString();
127 } else if (type == "tool_use") {
128 if (!block.has("id") || !block["id"].isString() || !block.has("name") ||
129 !block["name"].isString() || !block.has("input")) {
130 response.error =
131 "malformed tool_use block (needs string id, string name, input)";
132 response.stopReason = StopReason::Error;
133 return response;
134 }
135 ToolCall call;
136 call.id = block["id"].asString();
137 call.name = block["name"].asString();
138 call.arguments = block["input"];
139 response.toolCalls.push_back(call);
140 }
141 }
142 }
143
144 if (parsed.has("usage")) {
145 JSON& usage = parsed["usage"];
146 if (usage.has("input_tokens") && usage["input_tokens"].isNumber())
147 response.usage.inputTokens = static_cast<long long>(usage["input_tokens"].asNumber());
148 if (usage.has("output_tokens") && usage["output_tokens"].isNumber())
149 response.usage.outputTokens =
150 static_cast<long long>(usage["output_tokens"].asNumber());
151 }
152
153 const std::string stop = parsed.has("stop_reason") && parsed["stop_reason"].isString()
154 ? parsed["stop_reason"].asString()
155 : "end_turn";
156 if (stop == "tool_use") {
157 response.stopReason = StopReason::ToolUse;
158 } else if (stop == "max_tokens") {
159 response.stopReason = StopReason::MaxTokens;
160 } else if (stop == "refusal") {
161 response.stopReason = StopReason::Refusal;
162 } else {
163 response.stopReason = StopReason::EndTurn;
164 }
165 return response;
166}
