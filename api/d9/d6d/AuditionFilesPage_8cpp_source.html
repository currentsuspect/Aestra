22void AuditionFilesPage::setProjectDir(const std::string& dir) {
23 if (dir == m_projectDir && !m_places.empty()) return;
24 m_projectDir = dir;
25 m_places = AuditionFiles::places(dir, m_home);

42AuditionFilesPage::Layout AuditionFilesPage::layout() const {
43 Layout l;
44 const NUIRect b = getBounds();
45 l.header = {b.x + 24.0f, b.y + 24.0f, b.width - 48.0f, 32.0f};
46 l.up = {b.right() - 24.0f - 56.0f, b.y + 56.0f, 56.0f, 24.0f};
47 l.path = {b.x + 24.0f, b.y + 56.0f, b.width - 48.0f - 64.0f, 24.0f};
48 // Places wrap in rows of chips under the path.
49 float x = b.x + 24.0f, y = b.y + 92.0f;
50 for (const auto& p : m_places) {
51 const float w = static_cast<float>(p.name.size()) * 7.0f + 24.0f;
52 if (x + w > b.right() - 24.0f && x > b.x + 24.0f) { x = b.x + 24.0f; y += 32.0f; }
53 l.places.push_back({x, y, w, 24.0f});
54 x += w + 8.0f;
55 }
56 l.placesArea = {b.x + 24.0f, b.y + 92.0f, b.width - 48.0f, y + 24.0f - (b.y + 92.0f)};
57 const float top = l.placesArea.bottom() + 16.0f;
58 l.list = {b.x, top, b.width, std::max(0.0f, b.bottom() - top)};
59 return l;
60}

62int AuditionFilesPage::rowAt(const NUIPoint& p, const Layout& l) const {
63 if (!l.list.contains(p)) return -1;
64 const int row = static_cast<int>((p.y - l.list.y + m_scroll) / rowHeight());
65 return row >= 0 && row < static_cast<int>(m_entries.size()) ? row : -1;
66}

83void AuditionFilesPage::onRender(NUIRenderer& r) {
84 if (!isVisible()) return;
85 auto& theme = NUIThemeManager::getInstance();
86 const NUIColor text = theme.getColor("textPrimary"), muted = theme.getColor("textSecondary"), rule = theme.getColor("borderSubtle");
87 const NUIRect b = getBounds();
88 const Layout l = layout();
89 r.fillRect(b, theme.getColor("backgroundPrimary"));
90 r.fillRect({b.right() - 1.0f, b.y, 1.0f, b.height}, rule);
91
92 r.drawText("Files", {l.header.x, l.header.y}, 20.0f, text);
93 // The path, from the right so the folder you are in is always the part you can read.
94 std::string path = m_dir;
95 float pw = r.measureText(path, 11.0f).width;
96 while (pw > l.path.width && path.size() > 4) { path.erase(0, 4); pw = r.measureText("..." + path, 11.0f).width; if (pw <= l.path.width) { path = "..." + path; break; } }
97 r.drawText(path, {l.path.x, std::round(r.calculateTextY(l.path, 11.0f))}, 11.0f, muted);
98 const bool upHot = m_hoverControl == "up";
99 r.fillRect(l.up, upHot ? theme.getColor("surfaceRaised") : theme.getColor("surfaceTertiary"));
100 const float uw = r.measureText("Up", 11.0f).width;
101 r.drawText("Up", {l.up.x + (l.up.width - uw) * 0.5f, std::round(r.calculateTextY(l.up, 11.0f))}, 11.0f, text);
102
103 for (size_t i = 0; i < l.places.size(); ++i) {
104 const bool here = m_places[i].path == m_dir;
105 const bool hot = m_hoverControl == "place" + std::to_string(i);
106 r.fillRect(l.places[i], here ? theme.getColor("accentPrimary").withAlpha(0.28f) : hot ? theme.getColor("surfaceRaised") : theme.getColor("surfaceTertiary"));
107 const float tw = r.measureText(m_places[i].name, 11.0f).width;
108 r.drawText(m_places[i].name, {l.places[i].x + (l.places[i].width - tw) * 0.5f, std::round(r.calculateTextY(l.places[i], 11.0f))}, 11.0f, here ? text : muted);
109 }
110
111 r.fillRect({l.list.x, l.list.y - 1.0f, l.list.width, 1.0f}, rule);
112 r.setClipRect(l.list);
113 for (size_t i = 0; i < m_entries.size(); ++i) {
114 const NUIRect row{l.list.x, l.list.y + static_cast<float>(i) * rowHeight() - m_scroll, l.list.width, rowHeight()};
115 if (row.bottom() < l.list.y || row.y > l.list.bottom()) continue;
116 const auto& e = m_entries[i];
117 const bool hot = static_cast<int>(i) == m_hover;
118 if (static_cast<int>(i) == m_selected) r.fillRect(row, theme.getColor("surfaceRaised"));
119 else if (hot) r.fillRect(row, theme.getColor("surfaceTertiary"));
120 // A folder is an outline, a sound is a small waveform: the two read apart at a glance.
121 const float cy = row.y + row.height * 0.5f;
122 if (e.folder) {
123 r.strokeRect({row.x + 24.0f, cy - 6.0f, 16.0f, 12.0f}, 1.2f, muted);
124 r.fillRect({row.x + 24.0f, cy - 8.0f, 7.0f, 3.0f}, muted);
125 } else {
126 static const float bars[] = {4, 9, 6, 12, 8, 5, 10, 4};
127 for (int k = 0; k < 8; ++k) r.fillRect({row.x + 24.0f + static_cast<float>(k) * 2.2f, cy - bars[k] * 0.5f, 1.4f, bars[k]}, theme.getColor("accentPrimary").withAlpha(0.85f));
128 }
129 std::string name = e.name;
130 const float maxName = row.width - 24.0f - 56.0f - 24.0f - (hot && !e.folder ? 150.0f : 74.0f);
131 while (name.size() > 4 && r.measureText(name, 12.5f).width > maxName) name.pop_back();
132 if (name != e.name) name += "...";
133 r.drawText(name, {row.x + 56.0f, std::round(r.calculateTextY(row, 12.5f))}, 12.5f, e.folder ? text : text.withAlpha(0.95f));
134 if (e.folder) continue;
135 if (hot) {
136 const NUIRect play = playButton(row), ref = refButton(row);
137 for (const auto& btn : {std::make_pair(play, std::string("Play")), std::make_pair(ref, std::string("+ Reference"))}) {
138 r.fillRect(btn.first, theme.getColor("backgroundSecondary"));
139 r.strokeRect(btn.first, 1.0f, rule);
140 const float tw = r.measureText(btn.second, 10.5f).width;
141 r.drawText(btn.second, {btn.first.x + (btn.first.width - tw) * 0.5f, std::round(r.calculateTextY(btn.first, 10.5f))}, 10.5f, muted);
142 }
143 } else {
144 const std::string size = AuditionFiles::sizeLabel(e.bytes);
145 const float sw = r.measureText(size, 10.5f).width;
146 r.drawText(size, {row.right() - 24.0f - sw, std::round(r.calculateTextY(row, 10.5f))}, 10.5f, muted.withAlpha(0.7f));
147 }
148 }
149 if (m_entries.empty()) r.drawTextCentered("No audio or folders here", l.list, 12.0f, muted.withAlpha(0.7f));
150 r.clearClipRect();
151}

153bool AuditionFilesPage::onMouseEvent(const NUIMouseEvent& event) {
154 if (!isVisible()) return false;
155 auto& drag = NUIDragDropManager::getInstance();
156 const Layout l = layout();
157 const NUIPoint p = event.position;
158
159 if (m_dragging) { // the drag belongs to this page until the button goes up
160 if (drag.isDragging()) drag.updateDrag(p);
161 if (!event.pressed && event.button == NUIMouseButton::Left && event.type != NUIMouseEventType::Move) {
162 drag.endDrag(p);
163 m_dragging = false;
164 m_pressed = -1;
165 }
166 return true;
167 }
168 if (!getBounds().contains(p)) return false;
169
170 if (event.type == NUIMouseEventType::Move) {
171 const int row = rowAt(p, l);
172 std::string control;
173 if (l.up.contains(p)) control = "up";
174 for (size_t i = 0; i < l.places.size(); ++i) if (l.places[i].contains(p)) control = "place" + std::to_string(i);
175 if (row != m_hover || control != m_hoverControl) { m_hover = row; m_hoverControl = control; setDirty(true); }
176 if (m_pressed >= 0 && m_pressed < static_cast<int>(m_entries.size()) && (event.pressed || m_pressPos.x != 0.0f)) {
177 const float dx = p.x - m_pressPos.x, dy = p.y - m_pressPos.y;
178 if (std::sqrt(dx * dx + dy * dy) >= drag.getDragThreshold() && !m_entries[static_cast<size_t>(m_pressed)].folder) {
179 const auto& e = m_entries[static_cast<size_t>(m_pressed)];
180 DragData data;
181 data.type = DragDataType::File;
182 data.filePath = e.path;
183 data.displayName = e.name;
184 data.accentColor = NUIThemeManager::getInstance().getColor("accentPrimary");
185 data.previewWidth = 180.0f;
186 data.previewHeight = 30.0f;
187 drag.beginDrag(data, m_pressPos, this);
188 m_dragging = true;
189 }
190 }
191 return true;
192 }
193 if (event.type == NUIMouseEventType::Scroll) {
194 m_scroll -= event.wheelDelta * 60.0f;
195 clampScroll();
196 setDirty(true);
197 return true;
198 }
199 if (event.pressed && event.button == NUIMouseButton::Left) {
200 if (l.up.contains(p)) { setDirectory(std::filesystem::path(m_dir).parent_path().string()); return true; }
201 for (size_t i = 0; i < l.places.size(); ++i) if (l.places[i].contains(p)) { setDirectory(m_places[i].path); return true; }
202 const int row = rowAt(p, l);
203 if (row < 0) return true;
204 const auto& e = m_entries[static_cast<size_t>(row)];
205 const NUIRect rowRect{l.list.x, l.list.y + static_cast<float>(row) * rowHeight() - m_scroll, l.list.width, rowHeight()};
206 if (e.folder) { setDirectory(e.path); return true; }
207 if (m_hover == row && playButton(rowRect).contains(p)) { if (m_onPlay) m_onPlay(e.path); return true; }
208 if (m_hover == row && refButton(rowRect).contains(p)) { if (m_onAddReference) m_onAddReference(e.path); return true; }
209 m_selected = row;
210 m_pressed = row;
211 m_pressPos = p;
212 if (event.type == NUIMouseEventType::DoubleClick && m_onPlay) m_onPlay(e.path);
213 setDirty(true);
214 return true;
215 }
216 if (event.released && event.button == NUIMouseButton::Left) { m_pressed = -1; m_pressPos = {0.0f, 0.0f}; return true; }
217 return true;
218}
