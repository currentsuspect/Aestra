51UISurfaceGeometry geometryFromJson(const JSON& obj) {
52 UISurfaceGeometry g;
53 const auto readAnchor = [&obj](const char* key, double& out) {
54 if (obj.has(key) && obj[key].isNumber() && std::isfinite(obj[key].asNumber())) {
55 out = clamp01(obj[key].asNumber());
56 }
57 };
58 readAnchor("anchorX", g.anchorX);
59 readAnchor("anchorY", g.anchorY);
60
61 // Pixels, not fractions. Nothing is clamped to fit: an oversized preference is
62 // still the user's preference, and resolution shrinks only what is displayed.
63 // Beyond any real display (or non-finite, or not positive) is corrupt data.
64 // static: MSVC rejects using a non-static constexpr local inside a lambda without capturing it (C3493).
65 static constexpr double kMaxSurfaceExtent = 16384.0;
66 const auto readExtent = [&obj](const char* key, double& out) {
67 if (obj.has(key) && obj[key].isNumber()) {
68 const double v = obj[key].asNumber();
69 if (std::isfinite(v) && v > 0.0 && v <= kMaxSurfaceExtent) {
70 out = v;
71 }
72 }
73 };
74 readExtent("width", g.width);
75 readExtent("height", g.height);
76
77 if (obj.has("maximized") && obj["maximized"].isBool()) {
78 g.maximized = obj["maximized"].asBool();
79 }
80 if (obj.has("lastUsedAt") && obj["lastUsedAt"].isNumber()) {
81 constexpr double kMaxLastUsedAt = 9007199254740992.0; // 2^53: exact in a double, so no cast UB or prune overflow.
82 const double raw = obj["lastUsedAt"].asNumber();
83 if (std::isfinite(raw) && std::trunc(raw) == raw && raw >= 0.0 && raw <= kMaxLastUsedAt) {
84 g.lastUsedAt = static_cast<int64_t>(raw);
85 }
86 }
87 return g;
88}

126JSON migrateToCurrent(JSON root, int fromVersion) {
127 if (fromVersion == UISurfaceStore::kCurrentSchemaVersion) {
128 return root;
129 }
130 // No prior versions exist yet; anything else is unrecognized/foreign and
131 // is handled by the caller falling back to defaults before this is ever
132 // reached (see UISurfaceStore::load).
133 return root;
134}

139void pruneStaleEditorEntries(std::map<std::string, UISurfaceGeometry>& surfaces, int64_t nowSeconds) {
140 for (auto it = surfaces.begin(); it != surfaces.end();) {
141 if (isEditorKey(it->first) && (nowSeconds - it->second.lastUsedAt) > kEditorEntryMaxAgeSeconds) {
142 it = surfaces.erase(it);
143 } else {
144 ++it;
145 }
146 }
147
148 std::vector<std::string> editorKeys;
149 for (const auto& [key, geom] : surfaces) {
150 if (isEditorKey(key)) {
151 editorKeys.push_back(key);
152 }
153 }
154 if (editorKeys.size() <= kEditorEntryMaxCount) {
155 return;
156 }
157 std::sort(editorKeys.begin(), editorKeys.end(), [&surfaces](const std::string& a, const std::string& b) {
158 return surfaces.at(a).lastUsedAt < surfaces.at(b).lastUsedAt;
159 });
160 const size_t excess = editorKeys.size() - kEditorEntryMaxCount;
161 for (size_t i = 0; i < excess; ++i) {
162 surfaces.erase(editorKeys[i]);
163 }
164}

178UISurfaceStore UISurfaceStore::load(const std::string& path) {
179 UISurfaceStore store;
180
181 const std::optional<JSON> parsed = readJSONStrict(path);
182 if (!parsed.has_value()) {
183 return store; // Missing, unreadable, empty, or malformed: defaults.
184 }
185 const JSON& root = *parsed;
186 if (!root.isObject()) {
187 return store;
188 }
189
190 if (!root.has("schemaVersion") || !root["schemaVersion"].isNumber()) {
191 return store; // No recognizable version at all: treat as foreign/corrupt.
192 }
193 // Validate as a double first: asInt() is a bare static_cast, so 0.5 or -1 would pass as "older".
194 const double rawVersion = root["schemaVersion"].asNumber();
195 if (!std::isfinite(rawVersion) || std::trunc(rawVersion) != rawVersion || rawVersion < 1.0 ||
196 rawVersion > static_cast<double>(kCurrentSchemaVersion)) {
197 return store; // Invalid, pre-history, or from a future build: defaults.
198 }
199 const int fromVersion = static_cast<int>(rawVersion);
200
201 JSON migrated = migrateToCurrent(root, fromVersion);
202 store.schemaVersion = kCurrentSchemaVersion;
203
204 if (migrated.has("surfaces") && migrated["surfaces"].isObject()) {
205 for (const auto& [key, value] : migrated["surfaces"].asObject()) {
206 if (value.isObject()) {
207 store.surfaces[key] = geometryFromJson(value);
208 }
209 }
210 }
211
212 if (migrated.has("dialogExport") && migrated["dialogExport"].isObject()) {
213 store.dialogExport = dialogExportFromJson(migrated["dialogExport"]);
214 }
215
216 if (migrated.has("boolPreferences") && migrated["boolPreferences"].isObject()) {
217 for (const auto& [key, value] : migrated["boolPreferences"].asObject()) {
218 if (value.isBool()) {
219 store.boolPreferences[key] = value.asBool();
220 }
221 }
222 }
223
224 if (migrated.has("listPreferences") && migrated["listPreferences"].isObject()) {
225 for (auto& [key, value] : migrated["listPreferences"].asObject()) {
226 if (!value.isArray()) {
227 continue;
228 }
229 std::vector<std::string> items;
230 for (const JSON& item : value.asArray()) {
231 if (item.isString()) {
232 items.push_back(item.asString());
233 }
234 }
235 store.listPreferences[key] = std::move(items);
236 }
237 }
238
239 pruneStaleEditorEntries(store.surfaces, nowUnixSeconds());
240
241 return store;
242}

244bool UISurfaceStore::save(const std::string& path) const {
245 if (path.empty()) {
246 return false;
247 }
248
249 JSON root = JSON::object();
250 root.set("schemaVersion", JSON(static_cast<double>(schemaVersion)));
251
252 JSON surfacesJson = JSON::object();
253 for (const auto& [key, geom] : surfaces) {
254 surfacesJson.set(key, geometryToJson(geom));
255 }
256 root.set("surfaces", surfacesJson);
257
258 if (dialogExport.has_value()) {
259 root.set("dialogExport", dialogExportToJson(*dialogExport));
260 }
261
262 JSON boolPrefsJson = JSON::object();
263 for (const auto& [key, value] : boolPreferences) {
264 boolPrefsJson.set(key, JSON(value));
265 }
266 root.set("boolPreferences", boolPrefsJson);
267
268 JSON listPrefsJson = JSON::object();
269 for (const auto& [key, values] : listPreferences) {
270 JSON arr = JSON::array();
271 for (const std::string& item : values) {
272 arr.push(JSON(item));
273 }
274 listPrefsJson.set(key, arr);
275 }
276 root.set("listPreferences", listPrefsJson);
277
278 return writeJSONAtomic(path, root);
279}
