19bool isUnder(const std::string& path, const std::string& base) {
20 if (base.empty()) return false;
21 const std::string p = mapKeyForPath(path);
22 std::string b = mapKeyForPath(base);
23 if (p == b) return true;
24 if (!b.empty() && b.back() != '/' && b.back() != '\\') b += '/';
25 return p.rfind(b, 0) == 0;
26}

35std::string FileBrowser::currentPlaceLabel(std::string* placePath) const {
36 if (isShowingListing()) {
37 if (placePath) placePath->clear();
38 return listingTitle_;
39 }
40 // The longest place the current folder sits under names it.
41 std::string bestLabel, bestPath;
42 const auto consider = [&](const std::string& label, const std::string& path) {
43 if (path.empty() || !isUnder(currentPath_, path)) return;
44 if (mapKeyForPath(path).size() > mapKeyForPath(bestPath).size()) {
45 bestLabel = label;
46 bestPath = path;
47 }
48 };
49 const auto leaf = [](const std::string& path) {
50 const std::string name = std::filesystem::path(path).filename().string();
51 return name.empty() ? path : name;
52 };
53 if (!rootPath_.empty()) {
54 consider("Library", rootPath_);
55 consider("User Library", (std::filesystem::path(rootPath_) / "User Library").string());
56 consider("Packs", (std::filesystem::path(rootPath_) / "Packs").string());
57 }
58 for (const auto& place : systemPlaces_) consider(place.label, place.path);
59 for (const auto& place : customPlacePaths_) consider(leaf(place), place);
60 for (const auto& fav : favoritesPaths_) consider(leaf(fav), fav);
61 if (bestPath.empty()) {
62 bestPath = currentPath_;
63 bestLabel = currentPath_.empty() ? "Library" : leaf(currentPath_);
64 }
65 if (placePath) *placePath = bestPath;
66 return bestLabel;
67}

116void FileBrowser::renderLocationRow(NUIRenderer& renderer, const BrowserLayout& layout) {
117 if (layout.location.isEmpty()) return;
118 auto& theme = NUIThemeManager::getInstance();
119 const auto& props = theme.getCurrentTheme();
120 const NUIColor text = theme.getColor("textPrimary");
121 const NUIColor muted = theme.getColor("textSecondary");
122 const float font = props.fontSizeS;
123
124 // Back
125 {
126 const NUIRect& b = layout.backButton;
127 const float cx = b.x + b.width * 0.5f, cy = b.y + b.height * 0.5f;
128 const NUIColor c = muted.withAlpha(navHistoryIndex_ > 0 ? 0.85f : 0.28f);
129 renderer.drawLine({cx + 2.0f, cy - 4.0f}, {cx - 2.0f, cy}, 1.5f, c);
130 renderer.drawLine({cx - 2.0f, cy}, {cx + 2.0f, cy + 4.0f}, 1.5f, c);
131 }
132
133 // Place chip: the place this folder sits in, opening Places.
134 std::string placePath;
135 const std::string place = currentPlaceLabel(&placePath);
136 const NUIRect& area = layout.pathLabel;
137 const float maxChip = std::max(60.0f, area.width * 0.55f);
138 const std::string chipText = ellipsizeEnd(renderer, place, font, maxChip - 30.0f);
139 const float chipW = renderer.measureText(chipText, font).width + 30.0f;
140 const NUIRect chip(area.x, area.y, chipW, kChipH);
141 if (placesOpen_) renderer.fillRoundedRect(chip, 3.0f, theme.getColor("accentPrimary").withAlpha(0.20f));
142 renderer.strokeRoundedRect(chip, 3.0f, 1.0f, theme.getColor("border").withAlpha(0.75f));
143 const bool favourite = !placePath.empty() &&
144 std::find(favoritesPaths_.begin(), favoritesPaths_.end(), placePath) != favoritesPaths_.end();
145 renderer.fillCircle({chip.x + 9.0f, chip.y + chip.height * 0.5f}, 3.0f,
146 favourite ? theme.getColor("warning") : theme.getColor("accentPrimary"));
147 renderer.drawText(chipText, {chip.x + 17.0f, std::round(renderer.calculateTextY(chip, font))}, font, text);
148 drawDownChevron(renderer, chip.right() - 8.0f, chip.y + chip.height * 0.5f, muted);
149 {
150 ChromeHit hit;
151 hit.bounds = chip;
152 hit.placeChip = true;
153 chromeHits_.push_back(hit);
154 }
155
156 // Breadcrumbs from the place down to here; the last is bold-white, older ones
157 // fold into "…" from the left when the row runs out.
158 if (isShowingListing() || placePath.empty()) return;
159 std::vector<std::pair<std::string, std::string>> crumbs; // label, path
160 {
161 std::filesystem::path p(currentPath_);
162 const std::string base = mapKeyForPath(placePath);
163 while (!p.empty() && mapKeyForPath(p.string()) != base && p.has_parent_path() && p.parent_path() != p) {
164 crumbs.emplace_back(p.filename().string(), p.string());
165 p = p.parent_path();
166 }
167 std::reverse(crumbs.begin(), crumbs.end());
168 }
169 if (crumbs.empty()) return;
170 const float sepW = 14.0f;
171 float room = area.right() - chip.right() - 4.0f;
172 std::vector<float> widths;
173 for (const auto& c : crumbs) widths.push_back(renderer.measureText(c.first, font).width);
174 size_t first = 0;
175 float total = 0.0f;
176 for (float w : widths) total += w + sepW;
177 while (first + 1 < crumbs.size() && total > room) {
178 total -= widths[first] + sepW;
179 ++first;
180 }
181 float x = chip.right() + 4.0f;
182 const float cy = chip.y + chip.height * 0.5f;
183 if (first > 0) {
184 renderer.drawText("\xe2\x80\xa6", {x + 3.0f, std::round(renderer.calculateTextY(chip, font))}, font, muted);
185 x += sepW;
186 room -= sepW;
187 }
188 for (size_t i = first; i < crumbs.size(); ++i) {
189 // separator
190 renderer.drawLine({x + 5.0f, cy - 3.0f}, {x + 8.0f, cy}, 1.2f, muted.withAlpha(0.45f));
191 renderer.drawLine({x + 8.0f, cy}, {x + 5.0f, cy + 3.0f}, 1.2f, muted.withAlpha(0.45f));
192 x += sepW;
193 const bool lastCrumb = i + 1 == crumbs.size();
194 const float avail = std::max(0.0f, area.right() - x);
195 const std::string label = ellipsizeEnd(renderer, crumbs[i].first, font, avail);
196 renderer.drawText(label, {x, std::round(renderer.calculateTextY(chip, font))}, font,
197 lastCrumb ? text : muted.withAlpha(0.85f));
198 const float w = renderer.measureText(label, font).width;
199 if (!lastCrumb) {
200 ChromeHit hit;
201 hit.bounds = NUIRect(x - 2.0f, chip.y, w + 4.0f, chip.height);
202 hit.path = crumbs[i].second;
203 chromeHits_.push_back(hit);
204 }
205 x += w;
206 if (x >= area.right()) break;
207 }
208}

210void FileBrowser::renderTagChips(NUIRenderer& renderer, const BrowserLayout& layout) {
211 if (layout.chips.isEmpty()) return;
212 auto& theme = NUIThemeManager::getInstance();
213 const auto& props = theme.getCurrentTheme();
214 const float font = props.fontSizeXS;
215 float x = layout.chips.x + 10.0f;
216 const float y = layout.chips.y + (layout.chips.height - 20.0f) * 0.5f;
217 const auto drawChip = [&](const std::string& label, const std::string& tag, bool dot) {
218 const float w = renderer.measureText(label, font).width + (dot ? 28.0f : 18.0f);
219 if (x + w > layout.chips.right() - 10.0f) return false;
220 const NUIRect r(x, y, w, 20.0f);
221 const bool on = activeTagFilter_ == tag;

225 renderer.strokeRoundedRect(r, 10.0f, 1.0f, theme.getColor("border").withAlpha(0.75f));
226 }
227 float tx = r.x + 9.0f;
228 if (dot) {
229 renderer.fillCircle({tx + 3.0f, r.y + r.height * 0.5f}, 3.5f, collectionColor(tag));
230 tx += 10.0f;
231 }
232 renderer.drawText(label, {tx, std::round(renderer.calculateTextY(r, font))}, font,
233 on ? theme.getColor("backgroundPrimary") : theme.getColor("textSecondary"));
234 ChromeHit hit;
235 hit.bounds = r;
236 hit.hasTag = true;
237 hit.tag = tag;
238 chromeHits_.push_back(hit);
239 x += w + 4.0f;
240 return true;
241 };
242 drawChip("All", "", false);
243 for (const auto& name : collections_) {
244 if (!drawChip(name, name, true)) break;

318bool FileBrowser::handleChromeHits(const NUIMouseEvent& event) {
319 if (event.cursorCaptured) return false;
320 int hovered = -1;
321 for (int i = 0; i < static_cast<int>(chromeHits_.size()); ++i) {
322 if (chromeHits_[static_cast<size_t>(i)].bounds.contains(event.position)) {
323 hovered = i;
324 break;
325 }
326 }
327 if (hovered != hoveredChromeHit_) {
328 hoveredChromeHit_ = hovered;
329 setDirty(true); // hover overlay only
330 }
331 if (hovered < 0 || !event.pressed || event.button != NUIMouseButton::Left) return false;
332
333 const ChromeHit hit = chromeHits_[static_cast<size_t>(hovered)];
334 setFocused(true); // so Esc closes Places and arrows move through the rows
335 if (hit.kind >= 0) {
336 selectKind(hit.kind);
337 } else if (hit.placeChip) {
338 setPlacesOpen(!placesOpen_);
339 } else if (hit.scopeChip) {
340 setSearchWholeLibrary(!searchWholeLibrary_);
341 invalidateCache();
342 } else if (hit.hasTag) {
343 setPlacesOpen(false);
344 activeTagFilter_ = hit.tag;
345 applyFilter();
346 invalidateCache();
347 } else if (!hit.path.empty()) {
348 setPlacesOpen(false);
349 navigateTo(hit.path);
350 }
351 return true;
352}
