37void TrackManagerUI::updateScrollbar() {
38 if (!m_scrollbar)

40
41 AestraUI::NUIRect bounds = getBounds();
42 float headerHeight = kTimelineHeaderHeight;

48

54 m_scrollbar->setCurrentRange(m_scrollOffset, viewportHeight);

83void TrackManagerUI::setTimelineViewStartBeat(double viewStartBeat, bool isFinal) {
84 const float gridWidthPx = getTimelineGridWidthPixels();
85 if (!(m_pixelsPerBeat > 0.0f) || gridWidthPx <= 0.0f)
86 return;
87
88 const double viewWidthBeats = static_cast<double>(gridWidthPx / m_pixelsPerBeat);
89 const double domainStart = m_minimapDomainStartBeat;
90 const double domainEnd = std::max(m_minimapDomainEndBeat, domainStart + viewWidthBeats);
91 const double maxStart = std::max(domainStart, domainEnd - viewWidthBeats);
92
93 const double clampedStart = std::max(domainStart, std::min(viewStartBeat, maxStart));
94 float newScrollOffset = std::max(0.0f, static_cast<float>(clampedStart * static_cast<double>(m_pixelsPerBeat)));
95
96 // OPTIMIZATION: Only invalidate cache if scroll actually changed
97 // Use beat-based threshold to ensure responsiveness at all zoom levels
98 const double currentStartBeat = static_cast<double>(m_timelineScrollOffset / m_pixelsPerBeat);
99 const double beatThreshold = 0.01; // ~1/100th of a beat is perceptible
100 bool scrollChanged = std::abs(clampedStart - currentStartBeat) > beatThreshold;
101
102 if (scrollChanged) {
103 m_timelineScrollOffset = newScrollOffset;
104
105 for (auto& trackUI : m_trackUIComponents) {
106 trackUI->setTimelineScrollOffset(m_timelineScrollOffset);
107 }
108
109 invalidateCache();
110 setDirty(true);
111 }
112
113 if (!isFinal) {
114 updateTimelineMinimap(0.0);
115 }
116}

118void TrackManagerUI::resizeTimelineViewEdgeFromMinimap(AestraUI::TimelineMinimapResizeEdge edge, double anchorBeat,
119 double edgeBeat, bool isFinal) {
120 const float gridWidthPx = getTimelineGridWidthPixels();
121 if (gridWidthPx <= 0.0f)
122 return;
123
124 constexpr float kMinPixelsPerBeat = 1.0f; // Allow extreme zoom-out for long clips
125 constexpr float kMaxPixelsPerBeat = kMaxTimelinePixelsPerBeat;
126
127 const double domainStart = m_minimapDomainStartBeat;
128 const double domainEnd = std::max(m_minimapDomainEndBeat, domainStart + 1.0);
129
130 const double minWidthBeats = static_cast<double>(gridWidthPx / kMaxPixelsPerBeat);
131 // Max zoom-out is limited by domain size (can't view beyond domain via minimap resize)
132 const double maxWidthBeats = std::min(static_cast<double>(gridWidthPx / kMinPixelsPerBeat),
133 domainEnd - domainStart // Can't zoom out beyond current domain
134 );
135
136 const auto applyZoom = [this](float newPixelsPerBeat) {
137 m_pixelsPerBeat = newPixelsPerBeat;
138 m_targetPixelsPerBeat = newPixelsPerBeat;
139 for (auto& trackUI : m_trackUIComponents) {
140 trackUI->setPixelsPerBeat(m_pixelsPerBeat);
141 }
142 };
143
144 if (edge == AestraUI::TimelineMinimapResizeEdge::Left) {
145 // Dragging the left edge: keep right edge anchored.
146 const double clampedEdge =
147 std::max(domainStart, std::min(edgeBeat, anchorBeat - std::max(1e-6, minWidthBeats)));
148 const double desiredWidth = std::max(minWidthBeats, std::min(maxWidthBeats, anchorBeat - clampedEdge));
149 const float newPixelsPerBeat =
150 safeClampFloat(static_cast<float>(gridWidthPx / desiredWidth), kMinPixelsPerBeat, kMaxPixelsPerBeat);
151 applyZoom(newPixelsPerBeat);
152
153 const double viewWidthBeats = static_cast<double>(gridWidthPx / m_pixelsPerBeat);
154 const double viewStartBeat = anchorBeat - viewWidthBeats;
155 setTimelineViewStartBeat(viewStartBeat, isFinal);
156 } else {
157 // Dragging the right edge: keep left edge anchored.
158 // Clamp to domain - minimap resize does NOT expand domain (only edge-scroll during clip drag does)
159 const double clampedEdge = std::min(domainEnd, std::max(edgeBeat, anchorBeat + std::max(1e-6, minWidthBeats)));
160 const double desiredWidth = std::max(minWidthBeats, std::min(maxWidthBeats, clampedEdge - anchorBeat));
161 const float newPixelsPerBeat =
162 safeClampFloat(static_cast<float>(gridWidthPx / desiredWidth), kMinPixelsPerBeat, kMaxPixelsPerBeat);
163 applyZoom(newPixelsPerBeat);
164
165 setTimelineViewStartBeat(anchorBeat, isFinal);
166 }
167
168 invalidateCache(); // Always invalidate on zoom/resize changes
169 updateTimelineMinimap(0.0);
170}

182void TrackManagerUI::zoomTimelineAroundBeat(double anchorBeat, float zoomMultiplier) {
183 const float gridWidthPx = getTimelineGridWidthPixels();
184 if (gridWidthPx <= 0.0f)
185 return;
186
187 // Calculate min pixels/beat based on domain (can't zoom out beyond domain)
188 const double domainWidth = std::max(1.0, m_minimapDomainEndBeat - m_minimapDomainStartBeat);
189 float minPPB = std::max(1.0f, static_cast<float>(gridWidthPx / domainWidth));
190
191 // Minimap zoom must feel immediate; keep the smooth-zoom system in sync by updating both.
192 const float newPixelsPerBeat = safeClampFloat(m_pixelsPerBeat * zoomMultiplier, minPPB, kMaxTimelinePixelsPerBeat);
193 m_pixelsPerBeat = newPixelsPerBeat;
194 m_targetPixelsPerBeat = newPixelsPerBeat;
195
196 for (auto& trackUI : m_trackUIComponents) {
197 trackUI->setPixelsPerBeat(m_pixelsPerBeat);
198 }
199
200 const double viewWidthBeats = static_cast<double>(gridWidthPx / m_pixelsPerBeat);
201 const double viewStartBeat = anchorBeat - (viewWidthBeats * 0.5);
202 setTimelineViewStartBeat(viewStartBeat, true);
203 invalidateCache(); // Always invalidate on zoom changes
204 updateTimelineMinimap(0.0);
205}

207void TrackManagerUI::updateTimelineMinimap(double deltaTime) {
208 if (!m_timelineMinimap)
209 return;
210 if (!m_playlistVisible)
211 return;
212 if (!m_trackManager)
213 return;
214
215 const float gridWidthPx = getTimelineGridWidthPixels();
216 if (!(m_pixelsPerBeat > 0.0f) || gridWidthPx <= 0.0f)
217 return;
218
219 // Not const: a shrinking domain re-clamps the scroll offset further down, and the
220 // model published at the end of this function must describe the viewport we ended
221 // up with, not the one we started with.
222 double viewStartBeat = static_cast<double>(m_timelineScrollOffset / m_pixelsPerBeat);
223 const double viewWidthBeats = static_cast<double>(gridWidthPx / m_pixelsPerBeat);
224 double viewEndBeat = viewStartBeat + viewWidthBeats;
225
226 const double playheadBeat = secondsToBeats(m_trackManager->getUIPosition());
227
228 auto& playlist = m_trackManager->getPlaylistModel();
229 double clipEndBeat = playlist.getTotalDurationBeats();
230
231 // === SMART DOMAIN CALCULATION ===
232 const double beatsPerBarD = static_cast<double>(std::max(1, m_beatsPerBar));
233
234 // Has content?
235 bool hasContent = (clipEndBeat > 0.001);
236
237 double requiredEndBeat;
238 const bool allowViewDrivenDomainExpand = m_isDraggingClipInstant;
239 if (!hasContent) {
240 // EMPTY PROJECT: Fixed 16 bars - can't zoom out beyond this
241 const double emptyFixedBars = 16.0;
242 const double emptyMinBeats = beatsPerBarD * emptyFixedBars;
243
244 // Only an active clip drag may extend the domain beyond the empty-project floor.
245 if (allowViewDrivenDomainExpand && viewEndBeat > emptyMinBeats) {
246 // Edge scrolling pushed us out - expand domain to current view + small buffer
247 requiredEndBeat = viewEndBeat + (beatsPerBarD * 2.0);
248 } else {
249 // Locked at 16 bars
250 requiredEndBeat = emptyMinBeats;
251 }
252
253 // Playhead can also push domain (if playing beyond)
254 const double playheadBuffer = beatsPerBarD * 2.0;
255 if (playheadBeat + playheadBuffer > requiredEndBeat) {
256 requiredEndBeat = playheadBeat + playheadBuffer;
257 }
258 } else {
259 // HAS CONTENT: Dynamic padding that scales with content length
260 // Short clips (< 16 bars): Add 8 bars padding (50% headroom feels right)
261 // Medium clips (16-64 bars): Add ~25% padding
262 // Long clips (64+ bars): Add ~12.5% padding (efficient use of space)
263 double paddingBars;
264 double clipBars = clipEndBeat / beatsPerBarD;
265
266 if (clipBars < 16.0) {
267 paddingBars = 8.0; // Short clip: fixed 8 bar padding
268 } else if (clipBars < 64.0) {
269 paddingBars = clipBars * 0.25; // Medium: 25% of clip length
270 } else {
271 paddingBars = clipBars * 0.125; // Long: 12.5%, no cap - infinite if PC handles it
272 }
273
274 const double padBeats = beatsPerBarD * paddingBars;
275 const double playheadBuffer = beatsPerBarD * 4.0; // 4 bars ahead of playhead
276
277 // Normal scrolling/zooming should not mutate the domain; only a real clip edge-drag may do that.
278 requiredEndBeat = std::max(clipEndBeat + padBeats, playheadBeat + playheadBuffer);
279 if (allowViewDrivenDomainExpand) {
280 requiredEndBeat = std::max(requiredEndBeat, viewEndBeat + beatsPerBarD);
281 }
282
283 // No hard ceiling - let it grow as big as needed (infinite if PC can handle it)
284 }
285
286 // Update domain instantly (no cooldown needed)
287 bool domainShrank = false;
288 if (!(m_minimapDomainEndBeat > 0.0)) {
289 // First init
290 m_minimapDomainEndBeat = requiredEndBeat;
291 m_minimapNeedsRebuild = true;
292 } else if (requiredEndBeat > m_minimapDomainEndBeat + 1e-3) {
293 // Domain needs to grow - instant
294 m_minimapDomainEndBeat = requiredEndBeat;
295 m_minimapNeedsRebuild = true;
296 } else if (requiredEndBeat < m_minimapDomainEndBeat - 1e-3) {
297 // Domain can shrink - also instant, no reason to delay
298 m_minimapDomainEndBeat = requiredEndBeat;
299 m_minimapNeedsRebuild = true;
300 domainShrank = true;
301 }
302 // else: domain unchanged, nothing to do
303
304 // A shrinking domain can leave the viewport scrolled past the new end — delete the
305 // last clip in a long arrangement and the view stays out where that clip used to
306 // be, showing empty grid with a minimap thumb pinned off its own track. Growing is
307 // safe (the old view is still inside the larger domain), so only the shrink needs
308 // this. setTimelineViewStartBeat applies exactly the clamp this needs and is safe
309 // to call here: it only re-enters updateTimelineMinimap when isFinal is false.
310 if (domainShrank) {
311 setTimelineViewStartBeat(viewStartBeat, /*isFinal=*/true);
312
313 // Re-read what the clamp actually settled on. Zoom is untouched, so the width
314 // is unchanged and only the start moves. Without this the minimap model below
315 // would publish the pre-clamp viewport — the thumb would stay parked outside
316 // the domain for a frame, which is the exact symptom this block exists to fix.
317 viewStartBeat = static_cast<double>(m_timelineScrollOffset / m_pixelsPerBeat);
318 viewEndBeat = viewStartBeat + viewWidthBeats;
319 }
320
321 if (m_minimapNeedsRebuild) {
322 std::vector<AestraUI::TimelineMinimapClipSpan> spans;
323
324 const auto& laneIds = playlist.getLaneIDs();
325 for (size_t i = 0; i < laneIds.size(); ++i) {
326 const auto& laneId = laneIds[i];
327 if (const auto* lane = playlist.getLane(laneId)) {
328 for (const auto& clip : lane->clips) {
329 AestraUI::TimelineMinimapClipSpan span;
330 span.id = static_cast<AestraUI::TimelineMinimapClipId>(clip.id.high ^ clip.id.low);
331 span.type = AestraUI::TimelineMinimapClipType::Audio;
332
333 span.startBeat = clip.startBeat;
334 span.endBeat = clip.startBeat + clip.durationBeats;
335 span.trackIndex = static_cast<uint32_t>(i); // Track index for minimap matching
336
337 if (!(span.endBeat > span.startBeat))
338 continue;
339
340 spans.push_back(span);
341 }
342 }
343 }
344
345 m_timelineSummaryCache.requestRebuild(std::move(spans), m_minimapDomainStartBeat, m_minimapDomainEndBeat);
346 m_minimapNeedsRebuild = false;
347 }
348
349 m_timelineSummarySnapshot = m_timelineSummaryCache.getSnapshot();
350
351 if (m_isDrawingSelectionBox) {
352 auto& themeManager = AestraUI::NUIThemeManager::getInstance();
353 const auto& layout = themeManager.getLayoutDimensions();
354 const float controlAreaWidth = layout.trackControlsWidth;
355 const float gridStartXAbs = getBounds().x + controlAreaWidth + kTimelineGridInsetX;
356
357 const float minX = std::min(m_selectionBoxStart.x, m_selectionBoxEnd.x);
358 const float maxX = std::max(m_selectionBoxStart.x, m_selectionBoxEnd.x);
359
360 const double startBeat =
361 (static_cast<double>((minX - gridStartXAbs) + m_timelineScrollOffset)) / m_pixelsPerBeat;
362 const double endBeat = (static_cast<double>((maxX - gridStartXAbs) + m_timelineScrollOffset)) / m_pixelsPerBeat;
363 m_minimapSelectionBeatRange.start = std::max(0.0, std::min(startBeat, endBeat));
364 m_minimapSelectionBeatRange.end = std::max(0.0, std::max(startBeat, endBeat));
365 }
366
367 AestraUI::TimelineMinimapModel model;
368 model.summary = &m_timelineSummarySnapshot;
369 model.view.start = viewStartBeat;
370 model.view.end = viewEndBeat;
371 model.playheadBeat = playheadBeat;
372 model.selection = m_minimapSelectionBeatRange;
373 model.mode = m_minimapMode;
374 model.aggregation = m_minimapAggregation;
375 model.beatsPerBar = m_beatsPerBar;
376 model.showSelection = model.selection.isValid();
377 model.showLoop = false;
378 model.showMarkers = false;
379 model.showDiagnostics = false;
380 model.showPlayhead = !m_patternMode; // Hide playhead in Arsenal Pattern Mode
381
382 m_timelineMinimap->setModel(model);
383}

425// Calculate maximum timeline extent needed based on all samples
426// Calculate maximum timeline extent needed based on all clips
427double TrackManagerUI::getMaxTimelineExtent() const {
428 if (!m_trackManager)
429 return 0.0;
430
431 auto& playlist = m_trackManager->getPlaylistModel();
432 double totalDurationBeats = playlist.getTotalDurationBeats();
433
434 double bpm = std::max(playlist.getBPM(), 1.0);
435 double secondsPerBeat = 60.0 / bpm;
436
437 // Minimum extent - at least 8 bars even if empty
438 double minExtent = 8.0 * m_beatsPerBar * secondsPerBeat;
439
440 // Convert beats to seconds for extent
441 double totalDurationSeconds = totalDurationBeats * secondsPerBeat;
442
443 // Add 2 bars padding
444 double paddedEnd = totalDurationSeconds + (2.0 * m_beatsPerBar * secondsPerBeat);
445
446 return std::max(paddedEnd, minExtent);
447}
