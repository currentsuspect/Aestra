50 struct CacheKeyHash {
51 // Quantize float to an integer bucket so that epsilon-equal values hash the same.
52 static int quantize(float v) {
53 return static_cast<int>(std::round(v * 1e5f));
54 }
55
56 size_t operator()(const CacheKey& k) const {
57 // Combine hashes of all key components
58 size_t h1 = std::hash<std::string>()(k.svgContent);
59 size_t h2 = std::hash<int>()(k.width);
60 size_t h3 = std::hash<int>()(k.height);
61 // Use quantized tint values so hash agrees with epsilon equality
62 size_t h4 = std::hash<int>()(quantize(k.tint.r));
63 size_t h5 = std::hash<int>()(quantize(k.tint.g));
64 size_t h6 = std::hash<int>()(quantize(k.tint.b));
65 size_t h7 = std::hash<int>()(quantize(k.tint.a));
66
67 // Simple hash combination
68 return h1 ^ (h2 << 1) ^ (h3 << 2) ^ (h4 << 3) ^ (h5 << 4) ^ (h6 << 5) ^ (h7 << 6);
69 }
70 };
