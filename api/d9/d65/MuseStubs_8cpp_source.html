98DeleteTrackCommand::DeleteTrackCommand(TrackManager& manager, int trackIndex)
99 : m_manager(manager), m_trackIndex(trackIndex) {}

101void DeleteTrackCommand::execute() {
102 if (m_executed)
103 return;
104
105 MixerChannel* ch = m_manager.getChannel(static_cast<size_t>(m_trackIndex));
106 if (!ch)
107 return;
108
109 m_deletedName = ch->getName();
110 m_deletedId = ch->getChannelId();
111
112 if (!m_routesCaptured) {
113 for (UnitID unitId : m_manager.getUnitManager().getAllUnitIDs()) {
114 if (m_manager.getUnitManager().getUnitMixerChannel(unitId) == m_deletedId) {
115 m_routedUnits.push_back(unitId);
116 }
117 }
118 for (const auto& pattern : m_manager.getPatternManager().getAllPatterns()) {
119 if (pattern && pattern->isAudio() && pattern->getMixerChannelId() == m_deletedId) {
120 m_routedAudioPatterns.push_back(pattern->id);
121 }
122 }
123 for (const auto& channel : m_manager.getChannelsSnapshot()) {
124 if (!channel || channel->getChannelId() == m_deletedId)
125 continue;
126 const auto sends = channel->getSends();
127 const bool routesToDeleted = channel->getMainOutputId() == m_deletedId ||
128 std::any_of(sends.begin(), sends.end(), [this](const AudioRoute& route) {
129 return route.targetChannelId == m_deletedId;
130 });
131 if (routesToDeleted) {
132 m_mixerRoutes.push_back({channel->getChannelId(), channel->getMainOutputId(), sends});
133 }
134 }
135 m_routesCaptured = true;
136 }
137
138 // Detach rather than destroy: this command owns the channel for as long as
139 // the delete stands, so undo can put back the same object at the same id
140 // and index. See the m_detached comment in the header for why an equivalent
141 // channel is not good enough.
142 m_detached = m_manager.detachChannelById(m_deletedId, m_detachedIndex);
143 if (m_detached) {
144 m_manager.getUnitManager().resetMixerChannel(m_deletedId);
145 m_manager.getPatternManager().resetMixerChannel(m_deletedId);
146 m_manager.resetMixerRoutingDestination(m_deletedId);
147 m_manager.requestAudioGraphRebuild(GraphDirtyReason::RoutingChanged);
148 m_executed = true;
149 }
150}
151
152void DeleteTrackCommand::undo() {
153 if (!m_executed || !m_detached)
154 return;
155
156 if (m_manager.reinsertChannel(m_detached, m_detachedIndex)) {
157 for (UnitID unitId : m_routedUnits) {
158 m_manager.getUnitManager().setUnitMixerChannel(unitId, m_deletedId);
159 }
160 for (PatternID patternId : m_routedAudioPatterns) {
161 m_manager.getPatternManager().setPatternMixerChannel(patternId, m_deletedId);
162 }
163 for (const auto& routeSnapshot : m_mixerRoutes) {
164 if (auto* channel = m_manager.getChannelById(routeSnapshot.sourceChannelId)) {
165 channel->setMainOutputId(routeSnapshot.mainOutputId);
166 channel->replaceSends(routeSnapshot.sends);
167 }
168 }
169 m_manager.requestAudioGraphRebuild(GraphDirtyReason::RoutingChanged);
170 m_executed = false;
171 }
172}
173
174void DeleteTrackCommand::redo() {
175 if (m_executed)
176 return;
177
178 m_detached = m_manager.detachChannelById(m_deletedId, m_detachedIndex);
179 if (m_detached) {
180 m_manager.getUnitManager().resetMixerChannel(m_deletedId);
181 m_manager.getPatternManager().resetMixerChannel(m_deletedId);
182 m_manager.resetMixerRoutingDestination(m_deletedId);
183 m_manager.requestAudioGraphRebuild(GraphDirtyReason::RoutingChanged);
184 m_executed = true;
185 }
