25RenderSummary summarizeRender(const std::vector<float>& interleaved, uint32_t channels, uint32_t sampleRate,
26 double silenceDbfs) {
27 RenderSummary s;
28 s.channels = channels;
29 s.sampleRate = sampleRate;
30 s.hash64 = kFnvOffset;
31 if (channels == 0) {
32 return s;
33 }
34 s.frames = interleaved.size() / channels;
35 s.durationSeconds = sampleRate > 0 ? static_cast<double>(s.frames) / sampleRate : 0.0;
36 s.dcOffset.assign(channels, 0.0);
37
38 const double silenceLinear = std::pow(10.0, silenceDbfs / 20.0);
39 double peak = 0.0;
40 long double sumSquares = 0.0L;
41 uint64_t finiteCount = 0;
42 std::vector<long double> channelSums(channels, 0.0L);
43 bool any = false;
44 uint64_t firstLoud = 0;
45 uint64_t lastLoud = 0;
46
47 for (uint64_t f = 0; f < s.frames; ++f) {
48 bool loud = false;
49 for (uint32_t c = 0; c < channels; ++c) {
50 const float x = interleaved[static_cast<size_t>(f) * channels + c];
51 uint32_t bits = 0;
52 std::memcpy(&bits, &x, sizeof(bits));
53 for (int b = 0; b < 4; ++b) {
54 s.hash64 ^= (bits >> (8 * b)) & 0xFFu;
55 s.hash64 *= kFnvPrime;
56 }
57 if (!std::isfinite(x)) {
58 ++s.nonFiniteSamples;
59 continue;
60 }
61 const double a = std::fabs(static_cast<double>(x));
62 peak = std::max(peak, a);
63 sumSquares += static_cast<long double>(x) * x;
64 channelSums[c] += x;
65 ++finiteCount;
66 if (a >= 1.0) {
67 ++s.clippedSamples;
68 }
69 if (a > silenceLinear) {
70 loud = true;
71 }
72 }
73 if (loud) {
74 if (!any) {
75 firstLoud = f;
76 any = true;
77 }
78 lastLoud = f;
79 }
80 }
81
82 s.peakDbfs = toDb(peak);
83 s.rmsDbfs = finiteCount > 0 ? toDb(std::sqrt(static_cast<double>(sumSquares / finiteCount))) : -200.0;
84 for (uint32_t c = 0; c < channels; ++c) {
85 s.dcOffset[c] = s.frames > 0 ? static_cast<double>(channelSums[c] / static_cast<long double>(s.frames)) : 0.0;
86 }
87 s.silent = !any;
88 if (any && sampleRate > 0) {
89 s.leadingSilenceSeconds = static_cast<double>(firstLoud) / sampleRate;
90 s.trailingSilenceSeconds = static_cast<double>(s.frames - 1 - lastLoud) / sampleRate;
91 } else {
92 s.leadingSilenceSeconds = s.durationSeconds;
93 s.trailingSilenceSeconds = s.durationSeconds;
94 }
95 return s;
96}

107std::vector<std::string> compareToExpectation(const RenderSummary& actual, const RenderExpectation& expected,
108 double toleranceDb) {
109 std::vector<std::string> diffs;
110 char buf[160];
111 if (actual.frames != expected.frames) {
112 std::snprintf(buf, sizeof(buf), "frames: %llu, expected %llu",
113 static_cast<unsigned long long>(actual.frames), static_cast<unsigned long long>(expected.frames));
114 diffs.emplace_back(buf);
115 }
116 if (std::fabs(actual.peakDbfs - expected.peakDbfs) > toleranceDb) {
117 std::snprintf(buf, sizeof(buf), "peak: %.3f dBFS, expected %.3f (moved %+.3f dB)", actual.peakDbfs,
118 expected.peakDbfs, actual.peakDbfs - expected.peakDbfs);
119 diffs.emplace_back(buf);
120 }
121 if (std::fabs(actual.rmsDbfs - expected.rmsDbfs) > toleranceDb) {
122 std::snprintf(buf, sizeof(buf), "rms: %.3f dBFS, expected %.3f (moved %+.3f dB)", actual.rmsDbfs,
123 expected.rmsDbfs, actual.rmsDbfs - expected.rmsDbfs);
124 diffs.emplace_back(buf);
125 }
126 if (expected.hasHash && actual.hash64 != expected.hash64) {
127 diffs.emplace_back("hash: " + hashToHex(actual.hash64) + ", expected " + hashToHex(expected.hash64) +
128 " (not bit-identical)");
129 }
130 return diffs;
131}

141std::vector<std::string> compareToExpectation(const RenderSummary& actual, const AudioAnalysis& analysis,
142 const RenderExpectation& expected, double toleranceDb) {
143 std::vector<std::string> diffs = compareToExpectation(actual, expected, toleranceDb);
144 if (!expected.hasLoudness) {
145 return diffs;
146 }
147 char buf[160];
148 if (std::fabs(analysis.integratedLufs - expected.integratedLufs) > toleranceDb) {
149 std::snprintf(buf, sizeof(buf), "loudness: %.3f LUFS, expected %.3f (moved %+.3f LU)", analysis.integratedLufs,
150 expected.integratedLufs, analysis.integratedLufs - expected.integratedLufs);
151 diffs.emplace_back(buf);
152 }
153 if (std::fabs(analysis.truePeakDbtp - expected.truePeakDbtp) > toleranceDb) {
154 std::snprintf(buf, sizeof(buf), "true peak: %.3f dBTP, expected %.3f (moved %+.3f dB)", analysis.truePeakDbtp,
155 expected.truePeakDbtp, analysis.truePeakDbtp - expected.truePeakDbtp);
156 diffs.emplace_back(buf);
157 }
158 return diffs;
159}

167bool hashFromHex(const std::string& hex, uint64_t& out) {
168 if (hex.size() != 16) {
169 return false;
170 }
171 uint64_t v = 0;
172 for (const char ch : hex) {
173 v <<= 4;
174 if (ch >= '0' && ch <= '9') {
175 v |= static_cast<uint64_t>(ch - '0');
176 } else if (ch >= 'a' && ch <= 'f') {
177 v |= static_cast<uint64_t>(ch - 'a' + 10);
178 } else {
179 return false;
180 }
181 }
182 out = v;
183 return true;
184}
