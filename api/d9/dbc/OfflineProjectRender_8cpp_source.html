12AudioExporter::Result renderProjectOffline(AudioEngine& engine, TrackManager& trackManager,
13 const OfflineRenderRequest& request) {
14 // AudioExporter::render drives AudioEngine::processBlock -- the same path the
15 // live engine uses -- so committed MIDI clips synthesize through their
16 // sampler units (live/export parity, AGENTS.md §20).
17 //
18 // Restores everything this render mutates on every exit path, so a
19 // synchronous export leaves the caller's engine and session unchanged:
20 // - engine wiring (weak TrackManager ref, raw UnitManager*, owned slot map,
21 // graph copy, pattern-engine pointer) is cleared;
22 // - the pattern-playback engine is cleared so the timeline instances this
23 // render scheduled do not leak into the caller's TrackManager.
24 // The transport flags/position are never touched (see
25 // scheduleTimelineForOfflineRender below), so there is nothing else to undo.
26 struct RenderStateGuard {
27 AudioEngine& engine;
28 TrackManager& trackManager;
29 ~RenderStateGuard() {
30 engine.setGraph(AudioGraph{});
31 engine.setPatternPlaybackEngine(nullptr);
32 engine.setUnitManager(nullptr);
33 engine.setChannelSlotMap(nullptr);
34 engine.setTrackManager(nullptr);
35 // Remove, don't rewind: this render's instances must not survive into the
36 // caller's session (rewindScheduledInstances() would leave them scheduled and
37 // merely restarted).
38 trackManager.getPatternPlaybackEngine().clearScheduledInstances();
39 }
40 };
41
42 // Non-owning shared_ptr: setTrackManager requires shared ownership, but the
43 // caller owns trackManager and guarantees it outlives this render. The no-op
44 // deleter means the borrowed TrackManager is never freed here.
45 std::shared_ptr<TrackManager> borrowedTrackManager(&trackManager, [](TrackManager*) {});
46 RenderStateGuard renderGuard{engine, trackManager}; // destroyed before borrowedTrackManager
47
48 engine.setSampleRate(request.sampleRate);
49 engine.setBufferConfig(512, 2);
50 engine.setBPM(static_cast<float>(request.tempoBpm));
51 engine.setTrackManager(borrowedTrackManager);
52 engine.setUnitManager(&trackManager.getUnitManager());
53 engine.setPatternPlaybackEngine(&trackManager.getPatternPlaybackEngine());
54 trackManager.buildAndShareSlotMap();
55 if (auto slotMap = trackManager.getChannelSlotMapShared()) {
56 engine.setChannelSlotMap(slotMap);
57 }
58 engine.setGraph(AudioGraphBuilder::buildFromTrackManager(trackManager));
59 engine.initialize();
60
61 // MIDI clips reach their units through the pattern-playback engine
62 // (AudioEngine::processBlock pops scheduled notes into unit MIDI routes).
63 // Schedule the committed timeline into it WITHOUT starting live transport --
64 // the exporter drives the engine's own transport, so the caller's playing
65 // flag and position are not mutated. clearScheduledInstances() first removes
66 // any prior contents; the render guard clears again on exit so these
67 // instances don't leak. rewindScheduledInstances() cannot serve here -- it
68 // only REWINDS active instances, so anything already scheduled would have
69 // been rendered into the export alongside the timeline asked for.
70 trackManager.getPatternPlaybackEngine().clearScheduledInstances();
71 trackManager.scheduleTimelineForOfflineRender(0.0);
72
73 AudioExporter exporter(engine, trackManager);
74 if (request.progress) {
75 exporter.setProgressCallback(request.progress);
76 }
77
78 AudioExporter::Config config;
79 config.outputPath = request.outputPath;
80 config.sampleRate = request.sampleRate;
81 config.bitDepth = request.bitDepth;
82 config.scope = AudioExporter::RenderScope::FullSong;
83 // Result::maxTruePeakdBTP is measured on every render; validateTruePeak only
84 // turns it into a pass/fail against a ceiling, which is the caller's call.
85 return exporter.render(config);
86}
