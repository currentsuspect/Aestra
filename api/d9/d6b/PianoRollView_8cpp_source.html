15// Gap between the minimap (overview) and the ruler — both the layout and the
16// pitch-header geometry must agree on it so the key lane stays aligned.
17// Chrome band heights, in one place. onRender() and layoutChildren() each used
18// to carry their own copy of all four, so the header's painted geometry and the
19// grid's actual geometry were free to drift apart.
20//
21// Spec 2 §6: the editing canvas is the main object, so every band around it is
22// as small as it can be while its own contents still fit. The toolbar's buttons
23// are 30px, which 42 clears; the ruler and minimap are text and a strip.
24constexpr float kToolbarHeight = 42.0f;
25constexpr float kMinimapHeight = 22.0f;
26constexpr float kRulerHeight = 24.0f;
27constexpr float kMinimapGap = 4.0f;

32PianoRollView::PianoRollView()
33 // Spec 2 §6: the pitch lane keeps its key shapes and octave labels at 58px;
34 // the 18px it gives back goes straight to the grid.
35 : m_keyLaneWidth(58.0f), m_rulerHeight(kRulerHeight), m_pixelsPerBeat(80.0f), m_keyHeight(24.0f),
36 m_scrollX(0.0f), m_targetScrollX(0.0f)
37{
38 // [FIX] Canonical default octave: C3 (MIDI 48).
39 // scrollY = (127 - targetPitch) * keyHeight => (127 - 48) * 24 = 1896
40 m_scrollY = 1896.0f;
41 m_targetScrollY = 1896.0f;
42 m_keys = std::make_shared<PianoRollKeyLane>();
43 m_ruler = std::make_shared<PianoRollRuler>();
44 m_grid = std::make_shared<PianoRollGrid>();
45 m_notes = std::make_shared<PianoRollNoteLayer>();
46 m_controls = std::make_shared<PianoRollControlPanel>();
47
48 m_notes->setOnHoveredPitchChanged([this](int pitch) {
49 if (m_grid) m_grid->setHoveredPitch(pitch);
50 if (m_keys) m_keys->setHoveredKey(pitch);
51 });
52 m_keys->setOnHoveredKeyChanged([this](int pitch) {
53 if (m_grid) m_grid->setHoveredPitch(pitch);
54 });
55 // A note pressed in the grid presses its key (SPEC 3 §5.1).
56 m_notes->setOnKeyPressChanged([this](int pitch, bool down) {
57 if (!m_keys) return;
58 if (down) {
59 m_keys->pressKey(pitch);
60 } else {
61 m_keys->releaseKey(pitch);
62 }
63 });
64
65 // Toolbar
66 m_toolbar = std::make_shared<PianoRollToolbar>();
67 m_toolbar->setGrid(m_grid);
68 m_toolbar->setNoteLayer(m_notes);
69 m_toolbar->setPatternLengthBeats(m_patternLengthBeats);
70 m_toolbar->setOnShowShortcutHelp([this]() {
71 m_showShortcutHelp = !m_showShortcutHelp;
72 repaint();
73 });
74
75 m_controls->setNoteLayer(m_notes);
76
77 m_minimap = std::make_shared<PianoRollMinimap>(); // Local Minimap
78
79 m_vScroll = std::make_shared<NUIScrollbar>(NUIScrollbar::Orientation::Vertical);
80 m_vScroll->setOrientation(NUIScrollbar::Orientation::Vertical);
81 // No styling here: drawOverlayScrollbar() owns how every scrollbar looks
82 // (spec 2 §2), and the pixel thumb floor is enforced by the component.
83 m_vScroll->setMinimumThumbSize(0.06);
84
85 // Initial default layout config
86 m_minimap->setVisible(true);
87 m_vScroll->setVisible(true);
88
89 m_hScroll = std::make_shared<NUIScrollbar>(NUIScrollbar::Orientation::Horizontal);
90 m_hScroll->setOrientation(NUIScrollbar::Orientation::Horizontal);
91 m_hScroll->setMinimumThumbSize(0.06);
92 m_hScroll->setVisible(true);
93
94 // Ruler Zoom Callback
95 m_ruler->onZoomRequested = [this](float delta, float mouseX) {
96 const float zoomDelta = std::clamp(delta, -4.0f, 4.0f);
97 applyZoom(std::pow(1.15f, zoomDelta), mouseX);
98 };
99
100 m_ruler->onPlayheadScrubbed = [this](double beat, bool active) {
101 const double clampedBeat = std::clamp(beat, 0.0, m_totalDurationBeats);
102 setPlayheadBeat(clampedBeat, false);
103 if (m_onPlayheadScrubbed) {
104 m_onPlayheadScrubbed(clampedBeat, active);
105 }
106 };
107
108 // Ruler loop zone (SPEC 3 §5.2): snapped to the editor's grid, nearest line.
109 m_ruler->snapBeat = [this](double beat) {
110 const double step = m_notes ? MusicTheory::getSnapDuration(m_notes->getSnap()) : 0.0;
111 return step > 0.0 ? std::round(beat / step) * step : beat;
112 };
113 m_ruler->onLoopZoneDrawn = [this](double start, double end) {
114 // Clamped to the pattern, like the scrub: past its end there is only silence to loop. A
115 // zone entirely past the end becomes empty, and applyLoopZone drops it.
116 const double limit = std::max(0.0, m_totalDurationBeats);
117 applyLoopZone(std::make_pair(std::clamp(start, 0.0, limit), std::clamp(end, 0.0, limit)), true);
118 // The highlight selects the notes inside it (V82-PR1, FD-52), so Duplicate has something to act on.
119 if (m_loopZone) m_notes->selectStartingIn(m_loopZone->first, m_loopZone->second);
120 };
121 // Duplicate past the pattern's end grows the pattern by whole bars first (as the toolbar's + does).
122 m_notes->setOnNeedPatternEnd([this](double endBeat) {
123 const int bars = PianoRollDuplicate::barsToGrow(endBeat, m_patternLengthBeats, static_cast<double>(m_beatsPerBar));
124 if (bars > 0 && m_toolbar) m_toolbar->requestPatternBars(bars);
125 });
126 m_ruler->onLoopZoneCleared = [this]() {
127 if (m_loopZone) applyLoopZone(std::nullopt, true);
128 };
129
130 m_minimap->onViewChanged = [this](double start, double duration) {
131 m_scrollX = static_cast<float>(start * m_pixelsPerBeat);
132 m_targetScrollX = m_scrollX;
133
134 // ZOOM LOGIC:
135 // duration * ppb = visibleWidth
136 // ppb = visibleWidth / duration
137 float visibleW = m_grid->getWidth();
138 if (duration > 0.001) {
139 m_pixelsPerBeat = visibleW / static_cast<float>(duration);
140 }
141
142 syncChildren();
143 };
144
145 m_vScroll->setOnScroll([this](double val) {
146 float totalH = 128 * m_keyHeight;
147 float visibleH = m_grid->getHeight();
148 float maxScroll = std::max(0.0f, totalH - visibleH);
149 m_targetScrollY = safeClampRange(static_cast<float>(val), 0.0f, maxScroll);
150 });
151
152 // Horizontal scrollbar drives the target scroll across the scrollable
153 // domain (16-bar empty floor, growing with content — Track Manager parity).
154 m_hScroll->setOnScroll([this](double val) {
155 const float maxScrollX = horizontalScrollMax();
156 m_targetScrollX = safeClampRange(static_cast<float>(val), 0.0f, maxScrollX);
157 });
158
159 addChild(m_keys);
160 addChild(m_ruler);
161 addChild(m_grid);
162 addChild(m_notes);
163 addChild(m_controls);
164 addChild(m_minimap);
165 addChild(m_vScroll);
166 addChild(m_hScroll);
167 addChild(m_toolbar); // Top (Render Last)
168}

210 const float playheadX = snapVerticalLineX(

212
213 if (m_playheadBeat >= visibleStartBeat && m_playheadBeat <= visibleEndBeat &&
214 playheadX >= gridBounds.x && playheadX <= gridBounds.right()) {
215 const float playheadStartY = rulerBounds.bottom();
216 const float playheadEndY = m_controls ? m_controls->getBounds().bottom() : gridBounds.bottom();
217 Playhead::drawLine(renderer, playheadX, playheadStartY, playheadEndY, accent); // flat, 2 px
218 }
219 }
220
221 if (m_controls) {
222 const auto controlBounds = m_controls->getBounds();
223 const float gripWidth = 34.0f;
224 const NUIRect splitterGrip(controlBounds.x + (controlBounds.width - gripWidth) * 0.5f,
225 controlBounds.y - 2.0f,
226 gripWidth,
227 4.0f);
228 renderer.fillRoundedRect(splitterGrip,
229 2.0f,
230 theme.getColor("accentPrimary").withAlpha(

233

235 if (auto menu = m_toolbar->getActiveContextMenu(); menu && menu->isVisible()) {
236 menu->onRender(renderer);
237 }
238 }
239
240 if (m_showShortcutHelp) {
241 renderShortcutHelp(renderer);
242 }

349void PianoRollView::onUpdate(double deltaTime) {
350 NUIComponent::onUpdate(deltaTime);
351
352 const float totalH = 128 * m_keyHeight;
353 const float visibleH = m_grid ? m_grid->getHeight() : 0.0f;
354 const float maxScrollY = std::max(0.0f, totalH - visibleH);
355 m_targetScrollY = safeClampRange(m_targetScrollY, 0.0f, maxScrollY);
356 m_targetScrollX = safeClampRange(m_targetScrollX, 0.0f, horizontalScrollMax());
357
358 bool changed = false;
359 const float ease = 1.0f - std::exp(-static_cast<float>(deltaTime) * 18.0f);
360
361 const float dx = m_targetScrollX - m_scrollX;
362 if (std::abs(dx) > 0.1f) {
363 m_scrollX += dx * ease;
364 changed = true;
365 } else if (std::abs(dx) > 0.0f) {
366 m_scrollX = m_targetScrollX;
367 changed = true;
368 }
369
370 const float dy = m_targetScrollY - m_scrollY;
371 if (std::abs(dy) > 0.1f) {
372 m_scrollY += dy * ease;
373 changed = true;
374 } else if (std::abs(dy) > 0.0f) {
375 m_scrollY = m_targetScrollY;
376 changed = true;
377 }
378
379 if (changed) {
380 updateScrollbars();
381 syncChildren();
382 }
383}

406void PianoRollView::layoutChildren() {
407 auto b = getBounds();
408 const float sbSize = kOverlayScrollbarThickness;
409
410 // 0. Toolbar — zero once a host panel has taken it into its title bar.
411 float toolbarH = m_toolbarHosted ? 0.0f : kToolbarHeight;
412 if (m_toolbar && !m_toolbarHosted) m_toolbar->setBounds(NUIRect(b.x, b.y, b.width, toolbarH));
413
414 // 1. Scrollbar/Minimap Section (Below Toolbar)
415 float miniMapH = m_showLocalMinimap ? kMinimapHeight : 0.0f;
416
417 // 2. Ruler Section (Below Minimap if present)
418 float rulerH = kRulerHeight;
419
420 // Gap between minimap (overview) and ruler to visually separate them
421 float minimapGap = m_showLocalMinimap ? kMinimapGap : 0.0f;
422
423 float topTotalH = toolbarH + miniMapH + minimapGap + rulerH;
424
425 float keyW = std::max(40.0f, m_keyLaneWidth);
426 const float hScrollH = kOverlayScrollbarThickness; // Horizontal scrollbar row below the grid
427 float contentW = std::max(0.0f, b.width - keyW - sbSize);
428 float contentH = std::max(0.0f, b.height - topTotalH - m_controlPanelHeight - hScrollH); // Subtract control panel
429
430 // 1. Minimap (Top)
431 m_minimap->setVisible(m_showLocalMinimap);
432 if (m_showLocalMinimap) {
433 m_minimap->setBounds(NUIRect(b.x + keyW, b.y + toolbarH, contentW, miniMapH));
434 }
435
436 // 2. Ruler (leave a small gap below minimap)
437 m_ruler->setBounds(NUIRect(b.x + keyW, b.y + toolbarH + miniMapH + minimapGap, contentW, rulerH));
438
439 // 3. Grid/Notes (Below Ruler)
440 NUIRect contentRect(b.x + keyW, b.y + topTotalH, contentW, contentH);
441 m_grid->setBounds(contentRect);
442 m_notes->setBounds(contentRect);
443
444 // 4. Keys (Left, spans Grid height)
445 m_keys->setBounds(NUIRect(b.x, b.y + topTotalH, keyW, contentH));
446
447 // 5. V-Scroll (Right, spans Grid height only)
448 m_vScroll->setBounds(NUIRect(b.x + b.width - sbSize, b.y + topTotalH, sbSize, contentH));
449
450 // 5b. H-Scroll (below the grid, spans the content width)
451 if (m_hScroll) {
452 m_hScroll->setBounds(NUIRect(b.x + keyW, b.y + topTotalH + contentH, contentW, hScrollH));
453 }
454
455 // 6. Control Panel (Bottom) - Spans Full Width (Keys + Content)
456 // Ensures "Control" sidebar aligns with Keys
457 m_controls->setBounds(NUIRect(b.x, b.y + topTotalH + contentH + hScrollH, b.width, m_controlPanelHeight));
458 m_controls->setSidebarWidth(m_keyLaneWidth); // its beat 0 must sit under the grid's
459
460 updateScrollbars();
461 syncChildren();
462}

464float PianoRollView::horizontalScrollMax() const {
465 const float visibleW = m_grid ? m_grid->getWidth() : 0.0f;
466 const float totalW = static_cast<float>(std::max(0.0, m_scrollDomainEndBeats)) * m_pixelsPerBeat;
467 return std::max(0.0f, totalW - visibleW);

470void PianoRollView::updateScrollbarDomain() {
471 // Track Manager parity: 16 bars when empty, dynamic padding with content.
472 const double beatsPerBarD = static_cast<double>(std::max(1, m_beatsPerBar));
473 const double contentEnd = std::max(0.0, m_totalDurationBeats);
474 const bool hasContent = m_notes && !m_notes->getNotes().empty();
475
476 double requiredEnd;
477 if (!hasContent) {
478 // Empty: fixed 16-bar floor (matches the Track Manager timeline).
479 requiredEnd = beatsPerBarD * 16.0;
480 } else {
481 const double clipBars = contentEnd / beatsPerBarD;
482 double padBars;
483 if (clipBars < 16.0) {
484 padBars = 8.0;
485 } else if (clipBars < 64.0) {
486 padBars = clipBars * 0.25;
487 } else {
488 padBars = clipBars * 0.125;
489 }
490 requiredEnd = contentEnd + beatsPerBarD * padBars;
491 }
492
493 if (m_scrollDomainEndBeats <= 0.0 || requiredEnd > m_scrollDomainEndBeats + 1e-3) {
494 m_scrollDomainEndBeats = requiredEnd;
495 } else if (requiredEnd < m_scrollDomainEndBeats - 1e-3) {
496 // Shrink is allowed once content exists (Track Manager parity); the
497 // 16-bar floor applies to the empty case only.
498 m_scrollDomainEndBeats = std::max(requiredEnd, 0.0);
499 }
500}

502void PianoRollView::updateScrollbars() {
503 updateScrollbarDomain();
504
505 float totalBeats = static_cast<float>(m_scrollDomainEndBeats);
506 float visibleW = m_grid->getWidth();
507 double viewDur = visibleW / m_pixelsPerBeat;
508 double start = m_scrollX / m_pixelsPerBeat;
509
510 if (m_minimap && m_showLocalMinimap) {
511 m_minimap->setTotalDuration(totalBeats);
512 m_minimap->setView(start, viewDur);
513 }
514
515 if (m_hScroll) {
516 const double totalW = static_cast<double>(m_scrollDomainEndBeats) * m_pixelsPerBeat;
517 m_hScroll->setRangeLimit(0.0, totalW);
518 m_hScroll->setCurrentRange(m_scrollX, visibleW);
519 }
520
521 // Vertical
522 float totalH = 128 * m_keyHeight;
523 float visibleH = m_grid->getHeight();
524
525 m_vScroll->setRangeLimit(0.0, totalH);
526 m_vScroll->setCurrentRange(m_scrollY, visibleH);
527}

529void PianoRollView::syncChildren() {
530 if (!m_keys) return;
531
532 float x = m_scrollX;
533 float y = m_scrollY;
534
535 m_keys->setScrollOffsetY(y);
536 m_keys->setKeyHeight(m_keyHeight);
537
538 m_ruler->setScrollX(x);
539 m_ruler->setPixelsPerBeat(m_pixelsPerBeat);
540 m_ruler->setPlayheadBeat(m_playheadBeat);
541
542 m_grid->setPixelsPerBeat(m_pixelsPerBeat);
543 m_grid->setKeyHeight(m_keyHeight);
544 m_grid->setScrollOffsetX(x);
545 m_grid->setScrollOffsetY(y);
546 m_grid->setPlayheadBeat(m_playheadBeat);
547 m_grid->setTotalDurationBeats(m_totalDurationBeats);
548
549 m_notes->setPixelsPerBeat(m_pixelsPerBeat);

568bool PianoRollView::onMouseEvent(const NUIMouseEvent& event) {
569 // A release always reaches the children, wherever the button comes up. A drag that began
570 // inside (the minimap bar, a note, the velocity lane) must end even when released outside
571 // the editor: dropping it left the minimap "still clicked", scrolling the view on every
572 // later move (SPEC 3 §2.4). Every child's release path checks its own drag state, so a
573 // release that is not theirs does nothing.
574 if (!getBounds().contains(event.position) && !m_isResizingPanel && !event.released) return false;
575
576 // The shortcut sheet captures the pointer: any click or scroll dismisses it
577 // instead of reaching the editors underneath.
578 if (m_showShortcutHelp) {
579 if (event.pressed || event.wheelDelta != 0.0f) {
580 m_showShortcutHelp = false;
581 repaint();
582 }
583 return true;
584 }
585
586 if (m_toolbar) {
587 if (auto menu = m_toolbar->getActiveContextMenu(); menu && menu->isVisible()) {
588 if (menu->onMouseEvent(event)) return true;
589
590 if (event.pressed && event.button == NUIMouseButton::Left) {
591 const auto menuBounds = menu->getBounds();
592 const auto toolbarBounds = m_toolbar->getBounds();
593 if (!menuBounds.contains(event.position) &&
594 !toolbarBounds.contains(event.position)) {
595 m_toolbar->dismissActiveContextMenu();
596 return true;
597 }
598 }
599 }
600 }
601
602 auto b = getBounds();
603 float splitterY = b.y + b.height - m_controlPanelHeight;
604 float splitterZone = 7.0f;
605 const bool splitterHovered = std::abs(event.position.y - splitterY) < splitterZone;
606 if (m_splitterHovered != splitterHovered) {
607 m_splitterHovered = splitterHovered;
608 repaint();
609 }
610
611 if (event.pressed && event.button == NUIMouseButton::Left) {
612 if (std::abs(event.position.y - splitterY) < splitterZone) {
613 m_isResizingPanel = true;
614 m_dragStartPos = event.position;
615 m_dragStartPanelHeight = m_controlPanelHeight;
616 return true;
617 }
618 }
619 else if (m_isResizingPanel && !event.released) {
620 float dy = event.position.y - m_dragStartPos.y;
621 // Dragging UP increases height
622 float newH = m_dragStartPanelHeight - dy;
623 const float maxPanelHeight = std::max(20.0f, b.height * 0.5f);
624 m_controlPanelHeight = std::min(std::max(newH, 20.0f), maxPanelHeight);
625 layoutChildren();
626 return true;
627 }
628 else if (event.released) {
629 if (m_isResizingPanel) {
630 m_isResizingPanel = false;
631 return true;
632 }
633 }
634
635 // The ruler highlight clears two ways, so it is never a mystery: a left press on the grid OUTSIDE it, or a right press
636 // on the grid (a right-click on the ruler without dragging does too: PianoRollRuler). A press inside it leaves it be,
637 // and whatever the press does anyway (select, add or erase a note) still happens. Esc clears it as well.
638 if (m_loopZone && event.pressed && m_grid && m_pixelsPerBeat > 0.0f) {
639 const NUIRect g = m_grid->getBounds();
640 if (g.contains(event.position)) {
641 const double beat = static_cast<double>(event.position.x - g.x + m_scrollX) / static_cast<double>(m_pixelsPerBeat);
642 const bool outside = beat < m_loopZone->first - 1e-9 || beat >= m_loopZone->second - 1e-9;
643 if (event.button == NUIMouseButton::Right || (event.button == NUIMouseButton::Left && outside)) {
644 applyLoopZone(std::nullopt, true);
645 }
646 }
647 }
648
649 // 1. Give children priority (Ruler, Minimap, Grid, Notes)
650 if (NUIComponent::onMouseEvent(event)) return true;
651
652 // 2. View-level fallback for Grid Scrolling (if Grid didn't handle it)
653 if (event.wheelDelta != 0.0f) {
654 bool shift = (event.modifiers & NUIModifiers::Shift);
655 bool ctrl = (event.modifiers & NUIModifiers::Ctrl);
656
657 if (ctrl) {
658 // Zoom (Fallback) — same anchored, multiplicative semantics as the ruler.
659 // The ruler passes grid-local X (its bounds start after the key lane);
660 // mirror that basis or the beat under the cursor drifts with the lane
661 // width. Bounds are window-absolute, so grid-local X is the event X
662 // minus the grid's own origin — subtracting getBounds() as well would
663 // double-count the view position and desync the anchor whenever the
664 // piano-roll panel sits away from the window origin.
665 const float zoomDelta = std::clamp(event.wheelDelta, -4.0f, 4.0f);
666 applyZoom(std::pow(1.15f, zoomDelta), event.position.x - m_grid->getBounds().x);
667 } else if (shift) {
668 // H-Scroll
669 m_targetScrollX = std::max(0.0f, m_targetScrollX - event.wheelDelta * 40.0f);
670 } else {
671 // V-Scroll
672 float totalH = 128 * m_keyHeight;
673 float visibleH = m_grid->getHeight();
674 float maxScroll = std::max(0.0f, totalH - visibleH);
675
676 float newY = m_targetScrollY - event.wheelDelta * 30.0f;
677 m_targetScrollY = safeClampRange(newY, 0.0f, maxScroll);
678 }
679
680 if (ctrl) {
681 updateScrollbars();
682 syncChildren();
683 }
684 return true;
685 }
686
687 return false;
688}

690bool PianoRollView::onKeyEvent(const NUIKeyEvent& event) {
691 if (event.pressed && event.keyCode == NUIKeyCode::F1) {
692 m_showShortcutHelp = !m_showShortcutHelp;
693 repaint();
694 return true;
695 }
696 if (m_showShortcutHelp && event.pressed) {
697 // While the sheet is up, any key dismisses it rather than editing the
698 // notes it covers.
699 m_showShortcutHelp = false;
700 repaint();
701 return true;
702 }
703 // Esc is the exit: it clears the loop zone, and the note layer still gets it (deselect).
704 if (event.pressed && event.keyCode == NUIKeyCode::Escape && m_loopZone) {
705 applyLoopZone(std::nullopt, true);
706 m_notes->onKeyEvent(event);
707 return true;
708 }
709 // Duplicate with a ruler highlight copies the highlighted block and moves the highlight onto the copy (FL).
710 if (event.pressed && (event.modifiers & NUIModifiers::Ctrl) && event.keyCode == NUIKeyCode::D && m_loopZone) {
711 if (const auto next = m_notes->duplicateLikeFl(m_loopZone)) applyLoopZone(*next, true);
712 return true;
713 }
714 if (m_notes->onKeyEvent(event)) return true;
715 return NUIComponent::onKeyEvent(event);
716}

721 if (zone && zone->second <= zone->first + 1e-6) {

732 }
