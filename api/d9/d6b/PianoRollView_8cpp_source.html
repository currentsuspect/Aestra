15PianoRollView::PianoRollView()
16 : m_keyLaneWidth(76.0f), m_rulerHeight(28.0f), m_pixelsPerBeat(80.0f), m_keyHeight(24.0f),
17 m_scrollX(0.0f), m_targetScrollX(0.0f)
18{
19 // [FIX] Canonical default octave: C3 (MIDI 48).
20 // scrollY = (127 - targetPitch) * keyHeight => (127 - 48) * 24 = 1896
21 m_scrollY = 1896.0f;
22 m_targetScrollY = 1896.0f;
23 m_keys = std::make_shared<PianoRollKeyLane>();
24 m_ruler = std::make_shared<PianoRollRuler>();
25 m_grid = std::make_shared<PianoRollGrid>();
26 m_notes = std::make_shared<PianoRollNoteLayer>();
27 m_controls = std::make_shared<PianoRollControlPanel>();
28
29 m_notes->setOnHoveredPitchChanged([this](int pitch) {
30 if (m_grid) m_grid->setHoveredPitch(pitch);
31 if (m_keys) m_keys->setHoveredKey(pitch);
32 });
33 m_keys->setOnHoveredKeyChanged([this](int pitch) {
34 if (m_grid) m_grid->setHoveredPitch(pitch);
35 });
36
37 // Toolbar
38 m_toolbar = std::make_shared<PianoRollToolbar>();
39 m_toolbar->setGrid(m_grid);
40 m_toolbar->setNoteLayer(m_notes);
41 m_toolbar->setPatternLengthBeats(m_patternLengthBeats);
42 m_toolbar->setOnShowShortcutHelp([this]() {
43 m_showShortcutHelp = !m_showShortcutHelp;
44 repaint();
45 });
46
47 m_controls->setNoteLayer(m_notes);
48
49 m_minimap = std::make_shared<PianoRollMinimap>(); // Local Minimap
50
51 m_vScroll = std::make_shared<NUIScrollbar>(NUIScrollbar::Orientation::Vertical);
52 m_vScroll->setOrientation(NUIScrollbar::Orientation::Vertical);
53 {
54 auto& theme = NUIThemeManager::getInstance();
55 m_vScroll->setArrowSize(0.0f);
56 m_vScroll->setBorderWidth(0.0f);
57 m_vScroll->setBorderRadius(8.0f);
58 m_vScroll->setTrackColor(theme.getColor("surfaceRaised").withAlpha(0.55f));
59 m_vScroll->setThumbColor(theme.getColor("textPrimary").withAlpha(0.30f));
60 m_vScroll->setThumbHoverColor(theme.getColor("textPrimary").withAlpha(0.48f));
61 m_vScroll->setThumbPressedColor(theme.getColor("accentPrimary").withAlpha(0.68f));
62 m_vScroll->setMinimumThumbSize(0.06);
63 }
64
65 // Initial default layout config
66 m_minimap->setVisible(true);
67 m_vScroll->setVisible(true);
68
69 // Ruler Zoom Callback
70 m_ruler->onZoomRequested = [this](float delta, float mouseX) {
71 float oldPPB = m_pixelsPerBeat;
72 float zoomFactor = (delta > 0) ? 1.15f : 0.85f;
73 float newPPB = std::clamp(oldPPB * zoomFactor, 10.0f, 500.0f);
74
75 // Anchor logic: Keep beat under mouse stationary
76 float mouseBeat = (m_scrollX + mouseX) / oldPPB;
77 float newWorldX = mouseBeat * newPPB;
78 float newScrollX = newWorldX - mouseX;
79
80 if (newScrollX < 0) newScrollX = 0;
81
82 m_pixelsPerBeat = newPPB;
83 m_scrollX = newScrollX;
84 m_targetScrollX = newScrollX;
85
86 updateScrollbars();
87 syncChildren();
88 };
89
90 m_ruler->onPlayheadScrubbed = [this](double beat, bool active) {
91 const double clampedBeat = std::clamp(beat, 0.0, m_totalDurationBeats);
92 setPlayheadBeat(clampedBeat, false);
93 if (m_onPlayheadScrubbed) {
94 m_onPlayheadScrubbed(clampedBeat, active);
95 }
96 };
97
98 m_minimap->onViewChanged = [this](double start, double duration) {
99 m_scrollX = static_cast<float>(start * m_pixelsPerBeat);
100 m_targetScrollX = m_scrollX;
101
102 // ZOOM LOGIC:
103 // duration * ppb = visibleWidth
104 // ppb = visibleWidth / duration
105 float visibleW = m_grid->getWidth();
106 if (duration > 0.001) {
107 m_pixelsPerBeat = visibleW / static_cast<float>(duration);
108 }
109
110 syncChildren();
111 };
112
113 m_vScroll->setOnScroll([this](double val) {
114 float totalH = 128 * m_keyHeight;
115 float visibleH = m_grid->getHeight();
116 float maxScroll = std::max(0.0f, totalH - visibleH);
117 m_targetScrollY = safeClampRange(static_cast<float>(val), 0.0f, maxScroll);
118 });
119
120 addChild(m_keys);
121 addChild(m_ruler);
122 addChild(m_grid);
123 addChild(m_notes);
124 addChild(m_controls);
125 addChild(m_minimap);
126 addChild(m_vScroll);
127 addChild(m_toolbar); // Top (Render Last)
128}

320void PianoRollView::onUpdate(double deltaTime) {
321 NUIComponent::onUpdate(deltaTime);
322
323 const float totalH = 128 * m_keyHeight;
324 const float visibleH = m_grid ? m_grid->getHeight() : 0.0f;
325 const float maxScrollY = std::max(0.0f, totalH - visibleH);
326 m_targetScrollY = safeClampRange(m_targetScrollY, 0.0f, maxScrollY);
327 m_targetScrollX = std::max(0.0f, m_targetScrollX);
328
329 bool changed = false;
330 const float ease = 1.0f - std::exp(-static_cast<float>(deltaTime) * 18.0f);
331
332 const float dx = m_targetScrollX - m_scrollX;
333 if (std::abs(dx) > 0.1f) {
334 m_scrollX += dx * ease;
335 changed = true;
336 } else if (std::abs(dx) > 0.0f) {
337 m_scrollX = m_targetScrollX;
338 changed = true;
339 }
340
341 const float dy = m_targetScrollY - m_scrollY;
342 if (std::abs(dy) > 0.1f) {
343 m_scrollY += dy * ease;
344 changed = true;
345 } else if (std::abs(dy) > 0.0f) {
346 m_scrollY = m_targetScrollY;
347 changed = true;
348 }
349
350 if (changed) {
351 updateScrollbars();
352 syncChildren();
353 }
354}

356void PianoRollView::layoutChildren() {
357 auto b = getBounds();
358 float sbSize = 14.0f;
359
360 // 0. Toolbar (Standardized Aestra UI Height)
361 float toolbarH = 50.0f;
362 if (m_toolbar) m_toolbar->setBounds(NUIRect(b.x, b.y, b.width, toolbarH));
363
364 // 1. Scrollbar/Minimap Section (Below Toolbar)
365 float miniMapH = m_showLocalMinimap ? 28.0f : 0.0f;
366
367 // 2. Ruler Section (Below Minimap if present)
368 float rulerH = 28.0f;
369
370 float topTotalH = toolbarH + miniMapH + rulerH;
371
372 float keyW = std::max(40.0f, m_keyLaneWidth);
373 float contentW = std::max(0.0f, b.width - keyW - sbSize);
374 float contentH = std::max(0.0f, b.height - topTotalH - m_controlPanelHeight); // Subtract control panel
375
376 // 1. Minimap (Top)
377 m_minimap->setVisible(m_showLocalMinimap);
378 if (m_showLocalMinimap) {
379 m_minimap->setBounds(NUIRect(b.x + keyW, b.y + toolbarH, contentW, miniMapH));
380 }
381
382 // 2. Ruler
383 m_ruler->setBounds(NUIRect(b.x + keyW, b.y + toolbarH + miniMapH, contentW, rulerH));
384
385 // 3. Grid/Notes (Below Ruler)
386 NUIRect contentRect(b.x + keyW, b.y + topTotalH, contentW, contentH);
387 m_grid->setBounds(contentRect);
388 m_notes->setBounds(contentRect);
389
390 // 4. Keys (Left, spans Grid height)
391 m_keys->setBounds(NUIRect(b.x, b.y + topTotalH, keyW, contentH));
392
393 // 5. V-Scroll (Right, spans Grid height only)
394 m_vScroll->setBounds(NUIRect(b.x + b.width - sbSize, b.y + topTotalH, sbSize, contentH));
395
396 // 6. Control Panel (Bottom) - Spans Full Width (Keys + Content)
397 // Ensures "Control" sidebar aligns with Keys
398 m_controls->setBounds(NUIRect(b.x, b.y + topTotalH + contentH, b.width, m_controlPanelHeight));
399
400 updateScrollbars();
401 syncChildren();
402}

462bool PianoRollView::onMouseEvent(const NUIMouseEvent& event) {
463 if (!getBounds().contains(event.position) && !m_isResizingPanel) return false;
464
465 // The shortcut sheet captures the pointer: any click or scroll dismisses it
466 // instead of reaching the editors underneath.
467 if (m_showShortcutHelp) {
468 if (event.pressed || event.wheelDelta != 0.0f) {
469 m_showShortcutHelp = false;
470 repaint();
471 }
472 return true;
473 }
474
475 if (m_toolbar) {
476 if (auto menu = m_toolbar->getActiveContextMenu(); menu && menu->isVisible()) {
477 if (menu->onMouseEvent(event)) return true;
478
479 if (event.pressed && event.button == NUIMouseButton::Left) {
480 const auto menuBounds = menu->getGlobalBounds();
481 const auto toolbarBounds = m_toolbar->getBounds();
482 if (!menuBounds.contains(event.position) &&
483 !toolbarBounds.contains(event.position)) {
484 m_toolbar->dismissActiveContextMenu();
485 return true;
486 }
487 }
488 }
489 }
490
491 auto b = getBounds();
492 float splitterY = b.y + b.height - m_controlPanelHeight;
493 float splitterZone = 7.0f;
494 const bool splitterHovered = std::abs(event.position.y - splitterY) < splitterZone;
495 if (m_splitterHovered != splitterHovered) {
496 m_splitterHovered = splitterHovered;
497 repaint();
498 }
499
500 if (event.pressed && event.button == NUIMouseButton::Left) {
501 if (std::abs(event.position.y - splitterY) < splitterZone) {
502 m_isResizingPanel = true;
503 m_dragStartPos = event.position;
504 m_dragStartPanelHeight = m_controlPanelHeight;
505 return true;
506 }
507 }
508 else if (m_isResizingPanel && !event.released) {
509 float dy = event.position.y - m_dragStartPos.y;
510 // Dragging UP increases height
511 float newH = m_dragStartPanelHeight - dy;
512 const float maxPanelHeight = std::max(20.0f, b.height * 0.5f);
513 m_controlPanelHeight = std::min(std::max(newH, 20.0f), maxPanelHeight);
514 layoutChildren();
515 return true;
516 }
517 else if (event.released) {
518 if (m_isResizingPanel) {
519 m_isResizingPanel = false;
520 return true;
521 }
522 }
523
524 // 1. Give children priority (Ruler, Minimap, Grid, Notes)
525 if (NUIComponent::onMouseEvent(event)) return true;
526
527 // 2. View-level fallback for Grid Scrolling (if Grid didn't handle it)
528 if (event.wheelDelta != 0.0f) {
529 bool shift = (event.modifiers & NUIModifiers::Shift) || (event.modifiers & NUIModifiers::CapsLock);
530 bool ctrl = (event.modifiers & NUIModifiers::Ctrl);
531
532 if (ctrl) {
533 // Zoom (Fallback)
534 m_pixelsPerBeat = std::max(20.0f, m_pixelsPerBeat + event.wheelDelta * 5.0f);
535 } else if (shift) {
536 // H-Scroll
537 m_targetScrollX = std::max(0.0f, m_targetScrollX - event.wheelDelta * 40.0f);
538 } else {
539 // V-Scroll
540 float totalH = 128 * m_keyHeight;
541 float visibleH = m_grid->getHeight();
542 float maxScroll = std::max(0.0f, totalH - visibleH);
543
544 float newY = m_targetScrollY - event.wheelDelta * 30.0f;
545 m_targetScrollY = safeClampRange(newY, 0.0f, maxScroll);
546 }
547
548 if (ctrl) {
549 updateScrollbars();
550 syncChildren();
551 }
552 return true;
553 }
554
555 return false;
556}

558bool PianoRollView::onKeyEvent(const NUIKeyEvent& event) {
559 if (event.pressed && event.keyCode == NUIKeyCode::F1) {
560 m_showShortcutHelp = !m_showShortcutHelp;
561 repaint();
562 return true;
563 }
564 if (m_showShortcutHelp && event.pressed) {
565 // While the sheet is up, any key dismisses it rather than editing the
566 // notes it covers.
567 m_showShortcutHelp = false;
568 repaint();
569 return true;
570 }
571 if (m_notes->onKeyEvent(event)) return true;
572 return NUIComponent::onKeyEvent(event);
573}

686void PianoRollView::setPlayheadBeat(double beat, bool follow) {
687 m_playheadBeat = std::max(0.0, beat);
688
689 if (follow && m_grid) {
690 const float visibleW = m_grid->getWidth();
691 if (visibleW > 0.0f) {
692 const double visibleStart = static_cast<double>(m_scrollX) / m_pixelsPerBeat;
693 const double visibleDur = static_cast<double>(visibleW) / m_pixelsPerBeat;
694 const double leftGuard = visibleStart + visibleDur * 0.15;
695 const double rightGuard = visibleStart + visibleDur * 0.85;
696
697 if (m_playheadBeat < leftGuard || m_playheadBeat > rightGuard) {
698 const double targetStart = std::max(0.0, m_playheadBeat - visibleDur * 0.2);
699 m_scrollX = static_cast<float>(targetStart * m_pixelsPerBeat);
700 m_targetScrollX = m_scrollX;
701 updateScrollbars();
702 }
703 }
704 }
705
706 syncChildren();
707}

747void PianoRollView::applyEdgeAutoScroll(float scrollX, float scrollY) {
748 const float totalH = 128.0f * m_keyHeight;
749 const float visibleH = m_grid ? m_grid->getHeight() : 0.0f;
750 const float maxScrollY = std::max(0.0f, totalH - visibleH);
751 const float visibleW = m_grid ? m_grid->getWidth() : 0.0f;
752 const double dynamicTotalBeats = std::max(std::max(4.0, m_totalDurationBeats), getViewDurationBeats() + 8.0);
753 const float totalW = static_cast<float>(dynamicTotalBeats) * m_pixelsPerBeat;
754 const float maxScrollX = std::max(0.0f, totalW - visibleW);
755
756 m_scrollX = safeClampScroll(scrollX, maxScrollX);
757 m_scrollY = safeClampScroll(scrollY, maxScrollY);
758 m_targetScrollX = m_scrollX;
759 m_targetScrollY = m_scrollY;
760 updateScrollbars();
761 syncChildren();
762}

775void PianoRollView::setViewWindow(double startBeat, double durationBeats) {
776 const double clampedDuration = std::max(0.25, durationBeats);
777 if (m_grid) {
778 const float visibleW = m_grid->getWidth();
779 if (visibleW > 0.0f) {
