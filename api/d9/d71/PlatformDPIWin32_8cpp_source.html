12bool PlatformDPI::initialize() {
13 // Try to set per-monitor DPI awareness (Windows 10 1703+)
14 typedef BOOL(WINAPI * SetProcessDpiAwarenessContextProc)(DPI_AWARENESS_CONTEXT);
15
16 HMODULE user32 = LoadLibraryA("user32.dll");
17 if (user32) {
18 auto setProcessDpiAwarenessContext =
19 (SetProcessDpiAwarenessContextProc)GetProcAddress(user32, "SetProcessDpiAwarenessContext");
20
21 if (setProcessDpiAwarenessContext) {
22 // Try per-monitor V2 (best - handles non-client area scaling)
23 if (setProcessDpiAwarenessContext(DPI_AWARENESS_CONTEXT_PER_MONITOR_AWARE_V2)) {
24 AESTRA_LOG_INFO("DPI: Per-Monitor V2 awareness enabled");
25 FreeLibrary(user32);
26 return true;
27 }
28
29 // Fall back to per-monitor V1
30 if (setProcessDpiAwarenessContext(DPI_AWARENESS_CONTEXT_PER_MONITOR_AWARE)) {
31 AESTRA_LOG_INFO("DPI: Per-Monitor V1 awareness enabled");
32 FreeLibrary(user32);
33 return true;
34 }
35 }
36 FreeLibrary(user32);
37 }
38
39 // Try SetProcessDpiAwareness (Windows 8.1+)
40 HRESULT hr = SetProcessDpiAwareness(PROCESS_PER_MONITOR_DPI_AWARE);
41 if (SUCCEEDED(hr)) {
42 AESTRA_LOG_INFO("DPI: Per-Monitor awareness enabled (Windows 8.1)");
43 return true;
44 }
45
46 // Fall back to SetProcessDPIAware (Windows Vista+)
47 if (SetProcessDPIAware()) {
48 AESTRA_LOG_INFO("DPI: System awareness enabled (Windows Vista)");
49 return true;
50 }
51
52 AESTRA_LOG_WARNING("DPI: Failed to set DPI awareness - UI may be blurry on high-DPI displays");
53 return false;
54}

61int PlatformDPI::getDPI(HWND hwnd) {
62 // Try GetDpiForWindow (Windows 10 1607+)
63 typedef UINT(WINAPI * GetDpiForWindowProc)(HWND);
64
65 HMODULE user32 = LoadLibraryA("user32.dll");
66 if (user32) {
67 auto getDpiForWindow = (GetDpiForWindowProc)GetProcAddress(user32, "GetDpiForWindow");
68 if (getDpiForWindow) {
69 UINT dpi = getDpiForWindow(hwnd);
70 FreeLibrary(user32);
71 if (dpi > 0) {
72 return static_cast<int>(dpi);
73 }
74 }
75 FreeLibrary(user32);
76 }
77
78 // Fall back to GetDpiForMonitor (Windows 8.1+)
79 if (hwnd) {
80 HMONITOR monitor = MonitorFromWindow(hwnd, MONITOR_DEFAULTTONEAREST);
81 if (monitor) {
82 UINT dpiX = 96, dpiY = 96;
83 if (SUCCEEDED(GetDpiForMonitor(monitor, MDT_EFFECTIVE_DPI, &dpiX, &dpiY))) {
84 return static_cast<int>(dpiX);
85 }
86 }
87 }
88
89 // Fall back to system DPI
90 HDC hdc = GetDC(nullptr);
91 if (hdc) {
92 int dpi = GetDeviceCaps(hdc, LOGPIXELSX);
93 ReleaseDC(nullptr, hdc);
94 return dpi;
95 }
96
97 // Default to 96 DPI (100% scaling)
98 return 96;
99}
