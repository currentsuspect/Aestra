84 ReadHandle readRealtime() const noexcept {
85 // Default-constructed slot: nothing published yet, generation 0 holds null.
86 for (;;) {
87 const int index = m_active.load(std::memory_order_acquire);
88 m_readers[static_cast<size_t>(index)].fetch_add(1, std::memory_order_seq_cst);
89 if (index == m_active.load(std::memory_order_seq_cst)) {
90 return ReadHandle(this, index, m_raw[static_cast<size_t>(index)].load(std::memory_order_acquire));
91 }
92 m_readers[static_cast<size_t>(index)].fetch_sub(1, std::memory_order_release);
93 }
94 }

97 void publish(std::shared_ptr<T> object) {
98 std::lock_guard<std::mutex> lock(m_controlMutex);
99 int target = -1;
100 while (target < 0) {
101 const int active = m_active.load(std::memory_order_relaxed);
102 for (int i = 0; i < kGenerations; ++i) {
103 if (i != active && m_readers[static_cast<size_t>(i)].load(std::memory_order_seq_cst) == 0) {
104 target = i;
105 break;
106 }
107 }
108 if (target < 0) {
109 std::this_thread::yield();
110 }
111 }
112 m_raw[static_cast<size_t>(target)].store(object.get(), std::memory_order_relaxed);
113 m_owned[static_cast<size_t>(target)] = std::move(object);
114 m_active.store(target, std::memory_order_seq_cst);
115 // Let go of every other generation nobody has pinned, so the slot keeps no object
116 // alive beyond the current one. A reader that read a stale index and pins it after
117 // this check fails its re-check (the index is no longer active) and never uses it.
118 for (int i = 0; i < kGenerations; ++i) {
119 if (i != target && m_readers[static_cast<size_t>(i)].load(std::memory_order_seq_cst) == 0) {
120 m_raw[static_cast<size_t>(i)].store(nullptr, std::memory_order_relaxed);
121 m_owned[static_cast<size_t>(i)].reset();
122 }
123 }
124 }
125
127 std::shared_ptr<T> owner() const {
128 std::lock_guard<std::mutex> lock(m_controlMutex);
129 return m_owned[static_cast<size_t>(m_active.load(std::memory_order_relaxed))];
130 }

132private:
133 static constexpr int kGenerations = 3;
134 std::array<std::shared_ptr<T>, kGenerations> m_owned{};
135 std::array<std::atomic<T*>, kGenerations> m_raw{};
136 mutable std::array<std::atomic<uint32_t>, kGenerations> m_readers{};
137 std::atomic<int> m_active{0};
138 mutable std::mutex m_controlMutex;
139};
