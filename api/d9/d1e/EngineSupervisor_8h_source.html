141 struct alignas(64) PluginSlot { // cacheline-aligned to avoid false sharing
142 std::atomic<uint32_t> flags{0};
143
144 bool isBypassed() const noexcept { return flags.load(std::memory_order_relaxed) & 1u; }
145 bool isQuarantined() const noexcept { return flags.load(std::memory_order_relaxed) & 2u; }
146 uint32_t timeoutCount() const noexcept { return (flags.load(std::memory_order_relaxed) >> 2) & 0xFFu; }
147
148 void setBypassed(bool v) noexcept {
149 uint32_t expected = flags.load(std::memory_order_relaxed);
150 uint32_t desired;
151 do {
152 desired = v ? (expected | 1u) : (expected & ~1u);
153 } while (!flags.compare_exchange_weak(expected, desired, std::memory_order_relaxed));
154 }
155
156 void setQuarantined() noexcept {
157 flags.fetch_or(2u, std::memory_order_relaxed);
158 flags.fetch_or(1u, std::memory_order_relaxed); // quarantine implies bypass
159 }
160
161 uint32_t incrementTimeout(uint32_t threshold) noexcept {
162 uint32_t expected = flags.load(std::memory_order_relaxed);
163 uint32_t desired;
164 uint32_t count;
165 do {
166 count = ((expected >> 2) & 0xFFu) + 1;
167 if (count > 255) count = 255;
168 desired = (expected & ~(0xFFu << 2)) | (count << 2);
169 if (count >= threshold) desired |= 1u; // auto-bypass
170 } while (!flags.compare_exchange_weak(expected, desired, std::memory_order_relaxed));
171 return count;
172 }
173
174 void reset() noexcept { flags.store(0, std::memory_order_relaxed); }
