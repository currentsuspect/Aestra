13// The native engine surface (set_bpm, list_units, …) is conceptually the
14// audio.* domain, but those verbs are unprefixed for compatibility — renaming
15// 44 of them would break every existing caller and the published manifest.
16// Reserving the prefix keeps the conceptual model honest: nothing may register
17// a host verb that claims to be engine capability.
18constexpr const char* kReservedDomainPrefix = "audio.";

20bool isLowerCamelIdentifier(const std::string& text) {
21 if (text.empty() || !std::islower(static_cast<unsigned char>(text.front()))) {
22 return false;
23 }
24 return std::all_of(text.begin(), text.end(), [](unsigned char c) {
25 return std::isalnum(c) != 0;
26 });
27}

29// Validate a JSON value against one typed argument. JSON carries real types,
30// so unlike the text-flag path there is nothing to parse — only to check.
31bool valueMatches(const HostVerbArg& arg, const JSON& value, std::string& outError) {
32 switch (arg.type) {
33 case FlagType::Bool:
34 if (!value.isBool()) {
35 outError = "arg '" + arg.name + "' must be a bool";
36 return false;
37 }
38 return true;
39 case FlagType::String:
40 if (!value.isString()) {
41 outError = "arg '" + arg.name + "' must be a string";
42 return false;
43 }
44 return true;
45 case FlagType::Id: {
46 // Canonical 32-hex-char object id, exactly as the list_* queries print
47 // it, so a listed id can be handed straight back.
48 if (!value.isString()) {
49 outError = "arg '" + arg.name + "' must be a 32-character hex id string";
50 return false;
51 }
52 AestraUUID parsed;
53 if (!AestraUUID::tryParse(value.asString(), parsed) || (parsed.low == 0 && parsed.high == 0)) {
54 outError = "arg '" + arg.name + "' must be a 32-character hex id as returned by the list_* queries";
55 return false;
56 }
57 return true;
58 }
59 case FlagType::Int:
60 case FlagType::Float: {
61 if (!value.isNumber()) {
62 outError = "arg '" + arg.name + "' must be a number";
63 return false;
64 }
65 const double number = value.asNumber();
66 if (!std::isfinite(number)) {
67 outError = "arg '" + arg.name + "' must be finite";
68 return false;
69 }
70 if (arg.type == FlagType::Int && number != std::floor(number)) {
71 outError = "arg '" + arg.name + "' must be a whole number";
72 return false;
73 }
74 if (!std::isnan(arg.minValue) && number < arg.minValue) {
75 outError = "arg '" + arg.name + "' is below its minimum of " +
76 std::to_string(arg.minValue);
77 return false;
78 }
79 if (!std::isnan(arg.maxValue) && number > arg.maxValue) {
80 outError = "arg '" + arg.name + "' is above its maximum of " +
81 std::to_string(arg.maxValue);
82 return false;
83 }
84 return true;
85 }
86 }
87 outError = "arg '" + arg.name + "' has an unknown type";
88 return false;
89}

120HostVerbRegistry::RegisterStatus HostVerbRegistry::registerVerb(HostVerbSpec spec,
121 HostVerbHandler handler,
122 std::string& outError) {
123 if (!handler) {
124 outError = "no handler for " + spec.name;
125 return RegisterStatus::NoHandler;
126 }
127
128 if (spec.name.rfind(kReservedDomainPrefix, 0) == 0) {
129 outError = "the audio. domain is the native engine surface and cannot be registered into";
130 return RegisterStatus::ReservedDomain;
131 }
132
133 const std::string prefix = domainPrefixOf(spec.name);
134 if (prefix.empty()) {
135 outError = "host verb '" + spec.name +
136 "' must be named <domain>.<capability>, e.g. settings.setAudioDevice";
137 return RegisterStatus::NameMalformed;
138 }
139 const std::string capability = spec.name.substr(prefix.size() + 1);
140 if (!isLowerCamelIdentifier(capability) || capability.find('.') != std::string::npos) {
141 outError = "'" + capability + "' is not a lowerCamelCase capability name";
142 return RegisterStatus::NameMalformed;
143 }
144 if (prefix != domainName(spec.domain)) {
145 outError = "'" + spec.name + "' is prefixed '" + prefix + "' but declares domain '" +
146 domainName(spec.domain) + "'";
147 return RegisterStatus::DomainMismatch;
148 }
149
150 for (const auto& arg : spec.args) {
151 if (arg.name.empty()) {
152 outError = "'" + spec.name + "' declares an argument with no name";
153 return RegisterStatus::NameMalformed;
154 }
155 }
156
157 if (m_verbs.find(spec.name) != m_verbs.end()) {
158 outError = "'" + spec.name + "' is already registered";
159 return RegisterStatus::Duplicate;
160 }
161
162 const std::string name = spec.name;
163 m_verbs.emplace(name, Entry{std::move(spec), std::move(handler)});
164 return RegisterStatus::Ok;
165}

171HostVerbResult HostVerbRegistry::invoke(const std::string& name, const JSON& args,
172 bool hostUiThreadAvailable) const {
173 const auto it = m_verbs.find(name);
174 if (it == m_verbs.end()) {
175 return HostVerbResult::failure("unknown_verb", "no host verb named '" + name + "'");
176 }
177 const HostVerbSpec& spec = it->second.spec;
178
179 if (spec.affinity == HostThreadAffinity::HostUiThread && !hostUiThreadAvailable) {
180 // Not a marshalling problem: this process has no UI thread to marshal
181 // onto. Say so, rather than running host code somewhere undefined.
182 return HostVerbResult::failure(
183 "host_unavailable",
184 "'" + name + "' needs the application's UI thread; this session is headless");
185 }
186
187 // Args are optional only if every declared argument is optional.
188 JSON empty = JSON::object();
189 const JSON& provided = args.isObject() ? args : empty;
190 if (!args.isNull() && !args.isObject()) {
191 return HostVerbResult::failure("invalid_args", "args must be an object");
192 }
193
194 // Unknown args are refused by name, matching the rest of this surface: a
195 // misspelled argument that is silently dropped runs the verb with a default
196 // the caller never asked for and reports success.
197 //
198 // Checked BEFORE required args on purpose. Typing "framez" for "frames"
199 // otherwise reports the required arg as missing, which sends the caller
200 // looking for an argument they did in fact pass, spelled wrong.
201 // The non-const copy is load-bearing: JSON's const asObject() returns an
202 // EMPTY static, while the non-const overload returns the real map by value.
203 // Inlining provided.asObject() here would make this loop iterate nothing and
204 // silently stop rejecting misspelled args. Same footgun as jsonArgsToFlags
205 // in MuseService.cpp.
206 JSON mutableArgs = provided;
207 for (const auto& entry : mutableArgs.asObject()) {
208 const bool declared =
209 std::any_of(spec.args.begin(), spec.args.end(),
210 [&](const HostVerbArg& arg) { return arg.name == entry.first; });
211 if (!declared) {
212 return HostVerbResult::failure("invalid_args",
213 "unknown arg for " + name + ": " + entry.first);
214 }
215 }
216
217 for (const auto& arg : spec.args) {
218 const bool present = provided.has(arg.name);
219 if (!present) {
220 if (arg.required) {
221 return HostVerbResult::failure("missing_arg",
222 "'" + name + "' requires arg '" + arg.name + "' (" +
223 argTypeName(arg.type) + ")");
224 }
225 continue;
226 }
227 std::string typeError;
228 if (!valueMatches(arg, provided[arg.name], typeError)) {
229 return HostVerbResult::failure("invalid_args", typeError);
230 }
231 }
232
233 return it->second.handler(provided);
234}
