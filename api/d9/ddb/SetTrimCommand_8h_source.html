58 m_channel.setTrimDb(trimDb);
59 if (auto* continuous = m_trackManager.getContinuousParams().get()) {
60 continuous->setTrimDb(m_slotIndex, trimDb);
61 }
62 m_trackManager.requestAudioGraphRebuild(GraphDirtyReason::MixerStateChanged);
63 }
