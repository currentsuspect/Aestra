16constexpr double kPi = 3.14159265358979323846;
17constexpr double kAbsoluteGateLufs = -70.0;
18constexpr double kRelativeGateLu = -10.0; // BS.1770-4 integrated
19constexpr double kRangeRelativeGateLu = -20.0; // EBU Tech 3342 loudness range

21// Transposed direct form II, double precision: the analysis runs offline, so
22// accuracy matters more than speed.
23struct Filter {
24 KWeighting::Biquad c;
25 double z1 = 0.0, z2 = 0.0;
26 double process(double x) {
27 const double y = c.b0 * x + z1;
28 z1 = c.b1 * x - c.a1 * y + z2;
29 z2 = c.b2 * x - c.a2 * y;
30 return y;
31 }
32};

38double powerToLufs(double meanSquare) {
39 return meanSquare > 0.0 ? -0.691 + 10.0 * std::log10(meanSquare) : AudioAnalysis::kSilenceDb;

42// BS.1770 channel weight. 5.1 is L R C LFE Ls Rs: LFE excluded, surrounds +1.5 dB.
43double channelWeight(uint32_t channel, uint32_t channels) {
44 if (channels == 6) {
45 if (channel == 3) return 0.0;
46 if (channel >= 4) return 1.41;
47 }
48 return 1.0;
49}

55// RBJ cookbook Butterworth sections; two in series make a Linkwitz-Riley 4th order.
56KWeighting::Biquad butterworth(double sampleRate, double cutoffHz, bool highPass) {
57 const double w0 = 2.0 * kPi * cutoffHz / sampleRate;
58 const double cosw = std::cos(w0);
59 const double alpha = std::sin(w0) / (2.0 * 0.7071067811865476);
60 const double a0 = 1.0 + alpha;
61 const double b1 = highPass ? -(1.0 + cosw) : (1.0 - cosw);
62 const double b0 = highPass ? (1.0 + cosw) / 2.0 : (1.0 - cosw) / 2.0;
63 return {b0 / a0, b1 / a0, b0 / a0, -2.0 * cosw / a0, (1.0 - alpha) / a0};
64}
65
66struct LinkwitzRiley {
67 Filter a, b;
68 LinkwitzRiley(double sampleRate, double cutoffHz, bool highPass)
69 : a{butterworth(sampleRate, cutoffHz, highPass)}, b{butterworth(sampleRate, cutoffHz, highPass)} {}
70 double process(double x) { return b.process(a.process(x)); }
71};

73// Gated mean of block powers above @p threshold (LUFS), or 0 when none pass.
74double gatedMeanPower(const std::vector<double>& powers, double thresholdLufs) {
75 double sum = 0.0;
76 size_t n = 0;
77 for (double p : powers) {
78 if (powerToLufs(p) > thresholdLufs) {
79 sum += p;
80 ++n;
81 }
82 }
83 return n ? sum / static_cast<double>(n) : 0.0;
84}

86double percentile(std::vector<double> sorted, double p) {
87 std::sort(sorted.begin(), sorted.end());
88 const auto index = static_cast<size_t>(std::lround(p * static_cast<double>(sorted.size() - 1)));
89 return sorted[index];
90}

94AudioAnalysis analyzeAudio(const float* interleaved, uint64_t frames, uint32_t channels, uint32_t sampleRate) {
95 AudioAnalysis out;
96 out.frames = frames;
97 out.channels = channels;
98 out.sampleRate = sampleRate;
99 if (!interleaved || frames == 0 || channels == 0 || sampleRate == 0) {
100 return out;
101 }
102
103 // --- Loudness: K-weighted energy per 100 ms hop, summed over weighted channels.
104 const auto hop = std::max<uint64_t>(1, static_cast<uint64_t>(std::llround(sampleRate * 0.1)));
105 std::vector<Filter> pre(channels, Filter{KWeighting::preFilter(sampleRate)});
106 std::vector<Filter> rlb(channels, Filter{KWeighting::rlbHighPass(sampleRate)});
107 std::vector<double> hopEnergy;
108 hopEnergy.reserve(static_cast<size_t>(frames / hop) + 1);
109 double hopSum = 0.0;
110
111 // --- Peaks and correlation.
112 double samplePeak = 0.0;
113 double sumLL = 0.0, sumRR = 0.0, sumLR = 0.0;
114
115 // --- Bands, on the mid signal.
116 std::vector<LinkwitzRiley> lowPass, highPass;
117 size_t edgesInRange = 0; // edges above Nyquist leave their bands empty
118 for (double edge : AudioAnalysis::kBandEdgesHz) {
119 if (edge >= 0.49 * sampleRate) break;
120 lowPass.emplace_back(sampleRate, edge, false);
121 highPass.emplace_back(sampleRate, edge, true);
122 ++edgesInRange;
123 }
124 std::array<double, AudioAnalysis::kBandCount> bandEnergy{};
125
126 for (uint64_t f = 0; f < frames; ++f) {
127 const float* frame = interleaved + f * channels;
128 for (uint32_t c = 0; c < channels; ++c) {
129 const double x = sanitize(frame[c]);
130 samplePeak = std::max(samplePeak, std::abs(x));
131 const double k = rlb[c].process(pre[c].process(x));
132 hopSum += channelWeight(c, channels) * k * k;
133 }
134 if ((f + 1) % hop == 0) {
135 hopEnergy.push_back(hopSum);
136 hopSum = 0.0;
137 }
138
139 const double l = sanitize(frame[0]);
140 const double r = channels > 1 ? sanitize(frame[1]) : l;
141 sumLL += l * l;
142 sumRR += r * r;
143 sumLR += l * r;
144
145 double rest = channels > 1 ? 0.5 * (l + r) : l;
146 for (size_t e = 0; e < edgesInRange; ++e) {
147 const double below = lowPass[e].process(rest);
148 bandEnergy[e] += below * below;
149 rest = highPass[e].process(rest);
150 }
151 bandEnergy[edgesInRange] += rest * rest;
152 }
153
154 // Momentary (4 hops) and short-term (30 hops) windows on the 100 ms hop.
155 auto windowPowers = [&](size_t hops) {
156 std::vector<double> powers;
157 if (hopEnergy.size() < hops) return powers;
158 double running = 0.0;
159 for (size_t i = 0; i < hopEnergy.size(); ++i) {
160 running += hopEnergy[i];
161 if (i >= hops) running -= hopEnergy[i - hops];
162 if (i + 1 >= hops) powers.push_back(std::max(0.0, running) / static_cast<double>(hops * hop));
163 }
164 return powers;
165 };
166 const std::vector<double> momentary = windowPowers(4);
167 const std::vector<double> shortTerm = windowPowers(30);
168
169 for (double p : momentary) out.maxMomentaryLufs = std::max(out.maxMomentaryLufs, powerToLufs(p));
170 out.shortTermLufs.reserve(shortTerm.size());
171 for (double p : shortTerm) {
172 const double lufs = powerToLufs(p);
173 out.maxShortTermLufs = std::max(out.maxShortTermLufs, lufs);
174 out.shortTermLufs.push_back(static_cast<float>(lufs));
175 }
176
177 // Integrated: absolute gate, then relative gate 10 LU under the absolute-gated mean.
178 const double absoluteMean = gatedMeanPower(momentary, kAbsoluteGateLufs);
179 if (absoluteMean > 0.0) {
180 const double relativeGate = powerToLufs(absoluteMean) + kRelativeGateLu;
181 out.integratedLufs = powerToLufs(gatedMeanPower(momentary, std::max(kAbsoluteGateLufs, relativeGate)));
182 }
183
184 // Loudness range: short-term values past the absolute gate and 20 LU under their mean, P95 - P10.
185 const double rangeMean = gatedMeanPower(shortTerm, kAbsoluteGateLufs);
186 if (rangeMean > 0.0) {
187 const double gate = std::max(kAbsoluteGateLufs, powerToLufs(rangeMean) + kRangeRelativeGateLu);
188 std::vector<double> kept;
189 for (double p : shortTerm) {
190 const double lufs = powerToLufs(p);
191 if (lufs > gate) kept.push_back(lufs);
192 }
193 if (kept.size() >= 2) out.loudnessRangeLu = percentile(kept, 0.95) - percentile(kept, 0.10);
194 }
195
196 // True peak: the same 4x oversampling meter the engine uses, one channel at a time.
197 {
198 std::vector<float> channel(static_cast<size_t>(frames));
199 double truePeak = 0.0;
200 for (uint32_t c = 0; c < channels; ++c) {
201 for (uint64_t f = 0; f < frames; ++f) channel[f] = static_cast<float>(sanitize(interleaved[f * channels + c]));
202 TruePeakMeter meter;
203 meter.initialize(sampleRate);
204 constexpr uint64_t kChunk = 1u << 20;
205 for (uint64_t start = 0; start < frames; start += kChunk) {
206 meter.processMono(channel.data() + start, static_cast<uint32_t>(std::min(kChunk, frames - start)));
207 }
208 truePeak = std::max(truePeak, static_cast<double>(meter.getMaxTruePeak()));
209 }
210 out.truePeakDbtp = toDb(std::max(truePeak, samplePeak));
211 }
212 out.samplePeakDbfs = toDb(samplePeak);
213
214 if (channels == 1) {
215 out.correlation = samplePeak > 0.0 ? 1.0 : 0.0;
216 } else if (sumLL > 0.0 && sumRR > 0.0) {
217 out.correlation = std::clamp(sumLR / std::sqrt(sumLL * sumRR), -1.0, 1.0);
218 }
219
220 double totalBandEnergy = 0.0;
221 for (double e : bandEnergy) totalBandEnergy += e;
222 for (size_t b = 0; b < AudioAnalysis::kBandCount; ++b) {
223 out.bandRmsDbfs[b] = toDb(std::sqrt(bandEnergy[b] / static_cast<double>(frames)));
224 out.bandShare[b] = totalBandEnergy > 0.0 ? bandEnergy[b] / totalBandEnergy : 0.0;
225 }
226 return out;
227}

229std::shared_ptr<const AudioAnalysis> AudioAnalysisCache::getOrCompute(const std::string& key, const Compute& compute) {
230 if (auto hit = find(key)) return hit;
231 auto computed = std::make_shared<const AudioAnalysis>(compute());
232 std::lock_guard<std::mutex> lock(mutex_);
233 if (auto it = entries_.find(key); it != entries_.end()) return it->second; // another caller won the race
234 while (entries_.size() >= capacity_ && !order_.empty()) {
235 entries_.erase(order_.front());
236 order_.erase(order_.begin());
237 }
238 entries_.emplace(key, computed);
239 order_.push_back(key);
240 return computed;
241}

243std::shared_ptr<const AudioAnalysis> AudioAnalysisCache::find(const std::string& key) const {
244 std::lock_guard<std::mutex> lock(mutex_);
245 auto it = entries_.find(key);
246 return it == entries_.end() ? nullptr : it->second;
247}
