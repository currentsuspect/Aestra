57inline FileBrowserLocalLayout resolveFileBrowserLayout(float width, float height, const FileBrowserBands& bands,
58 float searchTrailing, Layout::NUILayoutRecorder* trace = nullptr) {
59 using namespace Layout;
60 const auto at = [trace](const char* scope, const char* from) {
61 if (trace) trace->scope(scope, from);
62 };
63 constexpr float kPad = 10.0f;
64 FileBrowserLocalLayout l;
65
66 at("kind", "");
67 const auto kind = splitVertical(NUILocalRect(0.0f, 0.0f, width, height), bands.kindRow, trace);
68 at("search", "kind.trailing");
69 const auto search = splitVertical(kind.trailing, bands.searchRow, trace);
70 at("location", "search.trailing");
71 const auto location = splitVertical(search.trailing, bands.locationRow, trace);
72 at("chips", "location.trailing");
73 const auto chips = splitVertical(location.trailing, bands.chipsRow, trace);
74 at("header", "chips.trailing");
75 const auto header = splitVertical(chips.trailing, bands.listHeader, trace);
76 at("list", "header.trailing");
77 const auto list = splitVertical(header.trailing, std::max(0.0f, header.trailing.height - bands.preview), trace);
78
79 l.kindRow = kind.leading;
80 // Three equal tabs inside a 28 px segment, 8 px from the top, 2 px inset.
81 const float segW = std::max(0.0f, width - kPad * 2.0f);
82 const float tabW = std::max(0.0f, (segW - 4.0f - 4.0f) / 3.0f);
83 for (int i = 0; i < 3; ++i) {
84 l.kinds[static_cast<size_t>(i)] =
85 NUILocalRect(kPad + 2.0f + static_cast<float>(i) * (tabW + 2.0f), 10.0f, tabW, 24.0f);
86 }
87
88 l.searchBar = search.leading;
89 l.search = NUILocalRect(kPad + 24.0f, search.leading.y + 4.0f,
90 std::max(0.0f, width - kPad - 24.0f - searchTrailing), 28.0f);
91
92 l.location = location.leading;
93 if (bands.locationRow > 0.0f) {
94 l.back = NUILocalRect(kPad - 4.0f, location.leading.y + 4.0f, 22.0f, 22.0f);
95 l.path = NUILocalRect(l.back.right() + 4.0f, location.leading.y + 4.0f,
96 std::max(0.0f, width - l.back.right() - 4.0f - kPad), 22.0f);
97 }
98 l.chips = chips.leading;
99
100 l.listHeader = header.leading;
101 if (bands.listHeader > 0.0f) {
102 const NUILocalRect controls(0.0f, header.leading.y + 2.0f, width, 22.0f);
103 const float sortWidth = width >= 240.0f ? 62.0f : 22.0f;
104 const auto right = arrangeTrailingRow(controls, {sortWidth, 22.0f}, 22.0f, 2.0f, kPad - 4.0f, trace);
105 l.sort = right[0];
106 l.filter = right[1];
107 }
108 l.list = list.leading;
109 l.body = search.trailing;
110
111 // Places drops over everything under the location band.
112 const float overlayTop = location.leading.bottom();
113 l.navPane = NUILocalRect(0.0f, overlayTop, width, std::max(0.0f, height - overlayTop));
114 const float navHeaderH = std::min(l.navPane.height, 34.0f);
115 l.navHeader = NUILocalRect(0.0f, overlayTop, width, navHeaderH);
116 l.navViewport = NUILocalRect(0.0f, overlayTop + navHeaderH, width, std::max(0.0f, l.navPane.height - navHeaderH));
117 return l;
118}
