4// V8-X2b phase 5: the file browser's panes and header chrome, resolved once by the
5// layout system. Before this, the folder pane's clipped, scrollable region was
6// computed twice — once where it was drawn and once where it was hit-tested — under
7// a comment saying the two must match, after rows clipped under the header had been
8// clickable. Now both read `navViewport`.
9//
10// Local space (origin at the browser's top-left); FileBrowser::computeBrowserLayout()
11// converts to window space once. Header-only, no widget dependencies, so a test can

37inline FileBrowserLocalLayout resolveFileBrowserLayout(float width, float height, float searchRowHeight,
38 float searchTrailing, float navWidth, float listHeaderHeight,
39 float previewHeight, Layout::NUILayoutRecorder* trace = nullptr) {
40 using namespace Layout;
41 const auto at = [trace](const char* scope, const char* from) {
42 if (trace) trace->scope(scope, from);
43 };
44 FileBrowserLocalLayout l;
45 at("rows", "");
46 const auto rows = splitVertical(NUILocalRect(0.0f, 0.0f, width, height), searchRowHeight, trace);
47 at("cols", "rows.trailing");
48 const auto cols = splitHorizontal(rows.trailing, navWidth, trace);
49 at("nav", "cols.leading");
50 const auto nav = splitVertical(cols.leading, listHeaderHeight, trace);
51 at("listCol", "cols.trailing");
52 const auto listCol = splitVertical(cols.trailing, listHeaderHeight, trace);
53 at("list", "listCol.trailing");
54 const auto list = splitVertical(listCol.trailing, listCol.trailing.height - previewHeight, trace);
55
56 // Header chrome sits 5 px down in a 24 px row: back · forward · up from the
57 // left, sort · filter from the right, each 5 px in and 2 px apart.
58 const NUILocalRect chrome(listCol.leading.x, listCol.leading.y + 5.0f, listCol.leading.width, 24.0f);
59 at("navButtons", "listCol.leading");
60 const auto left = arrangeLeadingRow(chrome, {22.0f, 22.0f, 22.0f}, 24.0f, 2.0f, 5.0f, trace);
61 at("tools", "listCol.leading");
62 const float sortWidth = listCol.leading.width >= 300.0f ? 62.0f : 22.0f;
63 const auto right = arrangeTrailingRow(chrome, {sortWidth, 22.0f}, 24.0f, 2.0f, 5.0f, trace);
64
65 l.searchBar = rows.leading;
66 l.search = NUILocalRect(26.0f, 4.0f, width - searchTrailing > 0.0f ? width - searchTrailing : 0.0f,
67 searchRowHeight - 8.0f);
68 l.navPane = cols.leading;
69 l.navHeader = nav.leading;
70 l.navViewport = nav.trailing;
71 l.listHeader = listCol.leading;
72 l.list = list.leading;
73 l.back = left[0];
74 l.forward = left[1];
75 l.up = left[2];
76 l.sort = right[0];
77 l.filter = right[1];
78 return l;
79}
