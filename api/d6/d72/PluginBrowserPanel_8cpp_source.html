67std::string fitText(NUIRenderer& renderer, const std::string& text, float fontSize, float maxWidth) {
68 if (text.empty() || renderer.measureText(text, fontSize).width <= maxWidth) {
69 return text;
70 }
71
72 constexpr const char* ellipsis = "...";
73 const float ellipsisW = renderer.measureText(ellipsis, fontSize).width;
74 if (ellipsisW >= maxWidth) {
75 return ellipsis;
76 }
77
78 std::string out = text;
79 while (!out.empty() && renderer.measureText(out, fontSize).width + ellipsisW > maxWidth) {
80 out.pop_back();
81 }
82 return out + ellipsis;
83}

86 const char* home = getenv("HOME");

103void PluginBrowserPanel::loadFavorites() {
104 m_favoritesSet.clear();
105 auto path = getFavoritesPath();
106 std::ifstream f(path);
107 if (!f.is_open()) return;
108 std::string content((std::istreambuf_iterator<char>(f)), std::istreambuf_iterator<char>());
109 f.close();
110 if (content.empty()) return;
111 try {
112 Aestra::JSON j = Aestra::JSON::parse(content);
113 if (j.has("favorites") && j["favorites"].isArray()) {
114 auto arr = j["favorites"].asArray();
115 for (size_t i = 0; i < arr.size(); ++i) {
116 if (arr[i].isString()) {
117 m_favoritesSet.insert(arr[i].asString());
118 }
119 }
120 }
121 } catch (...) {}
122}

124void PluginBrowserPanel::saveFavorites() {
125 Aestra::JSON j = Aestra::JSON::object();
126 j.set("version", Aestra::JSON(1.0));
127 Aestra::JSON arr = Aestra::JSON::array();
128 for (const auto& id : m_favoritesSet) {
129 arr.push(Aestra::JSON(id));
130 }
131 j.set("favorites", arr);
132 std::ofstream f(getFavoritesPath());
133 if (f.is_open()) f << j.toString(2);
134}

173void PluginBrowserPanel::renderFilterBar(NUIRenderer& renderer) {
174 auto bounds = getBounds();
175 float barY = bounds.y + HEADER_BAR_HEIGHT;
176 m_filterPillHits.clear();
177
178 renderer.fillRect({bounds.x, barY, bounds.width, FILTER_BAR_HEIGHT}, Colors::panelTop);
179 renderer.drawLine({bounds.x, barY + FILTER_BAR_HEIGHT}, {bounds.right(), barY + FILTER_BAR_HEIGHT}, 1.0f, Colors::divider);
180
181 float x = bounds.x + 12.0f;
182 const float y = barY + 7.0f;
183 const float pillH = 22.0f;
184 const float gap = 6.0f;
185
186 auto drawPill = [&](const std::string& label, bool active, FilterPillHit::Type type) {
187 float w = std::max(32.0f, renderer.measureText(label, 11.0f).width + 20.0f);
188 NUIRect rect = {x, y, w, pillH};
189
190 if (active) {
191 renderer.fillRoundedRect(rect, 11.0f, Colors::pillActiveBg);
192 } else {
193 renderer.strokeRoundedRect(rect, 11.0f, 1.0f, Colors::pillInactiveBorder);
194 }
195
196 NUIColor textColor = active ? Colors::pillActiveText : Colors::pillInactiveText;
197 auto measured = renderer.measureText(label, 11.0f);
198 renderer.drawText(label, {rect.x + (rect.width - measured.width) * 0.5f, rect.y + 4.5f}, 11.0f, textColor);
199
200 m_filterPillHits.push_back({type, rect});
201 x += w + gap;
202 };
203
204 bool allActive = (m_typeFilter == PluginTypeFilter::All && m_formatFilter == PluginFormatFilter::All && !m_showFavoritesOnly);
205 drawPill("All", allActive, FilterPillHit::TypeAll);
206 drawPill("FX", m_typeFilter == PluginTypeFilter::Effects, FilterPillHit::TypeFX);
207 drawPill("Inst", m_typeFilter == PluginTypeFilter::Instruments, FilterPillHit::TypeInst);
208
209 x += 4.0f;
210
211 drawPill("VST3", m_formatFilter == PluginFormatFilter::VST3, FilterPillHit::FormatVST3);
212 drawPill("CLAP", m_formatFilter == PluginFormatFilter::CLAP, FilterPillHit::FormatCLAP);
213
214 x += 4.0f;
215
216 drawPill(m_showFavoritesOnly ? "\xe2\x98\x85 Faves" : "\xe2\x98\x85", m_showFavoritesOnly, FilterPillHit::Fav);
217}

219void PluginBrowserPanel::renderPluginList(NUIRenderer& renderer) {
220 auto bounds = getBounds();
221 float listTop = bounds.y + HEADER_BAR_HEIGHT + FILTER_BAR_HEIGHT + 4.0f;
222 float listHeight = bounds.height - HEADER_BAR_HEIGHT - FILTER_BAR_HEIGHT - 4.0f;
223
224 renderer.setClipRect({bounds.x + 8.0f, listTop, bounds.width - 16.0f, listHeight});
225
226 // Rebuild star rects each render
227 m_starRects.assign(m_filteredPlugins.size(), NUIRect{});
228
229 float yOffset = listTop - m_scrollOffset;
230
231 for (size_t i = 0; i < m_filteredPlugins.size(); ++i) {
232 if (yOffset + ROW_HEIGHT > listTop && yOffset < listTop + listHeight) {
233 renderPluginRow(renderer, m_filteredPlugins[i], static_cast<int>(i), yOffset);
234 }
235 yOffset += ROW_HEIGHT;
236 }
237
238 if (m_filteredPlugins.empty() && !m_scanning) {
239 float centerY = listTop + listHeight * 0.5f;
240 renderer.drawText("No plugins found",
241 {bounds.x + bounds.width * 0.5f - 55, centerY - 14},
242 13.0f, Colors::textPrimary.withAlpha(0.85f));
243 renderer.drawText("Click Scan to discover installed plugins",
244 {bounds.x + bounds.width * 0.5f - 105, centerY + 4},
245 11.0f, Colors::textSecondary.withAlpha(0.40f));
246
247 // Show Scan button only if never scanned (allPlugins empty)
248 if (m_allPlugins.empty()) {
249 NUIRect scanBtn = {bounds.x + bounds.width * 0.5f - 30, centerY + 22, 60, 24};
250 renderer.fillRoundedRect(scanBtn, 6.0f, Colors::accentPrimary.withAlpha(0.25f));
251 renderer.strokeRoundedRect(scanBtn, 6.0f, 1.0f, Colors::accentPrimary.withAlpha(0.50f));
252 auto label = renderer.measureText("Scan", 11.0f);
253 renderer.drawText("Scan", {scanBtn.x + (scanBtn.width - label.width) * 0.5f, scanBtn.y + 5.5f}, 11.0f, Colors::accentPrimary);
254 }
255 }
256
257 renderer.clearClipRect();

260void PluginBrowserPanel::renderPluginRow(NUIRenderer& renderer,
261 const PluginListItem& plugin,
262 int index, float yOffset) {
263 auto bounds = getBounds();
264 NUIRect rowRect = {bounds.x + 10, yOffset + 2.0f, bounds.width - 20, ROW_HEIGHT - 4.0f};
265
266 // Row background
267 if (index == m_selectedIndex) {
268 renderer.fillRoundedRect(rowRect, 6.0f, Colors::listSelected);
269 renderer.fillRoundedRect({rowRect.x, rowRect.y + 4.0f, 3.0f, rowRect.height - 8.0f},
270 1.5f,
271 Colors::accentPrimary.withAlpha(0.85f));
272 } else if (index == m_hoveredIndex) {
273 renderer.fillRoundedRect(rowRect, 6.0f, Colors::listHover);
274 }
275
276 // Star rect (for hit-testing)
277 float starX = rowRect.x + 4;
278 float starY = rowRect.y + (rowRect.height - 14) * 0.5f;
279 float starSize = 14.0f;
280 m_starRects[index] = {starX, starY, starSize, starSize};
281
282 // Render star
283 if (plugin.isFavorite) {
284 renderer.drawText("\xe2\x98\x85", {starX + 1, starY + 1}, 12.0f, Colors::starActive);
285 } else if (index == m_hoveredRow) {
286 renderer.drawText("\xe2\x98\x85", {starX + 1, starY + 1}, 12.0f, Colors::starGhost);
287 }
288
289 // Type dot
290 float dotX = rowRect.x + 22;
291 float dotY = rowRect.y + (rowRect.height - 8) * 0.5f;
292 NUIColor dotColor = Colors::textDisabled.withAlpha(0.3f);
293 if (plugin.typeName == "Effect") dotColor = Colors::typeEffect;
294 else if (plugin.typeName == "Instrument") dotColor = Colors::typeInstrument;
295 else if (plugin.typeName == "Analyzer") dotColor = Colors::typeAnalyzer;
296 else if (plugin.typeName == "MIDI") dotColor = Colors::typeMidi;
297 renderer.fillRoundedRect({dotX, dotY, 8.0f, 8.0f}, 4.0f, dotColor);
298
299 // Name
300 float nameX = dotX + 14;
301 const float badgeW = 28.0f; // reserved for format badge
302 const float nameMaxW = rowRect.width - (nameX - rowRect.x) - badgeW - 8.0f;
303 std::string name = fitText(renderer, plugin.name, 13.0f, nameMaxW);
304 renderer.drawText(name, {nameX, rowRect.y + 5}, 13.0f, Colors::textPrimary.withAlpha(0.90f));
305
306 // Format badge (right-aligned)
307 float badgeX = rowRect.right() - badgeW - 4;
308 float badgeY = rowRect.y + (rowRect.height - 16) * 0.5f;
309 NUIRect badgeRect = {badgeX, badgeY, 0, 16};
310 NUIColor badgeBg = Colors::badgeIntBg;
311 NUIColor badgeText = Colors::badgeIntText;
312 std::string badgeLabel = "Int";
313 if (plugin.formatStr == "VST3") {
314 badgeLabel = "V3";
315 badgeBg = Colors::badgeVst3Bg;
316 badgeText = Colors::badgeVst3Text;
317 } else if (plugin.formatStr.find("CLAP") != std::string::npos) {
318 badgeLabel = "CL";
319 badgeBg = Colors::badgeClapBg;
320 badgeText = Colors::badgeClapText;
321 }
322 float badgeLabelW = renderer.measureText(badgeLabel, 9.0f).width;
323 badgeRect.width = badgeLabelW + 10;
324 badgeRect.x = rowRect.right() - badgeRect.width - 4;
325 renderer.fillRoundedRect(badgeRect, 4.0f, badgeBg);
326 renderer.drawText(badgeLabel, {badgeRect.x + (badgeRect.width - badgeLabelW) * 0.5f, badgeRect.y + 2.5f}, 9.0f, badgeText);
327
328 // Vendor · type (muted, between name and badge)
329 float vendorMaxW = badgeRect.x - nameX - 12;
330 std::string vendorMeta = plugin.vendor;
331 if (!plugin.typeName.empty()) vendorMeta += " · " + plugin.typeName;
332 vendorMeta = fitText(renderer, vendorMeta, 10.0f, vendorMaxW);
333 renderer.drawText(vendorMeta, {nameX, rowRect.y + 19}, 10.0f, Colors::textSecondary.withAlpha(0.45f));
334}

355bool PluginBrowserPanel::onMouseEvent(const NUIMouseEvent& event) {
356 // Early exit if not visible - don't lock mutex or consume events
357 if (!isVisible()) return false;
358
359 std::lock_guard<std::recursive_mutex> lock(m_uiMutex);
360 auto bounds = getBounds();
361 float mx = event.position.x;
362 float my = event.position.y;
363 const bool insideBounds = bounds.contains(event.position);
364
365 // Click handling
366 if (event.pressed && event.button == NUIMouseButton::Left) {
367 // Filter pill clicks
368 for (size_t i = 0; i < m_filterPillHits.size(); ++i) {
369 if (m_filterPillHits[i].bounds.contains(event.position)) {
370 m_searchFocused = false;
371 switch (m_filterPillHits[i].type) {
372 case FilterPillHit::TypeAll:
373 m_typeFilter = PluginTypeFilter::All;
374 m_formatFilter = PluginFormatFilter::All;
375 m_showFavoritesOnly = false;
376 break;
377 case FilterPillHit::TypeFX:
378 m_typeFilter = (m_typeFilter == PluginTypeFilter::Effects) ? PluginTypeFilter::All : PluginTypeFilter::Effects;
379 break;
380 case FilterPillHit::TypeInst:
381 m_typeFilter = (m_typeFilter == PluginTypeFilter::Instruments) ? PluginTypeFilter::All : PluginTypeFilter::Instruments;
382 break;
383 case FilterPillHit::FormatVST3:
384 m_formatFilter = (m_formatFilter == PluginFormatFilter::VST3) ? PluginFormatFilter::All : PluginFormatFilter::VST3;
385 break;
386 case FilterPillHit::FormatCLAP:
387 m_formatFilter = (m_formatFilter == PluginFormatFilter::CLAP) ? PluginFormatFilter::All : PluginFormatFilter::CLAP;
388 break;
389 case FilterPillHit::Fav:
390 m_showFavoritesOnly = !m_showFavoritesOnly;
391 break;
392 default:
393 break;
394 }
395 applyFilters();
396 repaint();
397 return true;
398 }
399 }
400
401 // Plugin list clicks (Press down)
402 float listTop = bounds.y + HEADER_BAR_HEIGHT + FILTER_BAR_HEIGHT + 4.0f;
403 if (insideBounds && my >= listTop) {
404 int rowIndex = hitTestRow(static_cast<int>(my));
405 if (rowIndex >= 0 && rowIndex < static_cast<int>(m_filteredPlugins.size())) {
406 // Check star hit first
407 if (rowIndex < static_cast<int>(m_starRects.size()) && m_starRects[rowIndex].contains(mx, my)) {
408 toggleFavorite(m_filteredPlugins[rowIndex].id);
409 repaint();
410 return true;
411 }
412 m_searchFocused = false;
413 repaint();
414 // Initiate potential drag or click
415 m_isPressed = true;
416 m_pressedIndex = rowIndex;
417 m_dragStartPos = event.position;
418 return true;
419 }
420 }
421
422 // Scan button (ignore if already scanning)
423 NUIRect scanBtn = getScanButtonRect();
424 if (insideBounds &&
425 mx >= scanBtn.x && mx < scanBtn.x + scanBtn.width &&
426 my >= scanBtn.y && my < scanBtn.y + scanBtn.height) {
427 if (!m_scanning && m_onScanRequested) {
428 m_onScanRequested();
429 }
430 return true;
431 }
432 if (insideBounds) {
433 m_searchFocused = false;
434 repaint();
435 }
436 }
437 // Handle Release (Click)
438 else if (event.released && event.button == NUIMouseButton::Left) {
439 if (m_isPressed) {
440 // Was a click
441 if (m_pressedIndex >= 0 && m_pressedIndex < static_cast<int>(m_filteredPlugins.size())) {
442 // Check for double-click (same index within 300ms)
443 auto now = std::chrono::steady_clock::now();
444 double elapsed = std::chrono::duration<double>(now.time_since_epoch()).count() - m_lastClickTime;
445
446 if (m_pressedIndex == m_lastClickIndex && elapsed < 0.3) {
447 // Double-click detected - trigger load
448 if (m_onPluginLoadRequested) {
449 m_onPluginLoadRequested(m_filteredPlugins[m_pressedIndex]);
450 }
451 m_lastClickIndex = -1; // Reset to prevent triple-click triggering
452 m_lastClickTime = 0.0;
453 } else {
454 // Single click - select and record for double-click detection
455 m_selectedIndex = m_pressedIndex;
456 if (m_onPluginSelected) {
457 m_onPluginSelected(m_filteredPlugins[m_pressedIndex]);
458 }
459 m_lastClickIndex = m_pressedIndex;
460 m_lastClickTime = std::chrono::duration<double>(now.time_since_epoch()).count();
461 }
462 }
463 m_isPressed = false;
464 m_pressedIndex = -1;
465 return true;
466 }
467 }
468 // Handle Drag
469 else if (!event.pressed && !event.released) { // Mouse Move
470 if (m_isPressed) {
471 float dx = event.position.x - m_dragStartPos.x;
472 float dy = event.position.y - m_dragStartPos.y;
473 float dist = std::sqrt(dx*dx + dy*dy);
474
475 if (dist > 5.0f) {
476 // Start Drag
477 if (m_pressedIndex >= 0 && m_pressedIndex < static_cast<int>(m_filteredPlugins.size())) {
478 const auto& plugin = m_filteredPlugins[m_pressedIndex];
479
480 AestraUI::DragData data;
481 data.type = AestraUI::DragDataType::Plugin;
482 data.displayName = plugin.name;
483 data.sourceClipIdString = plugin.id;
484 data.customData = plugin;
485
486 AestraUI::NUIDragDropManager::getInstance().beginDrag(data, m_dragStartPos, this);
487
488 // Consume interaction
489 m_isPressed = false;
490 m_pressedIndex = -1;
491 return true;
492 }
493 }
494 }
495 }
496
497 // Hover tracking
498 float listTop = bounds.y + HEADER_BAR_HEIGHT + FILTER_BAR_HEIGHT + 4.0f;
499 if (insideBounds && my >= listTop) {
500 m_hoveredIndex = hitTestRow(static_cast<int>(my));
501 m_hoveredRow = m_hoveredIndex;
502 } else {
503 m_hoveredIndex = -1;
504 m_hoveredRow = -1;
505 }
506
507 // Scroll handling
508 if (insideBounds && event.wheelDelta != 0.0f) {
509 float listHeight = bounds.height - HEADER_BAR_HEIGHT - FILTER_BAR_HEIGHT - 4.0f;
510 float contentHeight = m_filteredPlugins.size() * ROW_HEIGHT;
511 float maxScroll = std::max(0.0f, contentHeight - listHeight);
512
513 m_targetScrollOffset -= event.wheelDelta * 40.0f;
514 if (m_targetScrollOffset < 0.0f) m_targetScrollOffset = 0.0f;
515 if (m_targetScrollOffset > maxScroll) m_targetScrollOffset = maxScroll;
516 return true;
517 }
518
519 // Consume event if mouse is within our bounds to prevent click-through
520 return insideBounds;
521}

523bool PluginBrowserPanel::onKeyEvent(const NUIKeyEvent& event) {
524 if (!isVisible() || !m_searchFocused || !event.pressed) return false;
525
526 std::lock_guard<std::recursive_mutex> lock(m_uiMutex);
527
528 if (event.keyCode == NUIKeyCode::Escape) {
529 m_searchFocused = false;
530 setFocused(false);
531 repaint();
532 return true;
533 }
534 if (event.keyCode == NUIKeyCode::Backspace || event.keyCode == NUIKeyCode::Delete) {
535 if (!m_searchQuery.empty()) {
536 m_searchQuery.pop_back();
537 applyFilters();
538 repaint();
539 }
540 return true;
541 }
542 if (event.keyCode == NUIKeyCode::Enter) {
543 m_searchFocused = false;
544 setFocused(false);
545 repaint();
546 return true;
547 }
548
549 if (event.character >= 32 && event.character <= 126) {
550 m_searchQuery.push_back(event.character);
551 applyFilters();
552 repaint();
553 return true;
554 }
555 return false;
556}

602void PluginBrowserPanel::applyFilters() {
603 m_filteredPlugins.clear();
604
605 for (const auto& p : m_allPlugins) {
606 // Type filter
607 bool passType = true;
608 if (m_typeFilter == PluginTypeFilter::Effects) {
609 passType = (p.typeName == "Effect");
610 } else if (m_typeFilter == PluginTypeFilter::Instruments) {
611 passType = (p.typeName == "Instrument");
612 }
613 if (!passType) continue;
614
615 // Format filter
616 bool passFormat = true;
617 if (m_formatFilter == PluginFormatFilter::VST3) {
618 passFormat = (p.formatStr == "VST3");
619 } else if (m_formatFilter == PluginFormatFilter::CLAP) {
620 passFormat = (p.formatStr.find("CLAP") != std::string::npos);
621 }
622 if (!passFormat) continue;
623
624 // Favorites filter
625 if (m_showFavoritesOnly && !p.isFavorite) continue;
626
627 // Search filter
628 if (!m_searchQuery.empty()) {
629 std::string lowerQuery = m_searchQuery;
630 std::transform(lowerQuery.begin(), lowerQuery.end(), lowerQuery.begin(), ::tolower);
631
632 std::string lowerName = p.name;
633 std::transform(lowerName.begin(), lowerName.end(), lowerName.begin(), ::tolower);
634
635 std::string lowerVendor = p.vendor;
636 std::transform(lowerVendor.begin(), lowerVendor.end(), lowerVendor.begin(), ::tolower);
637
638 if (lowerName.find(lowerQuery) == std::string::npos &&
639 lowerVendor.find(lowerQuery) == std::string::npos) {
640 continue;
641 }
642 }
643
644 m_filteredPlugins.push_back(p);
645 }
646
647 if (m_selectedIndex >= static_cast<int>(m_filteredPlugins.size())) {
648 m_selectedIndex = -1;
649 }
650
651 m_scrollOffset = 0.0f;
652 m_targetScrollOffset = 0.0f;
653}

776void EffectChainRack::renderSlot(NUIRenderer& renderer, int index, float yOffset) {
777 NUIRect slotRect = slotRectForTop(yOffset);
778
779 const auto& slot = m_slots[index];
780 const bool isHovered = (index == m_hoveredSlot);
781
782 // Premium Glass Styling
783 NUIColor bgColor;
784 NUIColor borderColor;
785 // Drag Reorder: If this is the source slot, render faintly
786 bool isBeingDragged = (m_isDraggingReorder && index == m_draggingSlotIndex);
787
788 if (slot.isEmpty && !isBeingDragged) {
789 // Empty Slot: Subtle transparency or very faint glass
790 // Using Aestra "Deep Glass" tokens if available, otherwise manual
791 bgColor = isHovered ? Colors::buttonBackgroundHover.withAlpha(0.64f) : Colors::buttonBackground.withAlpha(0.48f);
792 borderColor = isHovered ? Colors::accentPrimary.withAlpha(0.30f) : Colors::panelBorder.withAlpha(0.30f);
793 } else {
794 // Populated: Solid dark glass
795 // If bypassed, make it slightly dimmer/transparent
796 if (isBeingDragged) {
797 bgColor = isHovered ? Colors::buttonBackgroundHover.withAlpha(0.72f) : Colors::buttonBackground.withAlpha(0.62f);
798 borderColor = Colors::accentPrimary.withAlpha(0.2f);
799 } else if (slot.bypassed) {
800 bgColor = Colors::buttonBackground.withAlpha(0.58f);
801 borderColor = Colors::panelBorder.withAlpha(0.5f);
802 } else {
803 bgColor = isHovered ? Colors::buttonBackgroundHover.withAlpha(0.84f) : Colors::buttonBackground.withAlpha(0.72f);
804 borderColor = isHovered ? Colors::accentPrimary.withAlpha(0.76f) : Colors::accentPrimary.withAlpha(0.20f);
805 }
806 }
807
808 renderer.fillRoundedRect(slotRect, 8.0f, bgColor);
809
810 // Dashed border for empty slots to signal droppability
811 if (slot.isEmpty && !isBeingDragged) {
812 renderer.strokeRoundedRect(slotRect, 8.0f, 1.0f, borderColor);
813 renderer.fillRoundedRect({slotRect.x + 6.0f, slotRect.y + slotRect.height * 0.5f - 0.5f,
814 slotRect.width - 12.0f, 1.0f},
815 0.5f,
816 isHovered ? Colors::accentPrimary.withAlpha(0.12f)
817 : Colors::panelBorder.withAlpha(0.16f));
818 } else {
819 renderer.strokeRoundedRect(slotRect, 8.0f, 1.0f, borderColor);
820 }
821
822 // DEBUG: Visual indicator for pending removal
823 if (slot.pendingRemoval) {
824 renderer.strokeRoundedRect(slotRect, 4.0f, 2.0f, NUIColor(1.0f, 0.0f, 0.0f, 0.8f));
825 }
826
827 // Shared vertical midline for the slot row — center everything around it
828 const float slotMid = slotRect.y + slotRect.height * 0.5f;
829
830 // Slot Number (Left side, stylistic) — centered on slotMid
831 char numBuf[8];
832 std::snprintf(numBuf, sizeof(numBuf), "%d", index + 1);
833 const float chipH = 14.0f;
834 const NUIRect indexChip{slotRect.x + 8.0f, slotMid - chipH * 0.5f, 18.0f, chipH};
835 renderer.fillRoundedRect(indexChip, 7.0f, Colors::buttonBackgroundHover.withAlpha(slot.isEmpty ? 0.62f : 0.76f));
836 renderer.strokeRoundedRect(indexChip, 7.0f, 1.0f, Colors::panelBorder.withAlpha(0.35f));
837 renderer.drawTextCentered(numBuf, indexChip, 9.0f, Colors::textDisabled.withAlpha(0.68f));
838
839 // Available text area to the right of the chip.
840 // Use the same vertical extent as the chip so drawTextCentered aligns identically.
841 const float textX = slotRect.x + 36.0f;
842 const float textW = slot.isEmpty ? slotRect.width - 44.0f : slotRect.width - 76.0f;
843 const float rowH = chipH; // 14 px — same height as the index chip
844
845 if (slot.isEmpty) {
846 const NUIRect textRect{textX, slotMid - rowH * 0.5f, textW, rowH};
847 if (isHovered) {
848 renderer.drawTextCentered("+ Add Insert", textRect, 10.0f, Colors::textPrimary);
849 } else {
850 const NUIColor dashColor = NUIColor(1.0f, 1.0f, 1.0f, 0.12f);
851 renderer.drawTextCentered("\xe2\x80\x94", textRect, 10.0f, dashColor);
852 }
853 } else {
854 // Name line plus state indicators. Active state is shown by the purple dot;
855 // bypass and auto-quarantine need a text label.
856 NUIColor nameColor = slot.bypassed ? Colors::textDisabled.withAlpha(0.6f) : Colors::textPrimary;
857 if (slot.bypassed) {
858 const NUIRect nameRect{textX, slotMid - rowH, textW, rowH};
859 renderer.drawText(fitText(renderer, slot.name, 10.5f, nameRect.width),
860 {nameRect.x, nameRect.y + 2.0f},
861 10.5f,
862 nameColor);
863 } else {
864 renderer.drawTextCentered(fitText(renderer, slot.name, 10.5f, textW),
865 {textX, slotRect.y, textW, slotRect.height},
866 10.5f,
867 nameColor);
868 }
869
870 if (slot.bypassed) {
871 const NUIRect statusRect{textX, slotMid, textW, rowH};
872 const bool autoFaulted = slot.nonFiniteOutputFault;
873 const char* statusText = autoFaulted ? "Output fault" : "Bypassed";
874 const NUIColor statusColor = autoFaulted
875 ? NUIThemeManager::getInstance().getColor("warning").withAlpha(0.92f)
876 : Colors::textDisabled.withAlpha(0.72f);
877 renderer.drawText(statusText, {statusRect.x, statusRect.y + 2.0f}, 8.5f, statusColor);
878 }
879
880 // Active indicator / Bypass toggle
881 float rightEdge = slotRect.x + slotRect.width;
882 float knobSize = 18.0f;
883 float knobX = rightEdge - knobSize - 4.0f;
884 float knobY = yOffset + (SLOT_HEIGHT - 4 - knobSize) * 0.5f;
885
886 // Dry/Wet Knob Rendering
887 NUIRect knobRect = {knobX, knobY, knobSize, knobSize};
888
889 // Helper to draw arc
890 auto drawArcPoly = [&](float startAngle, float endAngle, float width, NUIColor col) {
891 NUIPoint center = knobRect.center();
892 float radius = knobSize * 0.5f - 2.0f;
893 std::vector<NUIPoint> points;
894 int segments = 16;
895 for(int i=0; i<=segments; ++i) {
896 float t = (float)i / segments;
897 float ang = startAngle + (endAngle - startAngle) * t;
898 points.push_back({
899 center.x + std::cos(ang) * radius,
900 center.y + std::sin(ang) * radius
901 });
902 }
903 if (!points.empty())
904 renderer.drawPolyline(points.data(), (int)points.size(), width, col);
905 };
906
907 // Background Arc
908 drawArcPoly(0.75f * 3.14159f, 2.25f * 3.14159f, 2.0f, Colors::textDisabled.withAlpha(0.2f));
909
910 // Value Arc (Dim if bypassed)
911 float startAng = 0.75f * 3.14159f;
912 float range = 1.5f * 3.14159f;
913 float endAng = startAng + range * slot.dryWet;
914 NUIColor arcColor = slot.bypassed ? Colors::textDisabled.withAlpha(0.3f) : Colors::accentPrimary;
915 drawArcPoly(startAng, endAng, 2.0f, arcColor);
916
917 // Bypass Indicator (Dot left of knob)
918 // Center vertically better
919 float dotSize = 6.0f;
920 float dotY = slotRect.y + (slotRect.height - dotSize) * 0.5f;
921 NUIRect statusDot = {knobX - 12, dotY, dotSize, dotSize};
922
923 if (!slot.bypassed) {
924 // Active: LED On
925 renderer.fillRoundedRect(statusDot, 3.0f, Colors::accentPrimary);
926 // Glow
927 renderer.fillRoundedRect({statusDot.x - 2, statusDot.y - 2, 10, 10}, 5.0f, Colors::accentPrimary.withAlpha(0.4f));
928 } else {
929 // Bypassed: LED Off (Dark/Stroked)
930 renderer.strokeRoundedRect(statusDot, 3.0f, 1.0f, Colors::textDisabled.withAlpha(0.5f));
931 }
932 }
933}

944
945bool EffectChainRack::onMouseEvent(const NUIMouseEvent& event) {
946 if (!isVisible()) return false;
947
948 m_currentMousePos = event.position;
949
950 auto bounds = getBounds();
951
952 // Early exit if mouse is outside our bounds and not dragging
953 // Need to allow events if we are capturing mouse (like dragging knob or slot)
954 bool isCapturing = (m_activeKnobSlot != -1 || m_draggingSlotIndex != -1);
955 if (!bounds.contains(event.position) && !isCapturing) {
956 if (m_hoveredSlot != -1) {
957 m_hoveredSlot = -1;
958 repaint();
959 }
960 return false;
961 }
962
963 // Wheel support (Scroll)
964 if (std::abs(event.wheelDelta) > 0.001f && m_activeKnobSlot == -1) {
965 m_scrollOffset -= event.wheelDelta * 20.0f;
966
967 // Clamp scroll
968 float contentHeight = MAX_SLOTS * SLOT_HEIGHT + 10;
969 float viewHeight = getBounds().height;
970 m_scrollOffset = std::clamp(m_scrollOffset, 0.0f, std::max(0.0f, contentHeight - viewHeight));
971
972 repaint();
973 return true;
974 }
975
976 float my = event.position.y;
977 float mx = event.position.x;
978
979 // Update hover
980 m_hoveredSlot = hitTestSlot(my);
981
982 // Hit Testing Helpers
983 auto isOverKnob = [&](int index) {
984 if (index < 0) return false;
985 NUIRect slotRect = getSlotBounds(index);
986 float knobX = slotRect.x + slotRect.width - 22.0f;
987 return (mx >= knobX - 2 && mx <= knobX + 22) && (my >= slotRect.y + 2 && my <= slotRect.y + 26);
988 };
989
990 auto isOverBypass = [&](int index) {
991 if (index < 0) return false;
992 NUIRect slotRect = getSlotBounds(index);
993 float knobX = slotRect.x + slotRect.width - 22.0f;
994 return (mx >= knobX - 20 && mx <= knobX - 2) && (my >= slotRect.y + 2 && my <= slotRect.y + 26);
995 };
996
997 // RELEASED Event Handling (Must be checked before general Drag handling to allow drops)
998 if (event.released && event.button == NUIMouseButton::Left) {
999 if (m_activeKnobSlot != -1) {
1000 m_activeKnobSlot = -1;
1001 return true;
1002 }
1003
1004 // Handle Reorder Drop or Click
1005 if (m_isDraggingReorder && m_draggingSlotIndex != -1) {
1006 // Fix: Must account for scroll offset to map visual position back to slot index
1007 float contentY = event.position.y - (bounds.y + 8) + m_scrollOffset;
1008 int currentTarget = static_cast<int>(contentY / SLOT_HEIGHT);
1009
1010 if (currentTarget >= 0 && currentTarget < MAX_SLOTS && currentTarget != m_draggingSlotIndex) {
1011 if (m_onSlotMoveRequested) {
1012 m_onSlotMoveRequested(m_draggingSlotIndex, currentTarget);
1013 }
1014 }
1015 }
1016 else {
1017 // Single Click Action
1018 // Do nothing (User requested Double Click to open)
1019 }
1020
1021 m_draggingSlotIndex = -1;
1022 m_isDraggingReorder = false;
1023 repaint();
1024 return true;
1025 }
1026
1027 // ONGOING DRAG Handling
1028 // If we are in dragging mode, consume events (Move/Drag)
1029 if (m_draggingSlotIndex != -1) {
1030 // Slot Reorder Drag Check
1031 m_currentMousePos = event.position;
1032 float dist = std::abs(event.position.y - m_dragStartPos.y);
1033 if (dist > 5.0f && !m_isDraggingReorder) {
1034 m_isDraggingReorder = true;
1035 }
1036
1037 // Allow Move/Drag/None buttons to update the drag
1038 if (event.type == NUIMouseEventType::Drag || event.button == NUIMouseButton::Left || event.button == NUIMouseButton::None) {
1039 repaint();
1040 return true;
1041 }
1042 }
1043
1044 // KNOB DRAG Handling
1045 if (m_activeKnobSlot != -1) {
1046 const float dx = event.position.x - m_dragStartPos.x;
1047 const float dy = m_dragStartPos.y - event.position.y; // Up is positive (Values go up as mouse goes up)
1048 const float dragDelta = dx + dy;
1049
1050 float sensitivity = 0.005f;
1051 if (event.modifiers & NUIModifiers::Shift) sensitivity *= 0.1f;
1052
1053 float newValue = std::clamp(m_dragStartValue + dragDelta * sensitivity, 0.0f, 1.0f);
1054
1055 if (std::abs(newValue - m_slots[m_activeKnobSlot].dryWet) > 0.001f) {
1056 m_slots[m_activeKnobSlot].dryWet = newValue;
1057 if (m_onSlotMixChanged) {
1058 m_onSlotMixChanged(m_activeKnobSlot, newValue);
1059 }
1060 repaint();
1061 }
1062 return true;
1063 }
1064
1065 // CLICK (PRESSED) Event Handling
1066 int slotIdx = hitTestSlot(my);
1067
1068 if (event.pressed && event.button == NUIMouseButton::Left) {
1069 if (slotIdx >= 0 && slotIdx < MAX_SLOTS) {
1070 // 1. Knob Hit Test
1071 if (isOverKnob(slotIdx)) {
1072 if (!m_slots[slotIdx].isEmpty) {
1073 m_activeKnobSlot = slotIdx;
1074 m_dragStartValue = m_slots[slotIdx].dryWet;
1075 m_dragStartPos = event.position;
1076 return true;
1077 }
1078 }
1079 // 2. Bypass Hit Test
1080 else if (isOverBypass(slotIdx)) {
1081 if (!m_slots[slotIdx].isEmpty) {
1082 bool newState = !m_slots[slotIdx].bypassed;
1083 m_slots[slotIdx].bypassed = newState;
1084 m_bypassOverride[slotIdx] = newState ? 1 : 0;
1085 if (m_onSlotBypassToggled) m_onSlotBypassToggled(slotIdx, newState);
1086 repaint();
1087 return true;
1088 }
1089 }
1090 // 3. Slot Click (Selection / Drag Start)
1091 else {
1092 // Double Click Check
1093 auto now = std::chrono::steady_clock::now();
1094 auto elapsed = std::chrono::duration_cast<std::chrono::milliseconds>(now - m_lastClickTime).count();
1095 bool isDoubleClick = (slotIdx == m_lastClickSlot && elapsed < 300);
1096
1097 m_lastClickTime = now;
1098 m_lastClickSlot = slotIdx;
1099
1100 if (isDoubleClick) {
1101 if (!m_slots[slotIdx].isEmpty) {
1102 if (m_onSlotClicked) m_onSlotClicked(slotIdx);
1103 } else {
1104 if (m_onAddPluginRequested) m_onAddPluginRequested(slotIdx);
1105 }
1106 m_lastClickSlot = -1; // Reset to avoid triple-click issues
1107 } else {
1108 // Single click - Prepare for Drag
1109 if (!m_slots[slotIdx].isEmpty) {
1110 m_draggingSlotIndex = slotIdx;
1111 m_dragStartPos = event.position;
1112 }
1113 }
1114 return true;
1115 }
1116 }
1117 }
1118 else if (event.pressed && event.button == NUIMouseButton::Right) {
1119 if (slotIdx >= 0 && slotIdx < MAX_SLOTS && !m_slots[slotIdx].isEmpty) {
1120 // Close existing menu properly if it exists
1121 if (m_contextMenu) {
1122 if (m_contextMenu->getParent()) m_contextMenu->getParent()->removeChild(m_contextMenu);
1123 // Also try local remove just in case
1124 removeChild(m_contextMenu);
1125 m_contextMenu = nullptr;
1126 }
1127
1128 m_contextMenuSlot = slotIdx;
1129 m_contextMenu = std::make_shared<NUIContextMenu>();
1130
1131 // DELETE ACTION
1132 auto deleteItem = std::make_shared<NUIContextMenuItem>("Delete", NUIContextMenuItem::Type::Normal);
1133 deleteItem->setOnClick([this]() {
1134 if (m_onSlotRemoveRequested) {
1135 if (m_contextMenuSlot >= 0) {
1136 m_onSlotRemoveRequested(m_contextMenuSlot);
1137 }
1138 } else {
1139 Aestra::Log::warning("[Rack] m_onSlotRemoveRequested is NULL! Callback not bound.");
1140 }
1141 if (m_contextMenu) {

1175 root = root->getParent();

1188 m_contextMenu->showAt(event.position);
1189 }

1191 repaint();

1193 }
1194 }

1197 // which might think we are hovering "Add Send" or other overlapped widgets.
1198 if (m_hoveredSlot != -1) {
1199 return true;
1200 }

1202 return false;
