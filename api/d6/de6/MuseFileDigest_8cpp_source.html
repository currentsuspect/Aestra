12std::string fnv1aFileDigest(const std::string& path) {
13 // FNV-1a 64-bit. Chosen for being a few lines with no table, not for
14 // cryptographic strength: this detects "the render changed", and it is not
15 // defending the file from anyone.
16 constexpr uint64_t kOffsetBasis = 0xcbf29ce484222325ULL;
17 constexpr uint64_t kPrime = 0x100000001b3ULL;
18
19 std::ifstream file(path, std::ios::binary);
20 if (!file) return {};
21
22 uint64_t hash = kOffsetBasis;
23 std::array<char, 64 * 1024> buffer{};
24 while (file) {
25 file.read(buffer.data(), static_cast<std::streamsize>(buffer.size()));
26 const std::streamsize got = file.gcount();
27 for (std::streamsize i = 0; i < got; ++i) {
28 hash ^= static_cast<unsigned char>(buffer[static_cast<size_t>(i)]);
29 hash *= kPrime;
30 }
31 }
32 // A read that failed mid-file still leaves a hash of a prefix, which would
33 // compare equal to another truncated prefix. Report that as no digest.
34 if (file.bad()) return {};
35
36 char hex[17];
37 std::snprintf(hex, sizeof(hex), "%016llx", static_cast<unsigned long long>(hash));
38 return std::string(hex);
39}
