21// WaveformPeak - Min/Max sample pair
22// =============================================================================
23
27struct WaveformPeak {
28 float min = 0.0f;
29 float max = 0.0f;
30 float rms = 0.0f;
31 uint32_t count = 0;
32
33 WaveformPeak() = default;
34 WaveformPeak(float minVal, float maxVal, float rmsVal = 0.0f, uint32_t cnt = 0)
35 : min(minVal), max(maxVal), rms(rmsVal), count(cnt) {}
36
38 void merge(const WaveformPeak& other) {
39 min = std::min(min, other.min);
40 max = std::max(max, other.max);
41 uint64_t totalCount = static_cast<uint64_t>(count) + static_cast<uint64_t>(other.count);
42 if (totalCount > 0) {
43 double sumSq = static_cast<double>(rms) * rms * static_cast<double>(count)
44 + static_cast<double>(other.rms) * other.rms * static_cast<double>(other.count);
45 rms = static_cast<float>(std::sqrt(sumSq / static_cast<double>(totalCount)));
46 count = static_cast<uint32_t>(std::min(totalCount, static_cast<uint64_t>(std::numeric_limits<uint32_t>::max())));
47 }
48 }
49
51 void sanitize() {
52 if (std::isnan(min) || std::isinf(min)) min = 0.0f;
53 if (std::isnan(max) || std::isinf(max)) max = 0.0f;
54 if (std::isnan(rms) || std::isinf(rms)) rms = 0.0f;
55 }
56};

59// WaveformMipLevel - Single resolution level of peaks
60// =============================================================================
61
68struct WaveformMipLevel {
69 std::vector<WaveformPeak> peaks;
70 uint32_t samplesPerPeak = 1;
71 uint32_t numChannels = 0;
72 SampleIndex numPeaks = 0;
73
75 WaveformPeak getPeak(uint32_t channel, SampleIndex peakIndex) const {
76 if (channel >= numChannels || peakIndex < 0 || peakIndex >= numPeaks) {
77 return WaveformPeak();
78 }
79 size_t idx = static_cast<size_t>(peakIndex * numChannels + channel);
80 return idx < peaks.size() ? peaks[idx] : WaveformPeak();
81 }
82
85 WaveformPeak getInterpolatedPeak(uint32_t channel, double peakIndex) const {
86 if (channel >= numChannels || peakIndex < 0 || numPeaks == 0) {
87 return WaveformPeak();
88 }
89
90 SampleIndex idx0 = static_cast<SampleIndex>(peakIndex);
91 SampleIndex idx1 = std::min(idx0 + 1, numPeaks - 1);
92 float frac = static_cast<float>(peakIndex - idx0);
93
94 WaveformPeak p0 = getPeak(channel, idx0);
95 WaveformPeak p1 = getPeak(channel, idx1);
96
97 WaveformPeak result;
98 result.min = p0.min + frac * (p1.min - p0.min);
99 result.max = p0.max + frac * (p1.max - p0.max);
100 result.rms = (frac >= 0.5f) ? p1.rms : p0.rms;
101 result.count = (frac >= 0.5f) ? p1.count : p0.count;
102 return result;
103 }
104
106 WaveformPeak getPeakRange(uint32_t channel, SampleIndex startPeak, SampleIndex endPeak) const {
107 if (channel >= numChannels || startPeak >= numPeaks) {
108 return WaveformPeak();
109 }
110
111 startPeak = std::max(startPeak, static_cast<SampleIndex>(0));
112 endPeak = std::min(endPeak, numPeaks);
113
114 if (startPeak >= endPeak) {
115 return getPeak(channel, startPeak);
116 }
117
118 WaveformPeak result = getPeak(channel, startPeak);
119 for (SampleIndex i = startPeak + 1; i < endPeak; ++i) {
120 result.merge(getPeak(channel, i));
121 }
122 return result;
123 }
124};
