4// #866: projects saved by builds before #864 carry doubled track records. The old loader failed to restore a track,
5// created a fresh one per lane, and every later save wrote both. Two shapes exist in real files:
6//
7// A. Each lane owner routes to the master, and a record with the same name and no lanes holds the real mixer
8// channel. Restored as-is, every track loses its routing.
9// B. Each lane owner routes correctly, and a second record points only at lanes that no longer exist.
10//
11// The repair runs on the parsed records before they are restored. It merges A (the owner takes the channel, the
12// lane-less twin goes) and drops B's dead records. It never guesses: a name that matches more than one candidate is
13// left alone, and a lane owner on the master with no twin stays on the master (the user may have routed it there).
14

27struct LegacyTrackRepairReport {
28 size_t channelsMerged = 0;
29 size_t deadDropped = 0;
30};

33template <class LaneExists>
34LegacyTrackRepairReport repairLegacyTrackRecords(std::vector<Track>& records, LaneExists&& laneExists) {
35 LegacyTrackRepairReport report;
36 const auto ownsLiveLane = [&](const Track& t) {
37 return std::any_of(t.laneIds.begin(), t.laneIds.end(), [&](const PlaylistLaneID& id) { return laneExists(id); });
38 };
39
40 // How many lane-less records and how many live lane owners carry each name.
41 std::map<std::string, size_t> laneLessByName, ownersByName;
42 for (const Track& t : records) {
43 if (t.laneIds.empty())
44 ++laneLessByName[t.name];
45 else if (ownsLiveLane(t))
46 ++ownersByName[t.name];
47 }
48
49 std::vector<bool> drop(records.size(), false);
50 // Shape A: one owner on the master + exactly one lane-less twin of the same name.
51 for (size_t i = 0; i < records.size(); ++i) {
52 Track& owner = records[i];
53 if (owner.laneIds.empty() || !ownsLiveLane(owner) || owner.channelId != MASTER_MIXER_CHANNEL_ID)
54 continue;
55 if (laneLessByName[owner.name] != 1 || ownersByName[owner.name] != 1)
56 continue;
57 for (size_t j = 0; j < records.size(); ++j) {
58 const Track& twin = records[j];
59 if (j == i || drop[j] || !twin.laneIds.empty() || twin.name != owner.name)
60 continue;
61 if (twin.channelId == MASTER_MIXER_CHANNEL_ID)
62 break; // nothing to take
63 owner.channelId = twin.channelId;
64 drop[j] = true;
65 ++report.channelsMerged;
66 break;
67 }
68 }
69 // Shape B: a record that names lanes, none of which exists.
70 for (size_t i = 0; i < records.size(); ++i) {
71 if (!drop[i] && !records[i].laneIds.empty() && !ownsLiveLane(records[i])) {
72 drop[i] = true;
73 ++report.deadDropped;
74 }
75 }
76
77 size_t kept = 0;
78 for (size_t i = 0; i < records.size(); ++i) {
79 if (drop[i])
80 continue;
81 if (kept != i) // never self-move: a moved-onto-itself string or vector may come back empty
82 records[kept] = std::move(records[i]);
83 ++kept;
84 }
85 records.resize(kept);
86 return report;
87}
