4// V82-PR1 (FD-52): duplicate in the piano roll the way FL Studio does it. Founder, 2026-10-18: "Highlighting the piano
5// roll and commanding a duplication should act like FL, move the highlight next bar and duplicate the pattern within
6// it to that next bar", and "that highlight from the ruler too ... that doesn't select notes or do what i wanted".
7//
8// * Selection: Duplicate copies the selected notes one step on and moves the selection onto the copies, so pressing it
9// again keeps extending the phrase. The step is the selection's length rounded UP to the SNAP grid (one grid unit
10// at least), so the copy starts right after the last note on the grid and sits in the same place within its cell:
11// a phrase of 16ths repeats by 16ths, a bar by bars. (Owner, 2026-10-21: "by default it should duplicate to snap
12// with respect to the last one". It was whole bars whatever the snap, which threw a one-beat phrase four beats away.)
13// With snap off the step is the exact length.
14// * Ruler range: the highlight dragged on the ruler selects the notes inside it. Duplicate then copies that range
15// into the next block of the SAME length (not rounded: a 6-beat highlight copies into the next 6 beats) and moves
16// the highlight there. The highlight is still the loop zone, as in FL.
17//
18// Pure functions on the note list so the rules are pinned by a test without a window.

35inline double selectionStep(double selectionStart, double selectionEnd, double snapStep) {
36 const double span = std::max(0.0, selectionEnd - selectionStart);
37 if (snapStep <= 1e-9) return std::max(span, 0.0625);
38 const double cells = std::ceil(span / snapStep - 1e-9);
39 return std::max(1.0, cells) * snapStep;
40}

43inline bool startsIn(const MidiNote& n, const Range& r) {
44 return n.startBeat >= r.first - 1e-9 && n.startBeat < r.second - 1e-9;

61inline std::optional<Plan> plan(const std::vector<MidiNote>& notes, const std::optional<Range>& ruler,
62 double snapStep) {
63 std::vector<size_t> selected;
64 for (size_t i = 0; i < notes.size(); ++i)
65 if (!notes[i].isDeleted && notes[i].selected) selected.push_back(i);
66
67 if (ruler && ruler->second > ruler->first + 1e-9) {
68 std::vector<size_t> inRange;
69 for (size_t i = 0; i < notes.size(); ++i)
70 if (!notes[i].isDeleted && startsIn(notes[i], *ruler)) inRange.push_back(i);
71 const bool selectionInside = std::all_of(selected.begin(), selected.end(), [&](size_t i) { return startsIn(notes[i], *ruler); });
72 if (selectionInside) {
73 Plan p;
74 p.rangeMode = true;
75 p.shift = ruler->second - ruler->first;
76 p.sources = std::move(inRange);
77 p.from = *ruler;
78 p.to = {ruler->second, ruler->second + p.shift};
79 return p;
80 }
81 }
82
83 if (selected.empty()) return std::nullopt;
84 double start = notes[selected.front()].startBeat, end = 0.0;
85 for (const size_t i : selected) {
86 start = std::min(start, notes[i].startBeat);
87 end = std::max(end, notes[i].startBeat + notes[i].durationBeats);
88 }
89 Plan p;
90 p.shift = selectionStep(start, end, snapStep);
91 p.sources = std::move(selected);
92 p.from = {start, end};
93 p.to = {start + p.shift, end + p.shift};
94 return p;
95}

98inline std::vector<MidiNote> copies(const std::vector<MidiNote>& notes, const Plan& p) {
99 std::vector<MidiNote> out;
100 out.reserve(p.sources.size());
101 for (const size_t i : p.sources) {
102 MidiNote c = notes[i];
103 c.startBeat += p.shift;
104 c.selected = true;
105 c.isDeleted = false;
106 out.push_back(c);
107 }
108 return out;
109}

112inline double endOf(const std::vector<MidiNote>& made) {
113 double end = 0.0;
114 for (const auto& n : made) end = std::max(end, n.startBeat + n.durationBeats);
115 return end;
116}

119inline int barsToGrow(double endBeat, double patternLengthBeats, double beatsPerBar) {
120 const double bar = std::max(1.0, beatsPerBar);
121 return std::max(0, static_cast<int>(std::ceil((endBeat - patternLengthBeats) / bar - 1e-9)));
