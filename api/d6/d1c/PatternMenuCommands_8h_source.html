18class RenamePatternCommand final : public ICommand {
19public:
20 RenamePatternCommand(PatternManager& patterns, PatternID id, std::string newName)
21 : m_patterns(patterns), m_id(id), m_new(std::move(newName)) {}
22
23 void execute() override {
24 if (m_executed) return;
25 auto* p = m_patterns.getPattern(m_id);
26 if (!p || m_new.empty() || p->name == m_new) return;
27 m_old = p->name;
28 p->name = m_new;
29 m_executed = true;
30 }
31 void undo() override {
32 if (!m_executed) return;
33 if (auto* p = m_patterns.getPattern(m_id)) p->name = m_old;
34 m_executed = false;
35 }
36 void redo() override { execute(); }
37
38 std::string getName() const override { return "Rename Pattern"; }
39 size_t getSizeInBytes() const override { return sizeof(*this) + m_old.size() + m_new.size(); }
40 bool changesProjectState() const override { return true; }
41 bool wasApplied() const { return m_executed; }
42
43private:
44 PatternManager& m_patterns;
45 PatternID m_id;
46 std::string m_old, m_new;
47 bool m_executed = false;
48};

52class DuplicatePatternCommand final : public ICommand {
53public:
54 DuplicatePatternCommand(PatternManager& patterns, PatternID sourceId) : m_patterns(patterns), m_source(sourceId) {}
55
57 static std::string copyName(const PatternManager& patterns, const std::string& base) {
58 auto taken = [&](const std::string& candidate) {
59 bool found = false;
60 patterns.forEachPattern([&](const PatternSource& p) { found = found || p.name == candidate; });
61 return found;
62 };
63 const std::string stem = (base.empty() ? std::string("Pattern") : base) + " copy";
64 if (!taken(stem)) return stem;
65 for (int n = 2;; ++n) {
66 const std::string candidate = stem + " " + std::to_string(n);
67 if (!taken(candidate)) return candidate;
68 }
69 }
70
71 void execute() override {
72 if (m_executed) return;
73 if (m_detached) {
74 if (!m_patterns.reinsertPattern(m_detached)) return;
75 } else {
76 const auto* src = m_patterns.getPattern(m_source);
77 if (!src) return;
78 const std::string name = copyName(m_patterns, src->name);
79 m_copy = m_patterns.clonePattern(m_source);
80 if (!m_copy.isValid()) return;
81 if (auto* p = m_patterns.getPattern(m_copy)) p->name = name;
82 }
83 m_executed = true;
84 }
85 void undo() override {
86 if (!m_executed) return;
87 m_detached = m_patterns.detachPattern(m_copy);
88 m_executed = false;
89 }
90 void redo() override { execute(); }
91
92 std::string getName() const override { return "Duplicate Pattern"; }
93 size_t getSizeInBytes() const override { return sizeof(*this); }
94 bool changesProjectState() const override { return true; }
95 PatternID copyId() const { return m_copy; }
96
97private:
98 PatternManager& m_patterns;
99 PatternID m_source;
100 PatternID m_copy;
101 std::unique_ptr<PatternSource> m_detached;
102 bool m_executed = false;
103};
