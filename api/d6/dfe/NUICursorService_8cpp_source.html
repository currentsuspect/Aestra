6bool NUICursorService::beginDragCapture(NUICursorRestorePolicy policy, int grabOriginX, int grabOriginY) {
7 if (m_inTransition) {
8 // A host callback fired during end/cancel tried to start a new capture
9 // before the old one reached idle. Refuse: interleaving hide/show/grab
10 // sequences would corrupt both captures' host state.
11 return false;
12 }
13 if (m_captured) {
14 // A begin while captured means a release event was lost (focus churn,
15 // popup, teardown). Recover instead of wedging: restore visibility at
16 // the platform's current position, then start the new capture.
17 cancelDragCapture();
18 }
19 m_policy = policy;
20 m_grabOriginX = grabOriginX;
21 m_grabOriginY = grabOriginY;
22 m_captured = true;
23
24 m_lastPhysX = grabOriginX;
25 m_lastPhysY = grabOriginY;
26 m_expectRecenterEvent = false;
27
28 m_host.hostHideCursor();
29 // Confine the pointer for the duration of the capture: on native Wayland a
30 // warp is a silent no-op once the (hidden) pointer drifts out of the
31 // window and loses focus — the grab keeps the release-warp valid.
32 m_host.hostSetPointerGrab(true);
33 return true;
34}

36void NUICursorService::endDragCapture(int restoreX, int restoreY) {
37 if (!m_captured) {
38 return;
39 }
40 m_captured = false;
41 m_inTransition = true;
42
43 const int x = (m_policy == NUICursorRestorePolicy::GrabOrigin) ? m_grabOriginX : restoreX;
44 const int y = (m_policy == NUICursorRestorePolicy::GrabOrigin) ? m_grabOriginY : restoreY;
45
46 // Order matters: warp while still hidden so the cursor never renders a
47 // frame at the drifted pre-warp position, then unhide, then release the
48 // grab (releasing first would let the warp race the un-confinement).
49 m_host.hostWarpCursor(x, y);
50 m_host.hostShowCursor();
51 m_host.hostSetPointerGrab(false);
52 m_inTransition = false;
53}

55void NUICursorService::cancelDragCapture() {
56 if (!m_captured) {
57 return;
58 }
59 m_captured = false;
60 m_inTransition = true;
61 // No warp: a cancel means the warp target may no longer be valid (focus
62 // gone, window hidden). Just make the cursor visible wherever it is.
63 m_host.hostShowCursor();
64 m_host.hostSetPointerGrab(false);
65 m_inTransition = false;
66}

68NUICursorService::Delta NUICursorService::feedPhysicalMotion(int physX, int physY) {
69 if (!m_captured) {
70 return {};
71 }
72 // Suppress the synthetic motion produced by our own recenter warp: it lands
73 // on the anchor and must not be reported as drag motion.
74 if (m_expectRecenterEvent && physX == m_grabOriginX && physY == m_grabOriginY) {
75 m_expectRecenterEvent = false;
76 m_lastPhysX = m_grabOriginX;
77 m_lastPhysY = m_grabOriginY;
78 return {};
79 }
80
81 const Delta d{physX - m_lastPhysX, physY - m_lastPhysY};
82 if (d.dx == 0 && d.dy == 0) {
83 return d; // no motion, no recenter (avoids warp spam while idle)
84 }
85
86 // Recenter to the anchor so the next delta is measured from the anchor
87 // again — the pointer is pinned and can never drift to the window edge.
88 m_host.hostWarpCursor(m_grabOriginX, m_grabOriginY);
89 m_lastPhysX = m_grabOriginX;
90 m_lastPhysY = m_grabOriginY;
91 m_expectRecenterEvent = true;
92 return d;
93}
