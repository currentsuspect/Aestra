14void AestraApp::startUpdateChecks() {
15 if (m_updateNotifier) return;
16 UpdateNotifier::Hooks hooks;
17 if (IPlatformUtils* utils = Platform::getUtils()) {
18 const std::filesystem::path dir(utils->getAppDataPath("Aestra"));
19 if (!dir.empty()) hooks.settingsPath = (dir / "updates.json").string();
20 }
21 hooks.postToMain = [this](std::function<void()> task) { enqueueMainThreadTask(std::move(task)); };
22 hooks.requestQuit = [this]() { requestClose(); }; // the usual way out, so unsaved work is asked about first
23 hooks.showMessage = [this](const std::string& text) {
24 if (m_content) m_content->showToast(text, 9.0);
25 };
26 m_updateNotifier = std::make_shared<UpdateNotifier>(std::move(hooks));
27 m_updateNotifier->start();
28}
29
30void AestraApp::stopUpdateChecks() {
31 if (!m_updateNotifier) return;
32 m_updateNotifier->applyPendingRestart(); // the user chose "Restart and update": hand over to the updater as we exit
33 m_updateNotifier->stop(); // joins the checker thread: no result can arrive after this
34 m_updateNotifier.reset();
35}
