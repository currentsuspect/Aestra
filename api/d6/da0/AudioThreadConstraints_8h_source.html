94// Real-time thread detection lives in RealtimeThreadGuard.h and is the single
95// source of truth: isRealtimeAudioThread() / ScopedRealtimeAudioThread. This
96// header used to carry a parallel g_isAudioThread bool + AudioThreadGuard; those
97// were consolidated away so every constraint check keys off one flag. Mark the
98// audio callback scope with Aestra::Audio::ScopedRealtimeAudioThread.

103struct AudioThreadStats {
104 std::atomic<uint64_t> allocationViolations{0};
105 std::atomic<uint64_t> lockViolations{0};
106 std::atomic<uint64_t> ioViolations{0};
107 std::atomic<uint64_t> totalCallbacks{0};
108 std::atomic<uint64_t> underruns{0};
109 std::atomic<uint64_t> totalAllocations{0}; // B-007: Total tracked allocations
110 std::atomic<uint64_t> totalDeallocations{0}; // B-007: Total tracked deallocations
111 std::atomic<size_t> peakAllocationSize{0}; // B-007: Largest single allocation
112
113 void reset() {
114 allocationViolations.store(0);
115 lockViolations.store(0);
116 ioViolations.store(0);
117 totalCallbacks.store(0);
118 underruns.store(0);
119 totalAllocations.store(0);
120 totalDeallocations.store(0);
121 peakAllocationSize.store(0);
122 }
123
124 // B-007: Update peak if this allocation is larger
125 void updatePeak(size_t size) {
126 size_t current = peakAllocationSize.load(std::memory_order_relaxed);
127 while (size > current) {
128 if (peakAllocationSize.compare_exchange_weak(current, size,
129 std::memory_order_release, std::memory_order_relaxed)) {
130 break;
131 }
132 }
133 }
134
135 static AudioThreadStats& instance() {
136 static AudioThreadStats s_instance;
137 return s_instance;
138 }
139};
