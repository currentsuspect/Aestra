127struct AudioThreadStats {
128 std::atomic<uint64_t> allocationViolations{0};
129 std::atomic<uint64_t> lockViolations{0};
130 std::atomic<uint64_t> ioViolations{0};
131 std::atomic<uint64_t> totalCallbacks{0};
132 std::atomic<uint64_t> underruns{0};
133 std::atomic<uint64_t> totalAllocations{0}; // B-007: Total tracked allocations
134 std::atomic<uint64_t> totalDeallocations{0}; // B-007: Total tracked deallocations
135 std::atomic<size_t> peakAllocationSize{0}; // B-007: Largest single allocation
136
137 void reset() {
138 allocationViolations.store(0);
139 lockViolations.store(0);
140 ioViolations.store(0);
141 totalCallbacks.store(0);
142 underruns.store(0);
143 totalAllocations.store(0);
144 totalDeallocations.store(0);
145 peakAllocationSize.store(0);
146 }
147
148 // B-007: Update peak if this allocation is larger
149 void updatePeak(size_t size) {
150 size_t current = peakAllocationSize.load(std::memory_order_relaxed);
151 while (size > current) {
152 if (peakAllocationSize.compare_exchange_weak(current, size,
153 std::memory_order_release, std::memory_order_relaxed)) {
154 break;
155 }
156 }
157 }
158
159 static AudioThreadStats& instance() {
160 static AudioThreadStats s_instance;
161 return s_instance;
162 }
163};
