37inline bool isAudio(const std::filesystem::path& p) {
38 static const char* kExt[] = {".wav", ".flac", ".mp3", ".ogg", ".oga", ".opus", ".aif", ".aiff", ".aifc", ".m4a", ".aac"};
39 const std::string ext = lower(p.extension().string());
40 return std::any_of(std::begin(kExt), std::end(kExt), [&](const char* e) { return ext == e; });
41}

45inline std::vector<Entry> list(const std::string& dir) {
46 namespace fs = std::filesystem;
47 std::vector<Entry> out;
48 std::error_code ec;
49 for (fs::directory_iterator it(dir, fs::directory_options::skip_permission_denied, ec), end; !ec && it != end; it.increment(ec)) {
50 const std::string name = it->path().filename().string();
51 if (name.empty() || name[0] == '.') continue;
52 std::error_code e2;
53 if (it->is_directory(e2)) out.push_back({name, it->path().string(), true, 0});
54 else if (it->is_regular_file(e2) && isAudio(it->path())) out.push_back({name, it->path().string(), false, static_cast<uint64_t>(it->file_size(e2))});
55 }
56 std::sort(out.begin(), out.end(), [](const Entry& a, const Entry& b) {
57 if (a.folder != b.folder) return a.folder;
58 return lower(a.name) < lower(b.name);
59 });
60 return out;
61}

64inline std::vector<Place> places(const std::string& projectDir, const std::string& home) {
65 namespace fs = std::filesystem;
66 std::vector<Place> out;
67 const auto add = [&](const std::string& name, const fs::path& p) {
68 std::error_code ec;
69 if (!p.empty() && fs::is_directory(p, ec) && std::none_of(out.begin(), out.end(), [&](const Place& e) { return e.path == p.string(); })) out.push_back({name, p.string()});
70 };
71 if (!projectDir.empty()) add("Project", projectDir);
72 if (!home.empty()) {
73 add("Music", fs::path(home) / "Music");
74 add("Downloads", fs::path(home) / "Downloads");
75 add("Documents", fs::path(home) / "Documents");
76 add("Desktop", fs::path(home) / "Desktop");
77 add("Home", home);
78 }
79 return out;
80}
