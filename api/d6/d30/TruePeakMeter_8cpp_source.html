37double TruePeakMeter::besselI0(double x) noexcept {
38 // Series approximation, sufficient for |x| up to ~20.
39 double sum = 1.0;
40 double term = 1.0;
41 const double xx = (x * x) * 0.25;
42 for (int k = 1; k < 50; ++k) {
43 term *= xx / static_cast<double>(k * k);
44 sum += term;
45 if (term < 1e-18 * sum) {
46 break;
47 }
48 }
49 return sum;
50}

52double TruePeakMeter::kaiser(double n, double N, double beta) noexcept {
53 // Standard Kaiser window over n ∈ [0, N-1].
54 const double r = (2.0 * n / (N - 1.0)) - 1.0;
55 const double arg = beta * std::sqrt(1.0 - r * r);
56 return besselI0(arg) / besselI0(beta);
57}

59void TruePeakMeter::generateCoefficients() noexcept {
60 // Prototype FIR: windowed sinc, designed at the 4× output rate with
61 // cutoff at the Nyquist of the original rate (i.e. fs/2 → ω = π/4 in
62 // the 4× domain). Centre delay = (kFilterTaps - 1) / 2 = 23.5.
63 constexpr uint32_t L = kFilterTaps;
64 constexpr double centre = (L - 1.0) * 0.5;
65 constexpr double cutoff = 1.0 / static_cast<double>(kOversampleFactor); // π × cutoff
66
67 // Generate prototype with windowing.
68 double proto[L];
69 double sum = 0.0;
70 for (uint32_t n = 0; n < L; ++n) {
71 const double t = static_cast<double>(n) - centre;
72 double sinc;
73 if (std::abs(t) < 1e-12) {
74 sinc = cutoff;
75 } else {
76 const double arg = kPi * cutoff * t;
77 sinc = std::sin(arg) / (kPi * t);
78 }
79 const double w = kaiser(static_cast<double>(n), static_cast<double>(L), kKaiserBeta);
80 proto[n] = sinc * w;
81 sum += proto[n];
82 }
83
84 (void)sum; // global sum no longer used (we per-phase normalise below)
85
86 // Polyphase decomposition: e_k[m] = h[k + kOversampleFactor * m].
87 // History is "newest first" (hist[0] = x[n], hist[1] = x[n-1], ...),
88 // so when convolving y = Σ e_k[m] · x[n-m] we can index coeffs[tap] *
89 // hist[tap] directly.
90 //
91 // Normalise EACH polyphase sub-filter so its taps sum to 1.0. This
92 // guarantees unity DC gain at every fractional phase — without it the
93 // oversampled signal can drift several percent above/below |x| on DC,
94 // which would corrupt the true-peak measurement.
95 for (uint32_t phase = 0; phase < kOversampleFactor; ++phase) {
96 double phaseSum = 0.0;
97 for (uint32_t tap = 0; tap < kTapsPerPhase; ++tap) {
98 const uint32_t protoIdx = phase + kOversampleFactor * tap;
99 phaseSum += proto[protoIdx];
100 }
101 const double phaseScale = phaseSum != 0.0 ? 1.0 / phaseSum : 1.0;
102 for (uint32_t tap = 0; tap < kTapsPerPhase; ++tap) {
103 const uint32_t protoIdx = phase + kOversampleFactor * tap;
104 m_polyphase[phase][tap] = static_cast<float>(proto[protoIdx] * phaseScale);
105 }
106 }
107}

109void TruePeakMeter::processStereo(const float* interleavedStereo, uint32_t numFrames) noexcept {
110 if (!interleavedStereo || numFrames == 0) {
111 return;
112 }
113 for (uint32_t i = 0; i < numFrames; ++i) {
114 const float l = interleavedStereo[i * 2 + 0];
115 const float r = interleavedStereo[i * 2 + 1];
116
117 const float al = l < 0.0f ? -l : l;
118 const float ar = r < 0.0f ? -r : r;
119 if (al > m_samplePeakL) m_samplePeakL = al;
120 if (ar > m_samplePeakR) m_samplePeakR = ar;
121 if (al > m_truePeakL) m_truePeakL = al; // sample is one of the 4× outputs (phase 0 ≈ identity at centre)
122 if (ar > m_truePeakR) m_truePeakR = ar;
123
124 pushHistory(m_historyL, l);
125 pushHistory(m_historyR, r);
126
127 updateTruePeakFromHistory(m_historyL, m_truePeakL);
128 updateTruePeakFromHistory(m_historyR, m_truePeakR);
129 }
130}

132void TruePeakMeter::processMono(const float* mono, uint32_t numFrames) noexcept {
133 if (!mono || numFrames == 0) {
134 return;
135 }
136 for (uint32_t i = 0; i < numFrames; ++i) {
137 const float s = mono[i];
138 const float a = s < 0.0f ? -s : s;
139 if (a > m_samplePeakL) m_samplePeakL = a;
140 if (a > m_truePeakL) m_truePeakL = a;
141 pushHistory(m_historyL, s);
142 updateTruePeakFromHistory(m_historyL, m_truePeakL);
143 }
144 // Mirror to R so getMax... helpers behave for mono callers too.
145 if (m_samplePeakL > m_samplePeakR) m_samplePeakR = m_samplePeakL;
146 if (m_truePeakL > m_truePeakR) m_truePeakR = m_truePeakL;
147}
