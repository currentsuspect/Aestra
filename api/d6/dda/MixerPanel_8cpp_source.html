16MixerPanel::MixerPanel(std::shared_ptr<TrackManager> trackManager)
17 : WindowPanel("MIXER")
18 , m_trackManager(std::move(trackManager))
19{
20 // Create view model and modern mixer
21 m_viewModel = std::make_shared<Aestra::MixerViewModel>();
22 if (m_trackManager) {
23 // Scoped subscriptions for graph dirty / project modified callbacks
24 m_connections.add(m_viewModel->graphDirty.subscribe([trackManager = m_trackManager]() {
25 if (trackManager) {
26 trackManager->requestAudioGraphRebuild(
27 TrackManager::GraphDirtyReason::EffectChainChanged);
28 }
29 }));
30 m_connections.add(m_viewModel->projectModified.subscribe([trackManager = m_trackManager]() {
31 if (trackManager) {
32 trackManager->markModified();
33 }
34 }));
35 m_viewModel->setCommandHistory(&m_trackManager->getCommandHistory());
36 }
37 m_newMixer = std::make_shared<UIMixerPanel>(m_viewModel, m_trackManager);
38 m_newMixer->setId("UIMixerPanel_Inner");
39
40 // Set as content of WindowPanel
41 setContent(m_newMixer);
42
43 loadUIPreferences();
44 m_newMixer->onInspectorPreferenceChanged = [this](bool) { saveUIPreferences(); };
45
46 refreshChannels();
47}

49void MixerPanel::loadUIPreferences()
50{
51 if (!m_newMixer) return;
52
53 const std::string path = Aestra::MixerUIPreferences::settingsPath();
54 if (path.empty()) {
55 AESTRA_LOG_WARNING("No config directory available; mixer layout will not persist");
56 return;
57 }
58
59 // Absent keys keep the panel's own defaults; load() never invents a false.
60 const Aestra::MixerUIPreferences prefs = Aestra::MixerUIPreferences::load(path);
61 m_newMixer->setInspectorExpandedPreference(prefs.inspectorExpanded);
62}

77void MixerPanel::refreshChannels()
78{
79 if (!m_trackManager) return;
80
81 if (!m_viewModel || !m_newMixer) return;
82
83 auto slotMap = m_trackManager->getChannelSlotMapSnapshot();
84
85 // Refresh inputs from device manager
86 if (auto* deviceManager = Aestra::ServiceLocator::get<Aestra::Audio::AudioDeviceManager>()) {
87 m_viewModel->refreshInputs(*deviceManager);
88 }
89
90 m_viewModel->syncFromEngine(*m_trackManager, slotMap);
91
92 // Direct sync: read volume from MixerChannel and update ViewModel faderGainDb
93 // This ensures undo/redo reflects immediately without waiting for audio thread
94 for (size_t i = 0; i < m_trackManager->getChannelCount(); ++i) {
95 auto* channel = m_trackManager->getChannel(i);
96 if (!channel) continue;
97 auto* vmChannel = m_viewModel->getChannelById(channel->getChannelId());
98 if (vmChannel) {
99 float linearGain = channel->getVolume();
100 float db = (linearGain > 0.0001f) ? 20.0f * std::log10(linearGain) : -90.0f;
101 vmChannel->faderGainDb = db;
102 vmChannel->muted = channel->isMuted();
103 vmChannel->soloed = channel->isSoloed();
104 vmChannel->pan = channel->getPan();
105 }
106 }
107
108 m_newMixer->refreshChannels();
109}
