14MixerPanel::MixerPanel(std::shared_ptr<TrackManager> trackManager)
15 : WindowPanel("MIXER")
16 , m_trackManager(std::move(trackManager))
17{
18 // Create view model and modern mixer
19 m_viewModel = std::make_shared<Aestra::MixerViewModel>();
20 if (m_trackManager) {
21 // Scoped subscriptions for graph dirty / project modified callbacks
22 m_connections.add(m_viewModel->graphDirty.subscribe([trackManager = m_trackManager]() {
23 if (trackManager) {
24 trackManager->requestAudioGraphRebuild(
25 TrackManager::GraphDirtyReason::EffectChainChanged);
26 }
27 }));
28 m_connections.add(m_viewModel->projectModified.subscribe([trackManager = m_trackManager]() {
29 if (trackManager) {
30 trackManager->markModified();
31 }
32 }));
33 m_viewModel->setCommandHistory(&m_trackManager->getCommandHistory());
34 }
35 m_newMixer = std::make_shared<UIMixerPanel>(m_viewModel, m_trackManager);
36 m_newMixer->setId("UIMixerPanel_Inner");
37
38 // Set as content of WindowPanel

45void MixerPanel::refreshChannels()
46{
47 if (!m_trackManager) return;
48
49 if (!m_viewModel || !m_newMixer) return;
50
51 auto slotMap = m_trackManager->getChannelSlotMapSnapshot();
52
53 // Refresh inputs from device manager
54 if (auto* deviceManager = Aestra::ServiceLocator::get<Aestra::Audio::AudioDeviceManager>()) {
55 m_viewModel->refreshInputs(*deviceManager);
56 }
57
58 m_viewModel->syncFromEngine(*m_trackManager, slotMap);
59
60 // Direct sync: read volume from MixerChannel and update ViewModel faderGainDb
61 // This ensures undo/redo reflects immediately without waiting for audio thread
62 for (size_t i = 0; i < m_trackManager->getChannelCount(); ++i) {
63 auto* channel = m_trackManager->getChannel(i);
64 if (!channel) continue;
65 auto* vmChannel = m_viewModel->getChannelById(channel->getChannelId());
66 if (vmChannel) {
67 float linearGain = channel->getVolume();
68 float db = (linearGain > 0.0001f) ? 20.0f * std::log10(linearGain) : -90.0f;
69 vmChannel->faderGainDb = db;
70 vmChannel->muted = channel->isMuted();
71 vmChannel->soloed = channel->isSoloed();
72 vmChannel->pan = channel->getPan();
73 }
74 }
75
76 m_newMixer->refreshChannels();
77}
