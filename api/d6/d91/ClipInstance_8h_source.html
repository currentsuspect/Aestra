83 static float playbackRateFromSemitones(float semitones) noexcept {
84 if (!std::isfinite(semitones))
85 return 1.0f;
86 const float clamped = std::clamp(semitones, kMinPitchSemitones, kMaxPitchSemitones);
87 return std::pow(2.0f, clamped / 12.0f);
88 }
89
90 static float semitonesFromPlaybackRate(float playbackRate) noexcept {
91 if (!std::isfinite(playbackRate) || playbackRate <= 0.0f)
92 return 0.0f;
93 const float semitones = 12.0f * std::log2(playbackRate);
94 return std::clamp(semitones, kMinPitchSemitones, kMaxPitchSemitones);
95 }
96
102 float effectiveVarispeed() const noexcept {
103 const float rate = std::isfinite(playbackRate) ? std::clamp(playbackRate, 0.25f, 4.0f) : 1.0f;
104 const float pitch = std::isfinite(pitchSemitones)
105 ? std::clamp(pitchSemitones, kMinPitchSemitones, kMaxPitchSemitones)
106 : 0.0f;
107 return std::clamp(rate * std::pow(2.0f, pitch / 12.0f), 0.25f, 4.0f);
108 }

115 static ClipEdits forNewAudioClip() {

118 return edits;
