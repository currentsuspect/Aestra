14// A nonzero tag per thread. std::this_thread::get_id() wraps pthread_self()/GetCurrentThreadId():
15// no allocation, no lock, safe on the audio thread.
16size_t currentThreadTag() noexcept { return std::hash<std::thread::id>{}(std::this_thread::get_id()) | 1u; }

19void AudioEngine::beginManualRender() {
20 const size_t self = currentThreadTag();
21 if (m_manualRenderOwner.load(std::memory_order_acquire) == self) {
22 ++m_manualRenderNesting;
23 return;
24 }
25 // Another thread's manual render runs to completion first.
26 size_t expected = 0;
27 while (!m_manualRenderOwner.compare_exchange_weak(expected, self, std::memory_order_seq_cst)) {
28 expected = 0;
29 std::this_thread::sleep_for(std::chrono::milliseconds(1));
30 }
31 m_manualRenderNesting = 1;
32 // A live block may be mid-flight. It counts itself in before it reads the owner (seq_cst on
33 // both sides), so once the count drains every later live block sees the lease and plays silence.
34 for (int waited = 0; m_processBlocksInFlight.load(std::memory_order_seq_cst) != 0; ++waited) {
35 if (waited == 2000) {
36 Aestra::Log::error("[AudioEngine] manual render: a live block did not finish within 2 s");
37 break;
38 }
39 std::this_thread::sleep_for(std::chrono::milliseconds(1));
40 }
41}

43void AudioEngine::endManualRender() {
44 if (m_manualRenderOwner.load(std::memory_order_relaxed) != currentThreadTag() || m_manualRenderNesting == 0) {
45 return;
46 }
47 if (--m_manualRenderNesting == 0) {
48 m_manualRenderOwner.store(0, std::memory_order_release);
49 }
50}

52// Driver callback entry. Every per-block buffer the engine renders into is sized
53// to m_maxBufferFrames, but a backend is free to hand over more frames than the
54// size it reported (WASAPI shared mode delivers up to its device buffer, which
55// is larger than the requested period). Rendering such a block in one piece
56// read past the master buffer. Split it into capacity-sized chunks instead, so
57// an oversized callback costs extra passes, never memory safety or audio.
58int AudioEngine::processBlock(float* outputBuffer, const float* inputBuffer, uint32_t numFrames, double streamTime) {
59 m_processBlocksInFlight.fetch_add(1, std::memory_order_seq_cst);
60 struct InFlight {
61 std::atomic<uint32_t>& count;
62 ~InFlight() { count.fetch_sub(1, std::memory_order_release); }
63 } inFlight{m_processBlocksInFlight};
64 const size_t leaseOwner = m_manualRenderOwner.load(std::memory_order_seq_cst);
65 if (leaseOwner != 0 && leaseOwner != currentThreadTag()) {
66 // Another thread is rendering this engine by hand (export, bounce, Muse): stand down.
67 if (outputBuffer && numFrames > 0) {
68 const uint32_t channels = std::max<uint32_t>(1, m_outputChannels.load(std::memory_order_relaxed));
69 std::memset(outputBuffer, 0, static_cast<size_t>(numFrames) * channels * sizeof(float));
70 }
71 return 0;
72 }
73
74 const uint32_t capacity = m_maxBufferFrames.load(std::memory_order_relaxed);
75 if (capacity == 0 || numFrames <= capacity || !outputBuffer) {
76 return processBlockWithinCapacity(outputBuffer, inputBuffer, numFrames, streamTime);
77 }
78
79 // Recording input is consumed whole, once: the callback does not depend on
80 // the render block size, and the input channel count is not known here to
81 // slice it.
82 if (inputBuffer) {
83 if (auto cb = m_inputCallback.load(std::memory_order_relaxed)) {
84 cb(inputBuffer, numFrames, m_inputCallbackData.load(std::memory_order_relaxed));
85 }
86 }
87
88 const uint32_t channels = std::max<uint32_t>(1, m_outputChannels.load(std::memory_order_relaxed));
89 const uint32_t sampleRate = m_sampleRate.load(std::memory_order_relaxed);
90 uint32_t done = 0;
91 int result = 0;
92 while (done < numFrames) {
93 const uint32_t chunk = std::min(capacity, numFrames - done);
94 const double chunkTime =
95 sampleRate > 0 ? streamTime + static_cast<double>(done) / static_cast<double>(sampleRate) : streamTime;
96 result = processBlockWithinCapacity(outputBuffer + static_cast<size_t>(done) * channels, nullptr, chunk,
97 chunkTime);
98 done += chunk;
99 }
100 m_telemetry.incrementOversizedBlocks();
101 return result;
102}
