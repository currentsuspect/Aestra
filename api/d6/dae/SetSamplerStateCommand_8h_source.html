29 static SamplerEditState capture(const Plugins::SamplerPlugin& s) {
30 SamplerEditState e;

32 e.coarse = s.getCoarseSemitones(); e.fine = s.getFineTuneCents();

34 e.voices = s.getMaxVoices(); e.rootNote = s.getRootMidiNote();

42 s.setMaxVoices(voices); s.setRootMidiNote(rootNote);
43 s.setMonoMode(mono); s.setCutSelfMode(cutSelf); s.setPitchMode(pitchMode);

74 void write(const SamplerEditState& state) {
75 auto sampler = std::dynamic_pointer_cast<Plugins::SamplerPlugin>(m_units.getUnitPlugin(m_unit));
76 if (!sampler) return;
77 state.apply(*sampler);
78 m_units.captureUnitPluginState(m_unit);
79 if (m_onApplied) m_onApplied();
80 }
