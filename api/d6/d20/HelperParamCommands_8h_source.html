23inline std::string hexOf(const std::string& text) {
24 static const char* digits = "0123456789abcdef";
25 std::string out;
26 out.reserve(text.size() * 2);
27 for (const unsigned char c : text) {
28 out.push_back(digits[c >> 4]);
29 out.push_back(digits[c & 0x0F]);
30 }
31 return out;
32}

35class EditLog {
36public:
37 void add(uint32_t id, double value) {
38 std::lock_guard<std::mutex> lock(m_mutex);
39 for (auto& e : m_edits) {
40 if (e.first == id) {
41 e.second = value; // only the latest matters
42 return;
43 }
44 }
45 if (m_edits.size() < 4096) m_edits.emplace_back(id, value);
46 }
47 std::string take() {
48 std::vector<std::pair<uint32_t, double>> edits;
49 {
50 std::lock_guard<std::mutex> lock(m_mutex);
51 edits.swap(m_edits);
52 }
53 std::string out = "OK EDITS";
54 char buf[64];
55 for (const auto& e : edits) {
56 std::snprintf(buf, sizeof buf, " %u:%.9g", e.first, e.second);
57 out += buf;
58 }
59 return out;
60 }
61
62private:
63 std::mutex m_mutex;
64 std::vector<std::pair<uint32_t, double>> m_edits;
65};

68inline std::string paramsReply(Aestra::Audio::VST3PluginInstance& plugin) {
69 const auto params = plugin.getParameters();
70 std::string out = "OK " + std::to_string(params.size());
71 char buf[96];
72 for (const auto& p : params) {
73 const int flags = (p.isAutomatable ? 1 : 0) | (p.isBypass ? 2 : 0) | (p.isReadOnly ? 4 : 0);
74 std::snprintf(buf, sizeof buf, " %u:%d:%u:%.9g:", p.id, flags, p.stepCount, static_cast<double>(p.defaultValue));
75 out += buf;
76 out += hexOf(p.name) + ":" + hexOf(p.shortName) + ":" + hexOf(p.unit);
77 }
78 return out;
79}
