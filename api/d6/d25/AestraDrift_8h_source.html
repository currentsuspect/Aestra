64 void process(const float* const* inputs, float** outputs,
65 uint32_t numInputChannels, uint32_t numOutputChannels,
66 uint32_t numFrames, const MidiBuffer* midiInput = nullptr,
67 MidiBuffer* midiOutput = nullptr) override {
68 (void)midiInput;
69 (void)midiOutput;
70
71 if (!m_active.load(std::memory_order_relaxed) ||
72 m_params[kBypass].load(std::memory_order_relaxed) > 0.5f) {
73 for (uint32_t ch = 0; ch < numOutputChannels; ++ch) {
74 if (outputs[ch] && ch < numInputChannels && inputs[ch])
75 std::memcpy(outputs[ch], inputs[ch], numFrames * sizeof(float));
76 else if (outputs[ch])
77 std::memset(outputs[ch], 0, numFrames * sizeof(float));
78 }
79 return;
80 }
81
82 const uint32_t channels = std::min<uint32_t>(2, numOutputChannels);
83 const bool stereo = channels >= 2 && numInputChannels >= 2;
84 int writePos = m_writePos;
85 float tapPhase[2] = { m_tapPhase[0], m_tapPhase[1] };
86 const float baseDelay = m_baseDelay;
87
88 const float twoPi = 2.0f * kPi;
89
90 // Per-sample one-pole smoothing for the automatable params (Pitch, Mix),
91 // so automation or fast knob moves glide instead of zippering. Coeff is
92 // computed once per block; ~8 ms time constant.
93 const float smoothCoeff = 1.0f - std::exp(-1.0f / (static_cast<float>(m_sampleRate) * 0.008f));
94 // Once the smoother is within this of its target, snap exactly onto it.
95 // A one-pole only asymptotes, so without this the residual decays into
96 // denormal floats — a CPU penalty on the audio thread. 1e-6 is far below
97 // the audible resolution of these normalized (0..1) params.
98 constexpr float kSmoothSnapEps = 1.0e-6f;
99 const float pitchTarget = m_params[kPitch].load(std::memory_order_relaxed);
100 const float mixTarget = m_params[kMix].load(std::memory_order_relaxed);
101 float pitchSmoothed = m_pitchSmoothed;
102 float mixSmoothed = m_mixSmoothed;
103
104 for (uint32_t i = 0; i < numFrames; ++i) {
105 const float inL = (numInputChannels > 0 && inputs[0]) ? inputs[0][i] : 0.0f;
106 const float inR = stereo ? ((numInputChannels > 1 && inputs[1]) ? inputs[1][i] : inL) : inL;
107
108 m_buffer[writePos] = inL;
109 m_bufferR[writePos] = inR;
110
111 // Dual overlapping taps — always active, Hann-windowed.
112 // When one tap wraps (phase crosses 0/grainSize), its window is ~0
113 // and the other tap dominates. No audible discontinuity.
114 float outL = 0.0f, outR = 0.0f;
115 float winSum = 0.0f;
116 for (int t = 0; t < 2; ++t) {
117 const float phaseNorm = tapPhase[t] / kGrainSizeF;
118 // Hann window: 0 at phase=0 and phase=grainSize, 1 at phase=grainSize/2
119 const float win = 0.5f * (1.0f - std::cos(twoPi * phaseNorm));
120 winSum += win;
121
122 const float readPos = static_cast<float>(writePos) - baseDelay + tapPhase[t];
123 const float sL = readBuffer(m_buffer, readPos);
124 outL += sL * win;
125 if (stereo) {
126 const float sR = readBuffer(m_bufferR, readPos);
127 outR += sR * win;
128 }
129 }
130 // Normalize by window sum (should be ~1.0 for 50% overlap, but guard)
131 const float norm = 1.0f / std::max(winSum, 0.001f);
132 outL *= norm;
133 outR *= norm;
134
135 pitchSmoothed += smoothCoeff * (pitchTarget - pitchSmoothed);
136 if (std::fabs(pitchTarget - pitchSmoothed) < kSmoothSnapEps) pitchSmoothed = pitchTarget;
137 const float pitchNorm = pitchSmoothed;
138 const float pitchSemitones = -12.0f + pitchNorm * 24.0f;
139 const float pitchRatio = std::pow(2.0f, pitchSemitones / 12.0f);
140 const float delta = 1.0f - pitchRatio;
141
142 for (int t = 0; t < 2; ++t) {
143 tapPhase[t] += delta;
144 // Wrap: when one tap wraps, its Hann window is near 0
145 if (tapPhase[t] >= kGrainSizeF) tapPhase[t] -= kGrainSizeF;
146 if (tapPhase[t] < 0.0f) tapPhase[t] += kGrainSizeF;
147 }
148
149 writePos = (writePos + 1) & kBufferMask;
150
151 mixSmoothed += smoothCoeff * (mixTarget - mixSmoothed);
152 if (std::fabs(mixTarget - mixSmoothed) < kSmoothSnapEps) mixSmoothed = mixTarget;
153 const float mix = mixSmoothed;
154 const float wet = mix;
155 const float dry = 1.0f - wet;
156
157 if (numOutputChannels > 0 && outputs[0])
158 outputs[0][i] = inL * dry + outL * wet;
159 if (numOutputChannels > 1 && outputs[1])
160 outputs[1][i] = (stereo ? inR : inL) * dry + outR * wet;
161 for (uint32_t ch = 2; ch < numOutputChannels; ++ch) {
162 if (outputs[ch]) outputs[ch][i] = 0.0f;
163 }
164 }
165
166 m_writePos = writePos;
167 m_tapPhase[0] = tapPhase[0];
168 m_tapPhase[1] = tapPhase[1];
169 m_pitchSmoothed = pitchSmoothed;
170 m_mixSmoothed = mixSmoothed;
171 }

195 std::string getParameterDisplay(uint32_t id) const override {
196 if (id >= kParamCount) return "";
197 const float v = getParameter(id);
198 switch (id) {
199 case kPitch: {
200 const float st = -12.0f + v * 24.0f;
201 char buf[32];
202 std::snprintf(buf, sizeof(buf), "%+.1f st", st);
203 return buf;
204 }
205 case kGrain: {
206 const float ms = grainMsFromNorm(v);
207 char buf[32];
208 std::snprintf(buf, sizeof(buf), "%.0f ms", ms);
209 return buf;
210 }
211 case kMix:
212 return std::to_string(static_cast<int>(std::round(v * 100.0f))) + "%";
213 case kBypass:
214 return v > 0.5f ? "ON" : "OFF";
215 default: return "";
216 }

219 std::vector<uint8_t> saveState() const override {
220 struct Blob { uint32_t magic = kStateMagic; uint32_t version = 1; float params[kParamCount]; } blob;
221 for (uint32_t i = 0; i < kParamCount; ++i) blob.params[i] = getParameter(i);
222 const auto* data = reinterpret_cast<const uint8_t*>(&blob);
223 return { data, data + sizeof(blob) };
224 }

226 bool loadState(const std::vector<uint8_t>& state) override {
227 if (state.size() < sizeof(uint32_t) * 2) return false;
228 struct Header { uint32_t magic; uint32_t version; };
229 const auto* header = reinterpret_cast<const Header*>(state.data());
230 if (header->magic != kStateMagic) return false;
231 struct StateBlob { uint32_t magic; uint32_t version; float params[kParamCount]; };
232 if (state.size() < sizeof(StateBlob)) return false;
233 const auto* blob = reinterpret_cast<const StateBlob*>(state.data());
234 for (uint32_t i = 0; i < kParamCount; ++i) setParameter(i, blob->params[i]);
235 return true;
236 }

255private:
256 static constexpr uint32_t kStateMagic = 0x44524654;
257 static constexpr int kBufferSize = 8192;
258 static constexpr int kBufferMask = kBufferSize - 1;
259 static constexpr float kPi = 3.14159265358979323846f;
260 static constexpr float kGrainSizeF = 2048.0f;
261 static constexpr float kBaseDelay = static_cast<float>(kBufferSize) * 0.20f;
262
263 static float grainMsFromNorm(float norm) {
264 return 5.0f + norm * 35.0f;
265 }
266
267 // Cubic Hermite interpolation
268 static float readBuffer(const std::array<float, kBufferSize>& buf, float readPos) {
269 const int i0 = static_cast<int>(std::floor(readPos)) & kBufferMask;
270 const int i1 = (i0 + 1) & kBufferMask;
271 const int i2 = (i1 + 1) & kBufferMask;
272 const int i3 = (i2 + 1) & kBufferMask;
273 const float frac = readPos - std::floor(readPos);
274 const float a = buf[static_cast<size_t>(i3)] - buf[static_cast<size_t>(i2)]
275 + buf[static_cast<size_t>(i1)] - buf[static_cast<size_t>(i0)];
276 const float b = buf[static_cast<size_t>(i0)] - buf[static_cast<size_t>(i1)] - a;
277 const float c = buf[static_cast<size_t>(i2)] - buf[static_cast<size_t>(i0)];
278 const float d = buf[static_cast<size_t>(i1)];
279 return ((a * frac + b) * frac + c) * frac + d;
280 }

282 void resetRuntimeState() {
283 m_writePos = 0;
284 m_baseDelay = kBaseDelay;
285 m_tapPhase[0] = 0.0f;
286 m_tapPhase[1] = kGrainSizeF * 0.5f;
287 // Snap smoothers to the current targets so load/activate doesn't glide.
288 m_pitchSmoothed = m_params[kPitch].load(std::memory_order_relaxed);
289 m_mixSmoothed = m_params[kMix].load(std::memory_order_relaxed);
290 }

292 PluginInfo m_info;
293 double m_sampleRate = 48000.0;
294 std::atomic<bool> m_active{false};
295 std::atomic<bool> m_paramsInitialized{false};
296 std::array<std::atomic<float>, kParamCount> m_params{};
297
298 std::array<float, kBufferSize> m_buffer{};
299 std::array<float, kBufferSize> m_bufferR{};
300 int m_writePos = 0;
301 float m_baseDelay = kBaseDelay;
302 float m_tapPhase[2] = {0.0f, kGrainSizeF * 0.5f};
303
304 // Smoothed automatable params. Drift reads Pitch/Mix per-sample; without
305 // smoothing, automating Mix (a dry/wet gain crossfade) or Pitch zippers.
306 // setParameter only stores the atomic target; process() ramps toward it,
307 // matching the smoothing contract the other internal plugins follow.
308 float m_pitchSmoothed = 0.5f;
309 float m_mixSmoothed = 1.0f;
