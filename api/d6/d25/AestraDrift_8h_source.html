39 bool initialize(double sampleRate, uint32_t maxBlockSize) override {
40 (void)maxBlockSize;
41 m_sampleRate = std::max(1.0, sampleRate);
42 // Seed parameter defaults only on the first initialization of a fresh
43 // instance. EffectChain::prepare() re-calls initialize() on the live
44 // instance during sample-rate/device changes and must preserve the
45 // user's current parameter values (and any loaded project state).
46 if (!m_paramsInitialized.exchange(true)) {
47 for (const auto& param : getParameters()) {
48 m_params[param.id].store(param.defaultValue, std::memory_order_relaxed);
49 }
50 }
51 const uint32_t maxGrainSamples =
52 std::max(8u, static_cast<uint32_t>(std::ceil(kMaxGrainMilliseconds * 0.001 * m_sampleRate)));
53 m_latencySamples = maxGrainSamples / 2u + 4u;
54 const uint32_t requiredSamples = m_latencySamples + maxGrainSamples / 2u + 8u;
55 uint32_t bufferSize = 1u;
56 while (bufferSize < requiredSamples)
57 bufferSize <<= 1u;
58 m_bufferMask = static_cast<int>(bufferSize - 1u);
59 m_buffer.assign(bufferSize, 0.0f);
60 m_bufferR.assign(bufferSize, 0.0f);
61 resetRuntimeState();
62 return true;
63 }

77 void process(const float* const* inputs, float** outputs, uint32_t numInputChannels, uint32_t numOutputChannels,
78 uint32_t numFrames, const MidiBuffer* midiInput = nullptr, MidiBuffer* midiOutput = nullptr) override {
79 (void)midiInput;
80 (void)midiOutput;
81
82 if (!m_active.load(std::memory_order_relaxed) || m_params[kBypass].load(std::memory_order_relaxed) > 0.5f) {
83 for (uint32_t ch = 0; ch < numOutputChannels; ++ch) {
84 if (outputs[ch] && ch < numInputChannels && inputs[ch])
85 std::memcpy(outputs[ch], inputs[ch], numFrames * sizeof(float));
86 else if (outputs[ch])
87 std::memset(outputs[ch], 0, numFrames * sizeof(float));
88 }
89 return;
90 }
91
92 const uint32_t channels = std::min<uint32_t>(2, numOutputChannels);
93 const bool stereo = channels >= 2 && numInputChannels >= 2;
94 if (m_buffer.empty() || m_bufferR.empty()) {
95 for (uint32_t ch = 0; ch < numOutputChannels; ++ch) {
96 if (outputs[ch])
97 std::memset(outputs[ch], 0, numFrames * sizeof(float));
98 }
99 return;
100 }
101 int writePos = m_writePos;
102 std::array<float, kTapCount> tapPhase = m_tapPhase;
103 float motionPhase = m_motionPhase;
104 const int bufferMask = m_bufferMask;
105 const float latencySamples = static_cast<float>(m_latencySamples);
106
107 const float twoPi = 2.0f * kPi;
108
109 // Per-sample one-pole smoothing for the automatable params (Pitch, Mix),
110 // so automation or fast knob moves glide instead of zippering. Coeff is
111 // computed once per block; ~8 ms time constant.
112 const float smoothCoeff = 1.0f - std::exp(-1.0f / (static_cast<float>(m_sampleRate) * 0.008f));
113 // Once the smoother is within this of its target, snap exactly onto it.
114 // A one-pole only asymptotes, so without this the residual decays into
115 // denormal floats — a CPU penalty on the audio thread. 1e-6 is far below
116 // the audible resolution of these normalized (0..1) params.
117 constexpr float kSmoothSnapEps = 1.0e-6f;
118 const float pitchTarget = m_params[kPitch].load(std::memory_order_relaxed);
119 const float grainTarget = m_params[kGrain].load(std::memory_order_relaxed);
120 const float mixTarget = m_params[kMix].load(std::memory_order_relaxed);
121 const float fineTarget = m_params[kFine].load(std::memory_order_relaxed);
122 const float spreadTarget = m_params[kSpread].load(std::memory_order_relaxed);
123 const float motionTarget = m_params[kMotion].load(std::memory_order_relaxed);
124 const float motionRateTarget = m_params[kMotionRate].load(std::memory_order_relaxed);
125 const float outputTarget = m_params[kOutput].load(std::memory_order_relaxed);
126 const float textureTarget = m_params[kTexture].load(std::memory_order_relaxed);
127 float pitchSmoothed = m_pitchSmoothed;
128 float grainSmoothed = m_grainSmoothed;
129 float mixSmoothed = m_mixSmoothed;
130 float fineSmoothed = m_fineSmoothed;
131 float spreadSmoothed = m_spreadSmoothed;
132 float motionSmoothed = m_motionSmoothed;
133 float motionRateSmoothed = m_motionRateSmoothed;
134 float outputSmoothed = m_outputSmoothed;
135 float textureSmoothed = m_textureSmoothed;
136
137 for (uint32_t i = 0; i < numFrames; ++i) {
138 const float inL = (numInputChannels > 0 && inputs[0]) ? inputs[0][i] : 0.0f;
139 const float inR = stereo ? ((numInputChannels > 1 && inputs[1]) ? inputs[1][i] : inL) : inL;
140
141 m_buffer[writePos] = inL;
142 m_bufferR[writePos] = inR;
143
144 pitchSmoothed += smoothCoeff * (pitchTarget - pitchSmoothed);
145 if (std::fabs(pitchTarget - pitchSmoothed) < kSmoothSnapEps)
146 pitchSmoothed = pitchTarget;
147 grainSmoothed += smoothCoeff * (grainTarget - grainSmoothed);
148 if (std::fabs(grainTarget - grainSmoothed) < kSmoothSnapEps)
149 grainSmoothed = grainTarget;
150 mixSmoothed += smoothCoeff * (mixTarget - mixSmoothed);
151 if (std::fabs(mixTarget - mixSmoothed) < kSmoothSnapEps)
152 mixSmoothed = mixTarget;
153 fineSmoothed += smoothCoeff * (fineTarget - fineSmoothed);
154 if (std::fabs(fineTarget - fineSmoothed) < kSmoothSnapEps)
155 fineSmoothed = fineTarget;
156 spreadSmoothed += smoothCoeff * (spreadTarget - spreadSmoothed);
157 if (std::fabs(spreadTarget - spreadSmoothed) < kSmoothSnapEps)
158 spreadSmoothed = spreadTarget;
159 motionSmoothed += smoothCoeff * (motionTarget - motionSmoothed);
160 if (std::fabs(motionTarget - motionSmoothed) < kSmoothSnapEps)
161 motionSmoothed = motionTarget;
162 motionRateSmoothed += smoothCoeff * (motionRateTarget - motionRateSmoothed);
163 if (std::fabs(motionRateTarget - motionRateSmoothed) < kSmoothSnapEps)
164 motionRateSmoothed = motionRateTarget;
165 outputSmoothed += smoothCoeff * (outputTarget - outputSmoothed);
166 if (std::fabs(outputTarget - outputSmoothed) < kSmoothSnapEps)
167 outputSmoothed = outputTarget;
168 textureSmoothed += smoothCoeff * (textureTarget - textureSmoothed);
169 if (std::fabs(textureTarget - textureSmoothed) < kSmoothSnapEps)
170 textureSmoothed = textureTarget;
171
172 const float grainSamples =
173 std::max(8.0f, grainMsFromNorm(grainSmoothed) * 0.001f * static_cast<float>(m_sampleRate));
174 const float baseDelay = latencySamples + grainSamples * 0.5f;
175
176 // Complementary Hann taps crossfade at their zero-energy wrap points.
177 // The normalized grain phase makes this behavior sample-rate independent.
178 // When one normalized tap wraps, its window is ~0
179 // and the other tap dominates. No audible discontinuity.
180 float outL = 0.0f;
181 float outR = 0.0f;
182 float winSumL = 0.0f;
183 float winSumR = 0.0f;
184 const float stereoPhaseOffset = spreadSmoothed * 0.0625f;
185 for (size_t t = 0; t < kTapCount; ++t) {
186 float phaseNormL = tapPhase[t] - stereoPhaseOffset;
187 float phaseNormR = tapPhase[t] + stereoPhaseOffset;
188 if (phaseNormL < 0.0f)
189 phaseNormL += 1.0f;
190 if (phaseNormR >= 1.0f)
191 phaseNormR -= 1.0f;
192 // Texture OFF raises the Hann window to the sixth power, narrowing
193 // the overlap region where phase interference causes cyclic color.
194 // Texture blends toward the broader Hann overlap for users who want
195 // that motion as an intentional character control.
196 const float baseWinL = 0.5f * (1.0f - std::cos(twoPi * phaseNormL));
197 const float baseWinR = 0.5f * (1.0f - std::cos(twoPi * phaseNormR));
198 const float baseSquaredL = baseWinL * baseWinL;
199 const float baseSquaredR = baseWinR * baseWinR;
200 const float pureWinL = baseSquaredL * baseSquaredL * baseSquaredL;
201 const float pureWinR = baseSquaredR * baseSquaredR * baseSquaredR;
202 const float winL = pureWinL + textureSmoothed * (baseWinL - pureWinL);
203 const float winR = pureWinR + textureSmoothed * (baseWinR - pureWinR);
204 winSumL += winL;
205 winSumR += winR;
206
207 const float readPosL = static_cast<float>(writePos) - baseDelay + phaseNormL * grainSamples;
208 const float readPosR = static_cast<float>(writePos) - baseDelay + phaseNormR * grainSamples;
209 outL += readBuffer(m_buffer, readPosL, bufferMask) * winL;
210 outR += readBuffer(m_bufferR, readPosR, bufferMask) * winR;
211 }
212 outL /= std::max(winSumL, 0.001f);
213 outR /= std::max(winSumR, 0.001f);
214
215 const float pitchNorm = pitchSmoothed;
216 const float coarseSemitones = -12.0f + pitchNorm * 24.0f;
217 const float fineSemitones = fineSmoothed * 2.0f - 1.0f;
218 const float motionDepthSemitones = motionSmoothed * motionSmoothed * 0.25f;
219 const float motionRateHz = 0.05f * std::pow(40.0f, motionRateSmoothed);
220 motionPhase += motionRateHz / static_cast<float>(m_sampleRate);
221 if (motionPhase >= 1.0f)
222 motionPhase -= 1.0f;
223 const float pitchSemitones =
224 coarseSemitones + fineSemitones + std::sin(twoPi * motionPhase) * motionDepthSemitones;
225 const float pitchRatio = std::pow(2.0f, pitchSemitones / 12.0f);
226 const float delta = pitchRatio - 1.0f;
227
228 for (size_t t = 0; t < kTapCount; ++t) {
229 tapPhase[t] += delta / grainSamples;
230 // Wrap: when one tap wraps, its Hann window is near 0
231 if (tapPhase[t] >= 1.0f)
232 tapPhase[t] -= 1.0f;
233 if (tapPhase[t] < 0.0f)
234 tapPhase[t] += 1.0f;
235 }
236
237 const float dryL = readBuffer(m_buffer, static_cast<float>(writePos) - latencySamples, bufferMask);
238 const float dryR = readBuffer(m_bufferR, static_cast<float>(writePos) - latencySamples, bufferMask);
239 if (std::fabs(pitchSemitones) < 1.0e-6f) {
240 outL = dryL;
241 outR = dryR;
242 }
243
244 writePos = (writePos + 1) & bufferMask;
245
246 const float mix = mixSmoothed;
247 const float wet = mix;
248 const float dry = 1.0f - wet;
249 const float outputGain = std::pow(10.0f, (outputSmoothed * 24.0f - 12.0f) / 20.0f);
250
251 if (numOutputChannels > 0 && outputs[0])
252 outputs[0][i] = sanitizeOutput((dryL * dry + outL * wet) * outputGain);
253 if (numOutputChannels > 1 && outputs[1])
254 outputs[1][i] = sanitizeOutput((dryR * dry + outR * wet) * outputGain);
255 for (uint32_t ch = 2; ch < numOutputChannels; ++ch) {
256 if (outputs[ch])
257 outputs[ch][i] = 0.0f;
258 }
259 }
260
261 m_writePos = writePos;
262 m_tapPhase = tapPhase;
263 m_motionPhase = motionPhase;
264 m_pitchSmoothed = pitchSmoothed;
265 m_grainSmoothed = grainSmoothed;
266 m_mixSmoothed = mixSmoothed;
267 m_fineSmoothed = fineSmoothed;
268 m_spreadSmoothed = spreadSmoothed;
269 m_motionSmoothed = motionSmoothed;
270 m_motionRateSmoothed = motionRateSmoothed;
271 m_outputSmoothed = outputSmoothed;
272 m_textureSmoothed = textureSmoothed;
273 }

306 std::string getParameterDisplay(uint32_t id) const override {
307 if (id >= kParamCount)
308 return "";
309 const float v = getParameter(id);
310 switch (id) {
311 case kPitch: {
312 const float st = -12.0f + v * 24.0f;
313 char buf[32];
314 std::snprintf(buf, sizeof(buf), "%+.1f st", st);
315 return buf;
316 }
317 case kGrain: {
318 if (v <= 0.001f)
319 return "PURE";
320 const float ms = grainMsFromNorm(v);
321 char buf[32];
322 std::snprintf(buf, sizeof(buf), "%.0f ms", ms);
323 return buf;
324 }
325 case kMix:
326 return std::to_string(static_cast<int>(std::round(v * 100.0f))) + "%";
327 case kBypass:
328 return v > 0.5f ? "ON" : "OFF";
329 case kFine: {
330 char buf[32];
331 std::snprintf(buf, sizeof(buf), "%+.0f ct", v * 200.0f - 100.0f);
332 return buf;
333 }
334 case kSpread:
335 return std::to_string(static_cast<int>(std::round(v * 100.0f))) + "%";
336 case kMotion: {
337 char buf[32];
338 std::snprintf(buf, sizeof(buf), "%.0f ct", v * v * 25.0f);
339 return buf;
340 }
341 case kMotionRate: {
342 char buf[32];
343 std::snprintf(buf, sizeof(buf), "%.2f Hz", 0.05f * std::pow(40.0f, v));
344 return buf;
345 }
346 case kOutput: {
347 char buf[32];
348 std::snprintf(buf, sizeof(buf), "%+.1f dB", v * 24.0f - 12.0f);
349 return buf;
350 }
351 case kTexture:
352 return v <= 0.001f ? "OFF" : std::to_string(static_cast<int>(std::round(v * 100.0f))) + "%";
353 default:
354 return "";
355 }

358 std::vector<uint8_t> saveState() const override {
359 std::vector<uint8_t> state(sizeof(uint32_t) * 2u + sizeof(float) * kParamCount);
360 const uint32_t magic = kStateMagic;
361 const uint32_t version = kStateVersion;
362 std::memcpy(state.data(), &magic, sizeof(magic));
363 std::memcpy(state.data() + sizeof(magic), &version, sizeof(version));
364 for (uint32_t i = 0; i < kParamCount; ++i) {
365 const float value = getParameter(i);
366 std::memcpy(state.data() + sizeof(uint32_t) * 2u + sizeof(float) * i, &value, sizeof(value));
367 }
368 return state;
369 }

371 bool loadState(const std::vector<uint8_t>& state) override {
372 if (state.size() < sizeof(uint32_t) * 2)
373 return false;
374 uint32_t magic = 0;
375 uint32_t version = 0;
376 std::memcpy(&magic, state.data(), sizeof(magic));
377 std::memcpy(&version, state.data() + sizeof(magic), sizeof(version));
378 if (magic != kStateMagic)
379 return false;
380 const uint32_t storedParamCount = version == 1u ? kLegacyParamCount
381 : version == 2u ? kV2ParamCount
382 : version == kStateVersion ? kParamCount
383 : 0u;
384 if (storedParamCount == 0u || state.size() < sizeof(uint32_t) * 2u + sizeof(float) * storedParamCount)
385 return false;
386 if (version != kStateVersion) {
387 const auto defaults = getParameters();
388 for (uint32_t i = storedParamCount; i < kParamCount; ++i)
389 setParameter(i, defaults[i].defaultValue);
390 }
391 for (uint32_t i = 0; i < storedParamCount; ++i) {
392 float value = 0.0f;
393 std::memcpy(&value, state.data() + sizeof(uint32_t) * 2u + sizeof(float) * i, sizeof(value));
394 setParameter(i, value);
395 }
396 if (version == 1u)
397 setParameter(kGrain, 0.0f); // V1 stored this value but never applied it to DSP.
398 return true;
399 }

431 // Cubic Hermite interpolation
432 static float readBuffer(const std::vector<float>& buf, float readPos, int bufferMask) {
433 const int i1 = static_cast<int>(std::floor(readPos)) & bufferMask;
434 const int i0 = (i1 - 1) & bufferMask;
435 const int i2 = (i1 + 1) & bufferMask;
436 const int i3 = (i2 + 1) & bufferMask;
437 const float frac = readPos - std::floor(readPos);
438 const float y0 = buf[static_cast<size_t>(i0)];
439 const float y1 = buf[static_cast<size_t>(i1)];
440 const float y2 = buf[static_cast<size_t>(i2)];
441 const float y3 = buf[static_cast<size_t>(i3)];
442 const float a = -0.5f * y0 + 1.5f * y1 - 1.5f * y2 + 0.5f * y3;
443 const float b = y0 - 2.5f * y1 + 2.0f * y2 - 0.5f * y3;
444 const float c = -0.5f * y0 + 0.5f * y2;
445 return ((a * frac + b) * frac + c) * frac + y1;
446 }

448 void resetRuntimeState() {
449 m_writePos = 0;
450 for (size_t i = 0; i < kTapCount; ++i)
451 m_tapPhase[i] = static_cast<float>(i) / static_cast<float>(kTapCount);
452 m_motionPhase = 0.0f;
453 // Snap smoothers to the current targets so load/activate doesn't glide.
454 m_pitchSmoothed = m_params[kPitch].load(std::memory_order_relaxed);
455 m_grainSmoothed = m_params[kGrain].load(std::memory_order_relaxed);
456 m_mixSmoothed = m_params[kMix].load(std::memory_order_relaxed);
457 m_fineSmoothed = m_params[kFine].load(std::memory_order_relaxed);
458 m_spreadSmoothed = m_params[kSpread].load(std::memory_order_relaxed);
459 m_motionSmoothed = m_params[kMotion].load(std::memory_order_relaxed);
460 m_motionRateSmoothed = m_params[kMotionRate].load(std::memory_order_relaxed);
461 m_outputSmoothed = m_params[kOutput].load(std::memory_order_relaxed);
462 m_textureSmoothed = m_params[kTexture].load(std::memory_order_relaxed);
463 }
