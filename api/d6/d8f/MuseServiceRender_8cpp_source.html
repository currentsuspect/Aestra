36// Interleaved stereo float32 WAV — the format the offline test renderer uses.
37bool writeFloat32Wav(const std::string& path, const std::vector<float>& samples,
38 uint32_t sampleRate) {
39 std::ofstream file(path, std::ios::binary);
40 if (!file) return false;
41
42 const uint32_t dataBytes = static_cast<uint32_t>(samples.size() * sizeof(float));
43 const uint16_t channels = 2;
44 const uint16_t bitsPerSample = 32;
45 const uint16_t blockAlign = channels * bitsPerSample / 8;
46
47 auto u32 = [&](uint32_t v) { file.write(reinterpret_cast<const char*>(&v), 4); };
48 auto u16 = [&](uint16_t v) { file.write(reinterpret_cast<const char*>(&v), 2); };
49 file.write("RIFF", 4);
50 u32(36 + dataBytes);
51 file.write("WAVE", 4);
52 file.write("fmt ", 4);
53 u32(16);
54 u16(3); // IEEE float
55 u16(channels);
56 u32(sampleRate);
57 u32(sampleRate * blockAlign);
58 u16(blockAlign);
59 u16(bitsPerSample);
60 file.write("data", 4);
61 u32(dataBytes);
62 file.write(reinterpret_cast<const char*>(samples.data()), dataBytes);
63 file.close(); // a failed flush on close must not report success
64 return !file.fail();
65}

67// Render the whole timeline through AudioExporter and report the measurements
68// the exporter already made.
69//
70// Shared by render_song and verify_render so the two cannot drift: a verdict
71// computed from a different render path than the one the agent used to make the
72// file would be measuring something other than what it claims to confirm.
73//
74// The body is render_song's, lifted out unchanged. What it guarantees on every
75// exit path: transport stopped, pattern mode restored, slot map rebuilt for
76// tracks added since the engine was wired, and the audio graph drained — which
77// in the app happens continuously in AestraApp::run() and has to be done by
78// hand here.
79bool renderFullSong(AudioEngine& engine, TrackManager& trackManager,
80 const std::string& outputPath, double tailSeconds,
81 AudioExporter::Result& outResult) {
82 trackManager.buildAndShareSlotMap();
83 if (auto slotMap = trackManager.getChannelSlotMapShared()) {
84 engine.setChannelSlotMap(slotMap);
85 }
86
87 // The timeline render path routes tracks to master through the published
88 // audio graph; in the app AestraApp::run() drains rebuilds continuously,
89 // headless we drain here.
90 PlaybackGraphController graphController;
91 graphController.setTrackManager(&trackManager);
92 graphController.setAudioEngine(&engine);
93 graphController.requestRebuild(GraphDirtyReason::TimelineChanged);
94 graphController.drainIfDirty(static_cast<double>(engine.getSampleRate()));
95
96 {
97 // TrackManager::play() is what schedules timeline MIDI clip instances
98 // into the pattern engine — the exporter only drives the engine
99 // transport. Guarantee stop + pattern-mode restore on every exit.
100 struct TimelineGuard {
101 TrackManager& trackManager;
102 AudioEngine& engine;
103 ~TimelineGuard() {
104 trackManager.stop();
105 std::vector<float> settle(1024, 0.0f);
106 for (int i = 0; i < 2; ++i) {
107 engine.processBlock(settle.data(), nullptr, 512, 0.0);
108 engine.performNonRealtimeMaintenance();
109 }
110 }
111 } guard{trackManager, engine};
112
113 // Leave any Arsenal pattern-mode state behind: TrackManager's play() only
114 // schedules timeline MIDI clip instances when its own pattern-mode flag
115 // is clear.
116 trackManager.stopArsenalPlayback(false);
117 engine.setPatternPlaybackMode(false, 4.0);
118 trackManager.setPosition(0.0);
119 trackManager.play();
120
121 AudioExporter exporter(engine, trackManager);
122 AudioExporter::Config config;
123 config.outputPath = outputPath;
124 config.scope = AudioExporter::RenderScope::FullSong;
125 config.sampleRate = engine.getSampleRate();
126 config.bitDepth = AudioExporter::BitDepth::Float_32;
127 config.numChannels = 2;
128 config.tailSeconds = tailSeconds;
129 outResult = exporter.render(config);
130 }
131 return outResult.success;
132}

138std::optional<std::string> handleRenderVerbs(const RequestContext& ctx, const ResponseEnvelope& env) {
139 // Bound to the names the moved bodies already use, so no verb body had to be
140 // edited to make the move.
141 const double id = ctx.id;
142 const std::string& verb = ctx.verb;
143 JSON& request = *ctx.request;
144 TrackManager* const m_trackManager = ctx.trackManager;
145 AudioEngine* const m_engine = ctx.engine;
146 // A reference, not a pointer: get_project_load_report tests this for a value
147 // and then dereferences it, so it has to carry std::optional's own semantics.
148 // A pointer would be truthy whenever the address is non-null — including when
149 // the optional it points at is empty — which is not the same question.
150 const std::optional<JSON>& m_projectLoadReport = *ctx.projectLoadReport;
151
152 const auto makeOk = [&env]() { return env.ok(); };
153 const auto finish = [&env](JSON& response) { return env.finish(response); };
154
155 if (verb == "render_song") {
156 if (!m_trackManager || !m_engine) {
157 return makeError(id, "execution_error",
158 "render_song needs a track manager and an audio engine", verb)
159 .toString();
160 }
161 if (!request.has("args") || !request["args"].isObject()) {
162 return makeError(id, "validation_error",
163 "render_song requires args: {\"file\": <path>}", verb)
164 .toString();
165 }
166 JSON& args = request["args"];
167 for (auto& entry : args.asObject()) {
168 if (entry.first != "file" && entry.first != "tail") {
169 return makeError(id, "validation_error",
170 "unknown arg for render_song: " + entry.first, verb)
171 .toString();
172 }
173 }
174 if (!args.has("file") || !args["file"].isString() || args["file"].asString().empty()) {
175 return makeError(id, "validation_error", "arg 'file' must be a non-empty string",
176 verb)
177 .toString();
178 }
179 double tailSeconds = 1.0;
180 if (args.has("tail")) {
181 if (!args["tail"].isNumber()) {
182 return makeError(id, "validation_error", "arg 'tail' must be a number", verb)
183 .toString();
184 }
185 tailSeconds = args["tail"].asNumber();
186 if (!(tailSeconds >= 0.0 && tailSeconds <= 30.0)) {
187 return makeError(id, "validation_error", "arg 'tail' must be 0..30 seconds",
188 verb)
189 .toString();
190 }
191 }
192
193 AudioExporter::Result exportResult;
194 renderFullSong(*m_engine, *m_trackManager, args["file"].asString(), tailSeconds,
195 exportResult);
196
197 if (!exportResult.success) {
198 return makeError(id, "execution_error", exportResult.errorMessage, verb)
199 .toString();
200 }
201
202 JSON result = JSON::object();
203 result.set("file", JSON(exportResult.outputPath));
204 result.set("durationSeconds", JSON(exportResult.durationSeconds));
205 result.set("frames", JSON(static_cast<double>(exportResult.framesRendered)));
206 result.set("sampleRate", JSON(static_cast<double>(m_engine->getSampleRate())));
207 result.set("peakDb", JSON(exportResult.peakDb));
208 JSON response = makeOk();
209 response.set("result", result);
210 return finish(response);
211 }
212
213 if (verb == "render_pattern") {
214 if (!m_trackManager || !m_engine) {
215 return makeError(id, "execution_error",
216 "render_pattern needs a track manager and an audio engine", verb)
217 .toString();
218 }
219 if (!request.has("args") || !request["args"].isObject()) {
220 return makeError(
221 id, "validation_error",
222 "render_pattern requires args: {\"pattern\": <id>, \"file\": <path>}",
223 verb)
224 .toString();
225 }
226 JSON& args = request["args"];
227 for (auto& entry : args.asObject()) {
228 if (entry.first != "pattern" && entry.first != "file" && entry.first != "tail") {
229 return makeError(id, "validation_error",
230 "unknown arg for render_pattern: " + entry.first, verb)
231 .toString();
232 }
233 }
234 uint64_t patternValue = 0;
235 if (!args.has("pattern") || !args["pattern"].isNumber() ||
236 !numberToId(args["pattern"].asNumber(), patternValue)) {
237 return makeError(id, "validation_error",
238 "arg 'pattern' must be a non-negative integer", verb)
239 .toString();
240 }
241 if (!args.has("file") || !args["file"].isString() || args["file"].asString().empty()) {
242 return makeError(id, "validation_error", "arg 'file' must be a non-empty string",
243 verb)
244 .toString();
245 }
246 double tailSeconds = 1.0;
247 if (args.has("tail")) {
248 if (!args["tail"].isNumber()) {
249 return makeError(id, "validation_error", "arg 'tail' must be a number", verb)
250 .toString();
251 }
252 tailSeconds = args["tail"].asNumber();
253 if (!(tailSeconds >= 0.0 && tailSeconds <= 30.0)) {
254 return makeError(id, "validation_error", "arg 'tail' must be 0..30 seconds",
255 verb)
256 .toString();
257 }
258 }
259
260 const PatternID patternId{static_cast<uint64_t>(args["pattern"].asNumber())};
261 const PatternSource* pattern =
262 m_trackManager->getPatternManager().getPattern(patternId);
263 if (!pattern) {
264 return makeError(id, "execution_error",
265 "no such pattern: " + std::to_string(patternId.value), verb)
266 .toString();
267 }
268 if (!pattern->isMidi()) {
269 return makeError(id, "execution_error", "pattern is not a MIDI pattern", verb)
270 .toString();
271 }
272
273 const double bpm = std::max(1.0, static_cast<double>(m_engine->getBPM()));
274 // playPatternInArsenal resolves pattern length to at least 8 beats.
275 const double lengthBeats = std::max(8.0, pattern->lengthBeats);
276 const double durationSeconds = lengthBeats * 60.0 / bpm + tailSeconds;
277 const uint32_t sampleRate = m_engine->getSampleRate();
278 const uint64_t totalFrames =
279 static_cast<uint64_t>(durationSeconds * static_cast<double>(sampleRate));
280 constexpr uint32_t kBlockFrames = 512;
281
282 // Bound the render before touching engine state: the whole take
283 // is buffered in memory and the RIFF format caps a WAV at 4 GiB.
284 constexpr double kMaxRenderSeconds = 600.0;
285 const uint64_t dataBytes = totalFrames * 2ull * sizeof(float);
286 if (durationSeconds > kMaxRenderSeconds ||
287 36ull + dataBytes > 0xFFFFFFFFull) {
288 return makeError(id, "validation_error",
289 "render too long: " + std::to_string(durationSeconds) +
290 "s (max " + std::to_string(kMaxRenderSeconds) + "s)",
291 verb)
292 .toString();
293 }
294
295 // Pattern playback now shares the mixer graph with Timeline playback.
296 // Headless callers do not have the app loop to publish recent unit or
297 // channel changes, so prepare the current graph before pumping audio.
298 m_trackManager->buildAndShareSlotMap();
299 if (auto slotMap = m_trackManager->getChannelSlotMapShared()) {
300 m_engine->setChannelSlotMap(slotMap);
301 }
302 PlaybackGraphController graphController;
303 graphController.setTrackManager(m_trackManager);
304 graphController.setAudioEngine(m_engine);
305 graphController.requestRebuild(GraphDirtyReason::TimelineChanged);
306 graphController.drainIfDirty(static_cast<double>(sampleRate));
307
308 // Keep-length samplers play pre-rendered copies: never bounce the resampled fallback.
309 if (!m_trackManager->prewarmSamplerKeepLength(true)) {
310 return makeError(id, "execution_error", "keep-length sampler renders did not finish in time", verb)
311 .toString();
312 }
313
314 std::vector<float> rendered;
315 rendered.reserve(static_cast<size_t>(totalFrames) * 2u);
316 std::vector<float> block(static_cast<size_t>(kBlockFrames) * 2u, 0.0f);
317 float peak = 0.0f;
318 bool nonFinite = false;
319
320 {
321 // Everything after playback starts must be undone even if the
322 // pump throws: stop the transport, drain the stop command with
323 // settle blocks, and leave pattern mode (app default length).
324 struct TransportGuard {
325 AudioEngine& engine;
326 TrackManager& trackManager;
327 std::vector<float>& block;
328 ~TransportGuard() {
329 // Full Arsenal teardown: stop, clear the scheduled
330 // instance, and leave pattern mode — a lingering
331 // pattern-mode flag or instance would bleed into the
332 // next timeline play/render.
333 trackManager.stopArsenalPlayback(false);
334 for (int i = 0; i < 2; ++i) {
335 std::memset(block.data(), 0, block.size() * sizeof(float));
336 engine.processBlock(block.data(), nullptr, kBlockFrames, 0.0);
337 engine.performNonRealtimeMaintenance();
338 }
339 engine.setPatternPlaybackMode(false, 4.0);
340 }
341 } guard{*m_engine, *m_trackManager, block};
342
343 // Offline bounce through the exact live engine path, the same
344 // way the headless test renderer pumps it. Arsenal preview
345 // routing is what the user hears when a pattern plays, so it
346 // is what the agent gets back. Pattern playback needs both
347 // sides armed: the scheduler (playPatternInArsenal) and the
348 // engine's pattern mode (the app sets it wherever it starts
349 // pattern playback).
350 m_engine->setPatternPlaybackMode(true, lengthBeats);
351 m_trackManager->playPatternInArsenal(patternId, 0.0);
352
353 uint64_t framesRemaining = totalFrames;
354 while (framesRemaining > 0 && !nonFinite) {
355 const uint32_t framesThisBlock =
356 static_cast<uint32_t>(std::min<uint64_t>(kBlockFrames, framesRemaining));
357 std::memset(block.data(), 0, block.size() * sizeof(float));
358 m_engine->processBlock(block.data(), nullptr, framesThisBlock, 0.0);
359 // The pattern scheduler's RT queue is refilled from the
360 // control thread; offline that's us (AudioExporter does
361 // the same per block).
362 m_engine->performNonRealtimeMaintenance();
363 for (uint32_t i = 0; i < framesThisBlock * 2u; ++i) {
364 // Plugin output is untrusted: NaN/Inf must fail the
365 // render, not land in the file as "ok".
366 if (!std::isfinite(block[i])) {
367 nonFinite = true;
368 break;
369 }
370 peak = std::max(peak, std::abs(block[i]));
371 }
372 if (nonFinite) break;
373 rendered.insert(rendered.end(), block.begin(),
374 block.begin() + static_cast<size_t>(framesThisBlock) * 2u);
375 framesRemaining -= framesThisBlock;
376 }
377 }
378
379 if (nonFinite) {
380 return makeError(id, "execution_error",
381 "engine produced non-finite audio; render aborted", verb)
382 .toString();
383 }
384
385 if (!writeFloat32Wav(args["file"].asString(), rendered, sampleRate)) {
386 return makeError(id, "execution_error",
387 "cannot write output file: " + args["file"].asString(), verb)
388 .toString();
389 }
390
391 const double peakDb = peak > 0.0f ? 20.0 * std::log10(static_cast<double>(peak))
392 : -144.0;
393 JSON result = JSON::object();
394 result.set("file", JSON(args["file"].asString()));
395 result.set("durationSeconds", JSON(durationSeconds));
396 result.set("frames", JSON(static_cast<double>(totalFrames)));
397 result.set("sampleRate", JSON(static_cast<double>(sampleRate)));
398 result.set("peakDb", JSON(peakDb));
399 JSON response = makeOk();
400 response.set("result", result);
401 return finish(response);
402 }
403 // --- verify_render -------------------------------------------------------
404 //
405 // The confirmation verb. render_song reports what came out; this one says
406 // whether that is acceptable, so an agent is not left eyeballing peakDb and
407 // deciding for itself whether -96 dBFS means "quiet" or "broken".
408 //
409 // It reports the exporter's OWN measurements rather than re-deriving them
410 // from the samples: peakDb and the true-peak ceiling are already computed
411 // during the render, and a second estimate of the same number would be a
412 // second thing that can be wrong.
413 if (verb == "verify_render") {
414 if (!m_trackManager || !m_engine) {
415 return makeError(id, "execution_error",
416 "verify_render needs a track manager and an audio engine", verb)
417 .toString();
418 }
419 if (!request.has("args") || !request["args"].isObject()) {
420 return makeError(id, "validation_error",
421 "verify_render requires args: {\"file\": <path>}", verb)
422 .toString();
423 }
424 JSON& args = request["args"];
425 for (auto& entry : args.asObject()) {
426 const std::string& key = entry.first;
427 const bool known = key == "file" || key == "tail" || key == "against" ||
428 key == "expect_peak_min_db" || key == "expect_peak_max_db" ||
429 key == "expect_not_silent" || key == "expect_no_clipping";
430 if (!known) {
431 return makeError(id, "validation_error", "unknown arg for verify_render: " + key,
432 verb)
433 .toString();
434 }
435 }
436 if (!args.has("file") || !args["file"].isString() || args["file"].asString().empty()) {
437 return makeError(id, "validation_error", "arg 'file' must be a non-empty string",
438 verb)
439 .toString();
440 }
441
442 double tailSeconds = 1.0;
443 if (args.has("tail")) {
444 if (!args["tail"].isNumber()) {
445 return makeError(id, "validation_error", "arg 'tail' must be a number", verb)
446 .toString();
447 }
448 tailSeconds = args["tail"].asNumber();
449 if (!(tailSeconds >= 0.0 && tailSeconds <= 30.0)) {
450 return makeError(id, "validation_error", "arg 'tail' must be 0..30 seconds",
451 verb)
452 .toString();
453 }
454 }
455
456 // Defaults chosen to catch the two failures that actually happen: a
457 // render that came out silent, and one that clipped. Both default on,
458 // because "did I hear anything" and "is it too loud" are questions with
459 // no comfortable answer when you have to guess. A caller who wants a
460 // quiet render on purpose turns the first off explicitly.
461 double peakMinDb = -89.0;
462 if (args.has("expect_peak_min_db")) {
463 if (!args["expect_peak_min_db"].isNumber()) {
464 return makeError(id, "validation_error",
465 "arg 'expect_peak_min_db' must be a number", verb)
466 .toString();
467 }
468 peakMinDb = args["expect_peak_min_db"].asNumber();
469 }
470 double peakMaxDb = 0.0;
471 if (args.has("expect_peak_max_db")) {
472 if (!args["expect_peak_max_db"].isNumber()) {
473 return makeError(id, "validation_error",
474 "arg 'expect_peak_max_db' must be a number", verb)
475 .toString();
476 }
477 peakMaxDb = args["expect_peak_max_db"].asNumber();
478 }
479 bool expectNotSilent = true;
480 if (args.has("expect_not_silent")) {
481 if (!args["expect_not_silent"].isBool()) {
482 return makeError(id, "validation_error",
483 "arg 'expect_not_silent' must be true or false", verb)
484 .toString();
485 }
486 expectNotSilent = args["expect_not_silent"].asBool();
487 }
488 bool expectNoClipping = true;
489 if (args.has("expect_no_clipping")) {
490 if (!args["expect_no_clipping"].isBool()) {
491 return makeError(id, "validation_error",
492 "arg 'expect_no_clipping' must be true or false", verb)
493 .toString();
494 }
495 expectNoClipping = args["expect_no_clipping"].asBool();
496 }
497 std::string against;
498 if (args.has("against")) {
499 if (!args["against"].isString() || args["against"].asString().empty()) {
500 return makeError(id, "validation_error",
501 "arg 'against' must be a non-empty string", verb)
502 .toString();
503 }
504 against = args["against"].asString();
505 }
506
507 // Hash the reference BEFORE rendering. If 'against' names the same file as
508 // 'file', the render overwrites it, and hashing afterwards would compare
509 // the new file against itself — matches_previous_render would then pass
510 // unconditionally, which is the one failure mode this verb must not have.
511 // Reading it first also keeps "render again to the same path and compare"
512 // working, which is the natural way an agent checks reproducibility.
513 std::string wantDigest;
514 if (!against.empty()) {
515 wantDigest = fnv1aFileDigest(against);
516 if (wantDigest.empty()) {
517 return makeError(id, "execution_error",
518 "could not digest '" + against + "' for comparison", verb)
519 .toString();
520 }
521 }
522
523 AudioExporter::Result exportResult;
524 renderFullSong(*m_engine, *m_trackManager, args["file"].asString(), tailSeconds,
525 exportResult);
526 if (!exportResult.success) {
527 // A render that could not happen is not a verdict, so it stays an
528 // error rather than becoming verdict:"fail".
529 return makeError(id, "execution_error", exportResult.errorMessage, verb).toString();
530 }
531
532 JSON checks = JSON::array();
533 bool allPassed = true;
534 const auto addCheck = [&](const char* name, bool passed, const std::string& expected,
535 const std::string& actual) {
536 JSON check = JSON::object();
537 check.set("name", JSON(std::string(name)));
538 check.set("pass", JSON(passed));
539 check.set("expected", JSON(expected));
540 check.set("actual", JSON(actual));
541 checks.push(check);
542 if (!passed) allPassed = false;
543 };
544 const auto fmt = [](double value, int places) {
545 char buf[48];
546 std::snprintf(buf, sizeof(buf), "%.*f", places, value);
547 return std::string(buf);
548 };
549
550 if (expectNotSilent) {
551 const bool silent = !(exportResult.peakDb > peakMinDb);
552 addCheck("not_silent", !silent,
553 "peak above " + fmt(peakMinDb, 1) + " dBFS",
554 "peak " + fmt(exportResult.peakDb, 2) + " dBFS");
555 }
556 {
557 const bool inRange = exportResult.peakDb <= peakMaxDb;
558 addCheck("peak_in_range", inRange,
559 "peak at or below " + fmt(peakMaxDb, 1) + " dBFS",
560 "peak " + fmt(exportResult.peakDb, 2) + " dBFS");
561 }
562 if (expectNoClipping) {
563 const bool clipped = exportResult.truePeakCeilingExceeded;
564 addCheck("no_clipping", !clipped, "true peak under the ceiling",
565 fmt(static_cast<double>(exportResult.maxTruePeakdBTP), 2) + " dBTP" +
566 (clipped ? " (ceiling exceeded)" : ""));
567 }
568
569 JSON result = JSON::object();
570 result.set("file", JSON(exportResult.outputPath));
571 result.set("durationSeconds", JSON(exportResult.durationSeconds));
572 result.set("frames", JSON(static_cast<double>(exportResult.framesRendered)));
573 result.set("sampleRate", JSON(static_cast<double>(m_engine->getSampleRate())));
574 result.set("peakDb", JSON(exportResult.peakDb));
575 // dBTP, so the key reads maxTruePeakDbTp — not maxTruePeakDbTtp.
576 result.set("maxTruePeakDbTp", JSON(static_cast<double>(exportResult.maxTruePeakdBTP)));
577 result.set("truePeakCeilingExceeded", JSON(exportResult.truePeakCeilingExceeded));
578
579 if (!against.empty()) {
580 // Byte-for-byte reproducibility: an unchanged session must render
581 // to an identical file. A digest is the only way to say that.
582 // wantDigest was taken BEFORE the render, so this cannot end up
583 // comparing the new file against itself.
584 const std::string got = fnv1aFileDigest(exportResult.outputPath);
585 if (got.empty()) {
586 return makeError(id, "execution_error",
587 "could not digest '" + exportResult.outputPath +
588 "' for comparison", verb)
589 .toString();
590 }
591 result.set("digest", JSON(got));
592 result.set("comparedWith", JSON(against));
593 addCheck("matches_previous_render", got == wantDigest,
594 "identical bytes to " + against,
595 got == wantDigest ? got : got + " != " + wantDigest);
596 }
597
598 result.set("checks", checks);
599 result.set("verdict", JSON(std::string(allPassed ? "pass" : "fail")));
600
601 // status stays "ok" even when the verdict is "fail": the verb did its job
602 // and the thing it judged is wrong. Conflating the two would teach an
603 // agent to retry a failing render instead of fixing the mix.
604 JSON response = makeOk();
605 response.set("result", result);
606 return finish(response);
607 }
608
609 // No verb matched. In the single-file version this fell out of the if-chain
610 // into the next family's checks and eventually the mutation path, which is
611 // exactly what nullopt asks handleRequest to do.
612 return std::nullopt;
613}
