130 static bool validateRegistry(const std::vector<Migration>& migrations, int minSupportedVersion, int currentVersion,
131 std::string* error = nullptr) {
132 if (minSupportedVersion < 1 || currentVersion < minSupportedVersion) {
133 if (error) {
134 *error = "invalid supported project-version range";
135 }
136 return false;
137 }
138
139 std::vector<int> edgeCounts(static_cast<size_t>(currentVersion - minSupportedVersion), 0);
140 for (const auto& migration : migrations) {
141 if (migration.fromVersion < minSupportedVersion || migration.fromVersion >= currentVersion ||
142 migration.toVersion != migration.fromVersion + 1) {
143 if (error) {
144 *error = "migration edge must be adjacent and inside the supported range";
145 }
146 return false;
147 }
148 ++edgeCounts[static_cast<size_t>(migration.fromVersion - minSupportedVersion)];
149 }
150
151 for (size_t index = 0; index < edgeCounts.size(); ++index) {
152 if (edgeCounts[index] != 1) {
153 if (error) {
154 const int fromVersion = minSupportedVersion + static_cast<int>(index);
155 *error = edgeCounts[index] == 0 ? "missing migration edge from v" + std::to_string(fromVersion)
156 : "duplicate migration edge from v" + std::to_string(fromVersion);
157 }
158 return false;
159 }
160 }
161 return true;
162 }

170 static MigrationResult runMigrationsWith(JSON& root, int fromVersion, int toVersion,
171 const std::vector<Migration>& migrations) {
172 if (fromVersion > toVersion) {
173 return {MigrationOutcome::Failed, fromVersion};
174 }
175
176 MigrationOutcome outcome = MigrationOutcome::None;
177 int currentVersion = fromVersion;
178 while (currentVersion < toVersion) {
179 const Migration* edge = nullptr;
180 for (const auto& m : migrations) {
181 if (m.fromVersion == currentVersion && m.toVersion == currentVersion + 1) {
182 if (edge != nullptr) {
183 return {MigrationOutcome::Failed, currentVersion};
184 }
185 edge = &m;
186 }
187 }
188 if (edge == nullptr) {
189 return {MigrationOutcome::Failed, currentVersion};
190 }
191
192 const MigrationStepResult stepResult = edge->migrate(root);
193 if (stepResult == MigrationStepResult::Failed) {
194 return {MigrationOutcome::Failed, currentVersion};
195 }
196 if (stepResult == MigrationStepResult::Transformed) {
197 outcome = MigrationOutcome::Transformed;
198 }
199
200 currentVersion = edge->toVersion;
201 root.set("version", JSON(static_cast<double>(currentVersion)));
202 }
203
204 return {outcome, currentVersion};
205 }

207private:
208 // v1 and v2 differ only in the identity guarantees the writer made; every
209 // v1 field means in v2 exactly what it meant in v1. Nothing to transform.
210 static MigrationStepResult migrateV1ToV2(JSON&) { return MigrationStepResult::Unchanged; }
211
212 // Deliberately a no-op. The v3 loader reads v2's lane-local mixer state
213 // natively and derives channel records from it, rather than rewriting the
214 // JSON here.
215 //
216 // That is permitted only because the interpretation is
217 // representation-equivalent and round-trip stable, which is not asserted by
218 // this comment but PROVEN by the historical-fixture tests: an authentic v2
219 // document with a topology that would expose merging or positional
220 // collision loads, re-saves as v3, and reloads with identical semantics.
221 // If that ever stops holding, this function must start reporting
222 // `Transformed` (and do the rewrite) rather than the loader quietly
223 // upgrading meaning.
224 static MigrationStepResult migrateV2ToV3(JSON&) { return MigrationStepResult::Unchanged; }
225};
