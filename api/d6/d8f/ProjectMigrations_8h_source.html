131 static bool validateRegistry(const std::vector<Migration>& migrations, int minSupportedVersion, int currentVersion,
132 std::string* error = nullptr) {
133 if (minSupportedVersion < 1 || currentVersion < minSupportedVersion) {
134 if (error) {
135 *error = "invalid supported project-version range";
136 }
137 return false;
138 }
139
140 std::vector<int> edgeCounts(static_cast<size_t>(currentVersion - minSupportedVersion), 0);
141 for (const auto& migration : migrations) {
142 if (migration.fromVersion < minSupportedVersion || migration.fromVersion >= currentVersion ||
143 migration.toVersion != migration.fromVersion + 1) {
144 if (error) {
145 *error = "migration edge must be adjacent and inside the supported range";
146 }
147 return false;
148 }
149 ++edgeCounts[static_cast<size_t>(migration.fromVersion - minSupportedVersion)];
150 }
151
152 for (size_t index = 0; index < edgeCounts.size(); ++index) {
153 if (edgeCounts[index] != 1) {
154 if (error) {
155 const int fromVersion = minSupportedVersion + static_cast<int>(index);
156 *error = edgeCounts[index] == 0 ? "missing migration edge from v" + std::to_string(fromVersion)
157 : "duplicate migration edge from v" + std::to_string(fromVersion);
158 }
159 return false;
160 }
161 }
162 return true;
163 }

171 static MigrationResult runMigrationsWith(JSON& root, int fromVersion, int toVersion,
172 const std::vector<Migration>& migrations) {
173 if (fromVersion > toVersion) {
174 return {MigrationOutcome::Failed, fromVersion};
175 }
176
177 MigrationOutcome outcome = MigrationOutcome::None;
178 int currentVersion = fromVersion;
179 while (currentVersion < toVersion) {
180 const Migration* edge = nullptr;
181 for (const auto& m : migrations) {
182 if (m.fromVersion == currentVersion && m.toVersion == currentVersion + 1) {
183 if (edge != nullptr) {
184 return {MigrationOutcome::Failed, currentVersion};
185 }
186 edge = &m;
187 }
188 }
189 if (edge == nullptr) {
190 return {MigrationOutcome::Failed, currentVersion};
191 }
192
193 const MigrationStepResult stepResult = edge->migrate(root);
194 if (stepResult == MigrationStepResult::Failed) {
195 return {MigrationOutcome::Failed, currentVersion};
196 }
197 if (stepResult == MigrationStepResult::Transformed) {
198 outcome = MigrationOutcome::Transformed;
199 }
200
201 currentVersion = edge->toVersion;
202 root.set("version", JSON(static_cast<double>(currentVersion)));
203 }
204
205 return {outcome, currentVersion};
206 }

209 // v1 and v2 differ only in the identity guarantees the writer made; every
210 // v1 field means in v2 exactly what it meant in v1. Nothing to transform.
211 static MigrationStepResult migrateV1ToV2(JSON&) { return MigrationStepResult::Unchanged; }
212
213 // Deliberately a no-op. The v3 loader reads v2's lane-local mixer state
214 // natively and derives channel records from it, rather than rewriting the
215 // JSON here.
216 //
217 // That is permitted only because the interpretation is
218 // representation-equivalent and round-trip stable, which is not asserted by
219 // this comment but PROVEN by the historical-fixture tests: an authentic v2
220 // document with a topology that would expose merging or positional
221 // collision loads, re-saves as v3, and reloads with identical semantics.
222 // If that ever stops holding, this function must start reporting
223 // `Transformed` (and do the rewrite) rather than the loader quietly
224 // upgrading meaning.
225 static MigrationStepResult migrateV2ToV3(JSON&) { return MigrationStepResult::Unchanged; }
226
227 // v4 (V8-A6): automation tension is honoured (AutomationCurve.h, automationTensionShape).
228 // Before v4 it was serialized but evaluation was linear, and the UI stored 0.5 on every
229 // point, so honouring stored values would bend every curve ever drawn. The founder ruling
230 // (2026-09-12): no silent activation; same saved state, same audio. So every point's
231 // tension becomes 0, which is exactly the audio the file always produced, and any point
232 // that changes makes this a transformation: the project is marked upgraded.
233 static MigrationStepResult migrateV3ToV4(JSON& root) {
234 bool changed = false;
235 if (!root.has("lanes") || !root["lanes"].isArray()) {
236 return MigrationStepResult::Unchanged;
237 }
238 JSON& lanes = root["lanes"];
239 for (size_t l = 0; l < lanes.size(); ++l) {
240 JSON& lane = lanes[l];
241 if (!lane.isObject() || !lane.has("automation") || !lane["automation"].isArray()) continue;
242 JSON& curves = lane["automation"];
243 for (size_t c = 0; c < curves.size(); ++c) {
244 JSON& curve = curves[c];
245 if (!curve.isObject() || !curve.has("points") || !curve["points"].isArray()) continue;
246 JSON& points = curve["points"];
247 for (size_t p = 0; p < points.size(); ++p) {
248 JSON& point = points[p];
249 if (point.isObject() && point.has("c") && point["c"].isNumber() && point["c"].asNumber() != 0.0) {
250 point.set("c", JSON(0.0));
251 changed = true;
252 }
253 }
254 }
255 }
256 return changed ? MigrationStepResult::Transformed : MigrationStepResult::Unchanged;
257 }
