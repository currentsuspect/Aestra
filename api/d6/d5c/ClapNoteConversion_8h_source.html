6// Shared CLAP note-conversion rules used by BOTH CLAP hosting paths — the
7// in-process host (CLAPHost.cpp, real CLAP SDK) and the out-of-process child
8// (AestraPluginHostMain.cpp, hand-mirrored CLAP ABI). Keeping the dialect
9// selection and MIDI decoding here means the two paths cannot drift (#244).
10//
11// This header is deliberately SDK-free (the OOP child has no CLAP headers) and
12// allocation-free/branch-only, so it is safe to call from the audio thread. It
13// deals in plain integers; each caller builds its own event struct (real
14// clap_event_note vs. the child's mirror) from the decoded fields.

20// clap_note_dialect bits (clap/ext/note-ports.h), mirrored so this header needs
21// no SDK. Only CLAP and MIDI are handled here; MPE / MIDI2 are out of scope.
22enum : uint32_t {
23 kDialectClap = 1u << 0,
24 kDialectMidi = 1u << 1,
25 kDialectMidiMpe = 1u << 2,
26 kDialectMidi2 = 1u << 3,
27};

29enum class MidiNoteKind : uint8_t { NoteOn, NoteOff, Other };
30
31struct DecodedNote {
32 MidiNoteKind kind{MidiNoteKind::Other};
33 uint8_t channel{0}; // 0..15
34 uint8_t key{0}; // 0..127
35 uint8_t velocity{0}; // 0..127 (0 on a note-on means note-off)
36};

38// Decode a 3-byte MIDI channel-voice message. A note-on with velocity 0 is the
39// running-status note-off convention and is reported as NoteOff.
40inline DecodedNote decodeMidiMessage(const uint8_t* data) {
41 DecodedNote out;
42 if (!data) {
43 return out;
44 }
45 const uint8_t status = static_cast<uint8_t>(data[0] & 0xF0);
46 out.channel = static_cast<uint8_t>(data[0] & 0x0F);
47 out.key = static_cast<uint8_t>(data[1] & 0x7F);
48 out.velocity = static_cast<uint8_t>(data[2] & 0x7F);
49 if (status == 0x90) { // Note On
50 out.kind = out.velocity > 0 ? MidiNoteKind::NoteOn : MidiNoteKind::NoteOff;
51 } else if (status == 0x80) { // Note Off
52 out.kind = MidiNoteKind::NoteOff;
53 }
54 return out;
55}

57// Host-emittable note dialects (Aestra can produce native CLAP and raw MIDI;
58// MPE / MIDI2 are out of scope for #244).
59constexpr uint32_t kHostSupportedDialects = kDialectClap | kDialectMidi;
60
61// Pick the note dialect to deliver on a plugin note-input port, deterministically.
62// - Honor a validly-advertised preference: if preferred_dialect is a single
63// dialect present in BOTH the plugin's supported mask and the host's, use it.
64// (In particular, a plugin that supports CLAP+MIDI but prefers CLAP gets
65// CLAP — we never pick raw MIDI merely because MIDI is in supported_dialects.)
66// - If the preference is absent (0) or not mutually supported, fall back
67// deterministically to native CLAP when the port supports it, else MIDI.
68// This is a documented tie-break, not a MIDI-by-default.
69// - Return 0 when there is no mutually supported dialect; the caller must then
70// deliver no notes (and log a diagnostic).
71// @param supported supported_dialects bitmask from the note port.
72// @param preferred preferred_dialect from the note port (one bit, or 0).
73inline uint32_t selectDialect(uint32_t supported, uint32_t preferred) {
74 const uint32_t usable = supported & kHostSupportedDialects;
75 if (usable == 0) {
76 return 0; // no mutually supported dialect — caller delivers no notes
77 }
78 // A validly-advertised preference is a single dialect we both support.
79 const bool preferredIsSingleBit = preferred != 0 && (preferred & (preferred - 1)) == 0;
80 if (preferredIsSingleBit && (preferred & usable) == preferred) {
81 return preferred;
82 }
83 // Absent or unsupported preference: deterministic documented fallback.
84 return (usable & kDialectClap) ? kDialectClap : kDialectMidi;
85}

87// Normalized CLAP note velocity (0..1) from a 7-bit MIDI velocity.
88inline double normalizedVelocity(uint8_t midiVelocity) {
89 return static_cast<double>(midiVelocity) / 127.0;
90}
