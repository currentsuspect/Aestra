16 "command through the same undo history the user's UI edits use.\n\n"
17 "Working method:\n"
18 "- Look before you touch: use the query verbs (get_session_state, list_*, "

29std::string modeInstructions(AgentLoop::Mode mode) {
30 switch (mode) {
31 case AgentLoop::Mode::Direct:
32 return "Mode: direct. Execute exactly what the instruction asks — no "
33 "embellishment, no extra creative choices — then finish.\n";
34 case AgentLoop::Mode::Collaborative:
35 return "Mode: collaborative. Before major creative choices (style, key, "
36 "arrangement direction), call ask_user with one concise question. "
37 "Do not ask about trivia you can decide yourself.\n";
38 case AgentLoop::Mode::Autonomous:
39 return "Mode: autonomous. The user is not watching; never ask questions. "
40 "Make reasonable creative decisions yourself and continue until "
41 "the brief is met or you must stop.\n";
42 }
43 return "";
44}

65std::string AgentLoop::buildSystemPrompt(const std::string& manifestJson) const {
66 std::string prompt = kSystemPromptHeader;
67 prompt += modeInstructions(m_options.mode);
68 prompt += "\nBudget: at most " + std::to_string(m_options.maxIterations) +
69 " turns and " + std::to_string(m_options.maxToolCalls) +
70 " tool calls. Work efficiently; prefer batch and set_steps over "
71 "note-by-note edits.\n";
72
73 // The manifest's notes are the semantics an agent cannot discover through
74 // the protocol (sampler root = pitch 60, routing rules, step characters).
75 try {
76 JSON manifest = JSON::parse(manifestJson);
77 if (manifest.has("notes")) {
78 prompt += "\nEngine semantics you must respect:\n";
79 JSON& notes = manifest["notes"];
80 for (auto& [key, value] : notes.asObject()) {
81 prompt += "- " + key + ": " + value.asString() + "\n";
82 }
83 }
84 } catch (const std::exception&) {
85 // Manifest without notes still yields a usable prompt.
86 }
87 return prompt;
88}

90std::vector<ToolDefinition> AgentLoop::buildTools(const std::string& manifestJson) const {
91 std::vector<ToolDefinition> tools;
92 JSON manifest = JSON::parse(manifestJson);
93
94 if (manifest.has("commands")) {
95 JSON& commands = manifest["commands"];
96 for (size_t i = 0; i < commands.size(); ++i) {
97 JSON& command = commands[i];
98 ToolDefinition tool;
99 tool.name = command["verb"].asString();
100 tool.description =
101 command.has("description") ? command["description"].asString() : tool.name;
102
103 JSON properties = JSON::object();
104 JSON required = JSON::array();
105 if (command.has("flags")) {
106 JSON& flags = command["flags"];
107 for (size_t flagIndex = 0; flagIndex < flags.size(); ++flagIndex) {
108 JSON& flag = flags[flagIndex];
109 JSON property = JSON::object();
110 property.set("type",
111 JSON(schemaTypeName(flag["type"].asString())));
112 if (flag.has("min")) property.set("minimum", flag["min"]);
113 if (flag.has("max")) property.set("maximum", flag["max"]);
114 properties.set(flag["name"].asString(), property);
115 if (flag.has("required") && flag["required"].asBool()) {
116 required.push(JSON(flag["name"].asString()));
117 }
118 }
119 }
120 JSON schema = JSON::object();
121 schema.set("type", JSON("object"));
122 schema.set("properties", properties);
123 if (required.size() > 0) schema.set("required", required);
124 tool.parametersSchema = schema;
125 tools.push_back(tool);
126 }
127 }
128
129 // Queries and actions publish their arg shapes as prose; Muse validates,
130 // so a permissive schema plus the description is honest and sufficient.
131 const auto addLooseSection = [&tools](JSON& section) {
132 for (size_t i = 0; i < section.size(); ++i) {
133 JSON& entry = section[i];
134 ToolDefinition tool;
135 tool.name = entry["verb"].asString();
136 tool.description = entry.has("description") ? entry["description"].asString()
137 : tool.name;
138 if (entry.has("args") && entry["args"].isString() &&
139 entry["args"].asString() != "none") {
140 tool.description += " Args: " + entry["args"].asString();
141 }
142 JSON schema = JSON::object();
143 schema.set("type", JSON("object"));
144 schema.set("properties", JSON::object());
145 tool.parametersSchema = schema;
146 tools.push_back(tool);
147 }
148 };
149 if (manifest.has("queries")) addLooseSection(manifest["queries"]);
150 if (manifest.has("actions")) addLooseSection(manifest["actions"]);
151
152 // The explicit finish action — success is declared, not vibed.
153 {
154 ToolDefinition finish;
155 finish.name = "finish";
156 finish.description =
157 "Declare the brief complete (or explain why you must stop). Summarize what "
158 "was built, what was verified (renders, peak levels), and anything left "
159 "undone.";
160 JSON properties = JSON::object();
161 JSON summary = JSON::object();
162 summary.set("type", JSON("string"));
163 properties.set("summary", summary);
164 JSON schema = JSON::object();
165 schema.set("type", JSON("object"));
166 schema.set("properties", properties);
167 JSON required = JSON::array();
168 required.push(JSON("summary"));
169 schema.set("required", required);
170 finish.parametersSchema = schema;
171 tools.push_back(finish);
172 }
173
174 if (m_options.mode == Mode::Collaborative) {
175 ToolDefinition ask;
176 ask.name = "ask_user";
177 ask.description = "Ask the user one concise question at a major creative "
178 "decision point and wait for their answer.";
179 JSON properties = JSON::object();
180 JSON question = JSON::object();
181 question.set("type", JSON("string"));
182 properties.set("question", question);
183 JSON schema = JSON::object();
184 schema.set("type", JSON("object"));
185 schema.set("properties", properties);
186 JSON required = JSON::array();
187 required.push(JSON("question"));
188 schema.set("required", required);
189 ask.parametersSchema = schema;
190 tools.push_back(ask);
191 }
192 return tools;
193}

195AgentLoop::Outcome AgentLoop::run(const std::string& brief) {
196 Outcome outcome;
197 const auto deadline = std::chrono::steady_clock::now() +
198 std::chrono::seconds(m_options.maxSeconds);
199
200 // Bootstrap the tool manifest over the wire.
201 const std::string schemaResponse =
202 m_transport("{\"id\": 0, \"verb\": \"get_schema\"}");
203 std::string manifestJson;
204 try {
205 JSON parsed = JSON::parse(schemaResponse);
206 if (!parsed.has("result")) {
207 outcome.stopReason = "provider_error";
208 outcome.summary = "get_schema failed: " + schemaResponse;
209 return outcome;
210 }
211 manifestJson = parsed["result"].toString();
212 } catch (const std::exception& e) {
213 outcome.stopReason = "provider_error";
214 outcome.summary = std::string("get_schema unparseable: ") + e.what();
215 return outcome;
216 }
217
218 ModelRequest request;
219 request.systemPrompt = buildSystemPrompt(manifestJson);
220 request.tools = buildTools(manifestJson);
221 request.maxTokens = m_options.maxTokensPerTurn;
222
223 Message userMessage;
224 userMessage.role = "user";
225 userMessage.text = brief;
226 request.messages.push_back(userMessage);
227
228 long long museRequestId = 1;
229
230 while (outcome.iterations < m_options.maxIterations) {
231 const auto remaining = std::chrono::duration_cast<std::chrono::seconds>(
232 deadline - std::chrono::steady_clock::now())
233 .count();
234 if (remaining <= 0) {
235 outcome.stopReason = "max_seconds";
236 outcome.summary = "time budget exhausted after " +
237 std::to_string(outcome.iterations) + " turns";
238 return outcome;
239 }
240 ++outcome.iterations;
241
242 // The remaining budget becomes the transport deadline, so a hung
243 // provider cannot blow past --max-seconds. remaining >= 1 here (the
244 // truncated-to-seconds check above already returned at <= 0).
245 request.timeoutSeconds = static_cast<int>(remaining);
246 const ModelResponse response = m_provider.complete(request);
247 outcome.usage.inputTokens += response.usage.inputTokens;
248 outcome.usage.outputTokens += response.usage.outputTokens;
249
250 if (response.stopReason == StopReason::Error) {
251 outcome.stopReason = "provider_error";
252 outcome.summary = response.error;
253 return outcome;
254 }
255 if (response.stopReason == StopReason::Refusal) {
256 outcome.stopReason = "refusal";
257 outcome.summary = "the model declined the request";
258 return outcome;
259 }
260 if (response.stopReason == StopReason::MaxTokens && response.toolCalls.empty()) {
261 outcome.stopReason = "max_tokens";
262 outcome.summary = "the model hit its per-turn output token limit";
263 return outcome;
264 }
265
266 if (m_options.verbose && !response.text.empty()) {
267 std::cerr << "[model] " << response.text << "\n";
268 }
269
270 Message assistantMessage;
271 assistantMessage.role = "assistant";
272 assistantMessage.text = response.text;
273 assistantMessage.toolCalls = response.toolCalls;
274 request.messages.push_back(assistantMessage);
275
276 if (response.toolCalls.empty()) {
277 // Ending a turn without finish is a stop, not a success.
278 outcome.stopReason = "end_without_finish";
279 outcome.summary = response.text.empty() ? "the model ended without a summary"
280 : response.text;
281 return outcome;
282 }
283
284 for (const ToolCall& call : response.toolCalls) {
285 Message result;
286 result.role = "tool";
287 result.toolCallId = call.id;
288 result.toolName = call.name;
289
290 if (call.name == "finish") {
291 // Success is declared with substance: a finish without a real
292 // summary bounces back as an error so the model can correct it.
293 JSON arguments = call.arguments;
294 const bool validSummary = arguments.isObject() &&
295 arguments.has("summary") &&
296 arguments["summary"].isString() &&
297 !arguments["summary"].asString().empty();
298 if (!validSummary) {
299 result.text = "finish requires a non-empty string \"summary\" "
300 "describing what was done and verified";
301 result.isError = true;
302 request.messages.push_back(result);
303 continue;
304 }
305 outcome.finished = true;
306 outcome.stopReason = "finished";
307 outcome.summary = arguments["summary"].asString();
308 return outcome;
309 }
310 if (call.name == "ask_user") {
311 const std::string question =
312 call.arguments.has("question")
313 ? JSON(call.arguments)["question"].asString()
314 : "";
315 result.text = m_askUser(question);
316 request.messages.push_back(result);
317 continue;
318 }
319
320 if (outcome.toolCalls >= m_options.maxToolCalls) {
321 outcome.stopReason = "max_tool_calls";
322 outcome.summary = "tool-call budget exhausted";
323 return outcome;
324 }
325 ++outcome.toolCalls;
326
327 JSON museRequest = JSON::object();
328 museRequest.set("id", JSON(static_cast<double>(museRequestId++)));
329 museRequest.set("verb", JSON(call.name));
330 JSON arguments = call.arguments;
331 if (arguments.isObject() && arguments.size() > 0) {
332 museRequest.set("args", arguments);
333 }
334 const std::string museLine = museRequest.toString();
335 const std::string museResponse = m_transport(museLine);
336 if (m_options.verbose) {
337 std::cerr << "[muse] " << museLine << "\n " << museResponse << "\n";
338 }
339
340 result.text = museResponse;
341 try {
342 JSON parsed = JSON::parse(museResponse);
343 result.isError =
344 parsed.has("status") && parsed["status"].asString() != "ok";
345 } catch (const std::exception&) {
346 result.isError = true;
347 }
348 request.messages.push_back(result);
349 }
350 }
351
352 outcome.stopReason = "max_iterations";
353 outcome.summary = "iteration budget exhausted";
354 return outcome;
355}
